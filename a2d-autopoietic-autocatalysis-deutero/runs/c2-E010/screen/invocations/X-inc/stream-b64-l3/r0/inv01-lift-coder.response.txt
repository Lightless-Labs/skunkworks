use std::fmt;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped {
        width: usize,
        ending: LineEnding,
    },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

impl fmt::Display for Error {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            Error::InvalidByte { offset, byte } => {
                write!(f, "invalid byte {byte:#04x} at offset {offset}")
            }
            Error::BadPadding { offset } => write!(f, "bad padding at offset {offset}"),
            Error::Truncated { offset } => write!(f, "truncated input at offset {offset}"),
            Error::AlreadyFinished => write!(f, "stream already finished"),
            Error::InvalidWidth { width } => write!(f, "invalid wrap width {width}"),
            Error::BadWrap { offset } => write!(f, "bad wrapping at offset {offset}"),
        }
    }
}

impl std::error::Error for Error {}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn encode_digit(v: u8, alphabet: Alphabet) -> u8 {
    match v {
        0..=25 => b'A' + v,
        26..=51 => b'a' + (v - 26),
        52..=61 => b'0' + (v - 52),
        62 => if alphabet == Alphabet::Standard { b'+' } else { b'-' },
        _ => if alphabet == Alphabet::Standard { b'/' } else { b'_' },
    }
}

fn decode_digit(b: u8, alphabet: Alphabet) -> Option<u8> {
    match b {
        b'A'..=b'Z' => Some(b - b'A'),
        b'a'..=b'z' => Some(b - b'a' + 26),
        b'0'..=b'9' => Some(b - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn emit_group(input: &[u8; 3], n: usize) -> [u8; 4] {
    let a = input[0];
    let b = input[1];
    let c = input[2];
    let d = [
        a >> 2,
        ((a & 0x03) << 4) | (b >> 4),
        ((b & 0x0f) << 2) | (c >> 6),
        c & 0x3f,
    ];
    let _ = n;
    d
}

fn wrap_symbols(raw: Vec<u8>, whitespace: Whitespace) -> String {
    let mut out = Vec::with_capacity(raw.len() + raw.len() / 4);
    if let Whitespace::Wrapped { width, ending } = whitespace {
        for (i, &b) in raw.iter().enumerate() {
            if i != 0 && i % width == 0 {
                match ending {
                    LineEnding::Lf => out.push(b'\n'),
                    LineEnding::CrLf => {
                        out.push(b'\r');
                        out.push(b'\n');
                    }
                }
            }
            out.push(b);
        }
    } else {
        out = raw;
    }
    String::from_utf8(out).unwrap()
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    symbols_on_row: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: Vec::new(),
            symbols_on_row: 0,
            finished: false,
        })
    }

    fn encode_group(&mut self, bytes: &[u8]) -> String {
        let mut triple = [0u8; 3];
        triple[..bytes.len()].copy_from_slice(bytes);
        let digits = emit_group(&triple, bytes.len());
        let mut raw = vec![
            encode_digit(digits[0], self.options.alphabet),
            encode_digit(digits[1], self.options.alphabet),
        ];
        if bytes.len() == 3 {
            raw.push(encode_digit(digits[2], self.options.alphabet));
            raw.push(encode_digit(digits[3], self.options.alphabet));
        } else if bytes.len() == 2 {
            raw.push(encode_digit(digits[2], self.options.alphabet));
            if self.options.padding != Padding::Forbidden {
                raw.push(b'=');
            }
        } else if self.options.padding != Padding::Forbidden {
            raw.extend_from_slice(b"==");
        }

        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            let mut framed = Vec::with_capacity(raw.len() + 4);
            for b in raw {
                if self.symbols_on_row == width {
                    match ending {
                        LineEnding::Lf => framed.push(b'\n'),
                        LineEnding::CrLf => framed.extend_from_slice(b"\r\n"),
                    }
                    self.symbols_on_row = 0;
                }
                framed.push(b);
                self.symbols_on_row += 1;
            }
            String::from_utf8(framed).unwrap()
        } else {
            String::from_utf8(raw).unwrap()
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = String::new();
        for &b in input {
            self.tail.push(b);
            if self.tail.len() == 3 {
                let bytes = self.tail.clone();
                out.push_str(&self.encode_group(&bytes));
                self.tail.clear();
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        if self.tail.is_empty() {
            return Ok(String::new());
        }
        let tail = self.tail.clone();
        self.tail.clear();
        Ok(self.encode_group(&tail))
    }
}

pub struct Decoder {
    options: Options,
    offset: usize,
    group: Vec<(u8, usize)>,
    pending_second_padding: Option<usize>,
    padded: bool,
    finished: bool,
    sticky_error: Option<Error>,
    row_symbols: usize,
    frame_state: FrameState,
    after_separator: bool,
    group_started: Option<usize>,
}

#[derive(Clone, Copy)]
enum FrameState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            offset: 0,
            group: Vec::new(),
            pending_second_padding: None,
            padded: false,
            finished: false,
            sticky_error: None,
            row_symbols: 0,
            frame_state: FrameState::Normal,
            after_separator: false,
            group_started: None,
        })
    }

    fn sticky(&self) -> Option<Error> {
        self.sticky_error.clone()
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }

    fn emit_full_group(&mut self, out: &mut Vec<u8>) {
        let a = decode_digit(self.group[0].0, self.options.alphabet).unwrap();
        let b = decode_digit(self.group[1].0, self.options.alphabet).unwrap();
        let c = if self.group[2].0 == b'=' {
            0
        } else {
            decode_digit(self.group[2].0, self.options.alphabet).unwrap()
        };
        let d = if self.group[3].0 == b'=' {
            0
        } else {
            decode_digit(self.group[3].0, self.options.alphabet).unwrap()
        };
        out.push((a << 2) | (b >> 4));
        if self.group[2].0 != b'=' {
            out.push((b << 4) | (c >> 2));
        }
        if self.group[3].0 != b'=' {
            out.push((c << 6) | d);
        }
        if self.group[2].0 == b'=' || self.group[3].0 == b'=' {
            self.padded = true;
        }
        self.group.clear();
        self.group_started = None;
    }

    fn ascii_ignored(b: u8) -> bool {
        matches!(b, 9 | 10 | 11 | 12 | 13 | 32)
    }

    fn process_symbol(&mut self, b: u8, at: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.padded {
            return Err(Error::BadPadding { offset: at });
        }
        if self.pending_second_padding.is_some() && b != b'=' {
            return Err(Error::BadPadding { offset: at });
        }

        let pos = self.group.len();
        if b == b'=' {
            match pos {
                0 | 1 => return Err(Error::BadPadding { offset: at }),
                2 => {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset: at });
                    }
                    self.pending_second_padding = Some(at);
                    self.group.push((b, at));
                    self.group_started.get_or_insert(at);
                    return Ok(());
                }
                3 => {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset: at });
                    }
                    self.group.push((b, at));
                    self.emit_full_group(out);
                    return Ok(());
                }
                _ => return Err(Error::BadPadding { offset: at }),
            }
        }

        let Some(_) = decode_digit(b, self.options.alphabet) else {
            return Err(Error::InvalidByte { offset: at, byte: b });
        };
        self.group_started.get_or_insert(at);
        self.group.push((b, at));
        if self.group.len() == 4 {
            self.emit_full_group(out);
        }
        Ok(())
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(e) = self.sticky() {
            return Err(e);
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &b in input {
            let at = self.offset;
            self.offset += 1;

            match self.options.whitespace {
                Whitespace::None => {
                    if let Err(e) = self.process_symbol(b, at, &mut out) {
                        return self.fail(e);
                    }
                }
                Whitespace::IgnoreAscii => {
                    if !Self::ascii_ignored(b) {
                        if let Err(e) = self.process_symbol(b, at, &mut out) {
                            return self.fail(e);
                        }
                    }
                }
                Whitespace::Wrapped { width, ending } => {
                    match self.frame_state {
                        FrameState::NeedLf { cr_offset } => {
                            if b != b'\n' {
                                return self.fail(Error::BadWrap { offset: at });
                            }
                            self.frame_state = FrameState::NeedSymbol {
                                separator_offset: cr_offset,
                            };
                            continue;
                        }
                        FrameState::NeedSymbol { separator_offset } => {
                            self.frame_state = FrameState::Normal;
                            self.after_separator = true;
                            self.row_symbols = 0;
                            if b == b'\n' || b == b'\r' {
                                return self.fail(Error::BadWrap { offset: at });
                            }
                            let _ = separator_offset;
                        }
                        FrameState::Normal => {}
                    }

                    if self.row_symbols == width {
                        match ending {
                            LineEnding::Lf if b == b'\n' => {
                                self.frame_state = FrameState::NeedSymbol {
                                    separator_offset: at,
                                };
                                continue;
                            }
                            LineEnding::Lf => {
                                return self.fail(Error::BadWrap { offset: at });
                            }
                            LineEnding::CrLf if b == b'\r' => {
                                self.frame_state = FrameState::NeedLf { cr_offset: at };
                                continue;
                            }
                            LineEnding::CrLf => {
                                return self.fail(Error::BadWrap { offset: at });
                            }
                        }
                    }

                    if b == b'\r' || b == b'\n' || Self::ascii_ignored(b) {
                        return self.fail(Error::BadWrap { offset: at });
                    }
                    self.row_symbols += 1;
                    if let Err(e) = self.process_symbol(b, at, &mut out) {
                        return self.fail(e);
                    }
                }
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(e) = self.sticky() {
            return Err(e);
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Whitespace::Wrapped { width: _, .. } = self.options.whitespace {
            match self.frame_state {
                FrameState::NeedLf { cr_offset } => {
                    return self.fail(Error::BadWrap { offset: cr_offset });
                }
                FrameState::NeedSymbol { separator_offset } => {
                    return self.fail(Error::BadWrap {
                        offset: separator_offset,
                    });
                }
                FrameState::Normal => {}
            }
        }

        if self.pending_second_padding.is_some() {
            let at = self.pending_second_padding.unwrap();
            return self.fail(Error::BadPadding { offset: at });
        }

        let mut out = Vec::new();
        if self.group.is_empty() {
            self.finished = true;
            return Ok(out);
        }

        match self.group.len() {
            1 => {
                let at = self.group_started.unwrap_or(self.offset);
                return self.fail(Error::Truncated { offset: at });
            }
            2 => {
                if self.options.padding == Padding::Required {
                    let at = self.group_started.unwrap_or(self.offset);
                    return self.fail(Error::Truncated { offset: at });
                }
                let a = decode_digit(self.group[0].0, self.options.alphabet).unwrap();
                let b = decode_digit(self.group[1].0, self.options.alphabet).unwrap();
                out.push((a << 2) | (b >> 4));
            }
            3 => {
                if self.options.padding == Padding::Required {
                    let at = self.group_started.unwrap_or(self.offset);
                    return self.fail(Error::Truncated { offset: at });
                }
                let a = decode_digit(self.group[0].0, self.options.alphabet).unwrap();
                let b = decode_digit(self.group[1].0, self.options.alphabet).unwrap();
                let c = decode_digit(self.group[2].0, self.options.alphabet).unwrap();
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
            }
            _ => {}
        }

        self.group.clear();
        self.finished = true;
        Ok(out)
    }
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    let mut encoder = Encoder::new(options)?;
    let mut out = encoder.push(input)?;
    out.push_str(&encoder.finish()?);
    Ok(out)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend_from_slice(&decoder.finish()?);
    Ok(out)
}
