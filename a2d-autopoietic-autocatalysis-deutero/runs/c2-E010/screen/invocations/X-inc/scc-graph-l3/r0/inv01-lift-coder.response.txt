pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::<usize>::new(); n];
    let mut reverse = vec![Vec::<usize>::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse[to].push(from);
    }

    // Iterative Kosaraju traversal avoids recursion on long paths.
    let mut seen = vec![false; n];
    let mut finish = Vec::with_capacity(n);
    for start in 0..n {
        if seen[start] {
            continue;
        }
        seen[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((v, next_index)) = stack.pop() {
            if next_index < adjacency[v].len() {
                stack.push((v, next_index + 1));
                let w = adjacency[v][next_index];
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                finish.push(v);
            }
        }
    }

    seen.fill(false);
    let mut raw_components = Vec::<Vec<usize>>::new();
    for &start in finish.iter().rev() {
        if seen[start] {
            continue;
        }
        seen[start] = true;
        let mut component = Vec::new();
        let mut stack = vec![start];
        while let Some(v) = stack.pop() {
            component.push(v);
            for &w in &reverse[v] {
                if !seen[w] {
                    seen[w] = true;
                    stack.push(w);
                }
            }
        }
        component.sort_unstable();
        raw_components.push(component);
    }

    raw_components.sort_by(|a, b| {
        b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0]))
    });

    let components = raw_components;
    let mut component_of = vec![0usize; n];
    for (id, component) in components.iter().enumerate() {
        for &v in component {
            component_of[v] = id;
        }
    }

    let count = components.len();
    let mut cyclic = vec![false; count];
    for (id, component) in components.iter().enumerate() {
        if component.len() > 1 {
            cyclic[id] = true;
        }
    }

    let mut internal_edges = vec![0usize; count];
    let mut arc_map = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();
    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let a = component_of[from];
        let b = component_of[to];
        if a == b {
            internal_edges[a] += 1;
            if from == to {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_map.entry((a, b)).or_insert((0, edge_index));
            entry.0 += 1;
            entry.1 = entry.1.min(edge_index);
        }
    }

    let mut arcs: Vec<Arc> = arc_map
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::<usize>::new(); count];
    let mut indegree = vec![0usize; count];
    let mut predecessors = vec![Vec::<usize>::new(); count];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        predecessors[arc.to].push(arc.from);
        indegree[arc.to] += 1;
    }

    // Priority Kahn: greatest internal edge count, then greatest minimum vertex.
    let mut emitted = vec![false; count];
    let mut topo = Vec::with_capacity(count);
    let mut depth = vec![0usize; count];
    for _ in 0..count {
        let mut best: Option<usize> = None;
        for id in 0..count {
            if !emitted[id] && indegree[id] == 0 {
                best = match best {
                    None => Some(id),
                    Some(current) => {
                        let better = internal_edges[id] > internal_edges[current]
                            || (internal_edges[id] == internal_edges[current]
                                && components[id][0] > components[current][0]);
                        Some(if better { id } else { current })
                    }
                };
            }
        }
        let id = best.expect("condensation graph must be acyclic");
        emitted[id] = true;
        topo.push(id);

        for &next in &outgoing[id] {
            depth[next] = depth[next].max(depth[id] + 1);
            indegree[next] -= 1;
        }
    }

    // Ensure depth is the longest path from any source; Kahn processing updates
    // each successor once for every predecessor before it is emitted.
    let _ = predecessors;

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        let mut adjacency = vec![Vec::<usize>::new(); self.components.len()];
        for arc in &self.arcs {
            adjacency[arc.from].push(arc.to);
        }
        for neighbors in &mut adjacency {
            neighbors.sort_unstable();
            neighbors.dedup();
        }

        if let Some(route) = shortest_lex_route(&adjacency, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_lex_route(&adjacency, b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

fn shortest_lex_route(adjacency: &[Vec<usize>], start: usize, goal: usize) -> Option<Vec<usize>> {
    let n = adjacency.len();
    let mut distance = vec![usize::MAX; n];
    let mut queue = std::collections::VecDeque::new();
    distance[goal] = 0;
    queue.push_back(goal);

    let mut reverse = vec![Vec::<usize>::new(); n];
    for (v, neighbors) in adjacency.iter().enumerate() {
        for &w in neighbors {
            reverse[w].push(v);
        }
    }
    while let Some(v) = queue.pop_front() {
        for &previous in &reverse[v] {
            if distance[previous] == usize::MAX {
                distance[previous] = distance[v] + 1;
                queue.push_back(previous);
            }
        }
    }

    if distance[start] == usize::MAX {
        return None;
    }
    let mut route = vec![start];
    let mut current = start;
    while current != goal {
        let next = adjacency[current]
            .iter()
            .copied()
            .filter(|&v| distance[v] != usize::MAX && distance[v] + 1 == distance[current])
            .min()?;
        route.push(next);
        current = next;
    }
    Some(route)
}
