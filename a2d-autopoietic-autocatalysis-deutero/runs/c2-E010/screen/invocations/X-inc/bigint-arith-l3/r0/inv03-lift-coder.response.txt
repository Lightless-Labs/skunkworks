```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented by [0] with negative=false.
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let magnitude = &bytes[start..];

        if let Some(offset) = magnitude
            .iter()
            .position(|&b| b != b'_' && !b.is_ascii_digit())
        {
            return Err(Error::InvalidDigit {
                index: start + offset,
            });
        }

        for i in 0..magnitude.len() {
            if magnitude[i] == b'_'
                && (i == 0
                    || i + 1 == magnitude.len()
                    || !magnitude[i - 1].is_ascii_digit()
                    || !magnitude[i + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index: start + i });
            }
        }

        let first_digit = magnitude.iter().position(|&b| b != b'_').unwrap();
        let digit_count = magnitude.iter().filter(|&&b| b != b'_').count();

        if digit_count > 1 && magnitude[first_digit] == b'0' {
            return Err(Error::LeadingZero {
                index: start + first_digit,
            });
        }

        if digit_count > max_digits {
            let excess_ordinal = max_digits;
            let offset = magnitude
                .iter()
                .enumerate()
                .filter(|(_, &b)| b != b'_')
                .nth(excess_ordinal)
                .map(|(i, _)| i)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: start + offset,
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &b in magnitude.iter().rev() {
            if b != b'_' {
                digits.push(b - b'0');
            }
        }

        let mut result = Self { negative, digits };
        result.normalize();
        Ok(result)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = self.digits.len().max(min_digits).max(1);
        let padding = width - self.digits.len();

        let mut out = String::with_capacity(width + width / group.max(1) + 1);
        if self.negative {
            out.push('-');
        }

        for position in (0..width).rev() {
            if group > 0 && position + 1 < width && (width - position - 1) % group == 0 {
                out.push('_');
            }
            let digit = if position >= self.digits.len() {
                0
            } else {
                self.digits[position]
            };
            out.push(char::from(b'0' + digit));
        }

        let _ = padding;
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => Self::cmp_magnitude(&self.digits, &rhs.digits),
            (true, true) => Self::cmp_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self {
                negative: self.negative,
                digits: Self::add_magnitudes(&self.digits, &rhs.digits),
            };
            result.normalize();
            result
        } else {
            match Self::cmp_magnitude(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self {
                        negative: self.negative,
                        digits: Self::sub_magnitudes(&self.digits, &rhs.digits),
                    };
                    result.normalize();
                    result
                }
                Ordering::Less => {
                    let mut result = Self {
                        negative: rhs.negative,
                        digits: Self::sub_magnitudes(&rhs.digits, &self.digits),
                    };
                    result.normalize();
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut accum = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            for (j, &b) in rhs.digits.iter().enumerate() {
                accum[i + j] += u32::from(a) * u32::from(b);
            }
        }

        for i in 0..accum.len() - 1 {
            let carry = accum[i] / 10;
            accum[i] %= 10;
            accum[i + 1] += carry;
        }

        let mut digits: Vec<u8> = accum.into_iter().map(|v| v as u8).collect();
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }

        let mut result = Self {
            negative: self.negative != rhs.negative,
            digits,
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let (quotient_digits, remainder_digits) =
            Self::div_magnitudes(&self.digits, &rhs.digits);

        let mut quotient = Self {
            negative: self.negative != rhs.negative,
            digits: quotient_digits,
        };
        quotient.normalize();

        let mut remainder = Self {
            negative: self.negative,
            digits: remainder_digits,
        };
        remainder.normalize();

        Ok((quotient, remainder))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn normalize(&mut self) {
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.is_empty() {
            self.digits.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn cmp_magnitude(a: &[u8], b: &[u8]) -> Ordering {
        match a.len().cmp(&b.len()) {
            Ordering::Equal => {
                for (&x, &y) in a.iter().rev().zip(b.iter().rev()) {
                    match x.cmp(&y) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_magnitudes(a: &[u8], b: &[u8]) -> Vec<u8> {
        let len = a.len().max(b.len());
        let mut result = Vec::with_capacity(len + 1);
        let mut carry = 0u16;

        for i in 0..len {
            let sum = u16::from(*a.get(i).unwrap_or(&0))
                + u16::from(*b.get(i).unwrap_or(&0))
                + carry;
            result.push((sum % 10) as u8);
            carry = sum / 10;
        }

        if carry != 0 {
            result.push(carry as u8);
        }
        result
    }

    // Requires a >= b.
    fn sub_magnitudes(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut result = Vec::with_capacity(a.len());
        let mut borrow = 0i16;

        for (i, &digit) in a.iter().enumerate() {
            let mut value = i16::from(digit) - borrow - i16::from(*b.get(i).unwrap_or(&0));
            if value < 0 {
                value += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            result.push(value as u8);
        }

        while result.len() > 1 && result.last() == Some(&0) {
            result.pop();
        }
        result
    }

    fn mul_magnitude_digit(a: &[u8], digit: u8) -> Vec<u8> {
        if digit == 0 {
            return vec![0];
        }

        let mut result = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u16;
        for &value in a {
            let product = u16::from(value) * u16::from(digit) + carry;
            result.push((product % 10) as u8);
            carry = product / 10;
        }
        while carry != 0 {
            result.push((carry % 10) as u8);
            carry /= 10;
        }
        result
    }

    fn push_magnitude_digit(a: &mut Vec<u8>, digit: u8) {
        if a.len() == 1 && a[0] == 0 {
            a[0] = digit;
        } else {
            a.insert(0, digit);
        }
        while a.len() > 1 && a.last() == Some(&0) {
            a.pop();
        }
    }

    fn div_magnitudes(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
        let mut quotient_be = Vec::with_capacity(dividend.len());
        let mut remainder = vec![0];

        for &digit in dividend.iter().rev() {
            Self::push_magnitude_digit(&mut remainder, digit);

            let mut qdigit = 0u8;
            for candidate in (1..=9).rev() {
                let product = Self::mul_magnitude_digit(divisor, candidate);
                if Self::cmp_magnitude(&product, &remainder) != Ordering::Greater {
                    qdigit = candidate;
                    remainder = Self::sub_magnitudes(&remainder, &product);
                    break;
                }
            }
            quotient_be.push(qdigit);
        }

        while quotient_be.len() > 1 && quotient_be.first() == Some(&0) {
            quotient_be.remove(0);
        }

        let quotient = quotient_be.into_iter().rev().collect();
        (quotient, remainder)
    }
}
```
