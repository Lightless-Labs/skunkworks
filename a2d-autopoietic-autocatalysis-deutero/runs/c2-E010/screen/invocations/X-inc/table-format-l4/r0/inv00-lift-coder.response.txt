#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn invalid_char(ch: char) -> bool {
    ch == '\r'
        || (ch.is_control() && ch != '\n')
        || ch == '\u{2028}'
        || ch == '\u{2029}'
}

fn validate(rows: &[Vec<String>]) -> Result<(), TableError> {
    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            let mut iter = cell.char_indices().peekable();
            while let Some((byte, ch)) = iter.next() {
                if ch == '\r' {
                    if matches!(iter.peek(), Some((_, '\n'))) {
                        iter.next();
                    } else {
                        return Err(TableError::InvalidChar {
                            row: row_index,
                            column: column_index,
                            byte,
                        });
                    }
                } else if invalid_char(ch) {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
            }
        }
    }
    Ok(())
}

fn logical_lines(cell: &str, markdown: bool) -> Vec<Vec<Token>> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            let trimmed = line.trim_end_matches(' ');
            trimmed
                .chars()
                .map(|ch| {
                    let text = if markdown {
                        match ch {
                            '|' => "\\|".to_owned(),
                            '\\' => "\\\\".to_owned(),
                            _ => ch.to_string(),
                        }
                    } else {
                        ch.to_string()
                    };
                    let width = text.chars().count();
                    Token { text, width }
                })
                .collect()
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|token| token.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.iter().map(|token| token.text.as_str()).collect()];
            }

            let mut result = String::new();
            let mut used = 0;
            let limit = width.saturating_sub(1);
            for token in tokens {
                if used + token.width > limit {
                    break;
                }
                result.push_str(&token.text);
                used += token.width;
            }
            result.push('…');
            vec![result]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                // All tokens have width at most two and Markdown's minimum width is three.
                let text = tokens[start..end]
                    .iter()
                    .map(|token| token.text.as_str())
                    .collect();
                result.push(text);
                start = end;
            }
            result
        }
    }
}

fn aligned(text: &str, width: usize, alignment: &Alignment) -> String {
    let text_width = text.chars().count();
    let extra = width.saturating_sub(text_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    format!("{}{}{}", " ".repeat(left), text, " ".repeat(right))
}

fn border(widths: &[usize]) -> String {
    let mut line = String::from("+");
    for &width in widths {
        line.push_str(&"-".repeat(width + 2));
        line.push('+');
    }
    line.push('\n');
    line
}

fn emit_data_line(
    output: &mut String,
    mode: &Mode,
    widths: &[usize],
    alignments: &[Alignment],
    contents: &[String],
) {
    if *mode == Mode::Compact {
        for column in 0..widths.len() {
            if column > 0 {
                output.push_str("  ");
            }
            let mut value = aligned(&contents[column], widths[column], &alignments[column]);
            if column + 1 == widths.len() && alignments[column] != Alignment::Left {
                let trailing = widths[column]
                    .saturating_sub(contents[column].chars().count());
                let remove = match alignments[column] {
                    Alignment::Right => 0,
                    Alignment::Center => trailing - trailing / 2,
                    Alignment::Left => trailing,
                };
                for _ in 0..remove {
                    value.pop();
                }
            }
            output.push_str(&value);
        }
        output.push('\n');
    } else {
        output.push('|');
        for column in 0..widths.len() {
            output.push(' ');
            output.push_str(&aligned(
                &contents[column],
                widths[column],
                &alignments[column],
            ));
            output.push(' ');
            output.push('|');
        }
        output.push('\n');
    }
}

fn render_markdown_separator(
    output: &mut String,
    widths: &[usize],
    alignments: &[Alignment],
) {
    output.push('|');
    for (width, alignment) in widths.iter().zip(alignments) {
        output.push(' ');
        match alignment {
            Alignment::Left => {
                output.push(':');
                output.push_str(&"-".repeat(width - 1));
            }
            Alignment::Right => {
                output.push_str(&"-".repeat(width - 1));
                output.push(':');
            }
            Alignment::Center => {
                output.push(':');
                output.push_str(&"-".repeat(width - 2));
                output.push(':');
            }
        }
        output.push(' ');
        output.push('|');
    }
    output.push('\n');
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let column_count = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != column_count {
            return Err(TableError::Ragged {
                row: row_index,
                expected: column_count,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != column_count {
        return Err(TableError::AlignmentCount {
            expected: column_count,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != column_count {
        return Err(TableError::OverflowCount {
            expected: column_count,
            actual: options.overflow.len(),
        });
    }

    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_column_width = match options.mode {
        Mode::Markdown => 3,
        Mode::Grid | Mode::Compact => 1,
    };
    let overhead = match options.mode {
        Mode::Compact => 2 * (column_count - 1),
        Mode::Grid | Mode::Markdown => 3 * column_count + 1,
    };
    let minimum = overhead + column_count * minimum_column_width;

    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    validate(rows)?;

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; column_count]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; column_count]
    } else {
        options.overflow.clone()
    };

    let markdown = options.mode == Mode::Markdown;
    let table: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| logical_lines(cell, markdown))
                .collect()
        })
        .collect();

    let mut widths = vec![minimum_column_width; column_count];
    for row in &table {
        for (column, lines) in row.iter().enumerate() {
            for line in lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let candidate = (0..column_count)
                .filter(|&column| widths[column] > minimum_column_width)
                .max_by_key(|&column| (widths[column], column));
            if let Some(column) = candidate {
                widths[column] -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in table.iter().enumerate() {
        let cell_fragments: Vec<Vec<String>> = row
            .iter()
            .enumerate()
            .map(|(column, lines)| {
                let mut cell_output = Vec::new();
                for line in lines {
                    cell_output.extend(fragments(line, widths[column], &overflows[column]));
                }
                cell_output
            })
            .collect();
        let height = cell_fragments.iter().map(Vec::len).max().unwrap_or(1);

        for fragment_index in 0..height {
            let contents: Vec<String> = cell_fragments
                .iter()
                .map(|cell| cell.get(fragment_index).cloned().unwrap_or_default())
                .collect();
            emit_data_line(
                &mut output,
                &options.mode,
                &widths,
                &alignments,
                &contents,
            );
        }

        if options.mode == Mode::Grid && options.header && row_index == 0 && rows.len() > 1 {
            output.push_str(&border(&widths));
        }
        if options.mode == Mode::Markdown && row_index == 0 {
            render_markdown_separator(&mut output, &widths, &alignments);
        }
    }

    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    Ok(output)
}
