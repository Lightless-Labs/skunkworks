#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn validate_name(site: String, name: &str) -> Result<(), ResolveError> {
    if valid_name(name) {
        Ok(())
    } else {
        Err(ResolveError::InvalidName {
            site,
            name: name.to_owned(),
        })
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn matches(version: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|constraint| match *constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    })
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push(char::from_digit((byte >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            out.push(char::from_digit((byte & 0x0f) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

struct Search<'a> {
    catalog: &'a [Package],
    package_index: std::collections::BTreeMap<String, usize>,
    locks: std::collections::BTreeMap<String, Version>,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn inspect(
        &mut self,
        selected: &mut std::collections::BTreeMap<String, Version>,
    ) -> Result<(), ResolveError> {
        let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
            std::collections::BTreeMap::new();

        fn add(
            active: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
            requirement: &Requirement,
        ) {
            active
                .entry(requirement.package.clone())
                .or_default()
                .push(requirement.constraint);
        }

        // Roots are retained separately in the search state through this field.
        // The root list is supplied on each node by the caller wrapper.
        self.inspect_with_roots(selected, &mut active)
    }

    fn inspect_with_roots(
        &mut self,
        selected: &mut std::collections::BTreeMap<String, Version>,
        active: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
    ) -> Result<(), ResolveError> {
        // Root constraints are stored in a private side table set before search.
        // See `search_node` below.
        let roots = ROOTS.with(|cell| cell.borrow().clone());
        for root in &roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }
        for (name, version) in selected.iter() {
            if let Some(&index) = self.package_index.get(name) {
                let release = self.catalog[index]
                    .releases
                    .iter()
                    .find(|release| release.version == *version)
                    .expect("selected release must exist");
                for dependency in &release.deps {
                    active
                        .entry(dependency.package.clone())
                        .or_default()
                        .push(dependency.constraint);
                }
            }
        }
        for constraints in active.values_mut() {
            canonicalize(constraints);
        }

        if let Some(name) = active
            .keys()
            .filter(|name| !self.package_index.contains_key(*name))
            .next()
            .cloned()
        {
            return Err(ResolveError::Missing { package: name });
        }

        for (name, version) in selected.iter() {
            if let Some(constraints) = active.get(name) {
                if !matches(*version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if !selected.contains_key(name) {
                let package = &self.catalog[self.package_index[name]];
                if !package
                    .releases
                    .iter()
                    .any(|release| matches(release.version, constraints))
                {
                    return Err(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
        let Some(name) = next else {
            let solution: Vec<Selection> = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            let cost = self
                .locks
                .iter()
                .filter(|(package, version)| selected.get(*package) != Some(*version))
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_solution)) => {
                    cost < *best_cost
                        || (cost == *best_cost
                            && self.solution_vector_cmp(&solution, best_solution)
                                == std::cmp::Ordering::Greater)
                }
            };
            if replace {
                self.best = Some((cost, solution));
            }
            return Ok(());
        };

        let constraints = &active[&name];
        let package = &self.catalog[self.package_index[&name]];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|release| release.version)
            .filter(|version| matches(*version, constraints))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock_version) = self.locks.get(&name) {
            if let Some(position) = candidates.iter().position(|version| version == lock_version) {
                let locked = candidates.remove(position);
                candidates.insert(0, locked);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), candidate);
            let result = self.inspect(selected);
            selected.remove(&name);
            match result {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    return Err(ResolveError::Limit { attempts });
                }
                Err(error) => {
                    if self.first_error.is_none() {
                        self.first_error = Some(error);
                    }
                }
            }
        }
        Ok(())
    }

    fn solution_vector_cmp(
        &self,
        left: &[Selection],
        right: &[Selection],
    ) -> std::cmp::Ordering {
        let left_map: std::collections::BTreeMap<&str, Version> = left
            .iter()
            .map(|selection| (selection.package.as_str(), selection.version))
            .collect();
        let right_map: std::collections::BTreeMap<&str, Version> = right
            .iter()
            .map(|selection| (selection.package.as_str(), selection.version))
            .collect();
        for name in self.package_index.keys() {
            match (left_map.get(name.as_str()), right_map.get(name.as_str())) {
                (None, None) => {}
                (None, Some(_)) => return std::cmp::Ordering::Less,
                (Some(_), None) => return std::cmp::Ordering::Greater,
                (Some(a), Some(b)) => {
                    let ordering = a.cmp(b);
                    if ordering != std::cmp::Ordering::Equal {
                        return ordering;
                    }
                }
            }
        }
        std::cmp::Ordering::Equal
    }
}

thread_local! {
    static ROOTS: std::cell::RefCell<Vec<Requirement>> = const { std::cell::RefCell::new(Vec::new()) };
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        validate_name(format!("root[{i}]"), &root.package)?;
    }
    for (i, package) in catalog.iter().enumerate() {
        validate_name(format!("catalog[{i}]"), &package.name)?;
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dependency) in release.deps.iter().enumerate() {
                validate_name(
                    format!("catalog[{i}].release[{j}].dep[{k}]"),
                    &dependency.package,
                )?;
            }
        }
    }

    let mut package_names = std::collections::BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        validate_name(format!("lock[{i}]"), &lock.package)?;
    }
    let mut lock_names = std::collections::BTreeSet::new();
    let mut lock_map = std::collections::BTreeMap::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
        lock_map.insert(lock.package.clone(), lock.version);
    }

    let package_index: std::collections::BTreeMap<String, usize> = catalog
        .iter()
        .enumerate()
        .map(|(index, package)| (package.name.clone(), index))
        .collect();

    ROOTS.with(|cell| *cell.borrow_mut() = roots.to_vec());
    let mut search = Search {
        catalog,
        package_index,
        locks: lock_map.clone(),
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut selected = std::collections::BTreeMap::new();
    let result = search.inspect(&mut selected);
    ROOTS.with(|cell| cell.borrow_mut().clear());

    if let Err(ResolveError::Limit { attempts }) = result {
        return Err(ResolveError::Limit { attempts });
    }
    let Some((_, selected)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let final_constraints = {
        let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
            std::collections::BTreeMap::new();
        for root in roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }
        for selection in &selected {
            if let Some(&index) = search.package_index.get(&selection.package) {
                let release = catalog[index]
                    .releases
                    .iter()
                    .find(|release| release.version == selection.version)
                    .unwrap();
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
        for constraints in active.values_mut() {
            canonicalize(constraints);
        }
        active
    };

    let selected_map: std::collections::BTreeMap<String, Version> = selected
        .iter()
        .map(|selection| (selection.package.clone(), selection.version))
        .collect();
    let mut all_names = std::collections::BTreeSet::new();
    all_names.extend(lock_map.keys().cloned());
    all_names.extend(selected_map.keys().cloned());

    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = selected_map.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(locked), Some(_)) => {
                let exists = search
                    .package_index
                    .get(&package)
                    .map(|&index| {
                        catalog[index]
                            .releases
                            .iter()
                            .any(|release| release.version == locked)
                    })
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if final_constraints
                    .get(&package)
                    .is_some_and(|constraints| !matches(locked, constraints))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
