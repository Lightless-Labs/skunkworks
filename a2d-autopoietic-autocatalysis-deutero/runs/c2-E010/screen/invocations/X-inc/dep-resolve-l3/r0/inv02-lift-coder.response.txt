#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical(constraints.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches_constraint(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn matches_all(v: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|c| matches_constraint(v, *c))
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push('%');
            out.push(char::from_digit((b >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            out.push(char::from_digit((b & 0x0f) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn validate_names(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    Ok(())
}

fn catalog_checks(catalog: &[Package]) -> Result<(), ResolveError> {
    let mut seen = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    Ok(())
}

fn lock_checks(locks: &[Lock]) -> Result<(), ResolveError> {
    let mut seen = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

type ConstraintMap = std::collections::BTreeMap<String, Vec<Constraint>>;
type Assignments = std::collections::BTreeMap<String, Version>;

fn active_constraints(
    roots: &[Requirement],
    catalog: &[Package],
    assignments: &Assignments,
) -> ConstraintMap {
    let mut active = ConstraintMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in assignments {
        if let Some(package) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        *constraints = canonical(std::mem::take(constraints));
    }
    active
}

#[derive(Clone)]
struct SearchResult {
    assignment: Assignments,
    constraints: ConstraintMap,
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: std::collections::BTreeMap<String, Version>,
    max_attempts: usize,
    attempts: usize,
    best: Option<SearchResult>,
    best_cost: usize,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn inspect(&mut self, assignments: &mut Assignments) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, self.catalog, assignments);

        if let Some(name) = active
            .keys()
            .filter(|name| !self.catalog.iter().any(|p| p.name == ***name))
            .next()
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in assignments.iter() {
            if let Some(requirements) = active.get(name) {
                if !matches_all(*version, requirements) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: requirements.clone(),
                    });
                }
            }
        }

        for (name, requirements) in &active {
            if !assignments.contains_key(name) {
                let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
                if !package
                    .releases
                    .iter()
                    .any(|r| matches_all(r.version, requirements))
                {
                    return Err(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: requirements.clone(),
                    });
                }
            }
        }

        let next = active.keys().find(|name| !assignments.contains_key(*name));
        let Some(name) = next else {
            self.record_solution(assignments, &active);
            return Ok(());
        };

        let requirements = active.get(name).unwrap();
        let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .filter(|r| matches_all(r.version, requirements))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(locked) = self.locks.get(name) {
            if let Some(index) = candidates.iter().position(|v| v == locked) {
                let v = candidates.remove(index);
                candidates.insert(0, v);
            }
        }

        let mut first_failure = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assignments.insert(name.clone(), candidate);
            match self.inspect(assignments) {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    assignments.remove(name);
                    return Err(ResolveError::Limit { attempts });
                }
                Err(error) => {
                    if first_failure.is_none() {
                        first_failure = Some(error);
                    }
                }
            }
            assignments.remove(name);
        }

        if let Some(error) = first_failure {
            Err(error)
        } else {
            Ok(())
        }
    }

    fn record_solution(&mut self, assignments: &Assignments, constraints: &ConstraintMap) {
        let cost = self
            .locks
            .iter()
            .filter(|(name, version)| assignments.get(*name) != Some(*version))
            .count();

        let better = match &self.best {
            None => true,
            Some(best) if cost < self.best_cost => true,
            Some(best) if cost > self.best_cost => false,
            Some(best) => {
                let names: std::collections::BTreeSet<&str> = self
                    .catalog
                    .iter()
                    .map(|p| p.name.as_str())
                    .collect();
                for name in names {
                    match (assignments.get(name), best.assignment.get(name)) {
                        (Some(a), Some(b)) if a != b => {
                            if a > b {
                                break true;
                            } else {
                                break false;
                            }
                        }
                        (Some(_), None) => break true,
                        (None, Some(_)) => break false,
                        _ => {}
                    }
                }
                // If every vector position is equal, retain the earlier solution.
                false
            }
        };

        if better {
            self.best_cost = cost;
            self.best = Some(SearchResult {
                assignment: assignments.clone(),
                constraints: constraints.clone(),
            });
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_names(catalog, roots, locks)?;
    catalog_checks(catalog)?;
    lock_checks(locks)?;

    let lock_map: std::collections::BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();

    let mut search = Search {
        catalog,
        roots,
        locks: lock_map.clone(),
        max_attempts,
        attempts: 0,
        best: None,
        best_cost: usize::MAX,
        first_error: None,
    };

    let mut assignments = Assignments::new();
    match search.inspect(&mut assignments) {
        Ok(()) => {}
        Err(ResolveError::Limit { attempts }) => {
            return Err(ResolveError::Limit { attempts });
        }
        Err(error) => search.first_error = Some(error),
    }

    let Some(best) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = best
        .assignment
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut names: std::collections::BTreeSet<String> = lock_map.keys().cloned().collect();
    names.extend(best.assignment.keys().cloned());
    let mut changes = Vec::new();

    for package in names {
        let before = lock_map.get(&package).copied();
        let after = best.assignment.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if best
                    .constraints
                    .get(&package)
                    .is_some_and(|cs| !matches_all(old, cs))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
