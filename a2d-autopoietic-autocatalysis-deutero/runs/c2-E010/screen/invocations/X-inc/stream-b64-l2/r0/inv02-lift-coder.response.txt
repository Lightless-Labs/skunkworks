#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn encode_triplet(a: u8, b: u8, c: u8) -> [u8; 4] {
    [
        ALPHABET[(a >> 2) as usize],
        ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize],
        ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize],
        ALPHABET[(c & 0x3f) as usize],
    ]
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                }
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 3 {
                let encoded = encode_triplet(self.tail[0], self.tail[1], input[index - 1]);
                output.push_str(std::str::from_utf8(&encoded).expect("ASCII alphabet"));
                self.tail_len = 0;
            } else {
                return Ok(output);
            }
        }

        while index + 3 <= input.len() {
            let encoded = encode_triplet(input[index], input[index + 1], input[index + 2]);
            output.push_str(std::str::from_utf8(&encoded).expect("ASCII alphabet"));
            index += 3;
        }

        let remaining = input.len() - index;
        self.tail_len = remaining;
        self.tail[..remaining].copy_from_slice(&input[index..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let result = match self.tail_len {
            0 => String::new(),
            1 => {
                let a = self.tail[0];
                let encoded = [
                    ALPHABET[(a >> 2) as usize],
                    ALPHABET[((a & 0x03) << 4) as usize],
                    b'=',
                    b'=',
                ];
                std::str::from_utf8(&encoded).expect("ASCII alphabet").to_owned()
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                let encoded = [
                    ALPHABET[(a >> 2) as usize],
                    ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize],
                    ALPHABET[((b & 0x0f) << 2) as usize],
                    b'=',
                ];
                std::str::from_utf8(&encoded).expect("ASCII alphabet").to_owned()
            }
            _ => unreachable!(),
        };

        self.tail_len = 0;
        Ok(result)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    total_offset: usize,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            len: 0,
            total_offset: 0,
            terminal_padding: false,
            pending_second_padding: None,
            finished: false,
            sticky_error: None,
        }
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    fn emit_group(&mut self, output: &mut Vec<u8>) {
        let a = digit(self.group[0]).expect("validated digit");
        let b = digit(self.group[1]).expect("validated digit");
        output.push((a << 2) | (b >> 4));

        if self.group[2] != b'=' {
            let c = digit(self.group[2]).expect("validated digit");
            output.push((b << 4) | (c >> 2));
            if self.group[3] != b'=' {
                let d = digit(self.group[3]).expect("validated digit");
                output.push((c << 6) | d);
            }
        }

        if self.group[2] == b'=' || self.group[3] == b'=' {
            self.terminal_padding = true;
        }
        self.len = 0;
        self.pending_second_padding = None;
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let offset = self.total_offset;
            self.total_offset += 1;

            if self.terminal_padding {
                return Err(self.fail(Error::BadPadding { offset }));
            }

            if let Some(padding_offset) = self.pending_second_padding {
                if byte != b'=' {
                    return Err(self.fail(Error::BadPadding { offset }));
                }
                self.group[3] = b'=';
                self.offsets[3] = offset;
                self.emit_group(&mut output);
                let _ = padding_offset;
                continue;
            }

            let position = self.len;
            if position <= 1 && byte == b'=' {
                return Err(self.fail(Error::BadPadding { offset }));
            }

            if position == 2 && byte == b'=' {
                self.group[2] = b'=';
                self.offsets[2] = offset;
                self.pending_second_padding = Some(offset);
                self.len = 3;
                continue;
            }

            if byte == b'=' {
                if position == 3 {
                    self.group[3] = b'=';
                    self.offsets[3] = offset;
                    self.emit_group(&mut output);
                    continue;
                }
                return Err(self.fail(Error::BadPadding { offset }));
            }

            let value = match digit(byte) {
                Some(value) => value,
                None => return Err(self.fail(Error::InvalidByte { offset, byte })),
            };

            self.group[position] = byte;
            self.offsets[position] = offset;
            self.len += 1;

            if self.len == 4 {
                let _ = value;
                self.emit_group(&mut output);
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            return Err(self.fail(Error::BadPadding { offset }));
        }
        if self.len != 0 {
            return Err(self.fail(Error::Truncated {
                offset: self.offsets[0],
            }));
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}
