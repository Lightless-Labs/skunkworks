#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("new encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

fn encode_group(a: u8, b: u8, c: u8) -> [u8; 4] {
    [
        ALPHABET[(a >> 2) as usize],
        ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize],
        ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize],
        ALPHABET[(c & 0x3f) as usize],
    ]
}

fn digit(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&b| b == byte).map(|n| n as u8)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 2 && index < input.len() {
                self.tail[self.tail_len] = input[index];
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 2 && index < input.len() {
                let group = encode_group(self.tail[0], self.tail[1], input[index]);
                output.push_str(std::str::from_utf8(&group).expect("alphabet is ASCII"));
                self.tail_len = 0;
                index += 1;
            } else if self.tail_len < 2 || index == input.len() {
                return Ok(output);
            }
        }

        while input.len() - index >= 3 {
            let group = encode_group(input[index], input[index + 1], input[index + 2]);
            output.push_str(std::str::from_utf8(&group).expect("alphabet is ASCII"));
            index += 3;
        }

        self.tail_len = input.len() - index;
        if self.tail_len > 0 {
            self.tail[..self.tail_len].copy_from_slice(&input[index..]);
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            0 => {}
            1 => {
                output.push(ALPHABET[(self.tail[0] >> 2) as usize] as char);
                output.push(ALPHABET[((self.tail[0] & 0x03) << 4) as usize] as char);
                output.push_str("==");
            }
            2 => {
                output.push(ALPHABET[(self.tail[0] >> 2) as usize] as char);
                output.push(
                    ALPHABET[(((self.tail[0] & 0x03) << 4) | (self.tail[1] >> 4)) as usize]
                        as char,
                );
                output.push(ALPHABET[((self.tail[1] & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => unreachable!(),
        }
        self.tail_len = 0;
        Ok(output)
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
}

pub struct Decoder {
    group: [Symbol; 4],
    group_len: usize,
    offset: usize,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [Symbol { byte: 0, offset: 0 }; 4],
            group_len: 0,
            offset: 0,
            terminal_padding: false,
            pending_second_padding: None,
            finished: false,
            sticky_error: None,
        }
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    fn decode_complete_group(&mut self, output: &mut Vec<u8>) {
        let a = digit(self.group[0].byte).expect("complete group starts with a digit");
        let b = digit(self.group[1].byte).expect("complete group has a second digit");

        match self.group[3].byte {
            b'=' => {
                if self.group[2].byte == b'=' {
                    output.push((a << 2) | (b >> 4));
                } else {
                    let c = digit(self.group[2].byte).expect("third symbol is a digit");
                    output.push((a << 2) | (b >> 4));
                    output.push((b << 4) | (c >> 2));
                }
                self.terminal_padding = true;
            }
            _ => {
                let c = digit(self.group[2].byte).expect("third symbol is a digit");
                let d = digit(self.group[3].byte).expect("fourth symbol is a digit");
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                output.push((c << 6) | d);
            }
        }
        self.group_len = 0;
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let current_offset = self.offset;
            self.offset += 1;

            if self.terminal_padding {
                return Err(self.fail(Error::BadPadding {
                    offset: current_offset,
                }));
            }

            if let Some(padding_offset) = self.pending_second_padding {
                if byte != b'=' {
                    return Err(self.fail(Error::BadPadding {
                        offset: current_offset,
                    }));
                }
                self.pending_second_padding = None;
                self.group[3] = Symbol {
                    byte,
                    offset: current_offset,
                };
                self.group_len = 4;
                self.decode_complete_group(&mut output);
                let _ = padding_offset;
                continue;
            }

            let position = self.group_len;
            if byte == b'=' {
                match position {
                    0 | 1 => {
                        return Err(self.fail(Error::BadPadding {
                            offset: current_offset,
                        }));
                    }
                    2 => {
                        self.group[2] = Symbol {
                            byte,
                            offset: current_offset,
                        };
                        self.group_len = 3;
                        self.pending_second_padding = Some(current_offset);
                    }
                    3 => {
                        self.group[3] = Symbol {
                            byte,
                            offset: current_offset,
                        };
                        self.group_len = 4;
                        self.decode_complete_group(&mut output);
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            if let Some(value) = digit(byte) {
                self.group[position] = Symbol {
                    byte,
                    offset: current_offset,
                };
                self.group_len += 1;
                if self.group_len == 4 {
                    self.decode_complete_group(&mut output);
                }
                let _ = value;
            } else {
                return Err(self.fail(Error::InvalidByte {
                    offset: current_offset,
                    byte,
                }));
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            return Err(self.fail(Error::BadPadding { offset }));
        }
        if self.group_len != 0 {
            return Err(self.fail(Error::Truncated {
                offset: self.group[0].offset,
            }));
        }

        self.finished = true;
        Ok(Vec::new())
    }
}
