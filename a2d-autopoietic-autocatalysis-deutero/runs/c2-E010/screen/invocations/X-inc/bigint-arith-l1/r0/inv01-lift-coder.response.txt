#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Little-endian base-10 digits. The representation is always canonical:
    // zero is [0], and nonzero values have no leading zero digits.
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Scan all magnitude bytes for invalid characters before checking
        // separators, so InvalidDigit has priority everywhere in the input.
        if let Some(index) = (start..bytes.len())
            .find(|&i| !bytes[i].is_ascii_digit() && bytes[i] != b'_')
        {
            return Err(Error::InvalidDigit { index });
        }

        // Find the leftmost underscore that is not between two digits.
        if let Some(index) = (start..bytes.len()).find(|&i| {
            bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
        }) {
            return Err(Error::Separator { index });
        }

        let first_digit = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();

        if digit_count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero { index: first_digit });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0;
            let index = (start..bytes.len())
                .find(|&i| {
                    if bytes[i] == b'_' {
                        false
                    } else {
                        seen += 1;
                        seen == excess_number
                    }
                })
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let digits = bytes[start..]
            .iter()
            .filter(|&&b| b != b'_')
            .rev()
            .map(|&b| b - b'0')
            .collect();

        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let actual_len = self.digits.len().max(1);
        let width = min_digits.max(actual_len);
        let padding = width - actual_len;

        let mut out = String::with_capacity(width + if group > 0 { width / group } else { 0 });
        for i in 0..width {
            if group > 0 && i > 0 && (width - i) % group == 0 {
                out.push('_');
            }

            if i < padding {
                out.push('0');
            } else {
                let digit_index = width - 1 - i;
                out.push(char::from(b'0' + self.digits[digit_index]));
            }
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> std::cmp::Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            std::cmp::Ordering::Equal => self.digits.iter().rev().cmp(rhs.digits.iter().rev()),
            order => order,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let len = self.digits.len().max(rhs.digits.len());
        let mut digits = Vec::with_capacity(len + 1);
        let mut carry = 0u8;

        for i in 0..len {
            let a = self.digits.get(i).copied().unwrap_or(0);
            let b = rhs.digits.get(i).copied().unwrap_or(0);
            let sum = a + b + carry;
            digits.push(sum % 10);
            carry = sum / 10;
        }

        if carry != 0 {
            digits.push(carry);
        }

        Self { digits }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == std::cmp::Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut digits = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i16;

        for i in 0..self.digits.len() {
            let a = self.digits[i] as i16 - borrow;
            let b = rhs.digits.get(i).copied().unwrap_or(0) as i16;
            if a < b {
                digits.push((a + 10 - b) as u8);
                borrow = 1;
            } else {
                digits.push((a - b) as u8);
                borrow = 0;
            }
        }

        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }

        Ok(Self { digits })
    }
}
