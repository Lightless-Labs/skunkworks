fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort();
    cs.dedup();
    cs
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn evidence_sort(evidence: &mut Vec<Evidence>) {
    evidence.sort_by(|a, b| {
        a.constraint
            .cmp(&b.constraint)
            .then_with(|| a.path.cmp(&b.path))
    });
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            let mut ev = evidence.clone();
            evidence_sort(&mut ev);
            for e in ev {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    e.path.join(" -> ")
                ));
            }
            s
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let cs = canonical(requirements.clone())
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut s = format!("no-match|package={}|constraints={}", encode(package), cs);
            let mut ev = evidence.clone();
            evidence_sort(&mut ev);
            for e in ev {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    e.path.join(" -> ")
                ));
            }
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let cs = canonical(requirements.clone())
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                cs
            );
            let mut ev = evidence.clone();
            evidence_sort(&mut ev);
            for e in ev {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    e.path.join(" -> ")
                ));
            }
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn platform_matches(p: Platform, target: Target) -> bool {
    matches!(p, Platform::Any)
        || matches!(
            (p, target),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn dependency_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let fs = {
        let set: BTreeSet<_> = dep.features.iter().cloned().collect();
        set.into_iter().collect::<Vec<_>>().join(",")
    };
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        fs,
        gate,
        platform
    )
}

fn root_token(index: usize, req: &Requirement) -> String {
    let fs = {
        let set: BTreeSet<_> = req.features.iter().cloned().collect();
        set.into_iter().collect::<Vec<_>>().join(",")
    };
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        req.package,
        constraint_text(req.constraint),
        fs
    )
}

fn compute_closure(
    roots: &[Requirement],
    catalog: &BTreeMap<String, &Package>,
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut closure = Closure::default();
    let mut requests: BTreeMap<String, Vec<(String, Vec<String>)>> = BTreeMap::new();
    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(i, root)];
        closure
            .constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        closure
            .origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: path.clone(),
            });
        requests
            .entry(root.package.clone())
            .or_default()
            .push((root.features.join("\0"), path));
    }

    loop {
        let mut changed = false;
        for (pkg, entries) in requests.clone() {
            let Some(version) = selected.get(&pkg).copied() else {
                continue;
            };
            let Some(package) = catalog.get(&pkg) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let path_candidates = entries
                .iter()
                .map(|(_, p)| p.clone())
                .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.cmp(b)));
            if let Some(path) = path_candidates {
                let replace = closure
                    .paths
                    .get(&pkg)
                    .map(|old| path.len() < old.len() || (path.len() == old.len() && path < *old))
                    .unwrap_or(true);
                if replace {
                    closure.paths.insert(pkg.clone(), path);
                }
            }

            for (joined, _) in &entries {
                for feature in joined.split('\0').filter(|s| !s.is_empty()) {
                    if closure
                        .features
                        .entry(pkg.clone())
                        .or_default()
                        .insert(feature.to_string())
                    {
                        changed = true;
                    }
                }
            }

            let active_features = closure.features.get(&pkg).cloned().unwrap_or_default();
            for feature in &release.features {
                if active_features.contains(&feature.name) {
                    for enabled in &feature.enables {
                        let path = closure.paths.get(&pkg).cloned().unwrap_or_default();
                        let entry = requests.entry(pkg.clone()).or_default();
                        if !entry.iter().any(|(f, _)| f == enabled) {
                            entry.push((enabled.clone(), path));
                            changed = true;
                        }
                    }
                }
            }

            for (index, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target) {
                    continue;
                }
                if dep
                    .gate
                    .as_ref()
                    .is_some_and(|g| !active_features.contains(g))
                {
                    continue;
                }
                let Some(source_path) = closure.paths.get(&pkg).cloned() else {
                    continue;
                };
                let mut path = source_path;
                path.push(dependency_token(&pkg, version, index, dep));
                let origins = closure.origins.entry(dep.package.clone()).or_default();
                if !origins.iter().any(|o| o.constraint == dep.constraint && o.path == path) {
                    origins.push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                    closure
                        .constraints
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                    changed = true;
                }
                let requested = requests.entry(dep.package.clone()).or_default();
                for f in &dep.features {
                    if !requested.iter().any(|(old, old_path)| old == f && old_path == &path) {
                        requested.push((f.clone(), path.clone()));
                        changed = true;
                    }
                }
            }
        }

        if !changed {
            break;
        }
    }

    for paths in closure.paths.values_mut() {
        let _ = paths;
    }
    closure
}

fn canonical_path_for_package(
    package: &str,
    roots: &[Requirement],
    catalog: &BTreeMap<String, &Package>,
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Option<Vec<String>> {
    let mut best: Option<Vec<String>> = None;
    let mut queue: Vec<(String, Vec<String>)> = Vec::new();

    for (i, root) in roots.iter().enumerate() {
        if root.package == package {
            queue.push((root.package.clone(), vec![root_token(i, root)]));
        }
    }

    let mut best_to: BTreeMap<String, Vec<String>> = BTreeMap::new();
    while !queue.is_empty() {
        queue.sort_by(|a, b| a.1.len().cmp(&b.1.len()).then_with(|| a.1.cmp(&b.1)));
        let (current, path) = queue.remove(0);
        if best_to
            .get(&current)
            .is_some_and(|old| old.len() < path.len() || (old.len() == path.len() && old <= &path))
        {
            continue;
        }
        best_to.insert(current.clone(), path.clone());
        if current == package {
            best = Some(path.clone());
            break;
        }
        let Some(version) = selected.get(&current).copied() else {
            continue;
        };
        let Some(pkg) = catalog.get(&current) else {
            continue;
        };
        let Some(release) = pkg.releases.iter().find(|r| r.version == version) else {
            continue;
        };
        let enabled = selected_feature_set(current, roots, catalog, selected, target);
        for (i, dep) in release.deps.iter().enumerate() {
            if !platform_matches(dep.platform, target)
                || dep.gate.as_ref().is_some_and(|g| !enabled.contains(g))
            {
                continue;
            }
            let mut next = path.clone();
            next.push(dependency_token(&current, version, i, dep));
            queue.push((dep.package.clone(), next));
        }
    }
    best
}

fn selected_feature_set(
    package: &str,
    roots: &[Requirement],
    catalog: &BTreeMap<String, &Package>,
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> BTreeSet<String> {
    compute_closure(roots, catalog, selected, target)
        .features
        .remove(package)
        .unwrap_or_default()
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: &'a BTreeMap<String, &'a Package>,
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Version>, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn record_error(&mut self, e: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(e);
        }
    }

    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let closure = compute_closure(self.roots, self.catalog, selected, self.target);
        let active: BTreeSet<String> = closure.constraints.keys().cloned().collect();

        if let Some(pkg) = active.iter().find(|p| !self.catalog.contains_key(*p)) {
            let mut ev = closure
                .origins
                .get(pkg.as_str())
                .cloned()
                .unwrap_or_default()
                .into_iter()
                .map(|o| Evidence {
                    constraint: o.constraint,
                    path: o.path,
                })
                .collect::<Vec<_>>();
            evidence_sort(&mut ev);
            self.record_error(ResolveError::Missing {
                package: pkg.clone(),
                evidence: ev,
            });
            return Ok(());
        }

        for (pkg, version) in selected.iter() {
            let cs = canonical(closure.constraints.get(pkg).cloned().unwrap_or_default());
            if !cs.iter().all(|c| matches(*version, *c)) {
                let mut ev = closure
                    .origins
                    .get(pkg)
                    .cloned()
                    .unwrap_or_default()
                    .into_iter()
                    .map(|o| Evidence {
                        constraint: o.constraint,
                        path: o.path,
                    })
                    .collect::<Vec<_>>();
                evidence_sort(&mut ev);
                self.record_error(ResolveError::Conflict {
                    package: pkg.clone(),
                    selected: *version,
                    requirements: cs,
                    evidence: ev,
                });
                return Ok(());
            }
        }

        for pkg in selected.keys() {
            let Some(version) = selected.get(pkg).copied() else {
                continue;
            };
            let Some(package) = self.catalog.get(pkg) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };
            if let Some(feature) = closure
                .features
                .get(pkg)
                .into_iter()
                .flat_map(|s| s.iter())
                .find(|f| !release.features.iter().any(|decl| &decl.name == *f))
            {
                let path = canonical_path_for_package(
                    pkg,
                    self.roots,
                    self.catalog,
                    selected,
                    self.target,
                )
                .unwrap_or_default();
                self.record_error(ResolveError::MissingFeature {
                    package: pkg.clone(),
                    selected: version,
                    feature: feature.clone(),
                    path,
                });
                return Ok(());
            }
        }

        for pkg in active.iter().filter(|p| !selected.contains_key(*p)) {
            let package = self.catalog[*pkg];
            let cs = canonical(closure.constraints.get(pkg).cloned().unwrap_or_default());
            if !package
                .releases
                .iter()
                .any(|r| cs.iter().all(|c| matches(r.version, *c)))
            {
                let mut ev = closure
                    .origins
                    .get(pkg)
                    .cloned()
                    .unwrap_or_default()
                    .into_iter()
                    .map(|o| Evidence {
                        constraint: o.constraint,
                        path: o.path,
                    })
                    .collect::<Vec<_>>();
                evidence_sort(&mut ev);
                self.record_error(ResolveError::NoMatch {
                    package: pkg.clone(),
                    requirements: cs,
                    evidence: ev,
                });
                return Ok(());
            }
        }

        let Some(pkg) = active.iter().find(|p| !selected.contains_key(*p)).cloned() else {
            let cost = self
                .locks
                .iter()
                .filter(|(p, v)| selected.get(*p) != Some(*v))
                .count();
            let versions = self
                .catalog
                .keys()
                .map(|p| selected.get(p).copied().unwrap_or(Version {
                    major: 0,
                    minor: 0,
                    patch: 0,
                }))
                .collect::<Vec<_>>();
            let better = self.best.as_ref().map_or(true, |(bc, bv, _)| {
                cost < *bc || (cost == *bc && versions > *bv)
            });
            if better {
                self.best = Some((cost, versions, selected.clone()));
            }
            return Ok(());
        };

        let package = self.catalog[&pkg];
        let cs = canonical(closure.constraints.get(&pkg).cloned().unwrap_or_default());
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| cs.iter().all(|c| matches(*v, *c)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.get(&pkg) {
            if let Some(index) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(index);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(pkg.clone(), version);
            self.visit(selected)?;
            selected.remove(&pkg);
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let site = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site,
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].gate",
                                i, j, k
                            ),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut search = Search {
        roots,
        catalog: &catalog_map,
        locks: &lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.visit(&mut BTreeMap::new())?;

    let Some((_, _, chosen)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let final_closure = compute_closure(roots, &catalog_map, &chosen, target);
    let before: BTreeMap<String, Version> = lock_map.clone();
    let after = &chosen;
    let all_names: BTreeSet<String> = before.keys().chain(after.keys()).cloned().collect();
    let mut changes = Vec::new();
    for package in all_names {
        let b = before.get(&package).copied();
        let a = after.get(&package).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let cs = canonical(
                        final_closure
                            .constraints
                            .get(&package)
                            .cloned()
                            .unwrap_or_default(),
                    );
                    if !cs.iter().all(|c| matches(old, *c)) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
