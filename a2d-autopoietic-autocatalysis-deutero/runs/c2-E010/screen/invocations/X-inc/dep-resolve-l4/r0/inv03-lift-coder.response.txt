pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateFeature {
        package: String,
        version: Version,
        feature: String,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
        evidence: Vec<Evidence>,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,
    },

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    (1..=24).contains(&b.len())
        && b[0].is_ascii_lowercase()
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
        && b[b.len() - 1] != b'-'
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", b));
        }
    }
    out
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(items: &[Constraint]) -> Vec<Constraint> {
    let mut out = items.to_vec();
    out.sort_by_key(|c| constraint_key(*c));
    out.dedup();
    out
}

fn constraints_text(items: &[Constraint]) -> String {
    canonical_constraints(items)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn evidence_sorted(items: &[Evidence]) -> Vec<Evidence> {
    let mut out = items.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn append_evidence(mut text: String, items: &[Evidence]) -> String {
    for item in evidence_sorted(items) {
        text.push_str("\n  C:");
        text.push_str(&constraint_text(item.constraint));
        text.push_str(" via ");
        text.push_str(&item.path.join(" -> "));
    }
    text
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => append_evidence(
            format!("missing|package={}", encode(package)),
            evidence,
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn platform_matches(p: Platform, t: Target) -> bool {
    matches!(
        (p, t),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    let check = |site: String, name: &str| {
        if valid_name(name) {
            Ok(())
        } else {
            Err(ResolveError::InvalidName {
                site,
                name: name.to_string(),
            })
        }
    };
    for (i, root) in roots.iter().enumerate() {
        check(format!("root[{}]", i), &root.package)?;
        for (f, feature) in root.features.iter().enumerate() {
            check(format!("root[{}].feature[{}]", i, f), feature)?;
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        check(format!("catalog[{}]", i), &package.name)?;
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                check(
                    format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                    &dep.package,
                )?;
                if let Some(gate) = &dep.gate {
                    check(
                        format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                        gate,
                    )?;
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    check(
                        format!(
                            "catalog[{}].release[{}].dep[{}].feature[{}]",
                            i, j, k, f
                        ),
                        feature,
                    )?;
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                check(
                    format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                    &feature.name,
                )?;
                for (e, enabled) in feature.enables.iter().enumerate() {
                    check(
                        format!(
                            "catalog[{}].release[{}].feature[{}].enable[{}]",
                            i, j, f, e
                        ),
                        enabled,
                    )?;
                }
            }
        }
    }

    let mut names = BTreeSet::new();
    for package in catalog {
        if !names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }
    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        check(format!("lock[{}]", i), &lock.package)?;
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    active_features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

impl Active {
    fn add_origin(&mut self, package: &str, constraint: Constraint, path: Vec<String>) -> bool {
        self.origins
            .entry(package.to_string())
            .or_default()
            .push(Origin { constraint, path })
    }
}

fn canonical_path_better(candidate: &[String], current: &[String]) -> bool {
    candidate.len() < current.len() || (candidate.len() == current.len() && candidate < current)
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let mut features = dep.features.clone();
    features.sort();
    features.dedup();
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        gate,
        platform
    )
    .replacen(
        "+{}",
        &format!("+{{{}}}", features.join(",")),
        1,
    )
}

fn root_token(index: usize, root: &Requirement) -> String {
    let mut features = root.features.clone();
    features.sort();
    features.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        root.package,
        constraint_text(root.constraint),
        features.join(",")
    )
}

fn compute_active(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Active {
    let mut active = Active::default();
    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(i, root)];
        active.add_origin(&root.package, root.constraint, path.clone());
        active.paths
            .entry(root.package.clone())
            .and_modify(|p| {
                if canonical_path_better(&path, p) {
                    *p = path.clone();
                }
            })
            .or_insert(path);
        active
            .requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
    }

    loop {
        let mut changed = false;
        let package_names: Vec<String> = active.paths.keys().cloned().collect();
        for name in package_names {
            let Some(version) = selected.get(&name).copied() else {
                continue;
            };
            let Some(package) = catalog.get(&name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested = active.requests.get(&name).cloned().unwrap_or_default();
            for feature_name in requested {
                let inserted = active
                    .active_features
                    .entry(name.clone())
                    .or_default()
                    .insert(feature_name.clone());
                if !inserted {
                    continue;
                }
                changed = true;
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    active
                        .requests
                        .entry(name.clone())
                        .or_default()
                        .extend(feature.enables.iter().cloned());
                }
            }

            let active_features = active
                .active_features
                .get(&name)
                .cloned()
                .unwrap_or_default();
            for (index, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !active_features.contains(gate))
                {
                    continue;
                }
                let Some(source_path) = active.paths.get(&name).cloned() else {
                    continue;
                };
                let token = dep_token(&name, version, index, dep);
                let mut path = source_path;
                path.push(token);
                let before = active.origins.get(&dep.package).map_or(0, Vec::len);
                active.add_origin(&dep.package, dep.constraint, path.clone());
                if active.origins.get(&dep.package).map_or(0, Vec::len) != before {
                    changed = true;
                }

                let path_changed = match active.paths.get(&dep.package) {
                    None => true,
                    Some(old) => canonical_path_better(&path, old),
                };
                if path_changed {
                    active.paths.insert(dep.package.clone(), path);
                    changed = true;
                }
                let reqs = active.requests.entry(dep.package.clone()).or_default();
                let old_len = reqs.len();
                reqs.extend(dep.features.iter().cloned());
                if reqs.len() != old_len {
                    changed = true;
                }
            }
        }
        if !changed {
            break;
        }
    }
    active
}

fn active_constraints(active: &Active, package: &str) -> Vec<Constraint> {
    canonical_constraints(
        &active
            .origins
            .get(package)
            .into_iter()
            .flatten()
            .map(|o| o.constraint)
            .collect::<Vec<_>>(),
    )
}

fn evidence(active: &Active, package: &str) -> Vec<Evidence> {
    let mut out: Vec<Evidence> = active
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn dynamic_error(
    catalog: &BTreeMap<String, &Package>,
    active: &Active,
    selected: &BTreeMap<String, Version>,
) -> Option<ResolveError> {
    for name in active.paths.keys() {
        if !catalog.contains_key(name) {
            return Some(ResolveError::Missing {
                package: name.clone(),
                evidence: evidence(active, name),
            });
        }
    }
    for (name, version) in selected {
        if !active.origins.contains_key(name) {
            continue;
        }
        let cs = active_constraints(active, name);
        if !cs.iter().all(|c| matches(*c, *version)) {
            return Some(ResolveError::Conflict {
                package: name.clone(),
                selected: *version,
                requirements: cs,
                evidence: evidence(active, name),
            });
        }
    }
    for name in active.paths.keys() {
        if selected.contains_key(name) {
            continue;
        }
        let package = catalog[name];
        let cs = active_constraints(active, name);
        if !package
            .releases
            .iter()
            .any(|r| cs.iter().all(|c| matches(*c, r.version)))
        {
            return Some(ResolveError::NoMatch {
                package: name.clone(),
                requirements: cs,
                evidence: evidence(active, name),
            });
        }
    }
    for (name, version) in selected {
        let Some(requests) = active.active_features.get(name) else {
            continue;
        };
        let release = catalog[name]
            .releases
            .iter()
            .find(|r| r.version == *version)
            .unwrap();
        let declared: BTreeSet<&str> = release.features.iter().map(|f| f.name.as_str()).collect();
        for feature in requests {
            if !declared.contains(feature.as_str()) {
                return Some(ResolveError::MissingFeature {
                    package: name.clone(),
                    selected: *version,
                    feature: feature.clone(),
                    path: active.paths.get(name).cloned().unwrap_or_default(),
                });
            }
        }
    }
    None
}

#[derive(Clone)]
struct Best {
    selected: BTreeMap<String, Version>,
    cost: usize,
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<Best>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn cost(&self, selected: &BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|(name, version)| selected.get(*name) != Some(*version))
            .count()
    }

    fn better(&self, candidate: &BTreeMap<String, Version>, cost: usize) -> bool {
        let Some(best) = &self.best else {
            return true;
        };
        if cost != best.cost {
            return cost < best.cost;
        }
        let names: BTreeSet<String> = self
            .catalog
            .keys()
            .chain(best.selected.keys())
            .chain(candidate.keys())
            .cloned()
            .collect();
        for name in names {
            match (candidate.get(&name), best.selected.get(&name)) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }

    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let active = compute_active(self.catalog, self.roots, selected, self.target);
        if let Some(error) = dynamic_error(self.catalog, &active, selected) {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        let next = active.paths.keys().find(|name| !selected.contains_key(*name)).cloned();
        let Some(name) = next else {
            let cost = self.cost(selected);
            if self.better(selected, cost) {
                self.best = Some(Best {
                    selected: selected.clone(),
                    cost,
                });
            }
            return Ok(());
        };

        let package = self.catalog[&name];
        let cs = active_constraints(&active, &name);
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| cs.iter().all(|c| matches(*c, *v)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.get(&name) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                let version = candidates.remove(pos);
                candidates.insert(0, version);
            }
        }

        let mut first_child_error: Option<ResolveError> = None;
        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            let result = self.visit(selected);
            selected.remove(&name);
            if let Err(error) = result {
                return Err(error);
            }
            if first_child_error.is_none() {
                first_child_error = self.first_error.clone();
            }
        }
        if self.best.is_none() && self.first_error.is_none() {
            self.first_error = first_child_error;
        }
        Ok(())
    }
}

fn feature_rejected(active: &Active, catalog: &BTreeMap<String, &Package>, package: &str, version: Version) -> bool {
    let Some(requested) = active.active_features.get(package) else {
        return false;
    };
    let Some(release) = catalog[package].releases.iter().find(|r| r.version == version) else {
        return false;
    };
    let declared: BTreeSet<&str> = release.features.iter().map(|f| f.name.as_str()).collect();
    requested.iter().any(|f| !declared.contains(f.as_str()))
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;
    let packages: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut search = Search {
        catalog: &packages,
        roots,
        locks: &lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.visit(&mut BTreeMap::new())?;
    let Some(best) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_active = compute_active(&packages, roots, &best.selected, target);
    let mut selected: Vec<Selection> = best
        .selected
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();
    selected.sort_by(|a, b| a.package.cmp(&b.package));

    let mut all_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    all_names.extend(best.selected.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = best.selected.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(new)) => {
                let exists = packages
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !active_constraints(&final_active, &package)
                    .iter()
                    .all(|c| matches(*c, old))
                    || feature_rejected(&final_active, &packages, &package, old)
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
