pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },

        version: Version,

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    let (prefix, v) = match c {
        Constraint::Exact(v) => ("=", v),
        Constraint::AtLeast(v) => (">=", v),
        Constraint::Before(v) => ("<", v),
        Constraint::Caret(v) => ("^", v),
    };
    format!("{prefix}{}", version_text(v))
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut result = cs.to_vec();
    result.sort_by_key(|c| constraint_key(*c));
    result.dedup();
    result
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{b:02X}"));
        }
    }
    out
}

fn sort_evidence(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut result = evidence.to_vec();
    result.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    result
}

fn evidence_lines(evidence: &[Evidence]) -> String {
    sort_evidence(evidence)
        .iter()
        .map(|e| {
            format!(
                "\n  C:{} via {}",
                constraint_text(e.constraint),
                e.path.join(" -> ")
            )
        })
        .collect()
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            format!("missing|package={}{}", encode(package), evidence_lines(evidence))
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn check_name(name: &str, site: String) -> Result<(), ResolveError> {
    if valid_name(name) {
        Ok(())
    } else {
        Err(ResolveError::InvalidName {
            site,
            name: name.to_string(),
        })
    }
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn root_token(i: usize, r: &Requirement) -> String {
    let fs: BTreeSet<_> = r.features.iter().cloned().collect();
    format!(
        "root[{i}]:{}@{}+{{{}}}",
        r.package,
        constraint_text(r.constraint),
        fs.into_iter().collect::<Vec<_>>().join(",")
    )
}

fn platform_text(p: Platform) -> &'static str {
    match p {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn platform_matches(p: Platform, t: Target) -> bool {
    matches!(p, Platform::Any)
        || matches!(
            (p, t),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn dep_token(source: &str, version: Version, j: usize, d: &Dependency) -> String {
    let fs: BTreeSet<_> = d.features.iter().cloned().collect();
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        j,
        d.package,
        constraint_text(d.constraint),
        fs.into_iter().collect::<Vec<_>>().join(","),
        d.gate.as_deref().unwrap_or("*"),
        platform_text(d.platform)
    )
}

fn push_request(active: &mut Active, package: &str, feature: &str) -> bool {
    active
        .requests
        .entry(package.to_string())
        .or_default()
        .insert(feature.to_string())
}

fn compute_active(
    roots: &[Requirement],
    catalog: &BTreeMap<String, &Package>,
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Active {
    let mut active = Active::default();
    for (i, root) in roots.iter().enumerate() {
        let token = root_token(i, root);
        active
            .origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
        active
            .paths
            .entry(root.package.clone())
            .and_modify(|p| {
                if vec![token.clone()] < *p {
                    *p = vec![token.clone()];
                }
            })
            .or_insert_with(|| vec![token]);
        for feature in &root.features {
            push_request(&mut active, &root.package, feature);
        }
    }

    loop {
        let before_origins = active.origins.clone();
        let before_requests = active.requests.clone();
        let before_paths = active.paths.clone();

        let packages: Vec<(String, Version)> = selected
            .iter()
            .map(|(p, v)| (p.clone(), *v))
            .collect();

        for (package_name, version) in packages {
            let Some(package) = catalog.get(&package_name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested: Vec<String> = active
                .requests
                .get(&package_name)
                .map(|s| s.iter().cloned().collect())
                .unwrap_or_default();

            let mut enabled = BTreeSet::new();
            let mut queue = requested;
            while let Some(feature_name) = queue.pop() {
                if !enabled.insert(feature_name.clone()) {
                    continue;
                }
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for next in &feature.enables {
                        if !enabled.contains(next) {
                            queue.push(next.clone());
                        }
                    }
                }
            }

            for feature_name in &enabled {
                push_request(&mut active, &package_name, feature_name);
            }

            let source_path = active.paths.get(&package_name).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !enabled.contains(gate))
                {
                    continue;
                }
                let token = dep_token(&package_name, version, j, dep);
                let mut path = source_path.clone();
                path.push(token);
                active
                    .origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                active
                    .paths
                    .entry(dep.package.clone())
                    .and_modify(|p| {
                        if (path.len(), &path) < (p.len(), p) {
                            *p = path.clone();
                        }
                    })
                    .or_insert(path);
                for feature in &dep.features {
                    push_request(&mut active, &dep.package, feature);
                }
            }
        }

        if active.origins == before_origins
            && active.requests == before_requests
            && active.paths == before_paths
        {
            break;
        }

        // Rebuilding avoids accumulating duplicate incoming origins during fixed-point passes.
        active = rebuild_active_once(roots, catalog, selected, target, &active.requests);
        if active.origins == before_origins
            && active.requests == before_requests
            && active.paths == before_paths
        {
            break;
        }
    }

    active
}

fn rebuild_active_once(
    roots: &[Requirement],
    catalog: &BTreeMap<String, &Package>,
    selected: &BTreeMap<String, Version>,
    target: Target,
    seeded_requests: &BTreeMap<String, BTreeSet<String>>,
) -> Active {
    let mut active = Active::default();
    active.requests = seeded_requests.clone();

    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(i, root)];
        active
            .origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: path.clone(),
            });
        active
            .paths
            .entry(root.package.clone())
            .and_modify(|p| {
                if (path.len(), &path) < (p.len(), p) {
                    *p = path.clone();
                }
            })
            .or_insert(path);
        for f in &root.features {
            push_request(&mut active, &root.package, f);
        }
    }

    loop {
        let old_requests = active.requests.clone();
        let old_origins = active.origins.clone();
        let old_paths = active.paths.clone();

        let packages: Vec<(String, Version)> = selected
            .iter()
            .map(|(p, v)| (p.clone(), *v))
            .collect();
        for (package_name, version) in packages {
            let Some(package) = catalog.get(&package_name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested: Vec<_> = active
                .requests
                .get(&package_name)
                .into_iter()
                .flat_map(|s| s.iter().cloned())
                .collect();
            let mut enabled = BTreeSet::new();
            let mut queue = requested;
            while let Some(name) = queue.pop() {
                if !enabled.insert(name.clone()) {
                    continue;
                }
                if let Some(feature) = release.features.iter().find(|f| f.name == name) {
                    queue.extend(feature.enables.iter().cloned());
                }
            }
            for name in &enabled {
                push_request(&mut active, &package_name, name);
            }

            let source_path = active.paths.get(&package_name).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !enabled.contains(g))
                {
                    continue;
                }
                let token = dep_token(&package_name, version, j, dep);
                let mut path = source_path.clone();
                path.push(token);
                active
                    .origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                active
                    .paths
                    .entry(dep.package.clone())
                    .and_modify(|p| {
                        if (path.len(), &path) < (p.len(), p) {
                            *p = path.clone();
                        }
                    })
                    .or_insert(path);
                for f in &dep.features {
                    push_request(&mut active, &dep.package, f);
                }
            }
        }

        if old_requests == active.requests
            && old_origins == active.origins
            && old_paths == active.paths
        {
            break;
        }

        // Origins are recomputed from scratch on each pass to avoid duplicate paths.
        let requests = active.requests.clone();
        active.origins.clear();
        active.paths.clear();
        for (i, root) in roots.iter().enumerate() {
            let path = vec![root_token(i, root)];
            active
                .origins
                .entry(root.package.clone())
                .or_default()
                .push(Origin {
                    constraint: root.constraint,
                    path: path.clone(),
                });
            active.paths.entry(root.package.clone()).or_insert(path);
        }
        active.requests = requests;
    }

    active
}

fn canonical_path(active: &Active, package: &str) -> Vec<String> {
    active.paths.get(package).cloned().unwrap_or_default()
}

fn canonical_origins(active: &Active, package: &str) -> Vec<Evidence> {
    let mut evidence: Vec<Evidence> = active
        .origins
        .get(package)
        .into_iter()
        .flat_map(|os| os.iter())
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect();
    evidence.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    evidence
}

fn active_constraints(active: &Active, package: &str) -> Vec<Constraint> {
    canonical_constraints(
        &active
            .origins
            .get(package)
            .into_iter()
            .flat_map(|os| os.iter().map(|o| o.constraint))
            .collect::<Vec<_>>(),
    )
}

fn release_features(release: &Release) -> BTreeMap<String, &Feature> {
    release.features.iter().map(|f| (f.name.clone(), f)).collect()
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: &'a BTreeMap<String, &'a Package>,
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Option<Version>>, BTreeMap<String, Version>, Active)>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn inspect(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let active = compute_active(self.roots, self.catalog, selected, self.target);

        if let Some(package) = active
            .origins
            .keys()
            .filter(|p| !self.catalog.contains_key(*p))
            .next()
        {
            self.record_error(ResolveError::Missing {
                package: package.clone(),
                evidence: canonical_origins(&active, package),
            });
            return Ok(());
        }

        for (package, version) in selected.iter() {
            let constraints = active_constraints(&active, package);
            if constraints.iter().any(|c| !matches(*version, *c)) {
                self.record_error(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: constraints,
                    evidence: canonical_origins(&active, package),
                });
                return Ok(());
            }
        }

        for (package, version) in selected.iter() {
            let Some(cat_package) = self.catalog.get(package) else {
                continue;
            };
            let Some(release) = cat_package.releases.iter().find(|r| r.version == *version) else {
                continue;
            };
            let enabled = active.requests.get(package).cloned().unwrap_or_default();
            let declared = release_features(release);
            if let Some(feature) = enabled.iter().find(|f| !declared.contains_key(*f)) {
                self.record_error(ResolveError::MissingFeature {
                    package: package.clone(),
                    selected: *version,
                    feature: feature.clone(),
                    path: canonical_path(&active, package),
                });
                return Ok(());
            }
        }

        let mut unselected: Vec<String> = active
            .origins
            .keys()
            .filter(|p| !selected.contains_key(*p))
            .cloned()
            .collect();
        unselected.sort();

        for package in &unselected {
            let constraints = active_constraints(&active, package);
            let cat_package = &self.catalog[package];
            let has_match = cat_package.releases.iter().any(|r| {
                constraints.iter().all(|c| matches(r.version, *c))
            });
            if !has_match {
                self.record_error(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints,
                    evidence: canonical_origins(&active, package),
                });
                return Ok(());
            }
        }

        if unselected.is_empty() {
            self.record_solution(selected.clone(), active);
            return Ok(());
        }

        let package = unselected.remove(0);
        let constraints = active_constraints(&active, &package);
        let mut candidates: Vec<Version> = self.catalog[&package]
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| constraints.iter().all(|c| matches(*v, *c)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();
        if let Some(locked) = self.locks.get(&package) {
            if let Some(index) = candidates.iter().position(|v| v == locked) {
                let version = candidates.remove(index);
                candidates.insert(0, version);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            self.inspect(selected)?;
            selected.remove(&package);
        }
        Ok(())
    }

    fn record_solution(&mut self, selected: BTreeMap<String, Version>, active: Active) {
        let cost = self
            .locks
            .iter()
            .filter(|(p, v)| selected.get(*p) != Some(*v))
            .count();
        let names: Vec<String> = self.catalog.keys().cloned().collect();
        let vector: Vec<Option<Version>> = names.iter().map(|p| selected.get(p).copied()).collect();

        let replace = match &self.best {
            None => true,
            Some((best_cost, best_vector, _, _)) => {
                cost < *best_cost || (cost == *best_cost && vector > *best_vector)
            }
        };
        if replace {
            self.best = Some((cost, vector, selected, active));
        }
    }
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        check_name(&root.package, format!("root[{i}]"))?;
        for (f, feature) in root.features.iter().enumerate() {
            check_name(feature, format!("root[{i}].feature[{f}]"))?;
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        check_name(&package.name, format!("catalog[{i}]"))?;
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                check_name(
                    &dep.package,
                    format!("catalog[{i}].release[{j}].dep[{k}]"),
                )?;
                if let Some(gate) = &dep.gate {
                    check_name(
                        gate,
                        format!("catalog[{i}].release[{j}].dep[{k}].gate"),
                    )?;
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    check_name(
                        feature,
                        format!("catalog[{i}].release[{j}].dep[{k}].feature[{f}]"),
                    )?;
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                check_name(
                    &feature.name,
                    format!("catalog[{i}].release[{j}].feature[{f}]"),
                )?;
                for (e, enable) in feature.enables.iter().enumerate() {
                    check_name(
                        enable,
                        format!("catalog[{i}].release[{j}].feature[{f}].enable[{e}]"),
                    )?;
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen = BTreeSet::new();
        for release in &package.releases {
            if !seen.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        check_name(&lock.package, format!("lock[{i}]"))?;
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = BTreeSet::new();
            for feature in &release.features {
                if !seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut search = Search {
        roots,
        catalog: &catalog_map,
        locks: &lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };

    let mut selected = BTreeMap::new();
    search.inspect(&mut selected)?;

    if let Some((_, _, selected_map, active)) = search.best {
        let selected_vec = selected_map
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect();

        let mut changes = Vec::new();
        let all_names: BTreeSet<String> = lock_map
            .keys()
            .chain(selected_map.keys())
            .cloned()
            .collect();

        for package in all_names {
            let before = lock_map.get(&package).copied();
            let after = selected_map.get(&package).copied();
            if before == after {
                continue;
            }

            let reason = match (before, after) {
                (None, Some(_)) => ChangeReason::Added,
                (Some(_), None) => ChangeReason::Removed,
                (Some(old), Some(_)) => {
                    let exists = catalog_map
                        .get(&package)
                        .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                    if !exists {
                        ChangeReason::Unavailable
                    } else if active_constraints(&active, &package)
                        .iter()
                        .any(|c| !matches(old, *c))
                    {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
                (None, None) => continue,
            };
            changes.push(Change {
                package,
                before,
                after,
                reason,
            });
        }

        Ok(Resolution {
            selected: selected_vec,
            changes,
        })
    } else {
        Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }))
    }
}
