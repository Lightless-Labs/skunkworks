pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,

    Ignore,
    Redirect,

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {

    routes: Vec<Route>,

fn valid_method(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(s: &str) -> bool {
    let mut it = s.bytes();
    matches!(it.next(), Some(b'a'..=b'z'))
        && it.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-')
}

fn valid_capture_name(s: &str) -> bool {
    let mut it = s.bytes();
    matches!(it.next(), Some(b'a'..=b'z'))
        && it.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn decode_segment(s: &str, base: usize) -> Result<String, usize> {
    let bytes = s.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let Some(h1) = hex(bytes[i + 1]) else {
                return Err(base + i);
            };
            let Some(h2) = hex(bytes[i + 2]) else {
                return Err(base + i);
            };
            let decoded = h1 * 16 + h2;
            if !(0x20..=0x7e).contains(&decoded) || decoded == b'/' || unreserved(decoded) {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).unwrap_or_default())
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool, Vec<u8>), usize> {
    let bytes = pattern.as_bytes();
    if bytes.is_empty() || bytes[0] != b'/' {
        return Err(0);
    }
    if pattern == "/" {
        return Ok((Vec::new(), false, vec![3]));
    }
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut seen = HashSet::new();
    let mut pos = 1;
    while pos < end {
        let next = bytes[pos..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| pos + n)
            .unwrap_or(end);
        if next == pos {
            return Err(pos);
        }
        let raw = &pattern[pos..next];
        let first = bytes[pos];
        if first == b':' || first == b'*' {
            let name = &raw[1..];
            if !valid_capture_name(name) || !seen.insert(name.to_owned()) {
                return Err(pos);
            }
            if first == b'*' && next != end {
                return Err(pos);
            }
            segments.push(if first == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            let decoded = decode_segment(raw, pos)?;
            segments.push(Segment::Static(decoded));
        }
        pos = next + 1;
    }
    let mut rank = Vec::new();
    for seg in &segments {
        rank.push(match seg {
            Segment::Static(_) => 2,
            Segment::Param(_) => 1,
            Segment::Wildcard(_) => 0,
        });
    }
    if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
        rank.push(3);
    }
    Ok((segments, trailing, rank))
}

fn parse_path(head: &str) -> Result<(Vec<String>, bool), usize> {
    let bytes = head.as_bytes();
    if bytes.is_empty() || bytes[0] != b'/' {
        return Err(0);
    }
    if head == "/" {
        return Ok((Vec::new(), false));
    }
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut pos = 1;
    while pos < end {
        let next = bytes[pos..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| pos + n)
            .unwrap_or(end);
        if next == pos {
            return Err(pos);
        }
        segments.push(decode_segment(&head[pos..next], pos)?);
        pos = next + 1;
    }
    Ok((segments, trailing))
}

fn shape_conflict(a: &Route, b_segments: &[Segment], b_trailing: bool, policy: SlashPolicy) -> bool {
    if a.segments.len() != b_segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b_trailing {
        return false;
    }
    a.segments.iter().zip(b_segments).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn matches(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard = route.segments.last().is_some_and(|s| matches!(s, Segment::Wildcard(_)));
    if (!wildcard && route.segments.len() != path.len())
        || (wildcard && path.len() + 1 < route.segments.len())
    {
        return None;
    }
    let mut params = Vec::new();
    for (i, seg) in route.segments.iter().enumerate() {
        match seg {
            Segment::Static(s) => {
                if path.get(i) != Some(s) {
                    return None;
                }
            }
            Segment::Param(name) => {
                let value = path.get(i)?;
                params.push((name.clone(), value.clone()));
            }
            Segment::Wildcard(name) => {
                let value = path[i..].join("/");
                params.push((name.clone(), value));
            }
        }
    }
    Some(params)
}

fn encode_segment(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
            names: HashMap::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing, rank) = parse_pattern(pattern)
            .map_err(|at| Error::InvalidPattern { at })?;
        if self.names.contains_key(name) {
            return Err(Error::DuplicateName);
        }
        let id = self.routes.len();
        for route in &self.routes {
            if route.method == method && shape_conflict(route, &segments, trailing, self.policy) {
                return Err(Error::Conflict { with: route.id });
            }
        }
        self.names.insert(name.to_owned(), id);
        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .bytes()
            .position(|b| b == b'?' || b == b'#')
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (segments, trailing) =
            parse_path(head).map_err(|at| Error::InvalidPath { at })?;

        let candidates: Vec<(&Route, Vec<(String, String)>)> = self
            .routes
            .iter()
            .filter_map(|route| {
                if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                    return None;
                }
                matches(route, &segments).map(|params| (route, params))
            })
            .collect();

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<_> = if method == "HEAD" {
            if candidates.iter().any(|(r, _)| r.method == "HEAD") {
                candidates
                    .iter()
                    .filter(|(r, _)| r.method == "HEAD")
                    .collect()
            } else {
                candidates
                    .iter()
                    .filter(|(r, _)| r.method == "GET")
                    .collect()
            }
        } else {
            candidates
                .iter()
                .filter(|(r, _)| r.method == method)
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed = HashSet::new();
            for (route, _) in &candidates {
                allowed.insert(route.method.clone());
                if route.method == "GET" {
                    allowed.insert("HEAD".to_owned());
                }
            }
            let mut allowed: Vec<_> = allowed.into_iter().collect();
            allowed.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            return Err(Error::MethodNotAllowed { allowed });
        }

        let (route, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| a.rank.cmp(&b.rank).then_with(|| b.id.cmp(&a.id)))
            .unwrap();

        let redirect = if self.policy == SlashPolicy::Redirect
            && !segments.is_empty()
            && route.trailing != trailing
        {
            let mut s = head.to_owned();
            if route.trailing {
                s.push('/');
            } else if s.ends_with('/') {
                s.pop();
            }
            Some(s)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let Some(&id) = self.names.get(name) else {
            return Err(Error::UnknownName);
        };
        let route = &self.routes[id];

        let mut supplied = HashMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
            supplied.insert(*key, *value);
        }

        let mut required = Vec::new();
        for seg in &route.segments {
            match seg {
                Segment::Param(n) | Segment::Wildcard(n) => required.push(n),
                Segment::Static(_) => {}
            }
        }
        for key in &required {
            if !supplied.contains_key(key.as_str()) {
                return Err(Error::MissingParam { name: (*key).clone() });
            }
        }

        let required_set: HashSet<&str> = required.iter().map(|s| s.as_str()).collect();
        let mut extras: Vec<&str> = supplied
            .keys()
            .copied()
            .filter(|key| !required_set.contains(key))
            .collect();
        extras.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for seg in &route.segments {
            let name = match seg {
                Segment::Param(n) | Segment::Wildcard(n) => n,
                Segment::Static(_) => continue,
            };
            let value = supplied[name.as_str()];
            let valid = value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                && match seg {
                    Segment::Param(_) => !value.is_empty() && !value.contains('/'),
                    Segment::Wildcard(_) => {
                        value.is_empty()
                            || value.split('/').all(|part| !part.is_empty())
                    }
                    Segment::Static(_) => true,
                };
            if !valid {
                return Err(Error::InvalidValue { name: name.clone() });
            }
        }

        let mut output = String::new();
        for seg in &route.segments {
            match seg {
                Segment::Static(s) => {
                    output.push('/');
                    output.push_str(&encode_segment(s));
                }
                Segment::Param(name) => {
                    output.push('/');
                    output.push_str(&encode_segment(supplied[name.as_str()]));
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        for part in value.split('/') {
                            output.push('/');
                            output.push_str(&encode_segment(part));
                        }
                    }
                }
            }
        }
        if output.is_empty() {
            output.push('/');
        }
        if route.trailing && output != "/" {
            output.push('/');
        }
        Ok(output)
    }
}

use std::collections::{BTreeMap, HashMap, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Part {
    Lit(String),
    One(String),
    Tail(String),
}

struct Entry {
    id: usize,
    name: String,
    method: String,
    parts: Vec<Part>,
    slash: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    entries: Vec<Entry>,
    by_name: HashMap<String, usize>,
}

fn method_ok(value: &str) -> bool {
    !value.is_empty() && value.bytes().all(|b| b.is_ascii_uppercase())
}

fn name_ok(value: &str, hyphen: bool) -> bool {
    let mut bytes = value.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| {
            b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || (hyphen && b == b'-')
        })
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn plain(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn decode(s: &str, offset: usize) -> Result<String, usize> {
    let b = s.as_bytes();
    let mut v = Vec::new();
    let mut i = 0;
    while i < b.len() {
        if b[i] == b'%' {
            if i + 2 >= b.len() {
                return Err(offset + i);
            }
            let (Some(a), Some(c)) = (hex_value(b[i + 1]), hex_value(b[i + 2])) else {
                return Err(offset + i);
            };
            let x = a * 16 + c;
            if !(0x20..=0x7e).contains(&x) || x == b'/' || plain(x) {
                return Err(offset + i);
            }
            v.push(x);
            i += 3;
        } else if plain(b[i]) {
            v.push(b[i]);
            i += 1;
        } else {
            return Err(offset + i);
        }
    }
    Ok(String::from_utf8(v).unwrap_or_default())
}

fn parse_template(s: &str) -> Result<(Vec<Part>, bool, Vec<u8>), usize> {
    let b = s.as_bytes();
    if b.first() != Some(&b'/') {
        return Err(0);
    }
    if s == "/" {
        return Ok((Vec::new(), false, vec![3]));
    }
    let slash = b.last() == Some(&b'/');
    let limit = b.len() - usize::from(slash);
    let mut parts = Vec::new();
    let mut used = HashSet::new();
    let mut pos = 1;
    while pos < limit {
        let end = b[pos..limit]
            .iter()
            .position(|x| *x == b'/')
            .map_or(limit, |n| pos + n);
        if end == pos {
            return Err(pos);
        }
        let raw = &s[pos..end];
        if b[pos] == b':' || b[pos] == b'*' {
            let n = &raw[1..];
            if !name_ok(n, false) || !used.insert(n.to_owned()) {
                return Err(pos);
            }
            if b[pos] == b'*' && end != limit {
                return Err(pos);
            }
            parts.push(if b[pos] == b':' {
                Part::One(n.to_owned())
            } else {
                Part::Tail(n.to_owned())
            });
        } else {
            parts.push(Part::Lit(decode(raw, pos)?));
        }
        pos = end + 1;
    }
    let mut rank = parts
        .iter()
        .map(|p| match p {
            Part::Lit(_) => 2,
            Part::One(_) => 1,
            Part::Tail(_) => 0,
        })
        .collect::<Vec<_>>();
    if !parts.iter().any(|p| matches!(p, Part::Tail(_))) {
        rank.push(3);
    }
    Ok((parts, slash, rank))
}

fn parse_request(s: &str) -> Result<(Vec<String>, bool), usize> {
    let b = s.as_bytes();
    if b.first() != Some(&b'/') {
        return Err(0);
    }
    if s == "/" {
        return Ok((Vec::new(), false));
    }
    let slash = b.last() == Some(&b'/');
    let limit = b.len() - usize::from(slash);
    let mut parts = Vec::new();
    let mut pos = 1;
    while pos < limit {
        let end = b[pos..limit]
            .iter()
            .position(|x| *x == b'/')
            .map_or(limit, |n| pos + n);
        if end == pos {
            return Err(pos);
        }
        parts.push(decode(&s[pos..end], pos)?);
        pos = end + 1;
    }
    Ok((parts, slash))
}

fn same_shape(a: &Entry, b: &[Part], slash: bool, policy: SlashPolicy) -> bool {
    if a.parts.len() != b.len() || (policy == SlashPolicy::Strict && a.slash != slash) {
        return false;
    }
    a.parts.iter().zip(b).all(|(x, y)| match (x, y) {
        (Part::Lit(p), Part::Lit(q)) => p == q,
        (Part::One(_), Part::One(_)) | (Part::Tail(_), Part::Tail(_)) => true,
        _ => false,
    })
}

fn candidate(e: &Entry, path: &[String]) -> Option<Vec<(String, String)>> {
    let tail = matches!(e.parts.last(), Some(Part::Tail(_)));
    if (!tail && e.parts.len() != path.len()) || (tail && path.len() < e.parts.len() - 1) {
        return None;
    }
    let mut out = Vec::new();
    for (i, p) in e.parts.iter().enumerate() {
        match p {
            Part::Lit(s) if path.get(i) != Some(s) => return None,
            Part::Lit(_) => {}
            Part::One(n) => out.push((n.clone(), path.get(i)?.clone())),
            Part::Tail(n) => out.push((n.clone(), path[i..].join("/"))),
        }
    }
    Some(out)
}

fn encode(s: &str) -> String {
    const H: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::new();
    for b in s.bytes() {
        if plain(b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(H[(b >> 4) as usize] as char);
            out.push(H[(b & 15) as usize] as char);
        }
    }
    out
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self { policy, entries: Vec::new(), by_name: HashMap::new() }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !method_ok(method) {
            return Err(Error::InvalidMethod);
        }
        if !name_ok(name, true) {
            return Err(Error::InvalidName);
        }
        let (parts, slash, rank) = parse_template(pattern)
            .map_err(|at| Error::InvalidPattern { at })?;
        if self.by_name.contains_key(name) {
            return Err(Error::DuplicateName);
        }
        for old in &self.entries {
            if old.method == method && same_shape(old, &parts, slash, self.policy) {
                return Err(Error::Conflict { with: old.id });
            }
        }
        let id = self.entries.len();
        self.by_name.insert(name.to_owned(), id);
        self.entries.push(Entry {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            parts,
            slash,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !method_ok(method) {
            return Err(Error::InvalidMethod);
        }
        let cut = path.bytes().position(|b| b == b'?' || b == b'#').unwrap_or(path.len());
        let head = &path[..cut];
        let (segments, slash) = parse_request(head).map_err(|at| Error::InvalidPath { at })?;
        let all: Vec<_> = self.entries.iter().filter_map(|e| {
            if self.policy == SlashPolicy::Strict && e.slash != slash {
                None
            } else {
                candidate(e, &segments).map(|p| (e, p))
            }
        }).collect();
        if all.is_empty() {
            return Err(Error::NotFound);
        }

        let selected: Vec<_> = if method == "HEAD" {
            let exact_head = all.iter().any(|(e, _)| e.method == "HEAD");
            all.iter().filter(|(e, _)| e.method == if exact_head { "HEAD" } else { "GET" }).collect()
        } else {
            all.iter().filter(|(e, _)| e.method == method).collect()
        };
        if selected.is_empty() {
            let mut allowed = BTreeMap::<String, ()>::new();
            for (e, _) in &all {
                allowed.insert(e.method.clone(), ());
                if e.method == "GET" {
                    allowed.insert("HEAD".to_owned(), ());
                }
            }
            return Err(Error::MethodNotAllowed { allowed: allowed.into_keys().collect() });
        }
        let (winner, params) = selected.into_iter().min_by(|(a, _), (b, _)| {
            b.rank.cmp(&a.rank).then(a.id.cmp(&b.id))
        }).unwrap();

        let redirect = if self.policy == SlashPolicy::Redirect && !segments.is_empty() && winner.slash != slash {
            let mut value = head.to_owned();
            if winner.slash {
                value.push('/');
            } else {
                value.pop();
            }
            Some(value)
        } else {
            None
        };
        Ok(Match { route: winner.id, params, redirect })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let Some(&id) = self.by_name.get(name) else {
            return Err(Error::UnknownName);
        };
        let e = &self.entries[id];
        let mut values = HashMap::<&str, &str>::new();
        for &(key, value) in params {
            if values.contains_key(key) {
                return Err(Error::DuplicateParam { name: key.to_owned() });
            }
            values.insert(key, value);
        }
        let mut needed = Vec::new();
        for p in &e.parts {
            if let Part::One(n) | Part::Tail(n) = p {
                needed.push(n);
            }
        }
        for n in &needed {
            if !values.contains_key(n.as_str()) {
                return Err(Error::MissingParam { name: (*n).clone() });
            }
        }
        let valid: HashSet<&str> = needed.iter().map(|x| x.as_str()).collect();
        if let Some(k) = values.keys().filter(|k| !valid.contains(**k)).min_by(|a, b| a.as_bytes().cmp(b.as_bytes())) {
            return Err(Error::ExtraParam { name: (*k).to_owned() });
        }
        for p in &e.parts {
            let (n, tail) = match p {
                Part::One(n) => (n, false),
                Part::Tail(n) => (n, true),
                Part::Lit(_) => continue,
            };
            let v = values[n.as_str()];
            let ascii = v.bytes().all(|b| (0x20..=0x7e).contains(&b));
            let structure = if tail {
                v.is_empty() || v.split('/').all(|x| !x.is_empty())
            } else {
                !v.is_empty() && !v.contains('/')
            };
            if !ascii || !structure {
                return Err(Error::InvalidValue { name: n.clone() });
            }
        }
        let mut out = String::new();
        for p in &e.parts {
            match p {
                Part::Lit(s) => {
                    out.push('/');
                    out.push_str(&encode(s));
                }
                Part::One(n) => {
                    out.push('/');
                    out.push_str(&encode(values[n.as_str()]));
                }
                Part::Tail(n) => {
                    let v = values[n.as_str()];
                    for x in v.split('/').filter(|x| !x.is_empty()) {
                        out.push('/');
                        out.push_str(&encode(x));
                    }
                }
            }
        }
        if out.is_empty() {
            out.push('/');
        }
        if e.slash && out != "/" {
            out.push('/');
        }
        Ok(out)
    }
}
