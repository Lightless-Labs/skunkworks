- Method validation precedes every other add/resolve check.
- A single trailing slash is stored separately; root has no trailing-slash flag.
- Wildcards can capture zero remaining segments, including when the request has a trailing slash.
- Under `Redirect`, the selected route’s slash flag determines the redirect, preserving the encoded request head.
- Conflicts require identical structural shapes; capture names do not distinguish shapes.
- Build checks duplicate keys, missing keys, extra keys, then values, in that order.

### Solution A

```rust
use std::collections::{BTreeSet, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone, Debug, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self { policy, routes: Vec::new() }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_pattern(pattern)?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }
        for r in &self.routes {
            if r.method == method && self.same_shape(r, &segments, trailing) {
                return Err(Error::Conflict { with: r.id });
            }
        }
        let id = self.routes.len();
        let rank = rank_for(&segments);
        self.routes.push(Route {
            id, name: name.to_string(), method: method.to_string(),
            segments, trailing, rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path.char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) = parse_path(head)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if let Some(params) = path_match(&route.segments, &request_segments) {
                if self.policy != SlashPolicy::Strict || route.trailing == request_trailing {
                    candidates.push((route, params));
                }
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<usize> = if method == "HEAD" {
            let heads: Vec<usize> = candidates.iter().enumerate()
                .filter(|(_, (r, _))| r.method == "HEAD")
                .map(|(i, _)| i).collect();
            if heads.is_empty() {
                candidates.iter().enumerate()
                    .filter(|(_, (r, _))| r.method == "GET")
                    .map(|(i, _)| i).collect()
            } else { heads }
        } else {
            candidates.iter().enumerate()
                .filter(|(_, (r, _))| r.method == method)
                .map(|(i, _)| i).collect()
        };
        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (r, _) in &candidates {
                allowed.insert(r.method.clone());
                if r.method == "GET" { allowed.insert("HEAD".to_string()); }
            }
            return Err(Error::MethodNotAllowed { allowed: allowed.into_iter().collect() });
        }

        let mut best = eligible[0];
        for &i in eligible.iter().skip(1) {
            let a = candidates[i].0;
            let b = candidates[best].0;
            if a.rank > b.rank || (a.rank == b.rank && a.id < b.id) {
                best = i;
            }
        }
        let (route, params) = &candidates[best];
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != request_trailing
            && !request_segments.is_empty()
        {
            let mut s = head.to_string();
            if route.trailing { s.push('/'); }
            else if s.ends_with('/') { s.pop(); }
            Some(s)
        } else { None };
        Ok(Match { route: route.id, params: params.clone(), redirect })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = match self.routes.iter().find(|r| r.name == name) {
            Some(r) => r,
            None => return Err(Error::UnknownName),
        };
        let mut values = std::collections::HashMap::<&str, &str>::new();
        for &(key, value) in params {
            if values.insert(key, value).is_some() {
                return Err(Error::DuplicateParam { name: key.to_string() });
            }
        }
        let captures: Vec<&str> = route.segments.iter().filter_map(|s| match s {
            Segment::Param(n) | Segment::Wildcard(n) => Some(n.as_str()),
            _ => None,
        }).collect();
        for key in &captures {
            if !values.contains_key(key) {
                return Err(Error::MissingParam { name: (*key).to_string() });
            }
        }
        let allowed: HashSet<&str> = captures.iter().copied().collect();
        if let Some(key) = values.keys().filter(|k| !allowed.contains(**k)).min() {
            return Err(Error::ExtraParam { name: (*key).to_string() });
        }
        for key in &captures {
            let value = values[key];
            let wildcard = route.segments.iter().any(|s| matches!(s, Segment::Wildcard(n) if n == key));
            if !valid_build_value(value, wildcard) {
                return Err(Error::InvalidValue { name: (*key).to_string() });
            }
        }

        let mut out = String::from("/");
        let mut first = true;
        for seg in &route.segments {
            if !first { out.push('/'); }
            first = false;
            match seg {
                Segment::Static(s) => out.push_str(&encode_segment(s)),
                Segment::Param(n) => out.push_str(&encode_segment(values[n.as_str()])),
                Segment::Wildcard(n) => {
                    let v = values[n.as_str()];
                    for (i, part) in v.split('/').enumerate() {
                        if i > 0 { out.push('/'); }
                        out.push_str(&encode_segment(part));
                    }
                }
            }
        }
        if route.trailing && !route.segments.is_empty() { out.push('/'); }
        Ok(out)
    }

    fn same_shape(&self, route: &Route, segments: &[Segment], trailing: bool) -> bool {
        if self.policy == SlashPolicy::Strict && route.trailing != trailing { return false; }
        if route.segments.len() != segments.len() { return false; }
        route.segments.iter().zip(segments).all(|(a, b)| match (a, b) {
            (Segment::Static(x), Segment::Static(y)) => x == y,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        })
    }
}

fn valid_method(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_uppercase())
}
fn valid_route_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty() && b[0].is_ascii_lowercase()
        && b.iter().all(|x| x.is_ascii_lowercase() || x.is_ascii_digit() || *x == b'_' || *x == b'-')
}
fn valid_capture_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty() && b[0].is_ascii_lowercase()
        && b.iter().all(|x| x.is_ascii_lowercase() || x.is_ascii_digit() || *x == b'_')
}
fn rank_for(segments: &[Segment]) -> Vec<u8> {
    let mut v = Vec::new();
    for s in segments {
        match s {
            Segment::Static(_) => v.push(2),
            Segment::Param(_) => v.push(1),
            Segment::Wildcard(_) => { v.push(0); return v; }
        }
    }
    v.push(3);
    v
}
fn parse_pattern(s: &str) -> Result<(Vec<Segment>, bool), Error> {
    parse(s, true).map_err(|at| Error::InvalidPattern { at })
}
fn parse_path(s: &str) -> Result<(Vec<Segment>, bool), Error> {
    parse(s, false).map_err(|at| Error::InvalidPath { at })
}
fn parse(s: &str, pattern: bool) -> Result<(Vec<Segment>, bool), usize> {
    let bytes = s.as_bytes();
    if bytes.first() != Some(&b'/') { return Err(0); }
    if s == "/" { return Ok((Vec::new(), false)); }
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut seen = HashSet::new();
    let mut start = 1;
    while start < end {
        let rel_end = bytes[start..end].iter().position(|b| *b == b'/').map(|x| start + x).unwrap_or(end);
        if rel_end == start { return Err(start); }
        let raw = &s[start..rel_end];
        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_capture_name(name) || !seen.insert(name.to_string()) { return Err(start); }
            if marker == b'*' && rel_end != end { return Err(start); }
            segments.push(if marker == b':' { Segment::Param(name.to_string()) } else { Segment::Wildcard(name.to_string()) });
        } else {
            let decoded = decode_segment(&bytes[start..rel_end]).map_err(|i| start + i)?;
            segments.push(Segment::Static(decoded));
        }
        if rel_end == end { break; }
        start = rel_end + 1;
        if start == end { break; }
    }
    Ok((segments, trailing))
}
fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}
fn hex_upper(b: u8) -> Option<u8> {
    match b { b'0'..=b'9' => Some(b-b'0'), b'A'..=b'F' => Some(b-b'A'+10), _ => None }
}
fn decode_segment(raw: &[u8]) -> Result<String, usize> {
    let mut out = Vec::new();
    let mut i = 0;
    while i < raw.len() {
        let b = raw[i];
        if b == b'%' {
            if i + 2 >= raw.len() { return Err(i); }
            let hi = hex_upper(raw[i+1]).ok_or(i)?;
            let lo = hex_upper(raw[i+2]).ok_or(i)?;
            let d = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&d) || d == b'/' || is_unreserved(d) { return Err(i); }
            out.push(d);
            i += 3;
        } else {
            if !is_unreserved(b) { return Err(i); }
            out.push(b);
            i += 1;
        }
    }
    String::from_utf8(out).map_err(|_| 0)
}
fn path_match(pattern: &[Segment], path: &[Segment]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut i = 0;
    for seg in pattern {
        match seg {
            Segment::Wildcard(name) => {
                let rest = path[i..].iter().map(|s| match s { Segment::Static(v) => v.as_str(), _ => "" }).collect::<Vec<_>>().join("/");
                params.push((name.clone(), rest));
                return Some(params);
            }
            Segment::Static(expected) => match path.get(i) {
                Some(Segment::Static(actual)) if actual == expected => {},
                _ => return None,
            },
            Segment::Param(name) => match path.get(i) {
                Some(Segment::Static(value)) => params.push((name.clone(), value.clone())),
                _ => return None,
            },
        }
        i += 1;
    }
    if i == path.len() { Some(params) } else { None }
}
fn valid_build_value(value: &str, wildcard: bool) -> bool {
    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b)) { return false; }
    if wildcard {
        value.is_empty() || value.split('/').all(|p| !p.is_empty())
    } else {
        !value.is_empty() && !value.contains('/')
    }
}
fn encode_segment(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if is_unreserved(b) { out.push(b as char); }
        else {
            const H: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%'); out.push(H[(b >> 4) as usize] as char); out.push(H[(b & 15) as usize] as char);
        }
    }
    out
}
```

### Solution B

```rust
use std::collections::{BTreeSet, HashMap, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy { Strict, Ignore, Redirect }

#[derive(Clone)]
enum Part { Lit(String), Arg(String), Tail(String) }
struct Entry {
    id: usize,
    name: String,
    method: String,
    parts: Vec<Part>,
    slash: bool,
    rank: Vec<u8>,
}
pub struct Router { policy: SlashPolicy, entries: Vec<Entry> }

impl Router {
    pub fn new(policy: SlashPolicy) -> Self { Self { policy, entries: Vec::new() } }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !method_ok(method) { return Err(Error::InvalidMethod); }
        if !name_ok(name) { return Err(Error::InvalidName); }
        let (parts, slash) = parse(pattern, true).map_err(|at| Error::InvalidPattern { at })?;
        if self.entries.iter().any(|e| e.name == name) { return Err(Error::DuplicateName); }
        let id = self.entries.len();
        for e in &self.entries {
            if e.method == method && shape_equal(&e.parts, &parts)
                && (self.policy != SlashPolicy::Strict || e.slash == slash) {
                return Err(Error::Conflict { with: e.id });
            }
        }
        let rank = make_rank(&parts);
        self.entries.push(Entry { id, name: name.to_owned(), method: method.to_owned(), parts, slash, rank });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !method_ok(method) { return Err(Error::InvalidMethod); }
        let cut = path.bytes().position(|b| b == b'?' || b == b'#').unwrap_or(path.len());
        let head = &path[..cut];
        let (segments, slash) = parse(head, false).map_err(|at| Error::InvalidPath { at })?;

        let mut found: Vec<(&Entry, Vec<(String, String)>)> = Vec::new();
        for e in &self.entries {
            if self.policy == SlashPolicy::Strict && e.slash != slash { continue; }
            if let Some(p) = matches_parts(&e.parts, &segments) { found.push((e, p)); }
        }
        if found.is_empty() { return Err(Error::NotFound); }
        let use_head = method == "HEAD" && found.iter().any(|(e, _)| e.method == "HEAD");
        let wanted = if method == "HEAD" && !use_head { "GET" } else { method };
        let mut choices: Vec<usize> = found.iter().enumerate()
            .filter(|(_, (e, _))| e.method == wanted).map(|(i, _)| i).collect();
        if choices.is_empty() {
            let mut set = BTreeSet::new();
            for (e, _) in &found {
                set.insert(e.method.clone());
                if e.method == "GET" { set.insert("HEAD".to_string()); }
            }
            return Err(Error::MethodNotAllowed { allowed: set.into_iter().collect() });
        }
        choices.sort_by(|a, b| found[*b].0.rank.cmp(&found[*a].0.rank)
            .then_with(|| found[*a].0.id.cmp(&found[*b].0.id)));
        let (winner, params) = &found[choices[0]];
        let redirect = if self.policy == SlashPolicy::Redirect && winner.slash != slash
            && !segments.is_empty() {
            let mut p = head.to_string();
            if winner.slash { p.push('/'); } else { p.pop(); }
            Some(p)
        } else { None };
        Ok(Match { route: winner.id, params: params.clone(), redirect })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let e = self.entries.iter().find(|e| e.name == name).ok_or(Error::UnknownName)?;
        let mut values: HashMap<&str, &str> = HashMap::new();
        for &(k, v) in params {
            if values.contains_key(k) { return Err(Error::DuplicateParam { name: k.to_owned() }); }
            values.insert(k, v);
        }
        let names: Vec<&str> = e.parts.iter().filter_map(|p| match p {
            Part::Arg(n) | Part::Tail(n) => Some(n.as_str()), _ => None
        }).collect();
        for n in &names {
            if !values.contains_key(n) { return Err(Error::MissingParam { name: (*n).to_owned() }); }
        }
        let valid: HashSet<&str> = names.iter().copied().collect();
        if let Some(k) = values.keys().filter(|k| !valid.contains(**k)).min() {
            return Err(Error::ExtraParam { name: (*k).to_owned() });
        }
        for n in &names {
            let tail = e.parts.iter().any(|p| matches!(p, Part::Tail(x) if x == n));
            if !build_value_ok(values[n], tail) { return Err(Error::InvalidValue { name: (*n).to_owned() }); }
        }

        let mut out = String::from("/");
        for (i, part) in e.parts.iter().enumerate() {
            if i > 0 { out.push('/'); }
            match part {
                Part::Lit(s) => out.push_str(&encode(s)),
                Part::Arg(n) => out.push_str(&encode(values[n.as_str()])),
                Part::Tail(n) => {
                    for (j, component) in values[n.as_str()].split('/').enumerate() {
                        if j > 0 { out.push('/'); }
                        out.push_str(&encode(component));
                    }
                }
            }
        }
        if e.slash && !e.parts.is_empty() { out.push('/'); }
        Ok(out)
    }
}

fn method_ok(s: &str) -> bool { !s.is_empty() && s.bytes().all(|b| b'A' <= b && b <= b'Z') }
fn name_ok(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty() && b[0].is_ascii_lowercase()
        && b.iter().all(|x| x.is_ascii_lowercase() || x.is_ascii_digit() || *x == b'_' || *x == b'-')
}
fn capture_ok(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty() && b[0].is_ascii_lowercase()
        && b.iter().all(|x| x.is_ascii_lowercase() || x.is_ascii_digit() || *x == b'_')
}
fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}
fn hex(b: u8) -> Option<u8> {
    match b { b'0'..=b'9' => Some(b - b'0'), b'A'..=b'F' => Some(b - b'A' + 10), _ => None }
}
fn decode(raw: &[u8]) -> Result<String, usize> {
    let mut v = Vec::new();
    let mut i = 0;
    while i < raw.len() {
        if raw[i] == b'%' {
            if i + 2 >= raw.len() { return Err(i); }
            let h = hex(raw[i + 1]).ok_or(i)?;
            let l = hex(raw[i + 2]).ok_or(i)?;
            let b = h * 16 + l;
            if !(0x20..=0x7e).contains(&b) || b == b'/' || unreserved(b) { return Err(i); }
            v.push(b);
            i += 3;
        } else {
            if !unreserved(raw[i]) { return Err(i); }
            v.push(raw[i]);
            i += 1;
        }
    }
    Ok(String::from_utf8(v).unwrap())
}
fn parse(s: &str, pattern: bool) -> Result<(Vec<Part>, bool), usize> {
    let b = s.as_bytes();
    if b.first() != Some(&b'/') { return Err(0); }
    if s == "/" { return Ok((Vec::new(), false)); }
    let slash = b.last() == Some(&b'/');
    let stop = if slash { b.len() - 1 } else { b.len() };
    let mut result = Vec::new();
    let mut names = HashSet::new();
    let mut start = 1;
    while start < stop {
        let end = b[start..stop].iter().position(|x| *x == b'/').map(|x| start + x).unwrap_or(stop);
        if end == start { return Err(start); }
        let raw = &s[start..end];
        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let wildcard = raw.as_bytes()[0] == b'*';
            let n = &raw[1..];
            if !capture_ok(n) || !names.insert(n.to_owned()) { return Err(start); }
            if wildcard && end != stop { return Err(start); }
            result.push(if wildcard { Part::Tail(n.to_owned()) } else { Part::Arg(n.to_owned()) });
        } else {
            result.push(Part::Lit(decode(&b[start..end]).map_err(|x| start + x)?));
        }
        if end == stop { break; }
        start = end + 1;
    }
    Ok((result, slash))
}
fn make_rank(parts: &[Part]) -> Vec<u8> {
    let mut r = Vec::new();
    for p in parts {
        match p { Part::Lit(_) => r.push(2), Part::Arg(_) => r.push(1), Part::Tail(_) => { r.push(0); return r; } }
    }
    r.push(3);
    r
}
fn shape_equal(a: &[Part], b: &[Part]) -> bool {
    a.len() == b.len() && a.iter().zip(b).all(|(x, y)| match (x, y) {
        (Part::Lit(a), Part::Lit(b)) => a == b,
        (Part::Arg(_), Part::Arg(_)) | (Part::Tail(_), Part::Tail(_)) => true,
        _ => false,
    })
}
fn matches_parts(pattern: &[Part], path: &[Part]) -> Option<Vec<(String, String)>> {
    let mut out = Vec::new();
    let mut i = 0;
    for p in pattern {
        match p {
            Part::Tail(n) => {
                let tail = path[i..].iter().map(|x| match x { Part::Lit(s) => s.as_str(), _ => "" }).collect::<Vec<_>>().join("/");
                out.push((n.clone(), tail));
                return Some(out);
            }
            Part::Lit(s) => match path.get(i) { Some(Part::Lit(v)) if v == s => (), _ => return None },
            Part::Arg(n) => match path.get(i) { Some(Part::Lit(v)) => out.push((n.clone(), v.clone())), _ => return None },
        }
        i += 1;
    }
    if i == path.len() { Some(out) } else { None }
}
fn build_value_ok(s: &str, tail: bool) -> bool {
    if !s.bytes().all(|b| (0x20..=0x7e).contains(&b)) { return false; }
    if tail { s.is_empty() || s.split('/').all(|p| !p.is_empty()) }
    else { !s.is_empty() && !s.contains('/') }
}
fn encode(s: &str) -> String {
    let mut out = String::new();
    const H: &[u8; 16] = b"0123456789ABCDEF";
    for b in s.bytes() {
```
