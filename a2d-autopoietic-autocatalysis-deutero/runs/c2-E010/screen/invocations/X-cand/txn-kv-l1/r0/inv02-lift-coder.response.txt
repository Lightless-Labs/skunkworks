- Empty keys are rejected before map access; empty values and empty `count` arguments are valid.
- Name validation is left to right by UTF-8 byte offset, before registry lookup or capacity checks.
- The four-scope limit allows exactly four open scopes; the fifth `begin` changes nothing.
- Snapshots are independent copies and registry changes survive rollback.
- Committing a nested scope replaces its parent’s full map; rolling back an outer scope still discards that result.
- Solution A uses a vector of complete scope maps; Solution B uses a stack of scope maps with a separate visible-map helper.

### Solution A

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    NoTransaction,
    DepthLimit,
    SnapshotMissing(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Store {
    committed: BTreeMap<String, String>,
    scopes: Vec<BTreeMap<String, String>>,
    snapshots: BTreeMap<String, BTreeMap<String, String>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            committed: BTreeMap::new(),
            scopes: Vec::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn visible(&self) -> &BTreeMap<String, String> {
        self.scopes.last().unwrap_or(&self.committed)
    }

    fn visible_mut(&mut self) -> &mut BTreeMap<String, String> {
        if self.scopes.is_empty() {
            &mut self.committed
        } else {
            self.scopes.last_mut().unwrap()
        }
    }

    fn validate_key(key: &str) -> Result<(), Error> {
        if key.is_empty() {
            Err(Error::KeyEmpty)
        } else {
            Ok(())
        }
    }

    fn validate_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (i, b) in name.bytes().enumerate() {
            let valid = if i == 0 {
                b.is_ascii_lowercase()
            } else {
                b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_'
            };
            if !valid {
                return Err(Error::NameInvalid { at: i });
            }
        }
        Ok(())
    }

    pub fn begin(&mut self) -> Result<(), Error> {
        if self.scopes.len() >= 4 {
            return Err(Error::DepthLimit);
        }
        self.scopes.push(self.visible().clone());
        Ok(())
    }

    pub fn commit(&mut self) -> Result<(), Error> {
        let map = self.scopes.pop().ok_or(Error::NoTransaction)?;
        if let Some(parent) = self.scopes.last_mut() {
            *parent = map;
        } else {
            self.committed = map;
        }
        Ok(())
    }

    pub fn rollback(&mut self) -> Result<(), Error> {
        self.scopes.pop().map(|_| ()).ok_or(Error::NoTransaction)
    }

    pub fn set(&mut self, key: &str, value: &str) -> Result<(), Error> {
        Self::validate_key(key)?;
        self.visible_mut().insert(key.to_string(), value.to_string());
        Ok(())
    }

    pub fn delete(&mut self, key: &str) -> Result<bool, Error> {
        Self::validate_key(key)?;
        Ok(self.visible_mut().remove(key).is_some())
    }

    pub fn get(&mut self, key: &str) -> Result<Option<String>, Error> {
        Self::validate_key(key)?;
        Ok(self.visible().get(key).cloned())
    }

    pub fn count(&mut self, value: &str) -> Result<usize, Error> {
        Ok(self.visible().values().filter(|v| v.as_str() == value).count())
    }

    pub fn depth(&mut self) -> Result<usize, Error> {
        Ok(self.scopes.len())
    }

    pub fn snapshot(&mut self, name: &str) -> Result<(), Error> {
        Self::validate_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        self.snapshots.insert(name.to_string(), self.visible().clone());
        Ok(())
    }

    pub fn restore(&mut self, name: &str) -> Result<(), Error> {
        Self::validate_name(name)?;
        let saved = self
            .snapshots
            .get(name)
            .cloned()
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        *self.visible_mut() = saved;
        Ok(())
    }

    pub fn forget(&mut self, name: &str) -> Result<(), Error> {
        Self::validate_name(name)?;
        self.snapshots
            .remove(name)
            .map(|_| ())
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))
    }
}
```

### Solution B

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    NoTransaction,
    DepthLimit,
    SnapshotMissing(String),
}

pub struct Store {
    base: BTreeMap<String, String>,
    transactions: Vec<BTreeMap<String, String>>,
    saved: BTreeMap<String, BTreeMap<String, String>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            base: BTreeMap::new(),
            transactions: Vec::new(),
            saved: BTreeMap::new(),
        }
    }

    fn current(&self) -> &BTreeMap<String, String> {
        match self.transactions.last() {
            Some(map) => map,
            None => &self.base,
        }
    }

    fn current_mut(&mut self) -> &mut BTreeMap<String, String> {
        match self.transactions.last_mut() {
            Some(map) => map,
            None => &mut self.base,
        }
    }

    fn key_ok(key: &str) -> Result<(), Error> {
        if key.is_empty() {
            Err(Error::KeyEmpty)
        } else {
            Ok(())
        }
    }

    fn name_ok(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (offset, byte) in name.bytes().enumerate() {
            let ok = match offset {
                0 => byte.is_ascii_lowercase(),
                _ => byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'_',
            };
            if !ok {
                return Err(Error::NameInvalid { at: offset });
            }
        }
        Ok(())
    }

    pub fn begin(&mut self) -> Result<(), Error> {
        if self.transactions.len() == 4 {
            return Err(Error::DepthLimit);
        }
        let copy = self.current().clone();
        self.transactions.push(copy);
        Ok(())
    }

    pub fn commit(&mut self) -> Result<(), Error> {
        match self.transactions.pop() {
            None => Err(Error::NoTransaction),
            Some(map) => {
                if let Some(parent) = self.transactions.last_mut() {
                    *parent = map;
                } else {
                    self.base = map;
                }
                Ok(())
            }
        }
    }

    pub fn rollback(&mut self) -> Result<(), Error> {
        if self.transactions.pop().is_some() {
            Ok(())
        } else {
            Err(Error::NoTransaction)
        }
    }

    pub fn set(&mut self, key: &str, value: &str) -> Result<(), Error> {
        Self::key_ok(key)?;
        self.current_mut().insert(key.to_owned(), value.to_owned());
        Ok(())
    }

    pub fn delete(&mut self, key: &str) -> Result<bool, Error> {
        Self::key_ok(key)?;
        Ok(self.current_mut().remove(key).is_some())
    }

    pub fn get(&mut self, key: &str) -> Result<Option<String>, Error> {
        Self::key_ok(key)?;
        Ok(self.current().get(key).cloned())
    }

    pub fn count(&mut self, value: &str) -> Result<usize, Error> {
        Ok(self.current().values().filter(|item| item.as_str() == value).count())
    }

    pub fn depth(&mut self) -> Result<usize, Error> {
        Ok(self.transactions.len())
    }

    pub fn snapshot(&mut self, name: &str) -> Result<(), Error> {
        Self::name_ok(name)?;
        if !self.saved.contains_key(name) && self.saved.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        let copy = self.current().clone();
        self.saved.insert(name.to_owned(), copy);
        Ok(())
    }

    pub fn restore(&mut self, name: &str) -> Result<(), Error> {
        Self::name_ok(name)?;
        let copy = match self.saved.get(name) {
            Some(map) => map.clone(),
            None => return Err(Error::SnapshotMissing(name.to_owned())),
        };
        *self.current_mut() = copy;
        Ok(())
    }

    pub fn forget(&mut self, name: &str) -> Result<(), Error> {
        Self::name_ok(name)?;
        if self.saved.remove(name).is_some() {
            Ok(())
        } else {
            Err(Error::SnapshotMissing(name.to_owned()))
        }
    }
}
```
