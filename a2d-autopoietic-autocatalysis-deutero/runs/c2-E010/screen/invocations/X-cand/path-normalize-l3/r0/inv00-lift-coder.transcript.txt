#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .last()
            .is_some_and(|component| component == "." || component == "..")
}

fn raw_components(path: &str, base: usize) -> Vec<(String, usize)> {
    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((path[start..i].to_string(), base + start));
    }
    result
}

fn push_component(
    stack: &mut Vec<String>,
    component: &str,
    absolute: bool,
    at: usize,
) -> Result<(), PathError> {
    match component {
        "" | "." => Ok(()),
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_string());
            }
            Ok(())
        }
        name => {
            stack.push(name.to_string());
            Ok(())
        }
    }
}

fn render(stack: &[String], trailing: bool) -> String {
    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    if trailing && !stack.is_empty() {
        result.push('/');
    }
    result
}

fn normalize_valid(path: &str) -> Result<String, PathError> {
    let absolute = path.as_bytes()[0] == b'/';
    let mut stack = Vec::new();
    for (component, at) in raw_components(path, 0) {
        push_component(&mut stack, &component, absolute, at)?;
    }
    if absolute {
        Ok(render(&stack, directory_intent(path)))
    } else {
        let mut result = if stack.is_empty() {
            ".".to_string()
        } else {
            stack.join("/")
        };
        if directory_intent(path) && result != "." {
            result.push('/');
        }
        Ok(result)
    }
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    normalize_valid(path)
}

fn validate_context(path: &str) -> Result<Vec<String>, ()> {

        return Err(());

    let normalized = normalize(path).map_err(|_| ())?;
    Ok(normalized

        .filter(|component| !component.is_empty())
        .map(str::to_string)

}

    let home_stack = match home {
        Some(value) => Some(validate_context(value).map_err(|_| PathError::InvalidHome)?),

}

    next: usize,
    source: Source,
    active_link: Option<usize>,
}

    Request,
    Target,

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_stack = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_stack = match home {
        Some(value) => Some(validate_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    let mut table = HashMap::<String, (usize, String)>::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path)
            && link.path != "/";
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((first, _)) = table.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        table.insert(link.path.clone(), (index, link.target.clone()));
    }

    validate_raw(path)?;
    let (mut stack, request, request_base) = if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home_stack = home_stack.ok_or(PathError::MissingHome)?;
        let base = if bytes.len() > 1 { 2 } else { 1 };
        (home_stack, &path[base..], base)
    } else if path.starts_with('/') {
        (Vec::new(), path, 0)
    } else {
        (cwd_stack, path, 0)
    };

    let mut frames = Vec::new();
    frames.push(Frame {
        components: raw_components(request, request_base),
        next: 0,
        source: Source::Request,
        active_link: None,
    });
    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    for component in stack.clone() {
        push_component(&mut stack, &component, true, 0)?;
        let key = format!("/{}", stack.join("/"));
        if let Some((index, target)) = table.get(&key).cloned() {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;
            active.push(index);
            let target_components = raw_components(&target, 0);
            if target.starts_with('/') {
                stack.clear();
            } else {
                stack.pop();
            }
            frames.push(Frame {
                components: target_components,
                next: 0,
                source: Source::Target,
                active_link: Some(index),
            });
        }
    }

    while !frames.is_empty() {
        let frame_index = frames.len() - 1;
        if frames[frame_index].next >= frames[frame_index].components.len() {
            let frame = frames.pop().unwrap();
            if let Some(index) = frame.active_link {
                if let Some(position) = active.iter().rposition(|&item| item == index) {
                    active.remove(position);
                }
            }
            continue;
        }

        let (component, at) = {
            let frame = &mut frames[frame_index];
            let item = frame.components[frame.next].clone();
            frame.next += 1;
            item
        };

        if component == "." {
            continue;
        }
        if component == ".." {
            if stack.is_empty() {
                return Err(PathError::Escape { at });
            }
            stack.pop();
            continue;
        }

        stack.push(component);
        let key = format!("/{}", stack.join("/"));
        if let Some((index, target)) = table.get(&key).cloned() {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;
            active.push(index);
            if target.starts_with('/') {
                stack.clear();
            } else {
                stack.pop();
            }
            frames.push(Frame {
                components: raw_components(&target, 0),
                next: 0,
                source: Source::Target,
                active_link: Some(index),
            });
        }
    }

    let _ = request;
    let _ = request_base;
    let _ = Source::Request;
    let _ = Source::Target;
    Ok(render(&stack, directory_intent(path)))
}

use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    let bytes = path.as_bytes();
    if let Some(at) = bytes.iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = bytes.iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = bytes.iter().position(|&b| (1..=31).contains(&b) || b == 127) {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|s| !s.is_empty())
            .last()
            .is_some_and(|s| s == "." || s == "..")
}

fn components(path: &str, base: usize) -> Vec<(String, usize)> {
    let mut out = Vec::new();
    let bytes = path.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        if start < i {
            out.push((path[start..i].to_owned(), base + start));
        }
    }
    out
}

fn normalized_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    Ok(normalized
        .split('/')
        .filter(|s| !s.is_empty())
        .map(str::to_owned)
        .collect())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let mut stack: Vec<String> = Vec::new();
    for (component, at) in components(path, 0) {
        match component.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|s| s != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(component);
                }
            }
            _ => stack.push(component),
        }
    }

    let mut result = if absolute {
        if stack.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", stack.join("/"))
        }
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };
    if directory_intent(path) && result != "/" && result != "." {
        result.push('/');
    }
    Ok(result)
}

fn reduce_absolute(
    stack: &mut Vec<String>,
    path: &str,
    base: usize,
) -> Result<(), PathError> {
    for (component, at) in components(path, base) {
        match component.as_str() {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => stack.push(component),
        }
    }
    Ok(())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_stack = normalized_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_stack = home
        .map(normalized_context)
        .transpose()
        .map_err(|_| PathError::InvalidHome)?;

    validate_raw(path)?;
    let (mut stack, rest, base) = if path.starts_with('~') {
        if path.len() > 1 && path.as_bytes()[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home_stack = home_stack.ok_or(PathError::MissingHome)?;
        let base = if path.len() > 1 { 2 } else { 1 };
        (home_stack, &path[base..], base)
    } else if path.starts_with('/') {
        (Vec::new(), path, 0)
    } else {
        (cwd_stack, path, 0)
    };
    reduce_absolute(&mut stack, rest, base)?;
    let mut result = if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    };
    if directory_intent(path) && !stack.is_empty() {
        result.push('/');
    }
    Ok(result)
}

fn process_components(
    stack: &mut Vec<String>,
    items: &[(String, usize)],
    links: &HashMap<String, (usize, String)>,
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
    inherited_index: Option<usize>,
) -> Result<(), PathError> {
    for (component, at) in items {
        match component.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: *at });
                }
                continue;
            }
            _ => stack.push(component.clone()),
        }

        let key = format!("/{}", stack.join("/"));
        if let Some((index, target)) = links.get(&key) {
            if active.contains(index) {
                return Err(PathError::Loop { index: *index });
            }
            if *expansions >= max_expansions {
                return Err(PathError::Limit { index: *index });
            }
            *expansions += 1;
            active.push(*index);

            let parent_len = stack.len().saturating_sub(1);
            stack.truncate(parent_len);
            if target.starts_with('/') {
                stack.clear();
            }
            process_components(
                stack,
                &components(target, 0),
                links,
                active,
                expansions,
                max_expansions,
                Some(*index),
            )?;
            if active.last() == Some(index) {
                active.pop();
            }
        }
    }
    let _ = inherited_index;
    Ok(())
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_stack = normalized_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_stack = home
        .map(normalized_context)
        .transpose()
        .map_err(|_| PathError::InvalidHome)?;

    let mut table = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid = link.path.starts_with('/')
            && link.path != "/"
            && !link.path.ends_with('/')
            && normalize(&link.path).is_ok_and(|value| value == link.path)
            && validate_raw(&link.target).is_ok();
        if !valid {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((first, _)) = table.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        table.insert(link.path.clone(), (index, link.target.clone()));
    }

    validate_raw(path)?;
    let (mut stack, rest, base) = if path.starts_with('~') {
        if path.len() > 1 && path.as_bytes()[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home_stack = home_stack.ok_or(PathError::MissingHome)?;
        let base = if path.len() > 1 { 2 } else { 1 };
        (home_stack, &path[base..], base)
    } else if path.starts_with('/') {
        (Vec::new(), path, 0)
    } else {
        (cwd_stack, path, 0)
    };

    let mut active = Vec::new();
    let mut expansions = 0usize;
    process_components(
        &mut stack,
        &components(rest, base),
        &table,
        &mut active,
        &mut expansions,
        max_expansions,
        None,
    )?;

    let mut result = if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    };
    if directory_intent(path) && !stack.is_empty() {
        result.push('/');
    }
    Ok(result)
}
