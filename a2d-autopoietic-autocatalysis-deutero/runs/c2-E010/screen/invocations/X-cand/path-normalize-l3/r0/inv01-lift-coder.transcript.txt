fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()

        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.split('/').filter(|c| !c.is_empty()).last().is_some_and(|c| c == "." || c == "..")

fn normalized_parts(path: &str) -> Result<(bool, Vec<String>), PathError> {

    let absolute = path.as_bytes().first() == Some(&b'/');
    let mut parts: Vec<String> = Vec::new();

            "." => {}

                    parts.pop();
                } else if absolute {
                    let at = path.find(comp).unwrap_or(0);
                    return Err(PathError::Escape { at });
                } else {
                    parts.push("..".to_string());
                }
            }
            _ => parts.push(comp.to_string()),

    }
    Ok((absolute, parts))
}

        body
    };

        out.push('/');
    }
    out
}

}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {

    let home_parts = validate_home(home)?;

    let bytes = path.as_bytes();
    let mut stack: Vec<String>;
    let mut req_start = 0usize;
    if path.starts_with('~') {
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            stack = home_parts.ok_or(PathError::MissingHome)?;
            req_start = if bytes.len() > 1 { 2 } else { 1 };
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        stack = Vec::new();
    } else {
        stack = cwd_parts;
    }

    let mut off = req_start;
    for comp in path[req_start..].split('/') {
        let at = off;
        off += comp.len() + 1;
        if comp.is_empty() || comp == "." {
            continue;
        }

            if stack.pop().is_none() {
                return Err(PathError::Escape { at });
            }
        } else {
            stack.push(comp.to_string());
        }
    }
    Ok(render(&stack, true, intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (i, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path).ok().as_deref() == Some(link.path.as_str());
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index: i });
        }
        if let Some((_, first)) = seen.iter().find(|(p, _)| *p == link.path) {
            return Err(PathError::DuplicateLink { first: *first, second: i });
        }
        seen.push((&link.path, i));
    }
    Ok(())
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validate_home(home)?;
    validate_links(links)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let bytes = path.as_bytes();
    let (mut stack, req_start) = if path.starts_with('~') {
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            (home_parts.ok_or(PathError::MissingHome)?, if bytes.len() > 1 { 2 } else { 1 })
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        (Vec::new(), 0)
    } else {
        (cwd_parts, 0)
    };

    struct Frame {
        comps: Vec<(String, usize)>,
        pos: usize,
        source: String,
        link_index: Option<usize>,
    }

    let mut initial = Vec::new();
    let mut off = req_start;
    for c in path[req_start..].split('/') {
        let at = off;
        off += c.len() + 1;
        if !c.is_empty() {
            initial.push((c.to_string(), at));
        }
    }
    let mut frames = vec![Frame {
        comps: initial,
        pos: 0,
        source: path.to_string(),
        link_index: None,
    }];
    let mut active: Vec<usize> = Vec::new();
    let mut expansions = 0usize;

    while !frames.is_empty() {
        let frame_index = frames.len() - 1;
        if frames[frame_index].pos >= frames[frame_index].comps.len() {
            let done = frames.pop().unwrap();
            if let Some(i) = done.link_index {
                if active.last() == Some(&i) {
                    active.pop();
                } else if let Some(pos) = active.iter().rposition(|&x| x == i) {
                    active.remove(pos);
                }
            }
            continue;
        }
        let (comp, at) = {
            let frame = &mut frames[frame_index];
            let item = frame.comps[frame.pos].clone();
            frame.pos += 1;
            item
        };
        if comp == "." {
            continue;
        }
        if comp == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at });
            }
            continue;
        }
        stack.push(comp);
        let key = format!("/{}", stack.join("/"));
        if let Some((link_index, link)) = links.iter().enumerate().find(|(_, l)| l.path == key) {
            if active.contains(&link_index) {
                return Err(PathError::Loop { index: link_index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index: link_index });
            }
            expansions += 1;
            active.push(link_index);

            let target = &link.target;
            if target.starts_with('/') {
                stack.clear();
            } else {
                stack.pop();
            }
            let mut target_comps = Vec::new();
            let mut target_off = 0usize;
            for c in target.split('/') {
                let here = target_off;
                target_off += c.len() + 1;
                if !c.is_empty() {
                    target_comps.push((c.to_string(), here));
                }
            }
            frames.push(Frame {
                comps: target_comps,
                pos: 0,
                source: target.clone(),
                link_index: Some(link_index),
            });
        }
    }

    Ok(render(&stack, true, intent))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn check(s: &str) -> Result<(), PathError> {
    if s.is_empty() {
        return Err(PathError::Empty);
    }
    for (i, b) in s.bytes().enumerate() {
        if b == 0 {
            return Err(PathError::Nul { at: i });
        }
    }
    for (i, b) in s.bytes().enumerate() {
        if b == b'\\' {
            return Err(PathError::Backslash { at: i });
        }
    }
    for (i, b) in s.bytes().enumerate() {
        if (b >= 1 && b <= 31) || b == 127 {
            return Err(PathError::Control { at: i });
        }
    }
    Ok(())
}

fn intent(s: &str) -> bool {
    s.ends_with('/')
        || s.split('/').filter(|x| !x.is_empty()).last().is_some_and(|x| x == "." || x == "..")
}

fn reduce_component(stack: &mut Vec<String>, absolute: bool, comp: &str, at: usize) -> Result<(), PathError> {
    match comp {
        "" | "." => Ok(()),
        ".." => {
            if stack.last().is_some_and(|x| x != "..") {
                stack.pop();
            } else if absolute {
                Err(PathError::Escape { at })
            } else {
                stack.push("..".into());
                Ok(())
            }
        }
        _ => {
            stack.push(comp.to_string());
            Ok(())
        }
    }
}

fn output(stack: &[String], absolute: bool, dir: bool) -> String {
    let joined = stack.join("/");
    let mut value = if absolute {
        if joined.is_empty() { "/".into() } else { format!("/{joined}") }
    } else if joined.is_empty() {
        ".".into()
    } else {
        joined
    };
    if dir && value != "/" && value != "." {
        value.push('/');
    }
    value
}

fn reduce(path: &str, absolute: bool) -> Result<Vec<String>, PathError> {
    check(path)?;
    let mut stack = Vec::new();
    let mut offset = 0;
    for c in path.split('/') {
        let at = offset;
        offset += c.len() + 1;
        reduce_component(&mut stack, absolute, c, at)?;
    }
    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    check(path)?;
    let abs = path.starts_with('/');
    let stack = reduce(path, abs)?;
    Ok(output(&stack, abs, intent(path)))
}

fn context(s: &str, err: PathError) -> Result<Vec<String>, PathError> {
    if !s.starts_with('/') {
        return Err(err);
    }
    reduce(s, true).map_err(|_| err)
}

fn request_start<'a>(
    path: &'a str,
    cwd: Vec<String>,
    home: Option<Vec<String>>,
) -> Result<(Vec<String>, usize), PathError> {
    let b = path.as_bytes();
    if path.starts_with('~') {
        if b.len() == 1 || b.get(1) == Some(&b'/') {
            let h = home.ok_or(PathError::MissingHome)?;
            Ok((h, if b.len() == 1 { 1 } else { 2 }))
        } else {
            Err(PathError::UnsupportedTilde { at: 1 })
        }
    } else if path.starts_with('/') {
        Ok((Vec::new(), 0))
    } else {
        Ok((cwd, 0))
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(h) => Some(context(h, PathError::InvalidHome)?),
        None => None,
    };
    check(path)?;
    let (mut stack, start) = request_start(path, cwd_parts, home_parts)?;
    let mut offset = start;
    for c in path[start..].split('/') {
        let at = offset;
        offset += c.len() + 1;
        reduce_component(&mut stack, true, c, at)?;
    }
    Ok(output(&stack, true, intent(path)))
}

fn validate_table(links: &[Link]) -> Result<(), PathError> {
    let mut paths: Vec<&str> = Vec::new();
    for (i, l) in links.iter().enumerate() {
        let path_ok = l.path.starts_with('/')
            && l.path != "/"
            && !l.path.ends_with('/')
            && normalize(&l.path).ok().as_deref() == Some(l.path.as_str());
        let target_ok = check(&l.target).is_ok();
        if !path_ok || !target_ok {
            return Err(PathError::InvalidLink { index: i });
        }
        if let Some(first) = paths.iter().position(|p| *p == l.path) {
            return Err(PathError::DuplicateLink { first, second: i });
        }
        paths.push(&l.path);
    }
    Ok(())
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(h) => Some(context(h, PathError::InvalidHome)?),
        None => None,
    };
    validate_table(links)?;
    check(path)?;
    let (mut stack, start) = request_start(path, cwd_parts, home_parts)?;
    let request_intent = intent(path);

    let mut work: Vec<(String, usize, usize)> = Vec::new();
    let mut offset = start;
    let req: Vec<(String, usize)> = path[start..]
        .split('/')
        .map(|c| {
            let at = offset;
            offset += c.len() + 1;
            (c.to_string(), at)
        })
        .collect();
    for (c, at) in req.into_iter().rev() {
        work.push((c, at, usize::MAX));
    }

    let mut active = Vec::new();
    let mut expansions = 0usize;
    while let Some((comp, at, owner)) = work.pop() {
        if owner != usize::MAX && comp.is_empty() {
            if active.last() == Some(&owner) {
                active.pop();
            }
            continue;
        }
        match comp.as_str() {
            "" | "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => {
                stack.push(comp.clone());
                let key = format!("/{}", stack.join("/"));
                if let Some((idx, link)) = links.iter().enumerate().find(|(_, l)| l.path == key) {
                    if active.contains(&idx) {
                        return Err(PathError::Loop { index: idx });
                    }
                    if expansions >= max_expansions {
                        return Err(PathError::Limit { index: idx });
                    }
                    expansions += 1;
                    active.push(idx);
                    stack.pop();

                    let mut target: Vec<(String, usize, usize)> = Vec::new();
                    let mut pos = 0;
                    for c in link.target.split('/') {
                        let here = pos;
                        pos += c.len() + 1;
                        target.push((c.to_string(), here, idx));
                    }
                    if link.target.starts_with('/') {
                        stack.clear();
                    }
                    for item in target.into_iter().rev() {
                        work.push(item);
                    }
                    work.push((String::new(), 0, idx));
                }
            }
        }
    }
    Ok(output(&stack, true, request_intent))
}
