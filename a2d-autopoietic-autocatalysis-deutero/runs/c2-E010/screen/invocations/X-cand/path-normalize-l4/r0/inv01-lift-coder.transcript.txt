use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b < 32 || b == 127) {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.rsplit('/')
        .find(|part| !part.is_empty())
        .map_or(false, |part| part == "." || part == "..")
}

fn append_intent(mut path: String, intent: bool) -> String {
    if intent && path != "/" && path != "." && !path.ends_with('/') {
        path.push('/');
    }
    path
}

fn parts(path: &str) -> Vec<&str> {
    path.split('/').filter(|s| !s.is_empty()).collect()
}

fn reduce_component(stack: &mut Vec<String>, component: &str, absolute: bool, at: usize) -> Result<(), PathError> {
    match component {
        "." => Ok(()),
        ".." => {
            if stack.last().map_or(false, |s| s != "..") {
                stack.pop();
                Ok(())
            } else if absolute {
                Err(PathError::Escape { at })
            } else {
                stack.push("..".to_string());
                Ok(())
            }
        }
        _ => {
            stack.push(component.to_string());
            Ok(())
        }
    }
}

fn normalize_components(path: &str) -> Result<Vec<String>, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let mut stack = Vec::new();
    let mut cursor = 0usize;
    for component in path.split('/') {
        let start = cursor;
        cursor += component.len() + 1;
        if component.is_empty() {
            continue;
        }
        reduce_component(&mut stack, component, absolute, start)?;
    }
    Ok(stack)
}

fn format_path(stack: &[String], absolute: bool, intent: bool) -> String {
    let joined = stack.join("/");
    let base = if absolute {
        if joined.is_empty() { "/".to_string() } else { format!("/{joined}") }
    } else if joined.is_empty() {
        ".".to_string()
    } else {
        joined
    };
    append_intent(base, intent)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let stack = normalize_components(path)?;
    Ok(format_path(&stack, absolute, directory_intent(path)))
}

fn validated_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    normalize_components(path).map_err(|_| PathError::InvalidCwd)
}

fn validated_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    match home {
        None => Ok(None),
        Some(path) => {
            if !path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            normalize_components(path)
                .map(Some)
                .map_err(|_| PathError::InvalidHome)
        }
    }
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, (usize, String)>, PathError> {
    let mut map = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let path_ok = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize(&link.path).map_or(false, |n| n == link.path);
        if !path_ok || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((first, _)) = map.get(&link.path) {
            return Err(PathError::DuplicateLink { first: *first, second: index });
        }
        map.insert(link.path.clone(), (index, link.target.clone()));
    }
    Ok(map)
}

#[derive(Clone)]
struct Frame<'a> {
    raw: &'a str,
    pos: usize,
    request_offsets: bool,
    resume: bool,
    link_index: Option<usize>,
}

fn next_component(frame: &mut Frame<'_>) -> Option<(String, usize)> {
    let bytes = frame.raw.as_bytes();
    while frame.pos < bytes.len() && bytes[frame.pos] == b'/' {
        frame.pos += 1;
    }
    if frame.pos >= bytes.len() {
        return None;
    }
    let start = frame.pos;
    while frame.pos < bytes.len() && bytes[frame.pos] != b'/' {
        frame.pos += 1;
    }
    let end = frame.pos;
    let offset = if frame.request_offsets { start } else { start };
    Some((frame.raw[start..end].to_string(), offset))
}

fn absolute_key(stack: &[String]) -> String {
    if stack.is_empty() { "/".to_string() } else { format!("/{}", stack.join("/")) }
}

fn traverse(
    initial: Vec<String>,
    request: &str,
    start_pos: usize,
    links: &HashMap<String, (usize, String)>,
    max_expansions: usize,
) -> Result<Vec<String>, PathError> {
    let mut stack = Vec::<String>::new();
    let mut frames: Vec<Frame<'_>> = Vec::new();
    // Context components are traversed through the same push-and-expand path, with offset zero.
    for component in initial {
        stack.push(component);
        let key = absolute_key(&stack);
        if let Some((index, _)) = links.get(&key) {
            if index == &usize::MAX {
                unreachable!();
            }
        }
    }
    frames.push(Frame {
        raw: request,
        pos: start_pos,
        request_offsets: true,
        resume: false,
        link_index: None,
    });
    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while let Some(frame) = frames.last_mut() {
        match next_component(frame) {
            Some((component, at)) => {
                match component.as_str() {
                    "." => {}
                    ".." => {
                        if stack.pop().is_none() {
                            return Err(PathError::Escape { at });
                        }
                    }
                    _ => {
                        stack.push(component);
                        let key = absolute_key(&stack);
                        if let Some((index, target)) = links.get(&key) {
                            if active.contains(index) {
                                return Err(PathError::Loop { index: *index });
                            }
                            if expansions >= max_expansions {
                                return Err(PathError::Limit { index: *index });
                            }
                            expansions += 1;
                            active.push(*index);
                            frames.push(Frame {
                                raw: target,
                                pos: 0,
                                request_offsets: false,
                                resume: true,
                                link_index: Some(*index),
                            });
                        }
                    }
                }
            }
            None => {
                let finished = frames.pop().unwrap();
                if let Some(index) = finished.link_index {
                    if active.last() == Some(&index) {
                        active.pop();
                    } else if let Some(pos) = active.iter().rposition(|&x| x == index) {
                        active.remove(pos);
                    }
                }
            }
        }
    }
    Ok(stack)
}

fn expand_request<'a>(path: &'a str, home: Option<&[String]>) -> Result<(bool, usize, Option<Vec<String>>), PathError> {
    let absolute = path.starts_with('/');
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let home = home.ok_or(PathError::MissingHome)?;
            let pos = if bytes.len() > 1 { 2 } else { 1 };
            return Ok((false, pos, Some(home.to_vec())));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    Ok((absolute, 0, None))
}

fn resolve_internal(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validated_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = validated_home(home)?;
    let link_map = validate_links(links)?;
    validate_raw(path)?;
    let (absolute, start_pos, tilde_home) = expand_request(path, home_parts.as_deref())?;

    let initial = if let Some(h) = tilde_home {
        h
    } else if absolute {
        Vec::new()
    } else {
        cwd_parts
    };

    let stack = traverse(initial, path, start_pos, &link_map, max_expansions)?;
    Ok(format_path(&stack, true, directory_intent(path)))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    resolve_internal(path, cwd, home, &[], usize::MAX)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_internal(path, cwd, home, links, max_expansions)
}

fn components_of_absolute(path: &str) -> Vec<String> {
    parts(path).into_iter().map(str::to_string).collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let mut common = 0usize;
    while common < base.len() && common < target.len() && base[common] == target[common] {
        common += 1;
    }
    let mut out = Vec::new();
    for _ in common..base.len() {
        out.push("..".to_string());
    }
    out.extend(target[common..].iter().cloned());
    if out.is_empty() { ".".to_string() } else { out.join("/") }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let b = components_of_absolute(&resolved_base);
    let t = components_of_absolute(&resolved_target);
    let target_dir = resolved_target.ends_with('/');
    let mut candidates = vec![route(&b, &t)];

    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else { continue };
        let a = components_of_absolute(&probe);
        if a.len() <= t.len() && a.iter().zip(&t).all(|(x, y)| x == y) {
            let mut candidate = route(&b, &parts(&link.path).into_iter().map(str::to_string).collect::<Vec<_>>());
            let suffix: Vec<String> = t[a.len()..].to_vec();
            if !suffix.is_empty() {
                if candidate == "." {
                    candidate = suffix.join("/");
                } else {
                    candidate.push('/');
                    candidate.push_str(&suffix.join("/"));
                }
            }
            candidates.push(candidate);
        }
    }

    let mut valid: Vec<String> = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        candidate = append_intent(candidate, target_dir);
        let Ok(check) = resolve_links(&candidate, &resolved_base, home, links, max_expansions) else {
            continue;
        };
        if components_of_absolute(&check) == t && !valid.contains(&candidate) {
            valid.push(candidate);
        }
    }
    valid
        .into_iter()

use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate(s: &str) -> Result<(), PathError> {
    if s.is_empty() { return Err(PathError::Empty); }
    for (i, b) in s.bytes().enumerate() { if b == 0 { return Err(PathError::Nul { at: i }); } }
    for (i, b) in s.bytes().enumerate() { if b == b'\\' { return Err(PathError::Backslash { at: i }); } }
    for (i, b) in s.bytes().enumerate() { if b < 32 || b == 127 { return Err(PathError::Control { at: i }); } }
    Ok(())
}

fn intent(s: &str) -> bool {
    s.ends_with('/') || s.rsplit('/').find(|x| !x.is_empty()).map_or(false, |x| x == "." || x == "..")
}
fn push(stack: &mut Vec<String>, c: &str, absolute: bool, at: usize) -> Result<(), PathError> {
    if c == "." { return Ok(()); }
    if c == ".." {
        if stack.last().map_or(false, |x| x != "..") { stack.pop(); }
        else if absolute { return Err(PathError::Escape { at }); }
        else { stack.push("..".into()); }
    } else { stack.push(c.into()); }
    Ok(())
}
fn join(stack: &[String], abs: bool, dir: bool) -> String {
    let s = stack.join("/");
    let mut out = if abs { if s.is_empty() { "/".into() } else { format!("/{s}") } }
        else if s.is_empty() { ".".into() } else { s };
    if dir && out != "/" && out != "." && !out.ends_with('/') { out.push('/'); }
    out
}
fn normalize_parts(s: &str) -> Result<Vec<String>, PathError> {
    validate(s)?;
    let abs = s.starts_with('/');
    let mut v = Vec::new();
    let mut pos = 0;
    for c in s.split('/') {
        let at = pos;
        pos += c.len() + 1;
        if !c.is_empty() { push(&mut v, c, abs, at)?; }
    }
    Ok(v)
}
pub fn normalize(path: &str) -> Result<String, PathError> {
    let v = normalize_parts(path)?;
    Ok(join(&v, path.starts_with('/'), intent(path)))
}
fn context(s: &str, e: PathError) -> Result<Vec<String>, PathError> {
    if !s.starts_with('/') { return Err(e); }
    normalize_parts(s).map_err(|_| e)
}
fn raw_links(ls: &[Link]) -> Result<HashMap<String, (usize, String)>, PathError> {
    let mut m = HashMap::new();
    for (i, l) in ls.iter().enumerate() {
        if !l.path.starts_with('/') || l.path.ends_with('/') || normalize(&l.path).as_deref() != Ok(l.path.as_str()) || validate(&l.target).is_err() {
            return Err(PathError::InvalidLink { index: i });
        }
        if let Some((first, _)) = m.get(&l.path) { return Err(PathError::DuplicateLink { first: *first, second: i }); }
        m.insert(l.path.clone(), (i, l.target.clone()));
    }
    Ok(m)
}

#[derive(Clone)]
struct F<'a> { s: &'a str, p: usize, done: Option<usize> }
fn next(f: &mut F<'_>) -> Option<(&str, usize)> {
    while f.p < f.s.len() && f.s.as_bytes()[f.p] == b'/' { f.p += 1; }
    if f.p == f.s.len() { return None; }
    let st = f.p;
    while f.p < f.s.len() && f.s.as_bytes()[f.p] != b'/' { f.p += 1; }
    Some((&f.s[st..f.p], st))
}
fn abs(stack: &[String]) -> String { if stack.is_empty() { "/".into() } else { format!("/{}", stack.join("/")) } }

fn walk(initial: Vec<String>, req: &str, start: usize, m: &HashMap<String, (usize, String)>, max: usize) -> Result<Vec<String>, PathError> {
    enum Event<'a> { Scan(F<'a>), Exit(usize) }
    let mut events = Vec::<Event<'_>>::new();
    let mut stack = Vec::<String>::new();
    let mut active = Vec::<usize>::new();
    let mut count = 0;
    let mut push_scan = |s: &str, p: usize, done: Option<usize>, events: &mut Vec<Event<'_>>| {
        events.push(Event::Scan(F { s, p, done }));
    };
    // Initial absolute context is walked from root, allowing links on its components.
    let initial_owned = initial;
    let _ = &mut push_scan;
    let mut pending = vec![req];
    // Context components are seeded in order and link-expanded by a synthetic absolute path.
    // Each context prefix is scanned before the request.
    let context_string = if initial_owned.is_empty() { "/".to_string() } else { format!("/{}", initial_owned.join("/")) };
    pending.push(Box::leak(context_string.into_boxed_str()));
    let context_ref = pending.pop().unwrap();
    push_scan(context_ref, 0, None, &mut events);
    push_scan(req, start, None, &mut events);

    while let Some(event) = events.pop() {
        match event {
            Event::Exit(i) => { if active.last() == Some(&i) { active.pop(); } }
            Event::Scan(mut f) => {
                if let Some((c, at)) = next(&mut f) {
                    events.push(Event::Scan(f));
                    match c {
                        "." => {}
                        ".." => {
                            if stack.pop().is_none() { return Err(PathError::Escape { at }); }
                        }
                        _ => {
                            stack.push(c.to_string());
                            if let Some((i, t)) = m.get(&abs(&stack)) {
                                if active.contains(i) { return Err(PathError::Loop { index: *i }); }
                                if count >= max { return Err(PathError::Limit { index: *i }); }
                                count += 1;
                                active.push(*i);
                                events.push(Event::Exit(*i));
                                let leaked: &'static str = Box::leak(t.clone().into_boxed_str());
                                events.push(Event::Scan(F { s: leaked, p: 0, done: None }));
                            }
                        }
                    }
                }
            }
        }
    }
    Ok(stack)
}

fn resolve_impl(path: &str, cwd: &str, home: Option<&str>, links: &[Link], max: usize) -> Result<String, PathError> {
    let c = context(cwd, PathError::InvalidCwd)?;
    let h = match home { None => None, Some(s) => Some(context(s, PathError::InvalidHome)?) };
    let m = raw_links(links)?;
    validate(path)?;
    let mut start = 0;
    let initial;
    if path.starts_with('~') {
        if path.len() > 1 && path.as_bytes()[1] != b'/' { return Err(PathError::UnsupportedTilde { at: 1 }); }
        initial = h.ok_or(PathError::MissingHome)?;
        start = if path.len() == 1 { 1 } else { 2 };
    } else if path.starts_with('/') { initial = Vec::new(); }
    else { initial = c; }
    let v = walk(initial, path, start, &m, max)?;
    Ok(join(&v, true, intent(path)))
}
pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> { resolve_impl(path, cwd, home, &[], usize::MAX) }
pub fn resolve_links(path: &str, cwd: &str, home: Option<&str>, links: &[Link], max_expansions: usize) -> Result<String, PathError> {
    resolve_impl(path, cwd, home, links, max_expansions)
}
fn comps(s: &str) -> Vec<String> { s.split('/').filter(|x| !x.is_empty()).map(str::to_string).collect() }
fn route(b: &[String], t: &[String]) -> String {
    let mut i = 0;
    while i < b.len() && i < t.len() && b[i] == t[i] { i += 1; }
    let mut v = vec!["..".to_string(); b.len() - i];
    v.extend_from_slice(&t[i..]);
    if v.is_empty() { ".".into() } else { v.join("/") }
}
pub fn relative_to(base: &str, target: &str, cwd: &str, home: Option<&str>, links: &[Link], max_expansions: usize) -> Result<String, PathError> {
    let rb = resolve_links(base, cwd, home, links, max_expansions)?;
    let rt = resolve_links(target, cwd, home, links, max_expansions)?;
    let b = comps(&rb); let t = comps(&rt);
    let dir = rt.ends_with('/');
    let mut cs = vec![route(&b, &t)];
    for l in links {
        if let Ok(p) = resolve_links(&l.path, "/", None, links, max_expansions) {
            let a = comps(&p);
            if a.len() <= t.len() && a.iter().zip(&t).all(|(x,y)| x == y) {
                let key = comps(&l.path);
                let mut c = route(&b, &key);
                let suffix = &t[a.len()..];
                if !suffix.is_empty() { if c == "." { c = suffix.join("/"); } else { c.push('/'); c.push_str(&suffix.join("/")); } }
                cs.push(c);
            }
        }
    }
    let mut valid = Vec::new();
    for mut c in cs {
        if c.starts_with('~') { c = format!("./{c}"); }
        if dir && c != "." && !c.ends_with('/') { c.push('/'); }
        if let Ok(r) = resolve_links(&c, &rb, home, links, max_expansions) {
            if comps(&r) == t && !valid.contains(&c) { valid.push(c); }
        }
    }
    valid.into_iter().min_by(|x,y| x.len().cmp(&y.len()).then(x.as_bytes().cmp(y.as_bytes()))).ok_or(PathError::Escape { at: 0 })
}
