use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn raw_error(s: &str) -> Option<PathError> {
    if s.is_empty() {
        return Some(PathError::Empty);
    }
    if let Some(at) = s.as_bytes().iter().position(|&b| b == 0) {
        return Some(PathError::Nul { at });
    }
    if let Some(at) = s.as_bytes().iter().position(|&b| b == b'\\') {
        return Some(PathError::Backslash { at });
    }
    if let Some(at) = s.as_bytes().iter().position(|&b| (1..=31).contains(&b) || b == 127) {
        return Some(PathError::Control { at });
    }
    None
}

fn comps(s: &str) -> Vec<String> {
    s.split('/').filter(|x| !x.is_empty()).map(str::to_owned).collect()
}

fn canonical_abs(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn format_path(stack: &[String], absolute: bool, dir: bool) -> String {
    let mut out = if absolute {
        canonical_abs(stack)
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };
    if dir && out != "/" && out != "." {
        out.push('/');
    }
    out
}

fn directory_intent(s: &str, cs: &[String]) -> bool {
    s.ends_with('/') || cs.last().is_some_and(|x| x == "." || x == "..")
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if let Some(e) = raw_error(path) {
        return Err(e);
    }
    let absolute = path.as_bytes()[0] == b'/';
    let cs = comps(path);
    let dir = directory_intent(path, &cs);
    let mut stack: Vec<String> = Vec::new();
    for c in cs {
        match c.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|x| x != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape {
                        at: path.find("..").unwrap_or(0),
                    });
                } else {
                    stack.push(c);
                }
            }
            _ => stack.push(c),
        }
    }
    Ok(format_path(&stack, absolute, dir))
}

fn validate_context(s: &str, err: PathError) -> Result<Vec<String>, PathError> {
    if !s.starts_with('/') {
        return Err(err);
    }
    match normalize(s) {
        Ok(n) => Ok(comps(&n)),
        Err(_) => Err(err),
    }
}

fn prepare_request<'a>(
    path: &'a str,
    home: Option<&[String]>,
) -> Result<(bool, Vec<(String, usize)>), PathError> {
    if let Some(e) = raw_error(path) {
        return Err(e);
    }
    let bytes = path.as_bytes();
    if bytes[0] == b'~' {
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            let h = home.ok_or(PathError::MissingHome)?;
            let start = if bytes.len() == 1 { 1 } else { 2 };
            let mut v = Vec::new();
            for c in comps(&path[start..]) {
                let at = path[start..].find(&c).map(|x| x + start).unwrap_or(start);
                v.push((c, at));
            }
            return Ok((false, v));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    let absolute = bytes[0] == b'/';
    let cs = comps(path);
    let mut v = Vec::new();
    let mut from = 0usize;
    for c in cs {
        let at = path[from..].find(&c).map(|x| x + from).unwrap_or(from);
        from = at + c.len();
        v.push((c, at));
    }
    Ok((absolute, v))
}

fn reduce_simple(stack: &mut Vec<String>, c: &str, at: usize, root: bool) -> Result<(), PathError> {
    match c {
        "." => Ok(()),
        ".." => {
            if stack.last().is_some_and(|x| x != "..") {
                stack.pop();
                Ok(())
            } else if root {
                Err(PathError::Escape { at })
            } else {
                stack.push("..".to_owned());
                Ok(())
            }
        }
        _ => {
            stack.push(c.to_owned());
            Ok(())
        }
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwdc = validate_context(cwd, PathError::InvalidCwd)?;
    let homec = match home {
        Some(h) => Some(validate_context(h, PathError::InvalidHome)?),
        None => None,
    };
    let (absolute, request) = prepare_request(path, homec.as_deref())?;
    let mut stack = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        homec.clone().ok_or(PathError::MissingHome)?
    } else {
        cwdc
    };
    for (c, at) in request {
        reduce_simple(&mut stack, &c, at, true)?;
    }
    let rawcs = comps(path);
    let dir = directory_intent(path, &rawcs);
    Ok(format_path(&stack, true, dir))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut map = HashMap::new();
    for (i, l) in links.iter().enumerate() {
        let valid_path = l.path.starts_with('/')
            && !l.path.ends_with('/')
            && l.path != "/"
            && normalize(&l.path).is_ok_and(|n| n == l.path);
        let valid_target = raw_error(&l.target).is_none();
        if !valid_path || !valid_target {
            return Err(PathError::InvalidLink { index: i });
        }
        if let Some(&first) = map.get(&l.path) {
            return Err(PathError::DuplicateLink { first, second: i });
        }
        map.insert(l.path.clone(), i);
    }
    Ok(map)
}

#[derive(Clone)]
struct Item {
    c: String,
    at: usize,
}

fn traverse(
    stack: &mut Vec<String>,
    items: &[Item],
    links: &[Link],
    map: &HashMap<String, usize>,
    active: &mut Vec<usize>,
    count: &mut usize,
    max: usize,
    root: bool,
) -> Result<(), PathError> {
    for item in items {
        match item.c.as_str() {
            "." => continue,
            ".." => {
                if stack.last().is_some_and(|x| x != "..") {
                    stack.pop();
                } else if root {
                    return Err(PathError::Escape { at: item.at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            _ => {
                stack.push(item.c.clone());
                let key = canonical_abs(stack);
                if let Some(&idx) = map.get(&key) {
                    if active.contains(&idx) {
                        return Err(PathError::Loop { index: idx });
                    }
                    if *count >= max {
                        return Err(PathError::Limit { index: idx });
                    }
                    *count += 1;
                    active.push(idx);
                    let link = &links[idx];
                    let target_abs = link.target.starts_with('/');
                    if target_abs {
                        stack.clear();
                    } else {
                        stack.pop();
                    }
                    let target_items: Vec<Item> = {
                        let mut out = Vec::new();
                        let mut offset = 0usize;
                        for c in comps(&link.target) {
                            let at = link.target[offset..].find(&c).map(|x| x + offset).unwrap_or(offset);
                            offset = at + c.len();
                            out.push(Item { c, at });
                        }
                        out
                    };
                    traverse(stack, &target_items, links, map, active, count, max, true)?;
                    active.pop();
                }
            }
        }
    }
    Ok(())
}

fn items_from_request(path: &str, pairs: Vec<(String, usize)>) -> Vec<Item> {
    let _ = path;
    pairs.into_iter().map(|(c, at)| Item { c, at }).collect()
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwdc = validate_context(cwd, PathError::InvalidCwd)?;
    let homec = match home {
        Some(h) => Some(validate_context(h, PathError::InvalidHome)?),
        None => None,
    };
    let map = validate_links(links)?;
    let (absolute, pairs) = prepare_request(path, homec.as_deref())?;
    let mut stack = if absolute {
        Vec::new()
    } else if path.starts_with('~') {
        homec.clone().ok_or(PathError::MissingHome)?
    } else {
        cwdc
    };
    let req = items_from_request(path, pairs);
    let mut active = Vec::new();
    let mut count = 0;
    traverse(&mut stack, &req, links, &map, &mut active, &mut count, max_expansions, true)?;
    Ok(format_path(&stack, true, directory_intent(path, &comps(path))))
}

fn has_dir_suffix(s: &str) -> bool {
    s.ends_with('/')
}

fn components_equal(a: &str, b: &str) -> bool {
    comps(a) == comps(b)
}

fn route(base: &[String], target: &[String]) -> String {
    let mut i = 0;
    while i < base.len() && i < target.len() && base[i] == target[i] {
        i += 1;
    }
    let mut out = Vec::new();
    for _ in i..base.len() {
        out.push("..".to_owned());
    }
    out.extend(target[i..].iter().cloned());
    if out.is_empty() { ".".to_owned() } else { out.join("/") }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let b = resolve_links(base, cwd, home, links, max_expansions)?;
    let t = resolve_links(target, cwd, home, links, max_expansions)?;
    let bc = comps(&b);
    let tc = comps(&t);
    let target_dir = has_dir_suffix(&t);
    let mut candidates = vec![route(&bc, &tc)];
    for l in links {
        let a = match resolve_links(&l.path, "/", None, links, max_expansions) {
            Ok(x) => comps(&x),
            Err(_) => continue,
        };
        if a.len() <= tc.len() && a.iter().zip(&tc).all(|(x, y)| x == y) {
            let mut c = route(&bc, &comps(&l.path));
            let suffix = &tc[a.len()..];
            if !suffix.is_empty() {
                if c == "." { c = suffix.join("/"); }
                else { c.push('/'); c.push_str(&suffix.join("/")); }
            }
            candidates.push(c);
        }
    }
    let mut valid: Vec<String> = Vec::new();
    for mut c in candidates {
        if c.starts_with('~') {
            c.insert_str(0, "./");
        }
        if target_dir && c != "." && !c.ends_with('/') {
            c.push('/');
        }
        if let Ok(r) = resolve_links(&c, &b, home, links, max_expansions) {
            if components_equal(&r, &t) {
                valid.push(c);
            }
        }
    }
    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate(s: &str) -> Result<(), PathError> {
    if s.is_empty() { return Err(PathError::Empty); }
    for (i, b) in s.bytes().enumerate() { if b == 0 { return Err(PathError::Nul { at: i }); } }
    for (i, b) in s.bytes().enumerate() { if b == b'\\' { return Err(PathError::Backslash { at: i }); } }
    for (i, b) in s.bytes().enumerate() { if (1..=31).contains(&b) || b == 127 { return Err(PathError::Control { at: i }); } }
    Ok(())
}

fn split(s: &str) -> Vec<(String, usize)> {
    let mut v = Vec::new();
    let mut i = 0;
    while i < s.len() {
        while i < s.len() && s.as_bytes()[i] == b'/' { i += 1; }
        let st = i;
        while i < s.len() && s.as_bytes()[i] != b'/' { i += 1; }
        if st < i { v.push((s[st..i].to_owned(), st)); }
    }
    v
}

fn out(stack: &[String], trailing: bool) -> String {
    let mut s = if stack.is_empty() { "/".to_owned() } else { format!("/{}", stack.join("/")) };
    if trailing && s != "/" { s.push('/'); }
    s
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate(path)?;
    let abs = path.starts_with('/');
    let parts = split(path);
    let trailing = path.ends_with('/') || parts.last().is_some_and(|x| x.0 == "." || x.0 == "..");
    let mut st = Vec::new();
    for (c, at) in parts {
        if c == "." { continue; }
        if c == ".." {
            if st.last().is_some_and(|x| x != "..") { st.pop(); }
            else if abs { return Err(PathError::Escape { at }); }
            else { st.push(c); }
        } else { st.push(c); }
    }
    if abs { Ok(out(&st, trailing)) }
    else {
        let mut s = if st.is_empty() { ".".to_owned() } else { st.join("/") };
        if trailing && s != "." { s.push('/'); }
        Ok(s)
    }
}

fn context(s: &str, e: PathError) -> Result<Vec<String>, PathError> {
    if !s.starts_with('/') { return Err(e); }
    normalize(s).map(|x| split(&x).into_iter().map(|p| p.0).collect()).map_err(|_| e)
}

fn request_parts(path: &str, home: Option<&[String]>) -> Result<(Vec<String>, Vec<(String, usize)>, bool), PathError> {
    validate(path)?;
    if path.starts_with('~') {
        let b = path.as_bytes();
        if b.len() > 1 && b[1] != b'/' { return Err(PathError::UnsupportedTilde { at: 1 }); }
        let h = home.ok_or(PathError::MissingHome)?;
        let start = if b.len() == 1 { 1 } else { 2 };
        return Ok((h.to_vec(), split(&path[start..]), true));
    }
    Ok((Vec::new(), split(path), path.starts_with('/')))
}

fn step(st: &mut Vec<String>, c: &str, at: usize) -> Result<bool, PathError> {
    if c == "." { return Ok(false); }
    if c == ".." {
        if st.pop().is_some() { return Ok(false); }
        return Err(PathError::Escape { at });
    }
    st.push(c.to_owned());
    Ok(true)
}

fn link_map(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut m = HashMap::new();
    for (i, l) in links.iter().enumerate() {
        if !l.path.starts_with('/') || l.path == "/" || l.path.ends_with('/') || normalize(&l.path).as_deref() != Ok(l.path.as_str()) || validate(&l.target).is_err() {
            return Err(PathError::InvalidLink { index: i });
        }
        if let Some(&first) = m.get(&l.path) { return Err(PathError::DuplicateLink { first, second: i }); }
        m.insert(l.path.clone(), i);
    }
    Ok(m)
}

fn expand(
    st: &mut Vec<String>, input: &[(String, usize)], links: &[Link], map: &HashMap<String, usize>,
    active: &mut Vec<usize>, used: &mut usize, limit: usize
) -> Result<(), PathError> {
    for (c, at) in input {
        if c == "." { continue; }
        if c == ".." {
            if st.pop().is_none() { return Err(PathError::Escape { at: *at }); }
            continue;
        }
        st.push(c.clone());
        let key = if st.is_empty() { "/".to_owned() } else { format!("/{}", st.join("/")) };
        if let Some(&i) = map.get(&key) {
            if active.contains(&i) { return Err(PathError::Loop { index: i }); }
            if *used >= limit { return Err(PathError::Limit { index: i }); }
            *used += 1;
            active.push(i);
            if links[i].target.starts_with('/') { st.clear(); } else { st.pop(); }
            let t = split(&links[i].target);
            expand(st, &t, links, map, active, used, limit)?;
            active.pop();
        }
    }
    Ok(())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwdc = context(cwd, PathError::InvalidCwd)?;
    let hc = match home { Some(h) => Some(context(h, PathError::InvalidHome)?), None => None };
    let (mut st, p, abs) = request_parts(path, hc.as_deref())?;
    if abs { st.clear(); } else if !path.starts_with('~') { st = cwdc; }
    for (c, at) in p {
        step(&mut st, &c, at)?;
    }
    let parts = split(path);
    let dir = path.ends_with('/') || parts.last().is_some_and(|x| x.0 == "." || x.0 == "..");
    Ok(out(&st, dir))
}

pub fn resolve_links(path: &str, cwd: &str, home: Option<&str>, links: &[Link], max_expansions: usize) -> Result<String, PathError> {
    let cwdc = context(cwd, PathError::InvalidCwd)?;
    let hc = match home { Some(h) => Some(context(h, PathError::InvalidHome)?), None => None };
    let map = link_map(links)?;
    let (mut st, p, abs) = request_parts(path, hc.as_deref())?;
    if abs { st.clear(); } else if !path.starts_with('~') { st = cwdc; }
    let mut active = Vec::new();
    let mut used = 0;
    expand(&mut st, &p, links, &map, &mut active, &mut used, max_expansions)?;
    let parts = split(path);
    let dir = path.ends_with('/') || parts.last().is_some_and(|x| x.0 == "." || x.0 == "..");
    Ok(out(&st, dir))
}

fn route(b: &[String], t: &[String]) -> String {
    let mut i = 0;
    while i < b.len() && i < t.len() && b[i] == t[i] { i += 1; }
    let mut v = vec!["..".to_owned(); b.len() - i];
    v.extend_from_slice(&t[i..]);
    if v.is_empty() { ".".to_owned() } else { v.join("/") }
}

pub fn relative_to(base: &str, target: &str, cwd: &str, home: Option<&str>, links: &[Link], max_expansions: usize) -> Result<String, PathError> {
    let b = resolve_links(base, cwd, home, links, max_expansions)?;
    let t = resolve_links(target, cwd, home, links, max_expansions)?;
    let bc = split(&b).into_iter().map(|x| x.0).collect::<Vec<_>>();
    let tc = split(&t).into_iter().map(|x| x.0).collect::<Vec<_>>();
    let mut candidates = vec![route(&bc, &tc)];
    for l in links {
        let a = match resolve_links(&l.path, "/", None, links, max_expansions) { Ok(x) => split(&x).into_iter().map(|p| p.0).collect::<Vec<_>>(), Err(_) => continue };
        if a.len() <= tc.len() && a.iter().zip(&tc).all(|(x,y)| x == y) {
            let mut c = route(&bc, &split(&l.path).into_iter().map(|p| p.0).collect::<Vec<_>>());
            let suffix = &tc[a.len()..];
            if !suffix.is_empty() {
                if c == "." { c = suffix.join("/"); } else { c.push('/'); c.push_str(&suffix.join("/")); }
            }
            candidates.push(c);
        }
    }
    let dir = t.ends_with('/');
    let mut valid = Vec::new();
    for mut c in candidates {
        if c.starts_with('~') { c.insert_str(0, "./"); }
        if dir && c != "." && !c.ends_with('/') { c.push('/'); }
        if let Ok(r) = resolve_links(&c, &b, home, links, max_expansions) {
            if split(&r).into_iter().map(|x| x.0).collect::<Vec<_>>() == tc { valid.push(c); }
        }
    }
    valid.sort_by(|x,y| x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}
