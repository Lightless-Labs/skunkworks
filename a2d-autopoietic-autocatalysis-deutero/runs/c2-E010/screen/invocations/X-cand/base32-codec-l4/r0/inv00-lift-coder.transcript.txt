fn alphabet_value(alphabet: Alphabet, byte: u8) -> Option<u8> {

        Alphabet::Standard => match byte {

            _ => None,

        Alphabet::Hex => match byte {
            b'0'..=b'9' => Some(byte - b'0'),

    }

        0 => Some(0),

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config, position: 0, data_count: 0, bit_buffer: 0, bit_count: 0,
            last_value: 0, last_offset: 0, saw_padding: false, first_padding: None,
            padding_count: 0, expected_padding: 0, closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed { return Err(StreamError::Closed); }
        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            let value = if byte == b'=' {
                None
            } else {
                let upper = byte.to_ascii_uppercase();
                let Some(v) = alphabet_value(next.config.alphabet, upper) else {
                    return Err(StreamError::InvalidByte { offset, byte });
                };
                if next.config.canonical && byte != upper {
                    return Err(StreamError::NonCanonical { offset, expected: upper });
                }
                Some(v)
            };

            if byte == b'=' {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }
                if !next.saw_padding {
                    let residue = next.data_count % 8;
                    let Some(expected) = residue_padding(residue) else {
                        return Err(StreamError::Padding { offset });
                    };
                    if expected == 0 {
                        return Err(StreamError::Padding { offset });
                    }
                    next.saw_padding = true;
                    next.first_padding = Some(offset);
                    next.expected_padding = expected;
                }
                if next.padding_count >= next.expected_padding {
                    return Err(StreamError::Padding { offset });
                }
                next.padding_count += 1;
            } else {
                let v = value.unwrap();
                if next.saw_padding {
                    return Err(StreamError::Padding { offset });
                }
                next.data_count = next.data_count.checked_add(1)
                    .ok_or(StreamError::InvalidLength { offset })?;
                next.last_value = v;
                next.last_offset = offset;
                next.bit_buffer = (next.bit_buffer << 5) | v;
                next.bit_count += 5;
                while next.bit_count >= 8 {
                    next.bit_count -= 8;
                    output.push((next.bit_buffer >> next.bit_count) as u8);
                    next.bit_buffer &= (1u16.checked_shl(next.bit_count as u32).unwrap_or(0).wrapping_sub(1)) as u8;
                }
            }
            next.position = next.position.checked_add(1)
                .ok_or(StreamError::InvalidLength { offset })?;
        }

        self.copy_from(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed { return Err(StreamError::Closed); }
        let residue = self.data_count % 8;
        let Some(expected) = residue_padding(residue) else {
            return Err(StreamError::InvalidLength { offset: self.position });
        };
        if self.config.padding == Padding::Required && self.padding_count != expected {
            return Err(StreamError::Padding {
                offset: self.first_padding.unwrap_or(self.position),
            });
        }
        let unused = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused != 0 && (self.last_value & ((1u8 << unused) - 1)) != 0 {
            return Err(StreamError::TrailingBits { offset: self.last_offset });
        }
        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            position: self.position,
            data_count: self.data_count,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            last_value: self.last_value,
            last_offset: self.last_offset,
            saw_padding: self.saw_padding,
            first_padding: self.first_padding,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
            closed: self.closed,
        }
    }

    fn copy_from(&mut self, other: Self) {
        self.position = other.position;
        self.data_count = other.data_count;
        self.bit_buffer = other.bit_buffer;
        self.bit_count = other.bit_count;
        self.last_value = other.last_value;
        self.last_offset = other.last_offset;
        self.saw_padding = other.saw_padding;
        self.first_padding = other.first_padding;
        self.padding_count = other.padding_count;
        self.expected_padding = other.expected_padding;
        self.closed = other.closed;
    }
}

pub enum Alphabet { Standard, Hex }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding { Required, Omitted }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

pub struct Decoder {
    config: Config,
    pos: usize,
    symbols: usize,
    accumulator: u64,
    bits: u8,
    last: u8,
    last_pos: usize,
    padding_started: bool,
    pad_start: Option<usize>,
    pads: usize,
    pad_target: usize,
    closed: bool,
}

fn encode_symbol(a: Alphabet, v: u8) -> char {
    match a {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567"[v as usize] as char,
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV"[v as usize] as char,
    }
}
fn decode_symbol(a: Alphabet, b: u8) -> Option<u8> {
    match a {
        Alphabet::Standard => match b {
            b'A'..=b'Z' => Some(b - b'A'),
            b'2'..=b'7' => Some(b - b'2' + 26),
            _ => None,
        },
        Alphabet::Hex => match b {
            b'0'..=b'9' => Some(b - b'0'),
            b'A'..=b'V' => Some(b - b'A' + 10),
            _ => None,
        },
    }
}
fn pad_for(n: usize) -> Option<usize> {
    match n % 8 { 0 => Some(0), 2 => Some(6), 4 => Some(4), 5 => Some(3), 7 => Some(1), _ => None }
}

impl Encoder {
    pub fn new(config: Config) -> Self { Self { config, pending: Vec::new(), closed: false } }
    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed { return Err(StreamError::Closed); }
        let mut bytes = Vec::with_capacity(self.pending.len() + input.len());
        bytes.extend_from_slice(&self.pending);
        bytes.extend_from_slice(input);
        let full = bytes.len() / 5 * 5;
        let mut out = String::with_capacity(full / 5 * 8);
        let mut acc = 0u64;
        let mut bit_len = 0usize;
        for &b in &bytes[..full] {
            acc = (acc << 8) | b as u64;
            bit_len += 8;
            while bit_len >= 5 {
                bit_len -= 5;
                out.push(encode_symbol(self.config.alphabet, ((acc >> bit_len) & 31) as u8));
            }
            if bit_len == 0 { acc = 0; } else { acc &= (1u64 << bit_len) - 1; }
        }
        self.pending.clear();
        self.pending.extend_from_slice(&bytes[full..]);
        Ok(out)
    }
    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed { return Err(StreamError::Closed); }
        let mut out = String::new();
        if !self.pending.is_empty() {
            let mut acc = 0u64;
            for &b in &self.pending { acc = (acc << 8) | b as u64; }
            let bits = self.pending.len() * 8;
            let symbols = (bits + 4) / 5;
            for i in 0..symbols {
                let shift = bits.saturating_sub((i + 1) * 5);
                let value = if (i + 1) * 5 <= bits {
                    ((acc >> shift) & 31) as u8
                } else {
                    ((acc << ((i + 1) * 5 - bits)) & 31) as u8
                };
                out.push(encode_symbol(self.config.alphabet, value));
            }
            if self.config.padding == Padding::Required {
                while out.len() % 8 != 0 { out.push('='); }
            }
        }
        self.pending.clear();
        self.closed = true;
        Ok(out)
    }
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self { config, pos: 0, symbols: 0, accumulator: 0, bits: 0, last: 0, last_pos: 0, padding_started: false, pad_start: None, pads: 0, pad_target: 0, closed: false }
    }
    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed { return Err(StreamError::Closed); }
        let mut t = Self {
            config: self.config, pos: self.pos, symbols: self.symbols,
            accumulator: self.accumulator, bits: self.bits, last: self.last,
            last_pos: self.last_pos, padding_started: self.padding_started,
            pad_start: self.pad_start, pads: self.pads, pad_target: self.pad_target,
            closed: self.closed,
        };
        let mut out = Vec::new();
        for &b in input.as_bytes() {
            let at = t.pos;
            let value = if b == b'=' { None } else {
                let upper = b.to_ascii_uppercase();
                let Some(v) = decode_symbol(t.config.alphabet, upper) else {
                    return Err(StreamError::InvalidByte { offset: at, byte: b });
                };

                    return Err(StreamError::NonCanonical { offset: at, expected: upper });
                }
