pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::with_capacity(customers.len().saturating_mul(3));
    let mut waiting = VecDeque::new();
    let mut next_arrival = 0;
    let mut now = 0;
    let mut active: Option<(usize, u64)> = None;
    let mut previous_finish: Option<u64> = None;
    let mut finished_count = 0;

    while finished_count < customers.len() {
        let next_arrival_time = order
            .get(next_arrival)
            .map(|&i| customers[i].arrival);
        let next_finish_time = active.map(|(_, finish)| finish);

        let tick = match (next_arrival_time, next_finish_time) {
            (Some(a), Some(f)) => a.min(f),
            (Some(a), None) => a,
            (None, Some(f)) => f,
            (None, None) => break,
        };
        now = tick;

        if let Some((idx, finish)) = active {
            if finish == now {
                events.push(Event {
                    time: now,
                    customer: customers[idx].id,
                    kind: Kind::Finish,
                });
                active = None;
                previous_finish = Some(now);
                finished_count += 1;
            }
        }

        while next_arrival < order.len()
            && customers[order[next_arrival]].arrival == now
        {
            let idx = order[next_arrival];
            events.push(Event {
                time: now,
                customer: customers[idx].id,
                kind: Kind::Arrive,
            });
            waiting.push_back(idx);
            next_arrival += 1;
        }

        if active.is_none() {
            if let Some(idx) = waiting.pop_front() {
                let customer = &customers[idx];
                let wake = match previous_finish {
                    None => true,
                    Some(finish) => now > finish,
                };
                let base_finish = now.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: now,
                    },
                )?;
                let finish = if wake {
                    base_finish.checked_add(wake_cost).ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time: now,
                    })?
                } else {
                    base_finish
                };

                events.push(Event {
                    time: now,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });

                if finish == now {
                    events.push(Event {
                        time: now,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    previous_finish = Some(now);
                    finished_count += 1;
                } else {
                    active = Some((idx, finish));
                }
            }
        }
    }

    Ok(events)
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut result = Vec::with_capacity(customers.len().saturating_mul(3));
    let mut queue: VecDequeCompat = VecDequeCompat::new();
    let mut arrival_pos = 0;
    let mut time = 0u64;
    let mut in_service: Option<(&Customer, u64)> = None;
    let mut last_finish: Option<u64> = None;
    let mut completed = 0usize;

    while completed != customers.len() {
        let arrival_time = arrivals.get(arrival_pos).map(|c| c.arrival);
        let finish_time = in_service.map(|(_, t)| t);
        let next_time = match (arrival_time, finish_time) {
            (Some(a), Some(f)) => Some(a.min(f)),
            (Some(a), None) => Some(a),
            (None, Some(f)) => Some(f),
            (None, None) => None,
        };
        let Some(next_time) = next_time else {
            break;
        };
        time = next_time;

        if let Some((customer, finish)) = in_service {
            if finish == time {
                result.push(Event {
                    time,
                    customer: customer.id,
                    kind: Kind::Finish,
                });
                in_service = None;
                last_finish = Some(time);
                completed += 1;
            }
        }

        while arrival_pos < arrivals.len() && arrivals[arrival_pos].arrival == time {
            let customer = arrivals[arrival_pos];
            result.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            queue.push(customer);
            arrival_pos += 1;
        }

        if in_service.is_none() {
            if let Some(customer) = queue.pop() {
                let wake = last_finish.map_or(true, |finished_at| time > finished_at);
                let service_end = time.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time,
                    },
                )?;
                let finish = if wake {
                    service_end.checked_add(wake_cost).ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time,
                    })?
                } else {
                    service_end

                    time,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },

                        time,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    last_finish = Some(time);
                    completed += 1;
                } else {
                    in_service = Some((customer, finish));
                }
            }
        }
    }

    Ok(result)
}

struct VecDequeCompat {
    items: std::collections::VecDeque<&'static Customer>,
}

impl VecDequeCompat {
    fn new() -> Self {
        Self {
            items: std::collections::VecDeque::new(),
        }
    }

    fn push(&mut self, customer: &Customer) {
        let ptr = customer as *const Customer;
        let static_ref = unsafe { &*ptr };
        self.items.push_back(static_ref);
    }

    fn pop(&mut self) -> Option<&'static Customer> {
        self.items.pop_front()
    }
}
