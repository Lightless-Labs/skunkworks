        return a.priority < b.priority;

        return a.started;

                            } else if waiter_better(&jobs[b], &jobs[a]) {

            if !waiting.is_empty() {
                continue;
            }
            break;
        }

    a.priority.cmp(&b.priority).then_with(|| {

            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .arrival
                .cmp(&b.arrival)
                .then_with(|| b.id.cmp(&a.id)),
        }
    })
}

fn precedes(a: &Job, b: &Job) -> bool {
    let ap = a.customer.priority;
    let bp = b.customer.priority;
    if ap != bp {
        return ap < bp;
    }

    match (a.pause_time, b.pause_time) {
        (Some(at), Some(bt)) => at < bt || (at == bt && a.customer.id < b.customer.id),
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            a.customer.arrival < b.customer.arrival
                || (a.customer.arrival == b.customer.arrival
                    && a.customer.id > b.customer.id)
        }
    }
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut ids = std::collections::HashSet::new();
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
    }
    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut input = customers.to_vec();
    input.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then(a.priority.cmp(&b.priority))
            .then(b.id.cmp(&a.id))
    });

    let mut out = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut running: Option<(Customer, u64, u64, u64)> = None;
    let mut cursor = 0usize;
    let mut previous_dispatch = None;
    let mut time = input[0].arrival;

    loop {
        if let Some((c, _, _, finish)) = running.as_ref() {
            if *finish == time {
                out.push(Event { time, customer: c.id, kind: Kind::Finish });
                running = None;
            }
        }

        while cursor < input.len() && input[cursor].arrival == time {
            let c = input[cursor].clone();
            out.push(Event { time, customer: c.id, kind: Kind::Arrive });
            waiting.push(Job { remaining: c.service, customer: c, pause_time: None });
            cursor += 1;
        }

        if let Some((c, _, work_at, finish)) = running.as_ref() {
            if time >= *work_at {
                if let Some(i) = waiting.iter().enumerate().min_by(|(_, a), (_, b)| {
                    if precedes(a, b) { std::cmp::Ordering::Less }
                    else if precedes(b, a) { std::cmp::Ordering::Greater }
                    else { std::cmp::Ordering::Equal }
                }).map(|(i, _)| i) {
                    if waiting[i].customer.priority < c.priority {
                        let by = waiting[i].customer.id;
                        let rem = *finish - time;
                        let old = running.take().unwrap().0;
                        out.push(Event {
                            time,
                            customer: old.id,
                            kind: Kind::Preempt { remaining: rem, by },
                        });
                        waiting.push(Job {
                            customer: old,
                            remaining: rem,
                            pause_time: Some(time),
                        });
                    }
                }
            }
        }

        while running.is_none() && !waiting.is_empty() {
            let i = waiting.iter().enumerate().min_by(|(_, a), (_, b)| {
                if precedes(a, b) { std::cmp::Ordering::Less }
                else if precedes(b, a) { std::cmp::Ordering::Greater }
                else { std::cmp::Ordering::Equal }
            }).unwrap().0;
            let job = waiting.remove(i);
            let cost = if previous_dispatch.is_some() && previous_dispatch != Some(job.customer.id) {
                switch_cost
            } else {
                0
            };
            let work_at = time.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: job.customer.id,
                time,
            })?;
            let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                customer: job.customer.id,
                time,
            })?;
            let kind = if job.pause_time.is_some() {
                Kind::Resume { remaining: job.remaining, work_at, finish }
            } else {
                Kind::Start { remaining: job.remaining, work_at, finish }
            };
            out.push(Event { time, customer: job.customer.id, kind });
            previous_dispatch = Some(job.customer.id);
            running = Some((job.customer.clone(), job.remaining, work_at, finish));

            if finish == time {
                out.push(Event { time, customer: job.customer.id, kind: Kind::Finish });
                running = None;
            }
        }

        if cursor == input.len() && waiting.is_empty() && running.is_none() {
            break;
        }

        let mut next = if cursor < input.len() {
            Some(input[cursor].arrival)
        } else {
            None
        };
        if let Some((_, _, work_at, finish)) = running.as_ref() {
            for tick in [*work_at, *finish] {
                if tick > time && next.map_or(true, |n| tick < n) {
                    next = Some(tick);
                }
            }
        }
        let Some(next_time) = next else { break };
        time = next_time;
    }

    Ok(out)
}

            return Err(Error::ZeroId { index });
        }

            return Err(Error::DuplicateId { index, id: c.id });
        }
        if c.priority > 3 {

                index,
                value: c.priority,
            });
        }
    }
    if customers.is_empty() {
        return Ok(Vec::new());
    }

    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut out = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut arrival_i = 0usize;
    let mut finished_count = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut started_ids = std::collections::HashSet::new();
    let mut now: Option<u64> = None;

    while finished_count < customers.len() {
        let next_arrival = arrivals.get(arrival_i).map(|c| c.arrival);
        let next_finish = running.as_ref().map(|r| r.finish);
        let next_work = running
            .as_ref()
            .and_then(|r| (r.work_at > now.unwrap_or(0)).then_some(r.work_at));
        let t = [next_arrival, next_finish, next_work]
            .into_iter()
            .flatten()
            .min()
            .expect("unfinished customers imply a pending event");
        now = Some(t);

        if running.as_ref().is_some_and(|r| r.finish == t) {
            let r = running.take().unwrap();
            out.push(Event {
                time: t,
                customer: r.customer.id,
                kind: Kind::Finish,
            });
            finished_count += 1;
        }

        while arrival_i < arrivals.len() && arrivals[arrival_i].arrival == t {
            let c = arrivals[arrival_i].clone();
            out.push(Event {
                time: t,
                customer: c.id,
                kind: Kind::Arrive,
            });

                remaining: c.service,
                customer: c,
                started: false,
                pause_time: 0,
            });
            arrival_i += 1;

            if t >= r.work_at {
                if let Some(best_idx) = waiting
                    .iter()
                    .enumerate()
                    .min_by(|(_, a), (_, b)| waiting_cmp(a, b))
                    .map(|(i, _)| i)

                    if waiting[best_idx].customer.priority < r.customer.priority {
                        let best_id = waiting[best_idx].customer.id;
                        let r = running.take().unwrap();
                        let remaining = r.finish - t;
                        out.push(Event {

                            customer: r.customer.id,

                            customer: r.customer,
                            remaining,
                            started: true,
                            pause_time: t,
                        });
                    }
                }
            }
        }

        loop {
            if running.is_some() || waiting.is_empty() {
                break;
            }
            let best_idx = waiting
                .iter()
                .enumerate()
                .min_by(|(_, a), (_, b)| waiting_cmp(a, b))
                .map(|(i, _)| i)
                .unwrap();
            let w = waiting.remove(best_idx);
            let cost = match last_dispatched {
                None => 0,
                Some(id) if id == w.customer.id => 0,
                Some(_) => switch_cost,
            };
            let work_at = t.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: w.customer.id,
                time: t,
            })?;
            let finish = work_at.checked_add(w.remaining).ok_or(Error::TimeOverflow {
                customer: w.customer.id,
                time: t,
            })?;

            let kind = if started_ids.insert(w.customer.id) {
                Kind::Start {
                    remaining: w.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Resume {
                    remaining: w.remaining,
                    work_at,
                    finish,
                }
            };
            out.push(Event {
                time: t,
                customer: w.customer.id,
                kind,
            });
            last_dispatched = Some(w.customer.id);

            if finish == t {
                out.push(Event {
                    time: t,
                    customer: w.customer.id,
                    kind: Kind::Finish,
                });
                finished_count += 1;
            } else {
                running = Some(Running {
                    customer: w.customer,
                    remaining: w.remaining,
                    work_at,
                    finish,
                });
            }
        }

        if finished_count == customers.len() {
            break;
        }

        // If the running job has finished at this tick through an inline dispatch,
        // the dispatch loop above has already handled all immediate completions.
        // The next iteration advances to the next arrival, finish, or work boundary.
        let _ = Ordering::Equal;
    }

    Ok(out)
}

fn waiter_less(a: &Waiting, b: &Waiting) -> bool {
    if a.customer.priority != b.customer.priority {
        return a.customer.priority < b.customer.priority;
    }
    if a.started != b.started {
        return a.started;
    }
    if a.started {
        if a.pause_time != b.pause_time {
            return a.pause_time < b.pause_time;
        }
        a.customer.id < b.customer.id
    } else if a.customer.arrival != b.customer.arrival {
        a.customer.arrival < b.customer.arrival
    } else {
        a.customer.id > b.customer.id
    }
}

            best = Some(i);
        }
    }
    best

            }
        }

        if next_arrival >= arrivals.len() && running.is_none() && waiting.is_empty() {
            break;
        }

        if let Some(r) = &running {
            if r.finish > now || (r.work_at > now && r.work_at < r.finish) {
                let t = if r.work_at > now && r.work_at < r.finish {
                    r.work_at.min(r.finish)
                } else {
                    r.finish
                };
                if next_time.map_or(true, |n| t < n) {
                    next_time = Some(t);
                }
            }
        }

        if let Some(t) = next_time {
            if t > now {
                now = t;
            } else if next_arrival < arrivals.len() && arrivals[next_arrival].arrival > now {
                now = arrivals[next_arrival].arrival;
            } else if running.is_none() && waiting.is_empty() {
                break;
            } else {
                // All same-tick work is handled inline; this prevents a stalled loop.
                if let Some(r) = &running {
                    if r.finish <= now {
                        continue;
                    }
                }
                if next_arrival < arrivals.len() {
                    now = arrivals[next_arrival].arrival;
                }
            }
        } else {
            break;
        }
    }

    Ok(events)
}

use std::cmp::Ordering;
use std::collections::BTreeSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    id: u32,
    arrival: u64,
    priority: u8,
    started: bool,
    remaining: u64,
    pause_time: u64,
}

#[derive(Clone, Copy)]
struct Active {
    job: usize,
    work_at: u64,
    finish: u64,
}

fn compare_waiters(a: &Job, b: &Job) -> Ordering {
    a.priority
        .cmp(&b.priority)
        .then_with(|| b.started.cmp(&a.started))
        .then_with(|| {
            if a.started {
                a.pause_time.cmp(&b.pause_time)
                    .then_with(|| a.id.cmp(&b.id))
            } else {
                a.arrival.cmp(&b.arrival)
                    .then_with(|| b.id.cmp(&a.id))
            }
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut seen = BTreeSet::new();
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
    }
    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut jobs: Vec<Job> = customers
        .iter()
        .map(|c| Job {
            id: c.id,
            arrival: c.arrival,
            priority: c.priority,
            started: false,
            remaining: c.service,
            pause_time: 0,
        })
        .collect();
    let mut order: Vec<usize> = (0..jobs.len()).collect();
    order.sort_by(|&a, &b| {
        jobs[a]
            .arrival
            .cmp(&jobs[b].arrival)
            .then(jobs[a].priority.cmp(&jobs[b].priority))
            .then(jobs[b].id.cmp(&jobs[a].id))
    });

    let mut trace = Vec::new();
    let mut waiting = Vec::<usize>::new();
    let mut active: Option<Active> = None;
    let mut previous_id = None;
    let mut cursor = 0usize;
    let mut done = 0usize;
    let mut now = jobs[order[0]].arrival;

    while done < jobs.len() {
        if active.map_or(false, |a| a.finish == now) {
            let a = active.take().unwrap();
            trace.push(Event {
                time: now,
                customer: jobs[a.job].id,
                kind: Kind::Finish,
            });
            done += 1;
        }

        while cursor < order.len() && jobs[order[cursor]].arrival == now {
            let j = order[cursor];
            trace.push(Event {
                time: now,
                customer: jobs[j].id,
                kind: Kind::Arrive,
            });
            waiting.push(j);
            cursor += 1;
        }

        waiting.sort_by(|&a, &b| compare_waiters(&jobs[a], &jobs[b]));

        if let Some(a) = active {
            if now >= a.work_at {
                if let Some(&best) = waiting.first() {
                    if jobs[best].priority < jobs[a.job].priority {
                        let remaining = a.finish - now;
                        trace.push(Event {
                            time: now,
                            customer: jobs[a.job].id,
                            kind: Kind::Preempt {
                                remaining,
                                by: jobs[best].id,
                            },
                        });
                        jobs[a.job].remaining = remaining;
                        jobs[a.job].started = true;
                        jobs[a.job].pause_time = now;
                        waiting.push(a.job);
                        active = None;
                    }
                }
            }
        }

        while active.is_none() && !waiting.is_empty() {
            waiting.sort_by(|&a, &b| compare_waiters(&jobs[a], &jobs[b]));
            let j = waiting.remove(0);
            let id = jobs[j].id;
            let cost = match previous_id {
                Some(prev) if prev != id => switch_cost,
                _ => 0,
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
            })?;
            let finish = work_at
                .checked_add(jobs[j].remaining)
                .ok_or(Error::TimeOverflow {
                    customer: id,
                    time: now,
                })?;

            let kind = if jobs[j].started {
                Kind::Resume {
                    remaining: jobs[j].remaining,
                    work_at,
                    finish,
                }
            } else {
                jobs[j].started = true;
                Kind::Start {
                    remaining: jobs[j].remaining,
                    work_at,
                    finish,
                }
            };
            trace.push(Event {
                time: now,
                customer: id,
                kind,
            });
            previous_id = Some(id);

            if finish == now {
                trace.push(Event {
                    time: now,
                    customer: id,
                    kind: Kind::Finish,
                });
                done += 1;
            } else {
                active = Some(Active {
                    job: j,
                    work_at,
                    finish,
                });
            }
        }

        if done == jobs.len() {
            break;
        }

        let mut candidate = if cursor < order.len() {
            Some(jobs[order[cursor]].arrival)
        } else {
            None
        };
        if let Some(a) = active {
            for t in [a.work_at, a.finish] {
                if t > now {
                    candidate = Some(candidate.map_or(t, |old| old.min(t)));
                }
            }
        }
        match candidate {
            Some(t) => now = t,
            None if !waiting.is_empty() => continue,
            None => break,
        }
    }

    Ok(trace)
}
