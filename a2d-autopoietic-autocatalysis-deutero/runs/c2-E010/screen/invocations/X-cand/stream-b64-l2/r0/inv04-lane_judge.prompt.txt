            tail_len: 0,
            finished: false,
        }
    }

        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = String::with_capacity((self.tail_len + input.len()) / 3 * 4);
        let mut i = 0;

        if self.tail_len != 0 {

            i += 1;
        }

            offsets: [0; 4],
            len: 0,
            pending_second_pad: None,
            terminal: false,
            offset: 0,
            finished: false,
            sticky: None,
        }
    }

            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();

            let at = self.offset;
            self.offset += 1;

            if self.terminal {
                return self.fail(Error::BadPadding { offset: at });
            }

            if let Some(pad_at) = self.pending_second_pad {
                if byte != b'=' {
                    return self.fail(Error::BadPadding { offset: at });
                }
                let a = self.group[0];
                let b = self.group[1];
                out.push((a << 2) | (b >> 4));
                self.pending_second_pad = None;
                self.len = 0;
                self.terminal = true;
                let _ = pad_at;
                continue;
            }

            let pos = self.len;
            if pos <= 1 && byte == b'=' {
                return self.fail(Error::BadPadding { offset: at });
            }
            if pos == 2 && byte == b'=' {
                self.group[pos] = 0;
                self.offsets[pos] = at;
                self.len = 3;
                self.pending_second_pad = Some(at);
                continue;
            }
            if pos == 3 && byte == b'=' {
                let a = self.group[0];
                let b = self.group[1];
                let c = self.group[2];
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
                self.len = 0;
                self.terminal = true;
                continue;
            }

            let Some(value) = digit(byte) else {
                return self.fail(Error::InvalidByte { offset: at, byte });
            };
            self.group[pos] = value;
            self.offsets[pos] = at;
            self.len += 1;

            if self.len == 4 {
                let a = self.group[0];
                let b = self.group[1];
                let c = self.group[2];
                let d = self.group[3];
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
                out.push((c << 6) | d);
                self.len = 0;
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(at) = self.pending_second_pad {
            return self.fail(Error::BadPadding { offset: at });
        }
        if self.len != 0 {
            return self.fail(Error::Truncated {
                offset: self.offsets[0],
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

pub struct Decoder {
    symbols: Vec<(u8, usize)>,
    offset: usize,
    ended: bool,
    padded: bool,
    closed: bool,
    failure: Option<Error>,

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn value(b: u8) -> Option<u8> {
    ALPHABET.iter().position(|&x| x == b).map(|x| x as u8)
}

fn emit_three(a: u8, b: u8, c: u8, out: &mut String) {
    let v = [
        a >> 2,
        ((a & 3) << 4) | (b >> 4),
        ((b & 15) << 2) | (c >> 6),
        c & 63,
    ];
    for n in v {
        out.push(ALPHABET[n as usize] as char);
    }
}

fn emit_remainder(bytes: &[u8], out: &mut String) {
    if bytes.len() == 1 {
        let a = bytes[0];
        out.push(ALPHABET[(a >> 2) as usize] as char);
        out.push(ALPHABET[((a & 3) << 4) as usize] as char);
        out.push_str("==");
    } else if bytes.len() == 2 {
        let a = bytes[0];
        let b = bytes[1];
        out.push(ALPHABET[(a >> 2) as usize] as char);
        out.push(ALPHABET[(((a & 3) << 4) | (b >> 4)) as usize] as char);
        out.push(ALPHABET[((b & 15) << 2) as usize] as char);
        out.push('=');
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut out = String::new();
    let mut chunks = input.chunks_exact(3);
    for chunk in &mut chunks {
        emit_three(chunk[0], chunk[1], chunk[2], &mut out);
    }
    emit_remainder(chunks.remainder(), &mut out);
    out
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut d = Decoder::new();
    let mut out = d.push(input)?;
    out.extend(d.finish()?);
    Ok(out)
}

    pub fn new() -> Self {

            pending: Vec::with_capacity(2),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut out = String::new();
        let mut i = 0;

        if !self.pending.is_empty() {
            while self.pending.len() < 3 && i < input.len() {
                if self.pending.len() == 2 {
                    emit_three(self.pending[0], self.pending[1], input[i], &mut out);
                    self.pending.clear();
                    i += 1;
                    break;
                }
                self.pending.push(input[i]);
                i += 1;
            }
        }
        while i + 3 <= input.len() {
            emit_three(input[i], input[i + 1], input[i + 2], &mut out);
            i += 3;
        }
        self.pending.extend_from_slice(&input[i..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut out = String::new();
        emit_remainder(&self.pending, &mut out);
        self.pending.clear();
        self.closed = true;
        Ok(out)
    }
}

impl Decoder {
    pub fn new() -> Self {

            symbols: Vec::with_capacity(4),
            offset: 0,
            ended: false,
            padded: false,
            closed: false,
            failure: None,
        }

    fn scan(&mut self, b: u8, at: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.ended {
            return Err(Error::BadPadding { offset: at });
        }
        if let Some(&(last, pad_at)) = self.symbols.last() {
            if last == b'=' && self.symbols.len() == 3 {
                if b != b'=' {
                    return Err(Error::BadPadding { offset: at });
                }
                let a = self.symbols[0].0;
                let c = self.symbols[1].0;
                out.push((a << 2) | (c >> 4));
                self.symbols.clear();
                self.ended = true;
                self.padded = true;
                let _ = pad_at;
                return Ok(());
            }
        }

        let pos = self.symbols.len();
        if b == b'=' {
            match pos {
                0 | 1 => return Err(Error::BadPadding { offset: at }),
                2 => {
                    self.symbols.push((b, at));
                    return Ok(());
                }
                3 => {
                    let a = self.symbols[0].0;
                    let c = self.symbols[1].0;
                    let d = self.symbols[2].0;
                    out.push((a << 2) | (c >> 4));
                    out.push((c << 4) | (d >> 2));
                    self.symbols.clear();
                    self.ended = true;
                    self.padded = true;
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        let v = match value(b) {
            Some(x) => x,
            None => return Err(Error::InvalidByte { offset: at, byte: b }),
        };
        self.symbols.push((v, at));
        if self.symbols.len() == 4 {
            let a = self.symbols[0].0;
            let c = self.symbols[1].0;
            let d = self.symbols[2].0;
            let e = self.symbols[3].0;
            out.extend_from_slice(&[(a << 2) | (c >> 4), (c << 4) | (d >> 2), (d << 6) | e]);
            self.symbols.clear();
        }
        Ok(())
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(e) = &self.failure {
            return Err(e.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut out = Vec::new();
        for &b in input {
            let at = self.offset;
            let next = match self.offset.checked_add(1) {
                Some(x) => x,
                None => return self.fail(Error::Truncated { offset: at }),
            };
            self.offset = next;
            if let Err(e) = self.scan(b, at, &mut out) {
                return self.fail(e);
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(e) = &self.failure {
            return Err(e.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        if let Some(&(_, at)) = self.symbols.iter().find(|(b, _)| *b == b'=') {
            return self.fail(Error::BadPadding { offset: at });
        }
        if !self.symbols.is_empty() {
            return self.fail(Error::Truncated {
                offset: self.symbols[0].1,
            });
        }
        self.closed = true;
        Ok(Vec::new())
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }

            out.push(ALPHABET[((b & 15) << 2) as usize] as char);
            out.push('=');
        }
        _ => {}
    }
    out
}

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub struct Decoder {
    vals: [u8; 4],
    starts: [usize; 4],
    count: usize,
    second_pad: Option<usize>,
    padded_end: bool,
    absolute: usize,
    closed: bool,
    failed: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            vals: [0; 4],
            starts: [0; 4],
            count: 0,
            second_pad: None,
            padded_end: false,
            absolute: 0,
            closed: false,
            failed: None,
        }
    }

    fn error(&mut self, error: Error) -> Result<Vec<u8>, Error> {
        self.failed = Some(error.clone());
        Err(error)
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.failed {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let here = self.absolute;
            self.absolute += 1;

            if self.padded_end {
                return self.error(Error::BadPadding { offset: here });
            }
            if self.second_pad.is_some() {
                if byte == b'=' {
                    output.push((self.vals[0] << 2) | (self.vals[1] >> 4));
                    self.second_pad = None;
                    self.count = 0;
                    self.padded_end = true;
                    continue;
                }
                return self.error(Error::BadPadding { offset: here });
            }

            if self.count < 2 && byte == b'=' {
                return self.error(Error::BadPadding { offset: here });
            }
            if self.count == 2 && byte == b'=' {
                self.starts[2] = here;
                self.count = 3;
                self.second_pad = Some(here);
                continue;
            }
            if self.count == 3 && byte == b'=' {
                output.push((self.vals[0] << 2) | (self.vals[1] >> 4));
                output.push((self.vals[1] << 4) | (self.vals[2] >> 2));
                self.count = 0;
                self.padded_end = true;
                continue;
            }

            let Some(value) = digit(byte) else {
                return self.error(Error::InvalidByte { offset: here, byte });
            };

            self.vals[self.count] = value;
            self.starts[self.count] = here;
            self.count += 1;
            if self.count == 4 {
                output.push((self.vals[0] << 2) | (self.vals[1] >> 4));
                output.push((self.vals[1] << 4) | (self.vals[2] >> 2));
                output.push((self.vals[2] << 6) | self.vals[3]);
                self.count = 0;
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.failed {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        if let Some(at) = self.second_pad {
            return self.error(Error::BadPadding { offset: at });
        }
        if self.count != 0 {
            return self.error(Error::Truncated {
                offset: self.starts[0],
            });
        }
        self.closed = true;
        Ok(Vec::new())
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

const ALPHABET: &[u8; 64] =

pub fn encode(input: &[u8]) -> String {
    let mut out = String::with_capacity((input.len() + 2) / 3 * 4);
    let mut chunks = input.chunks_exact(3);
    for chunk in &mut chunks {
        let n = ((chunk[0] as u32) << 16) | ((chunk[1] as u32) << 8) | chunk[2] as u32;
        out.push(ALPHABET[((n >> 18) & 63) as usize] as char);
        out.push(ALPHABET[((n >> 12) & 63) as usize] as char);
        out.push(ALPHABET[((n >> 6) & 63) as usize] as char);
        out.push(ALPHABET[(n & 63) as usize] as char);
    }
    match chunks.remainder() {
        [a] => {
            out.push(ALPHABET[(a >> 2) as usize] as char);
            out.push(ALPHABET[((a & 3) << 4) as usize] as char);
            out.push_str("==");
        }
        [a, b] => {
            out.push(ALPHABET[(a >> 2) as usize] as char);
            out.push(ALPHABET[(((a & 3) << 4) | (b >> 4)) as usize] as char);
            out.push(ALPHABET[((b & 15) << 2) as usize] as char);
            out.push('=');
        }
        _ => {}
    }
    out
}

pub struct Encoder {
    tail: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self { tail: Vec::new(), closed: false }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut data = Vec::with_capacity(self.tail.len() + input.len());
        data.extend_from_slice(&self.tail);
        data.extend_from_slice(input);
        let complete_len = data.len() / 3 * 3;
        let output = encode(&data[..complete_len]);
        self.tail.clear();
        self.tail.extend_from_slice(&data[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;
        Ok(encode(&self.tail))
    }
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    pending_second_pad: Option<usize>,
    terminal: bool,
    next_offset: usize,
    closed: bool,
    sticky: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            len: 0,
            pending_second_pad: None,
            terminal: false,
            next_offset: 0,
            closed: false,
            sticky: None,
        }
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky = Some(error.clone());
        error
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.next_offset;
            self.next_offset = self.next_offset.saturating_add(1);

            if self.terminal {
                return Err(self.fail(Error::BadPadding { offset }));
            }
            if self.pending_second_pad.is_some() && byte != b'=' {
                return Err(self.fail(Error::BadPadding { offset }));
            }
            if self.len < 2 && byte == b'=' {
                return Err(self.fail(Error::BadPadding { offset }));
            }

            if byte == b'=' {
                if self.len == 2 {
                    self.pending_second_pad = Some(offset);
                    self.group[2] = 0;
                    self.offsets[2] = offset;
                    self.len = 3;
                    continue;
                } else if self.len == 3 {
                    self.group[3] = 0;
                    self.offsets[3] = offset;
                    self.len = 4;
                    self.terminal = true;
                    output.extend(self.emit_group());
                    continue;
                } else {
                    return Err(self.fail(Error::BadPadding { offset }));
                }
            }

            let Some(v) = value(byte) else {
                return Err(self.fail(Error::InvalidByte { offset, byte }));
            };
            self.group[self.len] = v;
            self.offsets[self.len] = offset;
            self.len += 1;

            if self.len == 4 {
                output.extend(self.emit_group());
            }
        }
        Ok(output)
    }

    fn emit_group(&mut self) -> Vec<u8> {
        let a = self.group[0];
        let b = self.group[1];
        let c = self.group[2];
        let d = self.group[3];
        let mut result = Vec::with_capacity(3);
        result.push((a << 2) | (b >> 4));
        if !self.terminal || self.pending_second_pad.is_none() {
            result.push((b << 4) | (c >> 2));
        }
        if !self.terminal {
            result.push((c << 6) | d);
        }
        self.group = [0; 4];
        self.offsets = [0; 4];
        self.len = 0;
        self.pending_second_pad = None;
        result
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        if let Some(offset) = self.pending_second_pad {
            return Err(self.fail(Error::BadPadding { offset }));
        }
        if self.len != 0 {
            return Err(self.fail(Error::Truncated { offset: self.offsets[0] }));
        }
        self.closed = true;
        Ok(Vec::new())
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i + 3 <= input.len() {
        let a = input[i];
        let b = input[i + 1];
        let c = input[i + 2];
        out.push(TABLE[(a >> 2) as usize] as char);
        out.push(TABLE[(((a & 3) << 4) | (b >> 4)) as usize] as char);
        out.push(TABLE[(((b & 15) << 2) | (c >> 6)) as usize] as char);
        out.push(TABLE[(c & 63) as usize] as char);
        i += 3;
    }
    match input.len() - i {
        1 => {
            let a = input[i];
            out.push(TABLE[(a >> 2) as usize] as char);
            out.push(TABLE[((a & 3) << 4) as usize] as char);

pub struct Encoder {

    pending_len: usize,
    done: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self { pending: [0; 2], pending_len: 0, done: false }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.done {
            return Err(Error::AlreadyFinished);
        }
        let mut out = String::new();
        let mut index = 0;
        if self.pending_len > 0 {
            while self.pending_len < 3 && index < input.len() {
                if self.pending_len < 2 {
                    self.pending[self.pending_len] = input[index];
                } else {
                    let a = self.pending[0];
                    let b = self.pending[1];
                    let c = input[index];
                    out.push(TABLE[(a >> 2) as usize] as char);
                    out.push(TABLE[(((a & 3) << 4) | (b >> 4)) as usize] as char);
                    out.push(TABLE[(((b & 15) << 2) | (c >> 6)) as usize] as char);
                    out.push(TABLE[(c & 63) as usize] as char);
                }
                self.pending_len += 1;
                index += 1;
            }
            if self.pending_len == 3 {
                self.pending_len = 0;
            }
        }
        while index + 3 <= input.len() {
            out.push_str(&encode(&input[index..index + 3]));
            index += 3;
        }
        while index < input.len() {
            self.pending[self.pending_len] = input[index];
            self.pending_len += 1;
            index += 1;
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.done {
            return Err(Error::AlreadyFinished);
        }
        self.done = true;
        Ok(encode(&self.pending[..self.pending_len]))
    }
}

pub struct Decoder {
    quartet: [u8; 4],
    offsets: [usize; 4],
    count: usize,
    pending_equal: Option<usize>,
    padded_done: bool,
    offset: usize,
    finished: bool,
    error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            quartet: [0; 4],
            offsets: [0; 4],
            count: 0,
            pending_equal: None,
            padded_done: false,
            offset: 0,
            finished: false,
            error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(e) = &self.error {
            return Err(e.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &b in input {
            let here = self.offset;
            self.offset = self.offset.saturating_add(1);

            let error = if self.padded_done {
                Some(Error::BadPadding { offset: here })
            } else if self.pending_equal.is_some() && b != b'=' {
                Some(Error::BadPadding { offset: here })
            } else if self.count < 2 && b == b'=' {
                Some(Error::BadPadding { offset: here })
            } else if b == b'=' && self.count == 2 {
                self.pending_equal = Some(here);
                self.quartet[2] = 0;
                self.offsets[2] = here;
                self.count = 3;
                None
            } else if b == b'=' && self.count == 3 {
                self.quartet[3] = 0;
                self.offsets[3] = here;
                self.count = 4;
                self.padded_done = true;
                None
            } else if b == b'=' {
                Some(Error::BadPadding { offset: here })
            } else if let Some(v) = decode_digit(b) {
                self.quartet[self.count] = v;
                self.offsets[self.count] = here;
                self.count += 1;
                None
            } else {
                Some(Error::InvalidByte { offset: here, byte: b })
            };

            if let Some(e) = error {
                self.error = Some(e.clone());
                return Err(e);
            }
            if self.count == 4 {
                let a = self.quartet[0];
                let b = self.quartet[1];
                let c = self.quartet[2];
                let d = self.quartet[3];
                output.push((a << 2) | (b >> 4));
                if !self.padded_done || self.pending_equal.is_none() {
                    output.push((b << 4) | (c >> 2));
                }
                if !self.padded_done {
                    output.push((c << 6) | d);
                }
                if !self.padded_done {
                    self.count = 0;
                    self.quartet = [0; 4];
                    self.offsets = [0; 4];
                }
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(e) = &self.error {
            return Err(e.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        if let Some(at) = self.pending_equal {
            let e = Error::BadPadding { offset: at };
            self.error = Some(e.clone());
            return Err(e);
        }
        if self.count != 0 && !self.padded_done {
            let e = Error::Truncated { offset: self.offsets[0] };
            self.error = Some(e.clone());
            return Err(e);
        }
        self.finished = true;
        Ok(Vec::new())
    }
}
