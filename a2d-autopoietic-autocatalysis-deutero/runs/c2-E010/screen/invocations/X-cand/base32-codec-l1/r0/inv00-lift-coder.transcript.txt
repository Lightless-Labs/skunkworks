pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

const ALPHABET: &[u8; 32] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";

pub fn encode(input: &[u8]) -> String {
    let mut out = String::new();
    let mut acc: u32 = 0;
    let mut bits: u32 = 0;

    for &byte in input {
        acc = (acc << 8) | u32::from(byte);
        bits += 8;
        while bits >= 5 {
            bits -= 5;
            out.push(ALPHABET[((acc >> bits) & 31) as usize] as char);
        }
        if bits == 0 {
            acc = 0;
        } else {
            acc &= (1u32 << bits) - 1;
        }
    }

    if bits != 0 {
        let value = (acc << (5 - bits)) & 31;
        out.push(ALPHABET[value as usize] as char);
    }

    let padding = (8 - (out.len() % 8)) % 8;
    for _ in 0..padding {
        out.push('=');
    }
    out
}

fn value(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'2'..=b'7' => Some(byte - b'2' + 26),
        _ => None,
    }
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let n = bytes.len();

    // Pass 1: lexical validation across the entire input.
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && value(byte).is_none() {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    // Pass 2: find the first padding byte and reject data after it.
    let p = bytes.iter().position(|&b| b == b'=').unwrap_or(n);
    for (offset, &byte) in bytes.iter().enumerate().skip(p) {
        if byte != b'=' {
            return Err(DecodeError::Padding { offset });
        }
    }

    // Pass 3: validate the number of data symbols.
    let residue = p % 8;

        2 => 6,
        4 => 4,
        5 => 3,
        7 => 1,
        _ => return Err(DecodeError::InvalidLength { offset: n }),

    if n - p != required_padding || n % 8 != 0 {

        2 => 2,

    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

const ALPHABET: &[u8; 32] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";

pub fn encode(input: &[u8]) -> String {
    let mut out = String::new();
    let mut bit_index = 0usize;
    let total_bits = input.len().saturating_mul(8);

    while bit_index < total_bits {
        let mut value = 0u8;
        for shift in (0..5).rev() {
            value <<= 1;
            if bit_index < total_bits {
                let byte = input[bit_index / 8];
                let bit_in_byte = 7 - (bit_index % 8);
                value |= (byte >> bit_in_byte) & 1;
                bit_index += 1;
            }
        }
        out.push(ALPHABET[value as usize] as char);
    }

    let padding = (8 - (out.len() % 8)) % 8;
    for _ in 0..padding {
        out.push('=');
    }
    out
}

fn symbol_value(byte: u8) -> Option<u8> {
    if byte >= b'A' && byte <= b'Z' {
        Some(byte - b'A')
    } else if byte >= b'2' && byte <= b'7' {
        Some(byte - b'2' + 26)
    } else {
        None
    }
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let n = bytes.len();

    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && symbol_value(byte).is_none() {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let mut p = n;
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte == b'=' {
            p = offset;
            break;
        }
    }
    for (offset, &byte) in bytes.iter().enumerate().skip(p) {
        if symbol_value(byte).is_some() {
            return Err(DecodeError::Padding { offset });
        }
    }

    let residue = p % 8;
    let expected = match residue {
        0 => 0,
        2 => 6,
        4 => 4,
        5 => 3,
        7 => 1,
        _ => return Err(DecodeError::InvalidLength { offset: n }),
    };

    let actual = n - p;
    if actual != expected || n % 8 != 0 {
        return Err(DecodeError::Padding { offset: p });
    }

    let trailing = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };
    if trailing > 0 {
        let final_value = symbol_value(bytes[p - 1]).expect("validated data symbol");
        if final_value & ((1u8 << trailing) - 1) != 0 {
            return Err(DecodeError::TrailingBits { offset: p - 1 });
        }
    }

    let mut result = Vec::with_capacity(p.saturating_mul(5) / 8);
    let mut current = 0u8;
    let mut filled = 0u8;

    for &byte in &bytes[..p] {
        let v = symbol_value(byte).expect("validated alphabet symbol");
        for bit in (0..5).rev() {
            current = (current << 1) | ((v >> bit) & 1);
            filled += 1;
            if filled == 8 {
                result.push(current);
                current = 0;
                filled = 0;
            }
        }
    }

    Ok(result)
}
