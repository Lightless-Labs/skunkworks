    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

const ALPHABET: &[u8; 32] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";

pub fn encode(input: &[u8]) -> String {
    let mut output = String::new();
    let mut buffer: u32 = 0;
    let mut bits: u32 = 0;

    for &byte in input {
        buffer = (buffer << 8) | u32::from(byte);
        bits += 8;

        while bits >= 5 {
            bits -= 5;
            let value = ((buffer >> bits) & 31) as usize;
            output.push(ALPHABET[value] as char);
        }

        if bits == 0 {
            buffer = 0;
        } else {
            buffer &= (1u32 << bits) - 1;
        }
    }

    if bits > 0 {
        let value = ((buffer << (5 - bits)) & 31) as usize;
        output.push(ALPHABET[value] as char);
    }

    while output.len() % 8 != 0 {
        output.push('=');
    }

    output
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

const ALPHABET: &[u8; 32] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";

fn value_of(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'2'..=b'7' => Some(byte - b'2' + 26),
        _ => None,
    }
}

pub fn encode(input: &[u8]) -> String {
    let symbols = input.len().saturating_mul(8).saturating_add(4) / 5;
    let capacity = symbols.saturating_add(7) / 8 * 8;
    let mut output = String::with_capacity(capacity);

    let mut bit_index = 0usize;
    let total_bits = input.len().saturating_mul(8);

    while bit_index < total_bits {
        let mut value = 0u8;
        for offset in 0..5 {
            value <<= 1;
            let source_bit = bit_index + offset;
            if source_bit < total_bits {
                let byte = input[source_bit / 8];
                let shift = 7 - (source_bit % 8);
                value |= (byte >> shift) & 1;
            }
        }
        output.push(ALPHABET[value as usize] as char);
        bit_index += 5;
    }

    while output.len() % 8 != 0 {
        output.push('=');
    }

    output
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let n = bytes.len();

    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && value_of(byte).is_none() {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let p = bytes.iter().position(|&byte| byte == b'=').unwrap_or(n);
    for (offset, &byte) in bytes.iter().enumerate().skip(p) {
        if byte != b'=' {
            return Err(DecodeError::Padding { offset });
        }
    }

    let residue = p % 8;
    let required_padding = match residue {
        0 => 0,
        2 => 6,
        4 => 4,
        5 => 3,
        7 => 1,
        _ => return Err(DecodeError::InvalidLength { offset: n }),
    };

    if n % 8 != 0 || n - p != required_padding {
        return Err(DecodeError::Padding { offset: p });
    }

    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };

    if unused_bits != 0 {
        let last = value_of(bytes[p - 1]).unwrap();
        let mask = (1u8 << unused_bits) - 1;
        if last & mask != 0 {
            return Err(DecodeError::TrailingBits { offset: p - 1 });
        }
    }

    let mut output = Vec::with_capacity(p.saturating_mul(5) / 8);
    let mut current = 0u8;
    let mut bit_count = 0u8;

    for &symbol in &bytes[..p] {
        current = (current << 5) | value_of(symbol).unwrap();
        bit_count += 5;

        if bit_count >= 8 {
            bit_count -= 8;
            output.push(current >> bit_count);
            current &= (1u8 << bit_count).wrapping_sub(1);
        }
    }

    Ok(output)
}
