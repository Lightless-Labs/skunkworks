- Check non-ASCII before searching for quotes or separators; offsets are byte offsets.
- Only an opening quote at byte zero protects `@`; a trailing escape leaves it unclosed.
- Quoted-content and escape errors precede decoded-length checks; an escape’s origin is its backslash.
- A single final domain dot is removed; its original position identifies an empty final label.
- Label length and hyphen checks precede character checks; the label-count check precedes numeric-TLD validation.
- Both solutions use these readings; Solution B uses a separate validation approach.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_prefix_end = if bytes[0] == b'"' {
        let mut i = 1;
        let mut end = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    end = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match end {
            Some(i) => Some(i),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quoted_prefix_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }
    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let mut local = String::new();
    let quoted;
    if bytes[0] == b'"' {
        let close = quoted_prefix_end.unwrap();
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }
        quoted = true;
        let mut origins = Vec::new();
        let mut j = 1;
        while j < close {
            if bytes[j] == b'\\' {
                if j + 1 >= close || !matches!(bytes[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: j });
                }
                local.push(bytes[j + 1] as char);
                origins.push(j);
                j += 2;
            } else {
                let b = bytes[j];
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: j });
                }
                local.push(b as char);
                origins.push(j);
                j += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        quoted = false;
        let len = sep;
        if len > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }
        for j in 0..len {
            let b = bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == len || bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: j });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: j });
            }
        }
        local.push_str(&input[..sep]);
    }

    let domain_start = sep + 1;
    let mut domain_end = bytes.len();
    let root_dot = if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut label_starts = Vec::new();
    let mut start = domain_start;
    for j in domain_start..=domain_end {
        if j == domain_end || bytes[j] == b'.' {
            if start == j {
                return Err(EmailError::EmptyLabel {
                    at: if j == domain_end {
                        root_dot.unwrap_or(start)
                    } else {
                        j
                    },
                });
            }
            let len = j - start;
            if len > 63 {
                return Err(EmailError::LabelTooLong { at: start + 63 });
            }
            if bytes[start] == b'-' {
                return Err(EmailError::LabelHyphen { at: start });
            }
            if bytes[j - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: j - 1 });
            }
            for k in start..j {
                if !bytes[k].is_ascii_alphanumeric() && bytes[k] != b'-' {
                    return Err(EmailError::DomainChar { at: k });
                }
            }
            label_starts.push(start);
            start = j + 1;
        }
    }

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    let last_start = *label_starts.last().unwrap();
    if bytes[last_start..domain_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let normalized_domain = input[domain_start..domain_end].to_ascii_lowercase();
    Ok(Address {
        local,
        domain: normalized_domain,
        quoted,
    })
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let b = input.as_bytes();
    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    for (i, byte) in b.iter().enumerate() {
        if *byte >= 128 {
            return Err(EmailError::NonAscii { at: i });
        }
    }

    let closing = if b[0] == b'"' {
        let mut i = 1usize;
        let mut found = None;
        while i < b.len() {
            if b[i] == b'\\' {
                i = i.saturating_add(2);
            } else if b[i] == b'"' {
                found = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        match found {
            Some(p) => Some(p),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut first_at = None;
    let mut second_at = None;
    let mut i = 0;
    while i < b.len() {
        if let Some(c) = closing {
            if i <= c {
                i = c + 1;
                continue;
            }
        }
        if b[i] == b'@' {
            if first_at.is_none() {
                first_at = Some(i);
            } else if second_at.is_none() {
                second_at = Some(i);
            }
        }
        i += 1;
    }
    let sep = match first_at {
        None => return Err(EmailError::MissingAt),
        Some(p) if second_at.is_some() => {
            return Err(EmailError::MultipleAt {
                at: second_at.unwrap(),
            });
        }
        Some(p) => p,
    };
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let (local, is_quoted) = if b[0] == b'"' {
        let end = closing.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal { at: end + 1 });
        }
        let mut value = Vec::new();
        let mut origin = Vec::new();
        let mut p = 1;
        while p < end {
            if b[p] == b'\\' {
                if p + 1 >= end || !matches!(b[p + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: p });
                }
                value.push(b[p + 1]);
                origin.push(p);
                p += 2;
            } else {
                if !(0x20..=0x7e).contains(&b[p]) {
                    return Err(EmailError::QuotedChar { at: p });
                }
                value.push(b[p]);
                origin.push(p);
                p += 1;
            }
        }
        if value.is_empty() {
            return Err(EmailError::EmptyLocal { at: end });
        }
        if value.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origin[64] });
        }
        (String::from_utf8(value).unwrap(), true)
    } else {
        if sep > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }
        let mut p = 0;
        while p < sep {
            if b[p] == b'.' {
                if p == 0 || p + 1 == sep || b[p - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: p });
                }
            } else if !b[p].is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b[p])
            {
                return Err(EmailError::LocalChar { at: p });
            }
            p += 1;
        }
        (input[..sep].to_owned(), false)
    };

    let ds = sep + 1;
    let mut de = b.len();
    let root = if de > ds && b[de - 1] == b'.' {
        de -= 1;
        Some(de)
    } else {
        None
    };
    if ds == de {
        return Err(EmailError::EmptyDomain { at: ds });
    }
    if de - ds > 253 {
        return Err(EmailError::DomainTooLong { at: ds + 253 });
    }

    let mut count = 0usize;
    let mut last_start = ds;
    let mut start = ds;
    let mut p = ds;
    loop {
        if p == de || b[p] == b'.' {
            if p == start {
                let at = if p == de { root.unwrap_or(start) } else { p };
                return Err(EmailError::EmptyLabel { at });
            }
            let n = p - start;
            if n > 63 {
                return Err(EmailError::LabelTooLong { at: start + 63 });
            }
            if b[start] == b'-' {
                return Err(EmailError::LabelHyphen { at: start });
            }
            if b[p - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: p - 1 });
            }
            for q in start..p {
                if !b[q].is_ascii_alphanumeric() && b[q] != b'-' {
                    return Err(EmailError::DomainChar { at: q });
                }
            }
            count += 1;
            last_start = start;
            if p == de {
                break;
            }
            start = p + 1;
        }
        p += 1;
    }

    if count < 2 {
        return Err(EmailError::DomainNeedsDot { at: ds });
    }
    if b[last_start..de].iter().all(|x| x.is_ascii_digit()) {
        return Err(EmailError::NumericTld { at: last_start });
    }
    Ok(Address {
        local,
        domain: input[ds..de].to_ascii_lowercase(),
        quoted: is_quoted,
    })
}
```
