        }
        i += 1;

        return Err(EmailError::MissingAt);

        return Err(EmailError::MultipleAt { at: separators[1] });

            return Err(EmailError::EmptyLocal { at: close });

            if b == b'.' {

                    return Err(EmailError::LocalDot { at: p });
                }

            {
                return Err(EmailError::LocalChar { at: p });
            }
        }
        String::from_utf8(local_bytes.to_vec()).expect("validated ASCII")

        bytes.len() - 1

        bytes.len()

        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,

    let mut label_start = 0usize;
    let mut last_label = &domain_bytes[0..0];

        if boundary != domain_bytes.len() && domain_bytes[boundary] != b'.' {
            continue;
        }

        if label.is_empty() {
            let at = if label_start == domain_bytes.len() && removed_dot {
                bytes.len() - 1
            } else {
                domain_start + label_start
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if label.len() > 63 {
            return Err(EmailError::LabelTooLong {
                at: domain_start + label_start + 63,
            });
        }
        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + label_start,
            });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + boundary - 1,
            });
        }
        for (offset, &b) in label.iter().enumerate() {
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: domain_start + label_start + offset,
                });
            }
        }

        label_count += 1;
        last_label = label;
        label_start = boundary + 1;
    }

    if label_count < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + domain_bytes.len() - last_label.len(),
        });
    }

    let domain = domain_bytes
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted: local_bytes[0] == b'"',
    })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    for (i, &byte) in bytes.iter().enumerate() {
        if byte > 127 {
            return Err(EmailError::NonAscii { at: i });
        }
    }

    let quote_close = if bytes.first() == Some(&b'"') {
        let mut cursor = 1;
        loop {
            if cursor >= bytes.len() {
                return Err(EmailError::UnclosedQuote { at: 0 });
            }
            match bytes[cursor] {
                b'\\' => {
                    cursor += 1;
                    if cursor < bytes.len() {
                        cursor += 1;
                    }
                }
                b'"' => break Some(cursor),
                _ => cursor += 1,
            }
        }
    } else {
        None
    };

    let mut separator = None;
    let mut second_separator = None;
    for i in 0..bytes.len() {
        if quote_close.is_some_and(|end| i <= end) {
            continue;
        }
        if bytes[i] == b'@' {
            if separator.is_none() {
                separator = Some(i);
            } else if second_separator.is_none() {
                second_separator = Some(i);
            }
        }
    }
    let sep = match separator {
        None => return Err(EmailError::MissingAt),
        Some(first) => {
            if let Some(second) = second_separator {
                return Err(EmailError::MultipleAt { at: second });
            }
            first
        }
    };
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let is_quoted = bytes[0] == b'"';
    let local = if is_quoted {
        let close = quote_close.unwrap();
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut cursor = 1;
        while cursor < close {
            let c = bytes[cursor];
            if c == b'\\' {
                let escaped = bytes.get(cursor + 1).copied();
                match escaped {
                    Some(b'"' | b'\\' | b'@') => {
                        decoded.push(escaped.unwrap());
                        origins.push(cursor);
                        cursor += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: cursor }),
                }
            } else if (0x20..=0x7e).contains(&c) {
                decoded.push(c);
                origins.push(cursor);
                cursor += 1;
            } else {
                return Err(EmailError::QuotedChar { at: cursor });
            }
        }
        if decoded.len() == 0 {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        String::from_utf8(decoded).unwrap()
    } else {
        let local_bytes = &bytes[..sep];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }
        let mut prev_dot = false;
        for (i, &c) in local_bytes.iter().enumerate() {
            if c == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || prev_dot {
                    return Err(EmailError::LocalDot { at: i });
                }
                prev_dot = true;
            } else {
                prev_dot = false;
                if !c.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&c) {
                    return Err(EmailError::LocalChar { at: i });
                }
            }
        }
        String::from_utf8(local_bytes.to_vec()).unwrap()
    };

    let domain_start = sep + 1;
    let has_root_dot = bytes.len() > domain_start && bytes[bytes.len() - 1] == b'.';
    let domain_end = if has_root_dot { bytes.len() - 1 } else { bytes.len() };
    let domain = &bytes[domain_start..domain_end];
    if domain.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels = Vec::new();
    let mut begin = 0;
    for cursor in 0..=domain.len() {
        if cursor == domain.len() || domain[cursor] == b'.' {
            labels.push((begin, cursor));
            begin = cursor + 1;
        }
    }
    for &(begin, end) in &labels {
        let label = &domain[begin..end];
        let at = domain_start + begin;
        if label.is_empty() {
            let position = if begin == domain.len() && has_root_dot {
                bytes.len() - 1
            } else {
                at
            };
            return Err(EmailError::EmptyLabel { at: position });
        }
        if label.len() > 63 {
            return Err(EmailError::LabelTooLong { at: at + 63 });
        }
        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen { at });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: at + label.len() - 1 });
        }
        if let Some(i) = label.iter().position(|&c| !c.is_ascii_alphanumeric() && c != b'-') {
            return Err(EmailError::DomainChar { at: at + i });
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    let (last_begin, last_end) = labels[labels.len() - 1];
    if domain[last_begin..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_begin,
        });
    }

    let lower = domain
        .iter()
        .map(u8::to_ascii_lowercase)
        .collect::<Vec<_>>();
    Ok(Address {
        local,
        domain: String::from_utf8(lower).unwrap(),
        quoted: is_quoted,
    })
}

    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    for (i, byte) in b.iter().enumerate() {
        if *byte >= 128 {
            return Err(EmailError::NonAscii { at: i });
        }
    }

    let quote_end = if b[0] == b'"' {
        let mut pos = 1usize;
        let mut found = None;
        while pos < b.len() {
            if b[pos] == b'\\' {
                pos = pos.saturating_add(2);
            } else if b[pos] == b'"' {
                found = Some(pos);
                break;
            } else {
                pos += 1;
            }
        }
        match found {
            Some(p) => Some(p),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None

    let mut second_at = None;
    let mut pos = 0usize;

        if let Some(end) = quote_end {
            if pos <= end {
                pos = end + 1;
                continue;
            }
        }
        if b[pos] == b'@' {
            if at_pos.is_some() {
                second_at = Some(pos);
                break;
            }
            at_pos = Some(pos);
        }
        pos += 1;
    }
    if let Some(pos) = second_at {
        return Err(EmailError::MultipleAt { at: pos });
    }
    let split = match at_pos {
        Some(p) => p,
        None => return Err(EmailError::MissingAt),
    };
    if split == 0 {
        return Err(EmailError::EmptyLocal { at: split });
    }

        let end = quote_end.unwrap();
        if end + 1 != split {
            return Err(EmailError::TrailingLocal { at: end + 1 });

        let mut source_positions = Vec::new();

        while p < end {
            let ch = b[p];

                let decoded = if p + 1 < end {
                    match b[p + 1] {
                        b'"' | b'\\' | b'@' => b[p + 1],
                        _ => return Err(EmailError::BadEscape { at: p }),

                    return Err(EmailError::BadEscape { at: p });
                };

                source_positions.push(p);
                p += 2;
            } else {
                if !(32..=126).contains(&ch) {
                    return Err(EmailError::QuotedChar { at: p });
                }
                result.push(ch);
                source_positions.push(p);
                p += 1;
            }
        }
        if result.is_empty() {
            return Err(EmailError::EmptyLocal { at: end });
        }
        if result.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: source_positions[64],
            });
        }
        String::from_utf8(result).unwrap()
    } else {
        if split > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }
        let mut p = 0;
        while p < split {
            let ch = b[p];
            if ch == b'.' {
                if p == 0 || p + 1 == split || b[p - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: p });
                }
            } else if !(ch.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&ch)) {
                return Err(EmailError::LocalChar { at: p });
            }
            p += 1;
        }
        String::from_utf8(b[..split].to_vec()).unwrap()
    };

    let domain_start = split + 1;
    let mut domain_end = b.len();
    let has_root_dot = domain_end > domain_start && b[domain_end - 1] == b'.';
    if has_root_dot {
        domain_end -= 1;
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut starts = Vec::new();
    let mut ends = Vec::new();
    let mut label_start = domain_start;
    for p in domain_start..domain_end {
        if b[p] == b'.' {
            starts.push(label_start);
            ends.push(p);
            label_start = p + 1;
        }
    }
    starts.push(label_start);
    ends.push(domain_end);

    let mut final_start = domain_start;
    let mut label_total = 0;
    for idx in 0..starts.len() {
        let start = starts[idx];
        let end = ends[idx];
        if start == end {
            let error_at = if has_root_dot && idx + 1 == starts.len() {
                b.len() - 1
            } else {
                start
            };
            return Err(EmailError::EmptyLabel { at: error_at });
        }
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong { at: start + 63 });
        }
        if b[start] == b'-' {
            return Err(EmailError::LabelHyphen { at: start });
        }
        if b[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: end - 1 });
        }
        for p in start..end {
            if !b[p].is_ascii_alphanumeric() && b[p] != b'-' {
                return Err(EmailError::DomainChar { at: p });
            }
        }
        label_total += 1;
        final_start = start;
    }

    if label_total < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    if b[final_start..domain_end].iter().all(|x| x.is_ascii_digit()) {
        return Err(EmailError::NumericTld { at: final_start });
    }

    let domain = b[domain_start..domain_end]
        .iter()
        .map(|x| char::from(x.to_ascii_lowercase()))
        .collect::<String>();

    Ok(Address {
        local,
        domain,
        quoted: local_is_quoted,
    })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let b = input.as_bytes();
    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    for (i, byte) in b.iter().enumerate() {
        if *byte >= 128 {
            return Err(EmailError::NonAscii { at: i });
        }
    }

    let closing = if b[0] == b'"' {
        let mut i = 1usize;
        let mut found = None;
        while i < b.len() {
            if b[i] == b'\\' {
                i = i.saturating_add(2);
            } else if b[i] == b'"' {
                found = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        match found {
            Some(p) => Some(p),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut first_at = None;
    let mut second_at = None;
    let mut i = 0;
    while i < b.len() {
        if let Some(c) = closing {
            if i <= c {
                i = c + 1;
                continue;
            }
        }
        if b[i] == b'@' {
            if first_at.is_none() {
                first_at = Some(i);
            } else if second_at.is_none() {
                second_at = Some(i);
            }
        }
        i += 1;
    }
    let sep = match first_at {
        None => return Err(EmailError::MissingAt),
        Some(p) if second_at.is_some() => {
            return Err(EmailError::MultipleAt {
                at: second_at.unwrap(),
            });
        }
        Some(p) => p,
    };
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let (local, is_quoted) = if b[0] == b'"' {
        let end = closing.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal { at: end + 1 });
        }
        let mut value = Vec::new();
        let mut origin = Vec::new();
        let mut p = 1;
        while p < end {
            if b[p] == b'\\' {
                if p + 1 >= end || !matches!(b[p + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: p });
                }
                value.push(b[p + 1]);
                origin.push(p);
                p += 2;
            } else {
                if !(0x20..=0x7e).contains(&b[p]) {
                    return Err(EmailError::QuotedChar { at: p });
                }
                value.push(b[p]);
                origin.push(p);
                p += 1;
            }
        }
        if value.is_empty() {
            return Err(EmailError::EmptyLocal { at: end });
        }
        if value.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origin[64] });
        }
        (String::from_utf8(value).unwrap(), true)
    } else {
        if sep > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }
        let mut p = 0;
        while p < sep {
            if b[p] == b'.' {
                if p == 0 || p + 1 == sep || b[p - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: p });
                }
            } else if !b[p].is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b[p])
            {
                return Err(EmailError::LocalChar { at: p });
            }
            p += 1;
        }
        (input[..sep].to_owned(), false)
    };

    let ds = sep + 1;
    let mut de = b.len();
    let root = if de > ds && b[de - 1] == b'.' {
        de -= 1;
        Some(de)
    } else {
        None
    };
    if ds == de {
        return Err(EmailError::EmptyDomain { at: ds });
    }
    if de - ds > 253 {
        return Err(EmailError::DomainTooLong { at: ds + 253 });
    }

    let mut count = 0usize;
    let mut last_start = ds;
    let mut start = ds;
    let mut p = ds;
    loop {
        if p == de || b[p] == b'.' {
            if p == start {
                let at = if p == de { root.unwrap_or(start) } else { p };
                return Err(EmailError::EmptyLabel { at });
            }
            let n = p - start;
            if n > 63 {
                return Err(EmailError::LabelTooLong { at: start + 63 });
            }
            if b[start] == b'-' {
                return Err(EmailError::LabelHyphen { at: start });
            }
            if b[p - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: p - 1 });
            }
            for q in start..p {
                if !b[q].is_ascii_alphanumeric() && b[q] != b'-' {
                    return Err(EmailError::DomainChar { at: q });
                }
            }
            count += 1;
            last_start = start;
            if p == de {
                break;
            }
            start = p + 1;
        }
        p += 1;
    }

    if count < 2 {
        return Err(EmailError::DomainNeedsDot { at: ds });
    }
    if b[last_start..de].iter().all(|x| x.is_ascii_digit()) {
        return Err(EmailError::NumericTld { at: last_start });
    }
    Ok(Address {
        local,
        domain: input[ds..de].to_ascii_lowercase(),
        quoted: is_quoted,
    })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let b = input.as_bytes();
    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = b.iter().position(|&c| c >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let quoted_prefix_end = if b[0] == b'"' {
        let mut i = 1;
        let mut found = None;
        while i < b.len() {
            match b[i] {
                b'\\' => {
                    if i + 1 < b.len() {
                        i += 2;
                    } else {
                        i += 1;
                    }
                }
                b'"' => {
                    found = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match found {
            Some(i) => Some(i),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut ats = Vec::new();
    let mut i = 0;
    while i < b.len() {
        if let Some(end) = quoted_prefix_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if b[i] == b'@' {
            ats.push(i);
        }
        i += 1;
    }

    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt { at: ats[1] });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let local_bytes = &b[..sep];
    let (local, is_quoted) = if local_bytes[0] == b'"' {
        let close = quoted_prefix_end.expect("opening quote has a close");
        if close >= sep || close + 1 != sep {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < close {
            if b[j] == b'\\' {
                if j + 1 >= close {
                    return Err(EmailError::BadEscape { at: j });
                }
                let value = match b[j + 1] {
                    b'"' | b'\\' | b'@' => b[j + 1],
                    _ => return Err(EmailError::BadEscape { at: j }),
                };
                decoded.push(value);
                origins.push(j);
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b[j]) {
                    return Err(EmailError::QuotedChar { at: j });
                }
                decoded.push(b[j]);
                origins.push(j);
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }
        for (j, &c) in local_bytes.iter().enumerate() {
            if c == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || local_bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: j });
                }
            } else if !c.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&c)
            {
                return Err(EmailError::LocalChar { at: j });
            }
        }
        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start = sep + 1;
    let mut domain_end = b.len();
    let root_dot = if domain_end > domain_start && b[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    let domain = &b[domain_start..domain_end];
    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut label_count = 0usize;
    let mut start = 0usize;
    let mut last_start = 0usize;
    for end in 0..=domain.len() {
        if end == domain.len() || domain[end] == b'.' {
            let label = &domain[start..end];
            let label_at = domain_start + start;
            if label.is_empty() {
                let at = if start == domain.len() {
                    root_dot.unwrap_or(label_at)
                } else {
                    label_at
                };
                return Err(EmailError::EmptyLabel { at });
            }
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong { at: label_at + 63 });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: label_at });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: label_at + label.len() - 1,
                });
            }
            for (k, &c) in label.iter().enumerate() {
                if !c.is_ascii_alphanumeric() && c != b'-' {
                    return Err(EmailError::DomainChar { at: label_at + k });
                }
            }
            label_count += 1;
            last_start = start;
            start = end + 1;
        }
    }

    if label_count < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    let last_label = &domain[last_start..];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

        .iter()
        .map(|c| c.to_ascii_lowercase())
        .collect::<Vec<_>>();
    Ok(Address {
        local,
        domain: String::from_utf8(lower).unwrap(),
        quoted: is_quoted,
    })
