#[derive(Clone)]
struct Gap {
    start: usize,
    end: usize,
}

fn horizontal(b: u8) -> bool {
    matches!(b, b' ' | b'\t')
}

fn folded_end(b: &[u8], p: usize) -> Option<usize> {
    if b.get(p) != Some(&b'\r') || b.get(p + 1) != Some(&b'\n') {
        return None;
    }
    let mut q = p + 2;
    if !b.get(q).copied().map(horizontal).unwrap_or(false) {
        return None;
    }
    while b.get(q).copied().map(horizontal).unwrap_or(false) {
        q += 1;
    }
    Some(q)
}

fn append_gap(gaps: &mut Vec<Gap>, start: usize, end: usize) {
    if let Some(g) = gaps.last_mut() {
        if g.end == start {
            g.end = end;
            return;
        }
    }
    gaps.push(Gap { start, end });
}

fn strip_cfws(src: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<Gap>), EmailError> {
    let mut kept = Vec::with_capacity(src.len());
    let mut offsets = Vec::with_capacity(src.len());
    let mut gaps = Vec::new();
    let mut p = 0usize;
    let mut first_is_quote = false;
    let mut quote_done = false;
    let mut escaped = false;

    while p < src.len() {
        if first_is_quote && !quote_done {
            kept.push(src[p]);
            offsets.push(p);
            if escaped {
                escaped = false;
            } else if src[p] == b'\\' {
                escaped = true;
            } else if src[p] == b'"' {
                quote_done = true;
            }
            p += 1;
            continue;
        }

        let c = src[p];
        if kept.is_empty() && !first_is_quote && c == b'"' {
            first_is_quote = true;
            kept.push(c);
            offsets.push(p);
            p += 1;
            continue;
        }

        if horizontal(c) {
            let begin = p;
            while p < src.len() && horizontal(src[p]) {
                p += 1;
            }
            append_gap(&mut gaps, begin, p);
            continue;
        }
        if c == b'\r' || c == b'\n' {
            let begin = p;
            p = folded_end(src, p).ok_or(EmailError::BadFold { at: p })?;
            append_gap(&mut gaps, begin, p);
            continue;
        }
        if c == b'(' {
            let begin = p;
            let outer = p;
            let mut stack = vec![p];
            p += 1;
            while p < src.len() {
                let x = src[p];
                if x == b'\\' {
                    let at = p;
                    match src.get(p + 1) {
                        Some(v) if (0x20..=0x7e).contains(v) => p += 2,
                        _ => return Err(EmailError::CommentEscape { at }),
                    }
                    continue;
                }
                if x == b'\r' || x == b'\n' {
                    p = folded_end(src, p).ok_or(EmailError::BadFold { at: p })?;
                    continue;
                }
                if x == b'(' {
                    if stack.len() == 3 {
                        return Err(EmailError::CommentDepth { at: p });
                    }
                    stack.push(p);
                    p += 1;
                    continue;
                }
                if x == b')' {
                    stack.pop();
                    p += 1;
                    if stack.is_empty() {
                        break;
                    }
                    continue;
                }
                if !(horizontal(x) || (0x20..=0x7e).contains(&x)) {
                    return Err(EmailError::CommentChar { at: p });
                }
                p += 1;
            }
            if !stack.is_empty() {
                return Err(EmailError::UnclosedComment { at: outer });
            }
            append_gap(&mut gaps, begin, p);
            continue;
        }
        if c == b')' {
            return Err(EmailError::UnexpectedClose { at: p });
        }
        kept.push(c);
        offsets.push(p);
        p += 1;
    }
    Ok((kept, offsets, gaps))
}

fn pos(offsets: &[usize], end: usize, i: usize) -> usize {
    offsets.get(i).copied().unwrap_or(end)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let src = input.as_bytes();
    if src.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(p) = src.iter().position(|x| *x >= 128) {
        return Err(EmailError::NonAscii { at: p });
    }

    let (b, offsets, gaps) = strip_cfws(src)?;
    for g in gaps {
        let at_edge = g.start == 0 || g.end == src.len();
        let adjacent_left_at = g.start > 0 && src[g.start - 1] == b'@';
        let adjacent_right_at = g.end < src.len() && src[g.end] == b'@';
        if !(at_edge || adjacent_left_at || adjacent_right_at) {
            return Err(EmailError::CfwsBoundary { at: g.start });
        }
    }
    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let close = if b[0] == b'"' {
        let mut q = 1usize;
        let mut found = None;
        while q < b.len() {
            match b[q] {
                b'\\' => q = (q + 2).min(b.len()),
                b'"' => {
                    found = Some(q);
                    break;
                }
                _ => q += 1,
            }
        }
        match found {
            Some(q) => Some(q),
            None => return Err(EmailError::UnclosedQuote { at: pos(&offsets, input.len(), 0) }),
        }
    } else {
        None
    };

    let mut separator = None;
    let mut second = None;
    let mut i = 0usize;
    while i < b.len() {
        if let Some(q) = close {
            if i <= q {
                i = q + 1;
                continue;
            }
        }
        if b[i] == b'@' {
            if separator.is_none() {
                separator = Some(i);
            } else if second.is_none() {
                second = Some(i);
                break;
            }
        }
        i += 1;
    }
    let at = match separator {
        None => return Err(EmailError::MissingAt),
        Some(v) => v,
    };
    if let Some(v) = second {
        return Err(EmailError::MultipleAt { at: pos(&offsets, input.len(), v) });
    }
    if at == 0 {
        return Err(EmailError::EmptyLocal { at: pos(&offsets, input.len(), at) });
    }

    let is_quoted = b[0] == b'"';
    let local;
    if !is_quoted {
        if at > 64 {
            return Err(EmailError::LocalTooLong { at: pos(&offsets, input.len(), 64) });
        }
        let mut prev_period = false;
        for k in 0..at {
            let c = b[k];
            if c == b'.' {
                if k == 0 || k + 1 == at || prev_period {
                    return Err(EmailError::LocalDot { at: pos(&offsets, input.len(), k) });
                }
                prev_period = true;
            } else {
                prev_period = false;
                if !(c.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&c)) {
                    return Err(EmailError::LocalChar { at: pos(&offsets, input.len(), k) });
                }
            }
        }
        local = String::from_utf8(b[..at].to_vec()).unwrap();
    } else {
        let q = close.unwrap();
        if q + 1 != at {
            return Err(EmailError::TrailingLocal { at: pos(&offsets, input.len(), q + 1) });
        }
        let mut value = Vec::new();
        let mut origin = Vec::new();
        let mut k = 1usize;
        while k < q {
            if b[k] == b'\\' {
                if k + 1 >= q || !matches!(b[k + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: offsets[k] });
                }
                value.push(b[k + 1]);
                origin.push(offsets[k]);
                k += 2;
            } else {
                if !(0x20..=0x7e).contains(&b[k]) {
                    return Err(EmailError::QuotedChar { at: offsets[k] });
                }
                value.push(b[k]);
                origin.push(offsets[k]);
                k += 1;
            }
        }
        if value.is_empty() {
            return Err(EmailError::EmptyLocal { at: offsets[q] });
        }
        if value.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origin[64] });
        }
        local = String::from_utf8(value).unwrap();
    }

    let ds = at + 1;
    let mut de = b.len();
    let removed_root = if de > ds && b[de - 1] == b'.' {
        de -= 1;
        Some(de)
    } else {
        None
    };
    if ds == de {
        return Err(EmailError::EmptyDomain { at: pos(&offsets, input.len(), ds) });
    }
    if de - ds > 253 {
        return Err(EmailError::DomainTooLong { at: pos(&offsets, input.len(), ds + 253) });
    }

    let mut bounds = Vec::new();
    let mut start = ds;
    let mut k = ds;
    while k <= de {
        if k == de || b[k] == b'.' {
            if start == k {
                let error_at = if start == de {
                    removed_root.map(|r| pos(&offsets, input.len(), r)).unwrap_or_else(|| pos(&offsets, input.len(), start))
                } else {
                    pos(&offsets, input.len(), start)
                };
                return Err(EmailError::EmptyLabel { at: error_at });
            }
            if k - start > 63 {
                return Err(EmailError::LabelTooLong { at: pos(&offsets, input.len(), start + 63) });
            }
            if b[start] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos(&offsets, input.len(), start) });
            }
            if b[k - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos(&offsets, input.len(), k - 1) });
            }
            for z in start..k {
                if !(b[z].is_ascii_alphanumeric() || b[z] == b'-') {
                    return Err(EmailError::DomainChar { at: pos(&offsets, input.len(), z) });
                }
            }
            bounds.push((start, k));
            start = k + 1;
        }
        k += 1;
    }
    if bounds.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: pos(&offsets, input.len(), ds) });
    }
    let (last_s, last_e) = bounds[bounds.len() - 1];
    if b[last_s..last_e].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos(&offsets, input.len(), last_s) });
    }

    let mut domain = String::with_capacity(de - ds);
    for &x in &b[ds..de] {
        domain.push(x.to_ascii_lowercase() as char);
    }
    Ok(Address { local, domain, quoted: is_quoted })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {

        return Err(EmailError::EmptyInput);
    }
    for (i, &b) in bytes.iter().enumerate() {
        if b >= 128 {
            return Err(EmailError::NonAscii { at: i });
        }
    }

    let n = bytes.len();
    let mut clean = Vec::<u8>::with_capacity(n);
    let mut map = Vec::<usize>::with_capacity(n);
    let mut runs = Vec::<Run>::new();
    let mut i = 0usize;
    let mut protected = false;
    let mut quote_closed = false;
    let mut quote_escape = false;
    let mut comment_depth = 0usize;
    let mut comment_outer = 0usize;
    let mut comment_stack = [0usize; 3];

    while i < n {
        let b = bytes[i];

        if protected {
            clean.push(b);
            map.push(i);
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                protected = false;
                quote_closed = true;

            continue;

            if b == b'\\' {
                if i + 1 >= n || !(0x20..=0x7e).contains(&bytes[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if b == b'\r' && i + 2 < n && bytes[i + 1] == b'\n' && is_hws(bytes[i + 2]) {
                    i += 3;
                    while i < n && is_hws(bytes[i]) {
                        i += 1;
                    }
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_stack[comment_depth] = i;
                comment_depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                comment_depth -= 1;
                i += 1;
                continue;
            }
            if b == b'\t' || (0x20..=0x7e).contains(&b) {
                i += 1;
                continue;
            }
            return Err(EmailError::CommentChar { at: i });
        }

        if !quote_closed && clean.is_empty() && b == b'"' {
            protected = true;
            clean.push(b);
            map.push(i);
            i += 1;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'(' || b == b'\r' || b == b'\n' {
            let start = i;
            loop {
                if i >= n {
                    break;
                }
                let x = bytes[i];
                if x == b' ' || x == b'\t' {
                    i += 1;
                    continue;
                }
                if x == b'\r' || x == b'\n' {
                    if x == b'\r' && i + 2 < n && bytes[i + 1] == b'\n' && is_hws(bytes[i + 2]) {
                        i += 3;
                        while i < n && is_hws(bytes[i]) {
                            i += 1;
                        }
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if x == b'(' {
                    comment_outer = i;
                    comment_stack[0] = i;
                    comment_depth = 1;
                    i += 1;
                    while i < n && comment_depth > 0 {
                        let c = bytes[i];
                        if c == b'\\' {
                            if i + 1 >= n || !(0x20..=0x7e).contains(&bytes[i + 1]) {
                                return Err(EmailError::CommentEscape { at: i });
                            }
                            i += 2;
                            continue;
                        }
                        if c == b'\r' || c == b'\n' {
                            if c == b'\r' && i + 2 < n && bytes[i + 1] == b'\n' && is_hws(bytes[i + 2]) {
                                i += 3;
                                while i < n && is_hws(bytes[i]) {
                                    i += 1;
                                }
                                continue;
                            }
                            return Err(EmailError::BadFold { at: i });
                        }
                        if c == b'(' {
                            if comment_depth == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            comment_stack[comment_depth] = i;
                            comment_depth += 1;
                            i += 1;
                            continue;
                        }
                        if c == b')' {
                            comment_depth -= 1;
                            i += 1;
                            continue;
                        }
                        if c == b'\t' || (0x20..=0x7e).contains(&c) {
                            i += 1;
                            continue;
                        }
                        return Err(EmailError::CommentChar { at: i });
                    }
                    if comment_depth != 0 {
                        return Err(EmailError::UnclosedComment { at: comment_outer });
                    }
                    continue;
                }
                break;
            }
            runs.push(Run { start, end: i });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        clean.push(b);
        map.push(i);
        i += 1;
    }

    if comment_depth > 0 {
        return Err(EmailError::UnclosedComment { at: comment_outer });
    }

    let mut at_positions = Vec::new();
    let mut quote_end = None;
    if clean.first() == Some(&b'"') {
        let mut j = 1usize;
        let mut closed = false;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j += 2;
                continue;
            }
            if clean[j] == b'"' {
                quote_end = Some(j);
                closed = true;
                break;
            }
            j += 1;
        }
        if !closed {
            return Err(EmailError::UnclosedQuote { at: map[0] });
        }
    }
    let protected_end = quote_end.map(|x| x + 1).unwrap_or(0);
    for (j, &b) in clean.iter().enumerate() {
        if b == b'@' && j >= protected_end {
            at_positions.push(j);
        }
    }
    if at_positions.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if at_positions.len() >= 2 {
        return Err(EmailError::MultipleAt { at: map[at_positions[1]] });
    }
    let sep = at_positions[0];
    let sep_orig = map[sep];

    for run in &runs {
        let before = map.iter().position(|&p| p >= run.start).map(|k| k.checked_sub(1)).flatten();
        let after = map.iter().position(|&p| p >= run.end);
        let boundary = before.is_none()
            || after.is_none()
            || before.map(|k| clean[k] == b'@').unwrap_or(false)
            || after.map(|k| clean[k] == b'@').unwrap_or(false);
        if !boundary {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep_orig });
    }

    let local_bytes = &clean[..sep];
    let local_map = &map[..sep];
    let mut local = String::new();
    let quoted = local_bytes.first() == Some(&b'"');

    if quoted {
        let end = quote_end.unwrap_or(0);
        if end + 1 != sep {
            let at = if end + 1 < local_map.len() { local_map[end + 1] } else { sep_orig };
            return Err(EmailError::TrailingLocal { at });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1usize;
        while j < end {
            let b = local_bytes[j];
            if b == b'\\' {
                if j + 1 >= end || !matches!(local_bytes[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: local_map[j] });
                }
                decoded.push(local_bytes[j + 1]);
                origins.push(local_map[j]);
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: local_map[j] });
                }
                decoded.push(b);
                origins.push(local_map[j]);
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: local_map[end] });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        local = String::from_utf8(decoded).unwrap();
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: local_map[64] });
        }
        let mut prev_dot = false;
        for (j, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || prev_dot {
                    return Err(EmailError::LocalDot { at: local_map[j] });
                }
                prev_dot = true;
            } else {
                prev_dot = false;
                let allowed = b.is_ascii_alphanumeric()
                    || b"!#$%&'*+-/=?^_`{|}~".contains(&b);
                if !allowed {
                    return Err(EmailError::LocalChar { at: local_map[j] });
                }
            }
        }
        local = String::from_utf8(local_bytes.to_vec()).unwrap();
    }

    let domain_start_index = sep + 1;
    let domain_start = if domain_start_index < map.len() {
        map[domain_start_index]
    } else {
        input.len()
    };
    let mut dom_end = clean.len();
    let mut root_pos = None;
    if dom_end > domain_start_index && clean[dom_end - 1] == b'.' {
        dom_end -= 1;
        root_pos = Some(map[dom_end]);
    }
    if dom_end == domain_start_index {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    let domain_len = dom_end - domain_start_index;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong { at: map[domain_start_index + 253] });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start_index;
    for k in domain_start_index..=dom_end {
        if k == dom_end || clean[k] == b'.' {
            if start == k {
                let pos = if k == dom_end {
                    root_pos.unwrap_or_else(|| if k < map.len() { map[k] } else { input.len() })
                } else {
                    map[k]
                };
                return Err(EmailError::EmptyLabel { at: pos });
            }
            let len = k - start;
            if len > 63 {
                return Err(EmailError::LabelTooLong { at: map[start + 63] });
            }
            if clean[start] == b'-' {
                return Err(EmailError::LabelHyphen { at: map[start] });
            }
            if clean[k - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: map[k - 1] });
            }
            for p in start..k {
                if !(clean[p].is_ascii_alphanumeric() || clean[p] == b'-') {
                    return Err(EmailError::DomainChar { at: map[p] });
                }
            }
            labels.push((start, k));
            start = k + 1;
        }
    }
    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: map[last_start] });
    }
    let domain = clean[domain_start_index..dom_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address { local, domain, quoted })
}

pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone)]
struct Removed {
    start: usize,
    end: usize,
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let b = input.as_bytes();
    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = b.iter().position(|x| *x >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let mut out = Vec::with_capacity(b.len());
    let mut pos = Vec::with_capacity(b.len());
    let mut removed = Vec::<Removed>::new();
    let mut i = 0;
    let mut quote_mode = false;
    let mut quote_escaped = false;
    let mut quote_seen_closed = false;

    while i < b.len() {
        if quote_mode {
            out.push(b[i]);
            pos.push(i);
            if quote_escaped {
                quote_escaped = false;
            } else if b[i] == b'\\' {
                quote_escaped = true;
            } else if b[i] == b'"' {
                quote_mode = false;
                quote_seen_closed = true;
            }
            i += 1;
            continue;
        }

        if !quote_seen_closed && out.is_empty() && b[i] == b'"' {
            quote_mode = true;
            out.push(b[i]);
            pos.push(i);
            i += 1;
            continue;
        }

        let start = i;
        let is_run_start = b[i] == b' ' || b[i] == b'\t' || b[i] == b'(' || b[i] == b'\r' || b[i] == b'\n';
        if is_run_start {
            loop {
                if i == b.len() {
                    break;
                }
                if b[i] == b' ' || b[i] == b'\t' {
                    i += 1;
                    continue;
                }
                if b[i] == b'\r' || b[i] == b'\n' {
                    if b[i] == b'\r' && i + 2 < b.len() && b[i + 1] == b'\n' && (b[i + 2] == b' ' || b[i + 2] == b'\t') {
                        i += 3;
                        while i < b.len() && (b[i] == b' ' || b[i] == b'\t') {
                            i += 1;
                        }
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if b[i] == b'(' {
                    let outer = i;
                    let mut depth = 1usize;
                    i += 1;
                    while i < b.len() && depth != 0 {
                        let c = b[i];
                        if c == b'\\' {
                            if i + 1 >= b.len() || !(0x20..=0x7e).contains(&b[i + 1]) {
                                return Err(EmailError::CommentEscape { at: i });
                            }
                            i += 2;
                        } else if c == b'\r' || c == b'\n' {
                            if c == b'\r' && i + 2 < b.len() && b[i + 1] == b'\n' && (b[i + 2] == b' ' || b[i + 2] == b'\t') {
                                i += 3;
                                while i < b.len() && (b[i] == b' ' || b[i] == b'\t') {
                                    i += 1;
                                }
                            } else {
                                return Err(EmailError::BadFold { at: i });
                            }
                        } else if c == b'(' {
                            if depth == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            depth += 1;
                            i += 1;
                        } else if c == b')' {
                            depth -= 1;
                            i += 1;
                        } else if c == b'\t' || (0x20..=0x7e).contains(&c) {
                            i += 1;
                        } else {
                            return Err(EmailError::CommentChar { at: i });
                        }
                    }
                    if depth != 0 {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                    continue;
                }
                break;
            }
            removed.push(Removed { start, end: i });
            continue;
        }

        if b[i] == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        out.push(b[i]);
        pos.push(i);
        i += 1;
    }

    if out.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut ranges = Vec::<(usize, usize)>::new();
    if out.first() == Some(&b'"') {
        let mut q = 1;
        let mut found = None;
        while q < out.len() {
            if out[q] == b'\\' {
                q += 2;
            } else if out[q] == b'"' {
                found = Some(q);
                break;
            } else {
                q += 1;
            }
        }
        let qend = found.ok_or(EmailError::UnclosedQuote { at: pos[0] })?;
        ranges.push((0, qend + 1));
    }
    let protect = ranges.first().map(|x| x.1).unwrap_or(0);
    let ats: Vec<usize> = out.iter().enumerate().filter_map(|(j, c)| if *c == b'@' && j >= protect { Some(j) } else { None }).collect();
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt { at: pos[ats[1]] });
    }
    let sep = ats[0];

    for r in &removed {
        let before = pos.iter().rposition(|p| *p < r.start);
        let after = pos.iter().position(|p| *p >= r.end);
        if before.is_none()
            || after.is_none()
            || before.map(|x| out[x] == b'@').unwrap_or(false)
            || after.map(|x| out[x] == b'@').unwrap_or(false)
        {
            continue;
        }
        return Err(EmailError::CfwsBoundary { at: r.start });
    }

    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: pos[sep] });
    }
    let local_b = &out[..sep];
    let qlocal = local_b[0] == b'"';
    let local = if qlocal {
        let mut j = 1;
        let mut close = None;
        while j < local_b.len() {
            if local_b[j] == b'\\' {
                j += 2;
            } else if local_b[j] == b'"' {
                close = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        let close = close.ok_or(EmailError::UnclosedQuote { at: pos[0] })?;
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal { at: pos[close + 1] });
        }
        let mut val = Vec::new();
        let mut origins = Vec::new();
        j = 1;
        while j < close {
            let c = local_b[j];
            if c == b'\\' {
                if j + 1 >= close || !matches!(local_b[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: pos[j] });
                }
                val.push(local_b[j + 1]);
                origins.push(pos[j]);
                j += 2;
            } else if !(0x20..=0x7e).contains(&c) {
                return Err(EmailError::QuotedChar { at: pos[j] });
            } else {
                val.push(c);
                origins.push(pos[j]);
                j += 1;
            }
        }
        if val.is_empty() {
            return Err(EmailError::EmptyLocal { at: pos[close] });
        }
        if val.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        String::from_utf8(val).unwrap()
    } else {
        if local_b.len() > 64 {
            return Err(EmailError::LocalTooLong { at: pos[64] });
        }
        let mut previous_dot = false;
        for (j, c) in local_b.iter().enumerate() {
            if *c == b'.' {
                if j == 0 || j + 1 == local_b.len() || previous_dot {
                    return Err(EmailError::LocalDot { at: pos[j] });
                }
                previous_dot = true;
            } else {
                previous_dot = false;
                if !(c.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(c)) {
                    return Err(EmailError::LocalChar { at: pos[j] });
                }
            }
        }
        String::from_utf8(local_b.to_vec()).unwrap()
    };

    let domain_start_idx = sep + 1;
    let domain_start = pos.get(domain_start_idx).copied().unwrap_or(input.len());
    let mut domain_end = out.len();
    let root = if domain_end > domain_start_idx && out[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(pos[domain_end])
    } else {
        None
    };
    if domain_end == domain_start_idx {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_end - domain_start_idx > 253 {
        return Err(EmailError::DomainTooLong { at: pos[domain_start_idx + 253] });
    }

    let mut labels = Vec::new();
    let mut start = domain_start_idx;
    for k in domain_start_idx..=domain_end {
        if k == domain_end || out[k] == b'.' {
            if start == k {
                let at = if k == domain_end {
                    root.unwrap_or(input.len())
                } else {
                    pos[k]
                };
                return Err(EmailError::EmptyLabel { at });
            }
            if k - start > 63 {
                return Err(EmailError::LabelTooLong { at: pos[start + 63] });
            }
            if out[start] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos[start] });
            }
            if out[k - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos[k - 1] });
            }
            for p in start..k {
                if !(out[p].is_ascii_alphanumeric() || out[p] == b'-') {
                    return Err(EmailError::DomainChar { at: pos[p] });
                }
            }
            labels.push((start, k));
            start = k + 1;
        }
    }
    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    let (ls, le) = labels[labels.len() - 1];
    if out[ls..le].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos[ls] });
    }
    let domain = out[domain_start_idx..domain_end]
        .iter()
        .map(|c| c.to_ascii_lowercase() as char)
        .collect();
    Ok(Address { local, domain, quoted: qlocal })
}
