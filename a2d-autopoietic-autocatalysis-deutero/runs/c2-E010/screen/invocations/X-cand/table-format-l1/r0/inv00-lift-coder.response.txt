- Empty input succeeds before any validation.
- Empty rows are checked across the whole table before ragged rows.
- Character validation scans original cells in row-major order, before trimming.
- Only trailing ASCII spaces are trimmed; width is measured in bytes and has a minimum of 1.
- Widths above 32 report the lowest offending column after all character checks.
- Borders and data rows use exact LF-terminated formatting.

### Solution A
```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
}

pub fn render(rows: &[Vec<String>]) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let columns = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != columns {
            return Err(TableError::Ragged {
                row,
                expected: columns,
                actual: cells.len(),
            });
        }
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if !(' '..='~').contains(&ch) {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }
        }
    }

    let mut widths = vec![1usize; columns];
    for cells in rows {
        for (column, cell) in cells.iter().enumerate() {
            let trimmed_len = cell.trim_end_matches(' ').len();
            widths[column] = widths[column].max(trimmed_len);
        }
    }

    for (column, &width) in widths.iter().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let mut output = String::new();

    output.push('+');
    for &width in &widths {
        output.extend(std::iter::repeat_n('-', width + 2));
        output.push('+');
    }
    output.push('\n');

    for cells in rows {
        output.push('|');
        for (column, cell) in cells.iter().enumerate() {
            let content = cell.trim_end_matches(' ');
            output.push(' ');
            output.push_str(content);
            output.extend(std::iter::repeat_n(' ', widths[column] - content.len()));
            output.push(' ');
            output.push('|');
        }
        output.push('\n');
    }

    output.push('+');
    for &width in &widths {
        output.extend(std::iter::repeat_n('-', width + 2));
        output.push('+');
    }
    output.push('\n');

    Ok(output)
}
```

### Solution B
```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
}

pub fn render(rows: &[Vec<String>]) -> Result<String, TableError> {
    if rows.len() == 0 {
        return Ok(String::new());
    }

    let mut first_empty = None;
    for (i, row) in rows.iter().enumerate() {
        if row.len() == 0 {
            first_empty = Some(i);
            break;
        }
    }
    if let Some(row) = first_empty {
        return Err(TableError::EmptyRow { row });
    }

    let expected = rows[0].len();
    for (i, row) in rows.iter().enumerate() {
        let actual = row.len();
        if actual != expected {
            return Err(TableError::Ragged { row: i, expected, actual });
        }
    }

    for (i, row) in rows.iter().enumerate() {
        for (j, cell) in row.iter().enumerate() {
            for (byte, value) in cell.bytes().enumerate() {
                if !(0x20..=0x7e).contains(&value) {
                    return Err(TableError::InvalidChar {
                        row: i,
                        column: j,
                        byte,
                    });
                }
            }
        }
    }

    let mut widths = vec![1usize; expected];
    for row in rows.iter() {
        for (j, cell) in row.iter().enumerate() {
            let mut length = cell.len();
            while length > 0 && cell.as_bytes()[length - 1] == b' ' {
                length -= 1;
            }
            if length > widths[j] {
                widths[j] = length;
            }
        }
    }

    for (column, width) in widths.iter().copied().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let border = |widths: &[usize]| {
        let mut line = String::from("+");
        for width in widths {
            for _ in 0..(width + 2) {
                line.push('-');
            }
            line.push('+');
        }
        line.push('\n');
        line
    };

    let mut result = border(&widths);
    for row in rows {
        result.push('|');
        for (column, cell) in row.iter().enumerate() {
            let content = cell.trim_end_matches(' ');
            let content_len = content.len();
            result.push(' ');
            result.push_str(content);
            for _ in 0..(widths[column] - content_len) {
                result.push(' ');
            }
            result.push(' ');
            result.push('|');
        }
        result.push('\n');
    }
    result.push_str(&border(&widths));
    Ok(result)
}
```
