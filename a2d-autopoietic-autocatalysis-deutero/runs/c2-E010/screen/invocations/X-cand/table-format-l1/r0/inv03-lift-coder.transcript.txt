pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
}

pub fn render(rows: &[Vec<String>]) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let expected = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != expected {
            return Err(TableError::Ragged {
                row,
                expected,
                actual: cells.len(),
            });
        }
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if !(' '..='~').contains(&ch) {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }
        }
    }

    let mut widths = vec![1usize; expected];
    for cells in rows {
        for (column, cell) in cells.iter().enumerate() {
            let trimmed_len = cell.trim_end_matches(' ').len();
            widths[column] = widths[column].max(trimmed_len);
        }
    }

    for (column, &width) in widths.iter().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let mut out = String::new();

    for &width in &widths {
        out.push('+');
        let border_width = width + 2;
        for _ in 0..border_width {
            out.push('-');
        }
    }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
}

pub fn render(rows: &[Vec<String>]) -> Result<String, TableError> {
    if rows.len() == 0 {
        return Ok(String::new());
    }

    let mut first_empty = None;
    for (i, row) in rows.iter().enumerate() {
        if row.len() == 0 {
            first_empty = Some(i);
            break;
        }
    }
    if let Some(row) = first_empty {
        return Err(TableError::EmptyRow { row });
    }

    let n = rows[0].len();
    let mismatch = rows.iter().enumerate().find(|(_, row)| row.len() != n);
    if let Some((row, cells)) = mismatch {
        return Err(TableError::Ragged {
            row,
            expected: n,
            actual: cells.len(),
        });
    }

    let mut trimmed: Vec<Vec<&str>> = Vec::with_capacity(rows.len());
    for (row, cells) in rows.iter().enumerate() {
        let mut trimmed_row = Vec::with_capacity(n);
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if ch < ' ' || ch > '~' {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }
            trimmed_row.push(cell.trim_end_matches(' '));
        }
        trimmed.push(trimmed_row);
    }

    let mut widths = vec![1usize; n];
    for row in &trimmed {
        for (column, cell) in row.iter().enumerate() {
            if cell.len() > widths[column] {
                widths[column] = cell.len();
            }
        }
    }

    for column in 0..n {
        if widths[column] > 32 {
            return Err(TableError::TooWide {
                column,
                width: widths[column],
            });
        }
    }

    let mut out = String::new();
    let mut border = String::new();
    for &width in &widths {
        border.push('+');
        border.extend(std::iter::repeat('-').take(width + 2));
    }
    border.push_str("+\n");
    out.push_str(&border);

    for row in &trimmed {
        out.push('|');
        for (column, cell) in row.iter().enumerate() {
            out.push(' ');
            out.push_str(cell);
            out.extend(std::iter::repeat(' ').take(widths[column] - cell.len()));
            out.push(' ');
            out.push('|');
        }
        out.push('\n');
    }

    out.push_str(&border);
    Ok(out)
}
