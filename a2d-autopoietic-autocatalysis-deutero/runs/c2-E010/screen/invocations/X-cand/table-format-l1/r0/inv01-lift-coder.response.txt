- Check order: empty table, all empty rows, ragged rows, invalid characters, then width errors. I follow that precedence.
- Validate original cells before trimming trailing ASCII spaces; offsets are UTF-8 byte offsets.
- Width is the maximum trimmed byte length, with a minimum of 1; width 32 is allowed.
- Output uses LF after every line, with exactly one top and bottom border.
- Empty and all-space cells are valid and render as empty content with the required padding.
- No input-size limit is specified; formatting uses checked width arithmetic to avoid overflow.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
}

pub fn render(rows: &[Vec<String>]) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let columns = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != columns {
            return Err(TableError::Ragged {
                row,
                expected: columns,
                actual: cells.len(),
            });
        }
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if !(' '..='~').contains(&ch) {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }
        }
    }

    let trimmed: Vec<Vec<&str>> = rows
        .iter()
        .map(|cells| {
            cells
                .iter()
                .map(|cell| cell.trim_end_matches(' '))
                .collect()
        })
        .collect();

    let mut widths = vec![1usize; columns];
    for cells in &trimmed {
        for (column, cell) in cells.iter().enumerate() {
            widths[column] = widths[column].max(cell.len());
        }
    }

    for (column, &width) in widths.iter().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let mut output = String::new();

    let append_border = |output: &mut String| {
        output.push('+');
        for &width in &widths {
            output.extend(std::iter::repeat_n('-', width + 2));
            output.push('+');
        }
        output.push('\n');
    };

    append_border(&mut output);
    for cells in &trimmed {
        output.push('|');
        for (column, cell) in cells.iter().enumerate() {
            output.push(' ');
            output.push_str(cell);
            output.extend(std::iter::repeat_n(' ', widths[column] - cell.len()));
            output.push(' ');
            output.push('|');
        }
        output.push('\n');
    }
    append_border(&mut output);

    Ok(output)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
}

pub fn render(rows: &[Vec<String>]) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    if let Some(row) = rows.iter().position(Vec::is_empty) {
        return Err(TableError::EmptyRow { row });
    }

    let expected = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != expected {
            return Err(TableError::Ragged {
                row,
                expected,
                actual: cells.len(),
            });
        }
    }

    let mut widths = vec![1usize; expected];
    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if !(ch as u32 >= 0x20 && ch as u32 <= 0x7e) {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }

            let trimmed_len = cell.trim_end_matches(' ').len();
            widths[column] = widths[column].max(trimmed_len);
        }
    }

    for (column, width) in widths.iter().copied().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let mut result = String::new();
    for &width in &widths {
        result.push('+');
        result.extend(std::iter::repeat_n('-', width + 2));
    }
    result.push_str("+\n");

    for cells in rows {
        result.push('|');
        for (column, cell) in cells.iter().enumerate() {
            let content = cell.trim_end_matches(' ');
            result.push(' ');
            result.push_str(content);
            result.extend(std::iter::repeat_n(' ', widths[column] - content.len()));
            result.push_str(" |");
        }
        result.push('\n');
    }

    for &width in &widths {
        result.push('+');
        result.extend(std::iter::repeat_n('-', width + 2));
    }
    result.push_str("+\n");

    Ok(result)
}
```
