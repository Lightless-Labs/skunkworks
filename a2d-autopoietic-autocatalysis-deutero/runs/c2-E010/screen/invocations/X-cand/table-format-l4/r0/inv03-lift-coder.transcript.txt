pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment { Left, Right, Center }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow { Clip, Wrap }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode { Grid, Compact, Markdown }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,

struct Token {
    text: String,
    width: usize,
}

fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (i, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: i });
        }
    }

    let n = rows[0].len();
    for (i, row) in rows.iter().enumerate() {
        if row.len() != n {
            return Err(TableError::Ragged {
                row: i,
                expected: n,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let (overhead, min_col) = match options.mode {
        Mode::Compact => (2usize.saturating_mul(n - 1), 1usize),
        Mode::Grid | Mode::Markdown => (3usize.saturating_mul(n).saturating_add(1), 1usize),
    };
    let min_col = if matches!(options.mode, Mode::Markdown) { 3 } else { min_col };
    let minimum = overhead.saturating_add(n.saturating_mul(min_col));
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    let mut normalized: Vec<Vec<Vec<Token>>> = Vec::with_capacity(rows.len());
    for (ri, row) in rows.iter().enumerate() {
        let mut out_row = Vec::with_capacity(n);

            let mut chars = Vec::new();

            while byte < cell.len() {
                let ch = cell[byte..].chars().next().unwrap();
                if ch == '\r' {
                    let next_byte = byte + 1;
                    if cell[next_byte..].starts_with('\n') {
                        chars.push('\n');
                        byte = next_byte + 1;
                        continue;
                    }
                    return Err(TableError::InvalidChar { row: ri, column: ci, byte });
                }
                if (ch.is_control() && ch != '\n') || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar { row: ri, column: ci, byte });
                }
                chars.push(ch);
                byte += ch.len_utf8();
            }

            let normalized_cell: String = chars.into_iter().collect();
            let mut logical = Vec::new();
            for line in normalized_cell.split('\n') {
                let trimmed = line.trim_end_matches(' ');
                let mut tokens = Vec::new();
                for ch in trimmed.chars() {
                    if matches!(options.mode, Mode::Markdown) && ch == '|' {
                        tokens.push(Token { text: "\\|".to_owned(), width: 2 });
                    } else if matches!(options.mode, Mode::Markdown) && ch == '\\' {
                        tokens.push(Token { text: "\\\\".to_owned(), width: 2 });
                    } else {
                        tokens.push(Token { text: ch.to_string(), width: 1 });
                    }
                }
                logical.push(tokens);
            }
            out_row.push(logical);
        }
        normalized.push(out_row);
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let mut widths = vec![min_col; n];
    for row in &normalized {
        for (ci, cell) in row.iter().enumerate() {
            for line in cell {
                let w: usize = line.iter().map(|t| t.width).sum();
                widths[ci] = widths[ci].max(w);
            }
        }
    }

    if options.max_width != 0 {
        let mut total = overhead.saturating_add(widths.iter().sum::<usize>());
        while total > options.max_width {
            let candidate = (0..n)
                .filter(|&i| widths[i] > min_col)
                .max_by_key(|&i| (widths[i], i));
            if let Some(i) = candidate {
                widths[i] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<Vec<Token>> {
        if tokens.is_empty() {
            return vec![Vec::new()];
        }
        let total: usize = tokens.iter().map(|t| t.width).sum();
        if matches!(overflow, Overflow::Clip) {
            if total <= width {
                return vec![tokens.to_vec()];
            }
            let mut result = Vec::new();
            let limit = width.saturating_sub(1);
            let mut used = 0;
            for token in tokens {
                if used + token.width > limit { break; }
                used += token.width;
                result.push(token.clone());
            }
            result.push(Token { text: "…".to_owned(), width: 1 });
            vec![result]
        } else {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                if end == start {
                    end += 1;
                }
                result.push(tokens[start..end].to_vec());
                start = end;
            }
            result
        }
    }

    fn token_width(tokens: &[Token]) -> usize {
        tokens.iter().map(|t| t.width).sum()
    }
    fn token_text(tokens: &[Token]) -> String {
        tokens.iter().map(|t| t.text.as_str()).collect()
    }
    fn aligned(tokens: &[Token], width: usize, alignment: &Alignment, omit_right: bool) -> String {
        let text = token_text(tokens);
        let used = token_width(tokens);
        let extra = width.saturating_sub(used);
        let (left, right) = match alignment {
            Alignment::Left => (0, extra),
            Alignment::Right => (extra, 0),
            Alignment::Center => (extra / 2, extra - extra / 2),
        };
        format!("{}{}{}", " ".repeat(left), text, " ".repeat(if omit_right { 0 } else { right }))
    }

    fn grid_border(widths: &[usize]) -> String {
        let mut s = String::from("+");
        for &w in widths {
            s.push_str(&"-".repeat(w + 2));
            s.push('+');
        }
        s.push('\n');
        s
    }

    let mut output = String::new();
    if matches!(options.mode, Mode::Grid) {
        output.push_str(&grid_border(&widths));
    }

    for (ri, row) in normalized.iter().enumerate() {
        let mut cell_frags = Vec::with_capacity(n);
        let mut height = 1;
        for ci in 0..n {
            let mut all = Vec::new();
            for logical_line in &row[ci] {
                all.extend(fragments(logical_line, widths[ci], &overflows[ci]));
            }
            height = height.max(all.len());
            cell_frags.push(all);
        }

        for fi in 0..height {
            match options.mode {
                Mode::Grid | Mode::Markdown => output.push('|'),
                Mode::Compact => {}
            }
            for ci in 0..n {
                let empty = Vec::new();
                let frag = cell_frags[ci].get(fi).unwrap_or(&empty);
                let omit_right = matches!(options.mode, Mode::Compact) && ci + 1 == n;
                let text = aligned(frag, widths[ci], &alignments[ci], omit_right);
                match options.mode {
                    Mode::Grid | Mode::Markdown => {
                        output.push(' ');
                        output.push_str(&text);
                        output.push_str(" |");
                    }
                    Mode::Compact => {
                        if ci > 0 { output.push_str("  "); }
                        output.push_str(&text);
                    }
                }
            }
            output.push('\n');
        }

        if matches!(options.mode, Mode::Grid) {
            if options.header && ri == 0 && rows.len() > 1 {
                output.push_str(&grid_border(&widths));
            }
        }
        if matches!(options.mode, Mode::Markdown) && ri == 0 {
            output.push('|');
            for ci in 0..n {
                output.push(' ');
                let w = widths[ci];
                match alignments[ci] {
                    Alignment::Left => {
                        output.push(':');
                        output.push_str(&"-".repeat(w - 1));
                    }
                    Alignment::Right => {
                        output.push_str(&"-".repeat(w - 1));
                        output.push(':');
                    }
                    Alignment::Center => {
                        output.push(':');
                        output.push_str(&"-".repeat(w - 2));
                        output.push(':');
                    }
                }
                output.push_str(" |");
            }
            output.push('\n');
        }
    }

    Ok(output)
}

struct Tok { s: String, w: usize }

fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() { return Ok(String::new()); }
    for (r, row) in rows.iter().enumerate() {
        if row.is_empty() { return Err(TableError::EmptyRow { row: r }); }
    }
    let n = rows[0].len();
    for (r, row) in rows.iter().enumerate() {
        if row.len() != n {
            return Err(TableError::Ragged { row: r, expected: n, actual: row.len() });
        }
    }
    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount { expected: n, actual: options.alignments.len() });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount { expected: n, actual: options.overflow.len() });
    }
    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let (overhead, floor) = match options.mode {
        Mode::Compact => (2 * (n - 1), 1),
        Mode::Grid => (3 * n + 1, 1),
        Mode::Markdown => (3 * n + 1, 3),
    };
    let minimum = overhead + n * floor;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow { minimum, provided: options.max_width });
    }

    let mut table: Vec<Vec<Vec<Vec<Tok>>>> = Vec::new();
    for (ri, row) in rows.iter().enumerate() {
        let mut tr = Vec::new();
        for (ci, cell) in row.iter().enumerate() {
            let mut normalized = String::new();
            let mut it = cell.char_indices().peekable();
            while let Some((byte, ch)) = it.next() {
                if ch == '\r' {
                    if matches!(it.peek(), Some((_, '\n'))) {
                        it.next();
                        normalized.push('\n');

                        return Err(TableError::InvalidChar { row: ri, column: ci, byte });
                    }
                } else if (ch.is_control() && ch != '\n') || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar { row: ri, column: ci, byte });
                } else {
                    normalized.push(ch);
                }
            }

            for source_line in normalized.split('\n') {
                let line = source_line.trim_end_matches(' ');

                for ch in line.chars() {
                    if options.mode == Mode::Markdown && (ch == '|' || ch == '\\') {
                        let mut s = String::new();
                        s.push('\\'); s.push(ch);
                        ts.push(Tok { s, w: 2 });
                    } else {
                        ts.push(Tok { s: ch.to_string(), w: 1 });
                    }
                }
                lines.push(ts);
            }
            tr.push(lines);
        }
        table.push(tr);

    let overs = if options.overflow.is_empty() { vec![Overflow::Wrap; n] } else { options.overflow.clone() };

                    pick = Some(i);
                }
            }
            match pick {
                Some(i) => widths[i] -= 1,
                None => break,
            }
        }

    fn split<'a>(tokens: &'a [Tok], width: usize, mode: &Overflow) -> Vec<Vec<Tok>> {
        if tokens.is_empty() { return vec![Vec::new()]; }
        if *mode == Overflow::Clip {
            let total: usize = tokens.iter().map(|t| t.w).sum();
            if total <= width { return vec![tokens.to_vec()]; }
            let mut piece = Vec::new();
            let mut used = 0;
            for t in tokens {
                if used + t.w > width - 1 { break; }
                used += t.w;
                piece.push(t.clone());
            }
            piece.push(Tok { s: "…".to_string(), w: 1 });
            return vec![piece];
        }
        let mut result = Vec::new();
        let mut i = 0;
        while i < tokens.len() {
            let mut j = i;
            let mut used = 0;
            while j < tokens.len() && used + tokens[j].w <= width {
                used += tokens[j].w;
                j += 1;
            }
            result.push(tokens[i..j].to_vec());
            i = j;
        }
        result
    }
    fn count(ts: &[Tok]) -> usize { ts.iter().map(|t| t.w).sum() }
    fn txt(ts: &[Tok]) -> String { ts.iter().map(|t| t.s.as_str()).collect() }
    fn field(ts: &[Tok], w: usize, a: &Alignment, compact_last: bool) -> String {
        let used = count(ts);
        let extra = w - used;
        let (l, r) = match a {
            Alignment::Left => (0, extra),
            Alignment::Right => (extra, 0),
            Alignment::Center => (extra / 2, extra - extra / 2),
        };
        format!("{}{}{}", " ".repeat(l), txt(ts), " ".repeat(if compact_last { 0 } else { r }))
    }
    fn border(widths: &[usize]) -> String {

        for w in widths { s.push_str(&"-".repeat(w + 2)); s.push('+'); }
        s.push('\n');
        s
    }

    let mut out = String::new();
    if options.mode == Mode::Grid { out.push_str(&border(&widths)); }
    for (ri, row) in table.iter().enumerate() {
        let mut cells = Vec::new();
        let mut h = 1;
        for c in 0..n {
            let mut fs = Vec::new();
            for line in &row[c] { fs.extend(split(line, widths[c], &overs[c])); }
            h = h.max(fs.len());
            cells.push(fs);
        }
        for k in 0..h {
            if options.mode != Mode::Compact { out.push('|'); }
            for c in 0..n {
                let empty = Vec::new();
                let ts = cells[c].get(k).unwrap_or(&empty);
                let value = field(ts, widths[c], &aligns[c], options.mode == Mode::Compact && c + 1 == n);
                if options.mode == Mode::Compact {
                    if c > 0 { out.push_str("  "); }
                    out.push_str(&value);
                } else {
                    out.push(' ');
                    out.push_str(&value);
                    out.push_str(" |");
                }
            }
            out.push('\n');
        }
        if options.mode == Mode::Grid && options.header && ri == 0 && rows.len() > 1 {
            out.push_str(&border(&widths));
        }
        if options.mode == Mode::Markdown && ri == 0 {
            out.push('|');
            for c in 0..n {
                out.push(' ');
                match aligns[c] {
                    Alignment::Left => { out.push(':'); out.push_str(&"-".repeat(widths[c] - 1)); }
                    Alignment::Right => { out.push_str(&"-".repeat(widths[c] - 1)); out.push(':'); }
                    Alignment::Center => { out.push(':'); out.push_str(&"-".repeat(widths[c] - 2)); out.push(':'); }
                }
                out.push_str(" |");
            }
            out.push('\n');
        }
    }
    Ok(out)
}
