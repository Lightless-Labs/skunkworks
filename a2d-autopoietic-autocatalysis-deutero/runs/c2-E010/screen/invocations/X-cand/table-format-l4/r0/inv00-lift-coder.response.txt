- Validation follows the specified precedence exactly; empty tables bypass all checks.
- Widths and budgets count emitted Unicode scalar values, including both characters of Markdown escapes.
- Trailing ASCII spaces are removed per logical line before Markdown escaping; CRLF normalizes to LF.
- Clipping and wrapping keep escape tokens indivisible and take longest initial token prefixes.
- Grid adds a header separator only when body rows follow; Markdown always emits a marker after its header.
- Center alignment puts the odd extra padding space on the right; Compact omits only the last column’s right padding.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

struct Cell {
    lines: Vec<Vec<Token>>,
}

fn tokens_for_line(line: &str, markdown: bool) -> Vec<Token> {
    let mut tokens = Vec::new();
    for ch in line.chars() {
        if markdown && ch == '|' {
            tokens.push(Token {
                text: "\\|".to_owned(),
                width: 2,
            });
        } else if markdown && ch == '\\' {
            tokens.push(Token {
                text: "\\\\".to_owned(),
                width: 2,
            });
        } else {
            tokens.push(Token {
                text: ch.to_string(),
                width: 1,
            });
        }
    }
    tokens
}

fn invalid_char_offset(s: &str) -> Option<usize> {
    let mut iter = s.char_indices().peekable();
    while let Some((byte, ch)) = iter.next() {
        if ch == '\r' {
            if matches!(iter.peek(), Some((_, '\n'))) {
                iter.next();
                continue;
            }
            return Some(byte);
        }
        if ch == '\n' {
            continue;
        }
        if ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}' {
            return Some(byte);
        }
    }
    None
}

fn normalize_and_tokenize(s: &str, markdown: bool) -> Cell {
    let normalized = s.replace("\r\n", "\n");
    let lines = normalized.split('\n');
    let mut result = Vec::new();
    for line in lines {
        let trimmed = line.trim_end_matches(' ');
        result.push(tokens_for_line(trimmed, markdown));
    }
    Cell { lines: result }
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            if token_width(tokens) <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }
            let mut out = String::new();
            let mut used = 0;
            let limit = width.saturating_sub(1);
            for token in tokens {
                if used + token.width > limit {
                    break;
                }
                used += token.width;
                out.push_str(&token.text);
            }
            out.push('…');
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                // Width minima guarantee progress for all generated tokens.
                let mut part = String::new();
                for token in &tokens[start..end] {
                    part.push_str(&token.text);
                }
                result.push(part);
                start = end;
            }
            result
        }
    }
}

fn align_fragment(
    fragment: &str,
    width: usize,
    alignment: &Alignment,
    omit_right: bool,
) -> String {
    let content_width = fragment.chars().count();
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let mut out = String::new();
    out.push_str(&" ".repeat(left));
    out.push_str(fragment);
    if !omit_right {
        out.push_str(&" ".repeat(right));
    }
    out
}

fn render_data_line(
    row: &[Vec<String>],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    let mut out = String::new();
    match mode {
        Mode::Grid | Mode::Markdown => {
            out.push('|');
            for (i, fragment) in row.iter().enumerate() {
                out.push(' ');
                out.push_str(&align_fragment(fragment, widths[i], &alignments[i], false));
                out.push(' ');
                out.push('|');
            }
            out.push('\n');
        }
        Mode::Compact => {
            for (i, fragment) in row.iter().enumerate() {
                if i > 0 {
                    out.push_str("  ");
                }
                out.push_str(&align_fragment(
                    fragment,
                    widths[i],
                    &alignments[i],
                    i + 1 == row.len(),
                ));
            }
            out.push('\n');
        }
    }
    out
}

fn border(widths: &[usize]) -> String {
    let mut out = String::new();
    out.push('+');
    for &width in widths {
        out.push_str(&"-".repeat(width + 2));
        out.push('+');
    }
    out.push('\n');
    out
}

fn markdown_marker(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut out = String::new();
    out.push('|');
    for (i, &width) in widths.iter().enumerate() {
        out.push(' ');
        match alignments[i] {
            Alignment::Left => {
                out.push(':');
                out.push_str(&"-".repeat(width - 1));
            }
            Alignment::Right => {
                out.push_str(&"-".repeat(width - 1));
                out.push(':');
            }
            Alignment::Center => {
                out.push(':');
                out.push_str(&"-".repeat(width - 2));
                out.push(':');
            }
        }
        out.push(' ');
        out.push('|');
    }
    out.push('\n');
    out
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_column = if options.mode == Mode::Markdown { 3 } else { 1 };
    let overhead = match options.mode {
        Mode::Grid | Mode::Markdown => 3 * n + 1,
        Mode::Compact => 2 * n.saturating_sub(1),
    };
    let minimum = overhead + n * minimum_column;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            if let Some(byte) = invalid_char_offset(cell) {
                return Err(TableError::InvalidChar { row, column, byte });
            }
        }
    }

    let alignments = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let markdown = options.mode == Mode::Markdown;
    let cells: Vec<Vec<Cell>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| normalize_and_tokenize(cell, markdown))
                .collect()
        })
        .collect();

    let mut widths = vec![minimum_column; n];
    for row in &cells {
        for (column, cell) in row.iter().enumerate() {
            for line in &cell.lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let mut total = overhead + widths.iter().sum::<usize>();
        while total > options.max_width {
            let mut selected: Option<usize> = None;
            for column in 0..n {
                if widths[column] > minimum_column
                    && selected.map_or(true, |best| widths[column] >= widths[best])
                {
                    selected = Some(column);
                }
            }
            if let Some(column) = selected {
                widths[column] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in cells.iter().enumerate() {
        let mut cell_fragments: Vec<Vec<String>> = Vec::with_capacity(n);
        let mut height = 1;
        for column in 0..n {
            let mut all = Vec::new();
            for line in &row[column].lines {
                all.extend(fragments(line, widths[column], &overflows[column]));
            }
            height = height.max(all.len());
            cell_fragments.push(all);
        }

        for fragment_index in 0..height {
            let physical: Vec<String> = (0..n)
                .map(|column| {
                    cell_fragments[column]
                        .get(fragment_index)
                        .cloned()
                        .unwrap_or_default()
                })
                .collect();
            output.push_str(&render_data_line(
                &physical,
                &widths,
                &alignments,
                &options.mode,
            ));
        }

        if options.mode == Mode::Markdown && row_index == 0 {
            output.push_str(&markdown_marker(&widths, &alignments));
        }
        if options.mode == Mode::Grid && options.header && row_index == 0 && rows.len() > 1 {
            output.push_str(&border(&widths));
        }
    }

    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }
    Ok(output)
}
```
