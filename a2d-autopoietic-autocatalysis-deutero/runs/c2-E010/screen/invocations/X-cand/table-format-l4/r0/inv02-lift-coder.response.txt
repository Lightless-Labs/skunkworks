#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

struct Cell {
    lines: Vec<Vec<Token>>,
}

fn render_tokens(line: &[Token]) -> String {
    let mut result = String::new();
    for token in line {
        result.push_str(&token.text);
    }
    result
}

fn split_fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total: usize = tokens.iter().map(|token| token.width).sum();
            if total <= width {
                return vec![render_tokens(tokens)];
            }
            let available = width.saturating_sub(1);
            let mut used = 0usize;
            let mut end = 0usize;
            while end < tokens.len() && used + tokens[end].width <= available {
                used += tokens[end].width;
                end += 1;
            }
            let mut fragment = render_tokens(&tokens[..end]);
            fragment.push('…');
            vec![fragment]
        }
        Overflow::Wrap => {
            let mut fragments = Vec::new();
            let mut start = 0usize;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0usize;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                fragments.push(render_tokens(&tokens[start..end]));
                start = end;
            }
            fragments
        }
    }
}

fn aligned(text: &str, alignment: &Alignment, width: usize, omit_right: bool) -> String {
    let content_width = text.chars().count();
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let mut result = String::new();
    result.extend(std::iter::repeat_n(' ', left));
    result.push_str(text);
    if !omit_right {
        result.extend(std::iter::repeat_n(' ', right));
    }
    result
}

fn append_data_line(
    out: &mut String,
    pieces: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) {
    match mode {
        Mode::Grid | Mode::Markdown => {
            out.push('|');
            for (i, piece) in pieces.iter().enumerate() {
                out.push(' ');
                out.push_str(&aligned(piece, &alignments[i], widths[i], false));
                out.push(' ');
                out.push('|');
            }
            out.push('\n');
        }
        Mode::Compact => {
            for (i, piece) in pieces.iter().enumerate() {
                if i != 0 {
                    out.push_str("  ");
                }
                out.push_str(&aligned(
                    piece,
                    &alignments[i],
                    widths[i],
                    i + 1 == pieces.len(),
                ));
            }
            out.push('\n');
        }
    }
}

fn append_grid_border(out: &mut String, widths: &[usize]) {
    out.push('+');
    for width in widths {
        out.extend(std::iter::repeat_n('-', width + 2));
        out.push('+');
    }
    out.push('\n');
}

fn append_markdown_separator(out: &mut String, widths: &[usize], alignments: &[Alignment]) {
    out.push('|');
    for (width, alignment) in widths.iter().zip(alignments) {
        out.push(' ');
        match alignment {
            Alignment::Left => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 1));
            }
            Alignment::Right => {
                out.extend(std::iter::repeat_n('-', width - 1));
                out.push(':');
            }
            Alignment::Center => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 2));
                out.push(':');
            }
        }
        out.push(' ');
        out.push('|');
    }
    out.push('\n');
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_column = if matches!(options.mode, Mode::Markdown) { 3 } else { 1 };
    let overhead = if matches!(options.mode, Mode::Compact) {
        2 * (n - 1)
    } else {
        3 * n + 1
    };
    let minimum = overhead + n * minimum_column;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    // Validate original text and retain the byte position of each scalar.
    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            let chars: Vec<(usize, char)> = cell.char_indices().collect();
            let mut i = 0;
            while i < chars.len() {
                let (byte, ch) = chars[i];
                if ch == '\r' {
                    if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                        i += 2;
                        continue;
                    }
                    return Err(TableError::InvalidChar { row, column, byte });
                }
                if (ch.is_control() && ch != '\n') || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
                i += 1;
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let mut cells = Vec::with_capacity(rows.len());
    let mut widths = vec![minimum_column; n];

    for row in rows {
        let mut normalized_row = Vec::with_capacity(n);
        for (column, original) in row.iter().enumerate() {
            let normalized = original.replace("\r\n", "\n");
            let mut logical_lines = Vec::new();
            for raw_line in normalized.split('\n') {
                let trimmed = raw_line.trim_end_matches(' ');
                let mut tokens = Vec::new();
                for ch in trimmed.chars() {
                    if matches!(options.mode, Mode::Markdown) && (ch == '|' || ch == '\\') {
                        tokens.push(Token {
                            text: format!("\\{ch}"),
                            width: 2,
                        });
                    } else {
                        tokens.push(Token {
                            text: ch.to_string(),
                            width: 1,
                        });
                    }
                }
                let line_width: usize = tokens.iter().map(|token| token.width).sum();
                widths[column] = widths[column].max(line_width);
                logical_lines.push(tokens);
            }
            normalized_row.push(Cell { lines: logical_lines });
        }
        cells.push(normalized_row);
    }

    if options.max_width != 0 {
        let mut total: usize = widths.iter().sum::<usize>() + overhead;
        while total > options.max_width {
            let candidate = (0..n)
                .filter(|&column| widths[column] > minimum_column)
                .max_by_key(|&column| (widths[column], column));
            match candidate {
                Some(column) => {
                    widths[column] -= 1;
                    total -= 1;
                }
                None => break,
            }
        }
    }

    let mut out = String::new();
    if matches!(options.mode, Mode::Grid) {
        append_grid_border(&mut out, &widths);
    }

    for (row_index, row) in cells.iter().enumerate() {
        let fragments: Vec<Vec<Vec<String>>> = row
            .iter()
            .enumerate()
            .map(|(column, cell)| {
                cell.lines
                    .iter()
                    .map(|line| split_fragments(line, widths[column], &overflows[column]))
                    .collect()
            })
            .collect();

        let row_height = fragments
            .iter()
            .map(|column_lines| column_lines.iter().map(Vec::len).sum::<usize>())
            .max()
            .unwrap_or(1);

        // Flatten each cell's logical lines in input order, preserving per-line fragments.
        let flattened: Vec<Vec<String>> = fragments
            .iter()
            .map(|column_lines| column_lines.iter().flatten().cloned().collect())
            .collect();

        for fragment_index in 0..row_height {
            let pieces: Vec<String> = flattened
                .iter()
                .map(|column_fragments| {
                    column_fragments
                        .get(fragment_index)
                        .cloned()
                        .unwrap_or_default()
                })
                .collect();
            append_data_line(&mut out, &pieces, &widths, &alignments, &options.mode);
        }

        if matches!(options.mode, Mode::Grid) && options.header && row_index == 0 && rows.len() > 1 {
            append_grid_border(&mut out, &widths);
        }
        if matches!(options.mode, Mode::Markdown) && options.header && row_index == 0 {
            append_markdown_separator(&mut out, &widths, &alignments);
        }
    }

    if matches!(options.mode, Mode::Grid) {
        append_grid_border(&mut out, &widths);
    }

    Ok(out)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

fn escaped_line(source: &str, markdown: bool) -> Vec<(String, usize)> {
    source
        .chars()
        .map(|c| {
            if markdown && (c == '|' || c == '\\') {
                (format!("\\{c}"), 2)
            } else {
                (c.to_string(), 1)
            }
        })
        .collect()
}

fn fragments(
    tokens: &[(String, usize)],
    width: usize,
    rule: &Overflow,
) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    let mut result = Vec::new();
    let mut start = 0;
    while start < tokens.len() {
        let mut end = start;
        let mut used = 0;
        while end < tokens.len() && used + tokens[end].1 <= width {
            used += tokens[end].1;
            end += 1;
        }

        match rule {
            Overflow::Wrap => {
                result.push(tokens[start..end].iter().map(|x| x.0.as_str()).collect());
                start = end;
            }
            Overflow::Clip => {
                let remaining_width: usize = tokens[start..].iter().map(|x| x.1).sum();
                if remaining_width <= width {
                    result.push(tokens[start..].iter().map(|x| x.0.as_str()).collect());
                } else {
                    let cap = width.saturating_sub(1);
                    let mut prefix_end = start;
                    let mut prefix_width = 0;
                    while prefix_end < tokens.len()
                        && prefix_width + tokens[prefix_end].1 <= cap
                    {
                        prefix_width += tokens[prefix_end].1;
                        prefix_end += 1;
                    }
                    let mut clipped: String =
                        tokens[start..prefix_end].iter().map(|x| x.0.as_str()).collect();
                    clipped.push('…');
                    result.push(clipped);
                }
                break;
            }
        }
    }
    result
}

fn pad(text: &str, alignment: &Alignment, width: usize, last_compact: bool) -> String {
    let extra = width.saturating_sub(text.chars().count());
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let mut out = String::new();
    for _ in 0..left {
        out.push(' ');
    }
    out.push_str(text);
    if !last_compact {
        for _ in 0..right {
            out.push(' ');
        }
    }
    out
}

fn border(out: &mut String, widths: &[usize]) {
    out.push('+');
    for width in widths {
        for _ in 0..width + 2 {
            out.push('-');
        }
        out.push('+');
    }
    out.push('\n');
}

fn data_line(
    out: &mut String,
    texts: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) {
    match mode {
        Mode::Grid | Mode::Markdown => {
            out.push('|');
            for i in 0..texts.len() {
                out.push(' ');
                out.push_str(&pad(&texts[i], &alignments[i], widths[i], false));
                out.push(' ');
                out.push('|');
            }
            out.push('\n');
        }
        Mode::Compact => {
            for i in 0..texts.len() {
                if i > 0 {
                    out.push_str("  ");
                }
                out.push_str(&pad(
                    &texts[i],
                    &alignments[i],
                    widths[i],
                    i + 1 == texts.len(),
                ));
            }
            out.push('\n');
        }
    }
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }
    for (i, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: i });
        }
    }
    let n = rows[0].len();
    for (i, row) in rows.iter().enumerate() {
        if row.len() != n {
            return Err(TableError::Ragged {
                row: i,
                expected: n,
                actual: row.len(),
            });
        }
    }
    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let compact = options.mode == Mode::Compact;
    let markdown = options.mode == Mode::Markdown;
    let floor = if markdown { 3 } else { 1 };
    let overhead = if compact { 2 * (n - 1) } else { 3 * n + 1 };
    let min_total = overhead + floor * n;
    if options.max_width != 0 && options.max_width < min_total {
        return Err(TableError::TooNarrow {
            minimum: min_total,
            provided: options.max_width,
        });
    }

    for (row_i, row) in rows.iter().enumerate() {
        for (col_i, cell) in row.iter().enumerate() {
            let mut iter = cell.char_indices().peekable();
            while let Some((byte, ch)) = iter.next() {
                if ch == '\r' {
                    if matches!(iter.peek(), Some((_, '\n'))) {
                        iter.next();
                    } else {
                        return Err(TableError::InvalidChar {
                            row: row_i,
                            column: col_i,
                            byte,
                        });
                    }
                } else if (ch.is_control() && ch != '\n') || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar {
                        row: row_i,
                        column: col_i,
                        byte,
                    });
                }
            }
        }
    }

    let alignments = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let mut prepared: Vec<Vec<Vec<Vec<(String, usize)>>>> = Vec::new();
    let mut widths = vec![floor; n];
    for row in rows {
        let mut prepared_row = Vec::new();
        for (col, cell) in row.iter().enumerate() {
            let normalized = cell.replace("\r\n", "\n");
            let mut lines = Vec::new();
            for line in normalized.split('\n') {
                let trimmed = line.trim_end_matches(' ');
                let tokens = escaped_line(trimmed, markdown);
                let total: usize = tokens.iter().map(|x| x.1).sum();
                widths[col] = widths[col].max(total);
                lines.push(tokens);
            }
            prepared_row.push(lines);
        }
        prepared.push(prepared_row);
    }

    if options.max_width != 0 {
        let mut used = overhead + widths.iter().sum::<usize>();
        while used > options.max_width {
            let mut chosen = None;
            for i in 0..n {
                if widths[i] > floor
                    && chosen.map_or(true, |j: usize| widths[i] >= widths[j])
                {
                    chosen = Some(i);
                }
            }
            if let Some(i) = chosen {
                widths[i] -= 1;
                used -= 1;
            } else {
                break;
            }
        }
    }

    let mut out = String::new();
    if options.mode == Mode::Grid {
        border(&mut out, &widths);
    }

    for (row_i, row) in prepared.iter().enumerate() {
        let mut per_cell = Vec::new();
        let mut height = 1;
        for (col, lines) in row.iter().enumerate() {
            let mut all = Vec::new();
            for line in lines {
                let parts = fragments(line, widths[col], &overflows[col]);
                height = height.max(all.len() + parts.len());
                all.extend(parts);
            }
            per_cell.push(all);
        }

        for index in 0..height {
            let texts: Vec<String> = per_cell
                .iter()
                .map(|column| column.get(index).cloned().unwrap_or_default())
                .collect();
            data_line(&mut out, &texts, &widths, &alignments, &options.mode);
        }

        if options.mode == Mode::Grid && options.header && row_i == 0 && rows.len() > 1 {
            border(&mut out, &widths);
        }
        if markdown && row_i == 0 {
            out.push('|');
            for col in 0..n {
                out.push(' ');
                match alignments[col] {
                    Alignment::Left => {
                        out.push(':');
                        for _ in 1..widths[col] {
                            out.push('-');
                        }
                    }
                    Alignment::Right => {
                        for _ in 1..widths[col] {
                            out.push('-');
                        }
                        out.push(':');
                    }
                    Alignment::Center => {
                        out.push(':');
                        for _ in 2..widths[col] {
                            out.push('-');
                        }
                        out.push(':');
                    }
                }
                out.push(' ');
                out.push('|');
            }
            out.push('\n');
        }
    }

    if options.mode == Mode::Grid {
        border(&mut out, &widths);
    }
    Ok(out)
}
