- Validation follows the stated precedence exactly; empty tables bypass all checks.
- CRLF counts as one line break, while invalid-character offsets remain byte offsets in the original cell.
- Width and padding count Unicode scalars; Markdown escapes count as two emitted scalars and stay indivisible.
- Clipping and wrapping consume whole tokens from the start of each logical line; empty lines yield one empty fragment.
- Budget shrinking chooses the rightmost widest eligible column and never expands columns.
- Compact omits only the last column’s right alignment padding; all physical lines end in LF.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment { Left, Right, Center }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow { Clip, Wrap }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode { Grid, Compact, Markdown }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

type CellLines = Vec<Vec<Vec<Token>>>;

fn invalid(c: char) -> bool {
    (c == '\r') || (c.is_control() && c != '\n') || c == '\u{2028}' || c == '\u{2029}'
}

fn parse_cell(s: &str, row: usize, column: usize, markdown: bool) -> Result<Vec<Vec<Token>>, TableError> {
    let chars: Vec<(usize, char)> = s.char_indices().collect();
    let mut normalized = String::new();
    let mut i = 0;
    while i < chars.len() {
        let (byte, c) = chars[i];
        if c == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
                continue;
            }
            return Err(TableError::InvalidChar { row, column, byte });
        }
        if invalid(c) {
            return Err(TableError::InvalidChar { row, column, byte });
        }
        normalized.push(c);
        i += 1;
    }

    let mut lines = Vec::new();
    for raw in normalized.split('\n') {
        let trimmed = raw.trim_end_matches(' ');
        let mut tokens = Vec::new();
        for c in trimmed.chars() {
            if markdown && c == '|' {
                tokens.push(Token { text: "\\|".to_owned(), width: 2 });
            } else if markdown && c == '\\' {
                tokens.push(Token { text: "\\\\".to_owned(), width: 2 });
            } else {
                tokens.push(Token { text: c.to_string(), width: 1 });
            }
        }
        lines.push(tokens);
    }
    Ok(lines)
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, rule: &Overflow) -> Vec<Vec<Token>> {
    if tokens.is_empty() {
        return vec![Vec::new()];
    }
    match rule {
        Overflow::Clip => {
            if token_width(tokens) <= width {
                return vec![tokens.to_vec()];
            }
            let mut out = Vec::new();
            let mut used = 0;
            for token in tokens {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                used += token.width;
                out.push(token.clone());
            }
            out.push(Token { text: "…".to_owned(), width: 1 });
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                result.push(tokens[start..end].to_vec());
                start = end;
            }
            result
        }
    }
}

fn aligned(content: &[Token], width: usize, alignment: &Alignment, omit_right: bool) -> String {
    let text: String = content.iter().map(|t| t.text.as_str()).collect();
    let used = token_width(content);
    let extra = width - used;
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    format!("{}{}{}", " ".repeat(left), text, " ".repeat(if omit_right { 0 } else { right }))
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (r, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: r });
        }
    }
    let n = rows[0].len();
    for (r, row) in rows.iter().enumerate() {
        if row.len() != n {
            return Err(TableError::Ragged { row: r, expected: n, actual: row.len() });
        }
    }
    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount { expected: n, actual: options.alignments.len() });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount { expected: n, actual: options.overflow.len() });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let (overhead, min_col) = match options.mode {
        Mode::Grid | Mode::Markdown => (3usize.saturating_mul(n).saturating_add(1), if matches!(options.mode, Mode::Markdown) { 3 } else { 1 }),
        Mode::Compact => (2usize.saturating_mul(n.saturating_sub(1)), 1),
    };
    let minimum = overhead.saturating_add(n.saturating_mul(min_col));
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow { minimum, provided: options.max_width });
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else { options.alignments.clone() };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else { options.overflow.clone() };

    let mut parsed: Vec<Vec<CellLines>> = Vec::new();
    let mut widths = vec![min_col; n];
    for (r, row) in rows.iter().enumerate() {
        let mut parsed_row = Vec::new();
        for (c, cell) in row.iter().enumerate() {
            let lines = parse_cell(cell, r, c, matches!(options.mode, Mode::Markdown))?;
            for line in &lines {
                widths[c] = widths[c].max(token_width(line));
            }
            parsed_row.push(lines);
        }
        parsed.push(parsed_row);
    }

    if options.max_width != 0 {
        let mut total = widths.iter().sum::<usize>().saturating_add(overhead);
        while total > options.max_width {
            let mut candidate = None;
            for c in 0..n {
                if widths[c] > min_col {
                    match candidate {
                        None => candidate = Some(c),
                        Some(old) if widths[c] >= widths[old] => candidate = Some(c),
                        _ => {}
                    }
                }
            }
            if let Some(c) = candidate {
                widths[c] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    let mut out = String::new();
    let grid = matches!(options.mode, Mode::Grid);
    let markdown = matches!(options.mode, Mode::Markdown);

    let mut border = || {
        out.push('+');
        for &w in &widths {
            out.push_str(&"-".repeat(w + 2));
            out.push('+');
        }
        out.push('\n');
    };
    if grid { border(); }

    for r in 0..rows.len() {
        let mut cell_frags: Vec<Vec<Vec<Token>>> = Vec::new();
        let mut height = 1;
        for c in 0..n {
            let mut all = Vec::new();
            for line in &parsed[r][c] {
                all.extend(fragments(line, widths[c], &overflows[c]));
            }
            height = height.max(all.len());
            cell_frags.push(all);
        }
        for fi in 0..height {
            match options.mode {
                Mode::Grid | Mode::Markdown => {
                    out.push('|');
                    for c in 0..n {
                        let empty = Vec::new();
                        let frag = cell_frags[c].get(fi).unwrap_or(&empty);
                        out.push(' ');
                        out.push_str(&aligned(frag, widths[c], &alignments[c], false));
                        out.push_str(" |");
                    }
                    out.push('\n');
                }
                Mode::Compact => {
                    for c in 0..n {
                        if c > 0 { out.push_str("  "); }
                        let empty = Vec::new();
                        let frag = cell_frags[c].get(fi).unwrap_or(&empty);
                        out.push_str(&aligned(frag, widths[c], &alignments[c], c + 1 == n));
                    }
                    out.push('\n');
                }
            }
        }
        if markdown && r == 0 {
            out.push('|');
            for c in 0..n {
                let marker = match alignments[c] {
                    Alignment::Left => format!(":{}", "-".repeat(widths[c] - 1)),
                    Alignment::Right => format!("{}:", "-".repeat(widths[c] - 1)),
                    Alignment::Center => format!(":{}:", "-".repeat(widths[c] - 2)),
                };
                out.push(' ');
                out.push_str(&marker);
                out.push_str(" |");
            }
            out.push('\n');
        }
        if grid && options.header && r == 0 && r + 1 < rows.len() {
            border();
        }
    }
    if grid { border(); }
    Ok(out)
}
```
