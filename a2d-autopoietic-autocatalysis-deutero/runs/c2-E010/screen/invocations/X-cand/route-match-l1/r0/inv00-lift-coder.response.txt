- Request paths discard the raw query or fragment before validation; invalid bytes in the suffix are ignored.
- Segment errors are checked left to right before trailing-slash errors; offsets are UTF-8 byte positions in the original input.
- Percent escapes use uppercase hex and decode only to printable, non-unreserved ASCII other than slash.
- Capture names are validated as a whole segment and duplicate names fail at the segment start.
- Matching requires equal segment counts; static segments outrank parameters lexicographically, with registration order breaking ties.
- I see no meaningful ambiguity in the requirements, so Solution B uses a separate matching algorithm.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    segments: Vec<Segment>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(Route { id, segments });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request = parse_path(head)?;

        let mut best: Option<(Vec<u8>, Match)> = None;

        for route in &self.routes {
            if route.segments.len() != request.len() {
                continue;
            }

            let mut rank = Vec::with_capacity(route.segments.len());
            let mut params = Vec::new();
            let mut matches = true;

            for (pattern_seg, value) in route.segments.iter().zip(&request) {
                match pattern_seg {
                    Segment::Static(expected) => {
                        if expected != value {
                            matches = false;
                            break;
                        }
                        rank.push(2);
                    }
                    Segment::Param(name) => {
                        rank.push(1);
                        params.push((name.clone(), value.clone()));
                    }
                }
            }

            if matches {
                let candidate = Match {
                    route: route.id,
                    params,
                };
                let replace = match &best {
                    None => true,
                    Some((best_rank, best_match)) => {
                        rank > *best_rank
                            || (rank == *best_rank && candidate.route < best_match.route)
                    }
                };
                if replace {
                    best = Some((rank, candidate));
                }
            }
        }

        best.map(|(_, matched)| matched).ok_or(Error::NotFound)
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| start + n)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &input[start..end];
        if raw.starts_with(':') {
            let name = &raw[1..];
            if !valid_capture_name(name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if parts.iter().any(|seg| matches!(seg, Segment::Param(existing) if existing == name)) {
                return Err(Error::InvalidPattern { at: start });
            }
            parts.push(Segment::Param(name.to_owned()));
        } else if raw.starts_with('*') {
            return Err(Error::InvalidPattern { at: start });
        } else {
            let decoded = decode_static_segment(raw, start).map_err(|at| Error::InvalidPattern { at })?;
            parts.push(Segment::Static(decoded));
        }

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(Error::InvalidPattern { at: bytes.len() - 1 });
        }
    }

    Ok(parts)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| start + n)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPath { at: start });
        }

        let decoded = decode_static_segment(&input[start..end], start)
            .map_err(|at| Error::InvalidPath { at })?;
        parts.push(decoded);

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(Error::InvalidPath { at: bytes.len() - 1 });
        }
    }

    Ok(parts)
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_static_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_value(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_value(bytes[i + 2]).ok_or(base + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(base + i);
            }
            decoded.push(b);
            i += 1;
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

enum Part {
    Literal(String),
    Capture(String),
}

struct Entry {
    parts: Vec<Part>,
}

pub struct Router {
    entries: Vec<Entry>,
}

impl Router {
    pub fn new() -> Self {
        Self { entries: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let parts = parse_pattern(pattern)?;
        let id = self.entries.len();
        self.entries.push(Entry { parts });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let end = path
            .as_bytes()
            .iter()
            .position(|&b| b == b'?' || b == b'#')
            .unwrap_or(path.len());
        let values = parse_path(&path[..end])?;

        let mut selected: Option<(usize, Match)> = None;
        for (id, entry) in self.entries.iter().enumerate() {
            if entry.parts.len() != values.len() {
                continue;
            }

            let mut specificity = 0usize;
            let mut captures = Vec::new();
            let mut valid = true;

            for (part, value) in entry.parts.iter().zip(values.iter()) {
                match part {
                    Part::Literal(literal) if literal == value => specificity += 1,
                    Part::Literal(_) => {
                        valid = false;
                        break;
                    }
                    Part::Capture(name) => captures.push((name.clone(), value.clone())),
                }
            }

            if !valid {
                continue;
            }

            let replace = match &selected {
                None => true,
                Some((old_id, old_match)) => {
                    let old_entry = &self.entries[*old_id];
                    let new_rank = entry
                        .parts
                        .iter()
                        .map(|p| matches!(p, Part::Literal(_)))
                        .collect::<Vec<_>>();
                    let old_rank = old_entry
                        .parts
                        .iter()
                        .map(|p| matches!(p, Part::Literal(_)))
                        .collect::<Vec<_>>();
                    new_rank > old_rank || (new_rank == old_rank && id < old_match.route)
                }
            };

            if replace {
                selected = Some((
                    id,
                    Match {
                        route: id,
                        params: captures,
                    },
                ));
            }
            let _ = specificity;
        }

        selected.map(|(_, matched)| matched).ok_or(Error::NotFound)
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Part>, Error> {
    if input.as_bytes().first() != Some(&b'/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut result = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut start = 1usize;

    loop {
        let end = bytes[start..]
            .iter()
            .position(|&byte| byte == b'/')
            .map_or(bytes.len(), |offset| start + offset);

        if start == end {
            return Err(Error::InvalidPattern { at: start });
        }

        let segment = &input[start..end];
        if segment.starts_with(':') {
            let name = &segment[1..];
            if !capture_name_ok(name) || seen.iter().any(|prior| prior == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            seen.push(name.to_owned());
            result.push(Part::Capture(name.to_owned()));
        } else if segment.starts_with('*') {
            return Err(Error::InvalidPattern { at: start });
        } else {
            result.push(Part::Literal(
                decode_segment(segment, start).map_err(|at| Error::InvalidPattern { at })?,
            ));
        }

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(Error::InvalidPattern { at: end });
        }
    }

    Ok(result)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if input.as_bytes().first() != Some(&b'/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut result = Vec::new();
    let mut start = 1usize;

    loop {
        let end = bytes[start..]
            .iter()
            .position(|&byte| byte == b'/')
            .map_or(bytes.len(), |offset| start + offset);

        if start == end {
            return Err(Error::InvalidPath { at: start });
        }

        result.push(decode_segment(&input[start..end], start).map_err(|at| Error::InvalidPath { at })?);

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(Error::InvalidPath { at: end });
        }
    }

    Ok(result)
}

fn capture_name_ok(name: &str) -> bool {
    let bytes = name.as_bytes();
    bytes.first().is_some_and(u8::is_ascii_lowercase)
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(segment: &str, base: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut output = Vec::new();
    let mut index = 0usize;

    while index < bytes.len() {
        if bytes[index] == b'%' {
            if index + 2 >= bytes.len() {
                return Err(base + index);
            }
            let high = hex(bytes[index + 1]).ok_or(base + index)?;
            let low = hex(bytes[index + 2]).ok_or(base + index)?;
            let decoded = (high << 4) | low;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || unreserved(decoded)
            {
                return Err(base + index);
            }
            output.push(decoded);
            index += 3;
        } else {
            if !unreserved(bytes[index]) {
                return Err(base + index);
            }
            output.push(bytes[index]);
            index += 1;
        }
    }

    Ok(String::from_utf8(output).expect("validated ASCII segment"))
}
```
