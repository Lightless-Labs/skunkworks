fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),

fn parse_encoded_segment(input: &str, start: usize, pattern: bool) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(if pattern {
                    Error::InvalidPattern { at: start + i }
                } else {
                    Error::InvalidPath { at: start + i }
                });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            let value = match (hi, lo) {
                (Some(hi), Some(lo)) => (hi << 4) | lo,
                _ => {
                    return Err(if pattern {
                        Error::InvalidPattern { at: start + i }
                    } else {
                        Error::InvalidPath { at: start + i }
                    })
                }
            };
            if !(0x20..=0x7e).contains(&value) || value == b'/' || is_unreserved(value) {
                return Err(if pattern {
                    Error::InvalidPattern { at: start + i }
                } else {
                    Error::InvalidPath { at: start + i }
                });
            }
            decoded.push(value);
            i += 3;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: start + i }
            } else {
                Error::InvalidPath { at: start + i }
            });
        }
    }

    Ok(String::from_utf8(decoded).unwrap())
}

fn parse_pattern(pattern: &str) -> Result<Route, Error> {
    let bytes = pattern.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Route { segments: Vec::new() });
    }

    let mut end = bytes.len();
    let mut trailing = false;
    if bytes.last() == Some(&b'/') {
        trailing = true;
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut names = Vec::new();
    let mut start = 1;

    while start < end {
        let segment_end = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map_or(end, |offset| start + offset);
        if segment_end == start {
            return Err(Error::InvalidPattern { at: start });
        }
        let raw = &pattern[start..segment_end];
        if raw.starts_with(':') {
            let name = &raw[1..];
            let valid = !name.is_empty()
                && name.as_bytes()[0].is_ascii_lowercase()
                && name
                    .as_bytes()
                    .iter()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_');
            if !valid || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_owned());
            segments.push(Segment::Param(name.to_owned()));
        } else if raw.starts_with('*') {
            return Err(Error::InvalidPattern { at: start });
        } else {
            segments.push(Segment::Static(parse_encoded_segment(
                raw, start, true,
            )?));
        }
        start = segment_end + 1;
    }

    if trailing {
        return Err(Error::InvalidPattern { at: bytes.len() - 1 });
    }

    Ok(Route { segments })
}

        return Err(Error::InvalidPath { at: 0 });

        return Ok(Vec::new());

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Part {
    Literal(String),
    Capture(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Entry {
    parts: Vec<Part>,
    rank: Vec<u8>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Router {
    entries: Vec<Entry>,
}

impl Router {
    pub fn new() -> Self {
        Self { entries: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let parts = scan_pattern(pattern)?;
        let rank = parts
            .iter()
            .map(|part| if matches!(part, Part::Literal(_)) { 2 } else { 1 })
            .collect();
        let id = self.entries.len();
        self.entries.push(Entry { parts, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_len = path
            .as_bytes()
            .iter()
            .position(|b| *b == b'?' || *b == b'#')
            .unwrap_or(path.len());
        let segments = scan_path(&path[..head_len])?;

        let mut winner: Option<(usize, Vec<(String, String)>)> = None;
        for (id, entry) in self.entries.iter().enumerate() {
            if entry.parts.len() != segments.len() {
                continue;
            }
            let mut params = Vec::new();
            let mut success = true;
            for (part, segment) in entry.parts.iter().zip(segments.iter()) {
                match part {
                    Part::Literal(value) if value == segment => {}
                    Part::Literal(_) => {
                        success = false;
                        break;
                    }
                    Part::Capture(name) => params.push((name.clone(), segment.clone())),
                }
            }
            if !success {
                continue;
            }
            let replace = match &winner {
                None => true,
                Some((winner_id, _)) => {
                    let winner_entry = &self.entries[*winner_id];
                    entry.rank > winner_entry.rank
                }
            };
            if replace {
                winner = Some((id, params));
            }
        }
        winner
            .map(|(route, params)| Match { route, params })
            .ok_or(Error::NotFound)
    }
}

fn err_pattern(at: usize) -> Error {
    Error::InvalidPattern { at }
}

fn err_path(at: usize) -> Error {
    Error::InvalidPath { at }
}

fn decode(raw: &str, base: usize, is_pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if unreserved(b) {
            out.push(b);
            i += 1;
            continue;
        }
        if b != b'%' {
            return Err(if is_pattern { err_pattern(base + i) } else { err_path(base + i) });
        }
        if i + 2 >= bytes.len() {
            return Err(if is_pattern { err_pattern(base + i) } else { err_path(base + i) });
        }
        let a = upper_hex(bytes[i + 1]);
        let c = upper_hex(bytes[i + 2]);
        let Some((a, c)) = a.zip(c) else {
            return Err(if is_pattern { err_pattern(base + i) } else { err_path(base + i) });
        };
        let decoded = (a << 4) | c;
        if !(0x20..=0x7e).contains(&decoded) || decoded == b'/' || unreserved(decoded) {
            return Err(if is_pattern { err_pattern(base + i) } else { err_path(base + i) });
        }
        out.push(decoded);
        i += 3;
    }
    Ok(String::from_utf8(out).unwrap())
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn scan_pattern(pattern: &str) -> Result<Vec<Part>, Error> {
    let bytes = pattern.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(err_pattern(0));
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }
    let mut parts = Vec::new();
    let mut names: Vec<String> = Vec::new();
    let mut begin = 1;
    while begin < bytes.len() {
        let end = bytes[begin..]
            .iter()
            .position(|b| *b == b'/')
            .map_or(bytes.len(), |n| begin + n);
        if end == begin {
            return Err(err_pattern(begin));
        }
        let segment = &pattern[begin..end];
        if segment.starts_with(':') {
            let name = &segment[1..];
            let ok = !name.is_empty()
                && name.as_bytes()[0].is_ascii_lowercase()
                && name.as_bytes().iter().all(|b| {
                    b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_'
                });
            if !ok || names.iter().any(|n| n == name) {
                return Err(err_pattern(begin));
            }
            names.push(name.to_owned());
            parts.push(Part::Capture(name.to_owned()));
        } else if segment.starts_with('*') {
            return Err(err_pattern(begin));
        } else {
            parts.push(Part::Literal(decode(segment, begin, true)?));
        }
        begin = end + 1;
    }
    if bytes.last() == Some(&b'/') {
        return Err(err_pattern(bytes.len() - 1));
    }
    Ok(parts)
}

fn scan_path(path: &str) -> Result<Vec<String>, Error> {
    let bytes = path.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(err_path(0));
    }
    if path == "/" {
        return Ok(Vec::new());
    }
    let mut segments = Vec::new();
    let mut begin = 1;
    while begin < bytes.len() {
        let end = bytes[begin..]
            .iter()
            .position(|b| *b == b'/')
            .map_or(bytes.len(), |n| begin + n);
        if end == begin {
            return Err(err_path(begin));
        }
        segments.push(decode(&path[begin..end], begin, false)?);
        begin = end + 1;
    }
    if bytes.last() == Some(&b'/') {
        return Err(err_path(bytes.len() - 1));
    }
    Ok(segments)
}
