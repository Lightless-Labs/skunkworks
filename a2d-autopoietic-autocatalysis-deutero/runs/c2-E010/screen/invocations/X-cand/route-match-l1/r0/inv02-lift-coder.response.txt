- Request query and fragment stripping happens before validation; the earliest raw `?` or `#` ends the path head.
- Root `/` is the only valid zero-segment path; a trailing slash on any other path is rejected after preceding segments are validated.
- Percent escapes require uppercase hex and decode only to printable, non-slash, non-unreserved ASCII; errors point to the `%`.
- Capture and wildcard segment errors are reported at the segment start, before validating the remainder as static data.
- Routes rank by their full left-to-right static/parameter vectors; equal ranks keep the earliest registered route.
- Both implementations use `usize` route IDs and offsets; practical inputs are bounded by addressable memory.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
}

#[derive(Debug, Clone)]
struct Route {
    segments: Vec<Segment>,
    id: usize,
}

#[derive(Debug, Clone)]
pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(Route { segments, id });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request = parse_path(head)?;

        let mut best: Option<(usize, Vec<(String, String)>)> = None;
        let mut best_rank: Option<Vec<u8>> = None;

        for route in &self.routes {
            if route.segments.len() != request.len() {
                continue;
            }

            let mut rank = Vec::with_capacity(route.segments.len());
            let mut params = Vec::new();
            let mut matched = true;

            for (pattern_segment, value) in route.segments.iter().zip(request.iter()) {
                match pattern_segment {
                    Segment::Static(expected) => {
                        rank.push(2);
                        if expected != value {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Param(name) => {
                        rank.push(1);
                        params.push((name.clone(), value.clone()));
                    }
                }
            }

            if matched && best_rank.as_ref().is_none_or(|r| rank > *r) {
                best_rank = Some(rank);
                best = Some((route.id, params));
            }
        }

        match best {
            Some((route, params)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(input: &str, base: usize, pattern: bool) -> Result<String, usize> {
    let bytes = input.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(base + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(base + i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else if b.is_ascii() && is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            let _ = pattern;
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let bytes = pattern.as_bytes();
    let mut result = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'/')
            .map(|p| start + p)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..end];
        if raw.as_bytes()[0] == b':' {
            let name = &raw[1..];
            let nb = name.as_bytes();
            let valid = !nb.is_empty()
                && nb[0].is_ascii_lowercase()
                && nb[1..].iter().all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_');
            if !valid || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            result.push(Segment::Param(name.to_owned()));
        } else if raw.as_bytes()[0] == b'*' {
            return Err(Error::InvalidPattern { at: start });
        } else {
            let value = decode_segment(raw, start, true)
                .map_err(|at| Error::InvalidPattern { at })?;
            result.push(Segment::Static(value));
        }

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(Error::InvalidPattern { at: end });
        }
    }

    Ok(result)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'/')
            .map(|p| start + p)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPath { at: start });
        }

        let raw = &path[start..end];
        let value = decode_segment(raw, start, false)
            .map_err(|at| Error::InvalidPath { at })?;
        result.push(value);

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(Error::InvalidPath { at: end });
        }
    }

    Ok(result)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone)]
enum Part {
    Literal(String),
    Capture(String),
}

#[derive(Debug, Clone)]
struct Entry {
    id: usize,
    parts: Vec<Part>,
    rank: Vec<u8>,
}

#[derive(Debug, Clone)]
pub struct Router {
    entries: Vec<Entry>,
}

impl Router {
    pub fn new() -> Self {
        Self { entries: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let parts = parse_pattern_b(pattern)?;
        let rank = parts
            .iter()
            .map(|part| match part {
                Part::Literal(_) => 2,
                Part::Capture(_) => 1,
            })
            .collect();
        let id = self.entries.len();
        self.entries.push(Entry { id, parts, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let end = path
            .as_bytes()
            .iter()
            .position(|b| *b == b'?' || *b == b'#')
            .unwrap_or(path.len());
        let values = parse_path_b(&path[..end])?;

        let mut winner: Option<(&Entry, Vec<(String, String)>)> = None;
        for entry in &self.entries {
            if entry.parts.len() != values.len() {
                continue;
            }
            let mut captures = Vec::new();
            let mut ok = true;
            for (part, value) in entry.parts.iter().zip(&values) {
                match part {
                    Part::Literal(literal) if literal == value => {}
                    Part::Literal(_) => {
                        ok = false;
                        break;
                    }
                    Part::Capture(name) => captures.push((name.clone(), value.clone())),
                }
            }
            if ok && winner.as_ref().is_none_or(|(previous, _)| entry.rank > previous.rank) {
                winner = Some((entry, captures));
            }
        }

        winner
            .map(|(entry, params)| Match { route: entry.id, params })
            .ok_or(Error::NotFound)
    }
}

fn is_unreserved_b(b: u8) -> bool {
    matches!(b, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'.' | b'_' | b'~' | b'-')
}

fn hex_b(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_b(raw: &str, offset: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        match bytes[i] {
            b'%' => {
                if i + 2 >= bytes.len() {
                    return Err(offset + i);
                }
                let hi = hex_b(bytes[i + 1]).ok_or(offset + i)?;
                let lo = hex_b(bytes[i + 2]).ok_or(offset + i)?;
                let byte = (hi << 4) | lo;
                if !(0x20..=0x7e).contains(&byte) || byte == b'/' || is_unreserved_b(byte) {
                    return Err(offset + i);
                }
                decoded.push(byte);
                i += 3;
            }
            b if b.is_ascii() && is_unreserved_b(b) => {
                decoded.push(b);
                i += 1;
            }
            _ => return Err(offset + i),
        }
    }
    Ok(String::from_utf8(decoded).expect("valid decoded ASCII"))
}

fn parse_pattern_b(pattern: &str) -> Result<Vec<Part>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let mut parts = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let bytes = pattern.as_bytes();
    let mut pos = 1;

    loop {
        let stop = bytes[pos..]
            .iter()
            .position(|&b| b == b'/')
            .map(|x| pos + x)
            .unwrap_or(bytes.len());

        if stop == pos {
            return Err(Error::InvalidPattern { at: pos });
        }

        let segment = &pattern[pos..stop];
        match segment.as_bytes()[0] {
            b':' => {
                let name = &segment[1..];
                let n = name.as_bytes();
                if n.is_empty()
                    || !n[0].is_ascii_lowercase()
                    || n[1..]
                        .iter()
                        .any(|b| !b.is_ascii_lowercase() && !b.is_ascii_digit() && *b != b'_')
                    || !seen.insert(name.to_owned())
                {
                    return Err(Error::InvalidPattern { at: pos });
                }
                parts.push(Part::Capture(name.to_owned()));
            }
            b'*' => return Err(Error::InvalidPattern { at: pos }),
            _ => parts.push(
                decode_b(segment, pos).map_err(|at| Error::InvalidPattern { at })?,
            ),
        }

        if stop == bytes.len() {
            break;
        }
        pos = stop + 1;
        if pos == bytes.len() {
            return Err(Error::InvalidPattern { at: stop });
        }
    }

    Ok(parts)
}

fn parse_path_b(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut values = Vec::new();
    let mut pos = 1;

    loop {
        let stop = bytes[pos..]
            .iter()
            .position(|&b| b == b'/')
            .map(|x| pos + x)
            .unwrap_or(bytes.len());

        if stop == pos {
            return Err(Error::InvalidPath { at: pos });
        }

        let segment = &path[pos..stop];
        values.push(decode_b(segment, pos).map_err(|at| Error::InvalidPath { at })?);

        if stop == bytes.len() {
            break;
        }
        pos = stop + 1;
        if pos == bytes.len() {
            return Err(Error::InvalidPath { at: stop });
        }
    }

    Ok(values)
}
```
