            index += 1;
        }
    }

            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request = parse_path(head)?;

        let mut best: Option<(usize, Vec<(String, String)>)> = None;
        let mut best_rank: Option<Vec<u8>> = None;

        for route in &self.routes {
            if route.segments.len() != request.len() {
                continue;
            }

            let mut rank = Vec::with_capacity(route.segments.len());
            let mut params = Vec::new();
            let mut matched = true;

            for (pattern_segment, value) in route.segments.iter().zip(request.iter()) {
                match pattern_segment {
                    Segment::Static(expected) => {
                        rank.push(2);

                            matched = false;
                            break;
                        }
                    }
                    Segment::Param(name) => {
                        rank.push(1);
                        params.push((name.clone(), value.clone()));
                    }
                }
            }

                best_rank = Some(rank);
                best = Some((route.id, params));
            }
        }

        match best {
            Some((route, params)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }
}

        for (id, entry) in self.entries.iter().enumerate() {

                continue;

                        success = false;
                        break;

                }

                }

fn decode(raw: &str, base: usize, is_pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if unreserved(b) {
            out.push(b);
            i += 1;
            continue;
        }
        if b != b'%' {
            return Err(if is_pattern { err_pattern(base + i) } else { err_path(base + i) });
        }
        if i + 2 >= bytes.len() {
            return Err(if is_pattern { err_pattern(base + i) } else { err_path(base + i) });
        }
        let a = upper_hex(bytes[i + 1]);
        let c = upper_hex(bytes[i + 2]);
        let Some((a, c)) = a.zip(c) else {
            return Err(if is_pattern { err_pattern(base + i) } else { err_path(base + i) });
        };

        if !(0x20..=0x7e).contains(&decoded) || decoded == b'/' || unreserved(decoded) {
            return Err(if is_pattern { err_pattern(base + i) } else { err_path(base + i) });
        }
        out.push(decoded);
        i += 3;
    }
    Ok(String::from_utf8(out).unwrap())
}

                    b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_'
                });
            if !ok || names.iter().any(|n| n == name) {
                return Err(err_pattern(begin));
            }
            names.push(name.to_owned());
            parts.push(Part::Capture(name.to_owned()));
        } else if segment.starts_with('*') {
            return Err(err_pattern(begin));
        } else {
            parts.push(Part::Literal(decode(segment, begin, true)?));
        }
        begin = end + 1;
    }
    if bytes.last() == Some(&b'/') {
        return Err(err_pattern(bytes.len() - 1));
    }
    Ok(parts)
}

fn scan_path(path: &str) -> Result<Vec<String>, Error> {
    let bytes = path.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(err_path(0));
    }
    if path == "/" {
        return Ok(Vec::new());
    }
    let mut segments = Vec::new();
    let mut begin = 1;
    while begin < bytes.len() {
        let end = bytes[begin..]
            .iter()
            .position(|b| *b == b'/')
            .map_or(bytes.len(), |n| begin + n);
        if end == begin {
            return Err(err_path(begin));
        }
        segments.push(decode(&path[begin..end], begin, false)?);
        begin = end + 1;
    }
    if bytes.last() == Some(&b'/') {
        return Err(err_path(bytes.len() - 1));
    }
    Ok(segments)
}

#[derive(Debug, Clone)]

    Capture(String),

#[derive(Debug, Clone)]

    id: usize,
    parts: Vec<Part>,

#[derive(Debug, Clone)]
pub struct Router {
    entries: Vec<Entry>,
}

impl Router {
    pub fn new() -> Self {
        Self { entries: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {

        let id = self.entries.len();
        self.entries.push(Entry { id, parts });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_len = path
            .bytes()
            .position(|b| b == b'?' || b == b'#')
            .unwrap_or(path.len());
        let head = &path[..head_len];
        let values = request_segments(head)?;

        let mut winner: Option<(usize, Vec<(String, String)>)> = None;

        for entry in &self.entries {
            if entry.parts.len() != values.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut same = true;
            for (part, value) in entry.parts.iter().zip(&values) {
                match part {
                    Part::Literal(literal) => {
                        if literal != value {
                            same = false;
                            break;
                        }
                    }
                    Part::Capture(name) => params.push((name.clone(), value.clone())),
                }
            }

            if !same {
                continue;
            }

            let should_replace = match &winner {
                None => true,
                Some((old_index, _)) => {
                    specificity(&entry.parts) > specificity(&self.entries[*old_index].parts)
                }
            };
            if should_replace {
                winner = Some((entry.id, params));
            }
        }

        match winner {
            Some((id, params)) => Ok(Match { route: id, params }),
            None => Err(Error::NotFound),
        }
    }
}

fn compile(pattern: &str) -> Result<Vec<Part>, Error> {
    if pattern.as_bytes().first() != Some(&b'/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let bytes = pattern.as_bytes();
    let mut parts = Vec::new();
    let mut captures = std::collections::HashSet::new();
    let mut pos = 1;

    loop {
        let end = bytes[pos..]
            .iter()
            .position(|b| *b == b'/')
            .map(|off| pos + off)
            .unwrap_or(bytes.len());

        if end == pos {
            return Err(Error::InvalidPattern { at: pos });
        }

        let slice = &pattern[pos..end];
        let first = slice.as_bytes()[0];
        if first == b':' {
            let name = &slice[1..];
            let valid = name
                .as_bytes()
                .first()
                .is_some_and(u8::is_ascii_lowercase)
                && name
                    .as_bytes()
                    .iter()
                    .skip(1)
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_');
            if !valid || !captures.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: pos });
            }
            parts.push(Part::Capture(name.to_string()));
        } else if first == b'*' {
            return Err(Error::InvalidPattern { at: pos });
        } else {
            parts.push(Part::Literal(decode_static(slice, pos)?));
        }

        if end == bytes.len() {
            break;
        }
        pos = end + 1;
        if pos == bytes.len() {
            return Err(Error::InvalidPattern { at: end });
        }
    }

    Ok(parts)
}

fn request_segments(path: &str) -> Result<Vec<String>, Error> {
    if path.as_bytes().first() != Some(&b'/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut values = Vec::new();
    let mut pos = 1;

    loop {
        let end = bytes[pos..]
            .iter()
            .position(|b| *b == b'/')
            .map(|off| pos + off)
            .unwrap_or(bytes.len());

        if end == pos {
            return Err(Error::InvalidPath { at: pos });
        }

        values.push(decode_request(&path[pos..end], pos)?);

        if end == bytes.len() {
            break;
        }
        pos = end + 1;
        if pos == bytes.len() {
            return Err(Error::InvalidPath { at: end });
        }
    }

    Ok(values)
}

fn decode_static(segment: &str, base: usize) -> Result<String, Error> {
    decode(segment, base, true)
}

fn decode_request(segment: &str, base: usize) -> Result<String, Error> {
    decode(segment, base, false)
}

fn decode(segment: &str, base: usize, pattern: bool) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        match bytes[i] {
            b'%' => {
                if i + 2 >= bytes.len() {
                    return Err(invalid(pattern, base + i));
                }
                let h = hex(bytes[i + 1]).ok_or_else(|| invalid(pattern, base + i))?;
                let l = hex(bytes[i + 2]).ok_or_else(|| invalid(pattern, base + i))?;
                let decoded = h * 16 + l;
                if !(0x20..=0x7e).contains(&decoded)
                    || decoded == b'/'
                    || unreserved(decoded)
                {
                    return Err(invalid(pattern, base + i));
                }
                out.push(decoded);
                i += 3;
            }
            b if unreserved(b) => {
                out.push(b);
                i += 1;
            }
            _ => return Err(invalid(pattern, base + i)),
        }
    }
    String::from_utf8(out).map_err(|_| invalid(pattern, base))
}

fn invalid(pattern: bool, at: usize) -> Error {
    if pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    }
}

fn hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn specificity(parts: &[Part]) -> Vec<u8> {
    parts
        .iter()
        .map(|part| match part {
            Part::Literal(_) => 2,
            Part::Capture(_) => 1,
        })
        .collect()
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let bytes = input.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| start + n)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &input[start..end];
        if raw.starts_with(':') {
            let name = &raw[1..];
            if !valid_capture_name(name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if parts.iter().any(|seg| matches!(seg, Segment::Param(existing) if existing == name)) {
                return Err(Error::InvalidPattern { at: start });
            }
            parts.push(Segment::Param(name.to_owned()));
        } else if raw.starts_with('*') {
            return Err(Error::InvalidPattern { at: start });
        } else {
            let decoded = decode_static_segment(raw, start).map_err(|at| Error::InvalidPattern { at })?;
            parts.push(Segment::Static(decoded));
        }

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(Error::InvalidPattern { at: bytes.len() - 1 });
        }
    }

    Ok(parts)
}

                .segments
                .iter()
                .map(|segment| match segment {
                    Segment::Static(_) => 2,
                    Segment::Param(_) => 1,

                .collect();

                None => true,
                Some((best_id, _, best_params)) => {

fn parse_encoded_segment(input: &str, start: usize, pattern: bool) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(if pattern {
                    Error::InvalidPattern { at: start + i }
                } else {
                    Error::InvalidPath { at: start + i }
                });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            let value = match (hi, lo) {
                (Some(hi), Some(lo)) => (hi << 4) | lo,
                _ => {
                    return Err(if pattern {
                        Error::InvalidPattern { at: start + i }
                    } else {
                        Error::InvalidPath { at: start + i }
                    })
                }
            };
            if !(0x20..=0x7e).contains(&value) || value == b'/' || is_unreserved(value) {
                return Err(if pattern {
                    Error::InvalidPattern { at: start + i }
                } else {
                    Error::InvalidPath { at: start + i }
                });
            }
            decoded.push(value);
            i += 3;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: start + i }
            } else {
                Error::InvalidPath { at: start + i }
            });
        }
    }

    Ok(String::from_utf8(decoded).unwrap())
}

fn parse_pattern(pattern: &str) -> Result<Route, Error> {

        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Route { segments: Vec::new() });
    }

    let mut end = bytes.len();
    let mut trailing = false;
    if bytes.last() == Some(&b'/') {
        trailing = true;
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut names = Vec::new();
    let mut start = 1;

    while start < end {
        let segment_end = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map_or(end, |offset| start + offset);
        if segment_end == start {
            return Err(Error::InvalidPattern { at: start });
        }
        let raw = &pattern[start..segment_end];
        if raw.starts_with(':') {
            let name = &raw[1..];
            let valid = !name.is_empty()
                && name.as_bytes()[0].is_ascii_lowercase()
                && name
                    .as_bytes()
                    .iter()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_');
            if !valid || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_owned());
            segments.push(Segment::Param(name.to_owned()));
        } else if raw.starts_with('*') {
            return Err(Error::InvalidPattern { at: start });
        } else {
            segments.push(Segment::Static(parse_encoded_segment(
                raw, start, true,
            )?));
        }
        start = segment_end + 1;
    }

    if trailing {
        return Err(Error::InvalidPattern { at: bytes.len() - 1 });
    }

    Ok(Route { segments })
}

        for route in &self.routes {
            if route.segments.len() != decoded.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matched = true;
            for (pat, value) in route.segments.iter().zip(&decoded) {
                match pat {
                    Segment::Static(expected) if expected == value => {}
                    Segment::Static(_) => {
                        matched = false;
                        break;
                    }
                    Segment::Param(name) => params.push((name.clone(), value.clone())),
                }
            }

            if matched {
                let replace = match &best {
                    None => true,
                    Some((old, _)) => route.rank > old.rank,
                };
                if replace {
                    best = Some((route, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match { route: route.id, params }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let bytes = pattern.as_bytes();
    let mut result = Vec::new();
    let mut names = std::collections::HashSet::<String>::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| start + n)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..end];
        if raw.as_bytes()[0] == b':' {
            let name = &raw[1..];
            let nb = name.as_bytes();
            let valid = !nb.is_empty()
                && nb[0].is_ascii_lowercase()
                && nb[1..]
                    .iter()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_');
            if !valid || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            result.push(Segment::Param(name.to_owned()));
        } else if raw.as_bytes()[0] == b'*' {
            return Err(Error::InvalidPattern { at: start });
        } else {
            result.push(Segment::Static(decode_segment(raw, start, true)?));
        }

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(Error::InvalidPattern { at: bytes.len() - 1 });
        }
    }

    Ok(result)
}

fn parse_path(head: &str, original: &str) -> Result<Vec<String>, Error> {
    if !head.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if head == "/" {
        return Ok(Vec::new());
    }

    let bytes = head.as_bytes();
    let mut result = Vec::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| start + n)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPath {
                at: original_offset(original, head, start),
            });
        }

        let raw = &head[start..end];
        result.push(
            decode_segment(raw, start, false)
                .map_err(|e| shift_error(e, original, head))?,
        );

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(Error::InvalidPath {
                at: original_offset(original, head, bytes.len() - 1),
            });
        }
    }

    Ok(result)
}

fn decode_segment(raw: &str, base: usize, _pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(Error::InvalidPattern { at: base + i });
            }
            let hi = upper_hex(bytes[i + 1]);
            let lo = upper_hex(bytes[i + 2]);
            if hi.is_none() || lo.is_none() {
                return Err(Error::InvalidPattern { at: base + i });
            }
            let decoded = (hi.unwrap() << 4) | lo.unwrap();
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(Error::InvalidPattern { at: base + i });
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(Error::InvalidPattern { at: base + i });
        }
    }

    String::from_utf8(out).map_err(|_| Error::InvalidPattern { at: base })
}

fn shift_error(err: Error, original: &str, head: &str) -> Error {
    match err {
        Error::InvalidPattern { at } | Error::InvalidPath { at } => Error::InvalidPath {
            at: original_offset(original, head, at),
        },
        Error::NotFound => Error::NotFound,
    }
}

fn original_offset(original: &str, head: &str, at: usize) -> usize {
    let _ = original;
    at.min(head.len())
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}
