    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

const LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn rounded_length_milli(a: Point, b: Point) -> u64 {
    let dx = (a.x as i128 - b.x as i128) as u128;
    let dy = (a.y as i128 - b.y as i128) as u128;
    let n = (dx * dx + dy * dy) * 1_000_000;
    let mut lo = 0u128;
    let mut hi = 3_000_000_001u128;
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid * mid <= n { lo = mid; } else { hi = mid; }
    }
    if 4 * n >= (2 * lo + 1) * (2 * lo + 1) { (lo + 1) as u64 } else { lo as u64 }
}

fn endpoint_key(a: Point, b: Point) -> (Point, Point) {
    if a <= b { (a, b) } else { (b, a) }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, p) in points.iter().enumerate() {
        if p.x < -LIMIT || p.x > LIMIT || p.y < -LIMIT || p.y > LIMIT {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut n = points.len();
    if n >= 2 && points[n - 1] == points[0] { n -= 1; }
    if n < 3 { return Err(GeomError::TooFewVertices { len: n }); }

    let mut signed_area2 = 0i128;
    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }

    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    let mut edge_milli = Vec::with_capacity(n);
    let mut perimeter_milli = 0u64;
    let mut longest_edge = None;
    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        let length = rounded_length_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let replace = match longest_edge {
            None => true,
            Some(j) => {

                length > old || (length == old &&
                    (endpoint_key(a, b) < endpoint_key(points[j], points[(j + 1) % n]) ||
                     (endpoint_key(a, b) == endpoint_key(points[j], points[(j + 1) % n]) && i < j)))
            }
        };
        if replace { longest_edge = Some(i); }
    }

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point { pub x: i64, pub y: i64 }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation { Clockwise, Degenerate, CounterClockwise }

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

const LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn isqrt(n: u128) -> u128 {
    if n < 2 { return n; }
    let mut x = n;
    let mut y = (x + 1) / 2;
    while y < x {
        x = y;
        y = (x + n / x) / 2;
    }
    x
}

fn length_milli(a: Point, b: Point) -> u64 {
    let dx = (a.x as i128 - b.x as i128) as u128;
    let dy = (a.y as i128 - b.y as i128) as u128;
    let n = (dx * dx + dy * dy) * 1_000_000;
    let r = isqrt(n);
    if 4 * (n - r * r) >= 4 * r + 1 { (r + 1) as u64 } else { r as u64 }
}

fn key(a: Point, b: Point) -> (Point, Point) {
    if a <= b { (a, b) } else { (b, a) }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let raw_len = points.len();
    if raw_len > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: raw_len });
    }
    for (index, point) in points.iter().enumerate() {
        if !(-LIMIT..=LIMIT).contains(&point.x) || !(-LIMIT..=LIMIT).contains(&point.y) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let n = raw_len - usize::from(raw_len >= 2 && points[raw_len - 1] == points[0]);
    if n < 3 { return Err(GeomError::TooFewVertices { len: n }); }

    let mut area = 0i128;
    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        area += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    let orientation = match area.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    let mut lengths = Vec::with_capacity(n);
    let mut perimeter = 0u64;
    let mut best: Option<usize> = None;
    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        let len = length_milli(a, b);
        lengths.push(len);
        perimeter += len;
        if let Some(j) = best {
            if len > lengths[j] || (len == lengths[j] &&
                (key(a, b), i) < (key(points[j], points[(j + 1) % n]), j)) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }

    Ok(Measures {
        signed_area2: area,
        area2: area.unsigned_abs(),
        orientation,
        edge_milli: lengths,
        perimeter_milli: perimeter,
        longest_edge: best,
    })
}
