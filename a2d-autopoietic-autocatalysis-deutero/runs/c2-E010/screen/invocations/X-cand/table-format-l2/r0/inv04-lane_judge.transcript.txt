                expected: n,
                actual: cells.len(),

        return Err(TableError::AlignmentCount {
            expected: n,

                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column,
                        byte,

        append_data_row(&mut output, row_index);
        if options.header && row_index == 0 && rows.len() > 1 {
            append_border(&mut output);
        }
    }

}

pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    if let Some((row, _)) = rows.iter().enumerate().find(|(_, cells)| cells.is_empty()) {
        return Err(TableError::EmptyRow { row });
    }

    let columns = rows[0].len();
    if let Some((row, cells)) = rows
        .iter()
        .enumerate()
        .find(|(_, cells)| cells.len() != columns)
    {
        return Err(TableError::Ragged {
            row,
            expected: columns,
            actual: cells.len(),
        });
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }

    let mut contents = Vec::with_capacity(rows.len());
    for (row, cells) in rows.iter().enumerate() {
        let mut clean_row = Vec::with_capacity(columns);
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if !(ch as u32 >= 0x20 && ch as u32 <= 0x7e) {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }
            let end = cell
                .as_bytes()
                .iter()
                .rposition(|&b| b != b' ')
                .map_or(0, |i| i + 1);
            clean_row.push(&cell[..end]);
        }
        contents.push(clean_row);
    }

    let mut widths = vec![1usize; columns];
    for row in &contents {
        for (column, cell) in row.iter().enumerate() {
            widths[column] = std::cmp::max(widths[column], cell.as_bytes().len());
        }
    }

    for column in 0..columns {
        if widths[column] > 32 {
            return Err(TableError::TooWide {
                column,
                width: widths[column],
            });
        }
    }

    let mut result = String::new();
    let mut append_border = |dst: &mut String| {
        dst.push('+');
        for &width in &widths {
            dst.extend(std::iter::repeat_n('-', width + 2));
            dst.push('+');
        }
        dst.push('\n');
    };

    append_border(&mut result);
    for (row_index, row) in contents.iter().enumerate() {
        result.push('|');
        for (column, cell) in row.iter().enumerate() {
            let slack = widths[column] - cell.len();
            let (left, right) = match options.alignments.get(column) {
                None | Some(Alignment::Left) => (0, slack),
                Some(Alignment::Right) => (slack, 0),
                Some(Alignment::Center) => (slack / 2, (slack + 1) / 2),
            };

            result.push(' ');
            result.extend(std::iter::repeat_n(' ', left));
            result.push_str(cell);
            result.extend(std::iter::repeat_n(' ', right));
            result.push(' ');
            result.push('|');
        }
        result.push('\n');

        if options.header && row_index == 0 && contents.len() > 1 {
            append_border(&mut result);
        }
    }
    append_border(&mut result);
    Ok(result)
}

    pub header: bool,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.len() == 0 {
        return Ok(String::new());
    }

    let mut empty_row = None;
    for (i, row) in rows.iter().enumerate() {
        if row.len() == 0 {
            empty_row = Some(i);
            break;
        }
    }
    if let Some(row) = empty_row {
        return Err(TableError::EmptyRow { row });
    }

    let columns = rows[0].len();
    for (i, row) in rows.iter().enumerate() {
        if row.len() != columns {
            return Err(TableError::Ragged {
                row: i,
                expected: columns,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }

    let mut widths = vec![1usize; columns];
    let mut trimmed_rows: Vec<Vec<&str>> = Vec::with_capacity(rows.len());

    for (ri, row) in rows.iter().enumerate() {
        let mut trimmed = Vec::with_capacity(columns);
        for (ci, cell) in row.iter().enumerate() {
            if let Some((byte, _)) = cell.char_indices().find(|(_, c)| !(' '..='~').contains(c)) {
                return Err(TableError::InvalidChar {
                    row: ri,
                    column: ci,
                    byte,
                });
            }
            let value = cell.trim_end_matches(' ');
            widths[ci] = widths[ci].max(value.len());
            trimmed.push(value);
        }
        trimmed_rows.push(trimmed);
    }

    for ci in 0..columns {
        if widths[ci] > 32 {
            return Err(TableError::TooWide {
                column: ci,
                width: widths[ci],
            });
        }
    }

    let mut output = String::new();
    write_border(&mut output, &widths);

    for (ri, row) in trimmed_rows.iter().enumerate() {
        for ci in 0..columns {
            output.push('|');
            output.push(' ');
            let cell = row[ci];
            let spare = widths[ci] - cell.len();
            let align = if options.alignments.is_empty() {
                &Alignment::Left
            } else {
                &options.alignments[ci]
            };
            let (leading, trailing) = match align {
                Alignment::Left => (0, spare),
                Alignment::Right => (spare, 0),
                Alignment::Center => (spare / 2, spare - spare / 2),
            };
            for _ in 0..leading {
                output.push(' ');
            }
            output.push_str(cell);
            for _ in 0..trailing {
                output.push(' ');
            }
            output.push(' ');
        }
        output.push_str("|\n");

        if options.header && ri == 0 && trimmed_rows.len() > 1 {
            write_border(&mut output, &widths);
        }
    }

    write_border(&mut output, &widths);
    Ok(output)
}

fn write_border(out: &mut String, widths: &[usize]) {
    out.push('+');
    for width in widths {
        for _ in 0..(*width + 2) {
            out.push('-');
        }
        out.push('+');
    }
    out.push('\n');
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
    AlignmentCount { expected: usize, actual: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.len() == 0 {
        return Ok(String::new());
    }

    let mut row_index = 0usize;
    while row_index < rows.len() {
        if rows[row_index].len() == 0 {
            return Err(TableError::EmptyRow { row: row_index });
        }
        row_index += 1;
    }

    let column_count = rows[0].len();
    row_index = 0;
    while row_index < rows.len() {
        let actual = rows[row_index].len();
        if actual != column_count {
            return Err(TableError::Ragged {
                row: row_index,
                expected: column_count,
                actual,
            });
        }
        row_index += 1;
    }

    let align_count = options.alignments.len();
    if align_count != 0 && align_count != column_count {
        return Err(TableError::AlignmentCount {
            expected: column_count,
            actual: align_count,
        });
    }

    for (r, row) in rows.iter().enumerate() {
        for (c, cell) in row.iter().enumerate() {
            for (b, byte) in cell.as_bytes().iter().copied().enumerate() {
                if byte < 0x20 || byte > 0x7e {
                    return Err(TableError::InvalidChar {
                        row: r,
                        column: c,
                        byte: b,
                    });
                }
            }
        }
    }

    let mut widths = vec![1usize; column_count];
    for row in rows {
        for (column, cell) in row.iter().enumerate() {
            let mut length = cell.len();
            while length > 0 && cell.as_bytes()[length - 1] == b' ' {
                length -= 1;
            }
            widths[column] = widths[column].max(length);
        }
    }

    for (column, width) in widths.iter().copied().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let mut out = String::new();
    write_border(&mut out, &widths);

    for (r, row) in rows.iter().enumerate() {
        out.push('|');
        for c in 0..column_count {
            let cell = &row[c];
            let mut end = cell.len();
            while end > 0 && cell.as_bytes()[end - 1] == b' ' {
                end -= 1;
            }
            let content = &cell[..end];
            let gap = widths[c] - content.len();
            let alignment = if align_count == 0 {
                Alignment::Left
            } else {
                options.alignments[c].clone()
            };

            out.push(' ');
            match alignment {
                Alignment::Left => {
                    out.push_str(content);
                    push_spaces(&mut out, gap);
                }
                Alignment::Right => {
                    push_spaces(&mut out, gap);
                    out.push_str(content);
                }
                Alignment::Center => {
                    let left = gap / 2;
                    push_spaces(&mut out, left);
                    out.push_str(content);
                    push_spaces(&mut out, gap - left);
                }
            }
            out.push(' ');
            out.push('|');
        }
        out.push('\n');

        if options.header && r == 0 && rows.len() > 1 {
            write_border(&mut out, &widths);
        }
    }

    write_border(&mut out, &widths);
    Ok(out)
}

fn push_spaces(out: &mut String, count: usize) {
    for _ in 0..count {
        out.push(' ');
    }
}

fn write_border(out: &mut String, widths: &[usize]) {
    out.push('+');
    for width in widths {
        for _ in 0..(*width + 2) {
            out.push('-');
        }
        out.push('+');
    }
    out.push('\n');
}
