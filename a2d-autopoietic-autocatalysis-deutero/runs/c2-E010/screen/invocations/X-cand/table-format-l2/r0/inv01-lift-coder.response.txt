- Validation follows the specified precedence: empty rows, ragged rows, alignment count, invalid characters, then width.
- Character validity is ASCII `0x20..=0x7e`; trailing spaces are trimmed only after validation.
- Width is the maximum trimmed byte length, with a minimum of one; widths above 32 report the lowest offending column.
- Center alignment puts the odd extra space on the right.
- An empty table returns immediately, even if options would otherwise be invalid.
- Both implementations produce LF-terminated borders and data lines; Solution B uses a separate formatting approach.

### Solution A
```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
    AlignmentCount { expected: usize, actual: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    let mut contents = Vec::with_capacity(rows.len());
    for (row_index, cells) in rows.iter().enumerate() {
        let mut out_row = Vec::with_capacity(n);
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if !(' '..='~').contains(&ch) {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column,
                        byte,
                    });
                }
            }
            out_row.push(cell.trim_end_matches(' ').to_owned());
        }
        contents.push(out_row);
    }

    let mut widths = vec![1usize; n];
    for row in &contents {
        for (column, cell) in row.iter().enumerate() {
            widths[column] = widths[column].max(cell.len());
        }
    }
    for (column, &width) in widths.iter().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let alignment = |column: usize| -> &Alignment {
        if options.alignments.is_empty() {
            &Alignment::Left
        } else {
            &options.alignments[column]
        }
    };

    let mut result = String::new();
    append_border(&mut result, &widths);

    for (row_index, row) in contents.iter().enumerate() {
        for (column, cell) in row.iter().enumerate() {
            result.push('|');
            result.push(' ');
            let extra = widths[column] - cell.len();
            let (left, right) = match alignment(column) {
                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),
            };
            result.extend(std::iter::repeat_n(' ', left));
            result.push_str(cell);
            result.extend(std::iter::repeat_n(' ', right));
            result.push(' ');
        }
        result.push_str("|\n");

        if options.header && row_index == 0 && contents.len() > 1 {
            append_border(&mut result, &widths);
        }
    }

    append_border(&mut result, &widths);
    Ok(result)
}

fn append_border(out: &mut String, widths: &[usize]) {
    out.push('+');
    for &width in widths {
        out.extend(std::iter::repeat_n('-', width + 2));
        out.push('+');
    }
    out.push('\n');
}
```

### Solution B
```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
    AlignmentCount { expected: usize, actual: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.len() == 0 {
        return Ok(String::new());
    }

    let mut empty_row = None;
    for (i, row) in rows.iter().enumerate() {
        if row.len() == 0 {
            empty_row = Some(i);
            break;
        }
    }
    if let Some(row) = empty_row {
        return Err(TableError::EmptyRow { row });
    }

    let columns = rows[0].len();
    for (i, row) in rows.iter().enumerate() {
        if row.len() != columns {
            return Err(TableError::Ragged {
                row: i,
                expected: columns,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }

    let mut widths = vec![1usize; columns];
    let mut trimmed_rows: Vec<Vec<&str>> = Vec::with_capacity(rows.len());

    for (ri, row) in rows.iter().enumerate() {
        let mut trimmed = Vec::with_capacity(columns);
        for (ci, cell) in row.iter().enumerate() {
            if let Some((byte, _)) = cell.char_indices().find(|(_, c)| !(' '..='~').contains(c)) {
                return Err(TableError::InvalidChar {
                    row: ri,
                    column: ci,
                    byte,
                });
            }
            let value = cell.trim_end_matches(' ');
            widths[ci] = widths[ci].max(value.len());
            trimmed.push(value);
        }
        trimmed_rows.push(trimmed);
    }

    for ci in 0..columns {
        if widths[ci] > 32 {
            return Err(TableError::TooWide {
                column: ci,
                width: widths[ci],
            });
        }
    }

    let mut output = String::new();
    write_border(&mut output, &widths);

    for (ri, row) in trimmed_rows.iter().enumerate() {
        for ci in 0..columns {
            output.push('|');
            output.push(' ');
            let cell = row[ci];
            let spare = widths[ci] - cell.len();
            let align = if options.alignments.is_empty() {
                &Alignment::Left
            } else {
                &options.alignments[ci]
            };
            let (leading, trailing) = match align {
                Alignment::Left => (0, spare),
                Alignment::Right => (spare, 0),
                Alignment::Center => (spare / 2, spare - spare / 2),
            };
            for _ in 0..leading {
                output.push(' ');
            }
            output.push_str(cell);
            for _ in 0..trailing {
                output.push(' ');
            }
            output.push(' ');
        }
        output.push_str("|\n");

        if options.header && ri == 0 && trimmed_rows.len() > 1 {
            write_border(&mut output, &widths);
        }
    }

    write_border(&mut output, &widths);
    Ok(output)
}

fn write_border(out: &mut String, widths: &[usize]) {
    out.push('+');
    for width in widths {
        for _ in 0..(*width + 2) {
            out.push('-');
        }
        out.push('+');
    }
    out.push('\n');
}
```
