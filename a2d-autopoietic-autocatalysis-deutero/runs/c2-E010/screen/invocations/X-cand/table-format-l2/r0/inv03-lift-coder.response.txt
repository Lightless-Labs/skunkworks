- Empty tables return before inspecting options; for nonempty input, empty rows take precedence over ragged rows.
- Character validation scans original UTF-8 bytes in row-major order; only printable ASCII is accepted, and trailing spaces are trimmed afterward.
- Widths use trimmed byte lengths, have a minimum of 1, and the lowest over-wide column is reported after all character checks.
- Alignment vectors must be empty or match the column count; empty means all Left. Center puts the odd extra space on the right.
- Borders and data lines end in LF; a header separator appears only when a body row exists.
- The output uses checked width arithmetic and `String` formatting for valid inputs.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
    AlignmentCount { expected: usize, actual: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let column_count = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != column_count {
            return Err(TableError::Ragged {
                row: row_index,
                expected: column_count,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != column_count {
        return Err(TableError::AlignmentCount {
            expected: column_count,
            actual: options.alignments.len(),
        });
    }

    let mut contents = Vec::with_capacity(rows.len());
    let mut widths = vec![1usize; column_count];

    for (row_index, row) in rows.iter().enumerate() {
        let mut rendered_row = Vec::with_capacity(column_count);
        for (column_index, cell) in row.iter().enumerate() {
            for (byte_index, byte) in cell.bytes().enumerate() {
                if !(0x20..=0x7e).contains(&byte) {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte: byte_index,
                    });
                }
            }

            let trimmed_len = cell.trim_end_matches(' ').len();
            widths[column_index] = widths[column_index].max(trimmed_len);
            rendered_row.push(cell[..trimmed_len].to_owned());
        }
        contents.push(rendered_row);
    }

    for (column, &width) in widths.iter().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let mut output = String::new();

    let append_border = |out: &mut String| {
        out.push('+');
        for &width in &widths {
            for _ in 0..(width + 2) {
                out.push('-');
            }
            out.push('+');
        }
        out.push('\n');
    };

    let append_data_row = |out: &mut String, row_index: usize| {
        out.push('|');
        for column in 0..column_count {
            let content = &contents[row_index][column];
            let extra = widths[column] - content.len();
            let alignment = if options.alignments.is_empty() {
                &Alignment::Left
            } else {
                &options.alignments[column]
            };

            let (left, right) = match alignment {
                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),
            };

            out.push(' ');
            for _ in 0..left {
                out.push(' ');
            }
            out.push_str(content);
            for _ in 0..right {
                out.push(' ');
            }
            out.push(' ');
            out.push('|');
        }
        out.push('\n');
    };

    append_border(&mut output);
    for row_index in 0..rows.len() {
        append_data_row(&mut output, row_index);
        if options.header && row_index == 0 && rows.len() > 1 {
            append_border(&mut output);
        }
    }
    append_border(&mut output);

    Ok(output)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
    AlignmentCount { expected: usize, actual: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let columns = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != columns {
            return Err(TableError::Ragged {
                row: row_index,
                expected: columns,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }

    let mut trimmed_rows: Vec<Vec<&str>> = Vec::with_capacity(rows.len());
    let mut widths = vec![1usize; columns];

    for (row_index, row) in rows.iter().enumerate() {
        let mut trimmed_row = Vec::with_capacity(columns);
        for (column_index, cell) in row.iter().enumerate() {
            for (byte_index, byte) in cell.bytes().enumerate() {
                if !(b' '..=b'~').contains(&byte) {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte: byte_index,
                    });
                }
            }

            let trimmed = cell.trim_end_matches(' ');
            widths[column_index] = widths[column_index].max(trimmed.len());
            trimmed_row.push(trimmed);
        }
        trimmed_rows.push(trimmed_row);
    }

    for (column, width) in widths.iter().copied().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    fn border(out: &mut String, widths: &[usize]) {
        for (i, width) in widths.iter().enumerate() {
            if i == 0 {
                out.push('+');
            }
            for _ in 0..(width + 2) {
                out.push('-');
            }
            out.push('+');
        }
        out.push('\n');
    }

    fn data_row(
        out: &mut String,
        row: &[&str],
        widths: &[usize],
        alignments: &[Alignment],
    ) {
        out.push('|');
        for (column, content) in row.iter().enumerate() {
            let extra = widths[column] - content.len();
            let alignment = alignments.get(column).unwrap_or(&Alignment::Left);
            let left = match alignment {
                Alignment::Left => 0,
                Alignment::Right => extra,
                Alignment::Center => extra / 2,
            };
            let right = extra - left;

            out.push(' ');
            for _ in 0..left {
                out.push(' ');
            }
            out.push_str(content);
            for _ in 0..right {
                out.push(' ');
            }
            out.push(' ');
            out.push('|');
        }
        out.push('\n');
    }

    let mut output = String::new();
    border(&mut output, &widths);

    for (row_index, row) in trimmed_rows.iter().enumerate() {
        data_row(&mut output, row, &widths, &options.alignments);
        if options.header && row_index == 0 && trimmed_rows.len() > 1 {
            border(&mut output, &widths);
        }
    }

    border(&mut output, &widths);
    Ok(output)
}
```
