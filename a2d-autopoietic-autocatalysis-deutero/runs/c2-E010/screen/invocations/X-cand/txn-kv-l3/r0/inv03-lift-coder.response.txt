- **Clock and expiry order:** A backward time returns immediately; otherwise the clock advances and expired entries are purged from the visible map before method-specific validation.
- **Scope visibility:** Only the innermost map is visible. Commit replaces its parent; rollback reveals the parent and purges it at the admitted time.
- **Names and keys:** Names use the exact ASCII grammar and byte offsets; empty keys are rejected before key access or TTL arithmetic.
- **Ranges and cursors:** End is exclusive; invalid ranges precede cursor checks. A cursor resumes strictly after `last` and does not reapply the offset.
- **Numbers and sums:** Only canonical signed `i64` decimals are accepted; entries are parsed and added in ascending key order, with the first error returned.
- **Solution B:** Uses an independent map representation while preserving the same specified behavior.

### Solution A

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    NoTransaction,
    DepthLimit,
    SnapshotMissing(String),
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

type Entry = (String, Option<u64>);
type Data = BTreeMap<String, Entry>;

pub struct Store {
    committed: Data,
    scopes: Vec<Data>,
    snapshots: BTreeMap<String, Data>,
    clock: u64,
}

impl Store {
    pub fn new() -> Self {
        Self {
            committed: BTreeMap::new(),
            scopes: Vec::new(),
            snapshots: BTreeMap::new(),
            clock: 0,
        }
    }

    fn visible(&self) -> &Data {
        self.scopes.last().unwrap_or(&self.committed)
    }

    fn visible_mut(&mut self) -> &mut Data {
        self.scopes.last_mut().unwrap_or(&mut self.committed)
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        let map = self.visible_mut();
        map.retain(|_, (_, deadline)| deadline.map_or(true, |d| d > now));
        Ok(())
    }

    fn validate_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (i, b) in name.bytes().enumerate() {
            let valid = if i == 0 {
                b.is_ascii_lowercase()
            } else {
                b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_'
            };
            if !valid {
                return Err(Error::NameInvalid { at: i });
            }
        }
        Ok(())
    }

    fn matches(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.map_or(true, |e| key < e)
            && prefix.map_or(true, |p| key.starts_with(p))
    }

    pub fn begin(&mut self, now: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.scopes.len() >= 4 {
            return Err(Error::DepthLimit);
        }
        self.scopes.push(self.visible().clone());
        Ok(())
    }

    pub fn commit(&mut self, now: u64) -> Result<(), Error> {
        self.admit(now)?;
        let map = self.scopes.pop().ok_or(Error::NoTransaction)?;
        if let Some(parent) = self.scopes.last_mut() {
            *parent = map;
        } else {
            self.committed = map;
        }
        Ok(())
    }

    pub fn rollback(&mut self, now: u64) -> Result<(), Error> {
        self.admit(now)?;
        self.scopes.pop().ok_or(Error::NoTransaction)?;
        let map = self.visible_mut();
        map.retain(|_, (_, deadline)| deadline.map_or(true, |d| d > now));
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        let deadline = match ttl {
            None => None,
            Some(d) => Some(now.checked_add(d).ok_or(Error::DeadlineOverflow)?),
        };
        if deadline == Some(now) {
            self.visible_mut().remove(key);
        } else {
            self.visible_mut()
                .insert(key.to_owned(), (value.to_owned(), deadline));
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self.visible_mut().remove(key).is_some())
    }

    pub fn get(&mut self, now: u64, key: &str) -> Result<Option<String>, Error> {
        self.admit(now)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self.visible().get(key).map(|(value, _)| value.clone()))
    }

    pub fn count(&mut self, now: u64, value: &str) -> Result<usize, Error> {
        self.admit(now)?;
        Ok(self.visible().values().filter(|(v, _)| v == value).count())
    }

    pub fn depth(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        Ok(self.scopes.len())
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        self.snapshots.insert(name.to_owned(), self.visible().clone());
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        let map = self
            .snapshots
            .get(name)
            .cloned()
            .ok_or_else(|| Error::SnapshotMissing(name.to_owned()))?;
        let now = self.clock;
        let map: Data = map
            .into_iter()
            .filter(|(_, (_, deadline))| deadline.map_or(true, |d| d > now))
            .collect();
        *self.visible_mut() = map;
        Ok(())
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        self.snapshots
            .remove(name)
            .map(|_| ())
            .ok_or_else(|| Error::SnapshotMissing(name.to_owned()))
    }

    pub fn scan(
        &mut self,
        now: u64,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        if end.is_some_and(|e| e <= start) {
            return Err(Error::InvalidRange);
        }
        if let Some(c) = cursor {
            if c.start != start
                || c.end.as_deref() != end
                || c.prefix.as_deref() != prefix
                || c.offset != offset
                || c.last.is_empty()
                || !Self::matches(&c.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }
        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }

        let mut selected = Vec::new();
        let mut skipped = 0usize;
        for (key, (value, _)) in self.visible() {
            if !Self::matches(key, start, end, prefix) {
                continue;
            }
            if let Some(c) = cursor {
                if key <= &c.last {
                    continue;
                }
            } else if skipped < offset {
                skipped += 1;
                continue;
            }
            selected.push((key.clone(), value.clone()));
            if selected.len() > limit {
                break;
            }
        }

        let has_more = selected.len() > limit;
        if has_more {
            selected.pop();
        }
        let next = if has_more {
            selected.last().map(|(last, _)| Cursor {
                start: start.to_owned(),
                end: end.map(str::to_owned),
                prefix: prefix.map(str::to_owned),
                offset,
                last: last.clone(),
            })
        } else {
            None
        };
        Ok(Page {
            items: selected,
            next,
        })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        if end.is_some_and(|e| e <= start) {
            return Err(Error::InvalidRange);
        }
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, (value, _)) in self.visible() {
            if !Self::matches(key, start, end, prefix) {
                continue;
            }
            let parsed = parse_number(value).ok_or_else(|| Error::InvalidNumber {
                key: key.clone(),
                value: value.clone(),
            })?;
            sum = sum
                .checked_add(parsed)
                .ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }
}

fn parse_number(value: &str) -> Option<i64> {
    if value == "0" {
        return Some(0);
    }
    let digits = value.strip_prefix('-').unwrap_or(value);
    if digits.is_empty()
        || digits.starts_with('0')
        || !digits.bytes().all(|b| b.is_ascii_digit())
        || value == "-0"
    {
        return None;
    }
    if value.starts_with('-') && digits == "0" {
        return None;
    }
    value.parse::<i64>().ok()
}
```

### Solution B

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    NoTransaction,
    DepthLimit,
    SnapshotMissing(String),
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

type Data = BTreeMap<String, (String, Option<u64>)>;

pub struct Store {
    layers: Vec<Data>,
    snapshots: BTreeMap<String, Data>,
    time: u64,
}

impl Store {
    pub fn new() -> Self {
        Self {
            layers: vec![BTreeMap::new()],
            snapshots: BTreeMap::new(),
            time: 0,
        }
    }

    fn current(&self) -> &Data {
        self.layers.last().unwrap()
    }

    fn current_mut(&mut self) -> &mut Data {
        self.layers.last_mut().unwrap()
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.time {
            return Err(Error::TimeBackwards {
                previous: self.time,
                now,
            });
        }
        self.time = now;
        self.current_mut()
            .retain(|_, (_, expiry)| expiry.map_or(true, |d| d > now));
        Ok(())
    }

    fn valid_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (i, b) in name.bytes().enumerate() {
            if !(if i == 0 {
                b.is_ascii_lowercase()
            } else {
                b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_'
            }) {
                return Err(Error::NameInvalid { at: i });
            }
        }
        Ok(())
    }

    fn matches(k: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        k >= start && end.map_or(true, |e| k < e) && prefix.map_or(true, |p| k.starts_with(p))
    }

    pub fn begin(&mut self, now: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.layers.len() - 1 >= 4 {
            return Err(Error::DepthLimit);
        }
        self.layers.push(self.current().clone());
        Ok(())
    }

    pub fn commit(&mut self, now: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.layers.len() == 1 {
            return Err(Error::NoTransaction);
        }
        let finished = self.layers.pop().unwrap();
        *self.current_mut() = finished;
        Ok(())
    }

    pub fn rollback(&mut self, now: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.layers.len() == 1 {
            return Err(Error::NoTransaction);
        }
        self.layers.pop();
        self.current_mut()
            .retain(|_, (_, expiry)| expiry.map_or(true, |d| d > now));
        Ok(())
    }

    pub fn set(&mut self, now: u64, key: &str, value: &str, ttl: Option<u64>) -> Result<(), Error> {
        self.admit(now)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        match ttl {
            Some(d) => {
                let expiry = now.checked_add(d).ok_or(Error::DeadlineOverflow)?;
                if expiry == now {
                    self.current_mut().remove(key);
                } else {
                    self.current_mut().insert(key.into(), (value.into(), Some(expiry)));
                }
            }
            None => {
                self.current_mut().insert(key.into(), (value.into(), None));
            }
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self.current_mut().remove(key).is_some())
    }

    pub fn get(&mut self, now: u64, key: &str) -> Result<Option<String>, Error> {
        self.admit(now)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self.current().get(key).map(|(v, _)| v.clone()))
    }

    pub fn count(&mut self, now: u64, value: &str) -> Result<usize, Error> {
        self.admit(now)?;
        Ok(self.current().values().filter(|(v, _)| v == value).count())
    }

    pub fn depth(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        Ok(self.layers.len() - 1)
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::valid_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        self.snapshots.insert(name.into(), self.current().clone());
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::valid_name(name)?;
        let saved = self.snapshots.get(name).cloned().ok_or_else(|| {
            Error::SnapshotMissing(name.into())
        })?;
        let live = saved
            .into_iter()
            .filter(|(_, (_, expiry))| expiry.map_or(true, |d| d > now))
            .collect();
        *self.current_mut() = live;
        Ok(())
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::valid_name(name)?;
        if self.snapshots.remove(name).is_some() {
            Ok(())
        } else {
            Err(Error::SnapshotMissing(name.into()))
        }
    }

    pub fn scan(
        &mut self,
        now: u64,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        if end.map_or(false, |e| e <= start) {
            return Err(Error::InvalidRange);
        }
        if let Some(c) = cursor {
            if c.start != start
                || c.end.as_deref() != end
                || c.prefix.as_deref() != prefix
                || c.offset != offset
                || c.last.is_empty()
                || !Self::matches(&c.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }
        let mut items = Vec::new();
        let mut matching_seen = 0usize;
        let mut more = false;
        if limit > 0 {
            for (key, (value, _)) in self.current() {
                if !Self::matches(key, start, end, prefix) {
                    continue;
                }
                if let Some(c) = cursor {
                    if key <= &c.last {
                        continue;
                    }
                } else {
                    if matching_seen < offset {
                        matching_seen += 1;
                        continue;
                    }
                }
                if items.len() == limit {
                    more = true;
                    break;
                }
                items.push((key.clone(), value.clone()));
            }
        }
        let next = if more {
            items.last().map(|(last, _)| Cursor {
                start: start.to_owned(),
                end: end.map(str::to_owned),
                prefix: prefix.map(str::to_owned),
                offset,
                last: last.clone(),
            })
        } else {
            None
        };
        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        if end.map_or(false, |e| e <= start) {
            return Err(Error::InvalidRange);
        }
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, (value, _)) in self.current() {
            if !Self::matches(key, start, end, prefix) {
                continue;
            }
            let number = parse_number(value).ok_or_else(|| Error::InvalidNumber {
                key: key.clone(),
                value: value.clone(),
            })?;
            sum = sum.checked_add(number).ok_or_else(|| Error::SumOverflow {
                key: key.clone(),
            })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }
}

fn parse_number(value: &str) -> Option<i64> {
    if value == "0" {
        return Some(0);
    }
    let digits = value.strip_prefix('-').unwrap_or(value);
    if digits.is_empty() || digits.starts_with('0') || !digits.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    value.parse::<i64>().ok()
}
```
