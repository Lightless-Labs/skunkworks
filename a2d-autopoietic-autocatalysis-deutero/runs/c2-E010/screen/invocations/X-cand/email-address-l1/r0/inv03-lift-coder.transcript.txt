pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut separators = bytes.iter().enumerate().filter(|(_, &b)| b == b'@');
    let separator = match separators.next() {
        None => return Err(EmailError::MissingAt),
        Some((at, _)) => at,
    };
    if let Some((at, _)) = separators.next() {
        return Err(EmailError::MultipleAt { at });
    }

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local = &bytes[..separator];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    let allowed_specials = b"!#$%&'*+-/=?^_`{|}~";
    for (i, &b) in local.iter().enumerate() {
        let at = i;
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at });
            }
        } else if !(b.is_ascii_alphanumeric() || allowed_specials.contains(&b)) {
            return Err(EmailError::LocalChar { at });
        }
    }

    let domain_start = separator + 1;
    let mut domain_end = bytes.len();
    let mut removed_root_dot = None;
    if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        removed_root_dot = Some(domain_end);
    }

    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain = &bytes[domain_start..domain_end];
    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut label_count = 0usize;
    let mut label_start = 0usize;
    let mut labels = Vec::new();

    for i in 0..=domain.len() {
        if i == domain.len() || domain[i] == b'.' {
            let label = &domain[label_start..i];
            let absolute_start = domain_start + label_start;

            if label.is_empty() {
                let at = if i == domain.len() {
                    removed_root_dot.unwrap_or(domain_start + i)
                } else {
                    domain_start + i
                };
                return Err(EmailError::EmptyLabel { at });
            }

            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: absolute_start + 63,
                });
            }

            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: absolute_start });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: absolute_start + label.len() - 1,
                });
            }

            for (j, &b) in label.iter().enumerate() {
                if !(b.is_ascii_alphanumeric() || b == b'-') {
                    return Err(EmailError::DomainChar {
                        at: absolute_start + j,
                    });
                }
            }

            labels.push((label_start, i));
            label_count += 1;
            label_start = i + 1;
        }
    }

    if label_count < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = labels[label_count - 1];
    let last_label = &domain[last_start..last_end];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let local_string = String::from_utf8(local.to_vec()).expect("ASCII input");
    let domain_string = domain
        .iter()
        .map(|&b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local: local_string,
        domain: domain_string,
        quoted: false,
    })
}

    pub domain: String,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    for (at, &byte) in bytes.iter().enumerate() {
        if byte >= 128 {
            return Err(EmailError::NonAscii { at });
        }
    }

    let mut at_sign = None;
    for (at, &byte) in bytes.iter().enumerate() {
        if byte == b'@' {
            if at_sign.is_some() {
                return Err(EmailError::MultipleAt { at });
            }
            at_sign = Some(at);
        }
    }
    let at_sign = at_sign.ok_or(EmailError::MissingAt)?;
    if at_sign == 0 {
        return Err(EmailError::EmptyLocal { at: at_sign });
    }

    let local = &bytes[..at_sign];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    let specials = b"!#$%&'*+-/=?^_`{|}~";
    for (index, &byte) in local.iter().enumerate() {
        if byte == b'.' {
            if index == 0
                || index + 1 == local.len()
                || local.get(index.wrapping_sub(1)) == Some(&b'.')
            {
                return Err(EmailError::LocalDot { at: index });
            }
        } else if !byte.is_ascii_alphanumeric() && !specials.contains(&byte) {
            return Err(EmailError::LocalChar { at: index });
        }
    }

    let domain_start = at_sign + 1;
    let mut domain_end = bytes.len();
    let removed_dot = if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    let domain = &bytes[domain_start..domain_end];
    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut starts = Vec::new();
    let mut ends = Vec::new();
    let mut start = 0usize;

    for cursor in 0..=domain.len() {
        if cursor == domain.len() || domain[cursor] == b'.' {
            let label = &domain[start..cursor];
            let absolute_start = domain_start + start;

            if label.is_empty() {
                let error_at = if cursor == domain.len() {
                    removed_dot.unwrap_or(domain_start + cursor)
                } else {
                    domain_start + cursor
                };
                return Err(EmailError::EmptyLabel { at: error_at });
            }
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: absolute_start + 63,
                });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: absolute_start });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: absolute_start + label.len() - 1,
                });
            }
            for (offset, &byte) in label.iter().enumerate() {
                if !byte.is_ascii_alphanumeric() && byte != b'-' {
                    return Err(EmailError::DomainChar {
                        at: absolute_start + offset,
                    });
                }
            }

            starts.push(start);
            ends.push(cursor);
            start = cursor + 1;
        }
    }

    if starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let last_index = starts.len() - 1;
    let last_label = &domain[starts[last_index]..ends[last_index]];
    if last_label.iter().all(|byte| byte.is_ascii_digit()) {
        return Err(EmailError::NumericTld {
            at: domain_start + starts[last_index],
        });
    }

    let local_string = String::from_utf8(local.to_vec()).expect("ASCII validated");
    let mut normalized_domain = String::with_capacity(domain.len());
    for &byte in domain {
        normalized_domain.push(byte.to_ascii_lowercase() as char);
    }

    Ok(Address {
        local: local_string,
        domain: normalized_domain,
        quoted: false,
    })
}
