- Separator counting includes every `@`, including ones inside quotes.
- The 65th local byte and 254th domain byte are the reported length-error positions.
- A removed final dot still determines the position of an empty trailing label.
- Label length and edge-hyphen checks precede character checks.
- A one-label domain reports `DomainNeedsDot` before checking for a numeric final label.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut separators = bytes.iter().enumerate().filter(|(_, &b)| b == b'@');
    let separator = match separators.next() {
        None => return Err(EmailError::MissingAt),
        Some((at, _)) => at,
    };
    if let Some((at, _)) = separators.next() {
        return Err(EmailError::MultipleAt { at });
    }

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local = &bytes[..separator];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at: i });
            }
        } else if !b.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
        {
            return Err(EmailError::LocalChar { at: i });
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(has_root_dot);
    let domain = &raw_domain[..domain_len];

    if domain.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut label_starts = Vec::new();
    let mut start = 0;

    for end in 0..=domain.len() {
        if end == domain.len() || domain[end] == b'.' {
            if end == start {
                let at = if has_root_dot && end == domain.len() {
                    input.len() - 1
                } else {
                    domain_start + start
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let label = &domain[start..end];
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: domain_start + start + 63,
                });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: domain_start + start,
                });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: domain_start + end - 1,
                });
            }
            for (offset, &b) in label.iter().enumerate() {
                if !b.is_ascii_alphanumeric() && b != b'-' {
                    return Err(EmailError::DomainChar {
                        at: domain_start + start + offset,
                    });
                }
            }
            label_starts.push((start, end));
            start = end + 1;
        }
    }

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = *label_starts.last().unwrap();
    if domain[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let local_string = String::from_utf8(local.to_vec()).unwrap();
    let domain_string = domain
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local: local_string,
        domain: domain_string,
        quoted: false,
    })
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    for (i, &b) in bytes.iter().enumerate() {
        if b >= 128 {
            return Err(EmailError::NonAscii { at: i });
        }
    }

    let mut at = None;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'@' {
            if at.is_some() {
                return Err(EmailError::MultipleAt { at: i });
            }
            at = Some(i);
        }
    }
    let sep = at.ok_or(EmailError::MissingAt)?;
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let local = &bytes[..sep];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }
    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at: i });
            }
        } else if !(b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)) {
            return Err(EmailError::LocalChar { at: i });
        }
    }

    let domain_start = sep + 1;
    let raw = &bytes[domain_start..];
    let root_removed = raw.last() == Some(&b'.');
    let content_len = raw.len() - usize::from(root_removed);
    if content_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if content_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain = &raw[..content_len];
    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut begin = 0;
    for i in 0..=content_len {
        if i == content_len || domain[i] == b'.' {
            if begin == i {
                let position = if root_removed && i == content_len {
                    input.len() - 1
                } else {
                    domain_start + begin
                };
                return Err(EmailError::EmptyLabel { at: position });
            }

            let len = i - begin;
            if len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: domain_start + begin + 63,
                });
            }
            if domain[begin] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: domain_start + begin,
                });
            }
            if domain[i - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: domain_start + i - 1,
                });
            }
            for j in begin..i {
                let b = domain[j];
                if !(b.is_ascii_alphanumeric() || b == b'-') {
                    return Err(EmailError::DomainChar {
                        at: domain_start + j,
                    });
                }
            }
            labels.push((begin, i));
            begin = i + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_begin, last_end) = labels[labels.len() - 1];
    let mut numeric = true;
    for &b in &domain[last_begin..last_end] {
        if !b.is_ascii_digit() {
            numeric = false;
            break;
        }
    }
    if numeric {
        return Err(EmailError::NumericTld {
            at: domain_start + last_begin,
        });
    }

    let mut lowered_domain = String::with_capacity(content_len);
    for &b in domain {
        lowered_domain.push((b.to_ascii_lowercase()) as char);
    }

    Ok(Address {
        local: String::from_utf8(local.to_vec()).unwrap(),
        domain: lowered_domain,
        quoted: false,
    })
}
```
