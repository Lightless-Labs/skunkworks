- Check raw emptiness and then the first non-ASCII byte before looking for separators.
- Count every `@`; report the second one before validating either side.
- Check local length before scanning local bytes; detect a bad dot when it is encountered.
- Strip one terminal domain dot, but retain its original offset for an empty final label.
- Check total domain length before labels; within each label, length and edge hyphens precede character checks.
- Require two labels before checking whether the final label is numeric.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut separators = bytes.iter().enumerate().filter(|(_, &b)| b == b'@');
    let separator = match separators.next() {
        Some((at, _)) => at,
        None => return Err(EmailError::MissingAt),
    };
    if let Some((at, _)) = separators.next() {
        return Err(EmailError::MultipleAt { at });
    }

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local = &bytes[..separator];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (i, &b) in local.iter().enumerate() {
        let original_at = i;
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at: original_at });
            }
        } else if !b.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
        {
            return Err(EmailError::LocalChar { at: original_at });
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_end = if has_root_dot {
        input.len() - 1
    } else {
        input.len()
    };
    let domain = &bytes[domain_start..domain_end];
    let root_dot_at = if has_root_dot { Some(domain_end) } else { None };

    if domain.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut label_starts = Vec::new();
    let mut label_start = 0;
    for i in 0..=domain.len() {
        if i == domain.len() || domain[i] == b'.' {
            let label = &domain[label_start..i];
            let absolute_start = domain_start + label_start;

            if label.is_empty() {
                let at = if label_start == domain.len() {
                    root_dot_at.unwrap_or(domain_start + label_start)
                } else {
                    absolute_start
                };
                return Err(EmailError::EmptyLabel { at });
            }

            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: absolute_start + 63,
                });
            }

            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: absolute_start });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: absolute_start + label.len() - 1,
                });
            }

            for (j, &b) in label.iter().enumerate() {
                if !b.is_ascii_alphanumeric() && b != b'-' {
                    return Err(EmailError::DomainChar {
                        at: absolute_start + j,
                    });
                }
            }

            label_starts.push((label_start, i));
            label_start = i + 1;
        }
    }

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = label_starts[label_starts.len() - 1];
    let last_label = &domain[last_start..last_end];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let local_string = String::from_utf8(local.to_vec()).expect("ASCII local part");
    let domain_string = domain
        .iter()
        .map(|&b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local: local_string,
        domain: domain_string,
        quoted: false,
    })
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let b = input.as_bytes();
    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    for (i, &byte) in b.iter().enumerate() {
        if byte >= 128 {
            return Err(EmailError::NonAscii { at: i });
        }
    }

    let mut at_pos = None;
    for (i, &byte) in b.iter().enumerate() {
        if byte == b'@' {
            if at_pos.is_some() {
                return Err(EmailError::MultipleAt { at: i });
            }
            at_pos = Some(i);
        }
    }
    let at_pos = match at_pos {
        Some(pos) => pos,
        None => return Err(EmailError::MissingAt),
    };
    if at_pos == 0 {
        return Err(EmailError::EmptyLocal { at: at_pos });
    }

    let local = &b[..at_pos];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }
    let mut previous_dot = false;
    for (i, &byte) in local.iter().enumerate() {
        if byte == b'.' {
            if i == 0 || i + 1 == local.len() || previous_dot {
                return Err(EmailError::LocalDot { at: i });
            }
            previous_dot = true;
        } else {
            if !byte.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&byte) {
                return Err(EmailError::LocalChar { at: i });
            }
            previous_dot = false;
        }
    }

    let domain_start = at_pos + 1;
    let raw = &b[domain_start..];
    let removed_root_dot = raw.last() == Some(&b'.');
    let domain_end = b.len() - usize::from(removed_root_dot);
    let domain = &b[domain_start..domain_end];

    if domain.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut count = 0usize;
    let mut start = 0usize;
    let mut last_label = (0usize, 0usize);
    for end in 0..=domain.len() {
        if end != domain.len() && domain[end] != b'.' {
            continue;
        }

        let label = &domain[start..end];
        let abs_start = domain_start + start;
        if label.is_empty() {
            let empty_at = if start == domain.len() && removed_root_dot {
                domain_end
            } else {
                abs_start
            };
            return Err(EmailError::EmptyLabel { at: empty_at });
        }
        if label.len() > 63 {
            return Err(EmailError::LabelTooLong {
                at: abs_start + 63,
            });
        }
        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen { at: abs_start });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: abs_start + label.len() - 1,
            });
        }
        for (offset, &byte) in label.iter().enumerate() {
            if !byte.is_ascii_alphanumeric() && byte != b'-' {
                return Err(EmailError::DomainChar {
                    at: abs_start + offset,
                });
            }
        }

        count += 1;
        last_label = (start, end);
        start = end + 1;
    }

    if count < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    let tld = &domain[last_label.0..last_label.1];
    if tld.iter().all(|byte| byte.is_ascii_digit()) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_label.0,
        });
    }

    let local_string = String::from_utf8(local.to_vec()).expect("ASCII");
    let domain_string: String = domain
        .iter()
        .map(|byte| byte.to_ascii_lowercase() as char)
        .collect();
    Ok(Address {
        local: local_string,
        domain: domain_string,
        quoted: false,
    })
}
```
