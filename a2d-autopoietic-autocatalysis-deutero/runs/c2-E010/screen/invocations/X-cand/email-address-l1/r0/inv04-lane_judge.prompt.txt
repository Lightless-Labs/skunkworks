    pub local: String,
    pub domain: String,
    pub quoted: bool,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    for (i, &b) in bytes.iter().enumerate() {
        if b >= 128 {
            return Err(EmailError::NonAscii { at: i });
        }
    }

    let ats: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();

    let split = match ats.as_slice() {
        [] => return Err(EmailError::MissingAt),
        [one] => *one,
        [_, second, ..] => return Err(EmailError::MultipleAt { at: *second }),
    };

    if split == 0 {
        return Err(EmailError::EmptyLocal { at: split });
    }

    let local = &bytes[..split];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at: i });
            }
        } else if !b.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
            return Err(EmailError::LocalChar { at: i });
        }
    }

    let domain_start = split + 1;
    let raw_domain = &bytes[domain_start..];
    let stripped = raw_domain.strip_suffix(b".").unwrap_or(raw_domain);
    let root_removed = stripped.len() != raw_domain.len();

    if stripped.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if stripped.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels: Vec<(usize, &[u8])> = Vec::new();
    let mut start = 0usize;
    for (i, &b) in stripped.iter().enumerate() {
        if b == b'.' {
            labels.push((start, &stripped[start..i]));
            start = i + 1;
        }
    }
    labels.push((start, &stripped[start..]));

    for (relative_start, label) in &labels {
        let absolute_start = domain_start + relative_start;
        if label.is_empty() {
            let at = if root_removed && *relative_start == stripped.len() {
                input.len() - 1
            } else {
                absolute_start
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if label.len() > 63 {
            return Err(EmailError::LabelTooLong {
                at: absolute_start + 63,
            });
        }
        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen { at: absolute_start });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: absolute_start + label.len() - 1,
            });
        }
        for (offset, &b) in label.iter().enumerate() {
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: absolute_start + offset,
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_label) = labels[labels.len() - 1];
    if last_label.iter().all(|b| b.is_ascii_digit()) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let local = String::from_utf8(local.to_vec()).unwrap();
    let domain: String = stripped
        .iter()
        .map(|&b| char::from(b.to_ascii_lowercase()))
        .collect();

    Ok(Address {
        local,
        domain,
        quoted: false,
    })
}

pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },

    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut separators = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| if b == b'@' { Some(i) } else { None });

    let separator = match separators.next() {
        Some(i) => i,
        None => return Err(EmailError::MissingAt),
    };

    if let Some(second) = separators.next() {
        return Err(EmailError::MultipleAt { at: second });
    }

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local = &bytes[..separator];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    let mut previous_dot = false;
    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || previous_dot {
                return Err(EmailError::LocalDot { at: i });
            }
            previous_dot = true;
        } else {
            previous_dot = false;
            let allowed = b.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&b);
            if !allowed {
                return Err(EmailError::LocalChar { at: i });
            }
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(has_root_dot);

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain = &raw_domain[..domain_len];
    let mut label_count = 0usize;
    let mut label_start = 0usize;
    let mut last_label_start = 0usize;

    for i in 0..=domain.len() {
        if i == domain.len() || domain[i] == b'.' {
            let label = &domain[label_start..i];
            let original_start = domain_start + label_start;

            if label.is_empty() {
                let at = if has_root_dot && label_start == domain.len() {
                    input.len() - 1
                } else {
                    original_start
                };
                return Err(EmailError::EmptyLabel { at });
            }

            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: original_start + 63,
                });
            }

            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: original_start });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: original_start + label.len() - 1,
                });
            }

            for (offset, &b) in label.iter().enumerate() {
                if !(b.is_ascii_alphanumeric() || b == b'-') {
                    return Err(EmailError::DomainChar {
                        at: original_start + offset,
                    });
                }
            }

            label_count += 1;
            last_label_start = original_start;
            label_start = i + 1;
        }
    }

    if label_count < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let last_label = &domain[last_label_start - domain_start..];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: last_label_start,
        });
    }

    let local_string = String::from_utf8(local.to_vec()).unwrap();
    let domain_string = domain
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local: local_string,
        domain: domain_string,
        quoted: false,
    })
}

        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });

            continue;

        }
    }

            if at.is_some() {
                return Err(EmailError::MultipleAt { at: i });
            }
            at = Some(i);
        }

    }

            }
        } else if !(b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)) {
            return Err(EmailError::LocalChar { at: i });
        }
    }

    let domain_start = sep + 1;
    let raw = &bytes[domain_start..];
    let root_removed = raw.last() == Some(&b'.');
    let content_len = raw.len() - usize::from(root_removed);
    if content_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if content_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain = &raw[..content_len];
    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut begin = 0;
    for i in 0..=content_len {
        if i == content_len || domain[i] == b'.' {
            if begin == i {
                let position = if root_removed && i == content_len {
                    input.len() - 1
                } else {
                    domain_start + begin
                };
                return Err(EmailError::EmptyLabel { at: position });
            }

            let len = i - begin;
            if len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: domain_start + begin + 63,
                });
            }
            if domain[begin] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: domain_start + begin,
                });
            }
            if domain[i - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: domain_start + i - 1,
                });
            }
            for j in begin..i {
                let b = domain[j];
                if !(b.is_ascii_alphanumeric() || b == b'-') {
                    return Err(EmailError::DomainChar {
                        at: domain_start + j,
                    });
                }

            labels.push((begin, i));
            begin = i + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_begin, last_end) = labels[labels.len() - 1];
    let mut numeric = true;

        if !b.is_ascii_digit() {
            numeric = false;
            break;
        }

    if numeric {
        return Err(EmailError::NumericTld {
            at: domain_start + last_begin,
        });

    for &b in domain {
        lowered_domain.push((b.to_ascii_lowercase()) as char);
    }

    Ok(Address {
        local: String::from_utf8(local.to_vec()).unwrap(),
        domain: lowered_domain,
        quoted: false,
    })

    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    for (at, &byte) in bytes.iter().enumerate() {
        if byte >= 128 {
            return Err(EmailError::NonAscii { at });
        }
    }

    let mut at_sign = None;
    for (at, &byte) in bytes.iter().enumerate() {
        if byte == b'@' {
            if at_sign.is_some() {
                return Err(EmailError::MultipleAt { at });
            }
            at_sign = Some(at);
        }
    }
    let at_sign = at_sign.ok_or(EmailError::MissingAt)?;
    if at_sign == 0 {
        return Err(EmailError::EmptyLocal { at: at_sign });
    }

    let local = &bytes[..at_sign];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    let specials = b"!#$%&'*+-/=?^_`{|}~";
    for (index, &byte) in local.iter().enumerate() {
        if byte == b'.' {
            if index == 0
                || index + 1 == local.len()
                || local.get(index.wrapping_sub(1)) == Some(&b'.')
            {
                return Err(EmailError::LocalDot { at: index });
            }
        } else if !byte.is_ascii_alphanumeric() && !specials.contains(&byte) {
            return Err(EmailError::LocalChar { at: index });
        }
    }

    let domain_start = at_sign + 1;
    let mut domain_end = bytes.len();
    let removed_dot = if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    let domain = &bytes[domain_start..domain_end];
    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut starts = Vec::new();
    let mut ends = Vec::new();
    let mut start = 0usize;

    for cursor in 0..=domain.len() {
        if cursor == domain.len() || domain[cursor] == b'.' {
            let label = &domain[start..cursor];
            let absolute_start = domain_start + start;

            if label.is_empty() {
                let error_at = if cursor == domain.len() {
                    removed_dot.unwrap_or(domain_start + cursor)
                } else {
                    domain_start + cursor
                };
                return Err(EmailError::EmptyLabel { at: error_at });
            }
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: absolute_start + 63,
                });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: absolute_start });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: absolute_start + label.len() - 1,
                });
            }
            for (offset, &byte) in label.iter().enumerate() {
                if !byte.is_ascii_alphanumeric() && byte != b'-' {
                    return Err(EmailError::DomainChar {
                        at: absolute_start + offset,
                    });
                }
            }

            starts.push(start);
            ends.push(cursor);
            start = cursor + 1;
        }
    }

    if starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let last_index = starts.len() - 1;
    let last_label = &domain[starts[last_index]..ends[last_index]];
    if last_label.iter().all(|byte| byte.is_ascii_digit()) {
        return Err(EmailError::NumericTld {
            at: domain_start + starts[last_index],
        });
    }

    let local_string = String::from_utf8(local.to_vec()).expect("ASCII validated");
    let mut normalized_domain = String::with_capacity(domain.len());
    for &byte in domain {
        normalized_domain.push(byte.to_ascii_lowercase() as char);
    }

    Ok(Address {
        local: local_string,
        domain: normalized_domain,
        quoted: false,
    })
}
