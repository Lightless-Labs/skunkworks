fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty() && b.len() <= 24 && b[0].is_ascii_lowercase()
        && b[b.len()-1] != b'-'
        && b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn constraint_key(c: &Constraint) -> (u8, Version) {
    match *c {
        Constraint::Exact(v) => (0,v),
        Constraint::AtLeast(v) => (1,v),
        Constraint::Before(v) => (2,v),
        Constraint::Caret(v) => (3,v),
    }
}
fn canonical(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort_by_key(constraint_key);
    cs.dedup();
    cs
}
fn permits(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x { false }
            else if x.major > 0 { v.major == x.major }
            else if x.minor > 0 { v.major == 0 && v.minor == x.minor }
            else { v == x }
        }
    }
}
fn meets(v: Version, cs: &[Constraint]) -> bool { cs.iter().all(|c| permits(*c,v)) }
fn version_text(v: Version) -> String { format!("{}.{}.{}",v.major,v.minor,v.patch) }

        Constraint::AtLeast(v) => format!(">={}",version_text(v)),
        Constraint::Before(v) => format!("<{}",version_text(v)),
        Constraint::Caret(v) => format!("^{}",version_text(v)),

fn validation(catalog:&[Package], roots:&[Requirement], locks:&[Lock]) -> Result<(),ResolveError> {
    for (i,r) in roots.iter().enumerate() {
        if !valid_name(&r.package) { return Err(ResolveError::InvalidName{site:format!("root[{}]",i),name:r.package.clone()}); }
    }
    for (i,p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) { return Err(ResolveError::InvalidName{site:format!("catalog[{}]",i),name:p.name.clone()}); }
        for (j,r) in p.releases.iter().enumerate() {
            for (k,d) in r.deps.iter().enumerate() {
                if !valid_name(&d.package) {
                    return Err(ResolveError::InvalidName{site:format!("catalog[{}].release[{}].dep[{}]",i,j,k),name:d.package.clone()});
                }
            }
        }
    }
    let mut names=std::collections::BTreeSet::new();
    for p in catalog {
        if !names.insert(p.name.clone()) { return Err(ResolveError::DuplicatePackage{package:p.name.clone()}); }
    }
    for p in catalog {
        let mut versions=std::collections::BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion{package:p.name.clone(),version:r.version});
            }
        }
    }
    for (i,l) in locks.iter().enumerate() {
        if !valid_name(&l.package) { return Err(ResolveError::InvalidName{site:format!("lock[{}]",i),name:l.package.clone()}); }
    }
    let mut seen=std::collections::BTreeSet::new();
    for l in locks {
        if !seen.insert(l.package.clone()) { return Err(ResolveError::DuplicateLock{package:l.package.clone()}); }
    }
    Ok(())
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    attempts: usize,
    max: usize,
    best: Option<(usize, Vec<Selection>, std::collections::BTreeMap<String,Vec<Constraint>>)>,
    first_error: Option<ResolveError>,
    aborted: bool,
}
impl<'a> Search<'a> {
    fn package(&self,n:&str)->Option<&'a Package>{self.catalog.iter().find(|p|p.name==n)}
    fn lock(&self,n:&str)->Option<Version>{self.locks.iter().find(|l|l.package==n).map(|l|l.version)}
    fn inspect(&mut self, assigned:&mut std::collections::BTreeMap<String,Version>) {
        if self.aborted { return; }
        let mut active: std::collections::BTreeMap<String,Vec<Constraint>>=std::collections::BTreeMap::new();
        for r in self.roots { active.entry(r.package.clone()).or_default().push(r.constraint); }
        for (name,ver) in assigned.iter() {
            if let Some(p)=self.package(name) {
                if let Some(rel)=p.releases.iter().find(|r|r.version==*ver) {
                    for d in &rel.deps { active.entry(d.package.clone()).or_default().push(d.constraint); }
                }
            }
        }
        for cs in active.values_mut() { *cs=canonical(std::mem::take(cs)); }
        for n in active.keys() {
            if self.package(n).is_none() {
                self.note(ResolveError::Missing{package:n.clone()});
                return;
            }
        }
        for (n,v) in assigned.iter() {
            if let Some(cs)=active.get(n) {
                if !meets(*v,cs) {
                    self.note(ResolveError::Conflict{package:n.clone(),selected:*v,requirements:cs.clone()});
                    return;
                }
            }
        }
        let mut branch:Option<String>=None;
        let mut candidates:Vec<Version>=Vec::new();
        for (n,cs) in &active {
            if assigned.contains_key(n) { continue; }
            let p=self.package(n).unwrap();
            let mut vs:Vec<Version>=p.releases.iter().filter(|r|meets(r.version,cs)).map(|r|r.version).collect();
            vs.sort_by(|a,b|b.cmp(a));
            if vs.is_empty() {
                self.note(ResolveError::NoMatch{package:n.clone(),requirements:cs.clone()});
                return;
            }
            if branch.is_none() { branch=Some(n.clone()); candidates=vs; }
        }
        if let Some(n)=branch {
            if let Some(lv)=self.lock(&n) {
                if let Some(i)=candidates.iter().position(|v|*v==lv) { candidates.remove(i); candidates.insert(0,lv); }
            }
            for v in candidates {
                if self.attempts==self.max { self.aborted=true; return; }
                self.attempts+=1;
                assigned.insert(n.clone(),v);
                self.inspect(assigned);
                assigned.remove(&n);
                if self.aborted { return; }
            }
            return;
        }
        let selected:Vec<Selection>=assigned.iter().map(|(p,v)|Selection{package:p.clone(),version:*v}).collect();
        let cost=self.locks.iter().filter(|l| assigned.get(&l.package)!=Some(&l.version)).count();
        let better=match &self.best {
            None=>true,
            Some((bc,bs,_)) if cost<*bc=>true,
            Some((bc,bs,_)) if cost>*bc=>false,
            Some((_bc,bs,_))=>vector_cmp(self.catalog,&selected,bs)==std::cmp::Ordering::Greater,
        };
        if better { self.best=Some((cost,selected,active)); }
    }
    fn note(&mut self,e:ResolveError) { if self.first_error.is_none(){self.first_error=Some(e);} }
}
fn vector_cmp(catalog:&[Package],a:&[Selection],b:&[Selection])->std::cmp::Ordering {
    let am:std::collections::BTreeMap<&str,Version>=a.iter().map(|s|(s.package.as_str(),s.version)).collect();
    let bm:std::collections::BTreeMap<&str,Version>=b.iter().map(|s|(s.package.as_str(),s.version)).collect();
    let mut names:Vec<&str>=catalog.iter().map(|p|p.name.as_str()).collect();
    names.sort();
    for n in names {
        let x=am.get(n); let y=bm.get(n);
        let c=match (x,y) {(None,None)=>std::cmp::Ordering::Equal,(None,Some(_))=>std::cmp::Ordering::Less,(Some(_),None)=>std::cmp::Ordering::Greater,(Some(x),Some(y))=>x.cmp(y)};
        if c!=std::cmp::Ordering::Equal{return c;}
    }
    std::cmp::Ordering::Equal
}
pub fn resolve(catalog:&[Package],roots:&[Requirement],locks:&[Lock],max_attempts:usize)->Result<Resolution,ResolveError>{
    validation(catalog,roots,locks)?;
    let mut s=Search{catalog,roots,locks,attempts:0,max:max_attempts,best:None,first_error:None,aborted:false};
    let mut assigned=std::collections::BTreeMap::new();
    s.inspect(&mut assigned);
    if s.aborted{return Err(ResolveError::Limit{attempts:max_attempts});}
    let (selected,active)=match s.best {
        Some((_,sel,act))=>(sel,act),
        None=>return Err(s.first_error.unwrap_or(ResolveError::NoMatch{package:String::new(),requirements:vec![]})),
    };
    let before:std::collections::BTreeMap<String,Version>=locks.iter().map(|l|(l.package.clone(),l.version)).collect();
    let after:std::collections::BTreeMap<String,Version>=selected.iter().map(|x|(x.package.clone(),x.version)).collect();
    let names:std::collections::BTreeSet<String>=before.keys().chain(after.keys()).cloned().collect();
    let mut changes=Vec::new();
    for n in names {
        let b=before.get(&n).copied(); let a=after.get(&n).copied();
        if b==a {continue;}
        let reason=match (b,a) {
            (None,Some(_))=>ChangeReason::Added,
            (Some(_),None)=>ChangeReason::Removed,
            (Some(old),Some(_))=>{
                let exists=self_package_release(catalog,&n,old);
                if !exists {ChangeReason::Unavailable}
                else if active.get(&n).map(|cs|!meets(old,cs)).unwrap_or(false) {ChangeReason::Rejected}
                else {ChangeReason::Rebalanced}
            },
            _=>unreachable!(),
        };
        changes.push(Change{package:n,before:b,after:a,reason});
    }
    changes.sort_by(|a,b|a.package.cmp(&b.package));
    Ok(Resolution{selected,changes})
}
fn self_package_release(catalog:&[Package],name:&str,v:Version)->bool{
    catalog.iter().find(|p|p.name==name).map(|p|p.releases.iter().any(|r|r.version==v)).unwrap_or(false)
}
