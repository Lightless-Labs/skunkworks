fn name_ok(s: &str) -> bool {
    let b = s.as_bytes();
    (1..=24).contains(&b.len())
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && (if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                })
        }
    }
}

fn canon(cs: &mut Vec<Constraint>) {
    fn rank(c: Constraint) -> (u8, Version) {
        match c {
            Constraint::Exact(v) => (0, v),
            Constraint::AtLeast(v) => (1, v),
            Constraint::Before(v) => (2, v),
            Constraint::Caret(v) => (3, v),
        }
    }
    cs.sort_by_key(|&c| rank(c));
    cs.dedup();
}

fn fmt_version(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn fmt_constraint(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", fmt_version(v)),
        Constraint::AtLeast(v) => format!(">={}", fmt_version(v)),
        Constraint::Before(v) => format!("<{}", fmt_version(v)),
        Constraint::Caret(v) => format!("^{}", fmt_version(v)),
    }
}

fn pct(s: &str) -> String {
    let mut o = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || *b == b'.' || *b == b'_' || *b == b'-' {
            o.push(*b as char);
        } else {
            o.push_str(&format!("%{:02X}", b));
        }
    }
    o
}

fn fmt_constraints(cs: &[Constraint]) -> String {

    canon(&mut c);
    c.iter().map(|&x| fmt_constraint(x)).collect::<Vec<_>>().join(",")
}

fn validate(c: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !name_ok(&r.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: r.package.clone(),
            });
        }
    }
    for (i, p) in c.iter().enumerate() {
        if !name_ok(&p.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: p.name.clone(),
            });
        }
        for (j, rel) in p.releases.iter().enumerate() {
            for (k, d) in rel.deps.iter().enumerate() {
                if !name_ok(&d.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),

struct Solver<'a> {
    packages: BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    budget: usize,
    used: usize,
    best: Option<(BTreeMap<String, Version>, BTreeMap<String, Vec<Constraint>>)>,
    best_cost: usize,
    first_failure: Option<ResolveError>,
    limited: bool,
}

impl<'a> Solver<'a> {
    fn active(&self, chosen: &BTreeMap<String, Version>) -> BTreeMap<String, Vec<Constraint>> {
        let mut a = BTreeMap::<String, Vec<Constraint>>::new();
        for r in self.roots {
            a.entry(r.package.clone()).or_default().push(r.constraint);
        }
        for (n, v) in chosen {
            if let Some(p) = self.packages.get(n) {
                if let Some(rel) = p.releases.iter().find(|r| r.version == *v) {
                    for d in &rel.deps {
                        a.entry(d.package.clone()).or_default().push(d.constraint);
                    }
                }
            }
        }
        for cs in a.values_mut() {
            canon(cs);
        }
        a
    }

    fn error(
        &self,
        chosen: &BTreeMap<String, Version>,
        active: &BTreeMap<String, Vec<Constraint>>,
    ) -> Option<ResolveError> {
        if let Some(n) = active.keys().find(|n| !self.packages.contains_key(*n)) {
            return Some(ResolveError::Missing {
                package: n.clone(),
            });
        }
        for (n, v) in chosen {
            if let Some(cs) = active.get(n) {
                if !cs.iter().all(|&c| matches(c, *v)) {
                    return Some(ResolveError::Conflict {
                        package: n.clone(),
                        selected: *v,
                        requirements: cs.clone(),
                    });
                }
            }
        }
        for (n, cs) in active {
            if !chosen.contains_key(n) {
                let p = self.packages[n];
                if !p.releases.iter().any(|r| cs.iter().all(|&c| matches(c, r.version))) {
                    return Some(ResolveError::NoMatch {
                        package: n.clone(),
                        requirements: cs.clone(),
                    });
                }
            }
        }
        None
    }

    fn cost(&self, chosen: &BTreeMap<String, Version>) -> usize {
        self.locks.iter().filter(|(n, v)| chosen.get(*n) != Some(*v)).count()
    }

    fn better(&self, chosen: &BTreeMap<String, Version>) -> bool {
        let cost = self.cost(chosen);
        let Some((best, _)) = &self.best else {
            return true;
        };
        if cost != self.best_cost {
            return cost < self.best_cost;
        }
        for n in self.packages.keys() {
            match (chosen.get(n), best.get(n)) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }

    fn visit(&mut self, chosen: BTreeMap<String, Version>) {
        if self.limited {
            return;
        }
        let active = self.active(&chosen);
        if let Some(e) = self.error(&chosen, &active) {
            if self.first_failure.is_none() {
                self.first_failure = Some(e);
            }
            return;
        }
        let next = active.keys().find(|n| !chosen.contains_key(*n)).cloned();
        let Some(n) = next else {
            if self.better(&chosen) {
                self.best_cost = self.cost(&chosen);
                self.best = Some((chosen, active));
            }
            return;
        };

        let p = self.packages[n.as_str()];
        let cs = &active[&n];
        let mut versions: Vec<Version> = p
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|&v| cs.iter().all(|&c| matches(c, v)))
            .collect();
        versions.sort_by(|a, b| b.cmp(a));
        if let Some(v) = self.locks.get(&n) {
            if let Some(i) = versions.iter().position(|x| x == v) {
                let x = versions.remove(i);
                versions.insert(0, x);
            }
        }
        for v in versions {
            if self.used == self.budget {
                self.limited = true;
                return;
            }
            self.used += 1;
            let mut child = chosen.clone();
            child.insert(n.clone(), v);
            self.visit(child);
            if self.limited {
                return;
            }
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;
    let mut packages = BTreeMap::new();
    for p in catalog {
        if packages.contains_key(&p.name) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
        packages.insert(p.name.clone(), p);
    }
    for p in catalog {
        let mut seen = BTreeSet::new();
        for r in &p.releases {
            if !seen.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }
    let mut lock_map = BTreeMap::new();
    for l in locks {
        if lock_map.contains_key(&l.package) {
            return Err(ResolveError::DuplicateLock {
                package: l.package.clone(),
            });
        }
        lock_map.insert(l.package.clone(), l.version);
    }

    let mut solver = Solver {
        packages,
        roots,
        locks: lock_map.clone(),
        budget: max_attempts,
        used: 0,
        best: None,
        best_cost: usize::MAX,
        first_failure: None,
        limited: false,
    };
    solver.visit(BTreeMap::new());
    if solver.limited {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }
    let Some((chosen, active)) = solver.best else {
        return Err(solver.first_failure.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let names: BTreeSet<String> = lock_map.keys().chain(chosen.keys()).cloned().collect();
    let mut changes = Vec::new();
    for n in names {
        let before = lock_map.get(&n).copied();
        let after = chosen.get(&n).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = solver
                    .packages
                    .get(&n)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if active.get(&n).is_some_and(|cs| !cs.iter().all(|&c| matches(c, old))) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            _ => continue,
        };
        changes.push(Change {
            package: n,
            before,
            after,
            reason,
        });
    }
    let selected = chosen
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect();
    Ok(Resolution { selected, changes })
}

fn fits(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x && (if x.major != 0 { v.major == x.major }
                else if x.minor != 0 { v.major == 0 && v.minor == x.minor }
                else { v == x })
        }
    }

type Picked = BTreeMap<String, Version>;
type ConstraintsByPackage = BTreeMap<String, Vec<Constraint>>;

fn gather(roots: &[Requirement], picked: &Picked, catalog: &BTreeMap<String, Package>) -> ConstraintsByPackage {
    let mut m = BTreeMap::new();
    for req in roots {
        m.entry(req.package.clone()).or_insert_with(Vec::new).push(req.constraint);
    }
    for (package, version) in picked {
        if let Some(p) = catalog.get(package) {
            if let Some(release) = p.releases.iter().find(|r| r.version == *version) {
                for req in &release.deps {
                    m.entry(req.package.clone()).or_insert_with(Vec::new).push(req.constraint);
                }
            }
        }
    }
    m.into_iter().map(|(n, xs)| (n, canonicalize(&xs))).collect()
}

fn viable(p: &Package, cs: &[Constraint]) -> Vec<Version> {
    let mut result: Vec<Version> = p.releases.iter().map(|r| r.version)
        .filter(|v| cs.iter().all(|c| fits(*v, *c))).collect();
    result.sort_by(|a,b| b.cmp(a));
    result
}

struct Winner {
    picks: Picked,
    constraints: ConstraintsByPackage,
    cost: usize,
    vector: Vec<Option<Version>>,
}

struct Solver<'a> {
    roots: &'a [Requirement],
    catalog: &'a BTreeMap<String, Package>,
    locks: &'a BTreeMap<String, Version>,
    names: &'a [String],
    attempts: usize,
    budget: usize,
    winner: Option<Winner>,
    first_failure: Option<ResolveError>,
}

impl Solver<'_> {
    fn record_failure(&mut self, e: ResolveError) {
        if self.first_failure.is_none() { self.first_failure = Some(e); }
    }

    fn visit(&mut self, picks: &mut Picked) -> Result<(), ResolveError> {
        let constraints = gather(self.roots, picks, self.catalog);
        if let Some((n, _)) = constraints.iter().find(|(n, _)| !self.catalog.contains_key(*n)) {
            return Err(ResolveError::Missing { package: n.clone() });
        }
        for (n, v) in picks.iter() {
            if let Some(cs) = constraints.get(n) {
                if !cs.iter().all(|c| fits(*v, *c)) {
                    return Err(ResolveError::Conflict { package: n.clone(), selected: *v, requirements: cs.clone() });
                }
            }
        }

        let mut branch: Option<(String, Vec<Version>)> = None;
        for (n, cs) in &constraints {
            if picks.contains_key(n) { continue; }
            let mut versions = viable(&self.catalog[n], cs);
            if versions.is_empty() {
                return Err(ResolveError::NoMatch { package: n.clone(), requirements: cs.clone() });
            }
            if let Some(v) = self.locks.get(n) {
                if let Some(i) = versions.iter().position(|x| x == v) {
                    versions.remove(i);
                    versions.insert(0, *v);
                }
            }
            if branch.is_none() { branch = Some((n.clone(), versions)); }
        }

        let Some((name, versions)) = branch else {
            let cost = self.locks.iter().filter(|(n, v)| picks.get(*n) != Some(*v)).count();
            let vector: Vec<Option<Version>> = self.names.iter().map(|n| picks.get(n).copied()).collect();
            let better = self.winner.as_ref().map_or(true, |w| {
                cost < w.cost || (cost == w.cost && vector > w.vector)
            });
            if better { self.winner = Some(Winner { picks: picks.clone(), constraints, cost, vector }); }
            return Ok(());
        };

        for v in versions {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit { attempts: self.budget });
            }
            self.attempts += 1;
            picks.insert(name.clone(), v);
            let child = self.visit(picks);
            picks.remove(&name);
            match child {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => return Err(ResolveError::Limit { attempts }),
                Err(e) => self.record_failure(e),
            }
        }
        Ok(())
    }
}

pub fn resolve(catalog: &[Package], roots: &[Requirement], locks: &[Lock], max_attempts: usize)
    -> Result<Resolution, ResolveError>
{
    for (i, root) in roots.iter().enumerate() {
        if !name_ok(&root.package) {
            return Err(ResolveError::InvalidName { site: format!("root[{}]",i), name: root.package.clone() });
        }
    }
    for (i, pkg) in catalog.iter().enumerate() {
        if !name_ok(&pkg.name) {
            return Err(ResolveError::InvalidName { site: format!("catalog[{}]",i), name: pkg.name.clone() });
        }
        for (j, release) in pkg.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !name_ok(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]",i,j,k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut packages = BTreeSet::new();
    for p in catalog {
        if !packages.insert(p.name.clone()) { return Err(ResolveError::DuplicatePackage { package: p.name.clone() }); }
    }
    for p in catalog {
        let mut versions = BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion { package: p.name.clone(), version: r.version });
            }
        }
    }
    for (i, l) in locks.iter().enumerate() {
        if !name_ok(&l.package) {
            return Err(ResolveError::InvalidName { site: format!("lock[{}]",i), name: l.package.clone() });
        }
    }
    let mut lock_names = BTreeSet::new();
    for l in locks {
        if !lock_names.insert(l.package.clone()) {
            return Err(ResolveError::DuplicateLock { package: l.package.clone() });
        }
    }

    let catalog_map: BTreeMap<String, Package> = catalog.iter().cloned().map(|p| (p.name.clone(),p)).collect();
    let locks_map: BTreeMap<String, Version> = locks.iter().map(|l| (l.package.clone(),l.version)).collect();
    let names: Vec<String> = catalog_map.keys().cloned().collect();
    let mut solver = Solver {
        roots, catalog: &catalog_map, locks: &locks_map, names: &names,
        attempts: 0, budget: max_attempts, winner: None, first_failure: None,
    };
    let mut picks = BTreeMap::new();
    solver.visit(&mut picks)?;

    if let Some(w) = solver.winner {
        let selected = w.picks.iter().map(|(package, version)| Selection {
            package: package.clone(), version: *version
        }).collect();

        let mut names = BTreeSet::new();
        names.extend(locks_map.keys().cloned());
        names.extend(w.picks.keys().cloned());
        let mut changes = Vec::new();
        for package in names {
            let before = locks_map.get(&package).copied();
            let after = w.picks.get(&package).copied();
            if before == after { continue; }
            let reason = match (before, after) {
                (None, Some(_)) => ChangeReason::Added,
                (Some(_), None) => ChangeReason::Removed,
                (Some(old), Some(_)) => {
                    let available = catalog_map.get(&package)
                        .map(|p| p.releases.iter().any(|r| r.version == old)).unwrap_or(false);
                    if !available { ChangeReason::Unavailable }
                    else if !w.constraints.get(&package).map(|cs| cs.iter().all(|c| fits(old,*c))).unwrap_or(true) {
                        ChangeReason::Rejected
                    } else { ChangeReason::Rebalanced }
                }
                (None,None) => continue,
            };
            changes.push(Change { package, before, after, reason });
        }
        Ok(Resolution { selected, changes })
    } else {
        Err(solver.first_failure.unwrap_or(ResolveError::NoMatch { package: String::new(), requirements: Vec::new() }))
    }
}

fn validation(catalog:&[Package], roots:&[Requirement], locks:&[Lock]) -> Result<(),ResolveError> {
    for (i,r) in roots.iter().enumerate() {
        if !valid_name(&r.package) { return Err(ResolveError::InvalidName{site:format!("root[{}]",i),name:r.package.clone()}); }
    }
    for (i,p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) { return Err(ResolveError::InvalidName{site:format!("catalog[{}]",i),name:p.name.clone()}); }
        for (j,r) in p.releases.iter().enumerate() {
            for (k,d) in r.deps.iter().enumerate() {
                if !valid_name(&d.package) {
                    return Err(ResolveError::InvalidName{site:format!("catalog[{}].release[{}].dep[{}]",i,j,k),name:d.package.clone()});
                }
            }
        }
    }
    let mut names=std::collections::BTreeSet::new();
    for p in catalog {
        if !names.insert(p.name.clone()) { return Err(ResolveError::DuplicatePackage{package:p.name.clone()}); }
    }
    for p in catalog {
        let mut versions=std::collections::BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion{package:p.name.clone(),version:r.version});
            }
        }
    }
    for (i,l) in locks.iter().enumerate() {
        if !valid_name(&l.package) { return Err(ResolveError::InvalidName{site:format!("lock[{}]",i),name:l.package.clone()}); }
    }
    let mut seen=std::collections::BTreeSet::new();
    for l in locks {
        if !seen.insert(l.package.clone()) { return Err(ResolveError::DuplicateLock{package:l.package.clone()}); }
    }
    Ok(())
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    attempts: usize,
    max: usize,
    best: Option<(usize, Vec<Selection>, std::collections::BTreeMap<String,Vec<Constraint>>)>,
    first_error: Option<ResolveError>,
    aborted: bool,
}
impl<'a> Search<'a> {
    fn package(&self,n:&str)->Option<&'a Package>{self.catalog.iter().find(|p|p.name==n)}
    fn lock(&self,n:&str)->Option<Version>{self.locks.iter().find(|l|l.package==n).map(|l|l.version)}
    fn inspect(&mut self, assigned:&mut std::collections::BTreeMap<String,Version>) {
        if self.aborted { return; }
        let mut active: std::collections::BTreeMap<String,Vec<Constraint>>=std::collections::BTreeMap::new();
        for r in self.roots { active.entry(r.package.clone()).or_default().push(r.constraint); }
        for (name,ver) in assigned.iter() {
            if let Some(p)=self.package(name) {
                if let Some(rel)=p.releases.iter().find(|r|r.version==*ver) {
                    for d in &rel.deps { active.entry(d.package.clone()).or_default().push(d.constraint); }
                }
            }
        }
        for cs in active.values_mut() { *cs=canonical(std::mem::take(cs)); }
        for n in active.keys() {
            if self.package(n).is_none() {
                self.note(ResolveError::Missing{package:n.clone()});
                return;
            }
        }
        for (n,v) in assigned.iter() {
            if let Some(cs)=active.get(n) {
                if !meets(*v,cs) {
                    self.note(ResolveError::Conflict{package:n.clone(),selected:*v,requirements:cs.clone()});
                    return;
                }
            }
        }
        let mut branch:Option<String>=None;
        let mut candidates:Vec<Version>=Vec::new();
        for (n,cs) in &active {
            if assigned.contains_key(n) { continue; }
            let p=self.package(n).unwrap();
            let mut vs:Vec<Version>=p.releases.iter().filter(|r|meets(r.version,cs)).map(|r|r.version).collect();
            vs.sort_by(|a,b|b.cmp(a));
            if vs.is_empty() {
                self.note(ResolveError::NoMatch{package:n.clone(),requirements:cs.clone()});
                return;
            }
            if branch.is_none() { branch=Some(n.clone()); candidates=vs; }
        }
        if let Some(n)=branch {
            if let Some(lv)=self.lock(&n) {
                if let Some(i)=candidates.iter().position(|v|*v==lv) { candidates.remove(i); candidates.insert(0,lv); }
            }
            for v in candidates {
                if self.attempts==self.max { self.aborted=true; return; }
                self.attempts+=1;
                assigned.insert(n.clone(),v);
                self.inspect(assigned);
                assigned.remove(&n);
                if self.aborted { return; }
            }
            return;
        }
        let selected:Vec<Selection>=assigned.iter().map(|(p,v)|Selection{package:p.clone(),version:*v}).collect();
        let cost=self.locks.iter().filter(|l| assigned.get(&l.package)!=Some(&l.version)).count();
        let better=match &self.best {
            None=>true,
            Some((bc,bs,_)) if cost<*bc=>true,
            Some((bc,bs,_)) if cost>*bc=>false,
            Some((_bc,bs,_))=>vector_cmp(self.catalog,&selected,bs)==std::cmp::Ordering::Greater,
        };
        if better { self.best=Some((cost,selected,active)); }
    }
    fn note(&mut self,e:ResolveError) { if self.first_error.is_none(){self.first_error=Some(e);} }
}
fn vector_cmp(catalog:&[Package],a:&[Selection],b:&[Selection])->std::cmp::Ordering {
    let am:std::collections::BTreeMap<&str,Version>=a.iter().map(|s|(s.package.as_str(),s.version)).collect();
    let bm:std::collections::BTreeMap<&str,Version>=b.iter().map(|s|(s.package.as_str(),s.version)).collect();
    let mut names:Vec<&str>=catalog.iter().map(|p|p.name.as_str()).collect();
    names.sort();
    for n in names {
        let x=am.get(n); let y=bm.get(n);
        let c=match (x,y) {(None,None)=>std::cmp::Ordering::Equal,(None,Some(_))=>std::cmp::Ordering::Less,(Some(_),None)=>std::cmp::Ordering::Greater,(Some(x),Some(y))=>x.cmp(y)};
        if c!=std::cmp::Ordering::Equal{return c;}
    }
    std::cmp::Ordering::Equal
}
pub fn resolve(catalog:&[Package],roots:&[Requirement],locks:&[Lock],max_attempts:usize)->Result<Resolution,ResolveError>{
    validation(catalog,roots,locks)?;
    let mut s=Search{catalog,roots,locks,attempts:0,max:max_attempts,best:None,first_error:None,aborted:false};
    let mut assigned=std::collections::BTreeMap::new();
    s.inspect(&mut assigned);
    if s.aborted{return Err(ResolveError::Limit{attempts:max_attempts});}
    let (selected,active)=match s.best {
        Some((_,sel,act))=>(sel,act),
        None=>return Err(s.first_error.unwrap_or(ResolveError::NoMatch{package:String::new(),requirements:vec![]})),
    };
    let before:std::collections::BTreeMap<String,Version>=locks.iter().map(|l|(l.package.clone(),l.version)).collect();
    let after:std::collections::BTreeMap<String,Version>=selected.iter().map(|x|(x.package.clone(),x.version)).collect();
    let names:std::collections::BTreeSet<String>=before.keys().chain(after.keys()).cloned().collect();
    let mut changes=Vec::new();
    for n in names {
        let b=before.get(&n).copied(); let a=after.get(&n).copied();
        if b==a {continue;}
        let reason=match (b,a) {
            (None,Some(_))=>ChangeReason::Added,
            (Some(_),None)=>ChangeReason::Removed,
            (Some(old),Some(_))=>{
                let exists=self_package_release(catalog,&n,old);
                if !exists {ChangeReason::Unavailable}
                else if active.get(&n).map(|cs|!meets(old,cs)).unwrap_or(false) {ChangeReason::Rejected}
                else {ChangeReason::Rebalanced}
            },
            _=>unreachable!(),
        };
        changes.push(Change{package:n,before:b,after:a,reason});
    }
    changes.sort_by(|a,b|a.package.cmp(&b.package));
    Ok(Resolution{selected,changes})
}
fn self_package_release(catalog:&[Package],name:&str,v:Version)->bool{
    catalog.iter().find(|p|p.name==name).map(|p|p.releases.iter().any(|r|r.version==v)).unwrap_or(false)
}

struct Best {

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: &'a BTreeMap<String, Package>,
    locks: &'a BTreeMap<String, Version>,
    catalog_names: &'a [String],
    attempts: usize,
    max_attempts: usize,
    best: Option<Best>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn save_error(&mut self, e: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(e);
        }
    }

    fn walk(&mut self, selected: &mut Selected) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, selected, self.catalog);

        if let Some(name) = active.keys().find(|n| !self.catalog.contains_key(*n)) {
            return Err(ResolveError::Missing { package: name.clone() });
        }

        for (name, version) in selected.iter() {
            if let Some(cs) = active.get(name) {
                if !cs.iter().all(|c| matches(*version, *c)) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: cs.clone(),
                    });
                }
            }
        }

        let mut next: Option<(String, Vec<Version>)> = None;
        for (name, cs) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let pkg = &self.catalog[name];
            let mut vs = candidates(pkg, cs);
            if vs.is_empty() {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: cs.clone(),
                });
            }
            if let Some(lock) = self.locks.get(name) {
                if let Some(pos) = vs.iter().position(|v| *v == *lock) {
                    let v = vs.remove(pos);
                    vs.insert(0, v);
                }
            }
            if next.is_none() {
                next = Some((name.clone(), vs));
            }
        }

        let Some((name, vs)) = next else {
            let cost = self.locks.iter().filter(|(n, v)| selected.get(*n) != Some(*v)).count();
            let vector = self.catalog_names.iter().map(|n| selected.get(n).copied()).collect();
            let replace = match &self.best {
                None => true,
                Some(b) => cost < b.cost || (cost == b.cost && vector > b.vector),
            };
            if replace {
                self.best = Some(Best { selected: selected.clone(), active, cost, vector });
            }
            return Ok(());
        };

        for version in vs {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit { attempts: self.max_attempts });
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            let result = self.walk(selected);
            selected.remove(&name);
            match result {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => return Err(ResolveError::Limit { attempts }),
                Err(e) => self.save_error(e),
            }
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName { site: format!("root[{}]", i), name: r.package.clone() });
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) {
            return Err(ResolveError::InvalidName { site: format!("catalog[{}]", i), name: p.name.clone() });
        }
        for (j, release) in p.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for p in catalog {
        if !seen_packages.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage { package: p.name.clone() });
        }
    }

    for p in catalog {
        let mut seen_versions = BTreeSet::new();
        for r in &p.releases {
            if !seen_versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion { package: p.name.clone(), version: r.version });
            }
        }
    }

    for (i, l) in locks.iter().enumerate() {
        if !valid_name(&l.package) {
            return Err(ResolveError::InvalidName { site: format!("lock[{}]", i), name: l.package.clone() });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for l in locks {
        if !seen_locks.insert(l.package.clone()) {
            return Err(ResolveError::DuplicateLock { package: l.package.clone() });
        }
    }

    let catalog_map: BTreeMap<String, Package> = catalog.iter().cloned().map(|p| (p.name.clone(), p)).collect();
    let locks_map: BTreeMap<String, Version> = locks.iter().map(|l| (l.package.clone(), l.version)).collect();
    let catalog_names: Vec<String> = catalog_map.keys().cloned().collect();

    let mut search = Search {
        roots,
        catalog: &catalog_map,
        locks: &locks_map,
        catalog_names: &catalog_names,
        attempts: 0,
        max_attempts,
        best: None,
        first_error: None,
    };
    let mut selected = BTreeMap::new();
    search.walk(&mut selected)?;

    if let Some(best) = search.best {
        let selected_out: Vec<Selection> = best.selected.iter()
            .map(|(package, version)| Selection { package: package.clone(), version: *version })
            .collect();

        let mut all_names = BTreeSet::new();
        all_names.extend(locks_map.keys().cloned());
        all_names.extend(best.selected.keys().cloned());

        let mut changes = Vec::new();
        for name in all_names {
            let before = locks_map.get(&name).copied();
            let after = best.selected.get(&name).copied();
            if before == after {
                continue;
            }
            let reason = match (before, after) {
                (None, Some(_)) => ChangeReason::Added,
                (Some(_), None) => ChangeReason::Removed,
                (Some(old), Some(_)) => {
                    let exists = catalog_map.get(&name)
                        .map(|p| p.releases.iter().any(|r| r.version == old))
                        .unwrap_or(false);
                    if !exists {
                        ChangeReason::Unavailable
                    } else if !best.active.get(&name).map(|cs| cs.iter().all(|c| matches(old, *c))).unwrap_or(true) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
                (None, None) => continue,
            };
            changes.push(Change { package: name, before, after, reason });
        }
        Ok(Resolution { selected: selected_out, changes })
    } else {
        Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }))
    }
}

fn validate(c: &[Package], r: &[Requirement], l: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in r.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, pkg) in c.iter().enumerate() {
        if !valid_name(&pkg.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: pkg.name.clone(),
            });
        }
        for (j, rel) in pkg.releases.iter().enumerate() {
            for (k, dep) in rel.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for pkg in c {
        if !package_names.insert(pkg.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: pkg.name.clone(),
            });
        }
    }
    for pkg in c {
        let mut versions = BTreeSet::new();
        for rel in &pkg.releases {
            if !versions.insert(rel.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: pkg.name.clone(),
                    version: rel.version,
                });
            }
        }
    }
    for (i, lock) in l.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut locks = BTreeSet::new();
    for lock in l {
        if !locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

type Chosen = BTreeMap<String, Version>;
type RequirementsByName = BTreeMap<String, Vec<Constraint>>;

fn collect(
    roots: &[Requirement],
    chosen: &Chosen,
    packages: &BTreeMap<String, &Package>,
) -> RequirementsByName {
    let mut map = RequirementsByName::new();
    for root in roots {
        map.entry(root.package.clone()).or_default().push(root.constraint);
    }
    for (name, version) in chosen {
        if let Some(package) = packages.get(name) {
            if let Some(release) = package.releases.iter().find(|x| x.version == *version) {
                for dep in &release.deps {
                    map.entry(dep.package.clone()).or_default().push(dep.constraint);
                }
            }
        }
    }
    for cs in map.values_mut() {
        *cs = canon(cs);
    }
    map
}

struct Resolver<'a> {
    roots: &'a [Requirement],
    packages: BTreeMap<String, &'a Package>,
    locks: BTreeMap<String, Version>,
    names: Vec<String>,
    budget: usize,
    used: usize,
    best: Option<(usize, Vec<Option<Version>>, Chosen, RequirementsByName)>,
    first_leaf: Option<ResolveError>,
}

impl<'a> Resolver<'a> {
    fn leaf(&mut self, e: ResolveError) {
        if self.first_leaf.is_none() {
            self.first_leaf = Some(e);
        }
    }

    fn visit(&mut self, chosen: &mut Chosen) -> Result<(), ResolveError> {
        let reqs = collect(self.roots, chosen, &self.packages);

        if let Some(name) = reqs.keys().find(|name| !self.packages.contains_key(*name)) {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in chosen.iter() {
            if let Some(all) = reqs.get(name) {
                if !all.iter().all(|c| matches(*c, *version)) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: all.clone(),
                    });
                }
            }
        }

        for (name, all) in &reqs {
            if chosen.contains_key(name) {
                continue;
            }
            let pkg = self.packages[name];
            if !pkg.releases.iter().any(|rel| {
                all.iter().all(|c| matches(*c, rel.version))
            }) {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: all.clone(),
                });
            }
        }

        let pending = reqs.keys().find(|name| !chosen.contains_key(*name)).cloned();
        let Some(name) = pending else {
            let cost = self
                .locks
                .iter()
                .filter(|(name, version)| chosen.get(*name) != Some(*version))
                .count();
            let vector = self.names.iter().map(|n| chosen.get(n).copied()).collect::<Vec<_>>();
            let better = self.best.as_ref().map_or(true, |(bc, bv, _, _)| {
                cost < *bc || (cost == *bc && vector > *bv)
            });
            if better {
                self.best = Some((cost, vector, chosen.clone(), reqs));
            }
            return Ok(());
        };

        let all = reqs.get(&name).cloned().unwrap_or_default();
        let mut versions = self.packages[&name]
            .releases
            .iter()
            .filter(|rel| all.iter().all(|c| matches(*c, rel.version)))
            .map(|rel| rel.version)
            .collect::<Vec<_>>();
        versions.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.get(&name) {
            if let Some(pos) = versions.iter().position(|v| v == lock) {
                let locked = versions.remove(pos);
                versions.insert(0, locked);
            }
        }

        for version in versions {
            if self.used == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.used += 1;
            chosen.insert(name.clone(), version);
            match self.visit(chosen) {
                Ok(()) => {}
                Err(e @ ResolveError::Limit { .. }) => {
                    chosen.remove(&name);
                    return Err(e);
                }
                Err(e) => self.leaf(e),
            }
            chosen.remove(&name);
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;
    let packages = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect::<BTreeMap<_, _>>();
    let locks_map = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect::<BTreeMap<_, _>>();
    let mut names = catalog.iter().map(|p| p.name.clone()).collect::<Vec<_>>();
    names.sort();

    let mut resolver = Resolver {
        roots,
        packages,
        locks: locks_map,
        names,
        budget: max_attempts,
        used: 0,
        best: None,
        first_leaf: None,
    };
    let mut chosen = Chosen::new();
    resolver.visit(&mut chosen)?;

    let Some((_, _, chosen, final_reqs)) = resolver.best else {
        if let Some(error) = resolver.first_leaf {
            return Err(error);
        }
        return Err(ResolveError::NoMatch {
            package: roots.first().map(|r| r.package.clone()).unwrap_or_default(),
            requirements: Vec::new(),
        });
    };

    let selected = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let before = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect::<BTreeMap<_, _>>();
    let all_names = before
        .keys()
        .chain(chosen.keys())
        .cloned()
        .collect::<BTreeSet<_>>();
    let mut changes = Vec::new();

    for package in all_names {
        let old = before.get(&package).copied();
        let new = chosen.get(&package).copied();
        if old == new {
            continue;
        }
        let reason = match (old, new) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old_version), Some(_)) => {
                let available = resolver
                    .packages
                    .get(&package)
                    .map_or(false, |pkg| pkg.releases.iter().any(|r| r.version == old_version));
                if !available {
                    ChangeReason::Unavailable
                } else if final_reqs.get(&package).map_or(false, |all| {
                    !all.iter().all(|c| matches(*c, old_version))
                }) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before: old,
            after: new,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

fn b_valid(s:&str)->bool {
    let z=s.as_bytes();
    z.len()>0 && z.len()<=24 && z[0].is_ascii_lowercase() && z[z.len()-1]!=b'-'
        && z.iter().all(|x|x.is_ascii_lowercase()||x.is_ascii_digit()||*x==b'-')
}
fn b_key(c:&Constraint)->(u8,Version){match *c{Constraint::Exact(v)=>(0,v),Constraint::AtLeast(v)=>(1,v),Constraint::Before(v)=>(2,v),Constraint::Caret(v)=>(3,v)}}

fn b_validate(c:&[Package],r:&[Requirement],l:&[Lock])->Result<(),ResolveError>{
    for (i,x) in r.iter().enumerate(){if !b_valid(&x.package){return Err(ResolveError::InvalidName{site:format!("root[{}]",i),name:x.package.clone()});}}
    for (i,p) in c.iter().enumerate(){if !b_valid(&p.name){return Err(ResolveError::InvalidName{site:format!("catalog[{}]",i),name:p.name.clone()});}
        for(j,z) in p.releases.iter().enumerate(){for(k,d)in z.deps.iter().enumerate(){if !b_valid(&d.package){return Err(ResolveError::InvalidName{site:format!("catalog[{}].release[{}].dep[{}]",i,j,k),name:d.package.clone()});}}}}

    Ok(())

struct BSearch<'a>{c:&'a[Package],r:&'a[Requirement],l:&'a[Lock],budget:usize,used:usize,sols:Vec<(usize,Vec<Selection>,std::collections::BTreeMap<String,Vec<Constraint>>)>,first:Option<ResolveError>,limit:bool}
impl<'a>BSearch<'a>{

    fn locked(&self,n:&str)->Option<Version>{self.l.iter().find(|x|x.package==n).map(|x|x.version)}
    fn leaf_error(&mut self,e:ResolveError){if self.first.is_none(){self.first=Some(e);}}
    fn walk(&mut self,a:&mut std::collections::BTreeMap<String,Version>){
        if self.limit{return;}
        let mut m:std::collections::BTreeMap<String,Vec<Constraint>>=std::collections::BTreeMap::new();
        for x in self.r{m.entry(x.package.clone()).or_default().push(x.constraint);}
        for(n,v)in a.iter(){if let Some(p)=self.pkg(n){if let Some(q)=p.releases.iter().find(|x|x.version==*v){for d in &q.deps{m.entry(d.package.clone()).or_default().push(d.constraint);}}}}

    }

    let aa:std::collections::BTreeMap<&str,Version>=a.iter().map(|x|(x.package.as_str(),x.version)).collect();

pub fn resolve(catalog:&[Package],roots:&[Requirement],locks:&[Lock],max_attempts:usize)->Result<Resolution,ResolveError>{

    w.walk(&mut std::collections::BTreeMap::new());
