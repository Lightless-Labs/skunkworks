- Validation runs completely before duplicate checks; name errors follow the specified traversal.
- Dynamic errors are checked in order: missing package, selected-version conflict, then empty candidates.
- A zero attempt budget still permits root-level errors and empty success; only candidate assignments consume budget.
- Locks affect candidate order and the global optimization cost, but do not activate packages or constrain versions.
- The exhaustive search chooses minimum lock changes, then the lexicographically greatest version vector; `Limit` prevents claiming an unproven optimum.
- Constraint and error formatting is canonical, percent-encoded, and has no trailing newline.

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],

    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

type Assignments = BTreeMap<String, Version>;
type Active = BTreeMap<String, Vec<Constraint>>;

fn active_constraints(roots: &[Requirement], selected: &Assignments, catalog: &BTreeMap<String, &Package>) -> Active {
    let mut active = BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in selected {
        if let Some(package) = catalog.get(name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for requirements in active.values_mut() {
        *requirements = canonical_constraints(requirements);
    }
    active
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: BTreeMap<String, &'a Package>,
    locks: BTreeMap<String, Version>,
    names: Vec<String>,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Option<Version>>, Assignments, Active)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn inspect(&mut self, selected: &mut Assignments) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, selected, &self.catalog);

        if let Some(name) = active.keys().find(|name| !self.catalog.contains_key(*name)) {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in selected.iter() {
            if let Some(reqs) = active.get(name) {
                if !reqs.iter().all(|c| constraint_matches(*c, *version)) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: reqs.clone(),
                    });
                }
            }
        }

        for (name, reqs) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let package = self.catalog[name];
            if !package.releases.iter().any(|release| {
                reqs.iter()
                    .all(|c| constraint_matches(*c, release.version))
            }) {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: reqs.clone(),
                });
            }
        }

        let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
        let Some(name) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(package, version)| selected.get(*package) != Some(*version))
                .count();
            let vector = self
                .names
                .iter()
                .map(|name| selected.get(name).copied())
                .collect::<Vec<_>>();
            let replace = self.best.as_ref().map_or(true, |(best_cost, best_vector, _, _)| {
                cost < *best_cost || (cost == *best_cost && vector > *best_vector)
            });
            if replace {
                self.best = Some((cost, vector, selected.clone(), active));
            }
            return Ok(());
        };

        let reqs = active.get(&name).cloned().unwrap_or_default();
        let package = self.catalog[&name];
        let mut candidates = package
            .releases
            .iter()
            .filter(|release| {
                reqs.iter()
                    .all(|c| constraint_matches(*c, release.version))
            })
            .map(|release| release.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(locked) = self.locks.get(&name) {
            if let Some(index) = candidates.iter().position(|v| v == locked) {
                let value = candidates.remove(index);
                candidates.insert(0, value);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            match self.inspect(selected) {
                Ok(()) => {}
                Err(error @ ResolveError::Limit { .. }) => {
                    selected.remove(&name);
                    return Err(error);
                }
                Err(error) => {
                    self.record_error(error);
                }
            }
            selected.remove(&name);
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let catalog_map = catalog
        .iter()
        .map(|package| (package.name.clone(), package))
        .collect::<BTreeMap<_, _>>();
    let lock_map = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect::<BTreeMap<_, _>>();
    let mut names = catalog.iter().map(|p| p.name.clone()).collect::<Vec<_>>();
    names.sort();

    let mut search = Search {
        roots,
        catalog: catalog_map,
        locks: lock_map,
        names,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut selected = Assignments::new();
    search.inspect(&mut selected)?;

    let Some((_, _, chosen, final_active)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: roots
                .first()
                .map(|r| r.package.clone())
                .unwrap_or_default(),
            requirements: Vec::new(),
        }));
    };

    let selected_out = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let mut before = BTreeMap::<String, Version>::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }
    let mut after = chosen.clone();
    let all_packages = before
        .keys()
        .chain(after.keys())
        .cloned()
        .collect::<BTreeSet<_>>();

    let mut changes = Vec::new();
    for package in all_packages {
        let b = before.get(&package).copied();
        let a = after.remove(&package);
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .map_or(false, |p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active.get(&package).map_or(false, |reqs| {
                    !reqs.iter().all(|c| constraint_matches(*c, old))
                }) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}

fn validate(c: &[Package], r: &[Requirement], l: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in r.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, pkg) in c.iter().enumerate() {
        if !valid_name(&pkg.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: pkg.name.clone(),
            });
        }
        for (j, rel) in pkg.releases.iter().enumerate() {
            for (k, dep) in rel.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for pkg in c {
        if !package_names.insert(pkg.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: pkg.name.clone(),
            });
        }
    }
    for pkg in c {
        let mut versions = BTreeSet::new();
        for rel in &pkg.releases {
            if !versions.insert(rel.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: pkg.name.clone(),
                    version: rel.version,
                });
            }
        }
    }
    for (i, lock) in l.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut locks = BTreeSet::new();
    for lock in l {
        if !locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

fn collect(
    roots: &[Requirement],
    chosen: &Chosen,
    packages: &BTreeMap<String, &Package>,
) -> RequirementsByName {
    let mut map = RequirementsByName::new();
    for root in roots {
        map.entry(root.package.clone()).or_default().push(root.constraint);

    for (name, version) in chosen {
        if let Some(package) = packages.get(name) {

struct Resolver<'a> {
    roots: &'a [Requirement],
    packages: BTreeMap<String, &'a Package>,
    locks: BTreeMap<String, Version>,
    names: Vec<String>,
    budget: usize,
    used: usize,
    best: Option<(usize, Vec<Option<Version>>, Chosen, RequirementsByName)>,
    first_leaf: Option<ResolveError>,
}

impl<'a> Resolver<'a> {
    fn leaf(&mut self, e: ResolveError) {
        if self.first_leaf.is_none() {
            self.first_leaf = Some(e);
        }
    }

    fn visit(&mut self, chosen: &mut Chosen) -> Result<(), ResolveError> {
        let reqs = collect(self.roots, chosen, &self.packages);

        if let Some(name) = reqs.keys().find(|name| !self.packages.contains_key(*name)) {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in chosen.iter() {
            if let Some(all) = reqs.get(name) {
                if !all.iter().all(|c| matches(*c, *version)) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: all.clone(),
                    });
                }
            }
        }

        for (name, all) in &reqs {
            if chosen.contains_key(name) {
                continue;
            }
            let pkg = self.packages[name];
            if !pkg.releases.iter().any(|rel| {
                all.iter().all(|c| matches(*c, rel.version))
            }) {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: all.clone(),
                });
            }
        }

        let pending = reqs.keys().find(|name| !chosen.contains_key(*name)).cloned();
        let Some(name) = pending else {
            let cost = self
                .locks
                .iter()
                .filter(|(name, version)| chosen.get(*name) != Some(*version))
                .count();
            let vector = self.names.iter().map(|n| chosen.get(n).copied()).collect::<Vec<_>>();
            let better = self.best.as_ref().map_or(true, |(bc, bv, _, _)| {
                cost < *bc || (cost == *bc && vector > *bv)
            });
            if better {
                self.best = Some((cost, vector, chosen.clone(), reqs));
            }
            return Ok(());
        };

        let all = reqs.get(&name).cloned().unwrap_or_default();
        let mut versions = self.packages[&name]
            .releases
            .iter()
            .filter(|rel| all.iter().all(|c| matches(*c, rel.version)))
            .map(|rel| rel.version)
            .collect::<Vec<_>>();
        versions.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.get(&name) {
            if let Some(pos) = versions.iter().position(|v| v == lock) {
                let locked = versions.remove(pos);
                versions.insert(0, locked);
            }
        }

        for version in versions {
            if self.used == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.used += 1;
            chosen.insert(name.clone(), version);
            match self.visit(chosen) {
                Ok(()) => {}
                Err(e @ ResolveError::Limit { .. }) => {
                    chosen.remove(&name);
                    return Err(e);
                }
                Err(e) => self.leaf(e),
            }
            chosen.remove(&name);
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;
    let packages = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect::<BTreeMap<_, _>>();
    let locks_map = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect::<BTreeMap<_, _>>();
    let mut names = catalog.iter().map(|p| p.name.clone()).collect::<Vec<_>>();
    names.sort();

    let mut resolver = Resolver {
        roots,
        packages,
        locks: locks_map,
        names,
        budget: max_attempts,
        used: 0,
        best: None,
        first_leaf: None,
    };
    let mut chosen = Chosen::new();
    resolver.visit(&mut chosen)?;

    let Some((_, _, chosen, final_reqs)) = resolver.best else {
        if let Some(error) = resolver.first_leaf {
            return Err(error);
        }
        return Err(ResolveError::NoMatch {
            package: roots.first().map(|r| r.package.clone()).unwrap_or_default(),
            requirements: Vec::new(),
        });
    };

    let selected = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let before = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect::<BTreeMap<_, _>>();
    let all_names = before
        .keys()
        .chain(chosen.keys())
        .cloned()
        .collect::<BTreeSet<_>>();
    let mut changes = Vec::new();

    for package in all_names {
        let old = before.get(&package).copied();
        let new = chosen.get(&package).copied();
        if old == new {
            continue;
        }
        let reason = match (old, new) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old_version), Some(_)) => {
                let available = resolver
                    .packages
                    .get(&package)
                    .map_or(false, |pkg| pkg.releases.iter().any(|r| r.version == old_version));
                if !available {
                    ChangeReason::Unavailable
                } else if final_reqs.get(&package).map_or(false, |all| {
                    !all.iter().all(|c| matches(*c, old_version))
                }) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before: old,
            after: new,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
