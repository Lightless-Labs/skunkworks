        return Ok(String::new());
    }

    for i in 0..rows.len() {
        if rows[i].len() == 0 {
            first_empty = Some(i);
            break;
        }

        return Err(TableError::EmptyRow { row });
    }

                row: i,
                expected: n,
                actual: rows[i].len(),
            });

            expected: n,
            actual: options.alignments.len(),

        None => usize::MAX,

            minimum,
            provided: options.max_width,

                        column: ci,
                        byte,

        }

    }

    Center,

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {

        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

        if row.len() != columns {
            return Err(TableError::Ragged {
                row: row_index,
                expected: columns,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }

    let fixed = columns
        .checked_mul(3)
        .and_then(|v| v.checked_add(1))
        .unwrap_or(usize::MAX);
    let minimum = columns
        .checked_add(fixed)
        .unwrap_or(usize::MAX);

    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
            }
        }
    }

    let trimmed: Vec<Vec<&str>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| cell.trim_end_matches(' '))
                .collect()
        })
        .collect();

        for (column, cell) in row.iter().enumerate() {
            widths[column] = widths[column].max(cell.chars().count());
        }
    }

    if options.max_width != 0 {
        loop {
            let total = widths
                .iter()
                .try_fold(fixed, |sum, &width| sum.checked_add(width))
                .unwrap_or(usize::MAX);
            if total <= options.max_width {
                break;
            }

            let mut candidate: Option<usize> = None;

                    candidate = Some(column);

                    provided: options.max_width,

        output.push('\n');

                    clipped.push(ch);
                }
                clipped.push('…');
                clipped
            } else {
                (*cell).to_owned()

                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),

            output.push(' ');

            }
            output.push_str(&content);
            for _ in 0..right {
                output.push(' ');
            }
            output.push(' ');
            output.push('|');
        }
        output.push('\n');
    }

    let alignments: &[Alignment] = if options.alignments.is_empty() {
        left_alignments = vec![Alignment::Left; columns];
        &left_alignments
    } else {
        &options.alignments

        if options.header && row_index == 0 && rows.len() > 1 {
            border(&mut output, &widths);
        }
    }
    border(&mut output, &widths);

    Ok(output)
}

                }
            }
            if let Some(column) = selected {
                widths[column] -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    append_border(&mut output, &widths);

    for (row_index, row) in trimmed.iter().enumerate() {
        for (column, cell) in row.iter().enumerate() {
            let content: String = if cell.chars().count() > widths[column] {
                let prefix_len = widths[column] - 1;
                let mut prefix: String = cell.chars().take(prefix_len).collect();
                prefix.push('…');
                prefix
            } else {
                (*cell).to_owned()
            };

            let content_width = content.chars().count();
            let extra = widths[column] - content_width;
            let (left, right) = match alignment_at(options, column) {
                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),
            };

            output.push('|');
            output.push(' ');
            push_spaces(&mut output, left);
            output.push_str(&content);
            push_spaces(&mut output, right);
            output.push(' ');
        }
        output.push_str("|\n");

        if options.header && row_index == 0 && rows.len() > 1 {
            append_border(&mut output, &widths);
        }
    }

    append_border(&mut output, &widths);
    Ok(output)
}

fn alignment_at(options: &Options, column: usize) -> Alignment {
    if options.alignments.is_empty() {
        Alignment::Left
    } else {
        options.alignments[column].clone()
    }
}

fn total_width(widths: &[usize]) -> usize {
    widths.iter().fold(1usize, |sum, width| {
        sum.saturating_add(width.saturating_add(3))
    })
}

fn append_border(output: &mut String, widths: &[usize]) {
    output.push('+');
    for width in widths {
        push_repeated(output, '-', width.saturating_add(2));
        output.push('+');
    }
    output.push('\n');
}

fn push_repeated(output: &mut String, ch: char, count: usize) {
    for _ in 0..count {
        output.push(ch);
    }
}

fn push_spaces(output: &mut String, count: usize) {
    push_repeated(output, ' ', count);
}

                        row: r,
                        column: c,
                        byte,
                    });
                }

            cleaned.push(cell.trim_end_matches(' ').to_string());
        }
        contents.push(cleaned);
    }

    let mut widths = vec![1; columns];
    for row in &contents {
        for (column, cell) in row.iter().enumerate() {
            let count = cell.chars().count();
            if count > widths[column] {
                widths[column] = count;
            }
        }
    }

    if options.max_width > 0 {
        let mut sum: usize = widths.iter().fold(minimum - columns, |a, b| a.saturating_add(*b));
        while sum > options.max_width {
            let mut best: Option<usize> = None;
            for i in 0..columns {
                if widths[i] > 1 && best.is_none_or(|j| widths[i] >= widths[j]) {
                    best = Some(i);
                }
            }
            if let Some(i) = best {
                widths[i] -= 1;
                sum -= 1;
            } else {
                break;
            }
        }
    }

        let mut s = String::new();
        s.push('+');
        for width in widths {
            s.extend(std::iter::repeat_n('-', width.saturating_add(2)));
            s.push('+');
        }
        s.push('\n');
        s
    };

                Alignment::Left => 0,
                Alignment::Right => remaining,
                Alignment::Center => remaining / 2,

            result.push(' ');
            result.extend(std::iter::repeat_n(' ', left));
            result.push_str(&value);
            result.extend(std::iter::repeat_n(' ', right));
            result.push(' ');
            result.push('|');
        }
        result.push('\n');

        if options.header && ri == 0 && contents.len() > 1 {
            result.push_str(&make_border(&widths));
        }
    }
    result.push_str(&make_border(&widths));
    Ok(result)
}
