- Validation order is exact: empty table, all empty rows, ragged rows, alignment count, width budget, then invalid characters.
- Empty rows are checked across the whole input before any ragged-row error.
- The width budget counts Unicode scalar values and includes the fixed border and padding overhead; zero means unlimited.
- Width reduction shrinks the rightmost widest eligible column one scalar at a time, never below one.
- Trailing ASCII spaces are trimmed before measuring and truncating; all other accepted characters are preserved.
- Both implementations use checked arithmetic for width calculations; if sizes exceed `usize`, they return `TooNarrow` with saturated `minimum`.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let columns = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != columns {
            return Err(TableError::Ragged {
                row: row_index,
                expected: columns,
                actual: row.len(),
            });
        }
    }

    let alignment_count = options.alignments.len();
    if alignment_count != 0 && alignment_count != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: alignment_count,
        });
    }

    let minimum = columns
        .checked_mul(4)
        .and_then(|value| value.checked_add(1))
        .unwrap_or(usize::MAX);
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
            }
        }
    }

    let trimmed: Vec<Vec<&str>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| cell.trim_end_matches(' '))
                .collect()
        })
        .collect();

    let mut widths = vec![1usize; columns];
    for row in &trimmed {
        for (column, cell) in row.iter().enumerate() {
            widths[column] = widths[column].max(cell.chars().count());
        }
    }

    if options.max_width != 0 {
        while total_width(&widths) > options.max_width {
            let mut selected: Option<usize> = None;
            for column in 0..columns {
                if widths[column] > 1
                    && selected.map_or(true, |current| widths[column] >= widths[current])
                {
                    selected = Some(column);
                }
            }
            if let Some(column) = selected {
                widths[column] -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    append_border(&mut output, &widths);

    for (row_index, row) in trimmed.iter().enumerate() {
        for (column, cell) in row.iter().enumerate() {
            let content: String = if cell.chars().count() > widths[column] {
                let prefix_len = widths[column] - 1;
                let mut prefix: String = cell.chars().take(prefix_len).collect();
                prefix.push('…');
                prefix
            } else {
                (*cell).to_owned()
            };

            let content_width = content.chars().count();
            let extra = widths[column] - content_width;
            let (left, right) = match alignment_at(options, column) {
                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),
            };

            output.push('|');
            output.push(' ');
            push_spaces(&mut output, left);
            output.push_str(&content);
            push_spaces(&mut output, right);
            output.push(' ');
        }
        output.push_str("|\n");

        if options.header && row_index == 0 && rows.len() > 1 {
            append_border(&mut output, &widths);
        }
    }

    append_border(&mut output, &widths);
    Ok(output)
}

fn alignment_at(options: &Options, column: usize) -> Alignment {
    if options.alignments.is_empty() {
        Alignment::Left
    } else {
        options.alignments[column].clone()
    }
}

fn total_width(widths: &[usize]) -> usize {
    widths.iter().fold(1usize, |sum, width| {
        sum.saturating_add(width.saturating_add(3))
    })
}

fn append_border(output: &mut String, widths: &[usize]) {
    output.push('+');
    for width in widths {
        push_repeated(output, '-', width.saturating_add(2));
        output.push('+');
    }
    output.push('\n');
}

fn push_repeated(output: &mut String, ch: char, count: usize) {
    for _ in 0..count {
        output.push(ch);
    }
}

fn push_spaces(output: &mut String, count: usize) {
    push_repeated(output, ' ', count);
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.len() == 0 {
        return Ok(String::new());
    }

    let mut first_empty = None;
    for i in 0..rows.len() {
        if rows[i].len() == 0 {
            first_empty = Some(i);
            break;
        }
    }
    if let Some(row) = first_empty {
        return Err(TableError::EmptyRow { row });
    }

    let n = rows[0].len();
    for i in 0..rows.len() {
        if rows[i].len() != n {
            return Err(TableError::Ragged {
                row: i,
                expected: n,
                actual: rows[i].len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    let minimum = match n.checked_mul(4).and_then(|v| v.checked_add(1)) {
        Some(value) => value,
        None => usize::MAX,
    };
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (ri, row) in rows.iter().enumerate() {
        for (ci, cell) in row.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar {
                        row: ri,
                        column: ci,
                        byte,
                    });
                }
            }
        }
    }

    let mut cells: Vec<Vec<Vec<char>>> = Vec::with_capacity(rows.len());
    let mut widths = vec![1usize; n];

    for row in rows {
        let mut converted = Vec::with_capacity(n);
        for (ci, cell) in row.iter().enumerate() {
            let chars: Vec<char> = cell.trim_end_matches(' ').chars().collect();
            widths[ci] = widths[ci].max(chars.len());
            converted.push(chars);
        }
        cells.push(converted);
    }

    if options.max_width != 0 {
        let mut current_total = 1usize;
        for width in &widths {
            current_total = current_total.saturating_add(width.saturating_add(3));
        }
        while current_total > options.max_width {
            let mut candidate = None;
            for ci in 0..n {
                if widths[ci] > 1 && candidate.map_or(true, |old| widths[ci] >= widths[old]) {
                    candidate = Some(ci);
                }
            }
            match candidate {
                Some(ci) => {
                    widths[ci] -= 1;
                    current_total -= 1;
                }
                None => break,
            }
        }
    }

    let mut output = String::new();
    border(&mut output, &widths);

    for ri in 0..cells.len() {
        output.push('|');
        for ci in 0..n {
            let source = &cells[ri][ci];
            let width = widths[ci];
            let mut displayed = Vec::new();
            if source.len() > width {
                displayed.extend(source.iter().take(width - 1).copied());
                displayed.push('…');
            } else {
                displayed.extend(source.iter().copied());
            }

            let spare = width - displayed.len();
            let (left, right) = match if options.alignments.is_empty() {
                &Alignment::Left
            } else {
                &options.alignments[ci]
            } {
                Alignment::Left => (0, spare),
                Alignment::Right => (spare, 0),
                Alignment::Center => (spare / 2, spare - spare / 2),
            };

            spaces(&mut output, 1 + left);
            for ch in displayed {
                output.push(ch);
            }
            spaces(&mut output, right + 1);
            output.push('|');
        }
        output.push('\n');

        if options.header && ri == 0 && cells.len() > 1 {
            border(&mut output, &widths);
        }
    }

    border(&mut output, &widths);
    output
        .shrink_to_fit();
    Ok(output)
}

fn border(out: &mut String, widths: &[usize]) {
    out.push('+');
    for &width in widths {
        for _ in 0..width.saturating_add(2) {
            out.push('-');
        }
        out.push('+');
    }
    out.push('\n');
}

fn spaces(out: &mut String, count: usize) {
    for _ in 0..count {
        out.push(' ');
    }
}
```
