### Solution A

- Check all empty rows before checking for ragged rows; then check alignment count, width budget, and invalid characters in that order.
- Treat width as Unicode scalar count, with a minimum width of one per column.
- Remove trailing ASCII spaces before measuring, truncating, and aligning.
- When shrinking, decrement the widest eligible column, choosing the rightmost on ties.
- Interpret the budget as a maximum line width; never expand columns to use spare width.
- Use checked arithmetic for size calculations; if dimensions exceed representable sizes, return `TooNarrow`.

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let columns = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != columns {
            return Err(TableError::Ragged {
                row: row_index,
                expected: columns,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }

    let fixed = columns
        .checked_mul(3)
        .and_then(|v| v.checked_add(1))
        .unwrap_or(usize::MAX);
    let minimum = columns
        .checked_add(fixed)
        .unwrap_or(usize::MAX);

    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
            }
        }
    }

    let trimmed: Vec<Vec<&str>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| cell.trim_end_matches(' '))
                .collect()
        })
        .collect();

    let mut widths = vec![1usize; columns];
    for row in &trimmed {
        for (column, cell) in row.iter().enumerate() {
            widths[column] = widths[column].max(cell.chars().count());
        }
    }

    if options.max_width != 0 {
        loop {
            let total = widths
                .iter()
                .try_fold(fixed, |sum, &width| sum.checked_add(width))
                .unwrap_or(usize::MAX);
            if total <= options.max_width {
                break;
            }

            let mut candidate: Option<usize> = None;
            for column in 0..columns {
                if widths[column] > 1
                    && candidate.map_or(true, |best| widths[column] >= widths[best])
                {
                    candidate = Some(column);
                }
            }

            if let Some(column) = candidate {
                widths[column] -= 1;
            } else {
                return Err(TableError::TooNarrow {
                    minimum,
                    provided: options.max_width,
                });
            }
        }
    }

    let mut output = String::new();

    fn border(output: &mut String, widths: &[usize]) {
        output.push('+');
        for &width in widths {
            for _ in 0..width + 2 {
                output.push('-');
            }
            output.push('+');
        }
        output.push('\n');
    }

    fn data_line(
        output: &mut String,
        row: &[&str],
        widths: &[usize],
        alignments: &[Alignment],
    ) {
        output.push('|');
        for (column, cell) in row.iter().enumerate() {
            let width = widths[column];
            let count = cell.chars().count();
            let content = if count > width {
                let mut clipped = String::new();
                for ch in cell.chars().take(width - 1) {
                    clipped.push(ch);
                }
                clipped.push('…');
                clipped
            } else {
                (*cell).to_owned()
            };
            let content_width = content.chars().count();
            let extra = width - content_width;
            let (left, right) = match alignments[column] {
                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),
            };

            output.push(' ');
            for _ in 0..left {
                output.push(' ');
            }
            output.push_str(&content);
            for _ in 0..right {
                output.push(' ');
            }
            output.push(' ');
            output.push('|');
        }
        output.push('\n');
    }

    let left_alignments;
    let alignments: &[Alignment] = if options.alignments.is_empty() {
        left_alignments = vec![Alignment::Left; columns];
        &left_alignments
    } else {
        &options.alignments
    };

    border(&mut output, &widths);
    for (row_index, row) in trimmed.iter().enumerate() {
        data_line(&mut output, row, &widths, alignments);
        if options.header && row_index == 0 && rows.len() > 1 {
            border(&mut output, &widths);
        }
    }
    border(&mut output, &widths);

    Ok(output)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    if let Some((row, _)) = rows.iter().enumerate().find(|(_, row)| row.is_empty()) {
        return Err(TableError::EmptyRow { row });
    }

    let n = rows[0].len();
    if let Some((row, cells)) = rows.iter().enumerate().find(|(_, row)| row.len() != n) {
        return Err(TableError::Ragged {
            row,
            expected: n,
            actual: cells.len(),
        });
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    let overhead = n.checked_mul(3).and_then(|v| v.checked_add(1)).unwrap_or(usize::MAX);
    let minimum = overhead.checked_add(n).unwrap_or(usize::MAX);
    if options.max_width > 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (r, row) in rows.iter().enumerate() {
        for (c, cell) in row.iter().enumerate() {
            if let Some((byte, _)) = cell
                .char_indices()
                .find(|(_, ch)| ch.is_control() || *ch == '\u{2028}' || *ch == '\u{2029}')
            {
                return Err(TableError::InvalidChar {
                    row: r,
                    column: c,
                    byte,
                });
            }
        }
    }

    let cells: Vec<Vec<String>> = rows
        .iter()
        .map(|row| row.iter().map(|s| s.trim_end_matches(' ').to_owned()).collect())
        .collect();

    let mut widths = vec![1usize; n];
    for row in &cells {
        for (c, cell) in row.iter().enumerate() {
            widths[c] = widths[c].max(cell.chars().count());
        }
    }

    if options.max_width != 0 {
        loop {
            let sum = widths
                .iter()
                .try_fold(overhead, |acc, &w| acc.checked_add(w))
                .unwrap_or(usize::MAX);
            if sum <= options.max_width {
                break;
            }
            let widest = (0..n)
                .filter(|&c| widths[c] > 1)
                .max_by_key(|&c| (widths[c], c));
            match widest {
                Some(c) => widths[c] -= 1,
                None => {
                    return Err(TableError::TooNarrow {
                        minimum,
                        provided: options.max_width,
                    });
                }
            }
        }
    }

    let default_alignments = vec![Alignment::Left; n];
    let aligns = if options.alignments.is_empty() {
        &default_alignments
    } else {
        &options.alignments
    };

    let mut out = String::new();
    let write_border = |out: &mut String| {
        out.push('+');
        for &w in &widths {
            for _ in 0..w + 2 {
                out.push('-');
            }
            out.push('+');
        }
        out.push('\n');
    };

    write_border(&mut out);
    for (r, row) in cells.iter().enumerate() {
        out.push('|');
        for c in 0..n {
            let w = widths[c];
            let mut displayed = String::new();
            let chars: Vec<char> = row[c].chars().collect();
            if chars.len() > w {
                displayed.extend(chars.iter().take(w - 1));
                displayed.push('…');
            } else {
                displayed.push_str(&row[c]);
            }
            let content_len = displayed.chars().count();
            let padding = w - content_len;
            let (left, right) = match aligns[c] {
                Alignment::Left => (0, padding),
                Alignment::Right => (padding, 0),
                Alignment::Center => (padding / 2, padding - padding / 2),
            };
            out.push(' ');
            out.extend(std::iter::repeat(' ').take(left));
            out.push_str(&displayed);
            out.extend(std::iter::repeat(' ').take(right));
            out.push(' ');
            out.push('|');
        }
        out.push('\n');
        if options.header && r == 0 && cells.len() > 1 {
            write_border(&mut out);
        }
    }
    write_border(&mut out);
    Ok(out)
}
```
