- Dimension checks run before all other constructor checks; the limit is inclusive at 4096.
- COO entries are validated in input order before any summation; then each coordinate is summed in input order.
- CSR pointer endpoints are checked before pointer monotonicity, and monotonicity before any slot access.
- CSR slots are checked in row/slot order, with bounds before ordering, finiteness, and explicit-zero checks.
- Arithmetic overflow is checked after each specified multiplication or addition; later cancellation cannot repair it.
- Both implementations use standard `HashMap` accumulation for COO and ordered traversal for CSR operations; results are sorted where canonical order requires it.

### Solution A

```rust
use std::collections::HashMap;

pub const MAX_DIM: usize = 4096;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Axis {
    Row,
    Column,
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Entry {
    pub row: usize,
    pub col: usize,
    pub value: f64,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Sparse {
    rows: usize,
    cols: usize,
    entries: Vec<Entry>,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Csr {
    rows: usize,
    cols: usize,
    row_ptr: Vec<usize>,
    col_idx: Vec<usize>,
    values: Vec<f64>,
}

#[derive(Clone, Debug, PartialEq)]
pub enum Error {
    Shape { axis: Axis, size: usize },
    Bounds { entry: usize, axis: Axis, index: usize, limit: usize },
    NonFinite { entry: usize },
    SumOverflow { entry: usize },

    RowPtrLength { expected: usize, actual: usize },
    ColumnValueLength { columns: usize, values: usize },
    RowPtrStart { found: usize },
    RowPtrEnd { found: usize, expected: usize },
    RowPtrOrder { row: usize },
    ColumnBounds { slot: usize, column: usize, limit: usize },
    ColumnOrder { slot: usize },
    NonFiniteValue { slot: usize },
    ExplicitZero { slot: usize },
    VectorLength { expected: usize, actual: usize },
    NonFiniteVector { index: usize },
    ProductOverflow { row: usize, column: usize },
    AccumulationOverflow { row: usize, column: usize },

    DimensionMismatch { left_cols: usize, right_rows: usize },
    MultiplyOverflow { row: usize, inner: usize, column: usize },
    MultiplySumOverflow { row: usize, inner: usize, column: usize },
}

fn check_shape(rows: usize, cols: usize) -> Result<(), Error> {
    if rows > MAX_DIM {
        return Err(Error::Shape { axis: Axis::Row, size: rows });
    }
    if cols > MAX_DIM {
        return Err(Error::Shape { axis: Axis::Column, size: cols });
    }
    Ok(())
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        check_shape(rows, cols)?;

        for (i, e) in entries.iter().enumerate() {
            if e.row >= rows {
                return Err(Error::Bounds { entry: i, axis: Axis::Row, index: e.row, limit: rows });
            }
            if e.col >= cols {
                return Err(Error::Bounds { entry: i, axis: Axis::Column, index: e.col, limit: cols });
            }
            if !e.value.is_finite() {
                return Err(Error::NonFinite { entry: i });
            }
        }

        let mut sums: HashMap<(usize, usize), f64> = HashMap::new();
        for (i, e) in entries.iter().enumerate() {
            let sum = sums.entry((e.row, e.col)).or_insert(0.0);
            *sum += e.value;
            if !sum.is_finite() {
                return Err(Error::SumOverflow { entry: i });
            }
        }

        let mut canonical: Vec<Entry> = sums
            .into_iter()
            .filter_map(|((row, col), value)| {
                if value == 0.0 {
                    None
                } else {
                    Some(Entry { row, col, value })
                }
            })
            .collect();
        canonical.sort_by(|a, b| a.row.cmp(&b.row).then_with(|| a.col.cmp(&b.col)));
        Ok(Self { rows, cols, entries: canonical })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn entries(&self) -> &[Entry] {
        &self.entries
    }

    pub fn to_csr(&self) -> Csr {
        let mut row_ptr = vec![0; self.rows + 1];
        for e in &self.entries {
            row_ptr[e.row + 1] += 1;
        }
        for r in 0..self.rows {
            row_ptr[r + 1] += row_ptr[r];
        }
        Csr {
            rows: self.rows,
            cols: self.cols,
            row_ptr,
            col_idx: self.entries.iter().map(|e| e.col).collect(),
            values: self.entries.iter().map(|e| e.value).collect(),
        }
    }
}

impl Csr {
    pub fn from_parts(
        rows: usize,
        cols: usize,
        row_ptr: Vec<usize>,
        col_idx: Vec<usize>,
        values: Vec<f64>,
    ) -> Result<Self, Error> {
        check_shape(rows, cols)?;
        let expected_ptr = rows + 1;
        if row_ptr.len() != expected_ptr {
            return Err(Error::RowPtrLength { expected: expected_ptr, actual: row_ptr.len() });
        }
        if col_idx.len() != values.len() {
            return Err(Error::ColumnValueLength { columns: col_idx.len(), values: values.len() });
        }
        if row_ptr[0] != 0 {
            return Err(Error::RowPtrStart { found: row_ptr[0] });
        }
        if row_ptr[rows] != values.len() {
            return Err(Error::RowPtrEnd { found: row_ptr[rows], expected: values.len() });
        }
        for r in 0..rows {
            if row_ptr[r] > row_ptr[r + 1] {
                return Err(Error::RowPtrOrder { row: r });
            }
        }
        for r in 0..rows {
            let start = row_ptr[r];
            let end = row_ptr[r + 1];
            let mut prev: Option<usize> = None;
            for s in start..end {
                let column = col_idx[s];
                if column >= cols {
                    return Err(Error::ColumnBounds { slot: s, column, limit: cols });
                }
                if let Some(p) = prev {
                    if column <= p {
                        return Err(Error::ColumnOrder { slot: s });
                    }
                }
                if !values[s].is_finite() {
                    return Err(Error::NonFiniteValue { slot: s });
                }
                if values[s] == 0.0 {
                    return Err(Error::ExplicitZero { slot: s });
                }
                prev = Some(column);
            }
        }
        Ok(Self { rows, cols, row_ptr, col_idx, values })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn row_ptr(&self) -> &[usize] {
        &self.row_ptr
    }

    pub fn col_indices(&self) -> &[usize] {
        &self.col_idx
    }

    pub fn values(&self) -> &[f64] {
        &self.values
    }

    pub fn to_coo(&self) -> Sparse {
        let mut entries = Vec::with_capacity(self.values.len());
        for r in 0..self.rows {
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                entries.push(Entry { row: r, col: self.col_idx[s], value: self.values[s] });
            }
        }
        Sparse { rows: self.rows, cols: self.cols, entries }
    }

    pub fn mul_vector(&self, x: &[f64]) -> Result<Vec<f64>, Error> {
        if x.len() != self.cols {
            return Err(Error::VectorLength { expected: self.cols, actual: x.len() });
        }
        for (i, &v) in x.iter().enumerate() {
            if !v.is_finite() {
                return Err(Error::NonFiniteVector { index: i });
            }
        }

        let mut out = Vec::with_capacity(self.rows);
        for r in 0..self.rows {
            let mut acc = 0.0;
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                let c = self.col_idx[s];
                let p = self.values[s] * x[c];
                if !p.is_finite() {
                    return Err(Error::ProductOverflow { row: r, column: c });
                }
                acc += p;
                if !acc.is_finite() {
                    return Err(Error::AccumulationOverflow { row: r, column: c });
                }
            }
            if acc == 0.0 {
                acc = 0.0;
            }
            out.push(acc);
        }
        Ok(out)
    }

    pub fn transpose(&self) -> Csr {
        let mut counts = vec![0usize; self.cols];
        for &c in &self.col_idx {
            counts[c] += 1;
        }
        let mut row_ptr = vec![0usize; self.cols + 1];
        for r in 0..self.cols {
            row_ptr[r + 1] = row_ptr[r] + counts[r];
        }
        let mut next = row_ptr[..self.cols].to_vec();
        let mut col_idx = vec![0usize; self.values.len()];
        let mut values = vec![0.0; self.values.len()];
        for r in 0..self.rows {
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                let c = self.col_idx[s];
                let dest = next[c];
                next[c] += 1;
                col_idx[dest] = r;
                values[dest] = self.values[s];
            }
        }
        Csr { rows: self.cols, cols: self.rows, row_ptr, col_idx, values }
    }

    pub fn multiply(&self, rhs: &Csr) -> Result<Csr, Error> {
        if self.cols != rhs.rows {
            return Err(Error::DimensionMismatch { left_cols: self.cols, right_rows: rhs.rows });
        }

        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);

        for r in 0..self.rows {
            let mut accum: HashMap<usize, f64> = HashMap::new();
            for left_slot in self.row_ptr[r]..self.row_ptr[r + 1] {
                let k = self.col_idx[left_slot];
                let left_value = self.values[left_slot];
                for right_slot in rhs.row_ptr[k]..rhs.row_ptr[k + 1] {
                    let c = rhs.col_idx[right_slot];
                    let p = left_value * rhs.values[right_slot];
                    if !p.is_finite() {
                        return Err(Error::MultiplyOverflow { row: r, inner: k, column: c });
                    }
                    let sum = accum.entry(c).or_insert(0.0);
                    *sum += p;
                    if !sum.is_finite() {
                        return Err(Error::MultiplySumOverflow { row: r, inner: k, column: c });
                    }
                }
            }

            let mut row: Vec<(usize, f64)> = accum.into_iter().filter(|(_, v)| *v != 0.0).collect();
            row.sort_by_key(|(c, _)| *c);
            for (c, v) in row {
                col_idx.push(c);
                values.push(v);
            }
            row_ptr.push(values.len());
        }

        Ok(Csr { rows: self.rows, cols: rhs.cols, row_ptr, col_idx, values })
    }
}
```

```rust
impl Csr {
    pub fn from_parts(
        rows: usize,
        cols: usize,
        row_ptr: Vec<usize>,
        col_idx: Vec<usize>,
        values: Vec<f64>,
    ) -> Result<Self, Error> {
        check_shape(rows, cols)?;
        let expected = rows + 1;
        if row_ptr.len() != expected {
            return Err(Error::RowPtrLength { expected, actual: row_ptr.len() });
        }
        if col_idx.len() != values.len() {
            return Err(Error::ColumnValueLength { columns: col_idx.len(), values: values.len() });
        }
        if row_ptr[0] != 0 {
            return Err(Error::RowPtrStart { found: row_ptr[0] });

        if row_ptr[rows] != values.len() {
            return Err(Error::RowPtrEnd { found: row_ptr[rows], expected: values.len() });
        }
        for r in 0..rows {
            if row_ptr[r] > row_ptr[r + 1] {
                return Err(Error::RowPtrOrder { row: r });
            }
        }
        for r in 0..rows {
            let mut prev: Option<usize> = None;
            for s in row_ptr[r]..row_ptr[r + 1] {
                let column = col_idx[s];
                if column >= cols {
                    return Err(Error::ColumnBounds { slot: s, column, limit: cols });
                }
                if prev.is_some_and(|p| column <= p) {
                    return Err(Error::ColumnOrder { slot: s });
                }
                if !values[s].is_finite() {
                    return Err(Error::NonFiniteValue { slot: s });
                }
                if values[s] == 0.0 {
                    return Err(Error::ExplicitZero { slot: s });
                }
                prev = Some(column);
            }
        }
        Ok(Self { rows, cols, row_ptr, col_idx, values })
    }

        (self.rows, self.cols)

    pub fn row_ptr(&self) -> &[usize] {
        &self.row_ptr

        &self.col_idx

    pub fn values(&self) -> &[f64] {
        &self.values

    pub fn to_coo(&self) -> Sparse {

            }

        Sparse { rows: self.rows, cols: self.cols, entries }

    pub fn mul_vector(&self, x: &[f64]) -> Result<Vec<f64>, Error> {
        if x.len() != self.cols {
            return Err(Error::VectorLength { expected: self.cols, actual: x.len() });
        }
        for (i, &v) in x.iter().enumerate() {
            if !v.is_finite() {
                return Err(Error::NonFiniteVector { index: i });
            }
        }

        let mut out = Vec::with_capacity(self.rows);
        for r in 0..self.rows {
            let mut acc = 0.0;
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                let c = self.col_idx[s];
                let p = self.values[s] * x[c];
                if !p.is_finite() {
                    return Err(Error::ProductOverflow { row: r, column: c });
                }
                acc += p;
                if !acc.is_finite() {
                    return Err(Error::AccumulationOverflow { row: r, column: c });
                }
            }
            out.push(if acc == 0.0 { 0.0 } else { acc });
        }

    }

    pub fn transpose(&self) -> Csr {
        let mut counts = vec![0usize; self.cols];
        for &c in &self.col_idx {
            counts[c] += 1;
        }
        let mut row_ptr = vec![0usize; self.cols + 1];

        let mut col_idx = vec![0usize; self.values.len()];
        let mut values = vec![0.0; self.values.len()];

                let c = self.col_idx[s];
                let dest = next[c];
                next[c] += 1;
                col_idx[dest] = r;
                values[dest] = self.values[s];
            }
        }
        Csr { rows: self.cols, cols: self.rows, row_ptr, col_idx, values }
    }

        if self.cols != rhs.rows {
            return Err(Error::DimensionMismatch { left_cols: self.cols, right_rows: rhs.rows });
        }

        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);

            for left_slot in self.row_ptr[r]..self.row_ptr[r + 1] {

                    values.push(v);
                }
            }
            row_ptr.push(values.len());
        }
        Ok(Csr { rows: self.rows, cols: rhs.cols, row_ptr, col_idx, values })
    }
```
