- Validation checks `n`, then edge count, then endpoints in source-before-target order; incremental capacity is checked before batch endpoints.
- Component IDs follow decreasing component size, then increasing minimum vertex; all later component fields use those IDs.
- Topological priority is applied after every emitted component, with internal edge count and then largest member as the tie-break.
- Backward queries choose the lexicographically smallest shortest route from the destination to the source, then reverse it.
- A merge event includes every SCC joined by the insertion, including intermediate components made mutually reachable by existing paths.
- Both implementations use iterative graph traversal to support long paths without recursion; B uses a different SCC algorithm.

// Iterative Kosaraju SCC decomposition.
fn scc(n: usize, edges: &[(usize, usize)]) -> (Vec<Vec<usize>>, Vec<usize>) {
    let mut out = vec![Vec::new(); n];
    let mut rev = vec![Vec::new(); n];
    for &(u, v) in edges {
        out[u].push(v);
        rev[v].push(u);
    }

    let mut seen = vec![false; n];
    let mut finish = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((v, next)) = stack.last_mut() {
            if *next < out[*v].len() {
                let w = out[*v][*next];
                *next += 1;
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                finish.push(*v);
                stack.pop();
            }
        }
    }

    let mut raw_of = vec![usize::MAX; n];
    let mut raw = Vec::<Vec<usize>>::new();
    for &root in finish.iter().rev() {
        if raw_of[root] != usize::MAX {
            continue;
        }
        let id = raw.len();
        let mut members = Vec::new();
        let mut stack = vec![root];
        raw_of[root] = id;
        while let Some(v) = stack.pop() {
            members.push(v);
            for &w in &rev[v] {
                if raw_of[w] == usize::MAX {
                    raw_of[w] = id;
                    stack.push(w);
                }
            }
        }
        members.sort_unstable();
        raw.push(members);
    }

    let mut order: Vec<usize> = (0..raw.len()).collect();
    order.sort_by_key(|&i| (Reverse(raw[i].len()), raw[i][0]));
    let mut canonical_of_raw = vec![0; raw.len()];
    let mut components = Vec::with_capacity(raw.len());
    for (new_id, old_id) in order.into_iter().enumerate() {
        canonical_of_raw[old_id] = new_id;
        components.push(std::mem::take(&mut raw[old_id]));
    }
    let component_of = raw_of
        .into_iter()
        .map(|id| canonical_of_raw[id])
        .collect();
    (components, component_of)
}

fn build_graph(n: usize, edges: &[(usize, usize)]) -> Graph {
    let (components, component_of) = scc(n, edges);
    let k = components.len();
    let mut cyclic = vec![false; k];
    let mut internal_edges = vec![0usize; k];
    let mut arc_data: HashMap<(usize, usize), (usize, usize)> = HashMap::new();

    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_data.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
        }
    }
    for (id, members) in components.iter().enumerate() {
        if members.len() > 1 {
            cyclic[id] = true;
        }
    }

    let mut arcs: Vec<Arc> = arc_data
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by_key(|a| (a.from, Reverse(a.to)));

    let mut outgoing = vec![Vec::new(); k];
    let mut indegree = vec![0usize; k];
    for a in &arcs {
        outgoing[a.from].push(a.to);
        indegree[a.to] += 1;
    }
    let mut ready = BinaryHeap::new();
    for id in 0..k {
        if indegree[id] == 0 {
            ready.push((internal_edges[id], components[id][0], Reverse(id)));
        }
    }
    let mut topo = Vec::with_capacity(k);
    let mut depth = vec![0usize; k];
    while let Some((_, _, Reverse(id))) = ready.pop() {
        topo.push(id);
        for &to in &outgoing[id] {
            depth[to] = depth[to].max(depth[id] + 1);
            indegree[to] -= 1;
            if indegree[to] == 0 {
                ready.push((internal_edges[to], components[to][0], Reverse(to)));
            }
        }
    }
    Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    }
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    validate(n, edges)?;
    Ok(build_graph(n, edges))
}

fn shortest_route(g: &Graph, start: usize, target: usize) -> Option<Vec<usize>> {
    if start == target {
        return Some(vec![start]);
    }
    let k = g.components.len();
    let mut outgoing = vec![Vec::new(); k];
    for a in &g.arcs {
        outgoing[a.from].push(a.to);
    }
    for list in &mut outgoing {
        list.sort_unstable();
        list.dedup();
    }

    let mut dist = vec![usize::MAX; k];
    let mut q = VecDeque::new();
    dist[target] = 0;
    q.push_back(target);
    let mut incoming = vec![Vec::new(); k];
    for a in &g.arcs {
        incoming[a.to].push(a.from);
    }
    while let Some(v) = q.pop_front() {
        for &p in &incoming[v] {
            if dist[p] == usize::MAX {
                dist[p] = dist[v] + 1;
                q.push_back(p);
            }
        }
    }
    if dist[start] == usize::MAX {
        return None;
    }
    let mut route = vec![start];
    let mut cur = start;
    while cur != target {
        let next = outgoing[cur]
            .iter()
            .copied()
            .filter(|&v| dist[v].checked_add(1) == Some(dist[cur]))
            .min()
            .expect("distance invariant");
        route.push(next);
        cur = next;
    }
    Some(route)
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        let n = self.component_of.len();
        if from >= n {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }
        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            return Ok(if self.cyclic[a] {
                Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                }
            } else {
                Query {
                    kind: Relation::Trivial,
                    route: vec![a],
                }
            });
        }
        if let Some(route) = shortest_route(self, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_route(self, b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

#[derive(Clone, Debug)]
pub struct Incremental {
    n: usize,
    edges: Vec<(usize, usize)>,
    graph: Graph,
    history: Vec<Merge>,
    pairs: HashSet<(usize, usize)>,
}

impl Incremental {
    pub fn new(n: usize, edges: &[(usize, usize)]) -> Result<Self, GraphError> {
        let graph = analyze(n, edges)?;
        Ok(Self {
            n,
            edges: edges.to_vec(),
            graph,
            history: Vec::new(),
            pairs: edges.iter().copied().collect(),
        })
    }

    pub fn snapshot(&self) -> Graph {
        self.graph.clone()
    }

    pub fn history(&self) -> Vec<Merge> {
        self.history.clone()
    }

    pub fn insert(&mut self, edges: &[(usize, usize)]) -> Result<Batch, GraphError> {
        if self.edges.len() + edges.len() > MAX_EDGES {
            return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
        }
        for (i, &(u, v)) in edges.iter().enumerate() {
            if u >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 0,
                    vertex: u,
                });
            }
            if v >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 1,
                    vertex: v,
                });
            }
        }

        let mut steps = Vec::with_capacity(edges.len());
        let mut merges = Vec::new();
        for &(u, v) in edges {
            let edge_index = self.edges.len();
            let before_graph = self.graph.clone();
            let before = (before_graph.component_of[u], before_graph.component_of[v]);
            let duplicate = self.pairs.contains(&(u, v));
            self.edges.push((u, v));
            self.pairs.insert((u, v));
            self.graph = build_graph(self.n, &self.edges);
            let after = (self.graph.component_of[u], self.graph.component_of[v]);

            let status = if duplicate {
                EdgeStatus::Duplicate
            } else if before.0 == before.1 {
                EdgeStatus::Internal
            } else if after.0 == after.1 {
                EdgeStatus::Merge
            } else {
                EdgeStatus::Link
            };

            if status == EdgeStatus::Merge {
                let members = self.graph.components[after.0].clone();
                let mut before_ids = HashSet::new();
                for &vertex in &members {
                    before_ids.insert(before_graph.component_of[vertex]);
                }
                let mut before_ids: Vec<usize> = before_ids.into_iter().collect();
                before_ids.sort_unstable();
                let event = Merge {
                    edge_index,
                    before: before_ids,
                    after: after.0,
                    members,
                };
                self.history.push(event.clone());
                merges.push(event);
            }
            steps.push(Step {
                edge_index,
                status,
                before,
                after,
            });
        }

        Ok(Batch {
            steps,
            merges,
            graph: self.graph.clone(),
        })
    }
}

// Iterative Tarjan SCC decomposition.
fn scc(n: usize, edges: &[(usize, usize)]) -> (Vec<Vec<usize>>, Vec<usize>) {
    let mut out = vec![Vec::new(); n];
    for &(u, v) in edges {
        out[u].push(v);
    }

    let mut index = vec![usize::MAX; n];
    let mut low = vec![0usize; n];
    let mut on_stack = vec![false; n];
    let mut tarjan_stack = Vec::new();
    let mut next_index = 0usize;
    let mut raw = Vec::<Vec<usize>>::new();

    for root in 0..n {
        if index[root] != usize::MAX {
            continue;
        }
        index[root] = next_index;
        low[root] = next_index;
        next_index += 1;
        tarjan_stack.push(root);
        on_stack[root] = true;
        let mut dfs = vec![(root, 0usize)];

        while let Some((v, edge_pos)) = dfs.last_mut() {
            if *edge_pos < out[*v].len() {
                let w = out[*v][*edge_pos];
                *edge_pos += 1;
                if index[w] == usize::MAX {
                    index[w] = next_index;
                    low[w] = next_index;
                    next_index += 1;
                    tarjan_stack.push(w);
                    on_stack[w] = true;
                    dfs.push((w, 0));
                } else if on_stack[w] {
                    low[*v] = low[*v].min(index[w]);
                }
            } else {
                let finished = *v;
                dfs.pop();
                if let Some((parent, _)) = dfs.last() {
                    low[*parent] = low[*parent].min(low[finished]);
                }
                if low[finished] == index[finished] {
                    let mut members = Vec::new();
                    loop {
                        let w = tarjan_stack.pop().expect("Tarjan stack invariant");
                        on_stack[w] = false;
                        members.push(w);
                        if w == finished {
                            break;
                        }
                    }
                    members.sort_unstable();
                    raw.push(members);
                }
            }
        }
    }

    let mut raw_of = vec![usize::MAX; n];
    for (id, members) in raw.iter().enumerate() {
        for &v in members {
            raw_of[v] = id;
        }
    }
    let mut order: Vec<usize> = (0..raw.len()).collect();
    order.sort_by_key(|&id| (Reverse(raw[id].len()), raw[id][0]));
    let mut canonical = vec![0; raw.len()];
    let mut components = Vec::with_capacity(raw.len());
    for (new_id, old_id) in order.into_iter().enumerate() {
        canonical[old_id] = new_id;
        components.push(raw[old_id].clone());
    }
    let component_of = raw_of.into_iter().map(|id| canonical[id]).collect();
    (components, component_of)
}

fn build_graph(n: usize, edges: &[(usize, usize)]) -> Graph {
    let (components, component_of) = scc(n, edges);
    let k = components.len();
    let mut cyclic = vec![false; k];
    let mut internal_edges = vec![0usize; k];
    let mut arc_data: HashMap<(usize, usize), (usize, usize)> = HashMap::new();
    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let e = arc_data.entry((a, b)).or_insert((0, i));
            e.0 += 1;
        }
    }
    for (i, members) in components.iter().enumerate() {
        if members.len() > 1 {
            cyclic[i] = true;
        }
    }

    let mut arcs: Vec<Arc> = arc_data
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by_key(|a| (a.from, Reverse(a.to)));

    let mut outgoing = vec![Vec::new(); k];
    let mut indegree = vec![0usize; k];
    for a in &arcs {
        outgoing[a.from].push(a.to);
        indegree[a.to] += 1;
    }
    let mut ready = BinaryHeap::new();
    for id in 0..k {
        if indegree[id] == 0 {
            ready.push((internal_edges[id], components[id][0], Reverse(id)));
        }
    }
    let mut topo = Vec::with_capacity(k);
    let mut depth = vec![0usize; k];
    while let Some((_, _, Reverse(id))) = ready.pop() {
        topo.push(id);
        for &to in &outgoing[id] {
            depth[to] = depth[to].max(depth[id] + 1);
            indegree[to] -= 1;
            if indegree[to] == 0 {
                ready.push((internal_edges[to], components[to][0], Reverse(to)));
            }
        }
    }
    Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    }
}

fn shortest_route(g: &Graph, start: usize, target: usize) -> Option<Vec<usize>> {
    let k = g.components.len();
    let mut outgoing = vec![Vec::new(); k];
    let mut incoming = vec![Vec::new(); k];
    for a in &g.arcs {
        outgoing[a.from].push(a.to);
        incoming[a.to].push(a.from);
    }
    for list in &mut outgoing {
        list.sort_unstable();
        list.dedup();
    }

    let mut dist = vec![usize::MAX; k];
    let mut q = VecDeque::new();
    dist[target] = 0;
    q.push_back(target);
    while let Some(v) = q.pop_front() {
        for &p in &incoming[v] {
            if dist[p] == usize::MAX {
                dist[p] = dist[v] + 1;
                q.push_back(p);
            }
        }
    }
    if dist[start] == usize::MAX {
        return None;
    }
    let mut route = vec![start];
    let mut cur = start;
    while cur != target {
        let next = outgoing[cur]
            .iter()
            .copied()
            .filter(|&v| dist[v] != usize::MAX && dist[v] + 1 == dist[cur])
            .min()
            .unwrap();
        route.push(next);
        cur = next;
    }
    Some(route)
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        let n = self.component_of.len();
        if from >= n {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }
        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            return Ok(if self.cyclic[a] {
                Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                }
            } else {
                Query {
                    kind: Relation::Trivial,
                    route: vec![a],
                }
            });
        }
        if let Some(route) = shortest_route(self, a, b) {
            Ok(Query {
                kind: Relation::Forward,
                route,
            })
        } else if let Some(mut route) = shortest_route(self, b, a) {
            route.reverse();
            Ok(Query {
                kind: Relation::Backward,
                route,
            })
        } else {
            Ok(Query {
                kind: Relation::Separate,
                route: Vec::new(),
            })
        }
    }
}

        })
    }

    pub fn snapshot(&self) -> Graph {
        self.graph.clone()
    }

    pub fn history(&self) -> Vec<Merge> {
        self.history.clone()
    }

    pub fn insert(&mut self, edges: &[(usize, usize)]) -> Result<Batch, GraphError> {
        if self.edges.len() + edges.len() > MAX_EDGES {
            return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
        }
        for (i, &(u, v)) in edges.iter().enumerate() {
            if u >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 0,
                    vertex: u,
                });
            }
            if v >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 1,
                    vertex: v,
                });
            }
        }

        let mut steps = Vec::with_capacity(edges.len());
        let mut merges = Vec::new();
        for &(u, v) in edges {
            let edge_index = self.edges.len();
            let before_graph = self.graph.clone();
            let before = (before_graph.component_of[u], before_graph.component_of[v]);
            let duplicate = self.pairs.contains(&(u, v));
            self.edges.push((u, v));
            self.pairs.insert((u, v));
            self.graph = build_graph(self.n, &self.edges);
            let after = (self.graph.component_of[u], self.graph.component_of[v]);
            let status = if duplicate {
                EdgeStatus::Duplicate
            } else if before.0 == before.1 {
                EdgeStatus::Internal
            } else if after.0 == after.1 {
                EdgeStatus::Merge
            } else {
                EdgeStatus::Link
            };
            if status == EdgeStatus::Merge {
                let members = self.graph.components[after.0].clone();
                let mut before_ids: Vec<usize> = members
                    .iter()
                    .map(|&v| before_graph.component_of[v])
                    .collect();
                before_ids.sort_unstable();
                before_ids.dedup();
                let event = Merge {
                    edge_index,
                    before: before_ids,
                    after: after.0,
                    members,
                };
                self.history.push(event.clone());
                merges.push(event);
            }
            steps.push(Step {
                edge_index,
                status,
                before,
                after,
            });
        }

        Ok(Batch {
            steps,
            merges,
            graph: self.graph.clone(),
        })
    }
}
