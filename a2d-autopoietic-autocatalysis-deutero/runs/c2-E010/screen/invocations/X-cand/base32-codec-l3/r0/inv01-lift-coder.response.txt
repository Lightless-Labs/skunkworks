- Checked mode reserves the last non-hyphen byte as the check symbol, even when it is the only symbol; I follow that rule.
- Validation follows the specified pass order, so lexical errors take precedence over length, trailing bits, and checksum errors.
- Length and missing-check offsets use the original UTF-8 byte length; symbol offsets count original bytes, including skipped hyphens.
- Partial five-bit groups are zero-filled on encode, and their unused low bits must be zero on decode.
- `=` is a check symbol only in the reserved check position; elsewhere it is invalid data.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    MissingCheck { offset: usize },
    InvalidByte { offset: usize, byte: u8 },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    CheckMismatch { offset: usize, expected: u8, found: u8 },
}

const DATA_ALPHABET: &[u8; 32] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ";
const CHECK_ALPHABET: &[u8; 37] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ*~$=U";

pub fn encode(input: &[u8], checked: bool) -> String {
    let mut output = String::new();
    let mut accumulator: u32 = 0;
    let mut bit_count = 0u32;
    let mut remainder = 0u8;

    for &byte in input {
        accumulator = (accumulator << 8) | u32::from(byte);
        bit_count += 8;

        while bit_count >= 5 {
            bit_count -= 5;
            let value = ((accumulator >> bit_count) & 31) as u8;
            output.push(DATA_ALPHABET[value as usize] as char);
            remainder = ((u16::from(remainder) * 32 + u16::from(value)) % 37) as u8;
        }
    }

    if bit_count != 0 {
        let value = ((accumulator << (5 - bit_count)) & 31) as u8;
        output.push(DATA_ALPHABET[value as usize] as char);
        remainder = ((u16::from(remainder) * 32 + u16::from(value)) % 37) as u8;
    }

    if checked {
        output.push(CHECK_ALPHABET[remainder as usize] as char);
    }

    output
}

fn data_value(byte: u8) -> Option<u8> {
    match byte {
        b'0' | b'O' | b'o' => Some(0),
        b'1' | b'I' | b'i' | b'L' | b'l' => Some(1),
        b'2'..=b'9' => Some(byte - b'0'),
        b'A' | b'a' => Some(10),
        b'B' | b'b' => Some(11),
        b'C' | b'c' => Some(12),
        b'D' | b'd' => Some(13),
        b'E' | b'e' => Some(14),
        b'F' | b'f' => Some(15),
        b'G' | b'g' => Some(16),
        b'H' | b'h' => Some(17),
        b'J' | b'j' => Some(18),
        b'K' | b'k' => Some(19),
        b'M' | b'm' => Some(20),
        b'N' | b'n' => Some(21),
        b'P' | b'p' => Some(22),
        b'Q' | b'q' => Some(23),
        b'R' | b'r' => Some(24),
        b'S' | b's' => Some(25),
        b'T' | b't' => Some(26),
        b'V' | b'v' => Some(27),
        b'W' | b'w' => Some(28),
        b'X' | b'x' => Some(29),
        b'Y' | b'y' => Some(30),
        b'Z' | b'z' => Some(31),
        _ => None,
    }
}

fn check_value(byte: u8) -> Option<u8> {
    match byte {
        b'*' => Some(32),
        b'~' => Some(33),
        b'$' => Some(34),
        b'=' => Some(35),
        b'U' | b'u' => Some(36),
        _ => data_value(byte),
    }
}

pub fn decode(input: &str, checked: bool) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let significant: Vec<(usize, u8)> = bytes
        .iter()
        .copied()
        .enumerate()
        .filter(|(_, byte)| *byte != b'-')
        .collect();

    if checked && significant.is_empty() {
        return Err(DecodeError::MissingCheck { offset: bytes.len() });
    }

    let data_end = if checked {
        significant.len() - 1
    } else {
        significant.len()
    };

    let mut values = Vec::with_capacity(data_end);
    for &(offset, byte) in &significant[..data_end] {
        match data_value(byte) {
            Some(value) => values.push((offset, value)),
            None => return Err(DecodeError::InvalidByte { offset, byte }),
        }
    }

    let reserved_check = if checked {
        let (offset, byte) = significant[data_end];
        match check_value(byte) {
            Some(value) => Some((offset, value)),
            None => return Err(DecodeError::InvalidByte { offset, byte }),
        }
    } else {
        None
    };

    let residue = values.len() % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength { offset: bytes.len() });
    }

    let unused_bits = match residue {
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => 0,
    };
    if unused_bits != 0 {
        let (offset, value) = values[values.len() - 1];
        if value & ((1u8 << unused_bits) - 1) != 0 {
            return Err(DecodeError::TrailingBits { offset });
        }
    }

    let mut output = Vec::with_capacity(values.len() * 5 / 8);
    let mut accumulator: u32 = 0;
    let mut bit_count = 0u32;
    let mut remainder = 0u8;

    for &(_, value) in &values {
        remainder = ((u16::from(remainder) * 32 + u16::from(value)) % 37) as u8;
        accumulator = (accumulator << 5) | u32::from(value);
        bit_count += 5;
        if bit_count >= 8 {
            bit_count -= 8;
            output.push((accumulator >> bit_count) as u8);
        }
    }

    if let Some((offset, found)) = reserved_check {
        if found != remainder {
            return Err(DecodeError::CheckMismatch {
                offset,
                expected: remainder,
                found,
            });
        }
    }

    Ok(output)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    MissingCheck { offset: usize },
    InvalidByte { offset: usize, byte: u8 },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    CheckMismatch { offset: usize, expected: u8, found: u8 },
}

const DATA_ALPHABET: &[u8; 32] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ";
const CHECK_ALPHABET: &[u8; 37] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ*~$=U";

pub fn encode(input: &[u8], checked: bool) -> String {
    let mut output = String::with_capacity((input.len() * 8 + 4) / 5 + usize::from(checked));
    let mut remainder = 0u8;
    let total_bits = input.len() * 8;
    let mut bit_position = 0usize;

    while bit_position < total_bits {
        let mut value = 0u8;
        for shift in (0..5).rev() {
            value <<= 1;
            if bit_position < total_bits {
                let byte = input[bit_position / 8];
                let bit = (byte >> (7 - bit_position % 8)) & 1;
                value |= bit;
                bit_position += 1;
            }
            let _ = shift;
        }
        output.push(DATA_ALPHABET[value as usize] as char);
        remainder = ((u16::from(remainder) * 32 + u16::from(value)) % 37) as u8;
    }

    if checked {
        output.push(CHECK_ALPHABET[remainder as usize] as char);
    }
    output
}

fn value_for_data(byte: u8) -> Option<u8> {
    let upper = byte.to_ascii_uppercase();
    match upper {
        b'0' | b'O' => Some(0),
        b'1' | b'I' | b'L' => Some(1),
        b'2'..=b'9' => Some(upper - b'0'),
        b'A'..=b'H' => Some(upper - b'A' + 10),
        b'J'..=b'K' => Some(upper - b'J' + 18),
        b'M'..=b'N' => Some(upper - b'M' + 20),
        b'P'..=b'T' => Some(upper - b'P' + 22),
        b'V'..=b'Z' => Some(upper - b'V' + 27),
        _ => None,
    }
}

fn value_for_check(byte: u8) -> Option<u8> {
    match byte {
        b'*' => Some(32),
        b'~' => Some(33),
        b'$' => Some(34),
        b'=' => Some(35),
        b'U' | b'u' => Some(36),
        _ => value_for_data(byte),
    }
}

pub fn decode(input: &str, checked: bool) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let mut significant = Vec::new();
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'-' {
            significant.push((offset, byte));
        }
    }

    if checked && significant.is_empty() {
        return Err(DecodeError::MissingCheck { offset: bytes.len() });
    }

    let data_count = significant.len() - usize::from(checked);
    let mut values = Vec::with_capacity(data_count);
    for &(offset, byte) in significant.iter().take(data_count) {
        if let Some(value) = value_for_data(byte) {
            values.push((offset, value));
        } else {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let check = if checked {
        let (offset, byte) = significant[data_count];
        match value_for_check(byte) {
            Some(value) => Some((offset, value)),
            None => return Err(DecodeError::InvalidByte { offset, byte }),
        }
    } else {
        None
    };

    let residue = data_count % 8;
    if ![0, 2, 4, 5, 7].contains(&residue) {
        return Err(DecodeError::InvalidLength { offset: bytes.len() });
    }

    let zero_bits = match residue {
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => 0,
    };
    if zero_bits != 0 {
        let (offset, last) = values[values.len() - 1];
        if last & ((1 << zero_bits) - 1) != 0 {
            return Err(DecodeError::TrailingBits { offset });
        }
    }

    let mut output = Vec::with_capacity(data_count * 5 / 8);
    let mut accumulator = 0u8;
    let mut bits = 0u8;
    let mut remainder = 0u8;
    for &(_, value) in &values {
        remainder = ((u16::from(remainder) * 32 + u16::from(value)) % 37) as u8;
        for shift in (0..5).rev() {
            accumulator = (accumulator << 1) | ((value >> shift) & 1);
            bits += 1;
            if bits == 8 {
                output.push(accumulator);
                accumulator = 0;
                bits = 0;
            }
        }
    }

    if let Some((offset, found)) = check {
        if found != remainder {
            return Err(DecodeError::CheckMismatch {
                offset,
                expected: remainder,
                found,
            });
        }
    }

    Ok(output)
}
```
