pub enum DecodeError {
    MissingCheck { offset: usize },
    InvalidByte { offset: usize, byte: u8 },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    CheckMismatch { offset: usize, expected: u8, found: u8 },
}

const DATA_ALPHABET: &[u8; 32] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ";
const CHECK_ALPHABET: &[u8; 37] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ*~$=U";

pub fn encode(input: &[u8], checked: bool) -> String {
    let mut output = String::new();
    let mut accumulator: u32 = 0;
    let mut bit_count = 0u32;
    let mut remainder = 0usize;

    for &byte in input {
        accumulator = (accumulator << 8) | byte as u32;
        bit_count += 8;

        while bit_count >= 5 {
            bit_count -= 5;
            let value = ((accumulator >> bit_count) & 31) as u8;
            output.push(DATA_ALPHABET[value as usize] as char);
            remainder = (remainder * 32 + value as usize) % 37;
        }
    }

    if bit_count > 0 {
        let value = ((accumulator << (5 - bit_count)) & 31) as u8;
        output.push(DATA_ALPHABET[value as usize] as char);
        remainder = (remainder * 32 + value as usize) % 37;
    }

    if checked {
        output.push(CHECK_ALPHABET[remainder] as char);
    }
    output
}

pub fn decode(input: &str, checked: bool) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let significant: Vec<(usize, u8)> = bytes
        .iter()
        .copied()
        .enumerate()
        .filter(|(_, byte)| *byte != b'-')
        .collect();

    if checked && significant.is_empty() {
        return Err(DecodeError::MissingCheck { offset: input.len() });
    }

    let data_end = if checked {
        significant.len() - 1
    } else {
        significant.len()
    };

    let mut values = Vec::with_capacity(data_end);
    for &(offset, byte) in &significant[..data_end] {
        match data_value(byte) {
            Some(value) => values.push((offset, value)),
            None => return Err(DecodeError::InvalidByte { offset, byte }),
        }
    }

    let check = if checked {
        let (offset, byte) = significant[significant.len() - 1];
        match check_value(byte) {
            Some(value) => Some((offset, value)),
            None => return Err(DecodeError::InvalidByte { offset, byte }),
        }
    } else {
        None
    };

    let residue = values.len() % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength { offset: input.len() });
    }

    let (unused_bits, mask) = match residue {
        2 => (2, 0b11),
        4 => (4, 0b1111),
        5 => (1, 0b1),
        7 => (3, 0b111),
        _ => (0, 0),
    };
    if unused_bits != 0 && values.last().unwrap().1 & mask != 0 {
        return Err(DecodeError::TrailingBits {
            offset: values.last().unwrap().0,
        });
    }

    let mut output = Vec::with_capacity(values.len() * 5 / 8);
    let mut accumulator: u32 = 0;
    let mut bit_count = 0u32;
    let mut remainder = 0usize;

    for &(_, value) in &values {
        accumulator = (accumulator << 5) | value as u32;
        bit_count += 5;
        remainder = (remainder * 32 + value as usize) % 37;

        if bit_count >= 8 {
            bit_count -= 8;
            output.push(((accumulator >> bit_count) & 0xff) as u8);
        }
    }

    if let Some((offset, found)) = check {
        if found as usize != remainder {
            return Err(DecodeError::CheckMismatch {
                offset,
                expected: remainder as u8,
                found,
            });
        }
    }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    MissingCheck { offset: usize },
    InvalidByte { offset: usize, byte: u8 },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    CheckMismatch { offset: usize, expected: u8, found: u8 },
}

const DATA_ALPHABET: &[u8; 32] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ";
const CHECK_ALPHABET: &[u8; 37] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ*~$=U";

pub fn encode(input: &[u8], checked: bool) -> String {
    let mut result = String::new();
    let mut remainder = 0u8;
    let mut value = 0u8;
    let mut filled = 0u8;

    for &byte in input {
        for shift in (0..8).rev() {
            value = (value << 1) | ((byte >> shift) & 1);
            filled += 1;
            if filled == 5 {
                result.push(DATA_ALPHABET[value as usize] as char);
                remainder = ((remainder as u16 * 32 + value as u16) % 37) as u8;
                value = 0;
                filled = 0;
            }
        }
    }

    if filled != 0 {
        value <<= 5 - filled;
        result.push(DATA_ALPHABET[value as usize] as char);
        remainder = ((remainder as u16 * 32 + value as u16) % 37) as u8;
    }

    if checked {
        result.push(CHECK_ALPHABET[remainder as usize] as char);
    }
    result
}

fn value_for_data(byte: u8) -> Option<u8> {
    let upper = byte.to_ascii_uppercase();
    match upper {
        b'0' | b'O' => Some(0),
        b'1' | b'I' | b'L' => Some(1),
        b'2'..=b'9' => Some(upper - b'0'),
        b'A'..=b'H' => Some(upper - b'A' + 10),
        b'J'..=b'K' => Some(upper - b'A' + 10),
        b'M'..=b'N' => Some(upper - b'A' + 10),
        b'P'..=b'T' => Some(upper - b'A' + 10),
        b'V'..=b'Z' => Some(upper - b'A' + 10),
        _ => None,
    }
}

fn value_for_check(byte: u8) -> Option<u8> {
    match byte {
        b'*' => Some(32),
        b'~' => Some(33),
        b'$' => Some(34),
        b'=' => Some(35),
        b'U' | b'u' => Some(36),
        _ => value_for_data(byte),
    }
}

pub fn decode(input: &str, checked: bool) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let mut positions = Vec::with_capacity(bytes.len());
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'-' {
            positions.push((offset, byte));
        }
    }

    if checked && positions.is_empty() {
        return Err(DecodeError::MissingCheck { offset: bytes.len() });
    }

    let data_count = positions.len() - usize::from(checked);
    let mut normalized = Vec::with_capacity(data_count);

    for &(offset, byte) in positions.iter().take(data_count) {
        if let Some(value) = value_for_data(byte) {
            normalized.push((offset, value));
        } else {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let check = if checked {
        let (offset, byte) = positions[data_count];
        match value_for_check(byte) {
            Some(value) => Some((offset, value)),
            None => return Err(DecodeError::InvalidByte { offset, byte }),
        }
    } else {
        None
    };

    let residue = normalized.len() % 8;
    if residue != 0 && residue != 2 && residue != 4 && residue != 5 && residue != 7 {
        return Err(DecodeError::InvalidLength { offset: bytes.len() });
    }

    let unused = match residue {
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => 0,
    };
    if unused != 0 {
        let (offset, value) = normalized[normalized.len() - 1];
        let mask = (1u8 << unused) - 1;
        if value & mask != 0 {
            return Err(DecodeError::TrailingBits { offset });
        }
    }

    let mut output = Vec::with_capacity(normalized.len() * 5 / 8);
    let mut byte_acc = 0u8;
    let mut bits_in_byte = 0u8;
    let mut remainder = 0u8;

    for &(_, value) in &normalized {
        remainder = ((remainder as u16 * 32 + value as u16) % 37) as u8;
        for shift in (0..5).rev() {
            byte_acc = (byte_acc << 1) | ((value >> shift) & 1);
            bits_in_byte += 1;
            if bits_in_byte == 8 {
                output.push(byte_acc);
                byte_acc = 0;
                bits_in_byte = 0;
            }
        }
    }

    if let Some((offset, found)) = check {
        if found != remainder {
            return Err(DecodeError::CheckMismatch {
                offset,
                expected: remainder,
                found,
            });
        }
    }

    Ok(output)
}
