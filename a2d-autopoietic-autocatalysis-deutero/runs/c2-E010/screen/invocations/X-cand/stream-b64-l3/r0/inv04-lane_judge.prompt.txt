    }

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut out = encoder.push(input)?;
    out.push_str(&encoder.finish()?);
    Ok(out)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

pub struct Encoder {
    options: Options,
    tail: [u8; 2],
    tail_len: usize,
    column: usize,
    closed: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: [0; 2],
            tail_len: 0,
            column: 0,
            closed: false,
        })
    }

    fn append_symbols(&mut self, out: &mut String, bytes: &[u8]) {
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            if self.column == width {
                match ending {
                    LineEnding::Lf => out.push('\n'),
                    LineEnding::CrLf => out.push_str("\r\n"),
                }
                self.column = 0;
            }
            self.column += bytes.len();
        }
        for &b in bytes {
            out.push(char::from(b));
        }
    }

    fn triple(&mut self, a: u8, b: u8, c: u8, out: &mut String) {
        let vals = [
            a >> 2,
            ((a & 3) << 4) | (b >> 4),
            ((b & 15) << 2) | (c >> 6),
            c & 63,
        ];
        let bytes = vals.map(|v| symbol(v, self.options.alphabet));
        self.append_symbols(out, &bytes);
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut out = String::new();
        let mut i = 0;
        if self.tail_len > 0 {
            while self.tail_len < 2 && i < input.len() {
                self.tail[self.tail_len] = input[i];
                self.tail_len += 1;
                i += 1;
            }
            if self.tail_len == 2 && i < input.len() {
                self.triple(self.tail[0], self.tail[1], input[i], &mut out);
                self.tail_len = 0;
                i += 1;
            } else if self.tail_len < 2 || i == input.len() {
                if self.tail_len < 2 {
                    return Ok(out);
                }
            }
        }
        while i + 3 <= input.len() {
            self.triple(input[i], input[i + 1], input[i + 2], &mut out);
            i += 3;
        }
        let rem = input.len() - i;
        self.tail_len = rem;
        if rem > 0 {
            self.tail[0] = input[i];
        }
        if rem > 1 {
            self.tail[1] = input[i + 1];
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;
        let mut out = String::new();
        if self.tail_len != 0 {
            let a = self.tail[0];
            let b = if self.tail_len == 2 { self.tail[1] } else { 0 };
            let mut bytes = vec![
                symbol(a >> 2, self.options.alphabet),
                symbol(((a & 3) << 4) | (b >> 4), self.options.alphabet),
            ];
            if self.tail_len == 2 {
                bytes.push(symbol((b & 15) << 2, self.options.alphabet));
            }
            if self.options.padding != Padding::Forbidden {
                while bytes.len() < 4 {
                    bytes.push(b'=');
                }
            }
            self.append_symbols(&mut out, &bytes);
        }
        Ok(out)
    }
}

#[derive(Clone, Copy)]
enum FrameState {
    Normal,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    group: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    pending_second_eq: Option<usize>,
    terminal: bool,
    physical: usize,
    wrapped_col: usize,
    frame: FrameState,
    closed: bool,
    sticky: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: [0; 4],
            offsets: [0; 4],
            len: 0,
            pending_second_eq: None,
            terminal: false,
            physical: 0,
            wrapped_col: 0,
            frame: FrameState::Normal,
            closed: false,
            sticky: None,
        })
    }

    fn err(&mut self, e: Error) -> Error {
        self.sticky = Some(e.clone());
        e
    }

    fn decode_group(&mut self, out: &mut Vec<u8>) {
        let a = self.group[0];
        let b = self.group[1];
        out.push((a << 2) | (b >> 4));
        if self.group[2] != 64 {
            let c = self.group[2];
            out.push((b << 4) | (c >> 2));
            if self.group[3] != 64 {
                out.push((c << 6) | self.group[3]);
            }
        }
        if self.group[2] == 64 || self.group[3] == 64 {
            self.terminal = true;
        }
        self.len = 0;
        self.pending_second_eq = None;
    }

    fn symbol_byte(&mut self, b: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.terminal {
            return Err(Error::BadPadding { offset });
        }
        if self.pending_second_eq.is_some() && b != b'=' {
            return Err(Error::BadPadding { offset });
        }
        if b == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            match self.len {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    self.pending_second_eq = Some(offset);
                    self.group[2] = 64;
                    self.offsets[2] = offset;
                    self.len = 3;
                    return Ok(());
                }
                3 => {
                    if self.group[2] == 64 {
                        self.group[3] = 64;
                        self.offsets[3] = offset;
                        self.len = 4;
                        self.decode_group(out);
                    } else {
                        self.group[3] = 64;
                        self.offsets[3] = offset;
                        self.len = 4;
                        self.decode_group(out);
                    }
                    return Ok(());
                }
                _ => return Err(Error::BadPadding { offset }),
            }
        }
        let val = match digit(b, self.options.alphabet) {
            Some(v) => v,
            None => return Err(Error::InvalidByte { offset, byte: b }),
        };
        if self.len == 3 && self.group[2] == 64 {
            return Err(Error::BadPadding { offset });
        }
        self.group[self.len] = val;
        self.offsets[self.len] = offset;
        self.len += 1;
        if self.len == 4 {
            self.decode_group(out);
        }
        Ok(())
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(e) = &self.sticky {
            return Err(e.clone());
        }
        if self.closed {
            let e = Error::AlreadyFinished;
            self.sticky = Some(e.clone());
            return Err(e);
        }
        let mut out = Vec::new();
        for &b in input {
            let off = self.physical;
            self.physical = self.physical.saturating_add(1);
            match self.options.whitespace {
                Whitespace::IgnoreAscii if matches!(b, 9 | 10 | 11 | 12 | 13 | 32) => continue,
                Whitespace::Wrapped { width, ending } => {
                    match self.frame {
                        FrameState::NeedLf { cr_offset } => {
                            if b != b'\n' {
                                return Err(self.err(Error::BadWrap { offset: off }));
                            }
                            self.frame = FrameState::NeedSymbol { separator_offset: cr_offset };
                            continue;
                        }
                        FrameState::NeedSymbol { separator_offset } => {
                            self.frame = FrameState::Normal;
                            if b == b'\n' || b == b'\r' {
                                return Err(self.err(Error::BadWrap { offset: off }));
                            }
                            self.wrapped_col = 0;
                            let _ = separator_offset;
                        }
                        FrameState::Normal => {}
                    }
                    if self.wrapped_col == width {
                        match ending {
                            LineEnding::Lf if b == b'\n' => {
                                self.frame = FrameState::NeedSymbol { separator_offset: off };
                                continue;
                            }
                            LineEnding::CrLf if b == b'\r' => {
                                self.frame = FrameState::NeedLf { cr_offset: off };
                                continue;
                            }
                            _ => return Err(self.err(Error::BadWrap { offset: off })),
                        }
                    }
                    if b == b'\n' || b == b'\r' {
                        return Err(self.err(Error::BadWrap { offset: off }));
                    }
                    self.wrapped_col += 1;
                }
                _ => {}
            }
            if let Err(e) = self.symbol_byte(b, off, &mut out) {
                return Err(self.err(e));
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(e) = &self.sticky {
            return Err(e.clone());
        }
        if self.closed {
            let e = Error::AlreadyFinished;
            self.sticky = Some(e.clone());
            return Err(e);
        }
        if let FrameState::NeedLf { cr_offset } = self.frame {
            return Err(self.err(Error::BadWrap { offset: cr_offset }));
        }
        if let FrameState::NeedSymbol { separator_offset } = self.frame {
            return Err(self.err(Error::BadWrap { offset: separator_offset }));
        }
        let mut out = Vec::new();
        if self.pending_second_eq.is_some() {
            return Err(self.err(Error::BadPadding {
                offset: self.pending_second_eq.unwrap(),
            }));
        }
        if self.len != 0 {
            match self.options.padding {
                Padding::Required => {
                    return Err(self.err(Error::Truncated {
                        offset: self.offsets[0],
                    }));
                }
                Padding::Forbidden | Padding::Optional => match self.len {
                    1 => {
                        return Err(self.err(Error::Truncated {
                            offset: self.offsets[0],
                        }));
                    }
                    2 => {
                        out.push((self.group[0] << 2) | (self.group[1] >> 4));
                    }
                    3 => {
                        out.push((self.group[0] << 2) | (self.group[1] >> 4));
                        out.push((self.group[1] << 4) | (self.group[2] >> 2));
                    }
                    _ => {}
                },
            }
        }
        self.closed = true;
        Ok(out)
    }
}
