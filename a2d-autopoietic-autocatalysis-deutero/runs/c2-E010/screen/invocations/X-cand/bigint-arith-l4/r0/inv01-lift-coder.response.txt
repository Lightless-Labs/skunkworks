- Radix validation takes precedence over every input or formatting error; I validate it first.
- Invalid bytes anywhere take precedence over malformed underscores; after that, the leftmost malformed underscore takes precedence over leading-zero and digit-limit errors.
- Error offsets are UTF-8 byte offsets in the original string, including a sign; digit limits exclude signs and underscores.
- Zero is canonical and unsigned. Formatting emits at least one digit, with padding applied before right-counted grouping.
- Division truncates toward zero, and a nonzero remainder has the dividend’s sign.
- The exponent limit is checked before shortcuts, including for bases 0, 1, and -1.

### Solution A

```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1,000,000,000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;

    fn zero() -> Self {
        Self { negative: false, limbs: Vec::new() }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn validate_radix(radix: u32) -> Result<(), Error> {
        if !(2..=36).contains(&radix) {
            Err(Error::InvalidRadix { radix })
        } else {
            Ok(())
        }
    }

    fn cmp_mag(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let sum = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64 + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        out
    }

    // Requires a >= b.
    fn sub_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let mut v = a[i] as i64 - b.get(i).copied().unwrap_or(0) as i64 - borrow;
            if v < 0 {
                v += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(v as u32);
        }
        while out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_small_mag(&mut self, m: u32) {
        if m == 0 || self.is_zero() {
            self.limbs.clear();
            self.negative = false;
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let n = *limb as u64 * m as u64 + carry;
            *limb = (n % Self::BASE) as u32;
            carry = n / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_mag(&mut self, n: u32) {
        let mut carry = n as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[i] as u64 + carry;
            self.limbs[i] = (sum % Self::BASE) as u32;
            carry = sum / Self::BASE;
            i += 1;
        }
    }

    fn div_small_mag(&mut self, divisor: u32) -> u32 {
        let mut rem = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let n = rem * Self::BASE + *limb as u64;
            *limb = (n / divisor as u64) as u32;
            rem = n % divisor as u64;
        }
        self.normalize();
        rem as u32
    }

    fn div_rem_mag(a: &Self, b: &Self) -> (Self, Self) {
        debug_assert!(!b.is_zero());
        if a.cmp_mag(b) == Ordering::Less {
            let mut r = a.clone();
            r.negative = false;
            return (Self::zero(), r);
        }
        let mut q_be = Vec::with_capacity(a.limbs.len());
        let mut rem = Self::zero();
        for &limb in a.limbs.iter().rev() {
            rem.mul_small_mag(Self::BASE as u32);
            rem.add_small_mag(limb);
            let mut lo = 0u64;
            let mut hi = Self::BASE - 1;
            while lo < hi {
                let mid = (lo + hi + 1) / 2;
                let mut product = b.clone();
                product.negative = false;
                product.mul_small_mag(mid as u32);
                if product.cmp_mag(&rem) != Ordering::Greater {
                    lo = mid;
                } else {
                    hi = mid - 1;
                }
            }
            let digit = lo as u32;
            if digit != 0 {
                let mut product = b.clone();
                product.negative = false;
                product.mul_small_mag(digit);
                rem.limbs = Self::sub_mag(&rem.limbs, &product.limbs);
            }
            q_be.push(digit);
        }
        q_be.reverse();
        let mut q = Self { negative: false, limbs: q_be };
        q.normalize();
        rem.normalize();
        (q, rem)
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        Self::validate_radix(radix)?;
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let digit_value = |b: u8| -> Option<u32> {
            match b {
                b'0'..=b'9' => Some((b - b'0') as u32),
                b'a'..=b'z' => Some((b - b'a') as u32 + 10),
                b'A'..=b'Z' => Some((b - b'A') as u32 + 10),
                _ => None,
            }
        };

        for i in start..bytes.len() {
            if bytes[i] != b'_' && digit_value(bytes[i]).map_or(true, |v| v >= radix) {
                return Err(Error::InvalidDigit { index: i });
            }
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let left_ok = i > start && digit_value(bytes[i - 1]).is_some_and(|v| v < radix);
                let right_ok = i + 1 < bytes.len()
                    && digit_value(bytes[i + 1]).is_some_and(|v| v < radix);
                if !left_ok || !right_ok {
                    return Err(Error::Separator { index: i });
                }
            }
        }

        let digits: Vec<(usize, u32)> = (start..bytes.len())
            .filter_map(|i| digit_value(bytes[i]).map(|v| (i, v)))
            .collect();

        if digits.len() > 1 && digits[0].1 == 0 {
            return Err(Error::LeadingZero { index: digits[0].0 });
        }
        if digits.len() > max_digits {
            return Err(Error::TooManyDigits { limit: max_digits, index: digits[max_digits].0 });
        }

        let mut result = Self::zero();
        for &(_, digit) in &digits {
            result.mul_small_mag(radix);
            result.add_small_mag(digit);
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        Self::validate_radix(radix)?;
        let mut n = self.clone();
        n.negative = false;
        let mut digits = Vec::new();
        if n.is_zero() {
            digits.push(0u8);
        } else {
            while !n.is_zero() {
                digits.push(n.div_small_mag(radix) as u8);
            }
            digits.reverse();
        }
        if digits.len() < min_digits {
            let mut padded = vec![0u8; min_digits - digits.len()];
            padded.extend(digits);
            digits = padded;
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, d) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(if *d < 10 {
                (b'0' + *d) as char
            } else {
                (b'a' + (*d - 10)) as char
            });
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_mag(rhs),
            (true, true) => self.cmp_mag(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self { negative: self.negative, limbs: Self::add_mag(&self.limbs, &rhs.limbs) };
            out.normalize();
            out
        } else {
            match self.cmp_mag(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self { negative: self.negative, limbs: Self::sub_mag(&self.limbs, &rhs.limbs) };
                    out.normalize();
                    out
                }
                Ordering::Less => {
                    let mut out = Self { negative: rhs.negative, limbs: Self::sub_mag(&rhs.limbs, &self.limbs) };
                    out.normalize();
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u64; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let n = out[k] + a as u64 * b as u64 + carry;
                out[k] = n % Self::BASE;
                carry = n / Self::BASE;
            }
            let mut k = i + rhs.limbs.len();
            while carry != 0 {
                if k == out.len() {
                    out.push(0);
                }
                let n = out[k] + carry;
                out[k] = n % Self::BASE;
                carry = n / Self::BASE;
                k += 1;
            }
        }
        let mut result = Self {
            negative: self.negative ^ rhs.negative,
            limbs: out.into_iter().map(|v| v as u32).collect(),
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let mut a = self.clone();
        let mut b = rhs.clone();
        a.negative = false;
        b.negative = false;
        let (mut q, mut r) = Self::div_rem_mag(&a, &b);
        q.negative = (self.negative ^ rhs.negative) && !q.is_zero();
        r.negative = self.negative && !r.is_zero();
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: LIMIT });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while e != 0 {
            if e & 1 != 0 {
                result = result.mul(&base);
            }
            e >>= 1;
            if e != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.clone();
        let mut b = rhs.clone();
        a.negative = false;
        b.negative = false;
        while !b.is_zero() {
            let (_, r) = Self::div_rem_mag(&a, &b);
            a = b;
            b = r;
        }
        a.negative = false;
        a
    }
}
```

### Solution B

```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian binary magnitude.
    limbs: Vec<u32>,
}

impl Big {
    fn zero() -> Self { Self { negative: false, limbs: Vec::new() } }
    fn one() -> Self { Self { negative: false, limbs: vec![1] } }
    fn is_zero(&self) -> bool { self.limbs.is_empty() }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) { self.limbs.pop(); }
        if self.limbs.is_empty() { self.negative = false; }
    }

    fn radix_ok(radix: u32) -> Result<(), Error> {
        if (2..=36).contains(&radix) { Ok(()) } else { Err(Error::InvalidRadix { radix }) }
    }

    fn cmp_mag(a: &[u32], b: &[u32]) -> Ordering {
        match a.len().cmp(&b.len()) {
            Ordering::Equal => {
                for i in (0..a.len()).rev() {
                    match a[i].cmp(&b[i]) {
                        Ordering::Equal => {}
                        x => return x,
                    }
                }
                Ordering::Equal
            }
            x => x,
        }
    }

    fn add_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let x = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64 + carry;
            out.push((x & 0xffff_ffff) as u32);
            carry = x >> 32;
        }
        if carry != 0 { out.push(carry as u32); }
        out
    }

    // Requires a >= b.
    fn sub_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0u64;
        for i in 0..a.len() {
            let x = a[i] as u64;
            let y = b.get(i).copied().unwrap_or(0) as u64 + borrow;
            if x >= y {
                out.push((x - y) as u32);
                borrow = 0;
            } else {
                out.push(((1u64 << 32) + x - y) as u32);
                borrow = 1;
            }
        }
        while out.last() == Some(&0) { out.pop(); }
        out
    }

    fn mul_small(&mut self, m: u32) {
        if m == 0 || self.is_zero() { self.limbs.clear(); self.negative = false; return; }
        let mut carry = 0u64;
        for x in &mut self.limbs {
            let v = *x as u64 * m as u64 + carry;
            *x = v as u32;
            carry = v >> 32;
        }
        while carry != 0 { self.limbs.push(carry as u32); carry >>= 32; }
    }

    fn add_small(&mut self, n: u32) {
        let mut carry = n as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() { self.limbs.push(0); }
            let v = self.limbs[i] as u64 + carry;
            self.limbs[i] = v as u32;
            carry = v >> 32;
            i += 1;
        }
    }

    fn div_small(&mut self, divisor: u32) -> u32 {
        let mut rem = 0u64;
        for x in self.limbs.iter_mut().rev() {
            let v = (rem << 32) | *x as u64;
            *x = (v / divisor as u64) as u32;
            rem = v % divisor as u64;
        }
        self.normalize();
        rem as u32
    }

    fn div_rem_mag(a: &Self, b: &Self) -> (Self, Self) {
        debug_assert!(!b.is_zero());
        if Self::cmp_mag(&a.limbs, &b.limbs) == Ordering::Less {
            let mut r = a.clone(); r.negative = false; return (Self::zero(), r);
        }
        let mut quotient = vec![0u32; a.limbs.len()];
        let mut rem = Self::zero();
        for bit_index in (0..a.limbs.len() * 32).rev() {
            rem.mul_small(2);
            let bit = (a.limbs[bit_index / 32] >> (bit_index % 32)) & 1;
            rem.add_small(bit);
            if Self::cmp_mag(&rem.limbs, &b.limbs) != Ordering::Less {
                rem.limbs = Self::sub_mag(&rem.limbs, &b.limbs);
                quotient[bit_index / 32] |= 1u32 << (bit_index % 32);
            }
        }
        let mut q = Self { negative: false, limbs: quotient };
        q.normalize();
        rem.normalize();
        (q, rem)
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        Self::radix_ok(radix)?;
        if text.is_empty() { return Err(Error::Empty); }
        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() { return Err(Error::SignOnly); }

        fn digit(b: u8) -> Option<u32> {
            match b {
                b'0'..=b'9' => Some((b - b'0') as u32),
                b'a'..=b'z' => Some((b - b'a') as u32 + 10),
                b'A'..=b'Z' => Some((b - b'A') as u32 + 10),
                _ => None,
            }
        }

        for i in start..bytes.len() {
            if bytes[i] != b'_' && digit(bytes[i]).map_or(true, |d| d >= radix) {
                return Err(Error::InvalidDigit { index: i });
            }
        }
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let left = i > start && digit(bytes[i - 1]).is_some_and(|d| d < radix);
                let right = i + 1 < bytes.len() && digit(bytes[i + 1]).is_some_and(|d| d < radix);
                if !left || !right { return Err(Error::Separator { index: i }); }
            }
        }

        let digits: Vec<(usize, u32)> = (start..bytes.len())
            .filter_map(|i| digit(bytes[i]).map(|d| (i, d))).collect();
        if digits.len() > 1 && digits[0].1 == 0 {
            return Err(Error::LeadingZero { index: digits[0].0 });
        }
        if digits.len() > max_digits {
            return Err(Error::TooManyDigits { limit: max_digits, index: digits[max_digits].0 });
        }

        let mut n = Self::zero();
        for (_, d) in digits { n.mul_small(radix); n.add_small(d); }
        n.negative = negative && !n.is_zero();
        Ok(n)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        Self::radix_ok(radix)?;
        let mut n = self.clone(); n.negative = false;
        let mut digits = Vec::new();
        if n.is_zero() { digits.push(0u8); }
        while !n.is_zero() { digits.push(n.div_small(radix) as u8); }
        digits.reverse();
        while digits.len() < min_digits { digits.insert(0, 0); }

        let mut out = String::new();
        if self.negative && !self.is_zero() { out.push('-'); }
        for (i, d) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 { out.push('_'); }
            out.push(if *d < 10 { (b'0' + *d) as char } else { (b'a' + (*d - 10)) as char });
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => Self::cmp_mag(&self.limbs, &rhs.limbs),
            (true, true) => Self::cmp_mag(&self.limbs, &rhs.limbs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut n = Self { negative: self.negative, limbs: Self::add_mag(&self.limbs, &rhs.limbs) };
            n.normalize(); n
        } else {
            match Self::cmp_mag(&self.limbs, &rhs.limbs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut n = Self { negative: self.negative, limbs: Self::sub_mag(&self.limbs, &rhs.limbs) };
                    n.normalize(); n
                }
                Ordering::Less => {
                    let mut n = Self { negative: rhs.negative, limbs: Self::sub_mag(&rhs.limbs, &self.limbs) };
                    n.normalize(); n
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self { self.add(&rhs.neg()) }

    pub fn neg(&self) -> Self {
        let mut n = self.clone();
        if !n.is_zero() { n.negative = !n.negative; }
        n
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() { return Self::zero(); }
        let mut out = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let v = out[k] as u64 + a as u64 * b as u64 + carry;
                out[k] = v as u32;
                carry = v >> 32;
            }
            let mut k = i + rhs.limbs.len();
            while carry != 0 {
                if k == out.len() { out.push(0); }
                let v = out[k] as u64 + carry;
                out[k] = v as u32;
                carry = v >> 32;
                k += 1;
            }
        }
        let mut n = Self { negative: self.negative ^ rhs.negative, limbs: out };
        n.normalize(); n
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() { return Err(Error::DivisionByZero); }
        let mut a = self.clone(); a.negative = false;
        let mut b = rhs.clone(); b.negative = false;
        let (mut q, mut r) = Self::div_rem_mag(&a, &b);
        q.negative = !q.is_zero() && (self.negative ^ rhs.negative);
        r.negative = !r.is_zero() && self.negative;
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: LIMIT });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while e > 0 {
            if e & 1 == 1 { result = result.mul(&base); }
            e >>= 1;
            if e > 0 { base = base.mul(&base); }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.clone(); a.negative = false;
        let mut b = rhs.clone(); b.negative = false;
        while !b.is_zero() {
            let (_, r) = Self::div_rem_mag(&a, &b);
            a = b; b = r;
        }
        a.negative = false;
        a
    }
}
```
