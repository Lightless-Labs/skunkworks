- Invalid radix is checked first; then empty input, sign-only input, invalid digits, malformed separators, leading zero, and digit limit.
- Error offsets are UTF-8 byte offsets, including any sign; invalid digits anywhere take precedence over separator errors.
- Leading zero is checked before the digit limit, and separators do not count as digits.
- Zero is canonical and nonnegative; division truncates toward zero, with the remainder carrying the dividend’s sign.
- Formatting pads before grouping, and the sign sits outside the grouped magnitude.
- Exponents above 1024 fail before shortcuts; accepted powers use exact integer arithmetic.

### Solution A

```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base 1e9 limbs. Zero is represented by [0], nonnegative.
    limbs: Vec<u32>,
}

const BASE: u64 = 1_000_000_000;

impl Big {
    fn zero() -> Self {
        Self { negative: false, limbs: vec![0] }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.limbs.push(0);
        }
        if self.limbs.len() == 1 && self.limbs[0] == 0 {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn abs(&self) -> Self {
        let mut r = self.clone();
        r.negative = false;
        r
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        self.limbs.len().cmp(&rhs.limbs.len()).then_with(|| {
            for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                match a.cmp(b) {
                    Ordering::Equal => {}
                    o => return o,
                }
            }
            Ordering::Equal
        })
    }

    fn mul_small_add(&mut self, m: u32, a: u32) {
        let mut carry = a as u64;
        for limb in &mut self.limbs {
            let v = *limb as u64 * m as u64 + carry;
            *limb = (v % BASE) as u32;
            carry = v / BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % BASE) as u32);
            carry /= BASE;
        }
        self.normalize();
    }

    fn add_abs(a: &Self, b: &Self) -> Self {
        let n = a.limbs.len().max(b.limbs.len());
        let mut out = Vec::with_capacity(n + 1);
        let mut carry = 0u64;
        for i in 0..n {
            let x = *a.limbs.get(i).unwrap_or(&0) as u64;
            let y = *b.limbs.get(i).unwrap_or(&0) as u64;
            let s = x + y + carry;
            out.push((s % BASE) as u32);
            carry = s / BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        let mut r = Self { negative: false, limbs: out };
        r.normalize();
        r
    }

    // Requires |a| >= |b|.
    fn sub_abs(a: &Self, b: &Self) -> Self {
        let mut out = Vec::with_capacity(a.limbs.len());
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let x = a.limbs[i] as i64 - borrow;
            let y = *b.limbs.get(i).unwrap_or(&0) as i64;
            if x < y {
                out.push((x + BASE as i64 - y) as u32);
                borrow = 1;
            } else {
                out.push((x - y) as u32);
                borrow = 0;
            }
        }
        let mut r = Self { negative: false, limbs: out };
        r.normalize();
        r
    }

    fn mul_abs(a: &Self, b: &Self) -> Self {
        if a.is_zero() || b.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u64; a.limbs.len() + b.limbs.len() + 1];
        for (i, &x) in a.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &y) in b.limbs.iter().enumerate() {
                let k = i + j;
                let v = out[k] + x as u64 * y as u64 + carry;
                out[k] = v % BASE;
                carry = v / BASE;
            }
            let mut k = i + b.limbs.len();
            while carry != 0 {
                let v = out[k] + carry;
                out[k] = v % BASE;
                carry = v / BASE;
                k += 1;
            }
        }
        let mut r = Self {
            negative: false,
            limbs: out.into_iter().map(|x| x as u32).collect(),
        };
        r.normalize();
        r
    }

    fn div_rem_abs(a: &Self, b: &Self) -> (Self, Self) {
        debug_assert!(!b.is_zero());
        if a.cmp_abs(b) == Ordering::Less {
            return (Self::zero(), a.abs());
        }
        let mut quotient_be = Vec::with_capacity(a.limbs.len());
        let mut rem = Self::zero();
        for &digit in a.limbs.iter().rev() {
            rem.limbs.insert(0, digit);
            rem.normalize();
            let mut lo = 0u64;
            let mut hi = BASE - 1;
            while lo < hi {
                let mid = (lo + hi + 1) / 2;
                let prod = Self::mul_abs_small(b, mid as u32);
                if prod.cmp_abs(&rem) != Ordering::Greater {
                    lo = mid;
                } else {
                    hi = mid - 1;
                }
            }
            quotient_be.push(lo as u32);
            let prod = Self::mul_abs_small(b, lo as u32);
            rem = Self::sub_abs(&rem, &prod);
        }
        quotient_be.reverse();
        let mut q = Self { negative: false, limbs: quotient_be };
        q.normalize();
        rem.normalize();
        (q, rem)
    }

    fn mul_abs_small(a: &Self, m: u32) -> Self {
        if m == 0 || a.is_zero() { return Self::zero(); }
        let mut out = Vec::with_capacity(a.limbs.len() + 1);
        let mut carry = 0u64;
        for &x in &a.limbs {
            let v = x as u64 * m as u64 + carry;
            out.push((v % BASE) as u32);
            carry = v / BASE;
        }
        if carry != 0 { out.push(carry as u32); }
        let mut r = Self { negative: false, limbs: out };
        r.normalize();
        r
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }
        let mut start = 0;
        let mut negative = false;
        if bytes[0] == b'+' || bytes[0] == b'-' {
            negative = bytes[0] == b'-';
            start = 1;
            if start == bytes.len() {
                return Err(Error::SignOnly);
            }
        }

        let mut invalid = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' { continue; }
            let v = digit_value(bytes[i]);
            if v.map_or(true, |x| x >= radix) {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let left = i > start && bytes[i - 1] != b'_';
                let right = i + 1 < bytes.len() && bytes[i + 1] != b'_';
                if !left || !right {
                    return Err(Error::Separator { index: i });
                }
            }
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if digit_count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero { index: first_digit });
        }
        if digit_count > max_digits {
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen == max_digits.saturating_add(1) {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut value = Self::zero();
        for &b in &bytes[start..] {
            if b == b'_' { continue; }
            value.mul_small_add(radix, digit_value(b).unwrap());
        }
        value.negative = negative && !value.is_zero();
        Ok(value)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let mut n = self.abs();
        let mut digits = Vec::new();
        if n.is_zero() {
            digits.push(0u8);
        } else {
            while !n.is_zero() {
                let (q, r) = n.div_small(radix);
                digits.push(r as u8);
                n = q;
            }
            digits.reverse();
        }
        while digits.len() < min_digits {
            digits.insert(0, 0);
        }
        let mut out = String::new();
        if self.negative && !self.is_zero() { out.push('-'); }
        for (i, d) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(digit_char(*d));
        }
        Ok(out)
    }

    fn div_small(&self, d: u32) -> (Self, u32) {
        let mut out = vec![0u32; self.limbs.len()];
        let mut rem = 0u64;
        for i in (0..self.limbs.len()).rev() {
            let cur = rem * BASE + self.limbs[i] as u64;
            out[i] = (cur / d as u64) as u32;
            rem = cur % d as u64;
        }
        let mut q = Self { negative: false, limbs: out };
        q.normalize();
        (q, rem as u32)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => self.cmp_abs(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut r = Self::add_abs(self, rhs);
            r.negative = self.negative && !r.is_zero();
            r
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut r = Self::sub_abs(self, rhs);
                    r.negative = self.negative;
                    r
                }
                Ordering::Less => {
                    let mut r = Self::sub_abs(rhs, self);
                    r.negative = rhs.negative;
                    r
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut r = self.clone();
        if !r.is_zero() { r.negative = !r.negative; }
        r
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        let mut r = Self::mul_abs(self, rhs);
        r.negative = (self.negative != rhs.negative) && !r.is_zero();
        r
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() { return Err(Error::DivisionByZero); }
        let (mut q, mut r) = Self::div_rem_abs(self, rhs);
        q.negative = (self.negative != rhs.negative) && !q.is_zero();
        r.negative = self.negative && !r.is_zero();
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > 1024 {
            return Err(Error::ExponentTooLarge { exponent, limit: 1024 });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut out = Self::one();
        while e > 0 {
            if e & 1 == 1 { out = out.mul(&base); }
            e >>= 1;
            if e > 0 { base = base.mul(&base); }
        }
        Ok(out)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let r = a.div_rem(&b).expect("nonzero divisor").1;
            a = b;
            b = r.abs();
        }
        a.negative = false;
        a
    }
}

fn digit_value(b: u8) -> Option<u32> {
    match b {
        b'0'..=b'9' => Some((b - b'0') as u32),
        b'a'..=b'z' => Some((b - b'a' + 10) as u32),
        b'A'..=b'Z' => Some((b - b'A' + 10) as u32),
        _ => None,
    }
}

fn digit_char(d: u8) -> char {
    match d {
        0..=9 => (b'0' + d) as char,
        _ => (b'a' + d - 10) as char,
    }
}
```

### Solution B

```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base 2^32 words.
    words: Vec<u32>,
}

impl Big {
    fn zero() -> Self { Self { negative: false, words: vec![0] } }
    fn one() -> Self { Self { negative: false, words: vec![1] } }
    fn is_zero(&self) -> bool { self.words.len() == 1 && self.words[0] == 0 }

    fn normalize(&mut self) {
        while self.words.len() > 1 && self.words.last() == Some(&0) { self.words.pop(); }
        if self.words.is_empty() { self.words.push(0); }
        if self.is_zero() { self.negative = false; }
    }

    fn abs(&self) -> Self {
        let mut r = self.clone();
        r.negative = false;
        r
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        self.words.len().cmp(&rhs.words.len()).then_with(|| {
            for (a, b) in self.words.iter().rev().zip(rhs.words.iter().rev()) {
                match a.cmp(b) { Ordering::Equal => {}, o => return o }
            }
            Ordering::Equal
        })
    }

    fn add_abs(a: &Self, b: &Self) -> Self {
        let n = a.words.len().max(b.words.len());
        let mut words = Vec::with_capacity(n + 1);
        let mut carry = 0u64;
        for i in 0..n {
            let sum = *a.words.get(i).unwrap_or(&0) as u64
                + *b.words.get(i).unwrap_or(&0) as u64 + carry;
            words.push(sum as u32);
            carry = sum >> 32;
        }
        if carry != 0 { words.push(carry as u32); }
        let mut r = Self { negative: false, words };
        r.normalize();
        r
    }

    fn sub_abs(a: &Self, b: &Self) -> Self {
        let mut words = Vec::with_capacity(a.words.len());
        let mut borrow = 0u64;
        for i in 0..a.words.len() {
            let av = a.words[i] as u64;
            let bv = *b.words.get(i).unwrap_or(&0) as u64 + borrow;
            words.push(av.wrapping_sub(bv) as u32);
            borrow = (av < bv) as u64;
        }
        let mut r = Self { negative: false, words };
        r.normalize();
        r
    }

    fn mul_small(&mut self, m: u32) {
        let mut carry = 0u64;
        for w in &mut self.words {
            let v = *w as u64 * m as u64 + carry;
            *w = v as u32;
            carry = v >> 32;
        }
        if carry != 0 { self.words.push(carry as u32); }
        self.normalize();
    }

    fn add_small(&mut self, a: u32) {
        let mut carry = a as u64;
        let mut i = 0;
        while carry != 0 && i < self.words.len() {
            let v = self.words[i] as u64 + carry;
            self.words[i] = v as u32;
            carry = v >> 32;
            i += 1;
        }
        if carry != 0 { self.words.push(carry as u32); }
    }

    fn mul_small_add(&mut self, m: u32, a: u32) {
        self.mul_small(m);
        self.add_small(a);
    }

    fn shl1_add(&mut self, bit: u32) {
        let mut carry = bit as u64;
        for w in &mut self.words {
            let v = ((*w as u64) << 1) | carry;
            *w = v as u32;
            carry = v >> 32;
        }
        if carry != 0 { self.words.push(carry as u32); }
        self.normalize();
    }

    fn bit_len(&self) -> usize {
        if self.is_zero() { 0 } else {
            (self.words.len() - 1) * 32 + (32 - self.words.last().unwrap().leading_zeros() as usize)
        }
    }

    fn get_bit(&self, i: usize) -> u32 {
        self.words.get(i / 32).map_or(0, |w| (w >> (i % 32)) & 1)
    }

    fn set_bit(&mut self, i: usize) {
        while self.words.len() <= i / 32 { self.words.push(0); }
        self.words[i / 32] |= 1 << (i % 32);
    }

    fn div_abs(a: &Self, b: &Self) -> (Self, Self) {
        let mut q = Self::zero();
        let mut r = Self::zero();
        for i in (0..a.bit_len()).rev() {
            r.shl1_add(a.get_bit(i));
            if r.cmp_abs(b) != Ordering::Less {
                r = Self::sub_abs(&r, b);
                q.set_bit(i);
            }
        }
        q.normalize();
        r.normalize();
        (q, r)
    }

    fn div_small(&self, d: u32) -> (Self, u32) {
        let mut words = vec![0; self.words.len()];
        let mut rem = 0u64;
        for i in (0..self.words.len()).rev() {
            let cur = (rem << 32) | self.words[i] as u64;
            words[i] = (cur / d as u64) as u32;
            rem = cur % d as u64;
        }
        let mut q = Self { negative: false, words };
        q.normalize();
        (q, rem as u32)
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) { return Err(Error::InvalidRadix { radix }); }
        let bytes = text.as_bytes();
        if bytes.is_empty() { return Err(Error::Empty); }
        let mut start = 0;
        let mut negative = false;
        if bytes[0] == b'+' || bytes[0] == b'-' {
            negative = bytes[0] == b'-';
            start = 1;
            if start == bytes.len() { return Err(Error::SignOnly); }
        }

        let mut invalid = None;
        for i in start..bytes.len() {
            if bytes[i] != b'_' && digit_value(bytes[i]).map_or(true, |v| v >= radix) {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid { return Err(Error::InvalidDigit { index }); }

        for i in start..bytes.len() {
            if bytes[i] == b'_' &&
                !(i > start && i + 1 < bytes.len() && bytes[i - 1] != b'_' && bytes[i + 1] != b'_') {
                return Err(Error::Separator { index: i });
            }
        }

        let count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if count > 1 && bytes[first] == b'0' { return Err(Error::LeadingZero { index: first }); }
        if count > max_digits {
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen == max_digits.saturating_add(1) {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut n = Self::zero();
        for &b in &bytes[start..] {
            if b != b'_' { n.mul_small_add(radix, digit_value(b).unwrap()); }
        }
        n.negative = negative && !n.is_zero();
        Ok(n)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) { return Err(Error::InvalidRadix { radix }); }
        let mut n = self.abs();
        let mut ds = Vec::new();
        if n.is_zero() { ds.push(0); }
        while !n.is_zero() {
            let (q, r) = n.div_small(radix);
            ds.push(r as u8);
            n = q;
        }
        ds.reverse();
        while ds.len() < min_digits { ds.insert(0, 0); }

        let mut out = String::new();
        if self.negative && !self.is_zero() { out.push('-'); }
        for (i, &d) in ds.iter().enumerate() {
            if group > 0 && i > 0 && (ds.len() - i) % group == 0 { out.push('_'); }
            out.push(digit_char(d));
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => self.cmp_abs(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut r = Self::add_abs(self, rhs);
            r.negative = self.negative && !r.is_zero();
            r
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut r = Self::sub_abs(self, rhs);
                    r.negative = self.negative;
                    r
                }
                Ordering::Less => {
                    let mut r = Self::sub_abs(rhs, self);
                    r.negative = rhs.negative;
                    r
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self { self.add(&rhs.neg()) }

    pub fn neg(&self) -> Self {
        let mut r = self.clone();
        if !r.is_zero() { r.negative = !r.negative; }
        r
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() { return Self::zero(); }
        let mut out = vec![0u32; self.words.len() + rhs.words.len() + 1];
        for (i, &x) in self.words.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &y) in rhs.words.iter().enumerate() {
                let k = i + j;
                let v = out[k] as u64 + x as u64 * y as u64 + carry;
                out[k] = v as u32;
                carry = v >> 32;
            }
            let mut k = i + rhs.words.len();
            while carry != 0 {
                let v = out[k] as u64 + carry;
                out[k] = v as u32;
                carry = v >> 32;
                k += 1;
            }
        }
        let mut r = Self { negative: self.negative != rhs.negative, words: out };
        r.normalize();
        r
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() { return Err(Error::DivisionByZero); }
        let (mut q, mut r) = Self::div_abs(&self.abs(), &rhs.abs());
        q.negative = (self.negative != rhs.negative) && !q.is_zero();
        r.negative = self.negative && !r.is_zero();
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > 1024 {
            return Err(Error::ExponentTooLarge { exponent, limit: 1024 });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut out = Self::one();
        while e != 0 {
            if e & 1 != 0 { out = out.mul(&base); }
            e >>= 1;
            if e != 0 { base = base.mul(&base); }
        }
        Ok(out)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let r = Self::div_abs(&a, &b).1;
            a = b;
            b = r;
        }
        a.negative = false;
        a
    }
}

fn digit_value(b: u8) -> Option<u32> {
    match b {
        b'0'..=b'9' => Some((b - b'0') as u32),
        b'a'..=b'z' => Some((b - b'a' + 10) as u32),
        b'A'..=b'Z' => Some((b - b'A' + 10) as u32),
        _ => None,
    }
}

fn digit_char(d: u8) -> char {
    if d < 10 { (b'0' + d) as char } else { (b'a' + d - 10) as char }
}
```
