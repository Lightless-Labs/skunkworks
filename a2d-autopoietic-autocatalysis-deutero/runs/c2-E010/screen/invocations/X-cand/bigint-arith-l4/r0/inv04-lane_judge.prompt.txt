use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian binary magnitude.
    limbs: Vec<u32>,
}

impl Big {
    fn zero() -> Self { Self { negative: false, limbs: Vec::new() } }
    fn one() -> Self { Self { negative: false, limbs: vec![1] } }
    fn is_zero(&self) -> bool { self.limbs.is_empty() }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) { self.limbs.pop(); }
        if self.limbs.is_empty() { self.negative = false; }
    }

    fn radix_ok(radix: u32) -> Result<(), Error> {
        if (2..=36).contains(&radix) { Ok(()) } else { Err(Error::InvalidRadix { radix }) }
    }

    fn cmp_mag(a: &[u32], b: &[u32]) -> Ordering {
        match a.len().cmp(&b.len()) {
            Ordering::Equal => {
                for i in (0..a.len()).rev() {
                    match a[i].cmp(&b[i]) {
                        Ordering::Equal => {}
                        x => return x,
                    }
                }
                Ordering::Equal
            }
            x => x,
        }
    }

    fn add_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let x = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64 + carry;
            out.push((x & 0xffff_ffff) as u32);
            carry = x >> 32;
        }
        if carry != 0 { out.push(carry as u32); }
        out
    }

    // Requires a >= b.
    fn sub_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0u64;
        for i in 0..a.len() {
            let x = a[i] as u64;
            let y = b.get(i).copied().unwrap_or(0) as u64 + borrow;
            if x >= y {
                out.push((x - y) as u32);
                borrow = 0;
            } else {
                out.push(((1u64 << 32) + x - y) as u32);
                borrow = 1;
            }
        }
        while out.last() == Some(&0) { out.pop(); }
        out
    }

    fn mul_small(&mut self, m: u32) {
        if m == 0 || self.is_zero() { self.limbs.clear(); self.negative = false; return; }
        let mut carry = 0u64;
        for x in &mut self.limbs {
            let v = *x as u64 * m as u64 + carry;
            *x = v as u32;
            carry = v >> 32;
        }
        while carry != 0 { self.limbs.push(carry as u32); carry >>= 32; }
    }

    fn add_small(&mut self, n: u32) {
        let mut carry = n as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() { self.limbs.push(0); }
            let v = self.limbs[i] as u64 + carry;
            self.limbs[i] = v as u32;
            carry = v >> 32;
            i += 1;
        }
    }

    fn div_small(&mut self, divisor: u32) -> u32 {
        let mut rem = 0u64;
        for x in self.limbs.iter_mut().rev() {
            let v = (rem << 32) | *x as u64;
            *x = (v / divisor as u64) as u32;
            rem = v % divisor as u64;
        }
        self.normalize();
        rem as u32
    }

    fn div_rem_mag(a: &Self, b: &Self) -> (Self, Self) {
        debug_assert!(!b.is_zero());
        if Self::cmp_mag(&a.limbs, &b.limbs) == Ordering::Less {
            let mut r = a.clone(); r.negative = false; return (Self::zero(), r);
        }
        let mut quotient = vec![0u32; a.limbs.len()];
        let mut rem = Self::zero();
        for bit_index in (0..a.limbs.len() * 32).rev() {
            rem.mul_small(2);
            let bit = (a.limbs[bit_index / 32] >> (bit_index % 32)) & 1;
            rem.add_small(bit);
            if Self::cmp_mag(&rem.limbs, &b.limbs) != Ordering::Less {
                rem.limbs = Self::sub_mag(&rem.limbs, &b.limbs);
                quotient[bit_index / 32] |= 1u32 << (bit_index % 32);
            }
        }
        let mut q = Self { negative: false, limbs: quotient };
        q.normalize();
        rem.normalize();
        (q, rem)
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        Self::radix_ok(radix)?;
        if text.is_empty() { return Err(Error::Empty); }
        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() { return Err(Error::SignOnly); }

        fn digit(b: u8) -> Option<u32> {
            match b {
                b'0'..=b'9' => Some((b - b'0') as u32),
                b'a'..=b'z' => Some((b - b'a') as u32 + 10),
                b'A'..=b'Z' => Some((b - b'A') as u32 + 10),
                _ => None,
            }
        }

        for i in start..bytes.len() {
            if bytes[i] != b'_' && digit(bytes[i]).map_or(true, |d| d >= radix) {
                return Err(Error::InvalidDigit { index: i });
            }
        }
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let left = i > start && digit(bytes[i - 1]).is_some_and(|d| d < radix);
                let right = i + 1 < bytes.len() && digit(bytes[i + 1]).is_some_and(|d| d < radix);
                if !left || !right { return Err(Error::Separator { index: i }); }
            }
        }

        let digits: Vec<(usize, u32)> = (start..bytes.len())
            .filter_map(|i| digit(bytes[i]).map(|d| (i, d))).collect();
        if digits.len() > 1 && digits[0].1 == 0 {
            return Err(Error::LeadingZero { index: digits[0].0 });
        }
        if digits.len() > max_digits {
            return Err(Error::TooManyDigits { limit: max_digits, index: digits[max_digits].0 });
        }

        let mut n = Self::zero();
        for (_, d) in digits { n.mul_small(radix); n.add_small(d); }
        n.negative = negative && !n.is_zero();
        Ok(n)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        Self::radix_ok(radix)?;
        let mut n = self.clone(); n.negative = false;
        let mut digits = Vec::new();
        if n.is_zero() { digits.push(0u8); }
        while !n.is_zero() { digits.push(n.div_small(radix) as u8); }
        digits.reverse();
        while digits.len() < min_digits { digits.insert(0, 0); }

        let mut out = String::new();
        if self.negative && !self.is_zero() { out.push('-'); }
        for (i, d) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 { out.push('_'); }
            out.push(if *d < 10 { (b'0' + *d) as char } else { (b'a' + (*d - 10)) as char });
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => Self::cmp_mag(&self.limbs, &rhs.limbs),
            (true, true) => Self::cmp_mag(&self.limbs, &rhs.limbs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut n = Self { negative: self.negative, limbs: Self::add_mag(&self.limbs, &rhs.limbs) };
            n.normalize(); n
        } else {
            match Self::cmp_mag(&self.limbs, &rhs.limbs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut n = Self { negative: self.negative, limbs: Self::sub_mag(&self.limbs, &rhs.limbs) };
                    n.normalize(); n
                }
                Ordering::Less => {
                    let mut n = Self { negative: rhs.negative, limbs: Self::sub_mag(&rhs.limbs, &self.limbs) };
                    n.normalize(); n
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self { self.add(&rhs.neg()) }

    pub fn neg(&self) -> Self {
        let mut n = self.clone();
        if !n.is_zero() { n.negative = !n.negative; }
        n
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() { return Self::zero(); }
        let mut out = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let v = out[k] as u64 + a as u64 * b as u64 + carry;
                out[k] = v as u32;
                carry = v >> 32;
            }
            let mut k = i + rhs.limbs.len();
            while carry != 0 {
                if k == out.len() { out.push(0); }
                let v = out[k] as u64 + carry;
                out[k] = v as u32;
                carry = v >> 32;
                k += 1;
            }
        }
        let mut n = Self { negative: self.negative ^ rhs.negative, limbs: out };
        n.normalize(); n
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() { return Err(Error::DivisionByZero); }
        let mut a = self.clone(); a.negative = false;
        let mut b = rhs.clone(); b.negative = false;
        let (mut q, mut r) = Self::div_rem_mag(&a, &b);
        q.negative = !q.is_zero() && (self.negative ^ rhs.negative);
        r.negative = !r.is_zero() && self.negative;
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: LIMIT });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while e > 0 {
            if e & 1 == 1 { result = result.mul(&base); }
            e >>= 1;
            if e > 0 { base = base.mul(&base); }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.clone(); a.negative = false;
        let mut b = rhs.clone(); b.negative = false;
        while !b.is_zero() {
            let (_, r) = Self::div_rem_mag(&a, &b);
            a = b; b = r;
        }
        a.negative = false;
        a
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },

    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base 2^32 words.
    words: Vec<u32>,
}

impl Big {
    fn zero() -> Self { Self { negative: false, words: vec![0] } }
    fn one() -> Self { Self { negative: false, words: vec![1] } }
    fn is_zero(&self) -> bool { self.words.len() == 1 && self.words[0] == 0 }

    fn normalize(&mut self) {
        while self.words.len() > 1 && self.words.last() == Some(&0) { self.words.pop(); }
        if self.words.is_empty() { self.words.push(0); }
        if self.is_zero() { self.negative = false; }
    }

    fn abs(&self) -> Self {
        let mut r = self.clone();
        r.negative = false;
        r
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        self.words.len().cmp(&rhs.words.len()).then_with(|| {
            for (a, b) in self.words.iter().rev().zip(rhs.words.iter().rev()) {
                match a.cmp(b) { Ordering::Equal => {}, o => return o }
            }
            Ordering::Equal
        })
    }

    fn add_abs(a: &Self, b: &Self) -> Self {
        let n = a.words.len().max(b.words.len());
        let mut words = Vec::with_capacity(n + 1);
        let mut carry = 0u64;
        for i in 0..n {
            let sum = *a.words.get(i).unwrap_or(&0) as u64
                + *b.words.get(i).unwrap_or(&0) as u64 + carry;
            words.push(sum as u32);
            carry = sum >> 32;
        }
        if carry != 0 { words.push(carry as u32); }
        let mut r = Self { negative: false, words };
        r.normalize();
        r
    }

    fn sub_abs(a: &Self, b: &Self) -> Self {
        let mut words = Vec::with_capacity(a.words.len());
        let mut borrow = 0u64;
        for i in 0..a.words.len() {
            let av = a.words[i] as u64;
            let bv = *b.words.get(i).unwrap_or(&0) as u64 + borrow;
            words.push(av.wrapping_sub(bv) as u32);
            borrow = (av < bv) as u64;
        }
        let mut r = Self { negative: false, words };
        r.normalize();
        r
    }

    fn mul_small(&mut self, m: u32) {
        let mut carry = 0u64;
        for w in &mut self.words {
            let v = *w as u64 * m as u64 + carry;
            *w = v as u32;
            carry = v >> 32;
        }
        if carry != 0 { self.words.push(carry as u32); }
        self.normalize();
    }

    fn add_small(&mut self, a: u32) {
        let mut carry = a as u64;
        let mut i = 0;
        while carry != 0 && i < self.words.len() {
            let v = self.words[i] as u64 + carry;
            self.words[i] = v as u32;
            carry = v >> 32;
            i += 1;
        }
        if carry != 0 { self.words.push(carry as u32); }
    }

        self.mul_small(m);
        self.add_small(a);
    }

    fn shl1_add(&mut self, bit: u32) {
        let mut carry = bit as u64;
        for w in &mut self.words {
            let v = ((*w as u64) << 1) | carry;
            *w = v as u32;
            carry = v >> 32;
        }
        if carry != 0 { self.words.push(carry as u32); }
        self.normalize();
    }

    fn bit_len(&self) -> usize {
        if self.is_zero() { 0 } else {
            (self.words.len() - 1) * 32 + (32 - self.words.last().unwrap().leading_zeros() as usize)
        }
    }

    fn get_bit(&self, i: usize) -> u32 {
        self.words.get(i / 32).map_or(0, |w| (w >> (i % 32)) & 1)
    }

    fn set_bit(&mut self, i: usize) {
        while self.words.len() <= i / 32 { self.words.push(0); }
        self.words[i / 32] |= 1 << (i % 32);
    }

    fn div_abs(a: &Self, b: &Self) -> (Self, Self) {
        let mut q = Self::zero();
        let mut r = Self::zero();
        for i in (0..a.bit_len()).rev() {
            r.shl1_add(a.get_bit(i));
            if r.cmp_abs(b) != Ordering::Less {
                r = Self::sub_abs(&r, b);
                q.set_bit(i);
            }
        }
        q.normalize();
        r.normalize();
        (q, r)
    }

    fn div_small(&self, d: u32) -> (Self, u32) {
        let mut words = vec![0; self.words.len()];
        let mut rem = 0u64;
        for i in (0..self.words.len()).rev() {
            let cur = (rem << 32) | self.words[i] as u64;
            words[i] = (cur / d as u64) as u32;
            rem = cur % d as u64;
        }
        let mut q = Self { negative: false, words };
        q.normalize();
        (q, rem as u32)
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) { return Err(Error::InvalidRadix { radix }); }
        let bytes = text.as_bytes();
        if bytes.is_empty() { return Err(Error::Empty); }
        let mut start = 0;
        let mut negative = false;
        if bytes[0] == b'+' || bytes[0] == b'-' {
            negative = bytes[0] == b'-';
            start = 1;
            if start == bytes.len() { return Err(Error::SignOnly); }
        }

        let mut invalid = None;
        for i in start..bytes.len() {
            if bytes[i] != b'_' && digit_value(bytes[i]).map_or(true, |v| v >= radix) {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid { return Err(Error::InvalidDigit { index }); }

        for i in start..bytes.len() {
            if bytes[i] == b'_' &&
                !(i > start && i + 1 < bytes.len() && bytes[i - 1] != b'_' && bytes[i + 1] != b'_') {
                return Err(Error::Separator { index: i });
            }
        }

        let count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if count > 1 && bytes[first] == b'0' { return Err(Error::LeadingZero { index: first }); }
        if count > max_digits {
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen == max_digits.saturating_add(1) {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut n = Self::zero();
        for &b in &bytes[start..] {
            if b != b'_' { n.mul_small_add(radix, digit_value(b).unwrap()); }
        }
        n.negative = negative && !n.is_zero();
        Ok(n)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) { return Err(Error::InvalidRadix { radix }); }
        let mut n = self.abs();
        let mut ds = Vec::new();
        if n.is_zero() { ds.push(0); }
        while !n.is_zero() {
            let (q, r) = n.div_small(radix);
            ds.push(r as u8);
            n = q;
        }
        ds.reverse();
        while ds.len() < min_digits { ds.insert(0, 0); }

        let mut out = String::new();
        if self.negative && !self.is_zero() { out.push('-'); }
        for (i, &d) in ds.iter().enumerate() {
            if group > 0 && i > 0 && (ds.len() - i) % group == 0 { out.push('_'); }
            out.push(digit_char(d));
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => self.cmp_abs(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut r = Self::add_abs(self, rhs);
            r.negative = self.negative && !r.is_zero();
            r
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut r = Self::sub_abs(self, rhs);
                    r.negative = self.negative;
                    r
                }
                Ordering::Less => {
                    let mut r = Self::sub_abs(rhs, self);
                    r.negative = rhs.negative;
                    r
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self { self.add(&rhs.neg()) }

    pub fn neg(&self) -> Self {
        let mut r = self.clone();
        if !r.is_zero() { r.negative = !r.negative; }
        r
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() { return Self::zero(); }
        let mut out = vec![0u32; self.words.len() + rhs.words.len() + 1];
        for (i, &x) in self.words.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &y) in rhs.words.iter().enumerate() {
                let k = i + j;
                let v = out[k] as u64 + x as u64 * y as u64 + carry;
                out[k] = v as u32;
                carry = v >> 32;
            }
            let mut k = i + rhs.words.len();
            while carry != 0 {
                let v = out[k] as u64 + carry;
                out[k] = v as u32;
                carry = v >> 32;
                k += 1;
            }
        }
        let mut r = Self { negative: self.negative != rhs.negative, words: out };
        r.normalize();
        r
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() { return Err(Error::DivisionByZero); }
        let (mut q, mut r) = Self::div_abs(&self.abs(), &rhs.abs());
        q.negative = (self.negative != rhs.negative) && !q.is_zero();
        r.negative = self.negative && !r.is_zero();
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > 1024 {
            return Err(Error::ExponentTooLarge { exponent, limit: 1024 });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut out = Self::one();
        while e != 0 {
            if e & 1 != 0 { out = out.mul(&base); }
            e >>= 1;
            if e != 0 { base = base.mul(&base); }
        }
        Ok(out)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let r = Self::div_abs(&a, &b).1;
            a = b;
            b = r;
        }
        a.negative = false;
        a
    }
}

fn digit_value(b: u8) -> Option<u32> {
    match b {
        b'0'..=b'9' => Some((b - b'0') as u32),
        b'a'..=b'z' => Some((b - b'a' + 10) as u32),
        b'A'..=b'Z' => Some((b - b'A' + 10) as u32),
        _ => None,
    }
}

fn digit_char(d: u8) -> char {
    if d < 10 { (b'0' + d) as char } else { (b'a' + d - 10) as char }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const EXPONENT_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self { negative: false, limbs: Vec::new() }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn from_u32(n: u32) -> Self {
        if n == 0 { Self::zero() } else { Self { negative: false, limbs: vec![n] } }
    }

    fn mul_small_add(&mut self, multiplier: u32, add: u32) {
        let mut carry = add as u64;
        for limb in &mut self.limbs {
            let n = *limb as u64 * multiplier as u64 + carry;
            *limb = (n % Self::BASE) as u32;
            carry = n / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }
        let sign = match bytes[0] {
            b'+' => Some(false),
            b'-' => Some(true),
            _ => None,
        };
        let start = if sign.is_some() { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut first_invalid = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                continue;
            }
            let d = match bytes[i] {
                b'0'..=b'9' => (bytes[i] - b'0') as u32,
                b'a'..=b'z' => (bytes[i] - b'a' + 10) as u32,
                b'A'..=b'Z' => (bytes[i] - b'A' + 10) as u32,
                _ => {
                    first_invalid = Some(i);
                    break;
                }
            };
            if d >= radix {
                first_invalid = Some(i);
                break;
            }
        }
        if let Some(index) = first_invalid {
            return Err(Error::InvalidDigit { index });
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let prev_ok = i > start && bytes[i - 1] != b'_';
                let next_ok = i + 1 < bytes.len() && bytes[i + 1] != b'_';
                if !prev_ok || !next_ok {
                    return Err(Error::Separator { index: i });
                }
            }
        }

        let mut digit_count = 0usize;
        let mut first_digit_index = None;
        let mut first_digit = 0u32;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                continue;
            }
            if first_digit_index.is_none() {
                first_digit_index = Some(i);
                first_digit = match bytes[i] {
                    b'0'..=b'9' => (bytes[i] - b'0') as u32,
                    b'a'..=b'z' => (bytes[i] - b'a' + 10) as u32,
                    _ => (bytes[i] - b'A' + 10) as u32,
                };
            }
            digit_count += 1;
        }
        let first_index = first_digit_index.unwrap();
        if digit_count > 1 && first_digit == 0 {
            return Err(Error::LeadingZero { index: first_index });
        }
        if digit_count > max_digits {
            let excess = start + bytes[start..].iter().filter(|&&b| b != b'_').take(max_digits).count();
            let mut seen = 0usize;
            let mut excess_index = excess;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    if seen == max_digits {
                        excess_index = i;
                        break;
                    }
                    seen += 1;
                }
            }
            return Err(Error::TooManyDigits { limit: max_digits, index: excess_index });
        }

        let mut result = Self::zero();
        for &b in &bytes[start..] {
            if b == b'_' {
                continue;
            }
            let d = match b {
                b'0'..=b'9' => (b - b'0') as u32,
                b'a'..=b'z' => (b - b'a' + 10) as u32,
                _ => (b - b'A' + 10) as u32,
            };
            result.mul_small_add(radix, d);
        }
        result.negative = sign.unwrap_or(false) && !result.limbs.is_empty();
        Ok(result)
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn abs_add(&self, rhs: &Self) -> Self {
        let n = self.limbs.len().max(rhs.limbs.len());
        let mut out = Vec::with_capacity(n + 1);
        let mut carry = 0u64;
        for i in 0..n {
            let sum = self.limbs.get(i).copied().unwrap_or(0) as u64
                + rhs.limbs.get(i).copied().unwrap_or(0) as u64 + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 { out.push(carry as u32); }
        Self { negative: false, limbs: out }
    }

    fn abs_sub(&self, rhs: &Self) -> Self {
        let mut out = Vec::with_capacity(self.limbs.len());
        let mut borrow = 0i64;
        for i in 0..self.limbs.len() {
            let mut d = self.limbs[i] as i64
                - rhs.limbs.get(i).copied().unwrap_or(0) as i64 - borrow;
            if d < 0 {
                d += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(d as u32);
        }
        let mut r = Self { negative: false, limbs: out };
        r.normalize();
        r
    }

    fn abs(&self) -> Self {
        let mut r = self.clone();
        r.negative = false;
        r
    }

    fn mul_small_in_place(&mut self, m: u32) {
        if m == 0 || self.limbs.is_empty() {
            *self = Self::zero();
            return;
        }
        let mut carry = 0u64;
        for x in &mut self.limbs {
            let t = *x as u64 * m as u64 + carry;
            *x = (t % Self::BASE) as u32;
            carry = t / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_in_place(&mut self, a: u32) {
        let mut carry = a as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let t = self.limbs[i] as u64 + carry;
            self.limbs[i] = (t % Self::BASE) as u32;
            carry = t / Self::BASE;
            i += 1;
        }
    }

    fn div_small_in_place(&mut self, d: u32) -> u32 {
        let mut rem = 0u64;
        for x in self.limbs.iter_mut().rev() {
            let cur = rem * Self::BASE + *x as u64;
            *x = (cur / d as u64) as u32;
            rem = cur % d as u64;
        }
        self.normalize();
        rem as u32
    }

    fn abs_div_rem(a: &Self, b: &Self) -> (Self, Self) {
        if a.abs_cmp(b) == Ordering::Less {
            return (Self::zero(), a.abs());
        }
        let mut quotient_be = Vec::with_capacity(a.limbs.len());
        let mut rem = Self::zero();
        for &digit in a.limbs.iter().rev() {
            rem.limbs.insert(0, digit);
            rem.normalize();
            let mut lo = 0u32;
            let mut hi = (Self::BASE - 1) as u32;
            let mut best = 0u32;
            while lo <= hi {
                let mid = lo + (hi - lo) / 2;
                let mut product = b.clone();
                product.mul_small_in_place(mid);
                match product.abs_cmp(&rem) {
                    Ordering::Greater => {
                        if mid == 0 { break; }
                        hi = mid - 1;
                    }
                    _ => {
                        best = mid;
                        if mid == (Self::BASE - 1) as u32 { break; }
                        lo = mid + 1;
                    }
                }
            }
            quotient_be.push(best);
            if best != 0 {
                let mut product = b.clone();
                product.mul_small_in_place(best);
                rem = rem.abs_sub(&product);
            }
        }
        quotient_be.reverse();
        let mut q = Self { negative: false, limbs: quotient_be };
        q.normalize();
        (q, rem)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let alphabet = b"0123456789abcdefghijklmnopqrstuvwxyz";
        let mut n = self.abs();
        let mut digits = Vec::new();
        if n.limbs.is_empty() {
            digits.push(b'0');
        } else {
            while !n.limbs.is_empty() {
                let d = n.div_small_in_place(radix);
                digits.push(alphabet[d as usize]);
            }
            digits.reverse();
        }
        while digits.len() < min_digits {
            digits.insert(0, b'0');
        }
        let mut out = String::with_capacity(digits.len() + digits.len() / group.max(1) + 1);
        if self.negative && !self.limbs.is_empty() {
            out.push('-');
        }
        for (i, &d) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(d as char);
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => rhs.abs_cmp(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut r = self.abs_add(rhs);
            r.negative = self.negative;
            r.normalize();
            r
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Greater => {
                    let mut r = self.abs_sub(rhs);
                    r.negative = self.negative;
                    r.normalize();
                    r
                }
                Ordering::Less => {
                    let mut r = rhs.abs_sub(self);
                    r.negative = rhs.negative;
                    r.normalize();
                    r
                }
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut r = self.clone();
        if !r.limbs.is_empty() {
            r.negative = !r.negative;
        }
        r
    }

    pub fn signum(&self) -> i8 {
        if self.limbs.is_empty() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.limbs.is_empty() || rhs.limbs.is_empty() {
            return Self::zero();
        }
        let mut out = vec![0u64; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let t = out[k] + a as u64 * b as u64 + carry;
                out[k] = t % Self::BASE;
                carry = t / Self::BASE;
            }
            let mut k = i + rhs.limbs.len();
            while carry != 0 {
                if k == out.len() { out.push(0); }
                let t = out[k] + carry;
                out[k] = t % Self::BASE;
                carry = t / Self::BASE;
                k += 1;
            }
        }
        let mut r = Self { negative: self.negative ^ rhs.negative, limbs: out.into_iter().map(|x| x as u32).collect() };
        r.normalize();
        r
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.limbs.is_empty() {
            return Err(Error::DivisionByZero);
        }
        let (mut q, mut r) = Self::abs_div_rem(self, rhs);
        q.negative = !q.limbs.is_empty() && (self.negative ^ rhs.negative);
        r.negative = !r.limbs.is_empty() && self.negative;
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::EXPONENT_LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: Self::EXPONENT_LIMIT });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while e != 0 {
            if e & 1 == 1 {
                result = result.mul(&base);
            }
            e >>= 1;
            if e != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.limbs.is_empty() {
            let r = a.div_rem(&b).unwrap().1.abs();
            a = b;
            b = r;
        }
        a.negative = false;
        a
    }
}

            self.limbs.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let n = a.len().max(b.len());
        let mut out = Vec::with_capacity(n + 1);
        let mut carry = 0u64;
        for i in 0..n {
            let sum = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64 + carry;
            out.push((sum % BASE) as u32);
            carry = sum / BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        out
    }

    fn sub_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let mut v = a[i] as i64 - b.get(i).copied().unwrap_or(0) as i64 - borrow;
            if v < 0 {
                v += BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(v as u32);
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_small_abs(&mut self, n: u32) {
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let x = *limb as u64 * n as u64 + carry;
            *limb = (x % BASE) as u32;
            carry = x / BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % BASE) as u32);
            carry /= BASE;
        }
    }

    fn add_small_abs(&mut self, n: u32) {
        let mut carry = n as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let x = self.limbs[i] as u64 + carry;
            self.limbs[i] = (x % BASE) as u32;
            carry = x / BASE;
            i += 1;
        }
    }

    fn div_rem_small_abs(&self, divisor: u32) -> (Self, u32) {
        let mut q = vec![0; self.limbs.len()];
        let mut rem = 0u64;
        for i in (0..self.limbs.len()).rev() {
            let x = rem * BASE + self.limbs[i] as u64;
            q[i] = (x / divisor as u64) as u32;
            rem = x % divisor as u64;
        }
        let mut result = Self { negative: false, limbs: q };
        result.normalize();
        (result, rem as u32)
    }

    fn mul_abs(&self, rhs: &Self) -> Self {
        let mut out = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let v = out[k] as u64 + a as u64 * b as u64 + carry;
                out[k] = (v % BASE) as u32;
                carry = v / BASE;
            }
            let mut k = i + rhs.limbs.len();
            while carry != 0 {
                if k == out.len() {
                    out.push(0);
                }
                let v = out[k] as u64 + carry;
                out[k] = (v % BASE) as u32;
                carry = v / BASE;
                k += 1;
            }
        }
        let mut result = Self { negative: false, limbs: out };
        result.normalize();
        result
    }

    fn mul_small_value(&self, n: u32) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result.mul_small_abs(n);
        result.normalize();
        result
    }

    fn add_small_value(&self, n: u32) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result.add_small_abs(n);
        result.normalize();
        result
    }

    fn div_rem_abs(&self, rhs: &Self) -> (Self, Self) {
        if self.cmp_abs(rhs) == Ordering::Less {
            let mut r = self.clone();
            r.negative = false;
            return (Self::zero(), r);
        }
        let mut quotient_be = Vec::with_capacity(self.limbs.len());
        let mut rem = Self::zero();
        for &limb in self.limbs.iter().rev() {
            rem.limbs.insert(0, limb);
            rem.normalize();
            let mut lo = 0u64;
            let mut hi = BASE - 1;
            let mut best = 0u64;
            while lo <= hi {
                let mid = (lo + hi) / 2;
                let prod = rhs.mul_small_value(mid as u32);
                match prod.cmp_abs(&rem) {
                    Ordering::Greater => {
                        if mid == 0 { break; }
                        hi = mid - 1;
                    }
                    _ => {
                        best = mid;
                        lo = mid + 1;
                    }
                }
            }
            if best != 0 {
                let prod = rhs.mul_small_value(best as u32);
                rem = Self { negative: false, limbs: Self::sub_abs(&rem.limbs, &prod.limbs) };
                rem.normalize();
            }
            quotient_be.push(best as u32);
        }
        quotient_be.reverse();
        let mut q = Self { negative: false, limbs: quotient_be };
        q.normalize();
        rem.normalize();
        (q, rem)
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (start, negative) = match bytes[0] {
            b'+' => (1, false),
            b'-' => (1, true),
            _ => (0, false),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut first_invalid = None;
        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b == b'_' {
                continue;
            }
            let digit = match b {
                b'0'..=b'9' => (b - b'0') as u32,
                b'a'..=b'z' => (b - b'a') as u32 + 10,
                b'A'..=b'Z' => (b - b'A') as u32 + 10,
                _ => radix,
            };
            if digit >= radix {
                first_invalid = Some(i);
                break;
            }
        }
        if let Some(index) = first_invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut first_bad_sep = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let prev_valid = i > start && bytes[i - 1] != b'_';
                let next_valid = i + 1 < bytes.len() && bytes[i + 1] != b'_';
                if !prev_valid || !next_valid {
                    first_bad_sep = Some(i);
                    break;
                }
            }
        }
        if let Some(index) = first_bad_sep {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        if digit_count > 1 {
            let first = bytes[start..].iter().position(|&b| b != b'_').unwrap() + start;
            if bytes[first] == b'0' {
                return Err(Error::LeadingZero { index: first });
            }
        }
        let mut seen = 0usize;
        let mut excess = None;
        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b != b'_' {
                seen += 1;
                if seen > max_digits {
                    excess = Some(i);
                    break;
                }
            }
        }
        if let Some(index) = excess {
            return Err(Error::TooManyDigits { limit: max_digits, index });
        }

        let mut value = Self::zero();
        for &b in &bytes[start..] {
            if b == b'_' {
                continue;
            }
            let digit = match b {
                b'0'..=b'9' => (b - b'0') as u32,
                b'a'..=b'z' => (b - b'a') as u32 + 10,
                b'A'..=b'Z' => (b - b'A') as u32 + 10,
                _ => unreachable!(),
            };
            value.mul_small_abs(radix);
            value.add_small_abs(digit);
        }
        value.negative = negative && !value.is_zero();
        Ok(value)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let mut n = self.clone();
        n.negative = false;
        let mut digits = Vec::new();
        if n.is_zero() {
            digits.push(0);
        } else {
            while !n.is_zero() {
                let (q, r) = n.div_rem_small_abs(radix);
                digits.push(r);
                n = q;
            }
            digits.reverse();
        }
        while digits.len() < min_digits {
            digits.insert(0, 0);
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, &d) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(match d {
                0..=9 => (b'0' + d as u8) as char,
                _ => (b'a' + (d - 10) as u8) as char,
            });
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => self.cmp_abs(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self {
                negative: self.negative,
                limbs: Self::add_abs(&self.limbs, &rhs.limbs),
            };
            result.normalize();
            result
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self {
                        negative: self.negative,
                        limbs: Self::sub_abs(&self.limbs, &rhs.limbs),
                    };
                    result.normalize();
                    result
                }
                Ordering::Less => {
                    let mut result = Self {
                        negative: rhs.negative,
                        limbs: Self::sub_abs(&rhs.limbs, &self.limbs),
                    };
                    result.normalize();
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut result = self.mul_abs(rhs);
        result.negative = self.negative != rhs.negative;
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut q, mut r) = self.div_rem_abs(rhs);
        q.negative = !q.is_zero() && (self.negative != rhs.negative);
        r.negative = !r.is_zero() && self.negative;
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > POW_LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: POW_LIMIT });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while e != 0 {
            if e & 1 == 1 {
                result = result.mul(&base);
            }
            e >>= 1;
            if e != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.clone();
        let mut b = rhs.clone();
        a.negative = false;
        b.negative = false;
        while !b.is_zero() {
            let (_, r) = a.div_rem_abs(&b);
            a = b;
            b = r;
        }
        a.negative = false;
        a.normalize();
        a
    }

#[derive(Clone, Debug, PartialEq, Eq)]

    Empty,
    SignOnly,

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base 1e9 limbs. Zero is represented by [0], nonnegative.
    limbs: Vec<u32>,
}

const BASE: u64 = 1_000_000_000;

impl Big {
    fn zero() -> Self {
        Self { negative: false, limbs: vec![0] }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.limbs.push(0);
        }
        if self.limbs.len() == 1 && self.limbs[0] == 0 {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn abs(&self) -> Self {
        let mut r = self.clone();
        r.negative = false;
        r
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        self.limbs.len().cmp(&rhs.limbs.len()).then_with(|| {
            for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                match a.cmp(b) {
                    Ordering::Equal => {}
                    o => return o,
                }
            }
            Ordering::Equal
        })
    }

    fn mul_small_add(&mut self, m: u32, a: u32) {
        let mut carry = a as u64;
        for limb in &mut self.limbs {
            let v = *limb as u64 * m as u64 + carry;
            *limb = (v % BASE) as u32;
            carry = v / BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % BASE) as u32);
            carry /= BASE;
        }
        self.normalize();
    }

    fn add_abs(a: &Self, b: &Self) -> Self {
        let n = a.limbs.len().max(b.limbs.len());
        let mut out = Vec::with_capacity(n + 1);
        let mut carry = 0u64;
        for i in 0..n {
            let x = *a.limbs.get(i).unwrap_or(&0) as u64;
            let y = *b.limbs.get(i).unwrap_or(&0) as u64;
            let s = x + y + carry;
            out.push((s % BASE) as u32);
            carry = s / BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        let mut r = Self { negative: false, limbs: out };
        r.normalize();
        r
    }

    // Requires |a| >= |b|.
    fn sub_abs(a: &Self, b: &Self) -> Self {
        let mut out = Vec::with_capacity(a.limbs.len());
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let x = a.limbs[i] as i64 - borrow;
            let y = *b.limbs.get(i).unwrap_or(&0) as i64;
            if x < y {
                out.push((x + BASE as i64 - y) as u32);
                borrow = 1;
            } else {
                out.push((x - y) as u32);
                borrow = 0;
            }
        }
        let mut r = Self { negative: false, limbs: out };
        r.normalize();
        r
    }

    fn mul_abs(a: &Self, b: &Self) -> Self {
        if a.is_zero() || b.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u64; a.limbs.len() + b.limbs.len() + 1];
        for (i, &x) in a.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &y) in b.limbs.iter().enumerate() {
                let k = i + j;
                let v = out[k] + x as u64 * y as u64 + carry;
                out[k] = v % BASE;
                carry = v / BASE;
            }
            let mut k = i + b.limbs.len();
            while carry != 0 {
                let v = out[k] + carry;
                out[k] = v % BASE;
                carry = v / BASE;
                k += 1;
            }
        }
        let mut r = Self {
            negative: false,
            limbs: out.into_iter().map(|x| x as u32).collect(),
        };
        r.normalize();
        r
    }

    fn div_rem_abs(a: &Self, b: &Self) -> (Self, Self) {
        debug_assert!(!b.is_zero());
        if a.cmp_abs(b) == Ordering::Less {
            return (Self::zero(), a.abs());
        }
        let mut quotient_be = Vec::with_capacity(a.limbs.len());
        let mut rem = Self::zero();
        for &digit in a.limbs.iter().rev() {
            rem.limbs.insert(0, digit);
            rem.normalize();
            let mut lo = 0u64;
            let mut hi = BASE - 1;
            while lo < hi {
                let mid = (lo + hi + 1) / 2;
                let prod = Self::mul_abs_small(b, mid as u32);
                if prod.cmp_abs(&rem) != Ordering::Greater {
                    lo = mid;
                } else {
                    hi = mid - 1;
                }
            }
            quotient_be.push(lo as u32);
            let prod = Self::mul_abs_small(b, lo as u32);
            rem = Self::sub_abs(&rem, &prod);
        }
        quotient_be.reverse();
        let mut q = Self { negative: false, limbs: quotient_be };
        q.normalize();
        rem.normalize();
        (q, rem)
    }

    fn mul_abs_small(a: &Self, m: u32) -> Self {
        if m == 0 || a.is_zero() { return Self::zero(); }
        let mut out = Vec::with_capacity(a.limbs.len() + 1);
        let mut carry = 0u64;
        for &x in &a.limbs {
            let v = x as u64 * m as u64 + carry;
            out.push((v % BASE) as u32);
            carry = v / BASE;
        }
        if carry != 0 { out.push(carry as u32); }
        let mut r = Self { negative: false, limbs: out };
        r.normalize();
        r
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }
        let mut start = 0;
        let mut negative = false;
        if bytes[0] == b'+' || bytes[0] == b'-' {
            negative = bytes[0] == b'-';
            start = 1;
            if start == bytes.len() {
                return Err(Error::SignOnly);
            }
        }

        let mut invalid = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' { continue; }
            let v = digit_value(bytes[i]);
            if v.map_or(true, |x| x >= radix) {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let left = i > start && bytes[i - 1] != b'_';
                let right = i + 1 < bytes.len() && bytes[i + 1] != b'_';
                if !left || !right {
                    return Err(Error::Separator { index: i });
                }
            }
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if digit_count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero { index: first_digit });
        }
        if digit_count > max_digits {
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen == max_digits.saturating_add(1) {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut value = Self::zero();
        for &b in &bytes[start..] {
            if b == b'_' { continue; }
            value.mul_small_add(radix, digit_value(b).unwrap());
        }
        value.negative = negative && !value.is_zero();
        Ok(value)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });

        let mut digits = Vec::new();

                digits.push(r as u8);

    fn div_small(&self, d: u32) -> (Self, u32) {

        for i in (0..self.limbs.len()).rev() {
            let cur = rem * BASE + self.limbs[i] as u64;
            out[i] = (cur / d as u64) as u32;
            rem = cur % d as u64;
        }
        let mut q = Self { negative: false, limbs: out };
        q.normalize();
        (q, rem as u32)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => self.cmp_abs(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut r = Self::add_abs(self, rhs);
            r.negative = self.negative && !r.is_zero();
            r

                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut r = Self::sub_abs(self, rhs);
                    r.negative = self.negative;
                    r
                }
                Ordering::Less => {
                    let mut r = Self::sub_abs(rhs, self);
                    r.negative = rhs.negative;
                    r
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut r = self.clone();
        if !r.is_zero() { r.negative = !r.negative; }
        r
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        let mut r = Self::mul_abs(self, rhs);
        r.negative = (self.negative != rhs.negative) && !r.is_zero();
        r
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() { return Err(Error::DivisionByZero); }
        let (mut q, mut r) = Self::div_rem_abs(self, rhs);
        q.negative = (self.negative != rhs.negative) && !q.is_zero();
        r.negative = self.negative && !r.is_zero();
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > 1024 {
            return Err(Error::ExponentTooLarge { exponent, limit: 1024 });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut out = Self::one();
        while e > 0 {
            if e & 1 == 1 { out = out.mul(&base); }
            e >>= 1;
            if e > 0 { base = base.mul(&base); }
        }
        Ok(out)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let r = a.div_rem(&b).expect("nonzero divisor").1;
            a = b;
            b = r.abs();
        }
        a.negative = false;
        a
    }
}

fn digit_value(b: u8) -> Option<u32> {
    match b {
        b'0'..=b'9' => Some((b - b'0') as u32),
        b'a'..=b'z' => Some((b - b'a' + 10) as u32),
        b'A'..=b'Z' => Some((b - b'A' + 10) as u32),
        _ => None,
    }
}

fn digit_char(d: u8) -> char {
    match d {
        0..=9 => (b'0' + d) as char,
        _ => (b'a' + d - 10) as char,
    }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1,000,000,000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;

    fn zero() -> Self {
        Self { negative: false, limbs: Vec::new() }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn validate_radix(radix: u32) -> Result<(), Error> {
        if !(2..=36).contains(&radix) {
            Err(Error::InvalidRadix { radix })
        } else {
            Ok(())
        }
    }

    fn cmp_mag(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let sum = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64 + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        out
    }

    // Requires a >= b.
    fn sub_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let mut v = a[i] as i64 - b.get(i).copied().unwrap_or(0) as i64 - borrow;
            if v < 0 {
                v += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(v as u32);
        }
        while out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_small_mag(&mut self, m: u32) {
        if m == 0 || self.is_zero() {
            self.limbs.clear();
            self.negative = false;
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let n = *limb as u64 * m as u64 + carry;
            *limb = (n % Self::BASE) as u32;
            carry = n / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_mag(&mut self, n: u32) {
        let mut carry = n as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[i] as u64 + carry;
            self.limbs[i] = (sum % Self::BASE) as u32;
            carry = sum / Self::BASE;
            i += 1;
        }
    }

    fn div_small_mag(&mut self, divisor: u32) -> u32 {
        let mut rem = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let n = rem * Self::BASE + *limb as u64;
            *limb = (n / divisor as u64) as u32;
            rem = n % divisor as u64;
        }
        self.normalize();
        rem as u32
    }

    fn div_rem_mag(a: &Self, b: &Self) -> (Self, Self) {
        debug_assert!(!b.is_zero());
        if a.cmp_mag(b) == Ordering::Less {
            let mut r = a.clone();
            r.negative = false;
            return (Self::zero(), r);
        }
        let mut q_be = Vec::with_capacity(a.limbs.len());
        let mut rem = Self::zero();
        for &limb in a.limbs.iter().rev() {
            rem.mul_small_mag(Self::BASE as u32);
            rem.add_small_mag(limb);
            let mut lo = 0u64;
            let mut hi = Self::BASE - 1;
            while lo < hi {
                let mid = (lo + hi + 1) / 2;
                let mut product = b.clone();
                product.negative = false;
                product.mul_small_mag(mid as u32);
                if product.cmp_mag(&rem) != Ordering::Greater {
                    lo = mid;
                } else {
                    hi = mid - 1;
                }
            }
            let digit = lo as u32;
            if digit != 0 {
                let mut product = b.clone();
                product.negative = false;
                product.mul_small_mag(digit);
                rem.limbs = Self::sub_mag(&rem.limbs, &product.limbs);
            }
            q_be.push(digit);
        }
        q_be.reverse();
        let mut q = Self { negative: false, limbs: q_be };
        q.normalize();
        rem.normalize();
        (q, rem)
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        Self::validate_radix(radix)?;
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let digit_value = |b: u8| -> Option<u32> {
            match b {
                b'0'..=b'9' => Some((b - b'0') as u32),
                b'a'..=b'z' => Some((b - b'a') as u32 + 10),
                b'A'..=b'Z' => Some((b - b'A') as u32 + 10),
                _ => None,
            }
        };

        for i in start..bytes.len() {
            if bytes[i] != b'_' && digit_value(bytes[i]).map_or(true, |v| v >= radix) {
                return Err(Error::InvalidDigit { index: i });
            }
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let left_ok = i > start && digit_value(bytes[i - 1]).is_some_and(|v| v < radix);
                let right_ok = i + 1 < bytes.len()
                    && digit_value(bytes[i + 1]).is_some_and(|v| v < radix);
                if !left_ok || !right_ok {
                    return Err(Error::Separator { index: i });
                }
            }
        }

        let digits: Vec<(usize, u32)> = (start..bytes.len())
            .filter_map(|i| digit_value(bytes[i]).map(|v| (i, v)))
            .collect();

        if digits.len() > 1 && digits[0].1 == 0 {
            return Err(Error::LeadingZero { index: digits[0].0 });
        }
        if digits.len() > max_digits {
            return Err(Error::TooManyDigits { limit: max_digits, index: digits[max_digits].0 });
        }

        let mut result = Self::zero();
        for &(_, digit) in &digits {
            result.mul_small_mag(radix);
            result.add_small_mag(digit);
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        Self::validate_radix(radix)?;
        let mut n = self.clone();
        n.negative = false;
        let mut digits = Vec::new();
        if n.is_zero() {
            digits.push(0u8);
        } else {
            while !n.is_zero() {
                digits.push(n.div_small_mag(radix) as u8);
            }
            digits.reverse();
        }
        if digits.len() < min_digits {
            let mut padded = vec![0u8; min_digits - digits.len()];
            padded.extend(digits);
            digits = padded;
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, d) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(if *d < 10 {
                (b'0' + *d) as char
            } else {
                (b'a' + (*d - 10)) as char
            });
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_mag(rhs),
            (true, true) => self.cmp_mag(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self { negative: self.negative, limbs: Self::add_mag(&self.limbs, &rhs.limbs) };
            out.normalize();
            out
        } else {
            match self.cmp_mag(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self { negative: self.negative, limbs: Self::sub_mag(&self.limbs, &rhs.limbs) };
                    out.normalize();
                    out
                }
                Ordering::Less => {
                    let mut out = Self { negative: rhs.negative, limbs: Self::sub_mag(&rhs.limbs, &self.limbs) };
                    out.normalize();
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u64; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let n = out[k] + a as u64 * b as u64 + carry;
                out[k] = n % Self::BASE;
                carry = n / Self::BASE;
            }
            let mut k = i + rhs.limbs.len();
            while carry != 0 {
                if k == out.len() {
                    out.push(0);
                }
                let n = out[k] + carry;
                out[k] = n % Self::BASE;
                carry = n / Self::BASE;
                k += 1;
            }
        }
        let mut result = Self {
            negative: self.negative ^ rhs.negative,
            limbs: out.into_iter().map(|v| v as u32).collect(),
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let mut a = self.clone();
        let mut b = rhs.clone();
        a.negative = false;
        b.negative = false;
        let (mut q, mut r) = Self::div_rem_mag(&a, &b);
        q.negative = (self.negative ^ rhs.negative) && !q.is_zero();
        r.negative = self.negative && !r.is_zero();
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: LIMIT });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while e != 0 {
            if e & 1 != 0 {
                result = result.mul(&base);
            }
            e >>= 1;
            if e != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.clone();
        let mut b = rhs.clone();
        a.negative = false;
        b.negative = false;
        while !b.is_zero() {
            let (_, r) = Self::div_rem_mag(&a, &b);
            a = b;
            b = r;
        }
        a.negative = false;
        a
    }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    neg: bool,
    mag: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;

    fn zero() -> Self {
        Self { neg: false, mag: Vec::new() }
    }

    fn clean(&mut self) {
        while self.mag.last() == Some(&0) {
            self.mag.pop();
        }
        if self.mag.is_empty() { self.neg = false; }
    }

    fn abs_cmp(&self, b: &Self) -> Ordering {
        self.mag.len().cmp(&b.mag.len()).then_with(|| {
            self.mag.iter().rev().cmp(b.mag.iter().rev())
        })
    }

    fn abs_add(a: &Self, b: &Self) -> Self {
        let mut m = Vec::with_capacity(a.mag.len().max(b.mag.len()) + 1);
        let mut c = 0u64;
        for i in 0..a.mag.len().max(b.mag.len()) {
            let v = a.mag.get(i).copied().unwrap_or(0) as u64
                + b.mag.get(i).copied().unwrap_or(0) as u64 + c;
            m.push((v % Self::BASE) as u32);
            c = v / Self::BASE;
        }
        if c != 0 { m.push(c as u32); }
        Self { neg: false, mag: m }
    }

    fn abs_sub(a: &Self, b: &Self) -> Self {
        let mut m = Vec::with_capacity(a.mag.len());
        let mut borrow = 0i64;
        for i in 0..a.mag.len() {
            let v = a.mag[i] as i64 - b.mag.get(i).copied().unwrap_or(0) as i64 - borrow;
            if v < 0 {
                m.push((v + Self::BASE as i64) as u32);
                borrow = 1;
            } else {
                m.push(v as u32);
                borrow = 0;
            }
        }
        let mut r = Self { neg: false, mag: m };
        r.clean();
        r
    }

    fn abs(&self) -> Self {
        let mut x = self.clone();
        x.neg = false;
        x
    }

    fn mul_small(&mut self, m: u32, a: u32) {
        let mut c = a as u64;
        for limb in &mut self.mag {
            let v = *limb as u64 * m as u64 + c;
            *limb = (v % Self::BASE) as u32;
            c = v / Self::BASE;
        }
        while c != 0 {
            self.mag.push((c % Self::BASE) as u32);
            c /= Self::BASE;
        }
    }

    fn div_small(&mut self, d: u32) -> u32 {
        let mut r = 0u64;
        for limb in self.mag.iter_mut().rev() {
            let v = r * Self::BASE + *limb as u64;
            *limb = (v / d as u64) as u32;
            r = v % d as u64;
        }
        self.clean();
        r as u32
    }

    fn abs_div_rem(a: &Self, b: &Self) -> (Self, Self) {
        let mut q = Self::zero();
        let mut r = Self::zero();
        if a.abs_cmp(b) == Ordering::Less { return (q, a.abs()); }
        let mut q_be = Vec::with_capacity(a.mag.len());
        for &limb in a.mag.iter().rev() {
            r.mag.insert(0, limb);
            r.clean();
            let mut low = 0u32;
            let mut high = (Self::BASE - 1) as u32;
            let mut digit = 0u32;
            while low <= high {
                let mid = low + (high - low) / 2;
                let mut t = b.clone();
                t.mul_small(mid, 0);
                match t.abs_cmp(&r) {
                    Ordering::Greater => {
                        if mid == 0 { break; }
                        high = mid - 1;
                    }
                    _ => {
                        digit = mid;
                        if mid == (Self::BASE - 1) as u32 { break; }
                        low = mid + 1;
                    }
                }
            }
            q_be.push(digit);
            if digit != 0 {
                let mut t = b.clone();
                t.mul_small(digit, 0);
                r = Self::abs_sub(&r, &t);
            }
        }
        q_be.reverse();
        q.mag = q_be;
        q.clean();
        (q, r)
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) { return Err(Error::InvalidRadix { radix }); }
        let s = text.as_bytes();
        if s.is_empty() { return Err(Error::Empty); }
        let neg = s[0] == b'-';
        let start = if s[0] == b'-' || s[0] == b'+' { 1 } else { 0 };
        if start == s.len() { return Err(Error::SignOnly); }

        let digit = |b: u8| -> Option<u32> {
            match b {
                b'0'..=b'9' => Some((b - b'0') as u32),
                b'a'..=b'z' => Some((b - b'a' + 10) as u32),
                b'A'..=b'Z' => Some((b - b'A' + 10) as u32),
                _ => None,
            }
        };
        for i in start..s.len() {
            if s[i] != b'_' && digit(s[i]).map_or(true, |d| d >= radix) {
                return Err(Error::InvalidDigit { index: i });
            }
        }
        for i in start..s.len() {
            if s[i] == b'_' && (i == start || i + 1 == s.len() || s[i - 1] == b'_' || s[i + 1] == b'_') {
                return Err(Error::Separator { index: i });
            }
        }
        let indices: Vec<usize> = (start..s.len()).filter(|&i| s[i] != b'_').collect();
        if indices.len() > 1 && digit(s[indices[0]]) == Some(0) {
            return Err(Error::LeadingZero { index: indices[0] });
        }
        if indices.len() > max_digits {
            return Err(Error::TooManyDigits { limit: max_digits, index: indices[max_digits] });
        }
        let mut n = Self::zero();
        for &i in &indices {
            n.mul_small(radix, digit(s[i]).unwrap());
        }
        n.neg = neg && !n.mag.is_empty();
        Ok(n)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) { return Err(Error::InvalidRadix { radix }); }
        let alphabet = b"0123456789abcdefghijklmnopqrstuvwxyz";
        let mut n = self.abs();
        let mut ds = Vec::new();
        if n.mag.is_empty() { ds.push(b'0'); }
        while !n.mag.is_empty() {
            ds.push(alphabet[n.div_small(radix) as usize]);
        }
        ds.reverse();
        if ds.len() < min_digits {
            let mut padded = vec![b'0'; min_digits - ds.len()];
            padded.extend(ds);
            ds = padded;
        }
        let mut out = String::new();
        if self.neg && !self.mag.is_empty() { out.push('-'); }
        for (i, &d) in ds.iter().enumerate() {
            if group != 0 && i != 0 && (ds.len() - i) % group == 0 { out.push('_'); }
            out.push(d as char);
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.neg, rhs.neg) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => rhs.abs_cmp(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.neg == rhs.neg {
            let mut x = Self::abs_add(self, rhs);
            x.neg = self.neg;
            x.clean();
            x
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut x = Self::abs_sub(self, rhs);
                    x.neg = self.neg;
                    x
                }
                Ordering::Less => {
                    let mut x = Self::abs_sub(rhs, self);
                    x.neg = rhs.neg;
                    x
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self { self.add(&rhs.neg()) }

    pub fn neg(&self) -> Self {
        let mut x = self.clone();
        if !x.mag.is_empty() { x.neg = !x.neg; }
        x
    }

    pub fn signum(&self) -> i8 {
        if self.mag.is_empty() { 0 } else if self.neg { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.mag.is_empty() || rhs.mag.is_empty() { return Self::zero(); }
        let mut acc = vec![0u64; self.mag.len() + rhs.mag.len() + 1];
        for (i, &a) in self.mag.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.mag.iter().enumerate() {
                let k = i + j;
                let v = acc[k] + a as u64 * b as u64 + carry;
                acc[k] = v % Self::BASE;
                carry = v / Self::BASE;
            }
            let mut k = i + rhs.mag.len();
            while carry != 0 {
                let v = acc[k] + carry;
                acc[k] = v % Self::BASE;
                carry = v / Self::BASE;
                k += 1;
            }
        }
        let mut x = Self { neg: self.neg ^ rhs.neg, mag: acc.into_iter().map(|v| v as u32).collect() };
        x.clean();
        x
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.mag.is_empty() { return Err(Error::DivisionByZero); }
        let (mut q, mut r) = Self::abs_div_rem(self, rhs);
        q.neg = !q.mag.is_empty() && (self.neg ^ rhs.neg);
        r.neg = !r.mag.is_empty() && self.neg;
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > 1024 {
            return Err(Error::ExponentTooLarge { exponent, limit: 1024 });
        }
        let mut result = Self { neg: false, mag: vec![1] };
        let mut base = self.clone();
        let mut e = exponent;
        while e > 0 {
            if e & 1 == 1 { result = result.mul(&base); }
            e >>= 1;
            if e > 0 { base = base.mul(&base); }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.mag.is_empty() {
            let r = a.div_rem(&b).unwrap().1.abs();
            a = b;
            b = r;
        }
        a.neg = false;
        a
    }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    limbs: Vec<u32>,
}

const BASE: u64 = 1_000_000_000;
const POW_LIMIT: u32 = 1024;

impl Big {
    fn zero() -> Self {
        Self { negative: false, limbs: vec![0] }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        self.limbs.len().cmp(&rhs.limbs.len()).then_with(|| {
            for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                match a.cmp(b) {
                    Ordering::Equal => continue,
                    other => return other,
                }
            }
            Ordering::Equal
        })
    }

    fn add_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let value = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64 + carry;
            out.push((value % BASE) as u32);
            carry = value / BASE;
        }
        if carry > 0 {
            out.push(carry as u32);
        }
        out
    }

    fn sub_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let value = a[i] as i64 - b.get(i).copied().unwrap_or(0) as i64 - borrow;
            if value < 0 {
                out.push((value + BASE as i64) as u32);
                borrow = 1;
            } else {
                out.push(value as u32);
                borrow = 0;
            }
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_small_abs(&mut self, n: u32) {
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = *limb as u64 * n as u64 + carry;
            *limb = (value % BASE) as u32;
            carry = value / BASE;
        }
        while carry > 0 {
            self.limbs.push((carry % BASE) as u32);
            carry /= BASE;
        }
    }

    fn add_small_abs(&mut self, n: u32) {
        let mut carry = n as u64;
        let mut i = 0;
        while carry > 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let value = self.limbs[i] as u64 + carry;
            self.limbs[i] = (value % BASE) as u32;
            carry = value / BASE;
            i += 1;
        }
    }

    fn div_rem_small_abs(&self, divisor: u32) -> (Self, u32) {
        let mut quotient = vec![0; self.limbs.len()];
        let mut remainder = 0u64;
        for i in (0..self.limbs.len()).rev() {
            let value = remainder * BASE + self.limbs[i] as u64;
            quotient[i] = (value / divisor as u64) as u32;
            remainder = value % divisor as u64;
        }
        let mut q = Self { negative: false, limbs: quotient };
        q.normalize();
        (q, remainder as u32)
    }

    fn mul_abs(&self, rhs: &Self) -> Self {
        let mut out = vec![0u32; self.limbs.len() + rhs.limbs.len() + 1];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let value = out[k] as u64 + a as u64 * b as u64 + carry;
                out[k] = (value % BASE) as u32;
                carry = value / BASE;
            }
            let mut k = i + rhs.limbs.len();
            while carry > 0 {
                let value = out[k] as u64 + carry;
                out[k] = (value % BASE) as u32;
                carry = value / BASE;
                k += 1;
            }
        }
        let mut result = Self { negative: false, limbs: out };
        result.normalize();
        result
    }

    fn mul_small_value(&self, n: u64) -> Self {
        let mut result = self.clone();
        result.negative = false;
        let mut carry = 0u128;
        for limb in &mut result.limbs {
            let value = *limb as u128 * n as u128 + carry;
            *limb = (value % BASE as u128) as u32;
            carry = value / BASE as u128;
        }
        while carry > 0 {
            result.limbs.push((carry % BASE as u128) as u32);
            carry /= BASE as u128;
        }
        result.normalize();
        result
    }

    fn div_rem_abs(&self, rhs: &Self) -> (Self, Self) {
        let mut quotient = vec![0u32; self.limbs.len()];
        let mut remainder = Self::zero();
        for i in (0..self.limbs.len()).rev() {
            remainder.limbs.insert(0, self.limbs[i]);
            remainder.normalize();
            let mut low = 0u64;
            let mut high = BASE - 1;
            while low < high {
                let mid = (low + high + 1) / 2;
                let product = rhs.mul_small_value(mid);
                if product.cmp_abs(&remainder) != Ordering::Greater {
                    low = mid;
                } else {
                    high = mid - 1;
                }
            }
            quotient[i] = low as u32;
            if low != 0 {
                let product = rhs.mul_small_value(low);
                remainder = Self {
                    negative: false,
                    limbs: Self::sub_abs(&remainder.limbs, &product.limbs),
                };
                remainder.normalize();
            }
        }
        let mut q = Self { negative: false, limbs: quotient };
        q.normalize();
        remainder.normalize();
        (q, remainder)
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }
        let bytes = text.as_bytes();
        let (start, negative) = match bytes[0] {
            b'+' => (1, false),
            b'-' => (1, true),
            _ => (0, false),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let invalid = bytes.iter().enumerate().skip(start).find_map(|(i, &b)| {
            if b == b'_' {
                None
            } else {
                let d = match b {
                    b'0'..=b'9' => (b - b'0') as u32,
                    b'a'..=b'z' => (b - b'a') as u32 + 10,
                    b'A'..=b'Z' => (b - b'A') as u32 + 10,
                    _ => radix,
                };
                if d >= radix { Some(i) } else { None }
            }
        });
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        if let Some(index) = (start..bytes.len()).find(|&i| {
            bytes[i] == b'_'
                && (i == start || i + 1 == bytes.len()
                    || bytes[i - 1] == b'_' || bytes[i + 1] == b'_')
        }) {
            return Err(Error::Separator { index });
        }

        let positions: Vec<usize> = (start..bytes.len()).filter(|&i| bytes[i] != b'_').collect();
        if positions.len() > 1 && bytes[positions[0]] == b'0' {
            return Err(Error::LeadingZero { index: positions[0] });
        }
        if positions.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: positions[max_digits],
            });
        }

        let mut result = Self::zero();
        for &b in &bytes[start..] {
            if b == b'_' { continue; }
            let digit = match b {
                b'0'..=b'9' => (b - b'0') as u32,
                b'a'..=b'z' => (b - b'a') as u32 + 10,
                b'A'..=b'Z' => (b - b'A') as u32 + 10,
                _ => unreachable!(),
            };
            result.mul_small_abs(radix);
            result.add_small_abs(digit);
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let mut n = self.clone();
        n.negative = false;
        let mut digits = Vec::new();
        while !n.is_zero() {
            let (q, r) = n.div_rem_small_abs(radix);
            digits.push(r);
            n = q;
        }
        if digits.is_empty() {
            digits.push(0);
        }
        while digits.len() < min_digits {
            digits.push(0);
        }
        digits.reverse();

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, &d) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(if d < 10 { (b'0' + d as u8) as char } else { (b'a' + (d - 10) as u8) as char });
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => self.cmp_abs(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self { negative: self.negative, limbs: Self::add_abs(&self.limbs, &rhs.limbs) };
            out.normalize();
            out
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self { negative: self.negative, limbs: Self::sub_abs(&self.limbs, &rhs.limbs) };
                    out.normalize();
                    out
                }
                Ordering::Less => {
                    let mut out = Self { negative: rhs.negative, limbs: Self::sub_abs(&rhs.limbs, &self.limbs) };
                    out.normalize();
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut out = self.mul_abs(rhs);
        out.negative = self.negative != rhs.negative;
        out.normalize();
        out
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut q, mut r) = self.div_rem_abs(rhs);
        q.negative = !q.is_zero() && (self.negative != rhs.negative);
        r.negative = !r.is_zero() && self.negative;
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > POW_LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: POW_LIMIT });
        }
        let mut power = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while power > 0 {
            if power & 1 == 1 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power > 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.clone();
        let mut b = rhs.clone();
        a.negative = false;
        b.negative = false;
        while !b.is_zero() {
            let (_, r) = a.div_rem_abs(&b);
            a = b;
            b = r;
        }
        a.negative = false;
        a.normalize();
        a
    }
}
