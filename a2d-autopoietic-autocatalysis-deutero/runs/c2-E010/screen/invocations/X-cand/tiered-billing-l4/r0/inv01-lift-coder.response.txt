- Tier schedules are fully validated before quantity handling; the earliest scan error wins over a missing final tier.
- A finite tier covers units after the preceding bound up to and including its own bound; a tier is charged only when it receives positive units.
- Overflow is checked at each used band, including its start charge, before minimums or discounts can reduce an amount.
- Credit ranges must be valid and ordered within the service interval; proration rounds once after subtracting all credits.
- Names are checked byte-for-byte in input order before pricing; successful invoice lines sort by UTF-8 bytes.
- Percentage discounts use exact integer arithmetic; rendered output uses the specified LF-terminated format.

pub enum Error { Empty, TooMany, Bound { index: usize }, Unbounded { index: usize }, MissingFinal, Overflow { index: usize }, Month, Period, Credit { index: usize }, CreditOrder { index: usize }, InvoiceDiscount, InvalidName { index: usize }, DuplicateName { index: usize }, Discount, CombinedOverflow, Meter { index: usize, source: Box<Error> }, InvoiceOverflow }

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() { return Err(Error::Empty); }
    if tiers.len() > 64 { return Err(Error::TooMany); }
    let mut previous = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous { return Err(Error::Bound { index }); }
                previous = bound;
            }
            None => {
                if index + 1 != tiers.len() { return Err(Error::Unbounded { index }); }
            }
        }
    }
    if tiers.last().unwrap().up_to.is_some() { return Err(Error::MissingFinal); }
    Ok(())
}

fn band_amount(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    let product = (units as u128) * (unit as u128);
    let amount = product + (start as u128);
    if amount > u64::MAX as u128 { Err(Error::Overflow { index }) }
    else { Ok(amount as u64) }
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 { return Ok(Charge { amount: 0, bands: Vec::new() }); }
            let amount = band_amount(quantity, *unit, *start, 0)?;
            Ok(Charge { amount, bands: vec![Band { index: 0, units: quantity, amount }] })
        }
        Price::Tiers(tiers) => {
            validate_tiers(tiers)?;
            if quantity == 0 { return Ok(Charge { amount: 0, bands: Vec::new() }); }
            let mut bands = Vec::new();
            let mut total = 0u64;
            let mut previous = 0u64;
            let mut remaining = quantity;
            for (index, tier) in tiers.iter().enumerate() {
                let allocated = match tier.up_to {
                    Some(bound) => {
                        let width = bound - previous;
                        previous = bound;
                        remaining.min(width)
                    }
                    None => remaining,
                };
                if allocated > 0 {
                    let amount = band_amount(allocated, tier.unit, tier.start, index)?;
                    total = total.checked_add(amount).ok_or(Error::Overflow { index })?;
                    bands.push(Band { index, units: allocated, amount });
                    remaining -= allocated;
                }
                if remaining == 0 { break; }
            }
            Ok(Charge { amount: total, bands })
        }
    }
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Tiers(tiers), Mode::Volume) => {
            validate_tiers(tiers)?;
            if quantity == 0 {
                Charge { amount: 0, bands: Vec::new() }
            } else {
                let (index, tier) = tiers.iter().enumerate()
                    .find(|(_, t)| t.up_to.map_or(true, |bound| quantity <= bound))
                    .unwrap_or((tiers.len() - 1, tiers.last().unwrap()));
                let amount = band_amount(quantity, tier.unit, tier.start, index)?;
                Charge { amount, bands: vec![Band { index, units: quantity, amount }] }
            }
        }
        _ => quote(quantity, price)?,
    };
    let minimum_adjustment = if quantity > 0 && minimum > charge.amount { minimum - charge.amount } else { 0 };
    let total = charge.amount.checked_add(minimum_adjustment).ok_or(Error::CombinedOverflow)?;
    Ok(Bill { charge, minimum_adjustment, total })
}

pub fn prorate(fee: u64, days: u32, start: u32, end: u32, credits: &[DayRange], rounding: Rounding) -> Result<Proration, Error> {
    if !(28..=31).contains(&days) { return Err(Error::Month); }
    if start > end || end > days { return Err(Error::Period); }
    let mut previous_end = start;
    let mut excluded = 0u32;
    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < previous_end { return Err(Error::CreditOrder { index }); }
        previous_end = credit.end;
        excluded += credit.end - credit.start;
    }
    let active_days = (end - start) - excluded;
    let numerator = (fee as u128) * (active_days as u128);
    let denominator = days as u128;
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let rounded = match rounding {
        Rounding::Down => quotient,
        Rounding::Up => quotient + u128::from(remainder != 0),
        Rounding::HalfEven => {
            let twice = remainder * 2;
            if twice > denominator || (twice == denominator && quotient % 2 == 1) { quotient + 1 } else { quotient }
        }
    };
    Ok(Proration { active_days, amount: rounded as u64 })
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    (1..=32).contains(&bytes.len()) &&
        !bytes.iter().any(|b| *b <= 0x1f || *b == 0x7f || *b == b'|' || *b == b':')
}

fn discount(base: u64, bps: u32) -> u64 {
    ((base as u128) * (bps as u128) / 10_000u128) as u64
}

fn meter_error(index: usize, source: Error) -> Error {
    Error::Meter { index, source: Box::new(source) }
}

pub fn invoice(meters: &[Meter], invoice_discount_bps: u32, credit: u64) -> Result<Invoice, Error> {
    if invoice_discount_bps > 10_000 { return Err(Error::InvoiceDiscount); }

    for (index, meter) in meters.iter().enumerate() {
        if !valid_name(&meter.name) { return Err(Error::InvalidName { index }); }
        if meters[..index].iter().any(|earlier| earlier.name == meter.name) {
            return Err(Error::DuplicateName { index });
        }
    }

    let mut lines = Vec::with_capacity(meters.len());
    let mut subtotal = 0u64;
    for (index, meter) in meters.iter().enumerate() {
        if meter.discount_bps > 10_000 { return Err(meter_error(index, Error::Discount)); }
        let b = bill(meter.quantity, &meter.price, meter.mode.clone(), meter.minimum)
            .map_err(|e| meter_error(index, e))?;
        let p = prorate(meter.monthly_fee, meter.days, meter.start, meter.end, &meter.credits, meter.rounding.clone())
            .map_err(|e| meter_error(index, e))?;
        let base = b.total.checked_add(p.amount).ok_or_else(|| meter_error(index, Error::CombinedOverflow))?;
        let meter_discount = discount(base, meter.discount_bps);
        let total = base - meter_discount;
        subtotal = subtotal.checked_add(total).ok_or(Error::InvoiceOverflow)?;
        lines.push(InvoiceLine {
            name: meter.name.clone(),
            usage: b.charge.amount,
            minimum_adjustment: b.minimum_adjustment,
            monthly: p.amount,
            discount: meter_discount,
            total,
        });
    }

    lines.sort_by(|a, b| a.name.as_bytes().cmp(b.name.as_bytes()));
    let invoice_discount = discount(subtotal, invoice_discount_bps);
    let post_discount = subtotal - invoice_discount;
    let applied_credit = credit.min(post_discount);
    let total = post_discount - applied_credit;
    let mut rendered = String::new();
    for line in &lines {
        rendered.push_str(&line.name.as_bytes().len().to_string());
        rendered.push(':');
        rendered.push_str(&line.name);
        rendered.push('|');
        rendered.push_str(&line.usage.to_string());
        rendered.push('|');
        rendered.push_str(&line.minimum_adjustment.to_string());
        rendered.push('|');
        rendered.push_str(&line.monthly.to_string());
        rendered.push('|');
        rendered.push_str(&line.discount.to_string());
        rendered.push('|');
        rendered.push_str(&line.total.to_string());
        rendered.push('\n');
    }
    rendered.push_str("TOTAL|");
    rendered.push_str(&subtotal.to_string());
    rendered.push('|');
    rendered.push_str(&invoice_discount.to_string());
    rendered.push('|');
    rendered.push_str(&applied_credit.to_string());
    rendered.push('|');
    rendered.push_str(&total.to_string());
    rendered.push('\n');

    Ok(Invoice { lines, subtotal, discount: invoice_discount, credit: applied_credit, total, rendered })
}

pub enum Error { Empty, TooMany, Bound { index: usize }, Unbounded { index: usize }, MissingFinal, Overflow { index: usize }, Month, Period, Credit { index: usize }, CreditOrder { index: usize }, InvoiceDiscount, InvalidName { index: usize }, DuplicateName { index: usize }, Discount, CombinedOverflow, Meter { index: usize, source: Box<Error> }, InvoiceOverflow }

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    match tiers.len() {
        0 => return Err(Error::Empty),
        65.. => return Err(Error::TooMany),
        _ => {}
    }
    let mut prev: Option<u64> = Some(0);
    for (i, t) in tiers.iter().enumerate() {
        match t.up_to {
            Some(bound) => {
                let p = prev.unwrap();
                if bound <= p { return Err(Error::Bound { index: i }); }
                prev = Some(bound);
            }
            None => {
                if i + 1 < tiers.len() { return Err(Error::Unbounded { index: i }); }
                prev = None;
            }
        }
    }
    if prev.is_some() { Err(Error::MissingFinal) } else { Ok(()) }
}

fn compute_band(units: u64, tier: &Tier, index: usize) -> Result<u64, Error> {
    let value = (units as u128) * (tier.unit as u128) + tier.start as u128;
    u64::try_from(value).map_err(|_| Error::Overflow { index })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    let mut bands = Vec::new();
    let mut sum = 0u64;
    match price {
        Price::Flat { unit, start } => {
            if quantity > 0 {
                let product = quantity as u128 * *unit as u128 + *start as u128;
                let amt = u64::try_from(product).map_err(|_| Error::Overflow { index: 0 })?;
                bands.push(Band { index: 0, units: quantity, amount: amt });
                sum = amt;
            }
        }
        Price::Tiers(ts) => {
            validate_tiers(ts)?;
            let mut remaining = quantity;
            let mut lower = 0u64;
            for (i, t) in ts.iter().enumerate() {
                if remaining == 0 { break; }
                let units = match t.up_to {
                    Some(upper) => {
                        let width = upper - lower;
                        lower = upper;
                        remaining.min(width)
                    }
                    None => remaining,
                };
                if units != 0 {
                    let amt = compute_band(units, t, i)?;
                    sum = sum.checked_add(amt).ok_or(Error::Overflow { index: i })?;
                    bands.push(Band { index: i, units, amount: amt });
                    remaining -= units;
                }
            }
        }
    }
    Ok(Charge { amount: sum, bands })
}

fn volume_charge(quantity: u64, ts: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(ts)?;
    if quantity == 0 { return Ok(Charge { amount: 0, bands: Vec::new() }); }
    let i = ts.iter().position(|t| t.up_to.map_or(true, |u| quantity <= u)).unwrap();
    let t = &ts[i];
    let amt = compute_band(quantity, t, i)?;
    Ok(Charge { amount: amt, bands: vec![Band { index: i, units: quantity, amount: amt }] })
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Tiers(ts), Mode::Volume) => volume_charge(quantity, ts)?,
        (_, _) => quote(quantity, price)?,
    };
    let minimum_adjustment = if quantity > 0 { minimum.saturating_sub(charge.amount) } else { 0 };
    let total = charge.amount.checked_add(minimum_adjustment).ok_or(Error::CombinedOverflow)?;
    Ok(Bill { charge, minimum_adjustment, total })
}

pub fn prorate(fee: u64, days: u32, start: u32, end: u32, credits: &[DayRange], rounding: Rounding) -> Result<Proration, Error> {
    if days != 28 && days != 29 && days != 30 && days != 31 { return Err(Error::Month); }
    if start > end || end > days { return Err(Error::Period); }
    let mut prev_end = start;
    let mut active = end - start;
    for (i, c) in credits.iter().enumerate() {
        if c.start < start || c.start >= c.end || c.end > end { return Err(Error::Credit { index: i }); }
        if c.start < prev_end { return Err(Error::CreditOrder { index: i }); }
        prev_end = c.end;
        active -= c.end - c.start;
    }
    let n = fee as u128 * active as u128;
    let d = days as u128;
    let q = n / d;
    let r = n % d;
    let rounded = match rounding {
        Rounding::Down => q,
        Rounding::Up => q + if r == 0 { 0 } else { 1 },
        Rounding::HalfEven => {
            if r * 2 > d || (r * 2 == d && q & 1 == 1) { q + 1 } else { q }
        }
    };
    Ok(Proration { active_days: active, amount: rounded as u64 })
}

fn name_ok(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty() && b.len() <= 32 && !b.iter().any(|x| *x <= 31 || *x == 127 || *x == b'|' || *x == b':')
}

fn pct(x: u64, bps: u32) -> u64 {
    ((x as u128 * bps as u128) / 10000) as u64
}

pub fn invoice(meters: &[Meter], invoice_discount_bps: u32, credit: u64) -> Result<Invoice, Error> {
    if invoice_discount_bps > 10000 { return Err(Error::InvoiceDiscount); }
    for (i, m) in meters.iter().enumerate() {
        if !name_ok(&m.name) { return Err(Error::InvalidName { index: i }); }
        if meters[..i].iter().any(|x| x.name == m.name) { return Err(Error::DuplicateName { index: i }); }
    }
    let mut lines = Vec::with_capacity(meters.len());
    let mut subtotal = 0u64;
    for (i, m) in meters.iter().enumerate() {
        if m.discount_bps > 10000 { return Err(Error::Meter { index: i, source: Box::new(Error::Discount) }); }
        let b = bill(m.quantity, &m.price, m.mode.clone(), m.minimum)
            .map_err(|e| Error::Meter { index: i, source: Box::new(e) })?;

            .map_err(|e| Error::Meter { index: i, source: Box::new(e) })?;
        let base = b.total.checked_add(p.amount)
            .ok_or_else(|| Error::Meter { index: i, source: Box::new(Error::CombinedOverflow) })?;
        let d = pct(base, m.discount_bps);
        let total = base - d;
        subtotal = subtotal.checked_add(total).ok_or(Error::InvoiceOverflow)?;
        lines.push(InvoiceLine { name: m.name.clone(), usage: b.charge.amount, minimum_adjustment: b.minimum_adjustment, monthly: p.amount, discount: d, total });
    }
