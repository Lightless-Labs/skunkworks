fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() { return Err(Error::Empty); }
    if tiers.len() > 64 { return Err(Error::TooMany); }
    let mut previous = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous { return Err(Error::Bound { index }); }
                previous = bound;
            }
            None => {
                if index + 1 != tiers.len() { return Err(Error::Unbounded { index }); }
            }
        }
    }
    if tiers.last().and_then(|t| t.up_to).is_some() { return Err(Error::MissingFinal); }
    Ok(())
}

fn band_amount(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    let amount = (units as u128) * (unit as u128) + (start as u128);
    if amount > u64::MAX as u128 { Err(Error::Overflow { index }) } else { Ok(amount as u64) }
}

fn graduated(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;
    let mut remaining = quantity;
    let mut previous = 0u64;
    let mut amount = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous;
                previous = bound;
                remaining.min(width)
            }
            None => remaining,
        };
        if units > 0 {
            let band = band_amount(units, tier.unit, tier.start, index)?;
            let sum = (amount as u128) + (band as u128);
            if sum > u64::MAX as u128 { return Err(Error::Overflow { index }); }
            amount = sum as u64;
            bands.push(Band { index, units, amount: band });
            remaining -= units;
        }
    }
    Ok(Charge { amount, bands })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 { return Ok(Charge { amount: 0, bands: Vec::new() }); }
            let amount = band_amount(quantity, *unit, *start, 0)?;
            Ok(Charge { amount, bands: vec![Band { index: 0, units: quantity, amount }] })
        }
        Price::Tiers(tiers) => graduated(quantity, tiers),
    }
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Tiers(tiers), Mode::Volume) => {
            validate_tiers(tiers)?;
            if quantity == 0 {
                Charge { amount: 0, bands: Vec::new() }
            } else {
                let index = tiers.iter().position(|t| t.up_to.map_or(true, |bound| quantity <= bound)).unwrap();
                let tier = &tiers[index];
                let amount = band_amount(quantity, tier.unit, tier.start, index)?;
                Charge { amount, bands: vec![Band { index, units: quantity, amount }] }
            }
        }
        _ => quote(quantity, price)?,
    };

    let total = (charge.amount as u128) + (minimum_adjustment as u128);
    if total > u64::MAX as u128 { return Err(Error::Overflow { index: charge.bands.last().map_or(0, |b| b.index) }); }
    Ok(Bill { charge, minimum_adjustment, total: total as u64 })
}

pub fn prorate(fee: u64, days: u32, start: u32, end: u32, credits: &[DayRange], rounding: Rounding) -> Result<Proration, Error> {
    if !(28..=31).contains(&days) { return Err(Error::Month); }
    if start > end || end > days { return Err(Error::Period); }
    let mut previous_end = start;
    let mut removed = 0u32;
    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < previous_end { return Err(Error::CreditOrder { index }); }
        removed += credit.end - credit.start;
        previous_end = credit.end;
    }
    let active_days = end - start - removed;
    let numerator = (fee as u128) * (active_days as u128);
    let denominator = days as u128;
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let rounded = match rounding {
        Rounding::Down => quotient,
        Rounding::Up => quotient + u128::from(remainder != 0),
        Rounding::HalfEven => {
            if remainder * 2 > denominator || (remainder * 2 == denominator && quotient % 2 == 1) { quotient + 1 } else { quotient }
        }
    };
    Ok(Proration { active_days, amount: rounded as u64 })
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty() && bytes.len() <= 32 &&
        !bytes.iter().any(|&b| b <= 0x1f || b == 0x7f || b == b'|' || b == b':')
}

fn percent(base: u64, bps: u32) -> u64 {
    ((base as u128) * (bps as u128) / 10_000u128) as u64
}

pub fn invoice(meters: &[Meter], invoice_discount_bps: u32, credit: u64) -> Result<Invoice, Error> {
    if invoice_discount_bps > 10_000 { return Err(Error::InvoiceDiscount); }
    for (index, meter) in meters.iter().enumerate() {
        if !valid_name(&meter.name) { return Err(Error::InvalidName { index }); }
        if meters[..index].iter().any(|earlier| earlier.name == meter.name) {
            return Err(Error::DuplicateName { index });
        }
    }

    let mut lines = Vec::with_capacity(meters.len());
    let mut subtotal = 0u64;
    for (index, meter) in meters.iter().enumerate() {
        let result = (|| {
            if meter.discount_bps > 10_000 { return Err(Error::Discount); }
            let bill = bill(meter.quantity, &meter.price, meter.mode.clone(), meter.minimum)?;
            let monthly = prorate(meter.monthly_fee, meter.days, meter.start, meter.end, &meter.credits, meter.rounding.clone())?.amount;
            let base = (bill.total as u128) + (monthly as u128);
            if base > u64::MAX as u128 { return Err(Error::CombinedOverflow); }
            let base = base as u64;
            let discount = percent(base, meter.discount_bps);
            Ok(InvoiceLine {
                name: meter.name.clone(),
                usage: bill.charge.amount,
                minimum_adjustment: bill.minimum_adjustment,
                monthly,
                discount,
                total: base - discount,
            })
        })().map_err(|source| Error::Meter { index, source: Box::new(source) })?;
        let sum = (subtotal as u128) + (result.total as u128);
        if sum > u64::MAX as u128 { return Err(Error::InvoiceOverflow); }
        subtotal = sum as u64;
        lines.push(result);
    }

    lines.sort_by(|a, b| a.name.as_bytes().cmp(b.name.as_bytes()));
    let discount = percent(subtotal, invoice_discount_bps);
    let credit = credit.min(subtotal - discount);
    let total = subtotal - discount - credit;
    let mut rendered = String::new();
    for line in &lines {
        use std::fmt::Write;
        writeln!(&mut rendered, "{}:{}|{}|{}|{}|{}|{}", line.name.len(), line.name, line.usage, line.minimum_adjustment, line.monthly, line.discount, line.total).unwrap();
    }
    use std::fmt::Write;
    writeln!(&mut rendered, "TOTAL|{}|{}|{}|{}", subtotal, discount, credit, total).unwrap();
    Ok(Invoice { lines, subtotal, discount, credit, total, rendered })
}

        _ => {}

    let mut last_bound = 0;
    for (index, tier) in tiers.iter().enumerate() {
        if let Some(bound) = tier.up_to {
            if bound <= last_bound {
                return Err(Error::Bound { index });
            }
            last_bound = bound;
        } else if index != tiers.len() - 1 {
            return Err(Error::Unbounded { index });
        }
    }
    if tiers[tiers.len() - 1].up_to.is_some() {
        Err(Error::MissingFinal)
    } else {
        Ok(())
    }
}

fn calculate(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    let wide = u128::from(units) * u128::from(unit) + u128::from(start);
    u64::try_from(wide).map_err(|_| Error::Overflow { index })
}

fn quote_impl(quantity: u64, price: &Price, mode: &Mode) -> Result<Charge, Error> {
    if let Price::Flat { unit, start } = price {
        return if quantity == 0 {
            Ok(Charge { amount: 0, bands: Vec::new() })
        } else {
            let amount = calculate(quantity, *unit, *start, 0)?;
            Ok(Charge { amount, bands: vec![Band { index: 0, units: quantity, amount }] })
        };
    }

    let tiers = match price { Price::Tiers(tiers) => tiers, _ => unreachable!() };
    validate_tiers(tiers)?;
    if quantity == 0 {
        return Ok(Charge { amount: 0, bands: Vec::new() });
    }

    if *mode == Mode::Volume {
        let selected = tiers.iter().enumerate().find(|(_, t)| {
            t.up_to.map(|bound| quantity <= bound).unwrap_or(true)
        }).unwrap();
        let (index, tier) = selected;
        let amount = calculate(quantity, tier.unit, tier.start, index)?;
        return Ok(Charge { amount, bands: vec![Band { index, units: quantity, amount }] });
    }

    let mut consumed_through = 0u64;
    let mut result = Charge { amount: 0, bands: Vec::new() };
    for (index, tier) in tiers.iter().enumerate() {
        let allocated = match tier.up_to {
            Some(bound) => {
                let width = bound - consumed_through;
                consumed_through = bound;
                quantity.saturating_sub(consumed_through.saturating_sub(width)).min(width)
            }
            None => quantity.saturating_sub(consumed_through),
        };
        if allocated == 0 { continue; }
        let amount = calculate(allocated, tier.unit, tier.start, index)?;
        let sum = u128::from(result.amount) + u128::from(amount);
        if sum > u128::from(u64::MAX) {
            return Err(Error::Overflow { index });
        }
        result.amount = sum as u64;
        result.bands.push(Band { index, units: allocated, amount });
    }
    Ok(result)
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    quote_impl(quantity, price, &Mode::Graduated)
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = quote_impl(quantity, price, &mode)?;
    let minimum_adjustment = if quantity == 0 || charge.amount >= minimum { 0 } else { minimum - charge.amount };
    let total = u128::from(charge.amount) + u128::from(minimum_adjustment);
    if total > u128::from(u64::MAX) {
        return Err(Error::Overflow { index: 0 });
    }
    Ok(Bill { charge, minimum_adjustment, total: total as u64 })
}

pub fn prorate(fee: u64, days: u32, start: u32, end: u32, credits: &[DayRange], rounding: Rounding) -> Result<Proration, Error> {
    if days != 28 && days != 29 && days != 30 && days != 31 { return Err(Error::Month); }
    if start > end || end > days { return Err(Error::Period); }

    let mut active_days = end - start;
    let mut previous_end = start;
    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < previous_end {
            return Err(Error::CreditOrder { index });
        }
        active_days -= credit.end - credit.start;
        previous_end = credit.end;
    }

    let numerator = u128::from(fee) * u128::from(active_days);
    let denominator = u128::from(days);
    let q = numerator / denominator;
    let r = numerator % denominator;
    let amount = match rounding {
        Rounding::Down => q,
        Rounding::Up => if r == 0 { q } else { q + 1 },
        Rounding::HalfEven => {
            let twice = r * 2;
            if twice > denominator || (twice == denominator && q % 2 == 1) { q + 1 } else { q }
        }
    };
    Ok(Proration { active_days, amount: amount as u64 })
}

fn discount(base: u64, bps: u32) -> u64 {
    (u128::from(base) * u128::from(bps) / 10_000) as u64
}

pub fn invoice(meters: &[Meter], invoice_discount_bps: u32, credit: u64) -> Result<Invoice, Error> {
    if invoice_discount_bps > 10_000 { return Err(Error::InvoiceDiscount); }

    let mut seen: Vec<&str> = Vec::with_capacity(meters.len());
    for (index, meter) in meters.iter().enumerate() {
        let bytes = meter.name.as_bytes();
        if bytes.is_empty() || bytes.len() > 32 || bytes.iter().any(|&b| b <= 31 || b == 127 || b == b'|' || b == b':') {
            return Err(Error::InvalidName { index });
        }
        if seen.iter().any(|name| *name == meter.name) {
            return Err(Error::DuplicateName { index });
        }
        seen.push(&meter.name);
    }

    let mut subtotal = 0u64;
    let mut lines = Vec::with_capacity(meters.len());
    for (index, meter) in meters.iter().enumerate() {
        let line = (|| {
            if meter.discount_bps > 10_000 { return Err(Error::Discount); }
            let bill = bill(meter.quantity, &meter.price, meter.mode.clone(), meter.minimum)?;
            let proration = prorate(meter.monthly_fee, meter.days, meter.start, meter.end, &meter.credits, meter.rounding.clone())?;
            let base = bill.total.checked_add(proration.amount).ok_or(Error::CombinedOverflow)?;
            let meter_discount = discount(base, meter.discount_bps);
            Ok(InvoiceLine {
                name: meter.name.clone(),
                usage: bill.charge.amount,
                minimum_adjustment: bill.minimum_adjustment,
                monthly: proration.amount,
                discount: meter_discount,
                total: base - meter_discount,
            })
        })().map_err(|source| Error::Meter { index, source: Box::new(source) })?;

        subtotal = subtotal.checked_add(line.total).ok_or(Error::InvoiceOverflow)?;
        lines.push(line);
    }

    lines.sort_by(|a, b| a.name.as_bytes().cmp(b.name.as_bytes()));
    let invoice_discount = discount(subtotal, invoice_discount_bps);
    let remainder = subtotal - invoice_discount;
    let applied_credit = credit.min(remainder);
    let total = remainder - applied_credit;

    let mut rendered = String::new();
    for line in &lines {
        rendered.push_str(&line.name.len().to_string());
        rendered.push(':');
        rendered.push_str(&line.name);
        rendered.push('|');
        rendered.push_str(&line.usage.to_string());
        rendered.push('|');
        rendered.push_str(&line.minimum_adjustment.to_string());
        rendered.push('|');
        rendered.push_str(&line.monthly.to_string());
        rendered.push('|');
        rendered.push_str(&line.discount.to_string());
        rendered.push('|');
        rendered.push_str(&line.total.to_string());
        rendered.push('\n');
    }
    rendered.push_str("TOTAL|");
    rendered.push_str(&subtotal.to_string());
    rendered.push('|');
    rendered.push_str(&invoice_discount.to_string());
    rendered.push('|');
    rendered.push_str(&applied_credit.to_string());
    rendered.push('|');
    rendered.push_str(&total.to_string());
    rendered.push('\n');

    Ok(Invoice { lines, subtotal, discount: invoice_discount, credit: applied_credit, total, rendered })
}

pub fn prorate(fee: u64, days: u32, start: u32, end: u32, credits: &[DayRange], rounding: Rounding) -> Result<Proration, Error> {

    if start > end || end > days { return Err(Error::Period); }
    let mut previous_end = start;
    let mut excluded = 0u32;

            return Err(Error::Credit { index });

        previous_end = credit.end;
        excluded += credit.end - credit.start;

            if twice > denominator || (twice == denominator && quotient % 2 == 1) { quotient + 1 } else { quotient }
        }
    };
    Ok(Proration { active_days, amount: rounded as u64 })
}

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {
    if days != 28 && days != 29 && days != 30 && days != 31 {
        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);
    }

    let mut active_days = end - start;
    let mut prior_end = start;
    for (index, range) in credits.iter().enumerate() {
        if range.start < start || range.start >= range.end || range.end > end {
            return Err(Error::Credit { index });
        }
        if range.start < prior_end {
            return Err(Error::CreditOrder { index });
        }
        active_days -= range.end - range.start;
        prior_end = range.end;
    }

    let product = u128::from(fee) * u128::from(active_days);
    let denominator = u128::from(days);
    let quotient = product / denominator;
    let remainder = product % denominator;

    let rounded = match rounding {
        Rounding::Down => quotient,
        Rounding::Up if remainder != 0 => quotient + 1,
        Rounding::Up => quotient,
        Rounding::HalfEven => {
            match (remainder * 2).cmp(&denominator) {
                std::cmp::Ordering::Less => quotient,
                std::cmp::Ordering::Greater => quotient + 1,
                std::cmp::Ordering::Equal if quotient % 2 == 1 => quotient + 1,
                std::cmp::Ordering::Equal => quotient,
            }
        }
    };

    Ok(Proration { active_days, amount: rounded as u64 })
}

        *b <= 0x1f || *b == 0x7f || *b == b'|' || *b == b':'
    })
}

fn discount(base: u64, bps: u32) -> u64 {

pub fn invoice(meters: &[Meter], invoice_discount_bps: u32, credit: u64) -> Result<Invoice, Error> {
    if invoice_discount_bps > 10_000 {
        return Err(Error::InvoiceDiscount);
    }

    for (index, meter) in meters.iter().enumerate() {
        if !valid_name(&meter.name) {
            return Err(Error::InvalidName { index });
        }
        if meters[..index].iter().any(|previous| previous.name == meter.name) {
            return Err(Error::DuplicateName { index });
        }
    }

    let mut lines = Vec::with_capacity(meters.len());
    let mut subtotal = 0u64;

    for (index, meter) in meters.iter().enumerate() {
        let line = (|| {
            if meter.discount_bps > 10_000 {
                return Err(Error::Discount);
            }
            let bill = bill(meter.quantity, &meter.price, meter.mode.clone(), meter.minimum)?;
            let proration = prorate(
                meter.monthly_fee, meter.days, meter.start, meter.end,
                &meter.credits, meter.rounding.clone(),
            )?;
            let base = u128::from(bill.total) + u128::from(proration.amount);
            if base > u128::from(u64::MAX) {
                return Err(Error::CombinedOverflow);
            }
            let base = base as u64;
            let discount = discount(base, meter.discount_bps);
            Ok(InvoiceLine {
                name: meter.name.clone(),
                usage: bill.charge.amount,
                minimum_adjustment: bill.minimum_adjustment,
                monthly: proration.amount,
                discount,
                total: base - discount,
            })
        })().map_err(|source| Error::Meter { index, source: Box::new(source) })?;

        subtotal = u64::try_from(u128::from(subtotal) + u128::from(line.total))
            .map_err(|_| Error::InvoiceOverflow)?;
        lines.push(line);
    }

    lines.sort_by(|left, right| left.name.as_bytes().cmp(right.name.as_bytes()));

    let invoice_discount = discount(subtotal, invoice_discount_bps);
    let after_discount = subtotal - invoice_discount;
    let applied_credit = credit.min(after_discount);
    let total = after_discount - applied_credit;

    let mut rendered = String::new();
    for line in &lines {
        use std::fmt::Write;
        writeln!(
            rendered,
            "{}:{}|{}|{}|{}|{}|{}",
            line.name.len(), line.name, line.usage, line.minimum_adjustment,
            line.monthly, line.discount, line.total
        ).unwrap();
    }
    use std::fmt::Write;
    writeln!(rendered, "TOTAL|{}|{}|{}|{}", subtotal, invoice_discount, applied_credit, total).unwrap();

    Ok(Invoice {
        lines,
        subtotal,
        discount: invoice_discount,
        credit: applied_credit,
        total,
        rendered,
    })
}

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    match tiers.len() {
        0 => return Err(Error::Empty),
        65.. => return Err(Error::TooMany),
        _ => {}
    }
    let mut prev: Option<u64> = Some(0);
    for (i, t) in tiers.iter().enumerate() {
        match t.up_to {
            Some(bound) => {
                let p = prev.unwrap();
                if bound <= p { return Err(Error::Bound { index: i }); }
                prev = Some(bound);
            }
            None => {
                if i + 1 < tiers.len() { return Err(Error::Unbounded { index: i }); }
                prev = None;
            }
        }
    }
    if prev.is_some() { Err(Error::MissingFinal) } else { Ok(()) }
}

fn compute_band(units: u64, tier: &Tier, index: usize) -> Result<u64, Error> {
    let value = (units as u128) * (tier.unit as u128) + tier.start as u128;
    u64::try_from(value).map_err(|_| Error::Overflow { index })

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    let mut bands = Vec::new();
    let mut sum = 0u64;
    match price {
        Price::Flat { unit, start } => {
            if quantity > 0 {
                let product = quantity as u128 * *unit as u128 + *start as u128;
                let amt = u64::try_from(product).map_err(|_| Error::Overflow { index: 0 })?;
                bands.push(Band { index: 0, units: quantity, amount: amt });
                sum = amt;
            }
        }
        Price::Tiers(ts) => {
            validate_tiers(ts)?;
            let mut remaining = quantity;
            let mut lower = 0u64;
            for (i, t) in ts.iter().enumerate() {
                if remaining == 0 { break; }
                let units = match t.up_to {
                    Some(upper) => {
                        let width = upper - lower;
                        lower = upper;
                        remaining.min(width)
                    }
                    None => remaining,
                };
                if units != 0 {
                    let amt = compute_band(units, t, i)?;
                    sum = sum.checked_add(amt).ok_or(Error::Overflow { index: i })?;
                    bands.push(Band { index: i, units, amount: amt });
                    remaining -= units;
                }
            }
        }
    }
    Ok(Charge { amount: sum, bands })
}

fn volume_charge(quantity: u64, ts: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(ts)?;
    if quantity == 0 { return Ok(Charge { amount: 0, bands: Vec::new() }); }
    let i = ts.iter().position(|t| t.up_to.map_or(true, |u| quantity <= u)).unwrap();
    let t = &ts[i];
    let amt = compute_band(quantity, t, i)?;
    Ok(Charge { amount: amt, bands: vec![Band { index: i, units: quantity, amount: amt }] })
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Tiers(ts), Mode::Volume) => volume_charge(quantity, ts)?,
        (_, _) => quote(quantity, price)?,
    };
    let minimum_adjustment = if quantity > 0 { minimum.saturating_sub(charge.amount) } else { 0 };
    let total = charge.amount.checked_add(minimum_adjustment).ok_or(Error::CombinedOverflow)?;
    Ok(Bill { charge, minimum_adjustment, total })
}

pub fn prorate(fee: u64, days: u32, start: u32, end: u32, credits: &[DayRange], rounding: Rounding) -> Result<Proration, Error> {
    if days != 28 && days != 29 && days != 30 && days != 31 { return Err(Error::Month); }
    if start > end || end > days { return Err(Error::Period); }
    let mut prev_end = start;
    let mut active = end - start;
    for (i, c) in credits.iter().enumerate() {
        if c.start < start || c.start >= c.end || c.end > end { return Err(Error::Credit { index: i }); }
        if c.start < prev_end { return Err(Error::CreditOrder { index: i }); }
        prev_end = c.end;
        active -= c.end - c.start;
    }
    let n = fee as u128 * active as u128;
    let d = days as u128;
    let q = n / d;
    let r = n % d;
    let rounded = match rounding {
        Rounding::Down => q,
        Rounding::Up => q + if r == 0 { 0 } else { 1 },
        Rounding::HalfEven => {
            if r * 2 > d || (r * 2 == d && q & 1 == 1) { q + 1 } else { q }
        }
    };
    Ok(Proration { active_days: active, amount: rounded as u64 })
}

}

        if meters[..i].iter().any(|x| x.name == m.name) { return Err(Error::DuplicateName { index: i }); }

        lines.push(InvoiceLine { name: m.name.clone(), usage: b.charge.amount, minimum_adjustment: b.minimum_adjustment, monthly: p.amount, discount: d, total });

fn quote_mode(quantity: u64, price: &Price, mode: Mode) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                Ok(Charge { amount: 0, bands: Vec::new() })
            } else {
                let amount = band_amount(quantity, *unit, *start, 0)?;
                Ok(Charge { amount, bands: vec![Band { index: 0, units: quantity, amount }] })
            }
        }
        Price::Tiers(tiers) => {
            validate_tiers(tiers)?;
            if quantity == 0 { return Ok(Charge { amount: 0, bands: Vec::new() }); }
            if mode == Mode::Volume {
                let index = tiers.iter().position(|t| t.up_to.map_or(true, |bound| quantity <= bound)).unwrap();
                let tier = &tiers[index];
                let amount = band_amount(quantity, tier.unit, tier.start, index)?;
                return Ok(Charge { amount, bands: vec![Band { index, units: quantity, amount }] });
            }
            let mut remaining = quantity;
            let mut previous = 0u64;
            let mut total = 0u64;
            let mut bands = Vec::new();
            for (index, tier) in tiers.iter().enumerate() {
                let units = match tier.up_to {
                    Some(bound) => {
                        let width = bound - previous;
                        previous = bound;
                        remaining.min(width)
                    }
                    None => remaining,
                };
                if units > 0 {
                    let amount = band_amount(units, tier.unit, tier.start, index)?;
                    let sum = (total as u128) + (amount as u128);
                    if sum > u64::MAX as u128 { return Err(Error::Overflow { index }); }
                    total = sum as u64;
                    bands.push(Band { index, units, amount });
                    remaining -= units;
                }
            }
            Ok(Charge { amount: total, bands })
        }
    }

pub fn prorate(fee: u64, days: u32, start: u32, end: u32, credits: &[DayRange], rounding: Rounding) -> Result<Proration, Error> {
    if !(28..=31).contains(&days) { return Err(Error::Month); }
    if start > end || end > days { return Err(Error::Period); }
    let mut previous_end = start;
    let mut removed = 0u32;
    for (index, range) in credits.iter().enumerate() {
        if range.start < start || range.start >= range.end || range.end > end {
            return Err(Error::Credit { index });
        }
        if range.start < previous_end { return Err(Error::CreditOrder { index }); }
        removed += range.end - range.start;
        previous_end = range.end;
    }
    let active_days = end - start - removed;
    let numerator = (fee as u128) * (active_days as u128);
    let denominator = days as u128;
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;

        Rounding::Down => quotient,
        Rounding::Up => quotient + u128::from(remainder != 0),
        Rounding::HalfEven => {
            if remainder * 2 > denominator || (remainder * 2 == denominator && quotient % 2 == 1) { quotient + 1 } else { quotient }
        }
    };
    Ok(Proration { active_days, amount: rounded as u64 })
}

fn check_schedule(tiers: &[Tier]) -> Result<(), Error> {

fn graduate(q: u64, tiers: &[Tier]) -> Result<Charge, Error> {

}

pub fn prorate(fee: u64, days: u32, start: u32, end: u32, credits: &[DayRange], rounding: Rounding) -> Result<Proration, Error> {
    if days < 28 || days > 31 { return Err(Error::Month); }
    if start > end || end > days { return Err(Error::Period); }
    let mut cursor = start;
    let mut active = end - start;
    for (i, c) in credits.iter().enumerate() {
        if c.start < start || c.start >= c.end || c.end > end { return Err(Error::Credit { index: i }); }
        if c.start < cursor { return Err(Error::CreditOrder { index: i }); }
        active -= c.end - c.start;
        cursor = c.end;
    }
    let n = fee as u128 * active as u128;
    let d = days as u128;
    let q = n / d;
    let r = n % d;
    let amount = match rounding {
        Rounding::Down => q,
        Rounding::Up => q + if r == 0 { 0 } else { 1 },
        Rounding::HalfEven => if r * 2 > d || (r * 2 == d && q % 2 == 1) { q + 1 } else { q },
    };
    Ok(Proration { active_days: active, amount: amount as u64 })
}

        if meters[..i].iter().any(|p| p.name == m.name) { return Err(Error::DuplicateName { index: i }); }

    let mut subtotal = 0u64;
    for (i, m) in meters.iter().enumerate() {
        let line = (|| -> Result<InvoiceLine, Error> {
            if m.discount_bps > 10_000 { return Err(Error::Discount); }
            let b = bill(m.quantity, &m.price, m.mode.clone(), m.minimum)?;

        })().map_err(|e| Error::Meter { index: i, source: Box::new(e) })?;
        subtotal = subtotal.checked_add(line.total).ok_or(Error::InvoiceOverflow)?;
        lines.push(line);
    }

    let applied_credit = credit_requested.min(subtotal - disc);
    let total = subtotal - disc - applied_credit;

    use std::fmt::Write;
    writeln!(&mut rendered, "TOTAL|{}|{}|{}|{}", subtotal, disc, applied_credit, total).unwrap();
    Ok(Invoice { lines, subtotal, discount: disc, credit: applied_credit, total, rendered })
}

pub fn prorate(
    fee: u64, days: u32, start: u32, end: u32,
    credits: &[DayRange], rounding: Rounding,
) -> Result<Proration, Error> {
    if !(28..=31).contains(&days) { return Err(Error::Month); }
    if start > end || end > days { return Err(Error::Period); }
    let mut previous_end = start;
    let mut removed = 0u32;
    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < previous_end {
            return Err(Error::CreditOrder { index });
        }
        removed += credit.end - credit.start;
        previous_end = credit.end;
    }
    let active_days = end - start - removed;
    let numerator = (fee as u128) * (active_days as u128);
    let denominator = days as u128;
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let rounded = match rounding {
        Rounding::Down => quotient,
        Rounding::Up => quotient + u128::from(remainder != 0),
        Rounding::HalfEven => {
            let twice = remainder * 2;
            if twice > denominator || (twice == denominator && quotient % 2 == 1) {
                quotient + 1
            } else {
                quotient
            }
        }
    };
    Ok(Proration { active_days, amount: rounded as u64 })
}
