    let mut output = Vec::with_capacity(10);
    loop {
        let payload = (value & 0x7f) as u8;
        value >>= 7;

            break;

        output.push(payload | 0x80);

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(value: u64) -> Vec<u8> {
    let mut out = Vec::with_capacity(10);
    let mut remaining = value;
    while remaining >= 0x80 {
        out.push((remaining as u8 & 0x7f) | 0x80);
        remaining >>= 7;
    }
    out.push(remaining as u8);
    out
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut cursor = start;
    let mut result = 0u64;

    for shift in [0u32, 7, 14, 21, 28, 35, 42, 49, 56, 63] {
        if cursor == input.len() {
            return Err(Error::Truncated {
                offset: input.len(),
                start,
            });
        }

        let byte = input[cursor];
        let payload = (byte & 0x7f) as u64;

        if shift == 63 {
            if payload > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow {
                    offset: cursor,
                    start,
                });
            }
            result |= payload << 63;
            return Ok((result, cursor + 1));
        }

        result |= payload << shift;
        cursor += 1;
        if byte & 0x80 == 0 {
            return Ok((result, cursor));
        }
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut output = Vec::with_capacity(count);
    let mut next = start;
    let mut remaining = count;
    while remaining != 0 {
        let (value, after) = decode_u64(input, next)?;
        output.push(value);
        next = after;
        remaining -= 1;
    }

    Ok((output, next))
}
