- `start == input.len()` is valid; attempting to decode there is truncated.
- Invalid starts are checked before the batch count, including when `count == 0`.
- Byte ten overflows if its payload exceeds 1 or its continuation bit is set; that check precedes truncation.
- Nonminimal encodings are accepted unless they violate the byte-ten rules.
- Errors for batch decoding use the cursor of the integer that failed, and no partial values are returned.
- Both solutions use these readings; Solution B uses a different decoding approach.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut output = Vec::with_capacity(10);
    loop {
        let payload = (value & 0x7f) as u8;
        value >>= 7;
        if value == 0 {
            output.push(payload);
            break;
        }
        output.push(payload | 0x80);
    }
    output
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut value = 0u64;
    let mut index = start;

    for byte_number in 0..10 {
        if index >= input.len() {
            return Err(Error::Truncated {
                offset: input.len(),
                start,
            });
        }

        let byte = input[index];
        let payload = (byte & 0x7f) as u64;

        if byte_number == 9 {
            if payload > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow {
                    offset: index,
                    start,
                });
            }
            value |= payload << 63;
            return Ok((value, index + 1));
        }

        value |= payload << (byte_number * 7);
        index += 1;

        if byte & 0x80 == 0 {
            return Ok((value, index));
        }
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut values = Vec::with_capacity(count);
    let mut cursor = start;

    for _ in 0..count {
        let (value, next) = decode_u64(input, cursor)?;
        values.push(value);
        cursor = next;
    }

    Ok((values, cursor))
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(value: u64) -> Vec<u8> {
    let mut bytes = Vec::with_capacity(10);
    let mut remaining = value;

    while remaining >= 0x80 {
        bytes.push(((remaining & 0x7f) as u8) | 0x80);
        remaining >>= 7;
    }
    bytes.push(remaining as u8);
    bytes
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut result = 0u64;
    let mut shift = 0u32;
    let mut cursor = start;

    loop {
        if cursor == input.len() {
            return Err(Error::Truncated {
                offset: input.len(),
                start,
            });
        }

        let byte = input[cursor];
        let payload = byte & 0x7f;

        if shift == 63 {
            if payload > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow {
                    offset: cursor,
                    start,
                });
            }
            result |= (payload as u64) << 63;
            return Ok((result, cursor + 1));
        }

        result |= (payload as u64) << shift;
        cursor += 1;

        if byte & 0x80 == 0 {
            return Ok((result, cursor));
        }

        shift += 7;
    }
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut decoded = Vec::new();
    let mut cursor = start;

    for _ in 0..count {
        let (value, next_cursor) = decode_u64(input, cursor)?;
        decoded.push(value);
        cursor = next_cursor;
    }

    Ok((decoded, cursor))
}
```
