### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut output = Vec::with_capacity(10);
    loop {
        let payload = (value & 0x7f) as u8;
        value >>= 7;
        if value == 0 {
            output.push(payload);
            break;
        }
        output.push(payload | 0x80);
    }
    output
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut value = 0u64;
    let mut index = start;

    for byte_index in 0..10 {
        if index >= input.len() {
            return Err(Error::Truncated {
                offset: input.len(),
                start,
            });
        }

        let byte = input[index];
        let payload = (byte & 0x7f) as u64;

        if byte_index == 9 {
            if payload > 1 || (byte & 0x80) != 0 {
                return Err(Error::Overflow {
                    offset: index,
                    start,
                });
            }
            value |= payload << 63;
            return Ok((value, index + 1));
        }

        value |= payload << (byte_index * 7);
        index += 1;

        if (byte & 0x80) == 0 {
            return Ok((value, index));
        }
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut values = Vec::with_capacity(count);
    let mut cursor = start;

    for _ in 0..count {
        let (value, next) = decode_u64(input, cursor)?;
        values.push(value);
        cursor = next;
    }

    Ok((values, cursor))
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(value: u64) -> Vec<u8> {
    let mut bytes = Vec::new();
    let mut remaining = value;

    while remaining >= 0x80 {
        bytes.push(((remaining & 0x7f) as u8) | 0x80);
        remaining >>= 7;
    }
    bytes.push(remaining as u8);
    bytes
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut result = 0u64;
    let mut cursor = start;
    let mut shift = 0u32;

    loop {
        if cursor == input.len() {
            return Err(Error::Truncated {
                offset: input.len(),
                start,
            });
        }

        let byte = input[cursor];
        let payload = (byte & 0x7f) as u64;

        if shift == 63 {
            if payload > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow {
                    offset: cursor,
                    start,
                });
            }
            result |= payload << 63;
            return Ok((result, cursor + 1));
        }

        result |= payload << shift;
        cursor += 1;

        if byte & 0x80 == 0 {
            return Ok((result, cursor));
        }

        shift += 7;
    }
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut output = Vec::with_capacity(count);
    let mut cursor = start;
    let mut remaining = count;

    while remaining != 0 {
        let (value, next_cursor) = decode_u64(input, cursor)?;
        output.push(value);
        cursor = next_cursor;
        remaining -= 1;
    }

    Ok((output, cursor))
}
```
