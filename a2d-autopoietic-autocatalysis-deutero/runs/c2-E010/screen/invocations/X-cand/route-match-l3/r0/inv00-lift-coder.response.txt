- Method validation happens first in both `add` and `resolve`.
- A lone `/` is root; other paths allow one final slash, while interior empty segments are errors.
- Percent escapes decode once to printable ASCII punctuation; unreserved escapes and encoded `/` are invalid.
- Wildcards capture zero or more remaining segments, joining decoded values with `/`.
- Slash policy affects candidate eligibility or redirect only; route ranking uses segment specificity.
- For ties and ambiguous wildcard/trailing-slash cases, I follow the explicit ordering and zero-segment rules.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn name_valid(name: &str) -> bool {
    let mut bytes = name.bytes();
    match bytes.next() {
        Some(b'a'..=b'z') => {}
        _ => return false,
    }
    bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else {
            if !b.is_ascii() || !unreserved(b) {
                return Err(base + i);
            }
            out.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

struct ParsedPattern {
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

fn parse_pattern(pattern: &str) -> Result<ParsedPattern, usize> {
    let bytes = pattern.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if pattern == "/" {
        return Ok(ParsedPattern {
            segments: vec![],
            trailing: false,
            rank: vec![3],
            has_wildcard: false,
        });
    }

    let trailing = pattern.ends_with('/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut rank = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut pos = 1;
    let mut has_wildcard = false;

    while pos < end {
        let next = bytes[pos..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| pos + n)
            .unwrap_or(end);
        if next == pos {
            return Err(pos);
        }
        let raw = &pattern[pos..next];
        let first = bytes[pos];
        if first == b':' || first == b'*' {
            let name = &raw[1..];
            if !name_valid(name) || !names.insert(name.to_owned()) {
                return Err(pos);
            }
            if first == b'*' {
                if next != end {
                    return Err(pos);
                }
                has_wildcard = true;
                segments.push(Segment::Wildcard(name.to_owned()));
                rank.push(0);
            } else {
                segments.push(Segment::Param(name.to_owned()));
                rank.push(1);
            }
        } else {
            let decoded = decode_segment(raw, pos)?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }
        if next == end {
            break;
        }
        pos = next + 1;
        if pos == end {
            break;
        }
    }
    if !has_wildcard {
        rank.push(3);
    }
    Ok(ParsedPattern {
        segments,
        trailing,
        rank,
        has_wildcard,
    })
}

struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

fn parse_path(path: &str) -> Result<ParsedPath, usize> {
    let bytes = path.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if path == "/" {
        return Ok(ParsedPath {
            segments: vec![],
            trailing: false,
        });
    }
    let trailing = path.ends_with('/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut pos = 1;
    while pos < end {
        let next = bytes[pos..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| pos + n)
            .unwrap_or(end);
        if next == pos {
            return Err(pos);
        }
        segments.push(decode_segment(&path[pos..next], pos)?);
        if next == end {
            break;
        }
        pos = next + 1;
    }
    Ok(ParsedPath { segments, trailing })
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let parsed = parse_pattern(pattern).map_err(|at| Error::InvalidPattern { at })?;
        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_owned(),
            segments: parsed.segments,
            trailing: parsed.trailing,
            rank: parsed.rank,
            has_wildcard: parsed.has_wildcard,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_path(head).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                continue;
            }
            let mut params = Vec::new();
            let mut matched = true;
            let mut i = 0;
            for segment in &route.segments {
                match segment {
                    Segment::Static(s) => {
                        if parsed.segments.get(i) != Some(s) {
                            matched = false;
                            break;
                        }
                        i += 1;
                    }
                    Segment::Param(name) => {
                        if let Some(value) = parsed.segments.get(i) {
                            params.push((name.clone(), value.clone()));
                            i += 1;
                        } else {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Wildcard(name) => {
                        params.push((name.clone(), parsed.segments[i..].join("/")));
                        i = parsed.segments.len();
                    }
                }
            }
            if matched && i == parsed.segments.len() {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let methods: std::collections::BTreeSet<String> =
            candidates.iter().map(|(r, _)| r.method.clone()).collect();
        let eligible_method = if method == "HEAD" && methods.contains("HEAD") {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<_> = candidates
            .into_iter()
            .filter(|(r, _)| r.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = methods;
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_owned());
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        eligible.sort_by(|(a, _), (b, _)| b.rank.cmp(&a.rank).then(a.id.cmp(&b.id)));
        let (route, params) = eligible.remove(0);
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != parsed.trailing
            && !route.segments.is_empty()
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Part {
    Lit(String),
    One(String),
    Rest(String),
}

#[derive(Clone)]
struct Entry {
    method: String,
    parts: Vec<Part>,
    trailing: bool,
}

pub struct Router {
    policy: SlashPolicy,
    entries: Vec<Entry>,
}

fn method_ok(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_name(s: &str) -> bool {
    let mut it = s.bytes();
    matches!(it.next(), Some(b'a'..=b'z'))
        && it.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn static_decode(s: &str, offset: usize) -> Result<String, usize> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        if b[i] == b'%' {
            if i + 2 >= b.len() {
                return Err(offset + i);
            }
            let digit = |x: u8| match x {
                b'0'..=b'9' => Some(x - b'0'),
                b'A'..=b'F' => Some(x - b'A' + 10),
                _ => None,
            };
            let hi = digit(b[i + 1]).ok_or(offset + i)?;
            let lo = digit(b[i + 2]).ok_or(offset + i)?;
            let x = (hi << 4) | lo;
            let is_unreserved =
                x.is_ascii_alphanumeric() || matches!(x, b'.' | b'_' | b'~' | b'-');
            if !(0x20..=0x7e).contains(&x) || x == b'/' || is_unreserved {
                return Err(offset + i);
            }
            out.push(x);
            i += 3;
        } else {
            let x = b[i];
            let is_unreserved =
                x.is_ascii_alphanumeric() || matches!(x, b'.' | b'_' | b'~' | b'-');
            if !x.is_ascii() || !is_unreserved {
                return Err(offset + i);
            }
            out.push(x);
            i += 1;
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn split_path(s: &str) -> Result<(Vec<(String, usize)>, bool), usize> {
    if !s.starts_with('/') {
        return Err(0);
    }
    if s == "/" {
        return Ok((Vec::new(), false));
    }
    let trailing = s.ends_with('/');
    let end = if trailing { s.len() - 1 } else { s.len() };
    let mut out = Vec::new();
    let mut start = 1;
    for i in 1..=end {
        if i == end || s.as_bytes()[i] == b'/' {
            if i == start {
                return Err(start);
            }
            out.push((s[start..i].to_owned(), start));
            start = i + 1;
        }
    }
    Ok((out, trailing))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self { policy, entries: Vec::new() }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !method_ok(method) {
            return Err(Error::InvalidMethod);
        }
        let (raw, trailing) =
            split_path(pattern).map_err(|at| Error::InvalidPattern { at })?;
        let mut parts = Vec::new();
        let mut names = std::collections::HashSet::new();
        let mut wildcard = false;
        for (seg, at) in raw {
            if seg.starts_with(':') || seg.starts_with('*') {
                let marker = seg.as_bytes()[0];
                let name = &seg[1..];
                if !valid_name(name) || !names.insert(name.to_owned()) {
                    return Err(Error::InvalidPattern { at });
                }
                if marker == b'*' {
                    wildcard = true;
                    // Its position is checked against the original segment list below.
                    parts.push(Part::Rest(name.to_owned()));
                } else {
                    parts.push(Part::One(name.to_owned()));
                }
            } else {
                parts.push(Part::Lit(
                    static_decode(&seg, at).map_err(|at| Error::InvalidPattern { at })?,
                ));
            }
        }
        if let Some(index) = parts.iter().position(|p| matches!(p, Part::Rest(_))) {
            if index + 1 != parts.len() {
                let at = raw_start_for_index(pattern, index);
                return Err(Error::InvalidPattern { at });
            }
        }
        let _ = wildcard;
        let id = self.entries.len();
        self.entries.push(Entry {
            method: method.to_owned(),
            parts,
            trailing,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !method_ok(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (raw, trailing) =
            split_path(head).map_err(|at| Error::InvalidPath { at })?;
        let mut decoded = Vec::with_capacity(raw.len());
        for (seg, at) in raw {
            decoded.push(static_decode(&seg, at).map_err(|at| Error::InvalidPath { at })?);
        }

        let mut candidates: Vec<(usize, Vec<u8>, Vec<(String, String)>, bool)> = Vec::new();
        for (id, entry) in self.entries.iter().enumerate() {
            if self.policy == SlashPolicy::Strict && entry.trailing != trailing {
                continue;
            }
            let mut params = Vec::new();
            let mut index = 0;
            let mut rank = Vec::new();
            let mut ok = true;
            for part in &entry.parts {
                match part {
                    Part::Lit(s) => {
                        rank.push(2);
                        if decoded.get(index) != Some(s) {
                            ok = false;
                            break;
                        }
                        index += 1;
                    }
                    Part::One(name) => {
                        rank.push(1);
                        if let Some(value) = decoded.get(index) {
                            params.push((name.clone(), value.clone()));
                            index += 1;
                        } else {
                            ok = false;
                            break;
                        }
                    }
                    Part::Rest(name) => {
                        rank.push(0);
                        params.push((name.clone(), decoded[index..].join("/")));
                        index = decoded.len();
                        break;
                    }
                }
            }
            if ok && index == decoded.len() {
                if !entry.parts.iter().any(|p| matches!(p, Part::Rest(_))) {
                    rank.push(3);
                }
                candidates.push((id, rank, params, entry.trailing));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let mut method_set = std::collections::BTreeSet::new();
        for (id, _, _, _) in &candidates {
            method_set.insert(self.entries[*id].method.clone());
        }
        let chosen_method = if method == "HEAD" && method_set.contains("HEAD") {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };
        let mut matching: Vec<_> = candidates
            .into_iter()
            .filter(|(id, _, _, _)| self.entries[*id].method == chosen_method)
            .collect();

        if matching.is_empty() {
            if method_set.contains("GET") {
                method_set.insert("HEAD".to_owned());
            }
            return Err(Error::MethodNotAllowed {
                allowed: method_set.into_iter().collect(),
            });
        }

        matching.sort_by(|a, b| b.1.cmp(&a.1).then(a.0.cmp(&b.0)));
        let (id, _, params, route_trailing) = matching.remove(0);
        let redirect = if self.policy == SlashPolicy::Redirect
            && route_trailing != trailing
            && !self.entries[id].parts.is_empty()
        {
            let mut dest = head.to_owned();
            if route_trailing {
                dest.push('/');
            } else {
                dest.pop();
            }
            Some(dest)
        } else {
            None
        };
        Ok(Match { route: id, params, redirect })
    }
}

fn raw_start_for_index(pattern: &str, wanted: usize) -> usize {
    let bytes = pattern.as_bytes();
    let mut index = 0;
    let mut start = 1;
    while start < bytes.len() {
        if index == wanted {
            return start;
        }
        while start < bytes.len() && bytes[start] != b'/' {
            start += 1;
        }
        index += 1;
        start += 1;
    }
    pattern.len()
}
```
