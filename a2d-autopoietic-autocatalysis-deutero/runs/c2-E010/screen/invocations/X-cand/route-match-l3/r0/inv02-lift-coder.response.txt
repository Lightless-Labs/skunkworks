- **Validation order:** I validate the method first, then path segments left to right, then trailing-slash policy.
- **Root and trailing slash:** `/` has no segments and no trailing-slash flag; a non-root final slash is stored separately.
- **Percent escapes:** only canonical uppercase `%HH` escapes for reserved printable ASCII are accepted, decoded once before matching and capture.
- **Wildcard and ranking:** a final wildcard may capture zero segments; rank vectors use the specified terminal marker for non-wildcard routes.
- **HEAD fallback:** any matching HEAD path candidate suppresses GET fallback, regardless of relative rank.
- **Redirects:** redirects preserve the original encoded path head and adjust only its final slash; query and fragment are discarded.

### Solution A

```rust
use std::collections::BTreeSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    has_wildcard: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self { policy, routes: Vec::new() }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing, has_wildcard) =
            parse_pattern(pattern).map_err(|at| Error::InvalidPattern { at })?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }
        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_owned(),
            segments,
            trailing,
            has_wildcard,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (head, request_segments, request_trailing) =
            parse_path(path).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(usize, Vec<(String, String)>)> = Vec::new();
        for (idx, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            if let Some(params) = match_route(route, &request_segments) {
                candidates.push((idx, params));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let mut eligible: Vec<usize> = if method == "HEAD" {
            let has_head = candidates.iter().any(|(idx, _)| self.routes[*idx].method == "HEAD");
            let wanted = if has_head { "HEAD" } else { "GET" };
            candidates.iter().enumerate()
                .filter_map(|(i, (idx, _))| (self.routes[*idx].method == wanted).then_some(i))
                .collect()
        } else {
            candidates.iter().enumerate()
                .filter_map(|(i, (idx, _))| (self.routes[*idx].method == method).then_some(i))
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (idx, _) in &candidates {
                allowed.insert(self.routes[*idx].method.clone());
            }
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_owned());
            }
            return Err(Error::MethodNotAllowed { allowed: allowed.into_iter().collect() });
        }

        eligible.sort_by(|a, b| {
            let ra = &self.routes[candidates[*a].0].rank;
            let rb = &self.routes[candidates[*b].0].rank;
            rb.cmp(ra).then_with(|| candidates[*a].0.cmp(&candidates[*b].0))
        });
        let (route_index, params) = candidates[eligible[0]].clone();
        let route = &self.routes[route_index];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !head_is_root(head)
            && route.trailing != request_trailing
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {
            None
        };
        Ok(Match { route: route_index, params, redirect })
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn parse_path(input: &str) -> Result<(&str, Vec<String>, bool), usize> {
    let end = input.bytes().position(|b| b == b'?' || b == b'#').unwrap_or(input.len());
    let head = &input[..end];
    let segments = parse_segments(head, false)?;
    let trailing = head != "/" && head.ends_with('/');
    Ok((head, segments, trailing))
}

fn head_is_root(head: &str) -> bool {
    head == "/"
}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool, bool), usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false, false));
    }
    let trailing = input.ends_with('/');
    let body_end = if trailing { input.len() - 1 } else { input.len() };
    let mut raw: Vec<(usize, &str)> = Vec::new();
    let mut start = 1;
    for (i, b) in input.as_bytes().iter().enumerate().take(body_end).skip(1) {
        if *b == b'/' {
            if i == start {
                return Err(start);
            }
            raw.push((start, &input[start..i]));
            start = i + 1;
        }
    }
    if start < body_end {
        raw.push((start, &input[start..body_end]));
    } else if !trailing {
        return Err(start);
    }
    let mut segments = Vec::with_capacity(raw.len());
    let mut names = BTreeSet::new();
    let mut wildcard = false;
    for (i, (at, text)) in raw.iter().enumerate() {
        if text.starts_with(':') || text.starts_with('*') {
            let marker = text.as_bytes()[0];
            let name = &text[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(*at);
            }
            if marker == b'*' {
                if i + 1 != raw.len() {
                    return Err(*at);
                }
                wildcard = true;
                segments.push(Segment::Wildcard(name.to_owned()));
            } else {
                segments.push(Segment::Param(name.to_owned()));
            }
        } else {
            segments.push(Segment::Static(parse_encoded_segment(text, *at)?));
        }
    }
    Ok((segments, trailing, wildcard))
}

fn parse_segments(head: &str, _unused: bool) -> Result<Vec<String>, usize> {
    if !head.starts_with('/') {
        return Err(0);
    }
    if head == "/" {
        return Ok(Vec::new());
    }
    let trailing = head.ends_with('/');
    let end = if trailing { head.len() - 1 } else { head.len() };
    let mut out = Vec::new();
    let mut start = 1;
    for (i, b) in head.as_bytes().iter().enumerate().take(end).skip(1) {
        if *b == b'/' {
            if i == start {
                return Err(start);
            }
            out.push(parse_encoded_segment(&head[start..i], start)?);
            start = i + 1;
        }
    }
    if start < end {
        out.push(parse_encoded_segment(&head[start..end], start)?);
    } else if !trailing {
        return Err(start);
    }
    Ok(out)
}

fn parse_encoded_segment(text: &str, base: usize) -> Result<String, usize> {
    let bytes = text.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(base + i);
            }
            let value = (hex(bytes[i + 1]) << 4) | hex(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            out.push(value);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(b: u8) -> bool {
    b.is_ascii_digit() || (b'A'..=b'F').contains(&b)
}

fn hex(b: u8) -> u8 {
    if b.is_ascii_digit() { b - b'0' } else { b - b'A' + 10 }
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b[0].is_ascii_lowercase()
        && b[1..].iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_')
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut pos = 0;
    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if request.get(pos)? != value { return None; }
                pos += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), request.get(pos)?.clone()));
                pos += 1;
            }
            Segment::Wildcard(name) => {
                let value = request[pos..].join("/");
                params.push((name.clone(), value));
                pos = request.len();
            }
        }
    }
    (pos == request.len()).then_some(params)
}
```

### Solution B

```rust
use std::collections::BTreeSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self { policy, routes: Vec::new() }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing) = parse_pattern(pattern)
            .map_err(|at| Error::InvalidPattern { at })?;
        let has_wildcard = segments.iter().any(|s| matches!(s, Segment::Wildcard(_)));
        let mut rank: Vec<u8> = segments.iter().map(|s| match s {
            Segment::Static(_) => 2,
            Segment::Param(_) => 1,
            Segment::Wildcard(_) => 0,
        }).collect();
        if !has_wildcard {
            rank.push(3);
        }
        let id = self.routes.len();
        self.routes.push(Route { method: method.to_owned(), segments, trailing, rank });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let end = path.bytes().position(|b| b == b'?' || b == b'#').unwrap_or(path.len());
        let head = &path[..end];
        let (request, request_trailing) =
            parse_request_head(head).map_err(|at| Error::InvalidPath { at })?;

        let candidates: Vec<(usize, Vec<(String, String)>)> = self.routes.iter().enumerate()
            .filter_map(|(i, route)| {
                if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                    return None;
                }
                match_route(route, &request).map(|p| (i, p))
            }).collect();

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD" {
            if candidates.iter().any(|(i, _)| self.routes[*i].method == "HEAD") {
                "HEAD"
            } else {
                "GET"
            }
        } else {
            method
        };
        let mut eligible: Vec<_> = candidates.iter()
            .filter(|(i, _)| self.routes[*i].method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (i, _) in &candidates {
                allowed.insert(self.routes[*i].method.clone());
            }
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_owned());
            }
            return Err(Error::MethodNotAllowed { allowed: allowed.into_iter().collect() });
        }

        eligible.sort_by(|(ia, _), (ib, _)| {
            self.routes[*ib].rank.cmp(&self.routes[*ia].rank).then_with(|| ia.cmp(ib))
        });
        let (index, params) = eligible[0];
        let route = &self.routes[*index];
        let redirect = if self.policy == SlashPolicy::Redirect
            && head != "/"
            && route.trailing != request_trailing
        {
            let mut target = head.to_owned();
            if route.trailing { target.push('/'); } else { target.pop(); }
            Some(target)
        } else {
            None
        };
        Ok(Match { route: *index, params: params.clone(), redirect })
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b'A' <= b && b <= b'Z')
}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool), usize> {
    let (parts, trailing) = split_head(input)?;
    let mut segments = Vec::with_capacity(parts.len());
    let mut names = BTreeSet::new();
    for (i, (at, part)) in parts.iter().enumerate() {
        if part.starts_with(':') || part.starts_with('*') {
            let marker = part.as_bytes()[0];
            let name = &part[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(*at);
            }
            if marker == b'*' && i + 1 != parts.len() {
                return Err(*at);
            }
            segments.push(if marker == b'*' {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            segments.push(Segment::Static(parse_encoded_segment(part, *at)?));
        }
    }
    Ok((segments, trailing))
}

fn parse_request_head(input: &str) -> Result<(Vec<String>, bool), usize> {
    let (parts, trailing) = split_head(input)?;
    let mut segments = Vec::with_capacity(parts.len());
    for (at, part) in parts {
        segments.push(parse_encoded_segment(part, at)?);
    }
    Ok((segments, trailing))
}

fn split_head(input: &str) -> Result<(Vec<(usize, &str)>, bool), usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }
    let trailing = input.ends_with('/');
    let limit = input.len() - usize::from(trailing);
    let bytes = input.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1;
    let mut i = 1;
    while i < limit {
        if bytes[i] == b'/' {
            if i == start {
                return Err(start);
            }
            parts.push((start, &input[start..i]));
            start = i + 1;
        }
        i += 1;
    }
    if start < limit {
        parts.push((start, &input[start..limit]));
    } else if !trailing {
        return Err(start);
    }
    Ok((parts, trailing))
}

fn parse_encoded_segment(text: &str, base: usize) -> Result<String, usize> {
    let bytes = text.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() || !upper_hex(bytes[i + 1]) || !upper_hex(bytes[i + 2]) {
                return Err(base + i);
            }
            let v = (hex(bytes[i + 1]) << 4) | hex(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&v) || v == b'/' || unreserved(v) {
                return Err(base + i);
            }
            decoded.push(v);
            i += 3;
        } else if unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(decoded).unwrap())
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty() && b[0].is_ascii_lowercase()
        && b[1..].iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn upper_hex(b: u8) -> bool {
    b.is_ascii_digit() || (b'A'..=b'F').contains(&b)
}

fn hex(b: u8) -> u8 {
    if b.is_ascii_digit() { b - b'0' } else { b - b'A' + 10 }
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let mut captures = Vec::new();
    let mut p = 0;
    for segment in &route.segments {
        match segment {
            Segment::Static(expected) => {
                if request.get(p)? != expected { return None; }
                p += 1;
            }
            Segment::Param(name) => {
                captures.push((name.clone(), request.get(p)?.clone()));
                p += 1;
            }
            Segment::Wildcard(name) => {
                captures.push((name.clone(), request[p..].join("/")));
                p = request.len();
            }
        }
    }
    if p == request.len() { Some(captures) } else { None }
}
```
