    if path {
        Error::InvalidPath { at }
    } else {
        Error::InvalidPattern { at }
    }
}

fn decode(raw: &str, base: usize, path: bool) -> Result<String, Error> {
    let b = raw.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        if b[i] == b'%' {
            if i + 2 >= b.len() {
                return Err(err(path, base + i));
            }
            let h = |c: u8| match c {
                b'0'..=b'9' => Some(c - b'0'),
                b'A'..=b'F' => Some(c - b'A' + 10),
                _ => None,
            };
            let Some(v) = h(b[i + 1]).zip(h(b[i + 2])).map(|(a, z)| a * 16 + z) else {
                return Err(err(path, base + i));
            };
            if !(0x20..=0x7e).contains(&v) || v == b'/' || is_unreserved(v) {
                return Err(err(path, base + i));
            }
            out.push(v);
            i += 3;
        } else if is_unreserved(b[i]) {
            out.push(b[i]);
            i += 1;
        } else {
            return Err(err(path, base + i));
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn split(input: &str, path: bool) -> Result<(Vec<(&str, usize)>, bool), Error> {
    if !input.starts_with('/') {
        return Err(err(path, 0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }
    let bytes = input.as_bytes();
    let mut end = bytes.len();
    let trailing = bytes[end - 1] == b'/';
    if trailing {
        end -= 1;
    }
    let mut out = Vec::new();
    let mut start = 1;
    while start <= end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(err(path, start));
        }
        out.push((&input[start..stop], start));
        if stop == end {
            break;
        }
        start = stop + 1;
    }
    Ok((out, trailing))
}

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {

            return Err(Error::InvalidMethod);
        }
        let (raw, trailing) = split(pattern, false)?;
        let mut parts = Vec::new();
        let mut names = std::collections::HashSet::new();
        let mut wildcard = false;
        for (i, (s, at)) in raw.iter().enumerate() {

                if !valid_name(name) || !names.insert(name.to_string()) {
                    return Err(Error::InvalidPattern { at: *at });

                    return Err(Error::InvalidPattern { at: *at });
                }

                    parts.push(Part::Wildcard(name.to_string()));

                    parts.push(Part::Param(name.to_string()));
                }

                parts.push(Part::Static(decode(s, *at, false)?));
            }
        }
        let id = self.entries.len();
        self.entries.push(Entry { method: method.to_string(), parts, trailing, wildcard });

            return Err(Error::InvalidMethod);

        let head = &path[..head_end];

        let mut decoded = Vec::new();

            decoded.push(decode(s, at, true)?);
        }

            if self.policy == SlashPolicy::Strict && entry.trailing != trailing {
                return None;
            }
            if entry.wildcard {
                if entry.parts.len() - 1 > decoded.len() {
                    return None;
                }
            } else if entry.parts.len() != decoded.len() {
                return None;
            }
            let mut params = Vec::new();

                match part {
                    Part::Static(v) if decoded.get(i) != Some(v) => return None,
                    Part::Static(_) => {}
                    Part::Param(n) => params.push((n.clone(), decoded[i].clone())),
                    Part::Wildcard(n) => params.push((n.clone(), decoded[i..].join("/"))),
                }
            }
            Some(params)
        };

        for (id, entry) in self.entries.iter().enumerate() {
            if let Some(params) = matches(entry) {
                all.push((id, params));
            }
        }
        if all.is_empty() {
            return Err(Error::NotFound);
        }

        let mut eligible: Vec<_> = all.iter().filter(|(id, _)| self.entries[*id].method == wanted).cloned().collect();

        if eligible.is_empty() {
            let mut allowed = std::collections::BTreeSet::new();
            for (id, _) in &all {
                allowed.insert(self.entries[*id].method.clone());
            }
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_string());
            }
            return Err(Error::MethodNotAllowed { allowed: allowed.into_iter().collect() });
        }

        let rank = |entry: &Entry| {
            let mut r = Vec::new();
            for p in &entry.parts {
                r.push(match p { Part::Static(_) => 2, Part::Param(_) => 1, Part::Wildcard(_) => 0 });
            }
            if !entry.wildcard { r.push(3); }
            r
        };
        let mut best = eligible.remove(0);
        for candidate in eligible {
            if rank(&self.entries[candidate.0]) > rank(&self.entries[best.0]) {
                best = candidate;
            }
        }
        let entry = &self.entries[best.0];
        let redirect = if self.policy == SlashPolicy::Redirect && entry.trailing != trailing {
            let mut target = head.to_string();
            if trailing { target.pop(); } else if head != "/" { target.push('/'); }
            Some(target)
        } else { None };

        Ok(Match { route: best.0, params: best.1, redirect })
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn name_valid(name: &str) -> bool {
    let mut bytes = name.bytes();
    match bytes.next() {
        Some(b'a'..=b'z') => {}
        _ => return false,
    }
    bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else {
            if !b.is_ascii() || !unreserved(b) {
                return Err(base + i);
            }
            out.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

struct ParsedPattern {
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

fn parse_pattern(pattern: &str) -> Result<ParsedPattern, usize> {
    let bytes = pattern.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if pattern == "/" {
        return Ok(ParsedPattern {
            segments: vec![],
            trailing: false,
            rank: vec![3],
            has_wildcard: false,
        });
    }

    let trailing = pattern.ends_with('/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut rank = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut pos = 1;
    let mut has_wildcard = false;

    while pos < end {
        let next = bytes[pos..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| pos + n)
            .unwrap_or(end);
        if next == pos {
            return Err(pos);
        }
        let raw = &pattern[pos..next];
        let first = bytes[pos];
        if first == b':' || first == b'*' {
            let name = &raw[1..];
            if !name_valid(name) || !names.insert(name.to_owned()) {
                return Err(pos);
            }
            if first == b'*' {
                if next != end {
                    return Err(pos);
                }
                has_wildcard = true;
                segments.push(Segment::Wildcard(name.to_owned()));
                rank.push(0);
            } else {
                segments.push(Segment::Param(name.to_owned()));
                rank.push(1);
            }
        } else {
            let decoded = decode_segment(raw, pos)?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }
        if next == end {
            break;
        }
        pos = next + 1;
        if pos == end {
            break;
        }
    }
    if !has_wildcard {
        rank.push(3);
    }
    Ok(ParsedPattern {
        segments,
        trailing,
        rank,
        has_wildcard,
    })
}

struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

fn parse_path(path: &str) -> Result<ParsedPath, usize> {
    let bytes = path.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if path == "/" {
        return Ok(ParsedPath {
            segments: vec![],
            trailing: false,
        });
    }
    let trailing = path.ends_with('/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut pos = 1;
    while pos < end {
        let next = bytes[pos..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| pos + n)
            .unwrap_or(end);
        if next == pos {
            return Err(pos);
        }
        segments.push(decode_segment(&path[pos..next], pos)?);
        if next == end {
            break;
        }
        pos = next + 1;
    }

}

    pub fn new(policy: SlashPolicy) -> Self {

            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {

            method: method.to_owned(),

            trailing: parsed.trailing,
            rank: parsed.rank,
            has_wildcard: parsed.has_wildcard,
        });
        Ok(id)

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_path(head).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                continue;
            }
            let mut params = Vec::new();
            let mut matched = true;
            let mut i = 0;
            for segment in &route.segments {
                match segment {
                    Segment::Static(s) => {
                        if parsed.segments.get(i) != Some(s) {
                            matched = false;
                            break;
                        }
                        i += 1;
                    }
                    Segment::Param(name) => {
                        if let Some(value) = parsed.segments.get(i) {
                            params.push((name.clone(), value.clone()));
                            i += 1;
                        } else {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Wildcard(name) => {
                        params.push((name.clone(), parsed.segments[i..].join("/")));
                        i = parsed.segments.len();
                    }
                }
            }
            if matched && i == parsed.segments.len() {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let methods: std::collections::BTreeSet<String> =
            candidates.iter().map(|(r, _)| r.method.clone()).collect();
        let eligible_method = if method == "HEAD" && methods.contains("HEAD") {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<_> = candidates
            .into_iter()
            .filter(|(r, _)| r.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = methods;
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_owned());
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        eligible.sort_by(|(a, _), (b, _)| b.rank.cmp(&a.rank).then(a.id.cmp(&b.id)));
        let (route, params) = eligible.remove(0);
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != parsed.trailing
            && !route.segments.is_empty()
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }
}

fn method_ok(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_name(s: &str) -> bool {
    let mut it = s.bytes();
    matches!(it.next(), Some(b'a'..=b'z'))
        && it.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn static_decode(s: &str, offset: usize) -> Result<String, usize> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        if b[i] == b'%' {
            if i + 2 >= b.len() {
                return Err(offset + i);
            }
            let digit = |x: u8| match x {
                b'0'..=b'9' => Some(x - b'0'),
                b'A'..=b'F' => Some(x - b'A' + 10),
                _ => None,
            };
            let hi = digit(b[i + 1]).ok_or(offset + i)?;
            let lo = digit(b[i + 2]).ok_or(offset + i)?;
            let x = (hi << 4) | lo;
            let is_unreserved =
                x.is_ascii_alphanumeric() || matches!(x, b'.' | b'_' | b'~' | b'-');
            if !(0x20..=0x7e).contains(&x) || x == b'/' || is_unreserved {
                return Err(offset + i);
            }
            out.push(x);
            i += 3;
        } else {
            let x = b[i];
            let is_unreserved =
                x.is_ascii_alphanumeric() || matches!(x, b'.' | b'_' | b'~' | b'-');
            if !x.is_ascii() || !is_unreserved {
                return Err(offset + i);
            }
            out.push(x);
            i += 1;
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn split_path(s: &str) -> Result<(Vec<(String, usize)>, bool), usize> {
    if !s.starts_with('/') {
        return Err(0);
    }
    if s == "/" {
        return Ok((Vec::new(), false));
    }
    let trailing = s.ends_with('/');
    let end = if trailing { s.len() - 1 } else { s.len() };
    let mut out = Vec::new();
    let mut start = 1;
    for i in 1..=end {
        if i == end || s.as_bytes()[i] == b'/' {
            if i == start {
                return Err(start);
            }
            out.push((s[start..i].to_owned(), start));
            start = i + 1;
        }
    }
    Ok((out, trailing))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self { policy, entries: Vec::new() }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !method_ok(method) {
            return Err(Error::InvalidMethod);
        }
        let (raw, trailing) =
            split_path(pattern).map_err(|at| Error::InvalidPattern { at })?;
        let mut parts = Vec::new();
        let mut names = std::collections::HashSet::new();
        let mut wildcard = false;
        for (seg, at) in raw {
            if seg.starts_with(':') || seg.starts_with('*') {
                let marker = seg.as_bytes()[0];
                let name = &seg[1..];
                if !valid_name(name) || !names.insert(name.to_owned()) {
                    return Err(Error::InvalidPattern { at });
                }
                if marker == b'*' {
                    wildcard = true;
                    // Its position is checked against the original segment list below.
                    parts.push(Part::Rest(name.to_owned()));
                } else {
                    parts.push(Part::One(name.to_owned()));
                }
            } else {
                parts.push(Part::Lit(
                    static_decode(&seg, at).map_err(|at| Error::InvalidPattern { at })?,
                ));
            }
        }
        if let Some(index) = parts.iter().position(|p| matches!(p, Part::Rest(_))) {
            if index + 1 != parts.len() {
                let at = raw_start_for_index(pattern, index);
                return Err(Error::InvalidPattern { at });
            }
        }
        let _ = wildcard;
        let id = self.entries.len();
        self.entries.push(Entry {
            method: method.to_owned(),
            parts,
            trailing,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !method_ok(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (raw, trailing) =
            split_path(head).map_err(|at| Error::InvalidPath { at })?;
        let mut decoded = Vec::with_capacity(raw.len());
        for (seg, at) in raw {
            decoded.push(static_decode(&seg, at).map_err(|at| Error::InvalidPath { at })?);
        }

        let mut candidates: Vec<(usize, Vec<u8>, Vec<(String, String)>, bool)> = Vec::new();
        for (id, entry) in self.entries.iter().enumerate() {
            if self.policy == SlashPolicy::Strict && entry.trailing != trailing {
                continue;
            }
            let mut params = Vec::new();
            let mut index = 0;
            let mut rank = Vec::new();
            let mut ok = true;
            for part in &entry.parts {
                match part {
                    Part::Lit(s) => {
                        rank.push(2);
                        if decoded.get(index) != Some(s) {
                            ok = false;
                            break;
                        }
                        index += 1;
                    }
                    Part::One(name) => {
                        rank.push(1);
                        if let Some(value) = decoded.get(index) {
                            params.push((name.clone(), value.clone()));
                            index += 1;
                        } else {
                            ok = false;
                            break;
                        }
                    }
                    Part::Rest(name) => {
                        rank.push(0);
                        params.push((name.clone(), decoded[index..].join("/")));
                        index = decoded.len();
                        break;
                    }
                }
            }
            if ok && index == decoded.len() {
                if !entry.parts.iter().any(|p| matches!(p, Part::Rest(_))) {
                    rank.push(3);
                }
                candidates.push((id, rank, params, entry.trailing));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let mut method_set = std::collections::BTreeSet::new();
        for (id, _, _, _) in &candidates {
            method_set.insert(self.entries[*id].method.clone());
        }
        let chosen_method = if method == "HEAD" && method_set.contains("HEAD") {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };
        let mut matching: Vec<_> = candidates
            .into_iter()
            .filter(|(id, _, _, _)| self.entries[*id].method == chosen_method)
            .collect();

        if matching.is_empty() {
            if method_set.contains("GET") {
                method_set.insert("HEAD".to_owned());
            }
            return Err(Error::MethodNotAllowed {
                allowed: method_set.into_iter().collect(),
            });
        }

        matching.sort_by(|a, b| b.1.cmp(&a.1).then(a.0.cmp(&b.0)));
        let (id, _, params, route_trailing) = matching.remove(0);
        let redirect = if self.policy == SlashPolicy::Redirect
            && route_trailing != trailing
            && !self.entries[id].parts.is_empty()
        {
            let mut dest = head.to_owned();
            if route_trailing {
                dest.push('/');
            } else {
                dest.pop();
            }
            Some(dest)
        } else {
            None
        };
        Ok(Match { route: id, params, redirect })
    }
}

fn raw_start_for_index(pattern: &str, wanted: usize) -> usize {
    let bytes = pattern.as_bytes();
    let mut index = 0;
    let mut start = 1;
    while start < bytes.len() {
        if index == wanted {
            return start;
        }
        while start < bytes.len() && bytes[start] != b'/' {
            start += 1;
        }
        index += 1;
        start += 1;
    }
    pattern.len()
}

        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(raw: &str, base: usize, path_error: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(if path_error {
                    Error::InvalidPath { at: base + i }
                } else {
                    Error::InvalidPattern { at: base + i }
                });
            }
            let hi = hex_upper(bytes[i + 1]);
            let lo = hex_upper(bytes[i + 2]);
            let Some(value) = hi.zip(lo).map(|(h, l)| (h << 4) | l) else {
                return Err(if path_error {
                    Error::InvalidPath { at: base + i }
                } else {
                    Error::InvalidPattern { at: base + i }
                });
            };
            if !(0x20..=0x7e).contains(&value) || value == b'/' || unreserved(value) {
                return Err(if path_error {
                    Error::InvalidPath { at: base + i }
                } else {
                    Error::InvalidPattern { at: base + i }
                });
            }
            out.push(value);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(if path_error {
                Error::InvalidPath { at: base + i }
            } else {
                Error::InvalidPattern { at: base + i }
            });
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn invalid_path(at: usize, path_error: bool) -> Error {
    if path_error {
        Error::InvalidPath { at }
    } else {
        Error::InvalidPattern { at }
    }
}

fn split_head(input: &str, path_error: bool) -> Result<(Vec<(String, usize)>, bool), Error> {
    if !input.starts_with('/') {
        return Err(invalid_path(0, path_error));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let mut trailing = false;
    let mut end = bytes.len();
    if bytes[end - 1] == b'/' {
        trailing = true;
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut start = 1;
    while start <= end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(invalid_path(start, path_error));
        }
        segments.push((input[start..stop].to_string(), start));
        if stop == end {
            break;
        }
        start = stop + 1;
    }
    Ok((segments, trailing))
}

fn request_head(path: &str) -> &str {
    let end = path
        .bytes()
        .position(|b| b == b'?' || b == b'#')
        .unwrap_or(path.len());
    &path[..end]
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let (raw_segments, trailing) = split_head(pattern, false)?;
        let mut segments = Vec::with_capacity(raw_segments.len());
        let mut seen = std::collections::HashSet::new();
        let mut has_wildcard = false;

        for (index, (raw, start)) in raw_segments.iter().enumerate() {
            let bytes = raw.as_bytes();
            if bytes[0] == b':' || bytes[0] == b'*' {
                let is_wildcard = bytes[0] == b'*';
                let name = &raw[1..];
                if !name_valid(name) || !seen.insert(name.to_string()) {
                    return Err(Error::InvalidPattern { at: *start });
                }
                if is_wildcard && index + 1 != raw_segments.len() {
                    return Err(Error::InvalidPattern { at: *start });
                }
                if is_wildcard {
                    has_wildcard = true;
                    segments.push(Segment::Wildcard(name.to_string()));
                } else {
                    segments.push(Segment::Param(name.to_string()));
                }
            } else {
                segments.push(Segment::Static(decode_segment(raw, *start, false)?));
            }
        }

        let mut rank = Vec::with_capacity(segments.len() + 1);
        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_string(),
            segments,
            trailing,
            has_wildcard,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head = request_head(path);
        let (raw_segments, trailing) = split_head(head, true)?;
        let mut decoded = Vec::with_capacity(raw_segments.len());
        for (raw, start) in &raw_segments {
            decoded.push(decode_segment(raw, *start, true)?);
        }

        let mut candidates = Vec::new();
        for (id, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if !route.has_wildcard && route.segments.len() != decoded.len() {
                continue;
            }
            if route.has_wildcard {
                if route.segments.len() - 1 > decoded.len() {
                    continue;
                }
            } else if route.segments.len() != decoded.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut ok = true;
            for (i, seg) in route.segments.iter().enumerate() {
                match seg {
                    Segment::Static(value) => {
                        if decoded.get(i) != Some(value) {
                            ok = false;
                            break;
                        }
                    }
                    Segment::Param(name) => {
                        if let Some(value) = decoded.get(i) {
                            params.push((name.clone(), value.clone()));
                        } else {
                            ok = false;
                            break;
                        }
                    }
                    Segment::Wildcard(name) => {
                        let value = decoded[i..].join("/");
                        params.push((name.clone(), value));
                    }
                }
            }
            if ok {
                candidates.push((id, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let mut eligible: Vec<(usize, Vec<(String, String)>)> = if method == "HEAD" {
            let has_head = candidates
                .iter()
                .any(|(id, _)| self.routes[*id].method == "HEAD");
            let wanted = if has_head { "HEAD" } else { "GET" };
            candidates
                .into_iter()
                .filter(|(id, _)| self.routes[*id].method == wanted)
                .collect()
        } else {
            candidates
                .into_iter()
                .filter(|(id, _)| self.routes[*id].method == method)
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed = std::collections::BTreeSet::new();
            for (id, _) in self.routes.iter().enumerate() {
                let route = &self.routes[id];
                let path_matches = {
                    if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                        false
                    } else if route.has_wildcard {
                        if route.segments.len() - 1 > decoded.len() {
                            false
                        } else {
                            route.segments.iter().enumerate().all(|(i, seg)| match seg {
                                Segment::Static(v) => decoded.get(i) == Some(v),
                                Segment::Param(_) => decoded.get(i).is_some(),
                                Segment::Wildcard(_) => true,
                            })
                        }
                    } else {
                        route.segments.len() == decoded.len()
                            && route.segments.iter().enumerate().all(|(i, seg)| match seg {
                                Segment::Static(v) => decoded.get(i) == Some(v),
                                Segment::Param(_) => true,
                                Segment::Wildcard(_) => false,
                            })
                    }
                };
                if path_matches {
                    allowed.insert(route.method.clone());
                }
            }
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_string());
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        let mut best = eligible.remove(0);
        for candidate in eligible {
            let a = &self.routes[candidate.0].rank;
            let b = &self.routes[best.0].rank;
            if a > b {
                best = candidate;
            }
        }

        let route = &self.routes[best.0];
        let redirect = if self.policy == SlashPolicy::Redirect && route.trailing != trailing {
            let mut target = head.to_string();
            if trailing {
                target.pop();
            } else if head != "/" {
                target.push('/');
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: best.0,
            params: best.1,
            redirect,
        })
    }
}

        let head_end = path
            .bytes()
            .position(|b| b == b'?' || b == b'#')
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request, slash) = read_request(head)?;

        let mut matches = Vec::new();
        for (index, entry) in self.entries.iter().enumerate() {
            if self.slash_policy == SlashPolicy::Strict && entry.slash != slash {
                continue;
            }
            if let Some(captures) = find_match(entry, &request) {
                matches.push((index, captures));
            }
        }

        if matches.is_empty() {
            return Err(Error::NotFound);
        }

        let selected_method = if method == "HEAD"
            && !matches.iter().any(|(i, _)| self.entries[*i].method == "HEAD")
        {
            "GET"
        } else {
            method
        };

        let applicable: Vec<_> = matches
            .iter()
            .filter(|(i, _)| self.entries[*i].method == selected_method)
            .collect();

        if applicable.is_empty() {
            let mut allowed = matches
                .iter()
                .map(|(i, _)| self.entries[*i].method.clone())
                .collect::<Vec<_>>();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let mut winner = applicable[0];
        for candidate in applicable.iter().skip(1) {
            if self.entries[candidate.0].score > self.entries[winner.0].score {
                winner = candidate;
            }
        }
        let entry = &self.entries[winner.0];
        let redirect = if self.slash_policy == SlashPolicy::Redirect && entry.slash != slash {
            let mut new_path = head.to_owned();
            if entry.slash {
                new_path.push('/');
            } else {
                new_path.pop();
            }
            Some(new_path)
        } else {
            None
        };

        Ok(Match {
            route: winner.0,
            params: winner.1.clone(),
            redirect,
        })
    }
}

fn check_method(method: &str) -> Result<(), Error> {
    if method.is_empty() || !method.bytes().all(|b| (b'A'..=b'Z').contains(&b)) {
        Err(Error::InvalidMethod)
    } else {
        Ok(())
    }
}

fn valid_capture(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn read_pattern(pattern: &str) -> Result<(Vec<Part>, bool), Error> {
    read_segments(pattern, true).map_err(|at| Error::InvalidPattern { at })
}

fn read_request(path: &str) -> Result<(Vec<String>, bool), Error> {
    let (parts, slash) = read_segments(path, false).map_err(|at| Error::InvalidPath { at })?;
    let decoded = parts
        .into_iter()
        .map(|part| match part {
            Part::Literal(text) => text,
            _ => unreachable!(),
        })
        .collect();
    Ok((decoded, slash))
}

fn read_segments(input: &str, is_pattern: bool) -> Result<(Vec<Part>, bool), usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let raw = input.as_bytes();
    let slash = raw.last() == Some(&b'/');
    let limit = if slash { raw.len() - 1 } else { raw.len() };
    let mut parts = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut pos = 1;

    while pos < limit {
        let next = raw[pos..limit]
            .iter()
            .position(|b| *b == b'/')
            .map_or(limit, |x| pos + x);

        if next == pos {
            return Err(pos);
        }

        let segment = &input[pos..next];
        if is_pattern && (segment.starts_with(':') || segment.starts_with('*')) {
            let marker = segment.as_bytes()[0];
            let name = &segment[1..];
            if !valid_capture(name) || !seen.insert(name.to_string()) {
                return Err(pos);
            }
            if marker == b'*' && next != limit {
                return Err(pos);
            }
            parts.push(if marker == b'*' {
                Part::Rest(name.into())
            } else {
                Part::Capture(name.into())
            });
        } else {
            parts.push(Part::Literal(unescape(segment, pos)?));
        }

        pos = next + 1;
    }

    if slash && parts.is_empty() {
        return Err(1);
    }
    Ok((parts, slash))
}

fn unescape(segment: &str, start: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut output = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        match bytes[i] {
            b'%' => {
                if i + 2 >= bytes.len() {
                    return Err(start + i);
                }
                let hi = hex(bytes[i + 1]).ok_or(start + i)?;
                let lo = hex(bytes[i + 2]).ok_or(start + i)?;
                let value = hi * 16 + lo;
                if !(0x20..=0x7e).contains(&value)
                    || value == b'/'
                    || unreserved(value)
                {
                    return Err(start + i);
                }
                output.push(value);
                i += 3;
            }
            b if unreserved(b) => {
                output.push(b);
                i += 1;
            }
            _ => return Err(start + i),
        }
    }
    Ok(String::from_utf8(output).unwrap())
}

fn hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn find_match(entry: &Entry, request: &[String]) -> Option<Vec<(String, String)>> {
    let rest = matches!(entry.parts.last(), Some(Part::Rest(_)));
    let fixed = entry.parts.len() - usize::from(rest);
    if (!rest && request.len() != fixed) || (rest && request.len() < fixed) {
        return None;
    }

    let mut captures = Vec::new();
    for (part, value) in entry.parts.iter().take(fixed).zip(request) {
        match part {
            Part::Literal(s) if s != value => return None,
            Part::Literal(_) => {}
            Part::Capture(name) => captures.push((name.clone(), value.clone())),
            Part::Rest(_) => return None,
        }
    }
    if let Some(Part::Rest(name)) = entry.parts.last() {
        captures.push((name.clone(), request[fixed..].join("/")));
    }
    Some(captures)
}

struct Route {

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {

            policy,
            routes: Vec::new(),

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;
        let (segments, trailing) = parse_pattern(pattern)?;
        let wildcard = matches!(segments.last(), Some(Segment::Wildcard(_)));
        let mut rank: Vec<u8> = segments
            .iter()
            .map(|s| match s {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            })
            .collect();
        if !wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_owned(),
            segments,
            trailing,
            rank,
            wildcard,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, trailing) = parse_path(head)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if let Some(params) = match_route(route, &request_segments) {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<usize> = if method == "HEAD" {
            if candidates.iter().any(|(r, _)| r.method == "HEAD") {
                candidates
                    .iter()
                    .enumerate()
                    .filter_map(|(i, (r, _))| (r.method == "HEAD").then_some(i))
                    .collect()
            } else {
                candidates
                    .iter()
                    .enumerate()
                    .filter_map(|(i, (r, _))| (r.method == "GET").then_some(i))
                    .collect()
            }
        } else {
            candidates
                .iter()
                .enumerate()
                .filter_map(|(i, (r, _))| (r.method == method).then_some(i))
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(r, _)| r.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let mut best = eligible[0];
        for &i in &eligible[1..] {
            if candidates[i].0.rank > candidates[best].0.rank {
                best = i;
            }
        }
        let (route, params) = &candidates[best];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && route.trailing != trailing
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }
}

        Err(Error::InvalidMethod)
    } else {
        Ok(())
    }
}

fn name_valid(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| matches!(b, b'a'..=b'z' | b'0'..=b'9' | b'_'))
}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    parse_segments(input, true).map_err(|at| Error::InvalidPattern { at })
}

fn parse_path(input: &str) -> Result<(Vec<String>, bool), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    let (segments, trailing) =
        parse_segments(input, false).map_err(|at| Error::InvalidPath { at })?;
    let decoded = segments
        .into_iter()
        .map(|s| match s {
            Segment::Static(v) => v,
            _ => unreachable!(),
        })
        .collect();
    Ok((decoded, trailing))
}

fn parse_segments(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), usize> {
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };

    let mut out = Vec::new();
    let mut names = std::collections::HashSet::<String>::new();
    let mut start = 1;
    let mut wildcard_seen = false;

    while start < end {
        let seg_end = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|n| start + n)
            .unwrap_or(end);

        if seg_end == start {
            return Err(start);
        }

        let raw = &input[start..seg_end];
        if pattern && wildcard_seen {
            return Err(start);
        }

        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !name_valid(name) || !names.insert(name.to_owned()) {
                return Err(start);
            }
            if marker == b'*' && seg_end != end {
                return Err(start);
            }
            if marker == b'*' {
                wildcard_seen = true;
                out.push(Segment::Wildcard(name.to_owned()));
            } else {
                out.push(Segment::Param(name.to_owned()));
            }
        } else {
            let decoded = decode_static(raw, start)?;
            out.push(Segment::Static(decoded));
        }

        start = seg_end + 1;
    }

    if end == 1 && !trailing {
        // The only non-root path with no segments cannot occur, but keep parsing defensive.
        return Ok((out, trailing));
    }
    if trailing && out.is_empty() {
        return Err(1);
    }

    Ok((out, trailing))
}

fn decode_static(raw: &str, offset: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut result = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(offset + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(offset + i)?;
            let decoded = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(offset + i);
            }
            result.push(decoded);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(offset + i);
            }
            result.push(b);
            i += 1;
        }
    }

    Ok(String::from_utf8(result).expect("decoded bytes are ASCII"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let has_wildcard = route.wildcard;
    let fixed_len = if has_wildcard {
        route.segments.len() - 1
    } else {
        route.segments.len()
    };

    if (!has_wildcard && request.len() != fixed_len) || (has_wildcard && request.len() < fixed_len) {
        return None;
    }

    for (segment, value) in route.segments.iter().take(fixed_len).zip(request.iter()) {
        match segment {
            Segment::Static(expected) if expected != value => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), value.clone())),
            Segment::Wildcard(_) => return None,
        }
    }

    if let Some(Segment::Wildcard(name)) = route.segments.last() {
        let value = request[fixed_len..].join("/");
        params.push((name.clone(), value));
    }

    Some(params)
}

struct Route {

pub struct Router {

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self { policy, routes: Vec::new() }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing, has_wildcard) =
            parse_pattern(pattern).map_err(|at| Error::InvalidPattern { at })?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }
        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_owned(),
            segments,
            trailing,
            has_wildcard,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (head, request_segments, request_trailing) =
            parse_path(path).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(usize, Vec<(String, String)>)> = Vec::new();
        for (idx, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            if let Some(params) = match_route(route, &request_segments) {
                candidates.push((idx, params));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let mut eligible: Vec<usize> = if method == "HEAD" {
            let has_head = candidates.iter().any(|(idx, _)| self.routes[*idx].method == "HEAD");
            let wanted = if has_head { "HEAD" } else { "GET" };
            candidates.iter().enumerate()
                .filter_map(|(i, (idx, _))| (self.routes[*idx].method == wanted).then_some(i))
                .collect()
        } else {
            candidates.iter().enumerate()
                .filter_map(|(i, (idx, _))| (self.routes[*idx].method == method).then_some(i))
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (idx, _) in &candidates {
                allowed.insert(self.routes[*idx].method.clone());
            }
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_owned());
            }
            return Err(Error::MethodNotAllowed { allowed: allowed.into_iter().collect() });
        }

        eligible.sort_by(|a, b| {
            let ra = &self.routes[candidates[*a].0].rank;
            let rb = &self.routes[candidates[*b].0].rank;
            rb.cmp(ra).then_with(|| candidates[*a].0.cmp(&candidates[*b].0))
        });
        let (route_index, params) = candidates[eligible[0]].clone();
        let route = &self.routes[route_index];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !head_is_root(head)
            && route.trailing != request_trailing
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {
            None
        };
        Ok(Match { route: route_index, params, redirect })
    }

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn parse_path(input: &str) -> Result<(&str, Vec<String>, bool), usize> {
    let end = input.bytes().position(|b| b == b'?' || b == b'#').unwrap_or(input.len());
    let head = &input[..end];
    let segments = parse_segments(head, false)?;
    let trailing = head != "/" && head.ends_with('/');
    Ok((head, segments, trailing))
}

fn head_is_root(head: &str) -> bool {
    head == "/"
}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool, bool), usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false, false));
    }
    let trailing = input.ends_with('/');
    let body_end = if trailing { input.len() - 1 } else { input.len() };
    let mut raw: Vec<(usize, &str)> = Vec::new();
    let mut start = 1;
    for (i, b) in input.as_bytes().iter().enumerate().take(body_end).skip(1) {
        if *b == b'/' {
            if i == start {
                return Err(start);
            }
            raw.push((start, &input[start..i]));
            start = i + 1;
        }
    }
    if start < body_end {
        raw.push((start, &input[start..body_end]));
    } else if !trailing {
        return Err(start);
    }
    let mut segments = Vec::with_capacity(raw.len());
    let mut names = BTreeSet::new();
    let mut wildcard = false;
    for (i, (at, text)) in raw.iter().enumerate() {
        if text.starts_with(':') || text.starts_with('*') {
            let marker = text.as_bytes()[0];
            let name = &text[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(*at);
            }
            if marker == b'*' {
                if i + 1 != raw.len() {
                    return Err(*at);
                }
                wildcard = true;
                segments.push(Segment::Wildcard(name.to_owned()));
            } else {
                segments.push(Segment::Param(name.to_owned()));
            }
        } else {
            segments.push(Segment::Static(parse_encoded_segment(text, *at)?));
        }
    }
    Ok((segments, trailing, wildcard))
}

fn parse_segments(head: &str, _unused: bool) -> Result<Vec<String>, usize> {
    if !head.starts_with('/') {
        return Err(0);
    }
    if head == "/" {
        return Ok(Vec::new());
    }
    let trailing = head.ends_with('/');
    let end = if trailing { head.len() - 1 } else { head.len() };
    let mut out = Vec::new();
    let mut start = 1;
    for (i, b) in head.as_bytes().iter().enumerate().take(end).skip(1) {
        if *b == b'/' {
            if i == start {
                return Err(start);
            }
            out.push(parse_encoded_segment(&head[start..i], start)?);
            start = i + 1;
        }
    }
    if start < end {
        out.push(parse_encoded_segment(&head[start..end], start)?);
    } else if !trailing {
        return Err(start);
    }
    Ok(out)
}

fn parse_encoded_segment(text: &str, base: usize) -> Result<String, usize> {
    let bytes = text.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(base + i);
            }
            let value = (hex(bytes[i + 1]) << 4) | hex(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            out.push(value);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn is_upper_hex(b: u8) -> bool {

fn hex(b: u8) -> u8 {
    if b.is_ascii_digit() { b - b'0' } else { b - b'A' + 10 }
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b[0].is_ascii_lowercase()
        && b[1..].iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_')
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut pos = 0;
    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if request.get(pos)? != value { return None; }
                pos += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), request.get(pos)?.clone()));
                pos += 1;
            }
            Segment::Wildcard(name) => {
                let value = request[pos..].join("/");
                params.push((name.clone(), value));
                pos = request.len();
            }
        }
    }
    (pos == request.len()).then_some(params)
}

use std::collections::BTreeSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self { policy, routes: Vec::new() }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing) = parse_pattern(pattern)
            .map_err(|at| Error::InvalidPattern { at })?;
        let has_wildcard = segments.iter().any(|s| matches!(s, Segment::Wildcard(_)));
        let mut rank: Vec<u8> = segments.iter().map(|s| match s {
            Segment::Static(_) => 2,
            Segment::Param(_) => 1,
            Segment::Wildcard(_) => 0,
        }).collect();
        if !has_wildcard {
            rank.push(3);
        }
        let id = self.routes.len();
        self.routes.push(Route { method: method.to_owned(), segments, trailing, rank });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let end = path.bytes().position(|b| b == b'?' || b == b'#').unwrap_or(path.len());
        let head = &path[..end];
        let (request, request_trailing) =
            parse_request_head(head).map_err(|at| Error::InvalidPath { at })?;

        let candidates: Vec<(usize, Vec<(String, String)>)> = self.routes.iter().enumerate()
            .filter_map(|(i, route)| {
                if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                    return None;
                }
                match_route(route, &request).map(|p| (i, p))
            }).collect();

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD" {
            if candidates.iter().any(|(i, _)| self.routes[*i].method == "HEAD") {
                "HEAD"
            } else {
                "GET"
            }
        } else {
            method
        };
        let mut eligible: Vec<_> = candidates.iter()
            .filter(|(i, _)| self.routes[*i].method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (i, _) in &candidates {
                allowed.insert(self.routes[*i].method.clone());
            }
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_owned());
            }
            return Err(Error::MethodNotAllowed { allowed: allowed.into_iter().collect() });
        }

        eligible.sort_by(|(ia, _), (ib, _)| {
            self.routes[*ib].rank.cmp(&self.routes[*ia].rank).then_with(|| ia.cmp(ib))
        });
        let (index, params) = eligible[0];
        let route = &self.routes[*index];
        let redirect = if self.policy == SlashPolicy::Redirect
            && head != "/"
            && route.trailing != request_trailing
        {
            let mut target = head.to_owned();
            if route.trailing { target.push('/'); } else { target.pop(); }
            Some(target)
        } else {
            None
        };
        Ok(Match { route: *index, params: params.clone(), redirect })
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b'A' <= b && b <= b'Z')
}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool), usize> {
    let (parts, trailing) = split_head(input)?;
    let mut segments = Vec::with_capacity(parts.len());
    let mut names = BTreeSet::new();
    for (i, (at, part)) in parts.iter().enumerate() {
        if part.starts_with(':') || part.starts_with('*') {
            let marker = part.as_bytes()[0];
            let name = &part[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(*at);
            }
            if marker == b'*' && i + 1 != parts.len() {
                return Err(*at);
            }
            segments.push(if marker == b'*' {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            segments.push(Segment::Static(parse_encoded_segment(part, *at)?));
        }
    }
    Ok((segments, trailing))
}

fn parse_request_head(input: &str) -> Result<(Vec<String>, bool), usize> {
    let (parts, trailing) = split_head(input)?;
    let mut segments = Vec::with_capacity(parts.len());
    for (at, part) in parts {
        segments.push(parse_encoded_segment(part, at)?);
    }
    Ok((segments, trailing))
}

fn split_head(input: &str) -> Result<(Vec<(usize, &str)>, bool), usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }
    let trailing = input.ends_with('/');
    let limit = input.len() - usize::from(trailing);
    let bytes = input.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1;
    let mut i = 1;
    while i < limit {
        if bytes[i] == b'/' {
            if i == start {
                return Err(start);
            }
            parts.push((start, &input[start..i]));
            start = i + 1;
        }
        i += 1;
    }
    if start < limit {
        parts.push((start, &input[start..limit]));
    } else if !trailing {
        return Err(start);
    }
    Ok((parts, trailing))
}

fn parse_encoded_segment(text: &str, base: usize) -> Result<String, usize> {
    let bytes = text.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() || !upper_hex(bytes[i + 1]) || !upper_hex(bytes[i + 2]) {
                return Err(base + i);
            }
            let v = (hex(bytes[i + 1]) << 4) | hex(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&v) || v == b'/' || unreserved(v) {
                return Err(base + i);
            }
            decoded.push(v);
            i += 3;
        } else if unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(decoded).unwrap())
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty() && b[0].is_ascii_lowercase()
        && b[1..].iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn upper_hex(b: u8) -> bool {
    b.is_ascii_digit() || (b'A'..=b'F').contains(&b)
}

fn hex(b: u8) -> u8 {
    if b.is_ascii_digit() { b - b'0' } else { b - b'A' + 10 }
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let mut captures = Vec::new();
    let mut p = 0;
    for segment in &route.segments {
        match segment {
            Segment::Static(expected) => {
                if request.get(p)? != expected { return None; }
                p += 1;
            }
            Segment::Param(name) => {
                captures.push((name.clone(), request.get(p)?.clone()));
                p += 1;
            }
            Segment::Wildcard(name) => {
                captures.push((name.clone(), request[p..].join("/")));
                p = request.len();
            }
        }
    }
    if p == request.len() { Some(captures) } else { None }
}
