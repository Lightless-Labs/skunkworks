    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },

        Self {

            digits: vec![0],

        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.len() == 1 && self.digits[0] == 0 {
            self.negative = false;
        }

        self.digits.len() == 1 && self.digits[0] == 0

            Ordering::Equal => {
                for (a, b) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u16;
        for i in 0..a.len().max(b.len()) {
            let sum = carry
                + u16::from(*a.get(i).unwrap_or(&0))
                + u16::from(*b.get(i).unwrap_or(&0));
            out.push((sum % 10) as u8);
            carry = sum / 10;
        }
        if carry != 0 {
            out.push(carry as u8);
        }
        out
    }

    fn sub_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len());

        for i in 0..a.len() {
            let mut digit = i16::from(a[i]) - borrow - i16::from(*b.get(i).unwrap_or(&0));
            if digit < 0 {
                digit += 10;
                borrow = 1;

                borrow = 0;
            }

            out.pop();
        }
        out

            return Err(Error::Separator { index });

            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),

        self.add(&neg_rhs)

        } else if self.negative {
            -1
        } else {
            1
        }

            return Self::zero();

                    digits: vec![1],

            negative: self.negative,
            digits: remainder_le,

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1,000,000,000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = match bytes[0] {
            b'+' | b'-' => 1,
            _ => 0,
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b != b'_' && !b.is_ascii_digit() {
                return Err(Error::InvalidDigit { index: i });
            }
        }

        let mut bad_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && !(i > start
                    && i + 1 < bytes.len()
                    && bytes[i - 1].is_ascii_digit()
                    && bytes[i + 1].is_ascii_digit())
            {
                bad_separator.get_or_insert(i);
            }
        }
        if let Some(index) = bad_separator {
            return Err(Error::Separator { index });
        }

        let digit_positions: Vec<usize> = (start..bytes.len())
            .filter(|&i| bytes[i].is_ascii_digit())
            .collect();

        if digit_positions.len() > 1 && bytes[digit_positions[0]] == b'0' {
            return Err(Error::LeadingZero { index: digit_positions[0] });
        }
        if digit_positions.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_positions[max_digits],
            });
        }

        let mut limbs = vec![0u32];
        let mut digits = 0usize;
        for &b in &bytes[start..] {
            if b.is_ascii_digit() {
                mul_small_in_place(&mut limbs, 10);
                add_small_in_place(&mut limbs, u32::from(b - b'0'));
                digits += 1;
            }
        }
        debug_assert_eq!(digits, digit_positions.len());
        let negative = bytes[0] == b'-' && !(limbs.len() == 1 && limbs[0] == 0);
        Ok(Self { negative, limbs })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let mut body = self.limbs.last().copied().unwrap_or(0).to_string();
        for limb in self.limbs.iter().rev().skip(1) {
            body.push_str(&format!("{:09}", limb));
        }
        if body.len() < min_digits {
            let mut padded = String::with_capacity(min_digits);
            padded.extend(std::iter::repeat_n('0', min_digits - body.len()));
            padded.push_str(&body);
            body = padded;
        }

        let mut out = String::with_capacity(body.len().saturating_add(
            if group > 0 { (body.len() - 1) / group } else { 0 }
                + usize::from(self.negative),
        ));
        if self.negative {
            out.push('-');
        }
        for (i, ch) in body.chars().enumerate() {
            if group > 0 && i > 0 && (body.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(ch);
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_mag(&self.limbs, &rhs.limbs),
            (true, true) => cmp_mag(&rhs.limbs, &self.limbs),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::from_parts(
                self.negative,
                add_mag(&self.limbs, &rhs.limbs),
            )
        } else {
            match cmp_mag(&self.limbs, &rhs.limbs) {
                Ordering::Greater => Self::from_parts(
                    self.negative,
                    sub_mag(&self.limbs, &rhs.limbs),
                ),
                Ordering::Less => Self::from_parts(
                    rhs.negative,
                    sub_mag(&rhs.limbs, &self.limbs),
                ),
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            self.clone()
        } else {
            Self { negative: !self.negative, limbs: self.limbs.clone() }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u64; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let n = out[k] + u64::from(a) * u64::from(b) + carry;
                out[k] = n % Self::BASE;
                carry = n / Self::BASE;
            }
            let mut k = i + rhs.limbs.len();
            while carry > 0 {
                if k == out.len() {
                    out.push(0);
                }
                let n = out[k] + carry;
                out[k] = n % Self::BASE;
                carry = n / Self::BASE;
                k += 1;
            }
        }
        Self::from_parts(
            self.negative ^ rhs.negative,
            out.into_iter().map(|x| x as u32).collect(),
        )
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        match cmp_mag(&self.limbs, &rhs.limbs) {
            Ordering::Less => return Ok((Self::zero(), self.clone())),
            Ordering::Equal => {
                return Ok((
                    Self::from_parts(self.negative ^ rhs.negative, vec![1]),
                    Self::zero(),
                ));
            }
            Ordering::Greater => {}
        }

        let mut quotient_be = Vec::with_capacity(self.limbs.len());
        let mut remainder = vec![0u32];
        for &limb in self.limbs.iter().rev() {
            remainder = mul_base_add(&remainder, limb);
            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u32;
            while low <= high {
                let mid = low + (high - low) / 2;
                match cmp_mag(&mul_small(&rhs.limbs, mid), &remainder) {
                    Ordering::Greater => {
                        if mid == 0 { break; }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid as u32;
                        low = mid + 1;
                    }
                }
            }
            quotient_be.push(best);
            remainder = sub_mag(&remainder, &mul_small(&rhs.limbs, u64::from(best)));
        }
        quotient_be.reverse();
        Ok((
            Self::from_parts(self.negative ^ rhs.negative, quotient_be),
            Self::from_parts(self.negative, remainder),
        ))
    }

    fn zero() -> Self {
        Self { negative: false, limbs: vec![0] }
    }

    fn from_parts(negative: bool, mut limbs: Vec<u32>) -> Self {
        normalize(&mut limbs);
        let zero = limbs.len() == 1 && limbs[0] == 0;
        Self { negative: negative && !zero, limbs }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }
}

fn normalize(limbs: &mut Vec<u32>) {
    while limbs.len() > 1 && limbs.last() == Some(&0) {
        limbs.pop();
    }
    if limbs.is_empty() {
        limbs.push(0);
    }
}

fn cmp_mag(a: &[u32], b: &[u32]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => {
            for i in (0..a.len()).rev() {
                match a[i].cmp(&b[i]) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
    let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u64;
    for i in 0..a.len().max(b.len()) {
        let n = u64::from(a.get(i).copied().unwrap_or(0))
            + u64::from(b.get(i).copied().unwrap_or(0))
            + carry;
        out.push((n % Big::BASE) as u32);
        carry = n / Big::BASE;
    }
    if carry != 0 {
        out.push(carry as u32);
    }
    out
}

fn sub_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
    debug_assert!(cmp_mag(a, b) != Ordering::Less);
    let mut out = Vec::with_capacity(a.len());
    let mut borrow = 0i64;
    for i in 0..a.len() {
        let n = i64::from(a[i]) - i64::from(b.get(i).copied().unwrap_or(0)) - borrow;
        if n < 0 {
            out.push((n + Big::BASE as i64) as u32);
            borrow = 1;
        } else {
            out.push(n as u32);
            borrow = 0;
        }
    }
    normalize(&mut out);
    out
}

fn mul_small_in_place(a: &mut Vec<u32>, factor: u32) {
    let mut carry = 0u64;
    for limb in a.iter_mut() {
        let n = u64::from(*limb) * u64::from(factor) + carry;
        *limb = (n % Big::BASE) as u32;
        carry = n / Big::BASE;
    }
    if carry != 0 {
        a.push(carry as u32);
    }
}

fn add_small_in_place(a: &mut Vec<u32>, value: u32) {
    let mut carry = u64::from(value);
    let mut i = 0;
    while carry > 0 {
        if i == a.len() {
            a.push(0);
        }
        let n = u64::from(a[i]) + carry;
        a[i] = (n % Big::BASE) as u32;
        carry = n / Big::BASE;
        i += 1;
    }
}

fn mul_small(a: &[u32], factor: u64) -> Vec<u32> {
    if factor == 0 {
        return vec![0];
    }
    let mut out = Vec::with_capacity(a.len() + 1);
    let mut carry = 0u128;
    for &limb in a {
        let n = u128::from(limb) * u128::from(factor) + carry;
        out.push((n % u128::from(Big::BASE)) as u32);
        carry = n / u128::from(Big::BASE);
    }
    while carry > 0 {
        out.push((carry % u128::from(Big::BASE)) as u32);
        carry /= u128::from(Big::BASE);
    }
    normalize(&mut out);
    out
}

fn mul_base_add(a: &[u32], limb: u32) -> Vec<u32> {
    let mut out = Vec::with_capacity(a.len() + 1);
    out.push(limb);
    out.extend_from_slice(a);
    normalize(&mut out);
    out
}

use std::cmp::Ordering;

const BASE: u64 = 1_000_000_000;
const WIDTH: usize = 9;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1,000,000,000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    fn zero() -> Self {
        Self {
            negative: false,
            limbs: vec![0],
        }
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.len() == 1 && self.limbs[0] == 0 {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let sum = u64::from(*a.get(i).unwrap_or(&0))
                + u64::from(*b.get(i).unwrap_or(&0))
                + carry;
            out.push((sum % BASE) as u32);
            carry = sum / BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        out
    }

    // Requires a >= b.
    fn sub_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let mut digit =
                i64::from(a[i]) - borrow - i64::from(*b.get(i).unwrap_or(&0));
            if digit < 0 {
                digit += BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(digit as u32);
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_abs_small(a: &[u32], factor: u32) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u64;
        for &limb in a {
            let value = u64::from(limb) * u64::from(factor) + carry;
            out.push((value % BASE) as u32);
            carry = value / BASE;
        }
        while carry != 0 {
            out.push((carry % BASE) as u32);
            carry /= BASE;
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn cmp_limbs(a: &[u32], b: &[u32]) -> Ordering {
        match a.len().cmp(&b.len()) {
            Ordering::Equal => {
                for (x, y) in a.iter().rev().zip(b.iter().rev()) {
                    match x.cmp(y) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn div_abs(numerator: &[u32], denominator: &[u32]) -> (Vec<u32>, Vec<u32>) {
        let mut remainder = vec![0u32];
        let mut quotient_be = Vec::with_capacity(numerator.len());

        for &limb in numerator.iter().rev() {
            // remainder = remainder * BASE + limb, in little-endian form.
            if remainder.len() == 1 && remainder[0] == 0 {
                remainder[0] = limb;
            } else {
                remainder.insert(0, limb);
            }
            while remainder.len() > 1 && remainder.last() == Some(&0) {
                remainder.pop();
            }

            // Binary search the quotient limb in [0, BASE - 1].
            let mut low = 0u64;
            let mut high = BASE - 1;
            while low < high {
                let mid = (low + high + 1) / 2;
                let product = Self::mul_abs_small(&denominator, mid as u32);
                if Self::cmp_limbs(&product, &remainder) != Ordering::Greater {
                    low = mid;
                } else {
                    high = mid - 1;
                }
            }
            let q_limb = low as u32;
            if q_limb != 0 {
                let product = Self::mul_abs_small(denominator, q_limb);
                remainder = Self::sub_abs(&remainder, &product);
            }
            quotient_be.push(q_limb);
        }

        while quotient_be.len() > 1 && quotient_be[0] == 0 {
            quotient_be.remove(0);
        }
        (quotient_be.into_iter().rev().collect(), remainder)
    }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        if let Some(index) =
            (start..bytes.len()).find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())
        {
            return Err(Error::InvalidDigit { index });
        }

        if let Some(index) = (start..bytes.len()).find(|&i| {
            bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
        }) {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        if digit_count > 1 {
            let first = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
            if bytes[first] == b'0' {
                return Err(Error::LeadingZero { index: first });
            }
        }

        if digit_count > max_digits {
            let index = (start..bytes.len())
                .filter(|&i| bytes[i] != b'_')
                .nth(max_digits)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut limbs = vec![0u32];
        for &b in &bytes[start..] {
            if b == b'_' {
                continue;
            }
            let mut carry = u64::from(b - b'0');
            for limb in &mut limbs {
                let value = u64::from(*limb) * 10 + carry;
                *limb = (value % BASE) as u32;
                carry = value / BASE;
            }
            if carry != 0 {
                limbs.push(carry as u32);
            }
        }

        let mut result = Self { negative, limbs };
        result.normalize();
        Ok(result)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let mut magnitude = String::new();
        let last = *self.limbs.last().unwrap_or(&0);
        magnitude.push_str(&last.to_string());
        for limb in self.limbs.iter().rev().skip(1) {
            magnitude.push_str(&format!("{:0WIDTH$}", limb, WIDTH = WIDTH));
        }

        let target = min_digits.max(1);
        if magnitude.len() < target {
            let mut padded = String::with_capacity(target);
            for _ in 0..target - magnitude.len() {
                padded.push('0');
            }
            padded.push_str(&magnitude);
            magnitude = padded;
        }

        let mut out = String::with_capacity(
            magnitude.len()
                + if group > 0 {
                    (magnitude.len() - 1) / group
                } else {
                    0
                }
                + usize::from(self.negative),
        );
        if self.negative {
            out.push('-');
        }
        if group == 0 {
            out.push_str(&magnitude);
        } else {
            for (i, ch) in magnitude.chars().enumerate() {
                if i > 0 && (magnitude.len() - i) % group == 0 {
                    out.push('_');
                }
                out.push(ch);
            }
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self {
                negative: self.negative,
                limbs: Self::add_abs(&self.limbs, &rhs.limbs),
            };
            out.normalize();
            out
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self {
                        negative: self.negative,
                        limbs: Self::sub_abs(&self.limbs, &rhs.limbs),
                    };
                    out.normalize();
                    out
                }
                Ordering::Less => {
                    let mut out = Self {
                        negative: rhs.negative,
                        limbs: Self::sub_abs(&rhs.limbs, &self.limbs),
                    };
                    out.normalize();
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut accum = vec![0u64; self.limbs.len() + rhs.limbs.len() + 1];
        for (i, &a) in self.limbs.iter().enumerate() {
            for (j, &b) in rhs.limbs.iter().enumerate() {
                accum[i + j] += u64::from(a) * u64::from(b);
            }
        }
        for i in 0..accum.len() - 1 {
            let carry = accum[i] / BASE;
            accum[i] %= BASE;
            accum[i + 1] += carry;
        }
        let mut limbs: Vec<u32> = accum.into_iter().map(|x| x as u32).collect();
        while limbs.len() > 1 && limbs.last() == Some(&0) {
            limbs.pop();
        }
        let mut out = Self {
            negative: self.negative ^ rhs.negative,
            limbs,
        };
        out.normalize();
        out
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        match self.abs_cmp(rhs) {
            Ordering::Less => return Ok((Self::zero(), self.clone())),
            Ordering::Equal => {
                return Ok((
                    Self {
                        negative: self.negative ^ rhs.negative,
                        limbs: vec![1],
                    },
                    Self::zero(),
                ));
            }
            Ordering::Greater => {}
        }

        let (q_limbs, r_limbs) = Self::div_abs(&self.limbs, &rhs.limbs);
        let mut quotient = Self {
            negative: self.negative ^ rhs.negative,
            limbs: q_limbs,
        };
        quotient.normalize();
        let mut remainder = Self {
            negative: self.negative,
            limbs: r_limbs,
        };
        remainder.normalize();
        Ok((quotient, remainder))
    }
}

pub enum Error {

    SignOnly,

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {

impl Big {
    fn new(negative: bool, mut digits: Vec<u8>) -> Self {
        let first = digits.iter().position(|&d| d != 0).unwrap_or(digits.len());
        if first == digits.len() {
            return Self { negative: false, digits: vec![0] };
        }
        digits.drain(..first);
        Self { negative, digits }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }
        let (negative, start) = match bytes[0] {
            b'-' => (true, 1),
            b'+' => (false, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        for i in start..bytes.len() {
            if bytes[i] != b'_' && !bytes[i].is_ascii_digit() {
                return Err(Error::InvalidDigit { index: i });
            }
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index: i });
            }
        }

        let digit_positions: Vec<usize> = (start..bytes.len())
            .filter(|&i| bytes[i] != b'_')
            .collect();

        if digit_positions.len() > 1 && bytes[digit_positions[0]] == b'0' {
            return Err(Error::LeadingZero { index: digit_positions[0] });
        }
        if digit_positions.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_positions[max_digits],
            });
        }

        let digits = digit_positions.iter().map(|&i| bytes[i] - b'0').collect();
        Ok(Self::new(negative, digits))
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = min_digits.max(self.digits.len()).max(1);
        let padding = width - self.digits.len();
        let mut out = String::with_capacity(width + width / group.max(1) + 1);
        if self.negative {
            out.push('-');
        }
        for i in 0..width {
            if group > 0 && i > 0 && (width - i) % group == 0 {
                out.push('_');
            }
            let digit = if i < padding { 0 } else { self.digits[i - padding] };
            out.push((b'0' + digit) as char);
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_mag(&self.digits, &rhs.digits),
            (true, true) => cmp_mag(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::new(self.negative, add_mag(&self.digits, &rhs.digits))
        } else {
            match cmp_mag(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::new(false, vec![0]),
                Ordering::Greater => Self::new(self.negative, sub_mag(&self.digits, &rhs.digits)),
                Ordering::Less => Self::new(rhs.negative, sub_mag(&rhs.digits, &self.digits)),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        Self::new(!self.negative, self.digits.clone())
    }

    pub fn signum(&self) -> i8 {
        if is_zero(&self.digits) { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        Self::new(self.negative ^ rhs.negative, mul_mag(&self.digits, &rhs.digits))
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if is_zero(&rhs.digits) {
            return Err(Error::DivisionByZero);
        }
        let (q, r) = div_mag(&self.digits, &rhs.digits);
        Ok((
            Self::new(self.negative ^ rhs.negative, q),
            Self::new(self.negative, r),
        ))
    }
}

fn is_zero(a: &[u8]) -> bool {
    a.len() == 1 && a[0] == 0
}

fn cmp_mag(a: &[u8], b: &[u8]) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.cmp(b))
}

fn add_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
    let (mut i, mut j, mut carry) = (a.len(), b.len(), 0u8);
    while i > 0 || j > 0 || carry > 0 {
        let x = if i > 0 { i -= 1; a[i] } else { 0 };
        let y = if j > 0 { j -= 1; b[j] } else { 0 };
        let sum = x + y + carry;
        out.push(sum % 10);
        carry = sum / 10;
    }
    out.reverse();
    out
}

fn sub_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len());
    let (mut i, mut j, mut borrow) = (a.len(), b.len(), 0i16);
    while i > 0 {
        i -= 1;
        let x = a[i] as i16 - borrow;
        let y = if j > 0 { j -= 1; b[j] as i16 } else { 0 };
        if x < y {
            out.push((x + 10 - y) as u8);
            borrow = 1;
        } else {
            out.push((x - y) as u8);
            borrow = 0;
        }
    }
    out.reverse();
    let first = out.iter().position(|&d| d != 0).unwrap_or(out.len() - 1);
    out.drain(..first);
    out
}

fn mul_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    if is_zero(a) || is_zero(b) {
        return vec![0];
    }
    let mut acc = vec![0u32; a.len() + b.len()];
    for i in (0..a.len()).rev() {
        for j in (0..b.len()).rev() {
            acc[i + j + 1] += (a[i] as u32) * (b[j] as u32);
        }
    }
    for i in (1..acc.len()).rev() {
        let carry = acc[i] / 10;
        acc[i] %= 10;
        acc[i - 1] += carry;
    }
    acc.into_iter().map(|d| d as u8).skip_while(|&d| d == 0).collect()
}

fn mul_digit(a: &[u8], d: u8) -> Vec<u8> {
    if d == 0 { return vec![0]; }
    let mut out = Vec::with_capacity(a.len() + 1);
    let mut carry = 0u16;
    for &x in a.iter().rev() {
        let n = x as u16 * d as u16 + carry;
        out.push((n % 10) as u8);
        carry = n / 10;
    }
    while carry > 0 {
        out.push((carry % 10) as u8);
        carry /= 10;
    }
    out.reverse();
    out
}

fn div_mag(a: &[u8], b: &[u8]) -> (Vec<u8>, Vec<u8>) {
    if cmp_mag(a, b) == Ordering::Less {
        return (vec![0], a.to_vec());
    }
    let mut quotient = Vec::with_capacity(a.len());
    let mut remainder = vec![0];
    for &digit in a {
        if is_zero(&remainder) {
            remainder[0] = digit;
        } else {
            remainder.push(digit);
        }
        let first = remainder.iter().position(|&d| d != 0).unwrap_or(remainder.len() - 1);
        remainder.drain(..first);
        let mut q = 0;
        while cmp_mag(&remainder, b) != Ordering::Less {
            remainder = sub_mag(&remainder, b);
            q += 1;
        }
        quotient.push(q);
    }
    let first = quotient.iter().position(|&d| d != 0).unwrap_or(quotient.len() - 1);
    quotient.drain(..first);
    (quotient, remainder)
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    limbs: Vec<u32>,
}

const BASE: u64 = 1_000_000_000;

impl Big {
    fn from_limbs(negative: bool, mut limbs: Vec<u32>) -> Self {
        while limbs.last() == Some(&0) {
            limbs.pop();
        }
        if limbs.is_empty() {
            Self { negative: false, limbs: vec![0] }
        } else {
            Self { negative, limbs }
        }
    }

    fn zero() -> Self {
        Self { negative: false, limbs: vec![0] }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn mag_cmp(&self, rhs: &Self) -> Ordering {
        self.limbs.len().cmp(&rhs.limbs.len())
            .then_with(|| self.limbs.iter().rev().cmp(rhs.limbs.iter().rev()))
    }

    fn mag_add(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let n = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64 + carry;
            out.push((n % BASE) as u32);
            carry = n / BASE;
        }
        if carry != 0 { out.push(carry as u32); }
        out
    }

    fn mag_sub(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let mut n = a[i] as i64 - borrow - b.get(i).copied().unwrap_or(0) as i64;
            if n < 0 { n += BASE as i64; borrow = 1; } else { borrow = 0; }
            out.push(n as u32);
        }
        while out.len() > 1 && out.last() == Some(&0) { out.pop(); }
        out
    }

    fn mul_small(a: &[u32], d: u64) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u64;
        for &x in a {
            let n = x as u64 * d + carry;
            out.push((n % BASE) as u32);
            carry = n / BASE;
        }
        while carry != 0 {
            out.push((carry % BASE) as u32);
            carry /= BASE;
        }
        out
    }

    fn div_mag(a: &[u32], b: &[u32]) -> (Vec<u32>, Vec<u32>) {
        let mut q = vec![0u32; a.len()];
        let mut rem = vec![0u32];
        for i in (0..a.len()).rev() {
            if rem.len() == 1 && rem[0] == 0 {
                rem[0] = a[i];
            } else {
                rem.insert(0, a[i]);
            }
            while rem.len() > 1 && rem.last() == Some(&0) { rem.pop(); }
            let mut lo = 0u64;
            let mut hi = BASE - 1;
            while lo < hi {
                let mid = (lo + hi + 1) / 2;
                let prod = Self::mul_small(b, mid);
                if mag_slice_cmp(&prod, &rem) != Ordering::Greater {
                    lo = mid;
                } else {
                    hi = mid - 1;
                }
            }
            q[i] = lo as u32;
            if lo != 0 {
                rem = Self::mag_sub(&rem, &Self::mul_small(b, lo));
            }
        }
        while q.len() > 1 && q.last() == Some(&0) { q.pop(); }
        while rem.len() > 1 && rem.last() == Some(&0) { rem.pop(); }
        (q, rem)
    }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() { return Err(Error::Empty); }
        let (negative, start) = match bytes[0] {
            b'-' => (true, 1),
            b'+' => (false, 1),
            _ => (false, 0),
        };
        if start == bytes.len() { return Err(Error::SignOnly); }
        for i in start..bytes.len() {
            if bytes[i] != b'_' && !bytes[i].is_ascii_digit() {
                return Err(Error::InvalidDigit { index: i });
            }
        }
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit() || !bytes[i + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index: i });
            }
        }
        let positions: Vec<usize> = (start..bytes.len()).filter(|&i| bytes[i] != b'_').collect();
        if positions.len() > 1 && bytes[positions[0]] == b'0' {
            return Err(Error::LeadingZero { index: positions[0] });
        }
        if positions.len() > max_digits {
            return Err(Error::TooManyDigits { limit: max_digits, index: positions[max_digits] });
        }

        let mut limbs = vec![0u32];
        for &i in &positions {
            limbs = Self::mul_small(&limbs, 10);
            let mut carry = (bytes[i] - b'0') as u64;
            for x in &mut limbs {
                let n = *x as u64 + carry;
                *x = (n % BASE) as u32;
                carry = n / BASE;
            }
            if carry != 0 { limbs.push(carry as u32); }
        }
        Ok(Self::from_limbs(negative, limbs))
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let mut s = self.limbs.last().copied().unwrap_or(0).to_string();
        for &limb in self.limbs[..self.limbs.len().saturating_sub(1)].iter().rev() {
            s.push_str(&format!("{:09}", limb));
        }
        let width = min_digits.max(s.len()).max(1);
        let padding = width - s.len();
        let mut out = String::new();
        if self.negative { out.push('-'); }
        for i in 0..width {
            if group > 0 && i > 0 && (width - i) % group == 0 { out.push('_'); }
            out.push(if i < padding { '0' } else { s.as_bytes()[i - padding] as char });
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.mag_cmp(rhs),
            (true, true) => rhs.mag_cmp(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::from_limbs(self.negative, Self::mag_add(&self.limbs, &rhs.limbs))
        } else {
            match self.mag_cmp(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => Self::from_limbs(self.negative, Self::mag_sub(&self.limbs, &rhs.limbs)),
                Ordering::Less => Self::from_limbs(rhs.negative, Self::mag_sub(&rhs.limbs, &self.limbs)),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self { self.add(&rhs.neg()) }

    pub fn neg(&self) -> Self {
        Self::from_limbs(!self.negative, self.limbs.clone())
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() { return Self::zero(); }
        let mut out = vec![0u64; self.limbs.len() + rhs.limbs.len()];
        for i in 0..self.limbs.len() {
            let mut carry = 0u64;
            for j in 0..rhs.limbs.len() {
                let k = i + j;
                let n = out[k] + self.limbs[i] as u64 * rhs.limbs[j] as u64 + carry;
                out[k] = n % BASE;
                carry = n / BASE;
            }
            let mut k = i + rhs.limbs.len();
            while carry != 0 {
                if k == out.len() { out.push(0); }
                let n = out[k] + carry;
                out[k] = n % BASE;
                carry = n / BASE;
                k += 1;
            }
        }
        Self::from_limbs(self.negative ^ rhs.negative, out.into_iter().map(|x| x as u32).collect())
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() { return Err(Error::DivisionByZero); }
        let (q, r) = Self::div_mag(&self.limbs, &rhs.limbs);
        Ok((
            Self::from_limbs(self.negative ^ rhs.negative, q),
            Self::from_limbs(self.negative, r),
        ))
    }
}

fn mag_slice_cmp(a: &[u32], b: &[u32]) -> Ordering {
    let mut aa = a.len();
    while aa > 1 && a[aa - 1] == 0 { aa -= 1; }
    let mut bb = b.len();
    while bb > 1 && b[bb - 1] == 0 { bb -= 1; }
    aa.cmp(&bb).then_with(|| a[..aa].iter().rev().cmp(b[..bb].iter().rev()))
}
