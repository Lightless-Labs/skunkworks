- **Error precedence:** I scan for every invalid byte first, then malformed underscores, leading zeros, and the digit limit, as specified.
- **Byte offsets:** Indexed errors point into the original UTF-8 string, including any sign; a non-ASCII invalid character points to its first byte.
- **Zero:** All spellings and arithmetic paths canonicalize zero as nonnegative; division by zero includes `0 / 0`.
- **Division:** Quotients truncate toward zero, and nonzero remainders have the dividend’s sign.
- **Large values:** Both implementations use arbitrary-length decimal digits and avoid machine-integer conversion of the value.
- **Alternative in B:** Solution A uses little-endian base-10 digits; Solution B uses little-endian base-1,000,000,000 limbs.

### Solution A

```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is [0] and is never negative.
    digits: Vec<u8>,
}

impl Big {
    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn normalize(&mut self) {
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.len() == 1 && self.digits[0] == 0 {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => {
                for (a, b) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u16;
        for i in 0..a.len().max(b.len()) {
            let sum = carry
                + u16::from(*a.get(i).unwrap_or(&0))
                + u16::from(*b.get(i).unwrap_or(&0));
            out.push((sum % 10) as u8);
            carry = sum / 10;
        }
        if carry != 0 {
            out.push(carry as u8);
        }
        out
    }

    // Requires a >= b.
    fn sub_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i16;
        for i in 0..a.len() {
            let mut digit = i16::from(a[i]) - borrow - i16::from(*b.get(i).unwrap_or(&0));
            if digit < 0 {
                digit += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(digit as u8);
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_abs_digit(a: &[u8], digit: u8) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u16;
        for &x in a {
            let v = u16::from(x) * u16::from(digit) + carry;
            out.push((v % 10) as u8);
            carry = v / 10;
        }
        while carry != 0 {
            out.push((carry % 10) as u8);
            carry /= 10;
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn cmp_abs_digits(a: &[u8], b: &[u8]) -> Ordering {
        match a.len().cmp(&b.len()) {
            Ordering::Equal => {
                for (x, y) in a.iter().rev().zip(b.iter().rev()) {
                    match x.cmp(y) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn sub_abs_digits(a: &[u8], b: &[u8]) -> Vec<u8> {
        Self::sub_abs(a, b)
    }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        if let Some(index) = (start..bytes.len())
            .find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())
        {
            return Err(Error::InvalidDigit { index });
        }

        if let Some(index) = (start..bytes.len()).find(|&i| {
            bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
        }) {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();

        if digit_count > 1 {
            let first_digit = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
            if bytes[first_digit] == b'0' {
                return Err(Error::LeadingZero { index: first_digit });
            }
        }

        if digit_count > max_digits {
            let excess_number = max_digits;
            let index = (start..bytes.len())
                .filter(|&i| bytes[i] != b'_')
                .nth(excess_number)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut digits = Vec::with_capacity(digit_count.max(1));
        for &b in bytes[start..].iter().rev() {
            if b != b'_' {
                digits.push(b - b'0');
            }
        }
        if digits.is_empty() {
            digits.push(0);
        }

        let mut value = Self { negative, digits };
        value.normalize();
        Ok(value)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let mag_len = self.digits.len().max(min_digits).max(1);
        let pad = mag_len - self.digits.len();
        let mut out = String::with_capacity(
            mag_len + if group > 0 { (mag_len - 1) / group } else { 0 } + usize::from(self.negative),
        );
        if self.negative {
            out.push('-');
        }
        for pos in (0..mag_len).rev() {
            if group > 0 && pos != mag_len - 1 && (pos + 1) % group == 0 {
                out.push('_');
            }
            if pos >= self.digits.len() {
                out.push('0');
            } else {
                out.push((b'0' + self.digits[pos]) as char);
            }
        }
        let _ = pad;
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self {
                negative: self.negative,
                digits: Self::add_abs(&self.digits, &rhs.digits),
            };
            result.normalize();
            result
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self {
                        negative: self.negative,
                        digits: Self::sub_abs(&self.digits, &rhs.digits),
                    };
                    result.normalize();
                    result
                }
                Ordering::Less => {
                    let mut result = Self {
                        negative: rhs.negative,
                        digits: Self::sub_abs(&rhs.digits, &self.digits),
                    };
                    result.normalize();
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        let mut neg_rhs = rhs.clone();
        if !neg_rhs.is_zero() {
            neg_rhs.negative = !neg_rhs.negative;
        }
        self.add(&neg_rhs)
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut accum = vec![0u32; self.digits.len() + rhs.digits.len() + 1];
        for (i, &a) in self.digits.iter().enumerate() {
            for (j, &b) in rhs.digits.iter().enumerate() {
                accum[i + j] += u32::from(a) * u32::from(b);
            }
        }
        for i in 0..accum.len() - 1 {
            let carry = accum[i] / 10;
            accum[i] %= 10;
            accum[i + 1] += carry;
        }
        let mut digits: Vec<u8> = accum.into_iter().map(|x| x as u8).collect();
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        let mut result = Self {
            negative: self.negative ^ rhs.negative,
            digits,
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        match self.abs_cmp(rhs) {
            Ordering::Less => return Ok((Self::zero(), self.clone())),
            Ordering::Equal => {
                let q = Self {
                    negative: self.negative ^ rhs.negative,
                    digits: vec![1],
                };
                return Ok((q, Self::zero()));
            }
            Ordering::Greater => {}
        }

        let mut quotient_be = Vec::with_capacity(self.digits.len());
        let mut remainder_le = vec![0u8];

        for &digit in self.digits.iter().rev() {
            let mut shifted = Vec::with_capacity(remainder_le.len() + 1);
            shifted.push(digit);
            shifted.extend_from_slice(&remainder_le);
            remainder_le = shifted;
            while remainder_le.len() > 1 && remainder_le.last() == Some(&0) {
                remainder_le.pop();
            }

            let mut qdigit = 0u8;
            while Self::cmp_abs_digits(&remainder_le, &rhs.digits) != Ordering::Less {
                remainder_le = Self::sub_abs_digits(&remainder_le, &rhs.digits);
                qdigit += 1;
            }
            quotient_be.push(qdigit);
        }

        while quotient_be.len() > 1 && quotient_be[0] == 0 {
            quotient_be.remove(0);
        }
        let mut quotient_digits: Vec<u8> = quotient_be.into_iter().rev().collect();
        if quotient_digits.is_empty() {
            quotient_digits.push(0);
        }

        let mut quotient = Self {
            negative: self.negative ^ rhs.negative,
            digits: quotient_digits,
        };
        quotient.normalize();

        let mut remainder = Self {
            negative: self.negative,
            digits: remainder_le,
        };
        remainder.normalize();

        Ok((quotient, remainder))
    }
}
```

### Solution B

```rust
use std::cmp::Ordering;

const BASE: u64 = 1_000_000_000;
const WIDTH: usize = 9;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1,000,000,000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    fn zero() -> Self {
        Self {
            negative: false,
            limbs: vec![0],
        }
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.len() == 1 && self.limbs[0] == 0 {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let sum = u64::from(*a.get(i).unwrap_or(&0))
                + u64::from(*b.get(i).unwrap_or(&0))
                + carry;
            out.push((sum % BASE) as u32);
            carry = sum / BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        out
    }

    // Requires a >= b.
    fn sub_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let mut digit =
                i64::from(a[i]) - borrow - i64::from(*b.get(i).unwrap_or(&0));
            if digit < 0 {
                digit += BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(digit as u32);
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_abs_small(a: &[u32], factor: u32) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u64;
        for &limb in a {
            let value = u64::from(limb) * u64::from(factor) + carry;
            out.push((value % BASE) as u32);
            carry = value / BASE;
        }
        while carry != 0 {
            out.push((carry % BASE) as u32);
            carry /= BASE;
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn cmp_limbs(a: &[u32], b: &[u32]) -> Ordering {
        match a.len().cmp(&b.len()) {
            Ordering::Equal => {
                for (x, y) in a.iter().rev().zip(b.iter().rev()) {
                    match x.cmp(y) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn div_abs(numerator: &[u32], denominator: &[u32]) -> (Vec<u32>, Vec<u32>) {
        let mut remainder = vec![0u32];
        let mut quotient_be = Vec::with_capacity(numerator.len());

        for &limb in numerator.iter().rev() {
            // remainder = remainder * BASE + limb, in little-endian form.
            if remainder.len() == 1 && remainder[0] == 0 {
                remainder[0] = limb;
            } else {
                remainder.insert(0, limb);
            }
            while remainder.len() > 1 && remainder.last() == Some(&0) {
                remainder.pop();
            }

            // Binary search the quotient limb in [0, BASE - 1].
            let mut low = 0u64;
            let mut high = BASE - 1;
            while low < high {
                let mid = (low + high + 1) / 2;
                let product = Self::mul_abs_small(&denominator, mid as u32);
                if Self::cmp_limbs(&product, &remainder) != Ordering::Greater {
                    low = mid;
                } else {
                    high = mid - 1;
                }
            }
            let q_limb = low as u32;
            if q_limb != 0 {
                let product = Self::mul_abs_small(denominator, q_limb);
                remainder = Self::sub_abs(&remainder, &product);
            }
            quotient_be.push(q_limb);
        }

        while quotient_be.len() > 1 && quotient_be[0] == 0 {
            quotient_be.remove(0);
        }
        (quotient_be.into_iter().rev().collect(), remainder)
    }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        if let Some(index) =
            (start..bytes.len()).find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())
        {
            return Err(Error::InvalidDigit { index });
        }

        if let Some(index) = (start..bytes.len()).find(|&i| {
            bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
        }) {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        if digit_count > 1 {
            let first = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
            if bytes[first] == b'0' {
                return Err(Error::LeadingZero { index: first });
            }
        }

        if digit_count > max_digits {
            let index = (start..bytes.len())
                .filter(|&i| bytes[i] != b'_')
                .nth(max_digits)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut limbs = vec![0u32];
        for &b in &bytes[start..] {
            if b == b'_' {
                continue;
            }
            let mut carry = u64::from(b - b'0');
            for limb in &mut limbs {
                let value = u64::from(*limb) * 10 + carry;
                *limb = (value % BASE) as u32;
                carry = value / BASE;
            }
            if carry != 0 {
                limbs.push(carry as u32);
            }
        }

        let mut result = Self { negative, limbs };
        result.normalize();
        Ok(result)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let mut magnitude = String::new();
        let last = *self.limbs.last().unwrap_or(&0);
        magnitude.push_str(&last.to_string());
        for limb in self.limbs.iter().rev().skip(1) {
            magnitude.push_str(&format!("{:0WIDTH$}", limb, WIDTH = WIDTH));
        }

        let target = min_digits.max(1);
        if magnitude.len() < target {
            let mut padded = String::with_capacity(target);
            for _ in 0..target - magnitude.len() {
                padded.push('0');
            }
            padded.push_str(&magnitude);
            magnitude = padded;
        }

        let mut out = String::with_capacity(
            magnitude.len()
                + if group > 0 {
                    (magnitude.len() - 1) / group
                } else {
                    0
                }
                + usize::from(self.negative),
        );
        if self.negative {
            out.push('-');
        }
        if group == 0 {
            out.push_str(&magnitude);
        } else {
            for (i, ch) in magnitude.chars().enumerate() {
                if i > 0 && (magnitude.len() - i) % group == 0 {
                    out.push('_');
                }
                out.push(ch);
            }
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self {
                negative: self.negative,
                limbs: Self::add_abs(&self.limbs, &rhs.limbs),
            };
            out.normalize();
            out
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self {
                        negative: self.negative,
                        limbs: Self::sub_abs(&self.limbs, &rhs.limbs),
                    };
                    out.normalize();
                    out
                }
                Ordering::Less => {
                    let mut out = Self {
                        negative: rhs.negative,
                        limbs: Self::sub_abs(&rhs.limbs, &self.limbs),
                    };
                    out.normalize();
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut accum = vec![0u64; self.limbs.len() + rhs.limbs.len() + 1];
        for (i, &a) in self.limbs.iter().enumerate() {
            for (j, &b) in rhs.limbs.iter().enumerate() {
                accum[i + j] += u64::from(a) * u64::from(b);
            }
        }
        for i in 0..accum.len() - 1 {
            let carry = accum[i] / BASE;
            accum[i] %= BASE;
            accum[i + 1] += carry;
        }
        let mut limbs: Vec<u32> = accum.into_iter().map(|x| x as u32).collect();
        while limbs.len() > 1 && limbs.last() == Some(&0) {
            limbs.pop();
        }
        let mut out = Self {
            negative: self.negative ^ rhs.negative,
            limbs,
        };
        out.normalize();
        out
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        match self.abs_cmp(rhs) {
            Ordering::Less => return Ok((Self::zero(), self.clone())),
            Ordering::Equal => {
                return Ok((
                    Self {
                        negative: self.negative ^ rhs.negative,
                        limbs: vec![1],
                    },
                    Self::zero(),
                ));
            }
            Ordering::Greater => {}
        }

        let (q_limbs, r_limbs) = Self::div_abs(&self.limbs, &rhs.limbs);
        let mut quotient = Self {
            negative: self.negative ^ rhs.negative,
            limbs: q_limbs,
        };
        quotient.normalize();
        let mut remainder = Self {
            negative: self.negative,
            limbs: r_limbs,
        };
        remainder.normalize();
        Ok((quotient, remainder))
    }
}
```
