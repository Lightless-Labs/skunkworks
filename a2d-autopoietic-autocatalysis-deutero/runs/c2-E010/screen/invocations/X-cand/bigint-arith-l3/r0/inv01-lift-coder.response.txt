- **Error order:** Empty → SignOnly → invalid byte → malformed separator → leading zero → digit limit. Invalid bytes anywhere outrank earlier malformed underscores.
- **Offsets:** All indexed errors use zero-based UTF-8 byte offsets in the original input, including any sign.
- **Separators:** An underscore must have a digit on each side; separator checks happen only after the full invalid-byte scan.
- **Digit limits:** Count magnitude digits only, after leading-zero validation; zero counts as one digit, so a limit of zero rejects it.
- **Division:** Quotients truncate toward zero; a nonzero remainder has the dividend’s sign.
- **Representation and cost:** Both solutions use little-endian base-10 digits for exact arithmetic. Solution B uses long division; Solution A uses binary search over each quotient digit.

### Solution A

```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian decimal digits. Zero is represented as [0], nonnegative.
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = match bytes[0] {
            b'+' | b'-' => 1,
            _ => 0,
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over every separator error.
        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b != b'_' && !b.is_ascii_digit() {
                return Err(Error::InvalidDigit { index: i });
            }
        }

        let mut first_bad_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let left_is_digit = i > start && bytes[i - 1].is_ascii_digit();
                let right_is_digit = i + 1 < bytes.len() && bytes[i + 1].is_ascii_digit();
                if !(left_is_digit && right_is_digit) {
                    first_bad_separator.get_or_insert(i);
                }
            }
        }
        if let Some(index) = first_bad_separator {
            return Err(Error::Separator { index });
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut first_excess = None;
        let mut digits = Vec::new();

        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b.is_ascii_digit() {
                first_digit.get_or_insert((i, b));
                digit_count = digit_count.saturating_add(1);
                if digit_count == max_digits.saturating_add(1) {
                    first_excess = Some(i);
                }
                digits.push(b - b'0');
            }
        }

        let (first_index, first_value) = first_digit.expect("validated nonempty magnitude");
        if digit_count > 1 && first_value == 0 {
            return Err(Error::LeadingZero { index: first_index });
        }
        if digit_count > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: first_excess.unwrap_or_else(|| {
                    bytes
                        .iter()
                        .enumerate()
                        .skip(start)
                        .filter(|(_, b)| b.is_ascii_digit())
                        .nth(max_digits)
                        .map(|(i, _)| i)
                        .unwrap_or(first_index)
                }),
            });
        }

        digits.reverse();
        let negative = bytes[0] == b'-' && !(digits.len() == 1 && digits[0] == 0);
        Ok(Self { negative, digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude_len = self.digits.len().max(1);
        let width = magnitude_len.max(min_digits);
        let padding = width - magnitude_len;

        let mut out = String::with_capacity(
            width.saturating_add(if group > 0 { (width - 1) / group } else { 0 })
                .saturating_add(usize::from(self.negative)),
        );
        if self.negative {
            out.push('-');
        }

        for pos in 0..width {
            if group > 0 && pos > 0 && (width - pos) % group == 0 {
                out.push('_');
            }
            let from_right = width - 1 - pos;
            if from_right < padding {
                out.push('0');
            } else {
                let digit_index = from_right - padding;
                out.push(char::from(b'0' + self.digits[self.digits.len() - 1 - digit_index]));
            }
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_mag(&self.digits, &rhs.digits),
            (true, true) => cmp_mag(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let digits = add_mag(&self.digits, &rhs.digits);
            Self::from_parts(self.negative, digits)
        } else {
            match cmp_mag(&self.digits, &rhs.digits) {
                Ordering::Greater => Self::from_parts(
                    self.negative,
                    sub_mag(&self.digits, &rhs.digits),
                ),
                Ordering::Less => Self::from_parts(
                    rhs.negative,
                    sub_mag(&rhs.digits, &self.digits),
                ),
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            self.clone()
        } else {
            Self {
                negative: !self.negative,
                digits: self.digits.clone(),
            }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut acc = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            let mut carry = 0u32;
            for (j, &b) in rhs.digits.iter().enumerate() {
                let k = i + j;
                let total = acc[k] + u32::from(a) * u32::from(b) + carry;
                acc[k] = total % 10;
                carry = total / 10;
            }
            let mut k = i + rhs.digits.len();
            while carry > 0 {
                let total = acc[k] + carry;
                acc[k] = total % 10;
                carry = total / 10;
                k += 1;
                if k == acc.len() && carry > 0 {
                    acc.push(0);
                }
            }
        }

        let digits = acc.into_iter().map(|d| d as u8).collect();
        Self::from_parts(self.negative ^ rhs.negative, digits)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        match cmp_mag(&self.digits, &rhs.digits) {
            Ordering::Less => return Ok((Self::zero(), self.clone())),
            Ordering::Equal => {
                return Ok((
                    Self::from_parts(self.negative ^ rhs.negative, vec![1]),
                    Self::zero(),
                ));
            }
            Ordering::Greater => {}
        }

        let mut quotient_be = Vec::with_capacity(self.digits.len());
        let mut remainder = vec![0u8];

        for &digit in self.digits.iter().rev() {
            remainder = mul10_add(&remainder, digit);
            let mut low = 0u8;
            let mut high = 9u8;
            let mut best = 0u8;
            while low <= high {
                let mid = low + (high - low) / 2;
                let product = mul_small(&rhs.digits, mid);
                match cmp_mag(&product, &remainder) {
                    Ordering::Greater => {
                        if mid == 0 {
                            break;
                        }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        if mid == 9 {
                            break;
                        }
                        low = mid + 1;
                    }
                }
            }
            quotient_be.push(best);
            let product = mul_small(&rhs.digits, best);
            remainder = sub_mag(&remainder, &product);
        }

        quotient_be.reverse();
        let quotient = Self::from_parts(self.negative ^ rhs.negative, quotient_be);
        let rem = Self::from_parts(self.negative, remainder);
        Ok((quotient, rem))
    }

    fn zero() -> Self {
        Self { negative: false, digits: vec![0] }
    }

    fn from_parts(negative: bool, mut digits: Vec<u8>) -> Self {
        normalize(&mut digits);
        let zero = digits.len() == 1 && digits[0] == 0;
        Self { negative: negative && !zero, digits }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }
}

fn normalize(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    if digits.is_empty() {
        digits.push(0);
    }
}

fn cmp_mag(a: &[u8], b: &[u8]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => {
            for i in (0..a.len()).rev() {
                match a[i].cmp(&b[i]) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u8;
    for i in 0..a.len().max(b.len()) {
        let sum = a.get(i).copied().unwrap_or(0) + b.get(i).copied().unwrap_or(0) + carry;
        out.push(sum % 10);
        carry = sum / 10;
    }
    if carry != 0 {
        out.push(carry);
    }
    out
}

fn sub_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    debug_assert!(cmp_mag(a, b) != Ordering::Less);
    let mut out = Vec::with_capacity(a.len());
    let mut borrow = 0i16;
    for i in 0..a.len() {
        let mut d = i16::from(a[i]) - i16::from(b.get(i).copied().unwrap_or(0)) - borrow;
        if d < 0 {
            d += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        out.push(d as u8);
    }
    normalize(&mut out);
    out
}

fn mul_small(a: &[u8], factor: u8) -> Vec<u8> {
    if factor == 0 {
        return vec![0];
    }
    let mut out = Vec::with_capacity(a.len() + 1);
    let mut carry = 0u16;
    for &d in a {
        let n = u16::from(d) * u16::from(factor) + carry;
        out.push((n % 10) as u8);
        carry = n / 10;
    }
    while carry > 0 {
        out.push((carry % 10) as u8);
        carry /= 10;
    }
    normalize(&mut out);
    out
}

fn mul10_add(a: &[u8], digit: u8) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len() + 1);
    out.push(digit);
    out.extend_from_slice(a);
    normalize(&mut out);
    out
}
```

### Solution B

```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1,000,000,000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = match bytes[0] {
            b'+' | b'-' => 1,
            _ => 0,
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b != b'_' && !b.is_ascii_digit() {
                return Err(Error::InvalidDigit { index: i });
            }
        }

        let mut bad_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && !(i > start
                    && i + 1 < bytes.len()
                    && bytes[i - 1].is_ascii_digit()
                    && bytes[i + 1].is_ascii_digit())
            {
                bad_separator.get_or_insert(i);
            }
        }
        if let Some(index) = bad_separator {
            return Err(Error::Separator { index });
        }

        let digit_positions: Vec<usize> = (start..bytes.len())
            .filter(|&i| bytes[i].is_ascii_digit())
            .collect();

        if digit_positions.len() > 1 && bytes[digit_positions[0]] == b'0' {
            return Err(Error::LeadingZero { index: digit_positions[0] });
        }
        if digit_positions.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_positions[max_digits],
            });
        }

        let mut limbs = vec![0u32];
        let mut digits = 0usize;
        for &b in &bytes[start..] {
            if b.is_ascii_digit() {
                mul_small_in_place(&mut limbs, 10);
                add_small_in_place(&mut limbs, u32::from(b - b'0'));
                digits += 1;
            }
        }
        debug_assert_eq!(digits, digit_positions.len());
        let negative = bytes[0] == b'-' && !(limbs.len() == 1 && limbs[0] == 0);
        Ok(Self { negative, limbs })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let mut body = self.limbs.last().copied().unwrap_or(0).to_string();
        for limb in self.limbs.iter().rev().skip(1) {
            body.push_str(&format!("{:09}", limb));
        }
        if body.len() < min_digits {
            let mut padded = String::with_capacity(min_digits);
            padded.extend(std::iter::repeat_n('0', min_digits - body.len()));
            padded.push_str(&body);
            body = padded;
        }

        let mut out = String::with_capacity(body.len().saturating_add(
            if group > 0 { (body.len() - 1) / group } else { 0 }
                + usize::from(self.negative),
        ));
        if self.negative {
            out.push('-');
        }
        for (i, ch) in body.chars().enumerate() {
            if group > 0 && i > 0 && (body.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(ch);
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_mag(&self.limbs, &rhs.limbs),
            (true, true) => cmp_mag(&rhs.limbs, &self.limbs),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::from_parts(
                self.negative,
                add_mag(&self.limbs, &rhs.limbs),
            )
        } else {
            match cmp_mag(&self.limbs, &rhs.limbs) {
                Ordering::Greater => Self::from_parts(
                    self.negative,
                    sub_mag(&self.limbs, &rhs.limbs),
                ),
                Ordering::Less => Self::from_parts(
                    rhs.negative,
                    sub_mag(&rhs.limbs, &self.limbs),
                ),
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            self.clone()
        } else {
            Self { negative: !self.negative, limbs: self.limbs.clone() }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u64; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let n = out[k] + u64::from(a) * u64::from(b) + carry;
                out[k] = n % Self::BASE;
                carry = n / Self::BASE;
            }
            let mut k = i + rhs.limbs.len();
            while carry > 0 {
                if k == out.len() {
                    out.push(0);
                }
                let n = out[k] + carry;
                out[k] = n % Self::BASE;
                carry = n / Self::BASE;
                k += 1;
            }
        }
        Self::from_parts(
            self.negative ^ rhs.negative,
            out.into_iter().map(|x| x as u32).collect(),
        )
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        match cmp_mag(&self.limbs, &rhs.limbs) {
            Ordering::Less => return Ok((Self::zero(), self.clone())),
            Ordering::Equal => {
                return Ok((
                    Self::from_parts(self.negative ^ rhs.negative, vec![1]),
                    Self::zero(),
                ));
            }
            Ordering::Greater => {}
        }

        let mut quotient_be = Vec::with_capacity(self.limbs.len());
        let mut remainder = vec![0u32];
        for &limb in self.limbs.iter().rev() {
            remainder = mul_base_add(&remainder, limb);
            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u32;
            while low <= high {
                let mid = low + (high - low) / 2;
                match cmp_mag(&mul_small(&rhs.limbs, mid), &remainder) {
                    Ordering::Greater => {
                        if mid == 0 { break; }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid as u32;
                        low = mid + 1;
                    }
                }
            }
            quotient_be.push(best);
            remainder = sub_mag(&remainder, &mul_small(&rhs.limbs, u64::from(best)));
        }
        quotient_be.reverse();
        Ok((
            Self::from_parts(self.negative ^ rhs.negative, quotient_be),
            Self::from_parts(self.negative, remainder),
        ))
    }

    fn zero() -> Self {
        Self { negative: false, limbs: vec![0] }
    }

    fn from_parts(negative: bool, mut limbs: Vec<u32>) -> Self {
        normalize(&mut limbs);
        let zero = limbs.len() == 1 && limbs[0] == 0;
        Self { negative: negative && !zero, limbs }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }
}

fn normalize(limbs: &mut Vec<u32>) {
    while limbs.len() > 1 && limbs.last() == Some(&0) {
        limbs.pop();
    }
    if limbs.is_empty() {
        limbs.push(0);
    }
}

fn cmp_mag(a: &[u32], b: &[u32]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => {
            for i in (0..a.len()).rev() {
                match a[i].cmp(&b[i]) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
    let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u64;
    for i in 0..a.len().max(b.len()) {
        let n = u64::from(a.get(i).copied().unwrap_or(0))
            + u64::from(b.get(i).copied().unwrap_or(0))
            + carry;
        out.push((n % Big::BASE) as u32);
        carry = n / Big::BASE;
    }
    if carry != 0 {
        out.push(carry as u32);
    }
    out
}

fn sub_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
    debug_assert!(cmp_mag(a, b) != Ordering::Less);
    let mut out = Vec::with_capacity(a.len());
    let mut borrow = 0i64;
    for i in 0..a.len() {
        let n = i64::from(a[i]) - i64::from(b.get(i).copied().unwrap_or(0)) - borrow;
        if n < 0 {
            out.push((n + Big::BASE as i64) as u32);
            borrow = 1;
        } else {
            out.push(n as u32);
            borrow = 0;
        }
    }
    normalize(&mut out);
    out
}

fn mul_small_in_place(a: &mut Vec<u32>, factor: u32) {
    let mut carry = 0u64;
    for limb in a.iter_mut() {
        let n = u64::from(*limb) * u64::from(factor) + carry;
        *limb = (n % Big::BASE) as u32;
        carry = n / Big::BASE;
    }
    if carry != 0 {
        a.push(carry as u32);
    }
}

fn add_small_in_place(a: &mut Vec<u32>, value: u32) {
    let mut carry = u64::from(value);
    let mut i = 0;
    while carry > 0 {
        if i == a.len() {
            a.push(0);
        }
        let n = u64::from(a[i]) + carry;
        a[i] = (n % Big::BASE) as u32;
        carry = n / Big::BASE;
        i += 1;
    }
}

fn mul_small(a: &[u32], factor: u64) -> Vec<u32> {
    if factor == 0 {
        return vec![0];
    }
    let mut out = Vec::with_capacity(a.len() + 1);
    let mut carry = 0u128;
    for &limb in a {
        let n = u128::from(limb) * u128::from(factor) + carry;
        out.push((n % u128::from(Big::BASE)) as u32);
        carry = n / u128::from(Big::BASE);
    }
    while carry > 0 {
        out.push((carry % u128::from(Big::BASE)) as u32);
        carry /= u128::from(Big::BASE);
    }
    normalize(&mut out);
    out
}

fn mul_base_add(a: &[u32], limb: u32) -> Vec<u32> {
    let mut out = Vec::with_capacity(a.len() + 1);
    out.push(limb);
    out.extend_from_slice(a);
    normalize(&mut out);
    out
}
```
