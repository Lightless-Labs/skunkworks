#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian decimal digits; zero is [0].
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'-' => (true, 1),
            b'+' => (false, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut first_invalid = None;
        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b != b'_' && !b.is_ascii_digit() {
                first_invalid = Some(i);
                break;
            }
        }
        if let Some(index) = first_invalid {
            return Err(Error::InvalidDigit { index });
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || bytes[i - 1] == b'_'
                    || bytes[i + 1] == b'_')
            {
                return Err(Error::Separator { index: i });
            }
        }

        let mut count = 0usize;
        let mut first_digit = None;
        let mut digits = Vec::new();
        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b != b'_' {
                if first_digit.is_none() {
                    first_digit = Some((i, b));
                }
                count += 1;
                digits.push(b - b'0');
            }
        }

        if count > 1 {
            if let Some((index, b'0')) = first_digit {
                return Err(Error::LeadingZero { index });
            }
        }

        if count > max_digits {
            let target = max_digits + 1;
            let mut seen = 0usize;
            for (i, &b) in bytes.iter().enumerate().skip(start) {
                if b != b'_' {
                    seen += 1;
                    if seen == target {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        let is_zero = digits.iter().all(|&d| d == 0);
        Ok(Self {
            negative: negative && !is_zero,
            digits,
        })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let natural = self.digits.len().max(1);
        let width = natural.max(min_digits);
        let padding = width - natural;
        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for pos in (0..width).rev() {
            if group > 0 && pos < width - 1 && (pos + 1) % group == 0 {
                out.push('_');
            }
            if pos >= padding {
                out.push((b'0' + self.digits[pos - padding]) as char);
            } else {
                out.push('0');
            }
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.signum(), rhs.signum()) {
            (a, b) if a < b => Ordering::Less,
            (a, b) if a > b => Ordering::Greater,
            (0, 0) => Ordering::Equal,
            (1, 1) => Self::cmp_abs(self, rhs),
            (-1, -1) => Self::cmp_abs(rhs, self),
            _ => Ordering::Equal,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let digits = Self::add_abs(&self.digits, &rhs.digits);
            Self::from_parts(self.negative, digits)
        } else {
            match Self::cmp_abs(self, rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    Self::from_parts(self.negative, Self::sub_abs(&self.digits, &rhs.digits))
                }
                Ordering::Less => {
                    Self::from_parts(rhs.negative, Self::sub_abs(&rhs.digits, &self.digits))
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

        if self.is_zero() {
            Self::zero()
        } else {
            Self { negative: !self.negative, digits: self.digits.clone() }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u8; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            let mut carry = 0u16;
            for (j, &b) in rhs.digits.iter().enumerate() {

                let value = out[k] as u16 + a as u16 * b as u16 + carry;
                out[k] = (value % 10) as u8;
                carry = value / 10;

            while carry != 0 {
                let value = out[k] as u16 + carry;
                out[k] = (value % 10) as u8;
                carry = value / 10;
                k += 1;
            }
        }
        Self::from_parts(self.negative != rhs.negative, out)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (q, r) = Self::div_abs(&self.digits, &rhs.digits);
        Ok((
            Self::from_parts(self.negative != rhs.negative, q),
            Self::from_parts(self.negative, r),
        ))
    }

    fn zero() -> Self {
        Self { negative: false, digits: vec![0] }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn from_parts(negative: bool, mut digits: Vec<u8>) -> Self {
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        if digits.is_empty() {
            digits.push(0);
        }
        let zero = digits.len() == 1 && digits[0] == 0;
        Self { negative: negative && !zero, digits }
    }

    fn cmp_abs(a: &Self, b: &Self) -> Ordering {
        match a.digits.len().cmp(&b.digits.len()) {
            Ordering::Equal => {
                for i in (0..a.digits.len()).rev() {
                    match a.digits[i].cmp(&b.digits[i]) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u16;
        for i in 0..a.len().max(b.len()) {
            let value = a.get(i).copied().unwrap_or(0) as u16
                + b.get(i).copied().unwrap_or(0) as u16
                + carry;
            out.push((value % 10) as u8);
            carry = value / 10;
        }
        if carry != 0 {
            out.push(carry as u8);
        }
        out
    }

    // Assumes a >= b.
    fn sub_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i16;
        for i in 0..a.len() {
            let mut value = a[i] as i16 - b.get(i).copied().unwrap_or(0) as i16 - borrow;
            if value < 0 {
                value += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(value as u8);
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_abs_digit(a: &[u8], digit: u8) -> Vec<u8> {
        if digit == 0 {
            return vec![0];
        }
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u16;
        for &d in a {
            let value = d as u16 * digit as u16 + carry;
            out.push((value % 10) as u8);
            carry = value / 10;
        }
        while carry != 0 {
            out.push((carry % 10) as u8);
            carry /= 10;
        }
        out
    }

    fn div_abs(a: &[u8], b: &[u8]) -> (Vec<u8>, Vec<u8>) {
        if Self::cmp_digits(a, b) == Ordering::Less {
            return (vec![0], a.to_vec());
        }
        // Long division, reading most significant digit first.
        let mut q_be = Vec::with_capacity(a.len());

        for &digit in a.iter().rev() {
            rem.insert(0, digit);
            while rem.len() > 1 && rem.last() == Some(&0) {
                rem.pop();
            }
            let mut qdigit = 0u8;
            for candidate in (1..=9).rev() {
                let product = Self::mul_abs_digit(b, candidate);
                if Self::cmp_digits(&product, &rem) != Ordering::Greater {
                    rem = Self::sub_abs(&rem, &product);
                    qdigit = candidate;
                    break;
                }
            }
            q_be.push(qdigit);
        }
        while q_be.len() > 1 && q_be.first() == Some(&0) {
            q_be.remove(0);
        }
        q_be.reverse();
        (q_be, rem)
    }

    fn cmp_digits(a: &[u8], b: &[u8]) -> Ordering {

                for i in (0..a.len()).rev() {

                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Big-endian decimal digits; zero is [0].
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let b = text.as_bytes();
        if b.is_empty() {
            return Err(Error::Empty);
        }
        let (negative, start) = match b[0] {
            b'-' => (true, 1),
            b'+' => (false, 1),
            _ => (false, 0),
        };
        if start == b.len() {
            return Err(Error::SignOnly);
        }

        if let Some(i) = (start..b.len()).find(|&i| b[i] != b'_' && !b[i].is_ascii_digit()) {
            return Err(Error::InvalidDigit { index: i });
        }
        if let Some(i) = (start..b.len()).find(|&i| {
            b[i] == b'_'
                && (i == start || i + 1 == b.len() || b[i - 1] == b'_' || b[i + 1] == b'_')
        }) {
            return Err(Error::Separator { index: i });
        }

        let digit_positions: Vec<usize> = (start..b.len()).filter(|&i| b[i] != b'_').collect();
        if digit_positions.len() > 1 && b[digit_positions[0]] == b'0' {
            return Err(Error::LeadingZero { index: digit_positions[0] });
        }
        if digit_positions.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_positions[max_digits],
            });
        }

        let digits: Vec<u8> = digit_positions.iter().map(|&i| b[i] - b'0').collect();
        let is_zero = digits.iter().all(|&d| d == 0);
        Ok(Self { negative: negative && !is_zero, digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = self.digits.len().max(1).max(min_digits);
        let padding = width - self.digits.len();
        let mut result = String::new();
        if self.negative && !self.is_zero() {
            result.push('-');
        }
        for i in 0..width {
            if group > 0 && i > 0 && (width - i) % group == 0 {
                result.push('_');
            }
            if i < padding {
                result.push('0');
            } else {
                result.push((b'0' + self.digits[i - padding]) as char);
            }
        }
        result
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.signum(), rhs.signum()) {
            (x, y) if x < y => Ordering::Less,
            (x, y) if x > y => Ordering::Greater,
            (0, 0) => Ordering::Equal,
            (1, 1) => Self::cmp_abs(&self.digits, &rhs.digits),
            (-1, -1) => Self::cmp_abs(&rhs.digits, &self.digits),
            _ => Ordering::Equal,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::from_parts(self.negative, Self::add_abs(&self.digits, &rhs.digits))
        } else {
            match Self::cmp_abs(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    Self::from_parts(self.negative, Self::sub_abs(&self.digits, &rhs.digits))
                }
                Ordering::Less => {
                    Self::from_parts(rhs.negative, Self::sub_abs(&rhs.digits, &self.digits))
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            Self::zero()
        } else {
            Self { negative: !self.negative, digits: self.digits.clone() }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let a = self.digits.len();
        let b = rhs.digits.len();
        let mut out = vec![0u8; a + b];
        for i in (0..a).rev() {
            let mut carry = 0u16;
            for j in (0..b).rev() {
                let k = i + j + 1;
                let value = out[k] as u16 + self.digits[i] as u16 * rhs.digits[j] as u16 + carry;
                out[k] = (value % 10) as u8;
                carry = value / 10;
            }
            let mut k = i;
            while carry != 0 {
                let value = out[k] as u16 + carry;
                out[k] = (value % 10) as u8;
                carry = value / 10;
                if k == 0 && carry != 0 {
                    out.insert(0, 0);
                    k += 1;
                } else if carry != 0 {
                    k -= 1;
                }
            }
        }
        Self::from_parts(self.negative != rhs.negative, out)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (q, r) = Self::div_abs(&self.digits, &rhs.digits);
        Ok((
            Self::from_parts(self.negative != rhs.negative, q),
            Self::from_parts(self.negative, r),
        ))
    }

    fn zero() -> Self {
        Self { negative: false, digits: vec![0] }
    }

    fn is_zero(&self) -> bool {
        self.digits.iter().all(|&d| d == 0)
    }

    fn from_parts(negative: bool, digits: Vec<u8>) -> Self {
        let first = digits.iter().position(|&d| d != 0).unwrap_or(digits.len().saturating_sub(1));
        let normalized = digits[first..].to_vec();
        let zero = normalized.len() == 1 && normalized[0] == 0;
        Self { negative: negative && !zero, digits: normalized }
    }

    fn cmp_abs(a: &[u8], b: &[u8]) -> Ordering {
        match a.len().cmp(&b.len()) {
            Ordering::Equal => a.cmp(b),
            other => other,
        }
    }

    fn add_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut i = a.len();
        let mut j = b.len();
        let mut carry = 0u16;
        while i > 0 || j > 0 || carry > 0 {
            let da = if i > 0 { i -= 1; a[i] as u16 } else { 0 };
            let db = if j > 0 { j -= 1; b[j] as u16 } else { 0 };
            let value = da + db + carry;
            out.push((value % 10) as u8);
            carry = value / 10;
        }
        out.reverse();
        out
    }

    // Assumes a >= b.
    fn sub_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut rev = Vec::with_capacity(a.len());
        let mut i = a.len();
        let mut j = b.len();
        let mut borrow = 0i16;
        while i > 0 {
            i -= 1;
            let db = if j > 0 { j -= 1; b[j] as i16 } else { 0 };
            let mut value = a[i] as i16 - db - borrow;
            if value < 0 {
                value += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            rev.push(value as u8);
        }
        rev.reverse();
        let first = rev.iter().position(|&d| d != 0).unwrap_or(rev.len().saturating_sub(1));
        rev[first..].to_vec()
    }

    fn mul_abs_digit(a: &[u8], digit: u8) -> Vec<u8> {
        let mut rev = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u16;
        for &d in a.iter().rev() {
            let value = d as u16 * digit as u16 + carry;
            rev.push((value % 10) as u8);
            carry = value / 10;
        }
        while carry != 0 {
            rev.push((carry % 10) as u8);
            carry /= 10;
        }
        rev.reverse();
        let first = rev.iter().position(|&d| d != 0).unwrap_or(rev.len().saturating_sub(1));
        rev[first..].to_vec()
    }

    fn div_abs(a: &[u8], b: &[u8]) -> (Vec<u8>, Vec<u8>) {
        if Self::cmp_abs(a, b) == Ordering::Less {
            return (vec![0], a.to_vec());
        }
        let mut quotient = Vec::with_capacity(a.len());
        let mut remainder = vec![0u8];
        for &digit in a {
            if remainder.len() == 1 && remainder[0] == 0 {
                remainder[0] = digit;
            } else {
                remainder.push(digit);
            }
            let mut qdigit = 0u8;
            for candidate in (1..=9).rev() {
                let product = Self::mul_abs_digit(b, candidate);
                if Self::cmp_abs(&product, &remainder) != Ordering::Greater {
                    remainder = Self::sub_abs(&remainder, &product);
                    qdigit = candidate;
                    break;
                }
            }
            quotient.push(qdigit);
        }
        let first = quotient.iter().position(|&d| d != 0).unwrap_or(quotient.len().saturating_sub(1));
        (quotient[first..].to_vec(), remainder)
    }
