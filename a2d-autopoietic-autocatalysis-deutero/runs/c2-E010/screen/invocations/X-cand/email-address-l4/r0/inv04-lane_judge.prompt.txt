fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {

    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();

            let b = input[i];
            bytes.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    bytes.push(input[i]);
                    positions.push(i);
                    i += 1;
                }
            } else {
                if b == b'"' {
                    protected_quote = false;
                }
                i += 1;
            }
            continue;

        if bytes.is_empty() && b == b'"' {
            protected_quote = true;

            positions.push(i);

            continue;

fn dot_atom_valid(bytes: &[u8]) -> bool {
    if bytes.is_empty() || bytes.len() > 64 {
        return false;
    }
    let mut previous_dot = true;
    for &b in bytes {
        if b == b'.' {
            if previous_dot {
                return false;
            }
            previous_dot = true;
        } else {
            if !is_atext(b) {
                return false;
            }
            previous_dot = false;
        }
    }
    !previous_dot
}

fn is_atext(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn parse_cleaned(c: &Cleaned, input_len: usize, allow_single: bool) -> Result<(Address, usize), EmailError> {
    for run in &c.runs {
        let left_at = run.start > 0 && c.bytes.get(run.start - 1) == Some(&b'@');
        let right_at = run.end < c.bytes.len() && c.bytes.get(run.end) == Some(&b'@');
        let at_edge = run.start == 0 || run.end == input_len;
        if !at_edge && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if c.bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if c.bytes[0] == b'"' {
        let mut i = 1;
        while i < c.bytes.len() {
            if c.bytes[i] == b'\\' {
                i += 2;
            } else if c.bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: map_pos(c, 0, input_len),
            });
        }
    }

    let mut ats = Vec::new();
    let protected_end = quote_end.unwrap_or(0);
    for i in 0..c.bytes.len() {
        if c.bytes[i] == b'@' && !(quote_end.is_some() && i <= protected_end) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: map_pos(c, ats[1], input_len),
        });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: map_pos(c, sep, input_len),
        });
    }

    let mut local = Vec::new();
    let mut origins = Vec::new();
    let quoted;
    if c.bytes[0] == b'"' {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: map_pos(c, end + 1, input_len),
            });
        }
        quoted = true;
        let mut i = 1;
        while i < end {
            let b = c.bytes[i];
            if b == b'\\' {
                if i + 1 >= end {
                    return Err(EmailError::BadEscape {
                        at: map_pos(c, i, input_len),
                    });
                }
                let next = c.bytes[i + 1];
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape {
                        at: map_pos(c, i, input_len),
                    });
                }
                local.push(next);
                origins.push(map_pos(c, i, input_len));
                i += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: map_pos(c, i, input_len),
                    });
                }
                local.push(b);
                origins.push(map_pos(c, i, input_len));
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: map_pos(c, end, input_len),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
    } else {
        quoted = false;
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: map_pos(c, 64, input_len),
            });
        }
        let bytes = &c.bytes[..sep];
        for i in 0..bytes.len() {
            let b = bytes[i];
            if b == b'.' {
                if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: map_pos(c, i, input_len),
                    });
                }
            } else if !is_atext(b) {
                return Err(EmailError::LocalChar {
                    at: map_pos(c, i, input_len),
                });
            }
        }
        local.extend_from_slice(bytes);
        for i in 0..bytes.len() {
            origins.push(map_pos(c, i, input_len));
        }
    }

    let domain_start = sep + 1;
    if domain_start >= c.bytes.len() {
        return Err(EmailError::EmptyDomain {
            at: map_pos(c, domain_start, input_len),
        });
    }

    let mut domain_end = c.bytes.len();
    let mut removed_root = None;
    if domain_end > domain_start && c.bytes[domain_end - 1] == b'.' {
        removed_root = Some(domain_end - 1);
        domain_end -= 1;
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: map_pos(c, domain_start, input_len),
        });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: map_pos(c, domain_start + 253, input_len),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || c.bytes[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }
    for (li, &(s, e)) in labels.iter().enumerate() {
        if s == e {
            let at = if li + 1 == labels.len() && removed_root.is_some() {
                map_pos(c, removed_root.unwrap(), input_len)
            } else {
                map_pos(c, s, input_len)
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if e - s > 63 {
            return Err(EmailError::LabelTooLong {
                at: map_pos(c, s + 63, input_len),
            });
        }
        if c.bytes[s] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: map_pos(c, s, input_len),
            });
        }
        if c.bytes[e - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: map_pos(c, e - 1, input_len),
            });
        }
        for i in s..e {
            let b = c.bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: map_pos(c, i, input_len),
                });
            }
        }
    }
    if labels.len() < 2 && !allow_single {
        return Err(EmailError::DomainNeedsDot {
            at: map_pos(c, domain_start, input_len),
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if c.bytes[last_start..last_end].iter().all(|b| b.is_ascii_digit()) {
        return Err(EmailError::NumericTld {
            at: map_pos(c, last_start, input_len),
        });
    }

    let local_string = String::from_utf8(local).unwrap();
    let domain = String::from_utf8(
        c.bytes[domain_start..domain_end]
            .iter()
            .map(u8::to_ascii_lowercase)
            .collect(),
    )
    .unwrap();
    Ok((
        Address {
            local: local_string,
            domain,
            quoted,
        },
        origins.first().copied().unwrap_or(map_pos(c, 0, input_len)),
    ))
}

fn parse_with_policy(input: &str, policy: &Policy, allow_single: bool) -> Result<(Address, usize), EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }
    let cleaned = cleanup(raw)?;
    let (mut address, first_origin) = parse_cleaned(&cleaned, raw.len(), allow_single)?;

    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }
    if policy.strip_subaddress {
        if let Some(plus) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if plus == 0 {
                let at = if address.quoted {
                    let mut decoded_index = 0usize;
                    let mut source = 1usize;
                    while source < cleaned.bytes.len() && cleaned.bytes[source] != b'"' {
                        if cleaned.bytes[source] == b'\\' {
                            source += 2;
                        } else {
                            if decoded_index == 0 {
                                break;
                            }
                            decoded_index += 1;
                            source += 1;
                        }
                    }
                    map_pos(&cleaned, source, raw.len())
                } else {
                    first_origin
                };
                return Err(EmailError::EmptyNormalizedLocal { at });
            }
            address.local.truncate(plus);
        }
    }
    if dot_atom_valid(address.local.as_bytes()) {
        if policy.unquote_safe {
            address.quoted = false;
        }
    } else {
        address.quoted = true;
    }
    Ok((address, first_origin))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let policy = Policy::default();
    parse_with_policy(input, &policy, false).map(|x| x.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, policy, policy.allow_single_label).map(|x| x.0)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                if let Some((_, _, first)) = keys
                    .iter()
                    .find(|(local, domain, _)| local == &address.local && domain == &address.domain)
                {
                    duplicate_of[index] = Some(*first);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), index));
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}

#[derive(Clone)]
struct Clean {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize, usize)>,
}

fn is_hws(b: u8) -> bool { b == b' ' || b == b'\t' }
fn is_printable(b: u8) -> bool { (0x20..=0x7e).contains(&b) }
fn is_atom_special(b: u8) -> bool { b"!#$%&'*+-/=?^_`{|}~".contains(&b) }
fn is_atom_byte(b: u8) -> bool { b.is_ascii_alphanumeric() || is_atom_special(b) }

fn cleanup(input: &[u8]) -> Result<Clean, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut quote_prefix = false;
    let mut quote_closed = false;

    while i < input.len() {
        if quote_prefix && !quote_closed {
            let b = input[i];
            bytes.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    bytes.push(input[i]);
                    positions.push(i);
                    i += 1;
                }
                continue;
            }
            if b == b'"' { quote_closed = true; }
            i += 1;
            continue;
        }

        let start = i;
        let mut removed = false;
        loop {
            if i >= input.len() { break; }
            if is_hws(input[i]) {
                removed = true;
                i += 1;
                continue;
            }
            if input[i] == b'\r' || input[i] == b'\n' {
                let at = i;
                if input[i] != b'\r' || i + 2 >= input.len()
                    || input[i + 1] != b'\n' || !is_hws(input[i + 2]) {
                    return Err(EmailError::BadFold { at });
                }
                removed = true;
                i += 2;
                while i < input.len() && is_hws(input[i]) { i += 1; }
                continue;
            }
            if input[i] == b'(' {
                removed = true;
                let outer = i;
                let mut stack = Vec::new();
                stack.push(i);
                i += 1;
                while i < input.len() {
                    let b = input[i];
                    if b == b'\\' {
                        let esc = i;
                        if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                            return Err(EmailError::CommentEscape { at: esc });
                        }
                        i += 2;
                        continue;
                    }
                    if b == b'\r' || b == b'\n' {
                        if b != b'\r' || i + 2 >= input.len()
                            || input[i + 1] != b'\n' || !is_hws(input[i + 2]) {
                            return Err(EmailError::BadFold { at: i });
                        }
                        i += 2;
                        while i < input.len() && is_hws(input[i]) { i += 1; }
                        continue;
                    }
                    if b == b'(' {
                        if stack.len() == 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        stack.push(i);
                        i += 1;
                        continue;
                    }
                    if b == b')' {
                        stack.pop();
                        i += 1;
                        if stack.is_empty() { break; }
                        continue;
                    }
                    if !is_printable(b) && !is_hws(b) {
                        return Err(EmailError::CommentChar { at: i });
                    }
                    i += 1;
                }
                if !stack.is_empty() {
                    return Err(EmailError::UnclosedComment { at: outer });
                }
                continue;
            }
            if input[i] == b')' {
                return Err(EmailError::UnexpectedClose { at: i });
            }
            break;
        }

        if removed {
            runs.push((start, i, bytes.len()));
        }
        if i >= input.len() { break; }

        let b = input[i];
        bytes.push(b);
        positions.push(i);
        if bytes.len() == 1 && b == b'"' {
            quote_prefix = true;
        }
        i += 1;
    }

    Ok(Clean { bytes, positions, runs })
}

fn parse_internal(input: &str, allow_single: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() { return Err(EmailError::EmptyInput); }
    if let Some(at) = raw.iter().position(|b| *b >= 128) {
        return Err(EmailError::NonAscii { at });
    }
    let clean = cleanup(raw)?;
    for &(start, end, boundary) in &clean.runs {
        let before = boundary == 0 || clean.bytes.get(boundary.wrapping_sub(1)) == Some(&b'@');
        let after = boundary == clean.bytes.len() || clean.bytes.get(boundary) == Some(&b'@');
        if !before && !after {
            return Err(EmailError::CfwsBoundary { at: start });
        }
        let _ = end;
    }
    if clean.bytes.is_empty() { return Err(EmailError::EmptyInput); }

    let mut quote_end = None;
    if clean.bytes[0] == b'"' {
        let mut i = 1;
        while i < clean.bytes.len() {
            if clean.bytes[i] == b'\\' {
                i += 2;
                continue;
            }
            if clean.bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            }
            i += 1;
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: clean.positions[0] });
        }
    }

    let mut ats = Vec::new();
    let mut i = 0;
    while i < clean.bytes.len() {
        if let Some(qe) = quote_end {
            if i <= qe { i += 1; continue; }
        }
        if clean.bytes[i] == b'@' { ats.push(i); }
        i += 1;
    }
    if ats.is_empty() { return Err(EmailError::MissingAt); }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt { at: clean.positions[ats[1]] });
    }
    let sep = ats[0];
    if sep == 0 { return Err(EmailError::EmptyLocal { at: clean.positions[sep] }); }

    let mut origins = Vec::new();
    let (local, quoted) = if clean.bytes[0] == b'"' {
        let qe = quote_end.unwrap();
        if qe + 1 != sep {
            return Err(EmailError::TrailingLocal { at: clean.positions[qe + 1] });
        }
        let mut decoded = Vec::new();
        let mut decoded_origins = Vec::new();
        let mut j = 1;
        while j < qe {
            let b = clean.bytes[j];
            if b == b'\\' {
                if j + 1 >= qe || !matches!(clean.bytes[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: clean.positions[j] });
                }
                decoded.push(clean.bytes[j + 1]);
                decoded_origins.push(clean.positions[j]);
                j += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar { at: clean.positions[j] });
                }
                decoded.push(b);
                decoded_origins.push(clean.positions[j]);
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: clean.positions[qe] });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: decoded_origins[64] });
        }
        origins = decoded_origins;
        (decoded, true)
    } else {
        if sep > 64 {
            return Err(EmailError::LocalTooLong { at: clean.positions[64] });
        }
        let mut prev_dot = false;
        for j in 0..sep {
            let b = clean.bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == sep || prev_dot {
                    return Err(EmailError::LocalDot { at: clean.positions[j] });
                }
                prev_dot = true;
            } else {
                prev_dot = false;
                if !is_atom_byte(b) {
                    return Err(EmailError::LocalChar { at: clean.positions[j] });
                }
            }
            origins.push(clean.positions[j]);
        }
        (clean.bytes[..sep].to_vec(), false)
    };

    let domain_start = sep + 1;
    let raw_domain = &clean.bytes[domain_start..];
    let mut domain_end = clean.bytes.len();
    let root_pos = if domain_end > domain_start && clean.bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(clean.positions[domain_end])
    } else { None };

    if domain_end == domain_start {
        let at = clean.positions.get(domain_start).copied().unwrap_or(input.len());
        return Err(EmailError::EmptyDomain { at });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong { at: clean.positions[domain_start + 253] });
    }

    let mut label_starts = Vec::new();
    let mut label_ends = Vec::new();
    let mut s = domain_start;
    for j in domain_start..=domain_end {
        if j == domain_end || clean.bytes[j] == b'.' {
            label_starts.push(s);
            label_ends.push(j);
            if s == j {
                let at = if j == domain_end && j == domain_start {
                    clean.positions.get(j).copied().unwrap_or(input.len())
                } else if j == domain_end {
                    root_pos.unwrap_or_else(|| clean.positions.get(j).copied().unwrap_or(input.len()))
                } else {
                    clean.positions[j]
                };
                return Err(EmailError::EmptyLabel { at });
            }
            let len = j - s;
            if len > 63 {
                return Err(EmailError::LabelTooLong { at: clean.positions[s + 63] });
            }
            if clean.bytes[s] == b'-' {
                return Err(EmailError::LabelHyphen { at: clean.positions[s] });
            }
            if clean.bytes[j - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: clean.positions[j - 1] });
            }
            for k in s..j {
                let b = clean.bytes[k];
                if !b.is_ascii_alphanumeric() && b != b'-' {
                    return Err(EmailError::DomainChar { at: clean.positions[k] });
                }
            }
            s = j + 1;
        }
    }
    if label_starts.len() < 2 && !allow_single {
        return Err(EmailError::DomainNeedsDot { at: clean.positions[domain_start] });
    }
    let last_s = *label_starts.last().unwrap();
    let last_e = *label_ends.last().unwrap();
    if clean.bytes[last_s..last_e].iter().all(|b| b.is_ascii_digit()) {
        return Err(EmailError::NumericTld { at: clean.positions[last_s] });
    }

    let mut dom = raw_domain[..domain_len].to_vec();
    dom.make_ascii_lowercase();
    let address = Address {
        local: String::from_utf8(local).unwrap(),
        domain: String::from_utf8(dom).unwrap(),
        quoted,
    };
    Ok((address, origins))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_internal(input, false).map(|x| x.0)
}

fn valid_dot_atom(local: &[u8]) -> bool {
    if local.is_empty() || local.len() > 64 { return false; }
    let mut prev_dot = false;
    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || prev_dot { return false; }
            prev_dot = true;
        } else {
            if !is_atom_byte(b) { return false; }
            prev_dot = false;
        }
    }
    true
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_internal(input, policy.allow_single_label)?;
    let mut local = address.local.into_bytes();
    if policy.local_case == LocalCase::Lower {
        local.make_ascii_lowercase();
    }
    let mut origin_map = origins;
    if policy.strip_subaddress {
        if let Some(i) = local.iter().position(|b| *b == b'+') {
            if i == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origin_map.get(0).copied().unwrap_or(input.len()),
                });
            }
            local.truncate(i);
            origin_map.truncate(i);
        }
    }
    let safe = valid_dot_atom(&local);
    address.local = String::from_utf8(local).unwrap();
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' { out.push('\\'); }
            out.push(c);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(e) => errors.push((i, e)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first) = seen.get(&key) {
                    duplicate_of[i] = Some(first);
                } else {
                    seen.insert(key, i);
                    source_indices.push(i);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult { addresses, source_indices, duplicate_of, errors }
}

fn horizontal(b: u8) -> bool { b == b' ' || b == b'\t' }

fn fold(bytes: &[u8], at: usize) -> Option<usize> {
    if bytes.get(at) != Some(&b'\r') || bytes.get(at + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = at + 2;
    if !bytes.get(j).copied().map(horizontal).unwrap_or(false) { return None; }
    while j < bytes.len() && horizontal(bytes[j]) { j += 1; }
    Some(j)
}

fn lexical_cleanup(bytes: &[u8]) -> Result<(Vec<Byte>, Vec<Run>), EmailError> {
    let mut output = Vec::with_capacity(bytes.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut comment_depth = 0usize;
    let mut outer_comment = 0usize;
    let mut protected_quote = false;

    while i < bytes.len() {
        let b = bytes[i];

        if protected_quote {
            output.push(Byte { value: b, pos: i });
            if b == b'\\' {
                i += 1;
                if i < bytes.len() {
                    output.push(Byte { value: bytes[i], pos: i });
                    i += 1;
                }
            } else {
                if b == b'"' { protected_quote = false; }
                i += 1;
            }
            continue;
        }

        if comment_depth != 0 {
            if horizontal(b) || b == b'\r' || b == b'\n' {
                let start = i;
                loop {
                    if i < bytes.len() && horizontal(bytes[i]) { i += 1; continue; }
                    if i < bytes.len() && (bytes[i] == b'\r' || bytes[i] == b'\n') {
                        if let Some(end) = fold(bytes, i) { i = end; continue; }
                        return Err(EmailError::BadFold { at: i });
                    }
                    break;
                }
                runs.push(Run { start, end: i });
            } else if b == b'\\' {
                if i + 1 == bytes.len() || !(0x20..=0x7e).contains(&bytes[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
            } else if b == b'(' {
                if comment_depth == 3 { return Err(EmailError::CommentDepth { at: i }); }
                comment_depth += 1;
                i += 1;
            } else if b == b')' {
                comment_depth -= 1;
                i += 1;
            } else if (0x20..=0x7e).contains(&b) {
                i += 1;
            } else {
                return Err(EmailError::CommentChar { at: i });
            }
            continue;
        }

        if horizontal(b) || b == b'\r' || b == b'\n' {
            let start = i;
            loop {
                if i < bytes.len() && horizontal(bytes[i]) { i += 1; continue; }
                if i < bytes.len() && (bytes[i] == b'\r' || bytes[i] == b'\n') {
                    if let Some(end) = fold(bytes, i) { i = end; continue; }
                    return Err(EmailError::BadFold { at: i });
                }
                break;
            }
            runs.push(Run { start, end: i });
            continue;
        }

        if b == b'(' {
            outer_comment = i;
            comment_depth = 1;
            i += 1;
            continue;
        }
        if b == b')' { return Err(EmailError::UnexpectedClose { at: i }); }

        if b == b'"' && output.is_empty() { protected_quote = true; }
        output.push(Byte { value: b, pos: i });
        i += 1;
    }

    if comment_depth != 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }

    for run in &runs {
        let before = output.iter().take_while(|b| b.pos < run.start).last();
        let after = output.iter().find(|b| b.pos >= run.end);
        if before.is_some() && after.is_some()
            && before.map(|b| b.value != b'@').unwrap_or(true)
            && after.map(|b| b.value != b'@').unwrap_or(true)
        {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    Ok((output, runs))
}

fn valid_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() || bytes.len() > 64 { return false; }
    for i in 0..bytes.len() {
        let b = bytes[i];
        if b == b'.' {
            if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' { return false; }
        } else if !b.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
            return false;
        }
    }
    true
}

fn parse_impl(input: &str, allow_single: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() { return Err(EmailError::EmptyInput); }
    if let Some(at) = raw.iter().position(|b| *b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (mapped, _) = lexical_cleanup(raw)?;
    if mapped.is_empty() { return Err(EmailError::EmptyInput); }
    let bytes: Vec<u8> = mapped.iter().map(|b| b.value).collect();
    let pos: Vec<usize> = mapped.iter().map(|b| b.pos).collect();

    let quote_end = if bytes.first() == Some(&b'"') {
        let mut j = 1;
        let mut found = None;
        while j < bytes.len() {
            if bytes[j] == b'\\' { j += 2; }
            else if bytes[j] == b'"' { found = Some(j); break; }
            else { j += 1; }
        }
        match found {
            Some(x) => Some(x),
            None => return Err(EmailError::UnclosedQuote { at: pos[0] }),
        }
    } else { None };

    let mut separators = Vec::new();
    for i in 0..bytes.len() {
        if quote_end.map(|q| i <= q).unwrap_or(false) { continue; }
        if bytes[i] == b'@' { separators.push(i); }
    }
    if separators.is_empty() { return Err(EmailError::MissingAt); }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt { at: pos[separators[1]] });
    }
    let sep = separators[0];
    if sep == 0 { return Err(EmailError::EmptyLocal { at: pos[sep] }); }

    let local_part = &bytes[..sep];
    let quoted = local_part.first() == Some(&b'"');
    let mut local = Vec::new();
    let mut origins = Vec::new();

    if !quoted {
        if local_part.len() > 64 {
            return Err(EmailError::LocalTooLong { at: pos[64] });
        }
        for i in 0..local_part.len() {
            let b = local_part[i];
            if b == b'.' {
                if i == 0 || i + 1 == local_part.len() || (i > 0 && local_part[i - 1] == b'.') {
                    return Err(EmailError::LocalDot { at: pos[i] });
                }
            } else if !b.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
                return Err(EmailError::LocalChar { at: pos[i] });
            }
            local.push(b);
            origins.push(pos[i]);
        }
    } else {
        let q = quote_end.unwrap();
        if q + 1 != sep { return Err(EmailError::TrailingLocal { at: pos[q + 1] }); }
        let mut i = 1;
        while i < q {
            let b = local_part[i];
            if b == b'\\' {
                if i + 1 >= q { return Err(EmailError::BadEscape { at: pos[i] }); }
                let escaped = local_part[i + 1];
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape { at: pos[i] });
                }
                local.push(escaped);
                origins.push(pos[i]);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) { return Err(EmailError::QuotedChar { at: pos[i] }); }
                local.push(b);
                origins.push(pos[i]);
                i += 1;
            }
        }
        if local.is_empty() { return Err(EmailError::EmptyLocal { at: pos[q] }); }
        if local.len() > 64 { return Err(EmailError::LocalTooLong { at: origins[64] }); }
    }

    let ds = sep + 1;
    let domain_start = pos.get(ds).copied().unwrap_or(input.len());
    let mut de = bytes.len();
    let mut root_dot = None;
    if de > ds && bytes[de - 1] == b'.' {
        de -= 1;
        root_dot = Some(pos[de]);
    }
    if de == ds { return Err(EmailError::EmptyDomain { at: domain_start }); }
    if de - ds > 253 { return Err(EmailError::DomainTooLong { at: pos[ds + 253] }); }

    let mut starts = Vec::new();
    let mut ends = Vec::new();
    let mut start = ds;
    for i in ds..=de {
        if i == de || bytes[i] == b'.' {
            if i == start {
                let at = if i == de { root_dot.unwrap_or(domain_start) } else { pos[i] };
                return Err(EmailError::EmptyLabel { at });
            }
            starts.push(start);
            ends.push(i);
            start = i + 1;
        }
    }

    for (&s, &e) in starts.iter().zip(&ends) {
        if e - s > 63 { return Err(EmailError::LabelTooLong { at: pos[s + 63] }); }
        if bytes[s] == b'-' { return Err(EmailError::LabelHyphen { at: pos[s] }); }
        if bytes[e - 1] == b'-' { return Err(EmailError::LabelHyphen { at: pos[e - 1] }); }
        for i in s..e {
            if !bytes[i].is_ascii_alphanumeric() && bytes[i] != b'-' {
                return Err(EmailError::DomainChar { at: pos[i] });
            }
        }
    }

    if starts.len() < 2 && !allow_single {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    let last_start = *starts.last().unwrap();
    let last_end = *ends.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos[last_start] });
    }

    let mut domain = bytes[ds..de].to_vec();
    domain.make_ascii_lowercase();
    Ok((Address {
        local: String::from_utf8(local).unwrap(),
        domain: String::from_utf8(domain).unwrap(),
        quoted,
    }, origins))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_impl(input, false).map(|x| x.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_impl(input, policy.allow_single_label)?;
    let mut local = address.local.into_bytes();
    if policy.local_case == LocalCase::Lower { local.make_ascii_lowercase(); }
    let mut origins = origins;

    if policy.strip_subaddress {
        if let Some(plus) = local.iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.get(plus).copied().unwrap_or(input.len())
                });
            }
            local.truncate(plus);
            origins.truncate(plus);
        }
    }

    let safe = valid_atom(&local);
    address.local = String::from_utf8(local).unwrap();
    if safe {
        if policy.unquote_safe { address.quoted = false; }
    } else {
        address.quoted = true;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' { result.push('\\'); }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut output = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut first_by_key: HashMap<(String, String), usize> = HashMap::new();

    for (index, value) in inputs.iter().enumerate() {
        match normalize_email(value, policy) {
            Err(error) => output.errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                match first_by_key.get(&key) {
                    Some(first_index) => output.duplicate_of[index] = Some(*first_index),
                    None => {
                        first_by_key.insert(key, index);
                        output.source_indices.push(index);
                        output.addresses.push(address);
                    }
                }
            }
        }
    }
    output
}

struct Scan {
    kept: Vec<(u8, usize)>,
    runs: Vec<(usize, usize, usize)>,
}
fn hws(b: u8) -> bool { matches!(b, b' ' | b'\t') }
fn printable(b: u8) -> bool { (0x20..=0x7e).contains(&b) }
fn fold_at(s: &[u8], i: usize) -> bool {
    i + 2 < s.len() && s[i] == b'\r' && s[i + 1] == b'\n' && hws(s[i + 2])
}
fn push_run(runs: &mut Vec<(usize, usize, usize)>, start: usize, end: usize, boundary: usize, len: usize) {
    if end > start {
        if let Some(last) = runs.last_mut() {
            if last.1 == start && last.2 == boundary {
                last.1 = end;
                return;
            }
        }
        runs.push((start, end, boundary.min(len)));
    }
}
fn scan_cfws(s: &[u8]) -> Result<Scan, EmailError> {
    let mut kept = Vec::with_capacity(s.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut protected = false;
    let mut closed = false;
    while i < s.len() {
        if protected && !closed {
            let b = s[i];
            kept.push((b, i));
            if b == b'\\' {
                i += 1;
                if i < s.len() { kept.push((s[i], i)); i += 1; }
            } else {
                if b == b'"' { closed = true; }
                i += 1;
            }
            continue;
        }
        let run_start_i = i;
        let run_start_k = kept.len();
        let mut had = false;
        loop {
            if i >= s.len() { break; }
            let b = s[i];
            if hws(b) {
                had = true;
                i += 1;
            } else if b == b'\r' || b == b'\n' {
                if !fold_at(s, i) { return Err(EmailError::BadFold { at: i }); }
                had = true;
                i += 2;
                while i < s.len() && hws(s[i]) { i += 1; }
            } else if b == b'(' {
                had = true;
                let outer = i;
                let mut depth = 1usize;
                i += 1;
                while i < s.len() && depth > 0 {
                    let c = s[i];
                    if c == b'\\' {
                        let at = i;
                        if i + 1 >= s.len() || !printable(s[i + 1]) {
                            return Err(EmailError::CommentEscape { at });
                        }
                        i += 2;
                    } else if c == b'\r' || c == b'\n' {
                        if !fold_at(s, i) { return Err(EmailError::BadFold { at: i }); }
                        i += 2;
                        while i < s.len() && hws(s[i]) { i += 1; }
                    } else if c == b'(' {
                        if depth == 3 { return Err(EmailError::CommentDepth { at: i }); }
                        depth += 1;
                        i += 1;
                    } else if c == b')' {
                        depth -= 1;
                        i += 1;
                    } else if !printable(c) && !hws(c) {
                        return Err(EmailError::CommentChar { at: i });
                    } else {
                        i += 1;
                    }
                }
                if depth != 0 { return Err(EmailError::UnclosedComment { at: outer }); }
            } else {
                break;
            }
        }
        if had {
            push_run(&mut runs, run_start_i, i, run_start_k, s.len());
        }
        if i < s.len() {
            if s[i] == b')' { return Err(EmailError::UnexpectedClose { at: i }); }
            let b = s[i];
            kept.push((b, i));
            if kept.len() == 1 && b == b'"' { protected = true; }
            i += 1;
        }
    }
    Ok(Scan { kept, runs })
}

fn parse_impl(input: &str, allow_single: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let s = input.as_bytes();
    if s.is_empty() { return Err(EmailError::EmptyInput); }
    if let Some(i) = s.iter().position(|b| *b >= 128) { return Err(EmailError::NonAscii { at: i }); }
    let scan = scan_cfws(s)?;
    let bytes: Vec<u8> = scan.kept.iter().map(|x| x.0).collect();
    let pos: Vec<usize> = scan.kept.iter().map(|x| x.1).collect();

    for &(start, end, boundary) in &scan.runs {
        let left = boundary == 0 || bytes.get(boundary.wrapping_sub(1)) == Some(&b'@');
        let right = boundary == bytes.len() || bytes.get(boundary) == Some(&b'@');
        if !left && !right { return Err(EmailError::CfwsBoundary { at: start }); }
        let _ = end;
    }
    if bytes.is_empty() { return Err(EmailError::EmptyInput); }

    let mut qend = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' { i += 2; continue; }
            if bytes[i] == b'"' { qend = Some(i); break; }
            i += 1;
        }
        if qend.is_none() { return Err(EmailError::UnclosedQuote { at: pos[0] }); }
    }
    let mut ats = Vec::new();
    for i in 0..bytes.len() {
        if qend.map_or(false, |q| i <= q) { continue; }
        if bytes[i] == b'@' { ats.push(i); }
    }
    if ats.is_empty() { return Err(EmailError::MissingAt); }
    if ats.len() > 1 { return Err(EmailError::MultipleAt { at: pos[ats[1]] }); }
    let sep = ats[0];
    if sep == 0 { return Err(EmailError::EmptyLocal { at: pos[sep] }); }

    let mut origin = Vec::new();
    let (local, quoted) = if bytes[0] == b'"' {
        let q = qend.unwrap();
        if q + 1 != sep { return Err(EmailError::TrailingLocal { at: pos[q + 1] }); }
        let mut out = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < q {
            if bytes[j] == b'\\' {
                if j + 1 >= q || !matches!(bytes[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: pos[j] });
                }
                out.push(bytes[j + 1]);
                origins.push(pos[j]);
                j += 2;
            } else {
                if !printable(bytes[j]) { return Err(EmailError::QuotedChar { at: pos[j] }); }
                out.push(bytes[j]);
                origins.push(pos[j]);
                j += 1;
            }
        }
        if out.is_empty() { return Err(EmailError::EmptyLocal { at: pos[q] }); }
        if out.len() > 64 { return Err(EmailError::LocalTooLong { at: origins[64] }); }
        origin = origins;
        (out, true)
    } else {
        if sep > 64 { return Err(EmailError::LocalTooLong { at: pos[64] }); }
        let mut prev_dot = false;
        for j in 0..sep {
            let b = bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == sep || prev_dot { return Err(EmailError::LocalDot { at: pos[j] }); }
                prev_dot = true;
            } else {
                prev_dot = false;
                if !b.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
                    return Err(EmailError::LocalChar { at: pos[j] });
                }
            }
            origin.push(pos[j]);
        }
        (bytes[..sep].to_vec(), false)
    };

    let ds = sep + 1;
    let mut de = bytes.len();
    let root = if de > ds && bytes[de - 1] == b'.' { de -= 1; Some(pos[de]) } else { None };
    if de == ds {
        return Err(EmailError::EmptyDomain { at: pos.get(ds).copied().unwrap_or(input.len()) });
    }
    if de - ds > 253 { return Err(EmailError::DomainTooLong { at: pos[ds + 253] }); }

    let mut starts = Vec::new();
    let mut ends = Vec::new();
    let mut st = ds;
    for j in ds..=de {
        if j == de || bytes[j] == b'.' {
            if st == j {
                let at = if j == de {
                    root.unwrap_or_else(|| pos.get(j).copied().unwrap_or(input.len()))
                } else { pos[j] };
                return Err(EmailError::EmptyLabel { at });
            }
            let n = j - st;
            if n > 63 { return Err(EmailError::LabelTooLong { at: pos[st + 63] }); }
            if bytes[st] == b'-' { return Err(EmailError::LabelHyphen { at: pos[st] }); }
            if bytes[j - 1] == b'-' { return Err(EmailError::LabelHyphen { at: pos[j - 1] }); }
            for k in st..j {
                if !bytes[k].is_ascii_alphanumeric() && bytes[k] != b'-' {
                    return Err(EmailError::DomainChar { at: pos[k] });
                }
            }
            starts.push(st);
            ends.push(j);
            st = j + 1;
        }
    }
    if starts.len() < 2 && !allow_single { return Err(EmailError::DomainNeedsDot { at: pos[ds] }); }
    let ls = starts[starts.len() - 1];
    let le = ends[ends.len() - 1];
    if bytes[ls..le].iter().all(u8::is_ascii_digit) { return Err(EmailError::NumericTld { at: pos[ls] }); }

    let mut domain = bytes[ds..de].to_vec();
    domain.make_ascii_lowercase();
    Ok((Address {
        local: String::from_utf8(local).unwrap(),
        domain: String::from_utf8(domain).unwrap(),
        quoted,
    }, origin))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_impl(input, false).map(|v| v.0)
}

fn dot_atom(s: &[u8]) -> bool {
    if s.is_empty() || s.len() > 64 { return false; }
    let mut prev = false;
    for (i, &b) in s.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == s.len() || prev { return false; }
            prev = true;
        } else if b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
            prev = false;
        } else {
            return false;
        }
    }
    true
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut a, mut origins) = parse_impl(input, policy.allow_single_label)?;
    let mut local = a.local.into_bytes();
    if policy.local_case == LocalCase::Lower { local.make_ascii_lowercase(); }
    if policy.strip_subaddress {
        if let Some(p) = local.iter().position(|b| *b == b'+') {
            if p == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.first().copied().unwrap_or(input.len()),
                });
            }
            local.truncate(p);
            origins.truncate(p);
        }
    }
    let safe = dot_atom(&local);
    a.local = String::from_utf8(local).unwrap();
    if !safe { a.quoted = true; }
    else if policy.unquote_safe { a.quoted = false; }
    Ok(a)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' { out.push('\\'); }
            out.push(c);
        }
        out.push('"');
    } else { out.push_str(&address.local); }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();
    for (i, s) in inputs.iter().enumerate() {
        match normalize_email(s, policy) {
            Err(e) => errors.push((i, e)),
            Ok(a) => {
                let key = (a.local.clone(), a.domain.clone());
                if let Some(&first) = seen.get(&key) {
                    duplicate_of[i] = Some(first);
                } else {
                    seen.insert(key, i);
                    source_indices.push(i);
                    addresses.push(a);
                }
            }
        }
    }
    DedupResult { addresses, source_indices, duplicate_of, errors }
}

fn is_hws(b: u8) -> bool { b == b' ' || b == b'\t' }

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if i + 2 >= bytes.len() || bytes[i] != b'\r' || bytes[i + 1] != b'\n' || !is_hws(bytes[i + 2]) {
        return None;
    }
    let mut j = i + 2;
    while j < bytes.len() && is_hws(bytes[j]) { j += 1; }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let mut clean = Vec::with_capacity(input.len());
    let mut map = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut protected = false;
    let mut quote_closed = true;

    while i < input.len() {
        let b = input[i];
        if protected {
            clean.push(b); map.push(i);
            if b == b'\\' {
                if i + 1 < input.len() {
                    i += 1;
                    clean.push(input[i]); map.push(i);
                }
            } else if b == b'"' {
                protected = false;
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        if b == b'"' && clean.is_empty() {
            protected = true;
            quote_closed = false;
            clean.push(b); map.push(i); i += 1;
            continue;
        }

        if b == b'\r' || b == b'\n' {
            if let Some(end) = fold_end(input, i) {
                let start = i;
                i = end;
                runs.push(Run { start, end });
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }

        if is_hws(b) {
            let start = i;
            while i < input.len() && is_hws(input[i]) { i += 1; }
            runs.push(Run { start, end: i });
            continue;
        }

        if b == b'(' {
            let outer = i;
            let mut depth = 1usize;
            i += 1;
            while i < input.len() {
                let c = input[i];
                if c == b'\\' {
                    if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                        return Err(EmailError::CommentEscape { at: i });
                    }
                    i += 2;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if c == b'(' {
                    depth += 1;
                    if depth > 3 { return Err(EmailError::CommentDepth { at: i }); }
                } else if c == b')' {
                    depth -= 1;
                    i += 1;
                    if depth == 0 { break; }
                    continue;
                } else if !is_printable(c) && !is_hws(c) {
                    return Err(EmailError::CommentChar { at: i });
                }
                i += 1;
            }
            if depth != 0 { return Err(EmailError::UnclosedComment { at: outer }); }
            runs.push(Run { start: outer, end: i });
            continue;
        }

        if b == b')' { return Err(EmailError::UnexpectedClose { at: i }); }
        clean.push(b); map.push(i);
        i += 1;
    }

    let _ = quote_closed;
    Ok((clean, map, runs))
}

fn local_dot_valid(s: &[u8]) -> bool {
    if s.is_empty() { return false; }
    for (i, &b) in s.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == s.len() || s[i - 1] == b'.' { return false; }
        } else if !is_alnum(b) && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
            return false;
        }
    }
    true
}

fn is_alnum(b: u8) -> bool {

fn parse_core(clean: &[u8], map: &[usize], allow_single: bool) -> Result<(Address, Vec<usize>), EmailError> {
    if clean.is_empty() { return Err(EmailError::EmptyInput); }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut i = 1;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
                continue;
            }
            if clean[i] == b'"' { quote_end = Some(i); break; }
            i += 1;
        }
        if quote_end.is_none() { return Err(EmailError::UnclosedQuote { at: map[0] }); }
    }

    let mut ats = Vec::new();
    let mut i = 0;
    while i < clean.len() {
        if let Some(qe) = quote_end {
            if i <= qe { i = qe + 1; continue; }
        }
        if clean[i] == b'@' { ats.push(i); }
        i += 1;
    }
    if ats.is_empty() { return Err(EmailError::MissingAt); }
    if ats.len() > 1 { return Err(EmailError::MultipleAt { at: map[ats[1]] }); }
    let at_i = ats[0];
    if at_i == 0 { return Err(EmailError::EmptyLocal { at: map[at_i] }); }

    let local_raw = &clean[..at_i];
    let mut local = Vec::new();
    let mut origins = Vec::new();
    let quoted;

    if local_raw[0] == b'"' {
        let qe = quote_end.unwrap();
        if qe + 1 != at_i { return Err(EmailError::TrailingLocal { at: map[qe + 1] }); }
        quoted = true;
        let mut j = 1;
        while j < qe {
            if local_raw[j] == b'\\' {
                if j + 1 >= qe || !matches!(local_raw[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: map[j] });
                }
                local.push(local_raw[j + 1]);
                origins.push(map[j]);
                j += 2;
            } else {
                if !is_printable(local_raw[j]) {
                    return Err(EmailError::QuotedChar { at: map[j] });
                }
                local.push(local_raw[j]);
                origins.push(map[j]);
                j += 1;
            }
        }
        if local.is_empty() { return Err(EmailError::EmptyLocal { at: map[qe] }); }
        if local.len() > 64 { return Err(EmailError::LocalTooLong { at: origins[64] }); }
    } else {
        quoted = false;
        if local_raw.len() > 64 { return Err(EmailError::LocalTooLong { at: map[64] }); }
        for (j, &b) in local_raw.iter().enumerate() {
            if b == b'.' {
                if j == 0 || j + 1 == local_raw.len() || local_raw[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: map[j] });
                }
            } else if !is_alnum(b) && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
                return Err(EmailError::LocalChar { at: map[j] });
            }
            local.push(b);
            origins.push(map[j]);
        }
    }

    let domain_start_i = at_i + 1;
    let domain_start = map.get(domain_start_i).copied().unwrap_or_else(|| map.last().map(|x| x + 1).unwrap_or(0));
    if domain_start_i == clean.len() { return Err(EmailError::EmptyDomain { at: domain_start }); }
    let mut dom_end = clean.len();
    let mut removed_dot = None;
    if clean[dom_end - 1] == b'.' {
        removed_dot = Some(dom_end - 1);
        dom_end -= 1;
    }
    if dom_end == domain_start_i {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    let dlen = dom_end - domain_start_i;
    if dlen > 253 { return Err(EmailError::DomainTooLong { at: map[domain_start_i + 253] }); }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start_i;
    for j in domain_start_i..dom_end {
        if clean[j] == b'.' {
            labels.push((start, j));
            start = j + 1;
        }
    }
    labels.push((start, dom_end));

    let mut label_count = 0;
    for &(s, e) in &labels {
        if s == e {
            let pos = if s == dom_end {
                removed_dot.map(|x| map[x]).unwrap_or(domain_start)
            } else if s > domain_start_i {
                map[s - 1]
            } else {
                map[s]
            };
            return Err(EmailError::EmptyLabel { at: pos });
        }
        let len = e - s;
        if len > 63 { return Err(EmailError::LabelTooLong { at: map[s + 63] }); }
        if clean[s] == b'-' { return Err(EmailError::LabelHyphen { at: map[s] }); }
        if clean[e - 1] == b'-' { return Err(EmailError::LabelHyphen { at: map[e - 1] }); }
        for j in s..e {
            let b = clean[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: map[j] });
            }
        }
        label_count += 1;
    }
    if label_count < 2 && !allow_single { return Err(EmailError::DomainNeedsDot { at: domain_start }); }

    let (last_s, last_e) = *labels.last().unwrap();
    if clean[last_s..last_e].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: map[last_s] });
    }

    let domain = clean[domain_start_i..dom_end].iter().map(|b| b.to_ascii_lowercase() as char).collect();
    let address = Address { local: String::from_utf8(local).unwrap(), domain, quoted };
    Ok((address, origins))
}

fn has_cfws_between(r: Run, map: &[usize], clean: &[u8]) -> bool {
    let first = map.iter().position(|&p| p >= r.end);
    let before = map.iter().rposition(|&p| p < r.start);
    let left = before.map(|i| clean[i] == b'@').unwrap_or(true);
    let right = first.map(|i| clean[i] == b'@').unwrap_or(true);
    left || right
}

fn parse_with_policy(input: &str, policy: &Policy, strict_single: bool) -> Result<(Address, Vec<usize>), EmailError> {
    if input.is_empty() { return Err(EmailError::EmptyInput); }
    let bytes = input.as_bytes();
    if let Some(i) = bytes.iter().position(|&b| b >= 128) { return Err(EmailError::NonAscii { at: i }); }
    let (clean, map, runs) = cleanup(bytes)?;
    for r in runs {
        if !has_cfws_between(r, &map, &clean) { return Err(EmailError::CfwsBoundary { at: r.start }); }
    }
    let (mut address, origins) = parse_core(&clean, &map, if strict_single { false } else { policy.allow_single_label })?;

    if !strict_single && policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }
    if !strict_single && policy.strip_subaddress {
        if let Some(i) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if i == 0 { return Err(EmailError::EmptyNormalizedLocal { at: origins.get(0).copied().unwrap_or(0) }); }
            address.local.truncate(i);
        }
    }
    if !strict_single {
        let valid = local_dot_valid(address.local.as_bytes());
        if !valid { address.quoted = true; }
        else if policy.unquote_safe { address.quoted = false; }
    }
    Ok((address, origins))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_policy(input, &Policy::default(), true).map(|x| x.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, policy, false).map(|x| x.0)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' { out.push('\\'); }
            out.push(c);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut keys: std::collections::HashMap<(String, String), usize> = std::collections::HashMap::new();

    for (idx, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(e) => errors.push((idx, e)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first) = keys.get(&key) {
                    duplicate_of[idx] = Some(first);
                } else {
                    keys.insert(key, idx);
                    source_indices.push(idx);
                    addresses.push(address);
                }
            }
        }
    }
    DedupResult { addresses, source_indices, duplicate_of, errors }
}

fn cleanup(input: &[u8]) -> Result<(Vec<Byte>, Vec<Run>), EmailError> {
    let mut kept: Vec<Byte> = Vec::with_capacity(input.len());
    let mut runs: Vec<Run> = Vec::new();
    let mut i = 0usize;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut quote_protected = false;

    while i < input.len() {
        let b = input[i];

        if quote_protected {
            kept.push(Byte { value: b, pos: i });
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    kept.push(Byte { value: input[i], pos: i });
                    i += 1;
                }
            } else {
                if b == b'"' { quote_protected = false; }
                i += 1;
            }
            continue;
        }

        if depth > 0 {
            if is_hws(b) || b == b'\r' || b == b'\n' {
                let start = i;
                while i < input.len() {
                    if is_hws(input[i]) { i += 1; continue; }
                    if input[i] == b'\r' || input[i] == b'\n' {
                        if let Some(end) = fold_end(input, i) { i = end; continue; }
                        return Err(EmailError::BadFold { at: i });
                    }
                    break;
                }
                runs.push(Run { start, end: i });
                continue;
            }
            if b == b'\\' {
                if i + 1 >= input.len()
                    || !(0x20..=0x7e).contains(&input[i + 1])
                {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'(' {
                if depth == 3 { return Err(EmailError::CommentDepth { at: i }); }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                continue;
            }
            if !(0x20..=0x7e).contains(&b) {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        if is_hws(b) || b == b'\r' || b == b'\n' {
            let start = i;
            while i < input.len() {
                if is_hws(input[i]) { i += 1; continue; }
                if input[i] == b'\r' || input[i] == b'\n' {
                    if let Some(end) = fold_end(input, i) { i = end; continue; }
                    return Err(EmailError::BadFold { at: i });
                }
                break;
            }
            runs.push(Run { start, end: i });
            continue;
        }

        if b == b'(' {
            outer_open = i;
            depth = 1;
            i += 1;
            continue;
        }
        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if b == b'"' && kept.is_empty() {
            quote_protected = true;
        }
        kept.push(Byte { value: b, pos: i });
        i += 1;
    }

    if depth > 0 { return Err(EmailError::UnclosedComment { at: outer_open }); }

    for run in &runs {
        let left_is_at = kept.last().map(|_| false).unwrap_or(false);
        let _ = left_is_at;
        let before = kept.iter().rev().find(|b| b.pos < run.start);
        let after = kept.iter().find(|b| b.pos >= run.end);
        let at_edge = before.is_none() || after.is_none();
        let adjacent_at = before.map(|b| b.value == b'@').unwrap_or(false)
            || after.map(|b| b.value == b'@').unwrap_or(false);
        if !at_edge && !adjacent_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    Ok((kept, runs))
}

fn local_atom_valid(bytes: &[u8]) -> bool {
    if bytes.is_empty() || bytes.len() > 64 { return false; }
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' {
                return false;
            }
        } else if !(b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)) {
            return false;
        }
    }
    true
}

fn parse_core(input: &str, allow_single: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() { return Err(EmailError::EmptyInput); }
    if let Some(i) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let (clean, _) = cleanup(raw)?;
    if clean.is_empty() { return Err(EmailError::EmptyInput); }
    let bytes: Vec<u8> = clean.iter().map(|b| b.value).collect();
    let positions: Vec<usize> = clean.iter().map(|b| b.pos).collect();

    let mut quoted_end: Option<usize> = None;
    if bytes.first() == Some(&b'"') {
        let mut i = 1usize;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quoted_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quoted_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: positions[0] });
        }
    }

    let mut ats = Vec::new();
    let mut i = 0usize;
    while i < bytes.len() {
        if let Some(end) = quoted_end {
            if i <= end { i += 1; continue; }
        }
        if bytes[i] == b'@' { ats.push(i); }
        i += 1;
    }
    if ats.is_empty() { return Err(EmailError::MissingAt); }
    if ats.len() > 1 { return Err(EmailError::MultipleAt { at: positions[ats[1]] }); }
    let sep = ats[0];
    if sep == 0 { return Err(EmailError::EmptyLocal { at: positions[sep] }); }

    let local_bytes = &bytes[..sep];
    let mut local = Vec::new();
    let mut origins = Vec::new();
    let quoted = local_bytes.first() == Some(&b'"');

    if !quoted {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: positions[64] });
        }
        for (i, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len()
                    || (i > 0 && local_bytes[i - 1] == b'.')
                {
                    return Err(EmailError::LocalDot { at: positions[i] });
                }
            } else if !(b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)) {
                return Err(EmailError::LocalChar { at: positions[i] });
            }
            local.push(b);
            origins.push(positions[i]);
        }
    } else {
        let end = quoted_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal { at: positions[end + 1] });
        }
        let mut i = 1usize;
        while i < end {
            let b = local_bytes[i];
            if b == b'\\' {
                if i + 1 >= end {
                    return Err(EmailError::BadEscape { at: positions[i] });
                }
                let next = local_bytes[i + 1];
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape { at: positions[i] });
                }
                local.push(next);
                origins.push(positions[i]);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: positions[i] });
                }
                local.push(b);
                origins.push(positions[i]);
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: positions[end] });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    }

    let domain_start_idx = sep + 1;
    let domain_start_pos = positions.get(domain_start_idx).copied().unwrap_or(input.len());
    let mut domain_end = bytes.len();
    let mut root_pos = None;
    if domain_end > domain_start_idx && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_pos = Some(positions[domain_end]);
    }
    if domain_end == domain_start_idx {
        return Err(EmailError::EmptyDomain { at: domain_start_pos });
    }
    let domain_len = domain_end - domain_start_idx;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong { at: positions[domain_start_idx + 253] });
    }

    let mut label_starts = Vec::new();
    let mut label_ends = Vec::new();
    let mut s = domain_start_idx;
    for i in domain_start_idx..=domain_end {
        if i == domain_end || bytes[i] == b'.' {
            if i == s {
                let at = if i == domain_end {
                    root_pos.unwrap_or(domain_start_pos)
                } else {
                    positions[i]
                };
                return Err(EmailError::EmptyLabel { at });
            }
            label_starts.push(s);
            label_ends.push(i);
            s = i + 1;
        }
    }

    for (&s, &e) in label_starts.iter().zip(label_ends.iter()) {
        let len = e - s;
        if len > 63 { return Err(EmailError::LabelTooLong { at: positions[s + 63] }); }
        if bytes[s] == b'-' { return Err(EmailError::LabelHyphen { at: positions[s] }); }
        if bytes[e - 1] == b'-' { return Err(EmailError::LabelHyphen { at: positions[e - 1] }); }
        for i in s..e {
            if !(bytes[i].is_ascii_alphanumeric() || bytes[i] == b'-') {
                return Err(EmailError::DomainChar { at: positions[i] });
            }
        }
    }

    if label_starts.len() < 2 && !allow_single {
        return Err(EmailError::DomainNeedsDot { at: domain_start_pos });
    }
    let last_s = *label_starts.last().unwrap();
    let last_e = *label_ends.last().unwrap();
    if bytes[last_s..last_e].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: positions[last_s] });
    }

    let mut domain: Vec<u8> = bytes[domain_start_idx..domain_end].to_vec();
    domain.make_ascii_lowercase();
    Ok((Address {
        local: String::from_utf8(local).unwrap(),
        domain: String::from_utf8(domain).unwrap(),
        quoted,
    }, origins))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_core(input, false).map(|(a, _)| a)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_core(input, policy.allow_single_label)?;
    let mut bytes = address.local.into_bytes();
    if policy.local_case == LocalCase::Lower {
        bytes.make_ascii_lowercase();
    }
    let mut origin_positions = origins;

    if policy.strip_subaddress {
        if let Some(i) = bytes.iter().position(|&b| b == b'+') {
            if i == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origin_positions.get(i).copied().unwrap_or(input.len())
                });
            }
            bytes.truncate(i);
            origin_positions.truncate(i);
        }
    }

    let safe = local_atom_valid(&bytes);
    address.local = String::from_utf8(bytes).unwrap();
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' { out.push('\\'); }
            out.push(c);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(e) => result.errors.push((i, e)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first) = seen.get(&key) {
                    result.duplicate_of[i] = Some(first);
                } else {
                    seen.insert(key, i);
                    result.source_indices.push(i);
                    result.addresses.push(address);
                }
            }
        }
    }
    result
}

fn clean(b: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<R>), EmailError> {
    let mut c = Vec::new();
    let mut m = Vec::new();
    let mut rs = Vec::new();
    let mut i = 0;
    let mut q = false;
    while i < b.len() {
        let x = b[i];
        if q {
            c.push(x); m.push(i);
            if x == b'\\' {
                if i + 1 < b.len() { i += 1; c.push(b[i]); m.push(i); }
            } else if x == b'"' { q = false; }
            i += 1; continue;
        }
        if x == b'"' && c.is_empty() { q = true; c.push(x); m.push(i); i += 1; continue; }
        if x == b'\r' || x == b'\n' {
            if let Some(e) = fold(b, i) { rs.push(R{s:i,e}); i=e; continue; }
            return Err(EmailError::BadFold{at:i});
        }
        if h(x) {
            let s=i; while i<b.len() && h(b[i]) { i+=1; }
            rs.push(R{s,e:i}); continue;
        }
        if x == b'(' {
            let s=i; let mut d=1usize; i+=1;
            while i<b.len() {
                let z=b[i];
                if z==b'\\' {
                    if i+1>=b.len() || !p(b[i+1]) { return Err(EmailError::CommentEscape{at:i}); }
                    i+=2; continue;
                }
                if z==b'\r' || z==b'\n' {
                    if let Some(e)=fold(b,i) { i=e; continue; }
                    return Err(EmailError::BadFold{at:i});
                }
                if z==b'(' { d+=1; if d>3 { return Err(EmailError::CommentDepth{at:i}); } }
                else if z==b')' { d-=1; i+=1; if d==0 { break; } continue; }
                else if !p(z) && !h(z) { return Err(EmailError::CommentChar{at:i}); }
                i+=1;
            }
            if d!=0 { return Err(EmailError::UnclosedComment{at:s}); }
            rs.push(R{s,e:i}); continue;
        }
        if x==b')' { return Err(EmailError::UnexpectedClose{at:i}); }
        c.push(x); m.push(i); i+=1;
    }
    Ok((c,m,rs))
}

fn dot_ok(x: &[u8]) -> bool {
    if x.is_empty() { return false; }
    for i in 0..x.len() {
        let b=x[i];
        if b==b'.' {
            if i==0 || i+1==x.len() || x[i-1]==b'.' { return false; }
        } else if !alpha_num(b) && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) { return false; }
    }
    true
}

fn core(c:&[u8],m:&[usize], single:bool)->Result<(Address,Vec<usize>),EmailError>{
    if c.is_empty(){return Err(EmailError::EmptyInput);}
    let mut qe=None;
    if c[0]==b'"'{
        let mut i=1;
        while i<c.len(){
            if c[i]==b'\\'{i+=2;continue;}
            if c[i]==b'"'{qe=Some(i);break;}
            i+=1;
        }
        if qe.is_none(){return Err(EmailError::UnclosedQuote{at:m[0]});}
    }
    let mut ats=Vec::new(); let mut i=0;
    while i<c.len(){
        if let Some(q)=qe { if i<=q {i=q+1;continue;} }
        if c[i]==b'@'{ats.push(i);}
        i+=1;
    }
    if ats.is_empty(){return Err(EmailError::MissingAt);}
    if ats.len()>1{return Err(EmailError::MultipleAt{at:m[ats[1]]});}
    let a=ats[0]; if a==0{return Err(EmailError::EmptyLocal{at:m[a]});}
    let raw=&c[..a]; let mut loc=Vec::new(); let mut org=Vec::new(); let quoted;
    if raw[0]==b'"'{
        let q=qe.unwrap(); if q+1!=a{return Err(EmailError::TrailingLocal{at:m[q+1]});}
        quoted=true; let mut j=1;
        while j<q{
            if raw[j]==b'\\'{
                if j+1>=q || !matches!(raw[j+1],b'"'|b'\\'|b'@'){return Err(EmailError::BadEscape{at:m[j]});}
                loc.push(raw[j+1]);org.push(m[j]);j+=2;
            }else{
                if !p(raw[j]){return Err(EmailError::QuotedChar{at:m[j]});}
                loc.push(raw[j]);org.push(m[j]);j+=1;
            }
        }
        if loc.is_empty(){return Err(EmailError::EmptyLocal{at:m[q]});}
        if loc.len()>64{return Err(EmailError::LocalTooLong{at:org[64]});}
    }else{
        quoted=false;
        if raw.len()>64{return Err(EmailError::LocalTooLong{at:m[64]});}
        for (j,&x) in raw.iter().enumerate(){
            if x==b'.'{
                if j==0||j+1==raw.len()||raw[j-1]==b'.'{return Err(EmailError::LocalDot{at:m[j]});}
            }else if !alpha_num(x)&&!b"!#$%&'*+-/=?^_`{|}~".contains(&x){return Err(EmailError::LocalChar{at:m[j]});}
            loc.push(x);org.push(m[j]);
        }
    }
    let ds=a+1; let dpos=m.get(ds).copied().unwrap_or_else(||m.last().map(|x|x+1).unwrap_or(0));
    if ds==c.len(){return Err(EmailError::EmptyDomain{at:dpos});}
    let mut de=c.len();let mut root=None;if c[de-1]==b'.'{root=Some(de-1);de-=1;}
    if de==ds{return Err(EmailError::EmptyDomain{at:dpos});}
    if de-ds>253{return Err(EmailError::DomainTooLong{at:m[ds+253]});}
    let mut ls=Vec::new();let mut s=ds;
    for j in ds..de{if c[j]==b'.'{ls.push((s,j));s=j+1;}}
    ls.push((s,de));
    for &(s,e) in &ls{
        if s==e{
            let at=if s==de{root.map(|r|m[r]).unwrap_or(dpos)}else if s>ds{m[s-1]}else{m[s]};
            return Err(EmailError::EmptyLabel{at});
        }
        if e-s>63{return Err(EmailError::LabelTooLong{at:m[s+63]});}
        if c[s]==b'-'{return Err(EmailError::LabelHyphen{at:m[s]});}
        if c[e-1]==b'-'{return Err(EmailError::LabelHyphen{at:m[e-1]});}
        for j in s..e{if !alpha_num(c[j])&&c[j]!=b'-'{return Err(EmailError::DomainChar{at:m[j]});}}
    }
    if ls.len()<2&&!single{return Err(EmailError::DomainNeedsDot{at:dpos});}
    let (s,e)=*ls.last().unwrap();
    if c[s..e].iter().all(u8::is_ascii_digit){return Err(EmailError::NumericTld{at:m[s]});}
    let domain:String=c[ds..de].iter().map(|&x|x.to_ascii_lowercase() as char).collect();
    Ok((Address{local:String::from_utf8(loc).unwrap(),domain,quoted},org))
}

fn boundary(r:&R,c:&[u8],m:&[usize])->bool{
    let left=m.iter().rposition(|&x|x<r.s).map(|i|c[i]==b'@').unwrap_or(true);
    let right=m.iter().position(|&x|x>=r.e).map(|i|c[i]==b'@').unwrap_or(true);
    left||right
}

fn run(input:&str,p:&Policy,strict:bool)->Result<(Address,Vec<usize>),EmailError>{
    if input.is_empty(){return Err(EmailError::EmptyInput);}
    let b=input.as_bytes();
    if let Some(i)=b.iter().position(|&x|x>=128){return Err(EmailError::NonAscii{at:i});}
    let(c,m,rs)=clean(b)?;
    for r in &rs{if !boundary(r,&c,&m){return Err(EmailError::CfwsBoundary{at:r.s});}}
    let(mut a,org)=core(&c,&m,if strict{false}else{p.allow_single_label})?;
    if !strict&&p.local_case==LocalCase::Lower{a.local.make_ascii_lowercase();}
    if !strict&&p.strip_subaddress{
        if let Some(i)=a.local.bytes().position(|x|x==b'+'){
            if i==0{return Err(EmailError::EmptyNormalizedLocal{at:org.first().copied().unwrap_or(0)});}
            a.local.truncate(i);
        }
    }
    if !strict{
        if !dot_ok(a.local.as_bytes()){a.quoted=true;}
        else if p.unquote_safe{a.quoted=false;}
    }
    Ok((a,org))
}
