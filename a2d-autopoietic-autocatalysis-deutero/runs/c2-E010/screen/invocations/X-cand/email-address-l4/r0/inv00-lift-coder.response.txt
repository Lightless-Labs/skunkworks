- Cleanup scans the entire ASCII input before address syntax, so malformed comments or folds can take precedence over local-part errors.
- A CFWS run is allowed only at an edge or directly beside a retained `@`; I treat a run adjacent to either side of `@` as legal.
- A leading quoted local protects `@` only through its first unescaped closing quote; an unclosed quote is reported before separator errors.
- A final domain dot is removed once, but its original offset remains available for empty-label errors.
- Quoted-local decoding errors precede decoded emptiness and length checks; escapes count as originating at their backslash.
- For normalization, I validate the full address first, then lowercase, strip at the first `+`, and set `quoted` from the resulting dot-atom safety rules.

### Solution A

fn is_hws(b: u8) -> bool { b == b' ' || b == b'\t' }

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while bytes.get(j).copied().map(is_hws).unwrap_or(false) { j += 1; }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<(Vec<Byte>, Vec<Run>), EmailError> {
    let mut kept: Vec<Byte> = Vec::with_capacity(input.len());
    let mut runs: Vec<Run> = Vec::new();
    let mut i = 0usize;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut quote_protected = false;

    while i < input.len() {
        let b = input[i];

        if quote_protected {
            kept.push(Byte { value: b, pos: i });
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    kept.push(Byte { value: input[i], pos: i });
                    i += 1;
                }
            } else {
                if b == b'"' { quote_protected = false; }
                i += 1;
            }
            continue;
        }

        if depth > 0 {
            if is_hws(b) || b == b'\r' || b == b'\n' {
                let start = i;
                while i < input.len() {
                    if is_hws(input[i]) { i += 1; continue; }
                    if input[i] == b'\r' || input[i] == b'\n' {
                        if let Some(end) = fold_end(input, i) { i = end; continue; }
                        return Err(EmailError::BadFold { at: i });
                    }
                    break;
                }
                runs.push(Run { start, end: i });
                continue;
            }
            if b == b'\\' {
                if i + 1 >= input.len()
                    || !(0x20..=0x7e).contains(&input[i + 1])
                {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'(' {
                if depth == 3 { return Err(EmailError::CommentDepth { at: i }); }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                continue;
            }
            if !(0x20..=0x7e).contains(&b) {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        if is_hws(b) || b == b'\r' || b == b'\n' {
            let start = i;
            while i < input.len() {
                if is_hws(input[i]) { i += 1; continue; }
                if input[i] == b'\r' || input[i] == b'\n' {
                    if let Some(end) = fold_end(input, i) { i = end; continue; }
                    return Err(EmailError::BadFold { at: i });
                }
                break;
            }
            runs.push(Run { start, end: i });
            continue;
        }

        if b == b'(' {
            outer_open = i;
            depth = 1;
            i += 1;
            continue;
        }
        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if b == b'"' && kept.is_empty() {
            quote_protected = true;
        }
        kept.push(Byte { value: b, pos: i });
        i += 1;
    }

    if depth > 0 { return Err(EmailError::UnclosedComment { at: outer_open }); }

    for run in &runs {
        let left_is_at = kept.last().map(|_| false).unwrap_or(false);
        let _ = left_is_at;
        let before = kept.iter().rev().find(|b| b.pos < run.start);
        let after = kept.iter().find(|b| b.pos >= run.end);
        let at_edge = before.is_none() || after.is_none();
        let adjacent_at = before.map(|b| b.value == b'@').unwrap_or(false)
            || after.map(|b| b.value == b'@').unwrap_or(false);
        if !at_edge && !adjacent_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    Ok((kept, runs))
}

fn local_atom_valid(bytes: &[u8]) -> bool {
    if bytes.is_empty() || bytes.len() > 64 { return false; }
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' {
                return false;
            }
        } else if !(b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)) {
            return false;
        }
    }
    true
}

fn parse_core(input: &str, allow_single: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() { return Err(EmailError::EmptyInput); }
    if let Some(i) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let (clean, _) = cleanup(raw)?;
    if clean.is_empty() { return Err(EmailError::EmptyInput); }
    let bytes: Vec<u8> = clean.iter().map(|b| b.value).collect();
    let positions: Vec<usize> = clean.iter().map(|b| b.pos).collect();

    let mut quoted_end: Option<usize> = None;
    if bytes.first() == Some(&b'"') {
        let mut i = 1usize;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quoted_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quoted_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: positions[0] });
        }
    }

    let mut ats = Vec::new();
    let mut i = 0usize;
    while i < bytes.len() {
        if let Some(end) = quoted_end {
            if i <= end { i += 1; continue; }
        }
        if bytes[i] == b'@' { ats.push(i); }
        i += 1;
    }
    if ats.is_empty() { return Err(EmailError::MissingAt); }
    if ats.len() > 1 { return Err(EmailError::MultipleAt { at: positions[ats[1]] }); }
    let sep = ats[0];
    if sep == 0 { return Err(EmailError::EmptyLocal { at: positions[sep] }); }

    let local_bytes = &bytes[..sep];
    let mut local = Vec::new();
    let mut origins = Vec::new();
    let quoted = local_bytes.first() == Some(&b'"');

    if !quoted {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: positions[64] });
        }
        for (i, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len()
                    || (i > 0 && local_bytes[i - 1] == b'.')
                {
                    return Err(EmailError::LocalDot { at: positions[i] });
                }
            } else if !(b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)) {
                return Err(EmailError::LocalChar { at: positions[i] });
            }
            local.push(b);
            origins.push(positions[i]);
        }
    } else {
        let end = quoted_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal { at: positions[end + 1] });
        }
        let mut i = 1usize;
        while i < end {
            let b = local_bytes[i];
            if b == b'\\' {
                if i + 1 >= end {
                    return Err(EmailError::BadEscape { at: positions[i] });
                }
                let next = local_bytes[i + 1];
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape { at: positions[i] });
                }
                local.push(next);
                origins.push(positions[i]);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: positions[i] });
                }
                local.push(b);
                origins.push(positions[i]);
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: positions[end] });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    }

    let domain_start_idx = sep + 1;
    let domain_start_pos = positions.get(domain_start_idx).copied().unwrap_or(input.len());
    let mut domain_end = bytes.len();
    let mut root_pos = None;
    if domain_end > domain_start_idx && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_pos = Some(positions[domain_end]);
    }
    if domain_end == domain_start_idx {
        return Err(EmailError::EmptyDomain { at: domain_start_pos });
    }
    let domain_len = domain_end - domain_start_idx;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong { at: positions[domain_start_idx + 253] });
    }

    let mut label_starts = Vec::new();
    let mut label_ends = Vec::new();
    let mut s = domain_start_idx;
    for i in domain_start_idx..=domain_end {
        if i == domain_end || bytes[i] == b'.' {
            if i == s {
                let at = if i == domain_end {
                    root_pos.unwrap_or(domain_start_pos)
                } else {
                    positions[i]
                };
                return Err(EmailError::EmptyLabel { at });
            }
            label_starts.push(s);
            label_ends.push(i);
            s = i + 1;
        }
    }

    for (&s, &e) in label_starts.iter().zip(label_ends.iter()) {
        let len = e - s;
        if len > 63 { return Err(EmailError::LabelTooLong { at: positions[s + 63] }); }
        if bytes[s] == b'-' { return Err(EmailError::LabelHyphen { at: positions[s] }); }
        if bytes[e - 1] == b'-' { return Err(EmailError::LabelHyphen { at: positions[e - 1] }); }
        for i in s..e {
            if !(bytes[i].is_ascii_alphanumeric() || bytes[i] == b'-') {
                return Err(EmailError::DomainChar { at: positions[i] });
            }
        }
    }

    if label_starts.len() < 2 && !allow_single {
        return Err(EmailError::DomainNeedsDot { at: domain_start_pos });
    }
    let last_s = *label_starts.last().unwrap();
    let last_e = *label_ends.last().unwrap();
    if bytes[last_s..last_e].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: positions[last_s] });
    }

    let mut domain: Vec<u8> = bytes[domain_start_idx..domain_end].to_vec();
    domain.make_ascii_lowercase();
    Ok((Address {
        local: String::from_utf8(local).unwrap(),
        domain: String::from_utf8(domain).unwrap(),
        quoted,
    }, origins))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_core(input, false).map(|(a, _)| a)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_core(input, policy.allow_single_label)?;
    let mut bytes = address.local.into_bytes();
    if policy.local_case == LocalCase::Lower {
        bytes.make_ascii_lowercase();
    }
    let mut origin_positions = origins;

    if policy.strip_subaddress {
        if let Some(i) = bytes.iter().position(|&b| b == b'+') {
            if i == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origin_positions.get(i).copied().unwrap_or(input.len())
                });
            }
            bytes.truncate(i);
            origin_positions.truncate(i);
        }
    }

    let safe = local_atom_valid(&bytes);
    address.local = String::from_utf8(bytes).unwrap();
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' { out.push('\\'); }
            out.push(c);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(e) => result.errors.push((i, e)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first) = seen.get(&key) {
                    result.duplicate_of[i] = Some(first);
                } else {
                    seen.insert(key, i);
                    result.source_indices.push(i);
                    result.addresses.push(address);
                }
            }
        }
    }
    result
}

    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,

#[derive(Clone, Copy)]
struct Byte {
    value: u8,
    pos: usize,

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,

fn horizontal(b: u8) -> bool { b == b' ' || b == b'\t' }

fn fold(bytes: &[u8], at: usize) -> Option<usize> {
    if bytes.get(at) != Some(&b'\r') || bytes.get(at + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = at + 2;
    if !bytes.get(j).copied().map(horizontal).unwrap_or(false) { return None; }
    while j < bytes.len() && horizontal(bytes[j]) { j += 1; }
    Some(j)
}

fn lexical_cleanup(bytes: &[u8]) -> Result<(Vec<Byte>, Vec<Run>), EmailError> {
    let mut output = Vec::with_capacity(bytes.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut comment_depth = 0usize;
    let mut outer_comment = 0usize;
    let mut protected_quote = false;

    while i < bytes.len() {
        let b = bytes[i];

        if protected_quote {
            output.push(Byte { value: b, pos: i });
            if b == b'\\' {
                i += 1;
                if i < bytes.len() {
                    output.push(Byte { value: bytes[i], pos: i });
                    i += 1;
                }
            } else {
                if b == b'"' { protected_quote = false; }
                i += 1;
            }
            continue;
        }

        if comment_depth != 0 {
            if horizontal(b) || b == b'\r' || b == b'\n' {
                let start = i;
                loop {
                    if i < bytes.len() && horizontal(bytes[i]) { i += 1; continue; }
                    if i < bytes.len() && (bytes[i] == b'\r' || bytes[i] == b'\n') {
                        if let Some(end) = fold(bytes, i) { i = end; continue; }
                        return Err(EmailError::BadFold { at: i });
                    }
                    break;
                }
                runs.push(Run { start, end: i });
            } else if b == b'\\' {
                if i + 1 == bytes.len() || !(0x20..=0x7e).contains(&bytes[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
            } else if b == b'(' {
                if comment_depth == 3 { return Err(EmailError::CommentDepth { at: i }); }
                comment_depth += 1;
                i += 1;
            } else if b == b')' {
                comment_depth -= 1;
                i += 1;
            } else if (0x20..=0x7e).contains(&b) {
                i += 1;
            } else {
                return Err(EmailError::CommentChar { at: i });
            }
            continue;
        }

        if horizontal(b) || b == b'\r' || b == b'\n' {
            let start = i;
            loop {
                if i < bytes.len() && horizontal(bytes[i]) { i += 1; continue; }
                if i < bytes.len() && (bytes[i] == b'\r' || bytes[i] == b'\n') {
                    if let Some(end) = fold(bytes, i) { i = end; continue; }
                    return Err(EmailError::BadFold { at: i });
                }
                break;
            }
            runs.push(Run { start, end: i });
            continue;
        }

        if b == b'(' {
            outer_comment = i;
            comment_depth = 1;
            i += 1;
            continue;
        }
        if b == b')' { return Err(EmailError::UnexpectedClose { at: i }); }

        if b == b'"' && output.is_empty() { protected_quote = true; }
        output.push(Byte { value: b, pos: i });
        i += 1;
    }

    if comment_depth != 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }

    for run in &runs {
        let before = output.iter().take_while(|b| b.pos < run.start).last();
        let after = output.iter().find(|b| b.pos >= run.end);
        if before.is_some() && after.is_some()
            && before.map(|b| b.value != b'@').unwrap_or(true)
            && after.map(|b| b.value != b'@').unwrap_or(true)
        {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    Ok((output, runs))
}

fn valid_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() || bytes.len() > 64 { return false; }
    for i in 0..bytes.len() {
        let b = bytes[i];
        if b == b'.' {
            if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' { return false; }
        } else if !b.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
            return false;
        }
    }
    true
}

fn parse_impl(input: &str, allow_single: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() { return Err(EmailError::EmptyInput); }
    if let Some(at) = raw.iter().position(|b| *b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (mapped, _) = lexical_cleanup(raw)?;
    if mapped.is_empty() { return Err(EmailError::EmptyInput); }
    let bytes: Vec<u8> = mapped.iter().map(|b| b.value).collect();
    let pos: Vec<usize> = mapped.iter().map(|b| b.pos).collect();

    let quote_end = if bytes.first() == Some(&b'"') {
        let mut j = 1;
        let mut found = None;
        while j < bytes.len() {
            if bytes[j] == b'\\' { j += 2; }
            else if bytes[j] == b'"' { found = Some(j); break; }
            else { j += 1; }
        }
        match found {
            Some(x) => Some(x),
            None => return Err(EmailError::UnclosedQuote { at: pos[0] }),
        }
    } else { None };

    let mut separators = Vec::new();
    for i in 0..bytes.len() {
        if quote_end.map(|q| i <= q).unwrap_or(false) { continue; }
        if bytes[i] == b'@' { separators.push(i); }
    }
    if separators.is_empty() { return Err(EmailError::MissingAt); }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt { at: pos[separators[1]] });
    }
    let sep = separators[0];
    if sep == 0 { return Err(EmailError::EmptyLocal { at: pos[sep] }); }

    let local_part = &bytes[..sep];
    let quoted = local_part.first() == Some(&b'"');
    let mut local = Vec::new();
    let mut origins = Vec::new();

    if !quoted {
        if local_part.len() > 64 {
            return Err(EmailError::LocalTooLong { at: pos[64] });
        }
        for i in 0..local_part.len() {
            let b = local_part[i];
            if b == b'.' {
                if i == 0 || i + 1 == local_part.len() || (i > 0 && local_part[i - 1] == b'.') {
                    return Err(EmailError::LocalDot { at: pos[i] });
                }
            } else if !b.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
                return Err(EmailError::LocalChar { at: pos[i] });
            }
            local.push(b);
            origins.push(pos[i]);
        }
    } else {
        let q = quote_end.unwrap();
        if q + 1 != sep { return Err(EmailError::TrailingLocal { at: pos[q + 1] }); }
        let mut i = 1;
        while i < q {
            let b = local_part[i];
            if b == b'\\' {
                if i + 1 >= q { return Err(EmailError::BadEscape { at: pos[i] }); }
                let escaped = local_part[i + 1];
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape { at: pos[i] });
                }
                local.push(escaped);
                origins.push(pos[i]);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) { return Err(EmailError::QuotedChar { at: pos[i] }); }
                local.push(b);
                origins.push(pos[i]);
                i += 1;
            }
        }
        if local.is_empty() { return Err(EmailError::EmptyLocal { at: pos[q] }); }
        if local.len() > 64 { return Err(EmailError::LocalTooLong { at: origins[64] }); }
    }

    let ds = sep + 1;
    let domain_start = pos.get(ds).copied().unwrap_or(input.len());
    let mut de = bytes.len();
    let mut root_dot = None;
    if de > ds && bytes[de - 1] == b'.' {
        de -= 1;
        root_dot = Some(pos[de]);
    }
    if de == ds { return Err(EmailError::EmptyDomain { at: domain_start }); }
    if de - ds > 253 { return Err(EmailError::DomainTooLong { at: pos[ds + 253] }); }

    let mut starts = Vec::new();
    let mut ends = Vec::new();
    let mut start = ds;
    for i in ds..=de {
        if i == de || bytes[i] == b'.' {
            if i == start {
                let at = if i == de { root_dot.unwrap_or(domain_start) } else { pos[i] };
                return Err(EmailError::EmptyLabel { at });
            }
            starts.push(start);
            ends.push(i);
            start = i + 1;
        }
    }

    for (&s, &e) in starts.iter().zip(&ends) {
        if e - s > 63 { return Err(EmailError::LabelTooLong { at: pos[s + 63] }); }
        if bytes[s] == b'-' { return Err(EmailError::LabelHyphen { at: pos[s] }); }
        if bytes[e - 1] == b'-' { return Err(EmailError::LabelHyphen { at: pos[e - 1] }); }
        for i in s..e {
            if !bytes[i].is_ascii_alphanumeric() && bytes[i] != b'-' {
                return Err(EmailError::DomainChar { at: pos[i] });
            }
        }
    }

    if starts.len() < 2 && !allow_single {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    let last_start = *starts.last().unwrap();
    let last_end = *ends.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos[last_start] });
    }

    let mut domain = bytes[ds..de].to_vec();
    domain.make_ascii_lowercase();
    Ok((Address {
        local: String::from_utf8(local).unwrap(),
        domain: String::from_utf8(domain).unwrap(),
        quoted,
    }, origins))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_impl(input, false).map(|x| x.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_impl(input, policy.allow_single_label)?;
    let mut local = address.local.into_bytes();
    if policy.local_case == LocalCase::Lower { local.make_ascii_lowercase(); }
    let mut origins = origins;

    if policy.strip_subaddress {
        if let Some(plus) = local.iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.get(plus).copied().unwrap_or(input.len())
                });
            }
            local.truncate(plus);
            origins.truncate(plus);
        }
    }

    let safe = valid_atom(&local);
    address.local = String::from_utf8(local).unwrap();
    if safe {
        if policy.unquote_safe { address.quoted = false; }
    } else {
        address.quoted = true;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' { result.push('\\'); }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut output = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut first_by_key: HashMap<(String, String), usize> = HashMap::new();

    for (index, value) in inputs.iter().enumerate() {
        match normalize_email(value, policy) {
            Err(error) => output.errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                match first_by_key.get(&key) {
                    Some(first_index) => output.duplicate_of[index] = Some(*first_index),
                    None => {
                        first_by_key.insert(key, index);
                        output.source_indices.push(index);
                        output.addresses.push(address);
                    }
                }
            }
        }
    }
    output
}
