- Error precedence follows the explicitly ordered stages; cleanup lexical errors are collected before boundary errors.
- A CFWS run is legal at sequence edges or when directly adjacent to a retained `@`; its reported position is its first original byte.
- A terminal domain dot is removed once, but its original offset is retained for an empty final label.
- A quoted local’s escaped bytes use the backslash offset for later length and normalization errors.
- `parse_email` stays strict about single-label domains; only normalization policy may allow them.
- For Solution B, I take the less likely reading that CFWS runs adjacent to either side of `@` are independently permitted, even when a run is also between local bytes or domain labels.

fn is_hws(b: u8) -> bool { b == b' ' || b == b'\t' }
fn is_printable(b: u8) -> bool { (0x20..=0x7e).contains(&b) }

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if i + 2 >= bytes.len() || bytes[i] != b'\r' || bytes[i + 1] != b'\n' || !is_hws(bytes[i + 2]) {
        return None;
    }
    let mut j = i + 2;
    while j < bytes.len() && is_hws(bytes[j]) { j += 1; }
    Some(j)

fn cleanup(input: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let mut clean = Vec::with_capacity(input.len());
    let mut map = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut protected = false;
    let mut quote_closed = true;

    while i < input.len() {
        let b = input[i];
        if protected {
            clean.push(b); map.push(i);
            if b == b'\\' {
                if i + 1 < input.len() {
                    i += 1;
                    clean.push(input[i]); map.push(i);
                }
            } else if b == b'"' {
                protected = false;
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        if b == b'"' && clean.is_empty() {
            protected = true;
            quote_closed = false;
            clean.push(b); map.push(i); i += 1;
            continue;
        }

        if b == b'\r' || b == b'\n' {
            if let Some(end) = fold_end(input, i) {
                let start = i;
                i = end;
                runs.push(Run { start, end });
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }

        if is_hws(b) {
            let start = i;
            while i < input.len() && is_hws(input[i]) { i += 1; }
            runs.push(Run { start, end: i });
            continue;
        }

        if b == b'(' {
            let outer = i;
            let mut depth = 1usize;
            i += 1;
            while i < input.len() {
                let c = input[i];
                if c == b'\\' {
                    if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                        return Err(EmailError::CommentEscape { at: i });
                    }
                    i += 2;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if c == b'(' {
                    depth += 1;
                    if depth > 3 { return Err(EmailError::CommentDepth { at: i }); }
                } else if c == b')' {
                    depth -= 1;
                    i += 1;
                    if depth == 0 { break; }
                    continue;
                } else if !is_printable(c) && !is_hws(c) {
                    return Err(EmailError::CommentChar { at: i });
                }
                i += 1;
            }
            if depth != 0 { return Err(EmailError::UnclosedComment { at: outer }); }
            runs.push(Run { start: outer, end: i });
            continue;
        }

        if b == b')' { return Err(EmailError::UnexpectedClose { at: i }); }
        clean.push(b); map.push(i);
        i += 1;
    }

    let _ = quote_closed;
    Ok((clean, map, runs))
}

fn local_dot_valid(s: &[u8]) -> bool {
    if s.is_empty() { return false; }
    for (i, &b) in s.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == s.len() || s[i - 1] == b'.' { return false; }
        } else if !is_alnum(b) && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
            return false;
        }
    }
    true
}

fn is_alnum(b: u8) -> bool {

fn parse_core(clean: &[u8], map: &[usize], allow_single: bool) -> Result<(Address, Vec<usize>), EmailError> {
    if clean.is_empty() { return Err(EmailError::EmptyInput); }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut i = 1;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
                continue;
            }
            if clean[i] == b'"' { quote_end = Some(i); break; }
            i += 1;
        }
        if quote_end.is_none() { return Err(EmailError::UnclosedQuote { at: map[0] }); }
    }

    let mut ats = Vec::new();
    let mut i = 0;
    while i < clean.len() {
        if let Some(qe) = quote_end {
            if i <= qe { i = qe + 1; continue; }
        }
        if clean[i] == b'@' { ats.push(i); }
        i += 1;
    }
    if ats.is_empty() { return Err(EmailError::MissingAt); }
    if ats.len() > 1 { return Err(EmailError::MultipleAt { at: map[ats[1]] }); }
    let at_i = ats[0];
    if at_i == 0 { return Err(EmailError::EmptyLocal { at: map[at_i] }); }

    let local_raw = &clean[..at_i];
    let mut local = Vec::new();
    let mut origins = Vec::new();
    let quoted;

    if local_raw[0] == b'"' {
        let qe = quote_end.unwrap();
        if qe + 1 != at_i { return Err(EmailError::TrailingLocal { at: map[qe + 1] }); }
        quoted = true;
        let mut j = 1;
        while j < qe {
            if local_raw[j] == b'\\' {
                if j + 1 >= qe || !matches!(local_raw[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: map[j] });
                }
                local.push(local_raw[j + 1]);
                origins.push(map[j]);
                j += 2;
            } else {
                if !is_printable(local_raw[j]) {
                    return Err(EmailError::QuotedChar { at: map[j] });
                }
                local.push(local_raw[j]);
                origins.push(map[j]);
                j += 1;
            }
        }
        if local.is_empty() { return Err(EmailError::EmptyLocal { at: map[qe] }); }
        if local.len() > 64 { return Err(EmailError::LocalTooLong { at: origins[64] }); }
    } else {
        quoted = false;
        if local_raw.len() > 64 { return Err(EmailError::LocalTooLong { at: map[64] }); }
        for (j, &b) in local_raw.iter().enumerate() {
            if b == b'.' {
                if j == 0 || j + 1 == local_raw.len() || local_raw[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: map[j] });
                }
            } else if !is_alnum(b) && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
                return Err(EmailError::LocalChar { at: map[j] });
            }
            local.push(b);
            origins.push(map[j]);
        }
    }

    let domain_start_i = at_i + 1;
    let domain_start = map.get(domain_start_i).copied().unwrap_or_else(|| map.last().map(|x| x + 1).unwrap_or(0));
    if domain_start_i == clean.len() { return Err(EmailError::EmptyDomain { at: domain_start }); }
    let mut dom_end = clean.len();
    let mut removed_dot = None;
    if clean[dom_end - 1] == b'.' {
        removed_dot = Some(dom_end - 1);
        dom_end -= 1;
    }
    if dom_end == domain_start_i {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    let dlen = dom_end - domain_start_i;
    if dlen > 253 { return Err(EmailError::DomainTooLong { at: map[domain_start_i + 253] }); }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start_i;
    for j in domain_start_i..dom_end {
        if clean[j] == b'.' {
            labels.push((start, j));
            start = j + 1;
        }
    }
    labels.push((start, dom_end));

    let mut label_count = 0;
    for &(s, e) in &labels {
        if s == e {
            let pos = if s == dom_end {
                removed_dot.map(|x| map[x]).unwrap_or(domain_start)
            } else if s > domain_start_i {
                map[s - 1]
            } else {
                map[s]
            };
            return Err(EmailError::EmptyLabel { at: pos });
        }
        let len = e - s;
        if len > 63 { return Err(EmailError::LabelTooLong { at: map[s + 63] }); }
        if clean[s] == b'-' { return Err(EmailError::LabelHyphen { at: map[s] }); }
        if clean[e - 1] == b'-' { return Err(EmailError::LabelHyphen { at: map[e - 1] }); }
        for j in s..e {
            let b = clean[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: map[j] });
            }
        }
        label_count += 1;
    }
    if label_count < 2 && !allow_single { return Err(EmailError::DomainNeedsDot { at: domain_start }); }

    let (last_s, last_e) = *labels.last().unwrap();
    if clean[last_s..last_e].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: map[last_s] });
    }

    let domain = clean[domain_start_i..dom_end].iter().map(|b| b.to_ascii_lowercase() as char).collect();
    let address = Address { local: String::from_utf8(local).unwrap(), domain, quoted };
    Ok((address, origins))
}

fn has_cfws_between(r: Run, map: &[usize], clean: &[u8]) -> bool {
    let first = map.iter().position(|&p| p >= r.end);
    let before = map.iter().rposition(|&p| p < r.start);
    let left = before.map(|i| clean[i] == b'@').unwrap_or(true);
    let right = first.map(|i| clean[i] == b'@').unwrap_or(true);
    left || right
}

fn parse_with_policy(input: &str, policy: &Policy, strict_single: bool) -> Result<(Address, Vec<usize>), EmailError> {
    if input.is_empty() { return Err(EmailError::EmptyInput); }
    let bytes = input.as_bytes();
    if let Some(i) = bytes.iter().position(|&b| b >= 128) { return Err(EmailError::NonAscii { at: i }); }
    let (clean, map, runs) = cleanup(bytes)?;
    for r in runs {
        if !has_cfws_between(r, &map, &clean) { return Err(EmailError::CfwsBoundary { at: r.start }); }
    }
    let (mut address, origins) = parse_core(&clean, &map, if strict_single { false } else { policy.allow_single_label })?;

    if !strict_single && policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }
    if !strict_single && policy.strip_subaddress {
        if let Some(i) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if i == 0 { return Err(EmailError::EmptyNormalizedLocal { at: origins.get(0).copied().unwrap_or(0) }); }
            address.local.truncate(i);
        }
    }
    if !strict_single {
        let valid = local_dot_valid(address.local.as_bytes());
        if !valid { address.quoted = true; }
        else if policy.unquote_safe { address.quoted = false; }
    }
    Ok((address, origins))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_policy(input, &Policy::default(), true).map(|x| x.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, policy, false).map(|x| x.0)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' { out.push('\\'); }
            out.push(c);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut keys: std::collections::HashMap<(String, String), usize> = std::collections::HashMap::new();

    for (idx, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(e) => errors.push((idx, e)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first) = keys.get(&key) {
                    duplicate_of[idx] = Some(first);
                } else {
                    keys.insert(key, idx);
                    source_indices.push(idx);
                    addresses.push(address);
                }
            }
        }
    }
    DedupResult { addresses, source_indices, duplicate_of, errors }
}

fn h(b: u8) -> bool { b == b' ' || b == b'\t' }
fn p(b: u8) -> bool { (32..=126).contains(&b) }
fn alpha_num(b: u8) -> bool { b.is_ascii_alphanumeric() }
fn fold(b: &[u8], i: usize) -> Option<usize> {
    if i + 2 >= b.len() || b[i] != 13 || b[i + 1] != 10 || !h(b[i + 2]) { return None; }

#[derive(Clone)]
struct R { s: usize, e: usize }

fn clean(b: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<R>), EmailError> {
    let mut c = Vec::new();
    let mut m = Vec::new();
    let mut rs = Vec::new();
    let mut i = 0;
    let mut q = false;
    while i < b.len() {
        let x = b[i];
        if q {
            c.push(x); m.push(i);
            if x == b'\\' {
                if i + 1 < b.len() { i += 1; c.push(b[i]); m.push(i); }
            } else if x == b'"' { q = false; }
            i += 1; continue;
        }
        if x == b'"' && c.is_empty() { q = true; c.push(x); m.push(i); i += 1; continue; }
        if x == b'\r' || x == b'\n' {
            if let Some(e) = fold(b, i) { rs.push(R{s:i,e}); i=e; continue; }
            return Err(EmailError::BadFold{at:i});
        }
        if h(x) {
            let s=i; while i<b.len() && h(b[i]) { i+=1; }
            rs.push(R{s,e:i}); continue;
        }
        if x == b'(' {
            let s=i; let mut d=1usize; i+=1;
            while i<b.len() {
                let z=b[i];
                if z==b'\\' {
                    if i+1>=b.len() || !p(b[i+1]) { return Err(EmailError::CommentEscape{at:i}); }
                    i+=2; continue;
                }
                if z==b'\r' || z==b'\n' {
                    if let Some(e)=fold(b,i) { i=e; continue; }
                    return Err(EmailError::BadFold{at:i});
                }
                if z==b'(' { d+=1; if d>3 { return Err(EmailError::CommentDepth{at:i}); } }
                else if z==b')' { d-=1; i+=1; if d==0 { break; } continue; }
                else if !p(z) && !h(z) { return Err(EmailError::CommentChar{at:i}); }
                i+=1;
            }
            if d!=0 { return Err(EmailError::UnclosedComment{at:s}); }
            rs.push(R{s,e:i}); continue;
        }
        if x==b')' { return Err(EmailError::UnexpectedClose{at:i}); }
        c.push(x); m.push(i); i+=1;
    }
    Ok((c,m,rs))
}

fn dot_ok(x: &[u8]) -> bool {
    if x.is_empty() { return false; }
    for i in 0..x.len() {
        let b=x[i];

        } else if !alpha_num(b) && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) { return false; }
    }
    true
}

fn core(c:&[u8],m:&[usize], single:bool)->Result<(Address,Vec<usize>),EmailError>{
    if c.is_empty(){return Err(EmailError::EmptyInput);}
    let mut qe=None;
    if c[0]==b'"'{
        let mut i=1;
        while i<c.len(){
            if c[i]==b'\\'{i+=2;continue;}
            if c[i]==b'"'{qe=Some(i);break;}
            i+=1;
        }
        if qe.is_none(){return Err(EmailError::UnclosedQuote{at:m[0]});}
    }
    let mut ats=Vec::new(); let mut i=0;
    while i<c.len(){
        if let Some(q)=qe { if i<=q {i=q+1;continue;} }
        if c[i]==b'@'{ats.push(i);}
        i+=1;
    }
    if ats.is_empty(){return Err(EmailError::MissingAt);}
    if ats.len()>1{return Err(EmailError::MultipleAt{at:m[ats[1]]});}
    let a=ats[0]; if a==0{return Err(EmailError::EmptyLocal{at:m[a]});}
    let raw=&c[..a]; let mut loc=Vec::new(); let mut org=Vec::new(); let quoted;
    if raw[0]==b'"'{

        quoted=true; let mut j=1;

                if !p(raw[j]){return Err(EmailError::QuotedChar{at:m[j]});}
                loc.push(raw[j]);org.push(m[j]);j+=1;
            }
        }
        if loc.is_empty(){return Err(EmailError::EmptyLocal{at:m[q]});}
        if loc.len()>64{return Err(EmailError::LocalTooLong{at:org[64]});}
    }else{
        quoted=false;
        if raw.len()>64{return Err(EmailError::LocalTooLong{at:m[64]});}
        for (j,&x) in raw.iter().enumerate(){

            loc.push(x);org.push(m[j]);
        }
    }
    let ds=a+1; let dpos=m.get(ds).copied().unwrap_or_else(||m.last().map(|x|x+1).unwrap_or(0));

    let mut de=c.len();let mut root=None;if c[de-1]==b'.'{root=Some(de-1);de-=1;}
    if de==ds{return Err(EmailError::EmptyDomain{at:dpos});}

    let mut ls=Vec::new();let mut s=ds;
    for j in ds..de{if c[j]==b'.'{ls.push((s,j));s=j+1;}}
    ls.push((s,de));

            return Err(EmailError::EmptyLabel{at});
        }
        if e-s>63{return Err(EmailError::LabelTooLong{at:m[s+63]});}
        if c[s]==b'-'{return Err(EmailError::LabelHyphen{at:m[s]});}
        if c[e-1]==b'-'{return Err(EmailError::LabelHyphen{at:m[e-1]});}
        for j in s..e{if !alpha_num(c[j])&&c[j]!=b'-'{return Err(EmailError::DomainChar{at:m[j]});}}
    }
    if ls.len()<2&&!single{return Err(EmailError::DomainNeedsDot{at:dpos});}
    let (s,e)=*ls.last().unwrap();
    if c[s..e].iter().all(u8::is_ascii_digit){return Err(EmailError::NumericTld{at:m[s]});}
    let domain:String=c[ds..de].iter().map(|&x|x.to_ascii_lowercase() as char).collect();
    Ok((Address{local:String::from_utf8(loc).unwrap(),domain,quoted},org))
}

fn boundary(r:&R,c:&[u8],m:&[usize])->bool{
    let left=m.iter().rposition(|&x|x<r.s).map(|i|c[i]==b'@').unwrap_or(true);
    let right=m.iter().position(|&x|x>=r.e).map(|i|c[i]==b'@').unwrap_or(true);
    left||right
}

fn run(input:&str,p:&Policy,strict:bool)->Result<(Address,Vec<usize>),EmailError>{
    if input.is_empty(){return Err(EmailError::EmptyInput);}
    let b=input.as_bytes();
    if let Some(i)=b.iter().position(|&x|x>=128){return Err(EmailError::NonAscii{at:i});}
    let(c,m,rs)=clean(b)?;
    for r in &rs{if !boundary(r,&c,&m){return Err(EmailError::CfwsBoundary{at:r.s});}}
    let(mut a,org)=core(&c,&m,if strict{false}else{p.allow_single_label})?;
    if !strict&&p.local_case==LocalCase::Lower{a.local.make_ascii_lowercase();}
    if !strict&&p.strip_subaddress{
        if let Some(i)=a.local.bytes().position(|x|x==b'+'){
            if i==0{return Err(EmailError::EmptyNormalizedLocal{at:org.first().copied().unwrap_or(0)});}
            a.local.truncate(i);
        }
    }
    if !strict{
        if !dot_ok(a.local.as_bytes()){a.quoted=true;}
        else if p.unquote_safe{a.quoted=false;}
    }
    Ok((a,org))
}

pub fn parse_email(input:&str)->Result<Address,EmailError>{run(input,&Policy::default(),true).map(|x|x.0)}
pub fn normalize_email(input:&str,policy:&Policy)->Result<Address,EmailError>{run(input,policy,false).map(|x|x.0)}

pub fn format_email(a:&Address)->String{
    let mut s=String::new();
    if a.quoted{
        s.push('"');
        for ch in a.local.chars(){if ch=='"'||ch=='\\'{s.push('\\');}s.push(ch);}
        s.push('"');
    }else{s.push_str(&a.local);}
    s.push('@');s.push_str(&a.domain);s
}

pub fn deduplicate(inputs:&[&str],policy:&Policy)->DedupResult{
    let mut addresses=Vec::new();let mut source_indices=Vec::new();let mut duplicate_of=vec![None;inputs.len()];let mut errors=Vec::new();
    let mut seen:std::collections::HashMap<(String,String),usize>=std::collections::HashMap::new();
    for(i,x)in inputs.iter().enumerate(){
        match normalize_email(x,policy){
            Err(e)=>errors.push((i,e)),
            Ok(a)=>{
                let k=(a.local.clone(),a.domain.clone());
                if let Some(&first)=seen.get(&k){duplicate_of[i]=Some(first);}
                else{seen.insert(k,i);source_indices.push(i);addresses.push(a);}
            }
        }
    }
    DedupResult{addresses,source_indices,duplicate_of,errors}
}
