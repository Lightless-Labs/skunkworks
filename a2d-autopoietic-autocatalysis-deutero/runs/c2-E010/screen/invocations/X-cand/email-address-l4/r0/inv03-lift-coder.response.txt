- **Error order:** I check raw emptiness and non-ASCII first, then scan all CFWS for lexical errors before checking run boundaries.
- **Quoted local:** Only a quote at the first retained byte protects `@`; its first unescaped closing quote ends that protection.
- **CFWS boundary:** A removed run is allowed only at an edge or next to a retained `@`; the reported offset is the run’s first original byte.
- **Domain root dot:** Exactly one final dot is removed; an empty last label after that removal reports the removed dot’s original offset.
- **Normalization:** The entire address is validated before lowercasing or stripping the first `+`; an empty stripped local reports that plus’s original offset.
- **Independent implementation:** Both versions use byte offsets and linear scans; B uses a separate cleanup representation and validation flow.

### Solution A

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut protected_quote = false;
    let mut quote_open = 0;

    while i < input.len() {
        if protected_quote {
            let b = input[i];
            bytes.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    bytes.push(input[i]);
                    positions.push(i);
                    i += 1;
                }
            } else {
                if b == b'"' {
                    protected_quote = false;
                }
                i += 1;
            }
            continue;
        }

        let b = input[i];
        if bytes.is_empty() && b == b'"' {
            protected_quote = true;
            quote_open = i;
            bytes.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        let construct_start = i;
        let mut consumed = false;

        if b == b'\r' || b == b'\n' {
            if b == b'\r'
                && i + 2 < input.len()
                && input[i + 1] == b'\n'
                && is_hws(input[i + 2])
            {
                i += 2;
                while i < input.len() && is_hws(input[i]) {
                    i += 1;
                }
                consumed = true;
            } else {
                return Err(EmailError::BadFold { at: i });
            }
        } else if is_hws(b) {
            i += 1;
            while i < input.len() && is_hws(input[i]) {
                i += 1;
            }
            consumed = true;
        } else if b == b'(' {
            let outer = i;
            let depth = 1usize;
            let mut depth = depth;
            i += 1;
            loop {
                if i >= input.len() {
                    return Err(EmailError::UnclosedComment { at: outer });
                }
                let c = input[i];
                if c == b'\\' {
                    let slash = i;
                    if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                        return Err(EmailError::CommentEscape { at: slash });
                    }
                    i += 2;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if c == b'\r'
                        && i + 2 < input.len()
                        && input[i + 1] == b'\n'
                        && is_hws(input[i + 2])
                    {
                        i += 2;
                        while i < input.len() && is_hws(input[i]) {
                            i += 1;
                        }
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if c == b'(' {
                    if depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    depth += 1;
                    i += 1;
                    continue;
                }
                if c == b')' {
                    depth -= 1;
                    i += 1;
                    if depth == 0 {
                        break;
                    }
                    continue;
                }
                if !is_printable(c) && c != b'\t' {
                    return Err(EmailError::CommentChar { at: i });
                }
                i += 1;
            }
            consumed = true;
        } else if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if consumed {
            let run_start = construct_start;
            let mut run_end = i;
            loop {
                if run_end >= input.len() {
                    break;
                }
                let c = input[run_end];
                if is_hws(c) {
                    run_end += 1;
                    while run_end < input.len() && is_hws(input[run_end]) {
                        run_end += 1;
                    }
                    i = run_end;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if c == b'\r'
                        && run_end + 2 < input.len()
                        && input[run_end + 1] == b'\n'
                        && is_hws(input[run_end + 2])
                    {
                        run_end += 2;
                        while run_end < input.len() && is_hws(input[run_end]) {
                            run_end += 1;
                        }
                        i = run_end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: run_end });
                }
                if c == b'(' {
                    let outer = run_end;
                    let mut d = 1usize;
                    run_end += 1;
                    loop {
                        if run_end >= input.len() {
                            return Err(EmailError::UnclosedComment { at: outer });
                        }
                        let x = input[run_end];
                        if x == b'\\' {
                            if run_end + 1 >= input.len() || !is_printable(input[run_end + 1]) {
                                return Err(EmailError::CommentEscape { at: run_end });
                            }
                            run_end += 2;
                            continue;
                        }
                        if x == b'\r' || x == b'\n' {
                            if x == b'\r'
                                && run_end + 2 < input.len()
                                && input[run_end + 1] == b'\n'
                                && is_hws(input[run_end + 2])
                            {
                                run_end += 2;
                                while run_end < input.len() && is_hws(input[run_end]) {
                                    run_end += 1;
                                }
                                continue;
                            }
                            return Err(EmailError::BadFold { at: run_end });
                        }
                        if x == b'(' {
                            if d == 3 {
                                return Err(EmailError::CommentDepth { at: run_end });
                            }
                            d += 1;
                        } else if x == b')' {
                            d -= 1;
                            run_end += 1;
                            if d == 0 {
                                break;
                            }
                            continue;
                        } else if !is_printable(x) && x != b'\t' {
                            return Err(EmailError::CommentChar { at: run_end });
                        }
                        run_end += 1;
                    }
                    i = run_end;
                    continue;
                }
                break;
            }
            runs.push(Run {
                start: run_start,
                end: run_end,
            });
            i = run_end;
        } else {
            bytes.push(b);
            positions.push(i);
            i += 1;
        }
    }

    let _ = quote_open;
    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn map_pos(c: &Cleaned, index: usize, input_len: usize) -> usize {
    c.positions.get(index).copied().unwrap_or(input_len)
}

fn dot_atom_valid(bytes: &[u8]) -> bool {
    if bytes.is_empty() || bytes.len() > 64 {
        return false;
    }
    let mut previous_dot = true;
    for &b in bytes {
        if b == b'.' {
            if previous_dot {
                return false;
            }
            previous_dot = true;
        } else {
            if !is_atext(b) {
                return false;
            }
            previous_dot = false;
        }
    }
    !previous_dot
}

fn is_atext(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn parse_cleaned(c: &Cleaned, input_len: usize, allow_single: bool) -> Result<(Address, usize), EmailError> {
    for run in &c.runs {
        let left_at = run.start > 0 && c.bytes.get(run.start - 1) == Some(&b'@');
        let right_at = run.end < c.bytes.len() && c.bytes.get(run.end) == Some(&b'@');
        let at_edge = run.start == 0 || run.end == input_len;
        if !at_edge && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if c.bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if c.bytes[0] == b'"' {
        let mut i = 1;
        while i < c.bytes.len() {
            if c.bytes[i] == b'\\' {
                i += 2;
            } else if c.bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: map_pos(c, 0, input_len),
            });
        }
    }

    let mut ats = Vec::new();
    let protected_end = quote_end.unwrap_or(0);
    for i in 0..c.bytes.len() {
        if c.bytes[i] == b'@' && !(quote_end.is_some() && i <= protected_end) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: map_pos(c, ats[1], input_len),
        });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: map_pos(c, sep, input_len),
        });
    }

    let mut local = Vec::new();
    let mut origins = Vec::new();
    let quoted;
    if c.bytes[0] == b'"' {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: map_pos(c, end + 1, input_len),
            });
        }
        quoted = true;
        let mut i = 1;
        while i < end {
            let b = c.bytes[i];
            if b == b'\\' {
                if i + 1 >= end {
                    return Err(EmailError::BadEscape {
                        at: map_pos(c, i, input_len),
                    });
                }
                let next = c.bytes[i + 1];
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape {
                        at: map_pos(c, i, input_len),
                    });
                }
                local.push(next);
                origins.push(map_pos(c, i, input_len));
                i += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: map_pos(c, i, input_len),
                    });
                }
                local.push(b);
                origins.push(map_pos(c, i, input_len));
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: map_pos(c, end, input_len),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
    } else {
        quoted = false;
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: map_pos(c, 64, input_len),
            });
        }
        let bytes = &c.bytes[..sep];
        for i in 0..bytes.len() {
            let b = bytes[i];
            if b == b'.' {
                if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: map_pos(c, i, input_len),
                    });
                }
            } else if !is_atext(b) {
                return Err(EmailError::LocalChar {
                    at: map_pos(c, i, input_len),
                });
            }
        }
        local.extend_from_slice(bytes);
        for i in 0..bytes.len() {
            origins.push(map_pos(c, i, input_len));
        }
    }

    let domain_start = sep + 1;
    if domain_start >= c.bytes.len() {
        return Err(EmailError::EmptyDomain {
            at: map_pos(c, domain_start, input_len),
        });
    }

    let mut domain_end = c.bytes.len();
    let mut removed_root = None;
    if domain_end > domain_start && c.bytes[domain_end - 1] == b'.' {
        removed_root = Some(domain_end - 1);
        domain_end -= 1;
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: map_pos(c, domain_start, input_len),
        });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: map_pos(c, domain_start + 253, input_len),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || c.bytes[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }
    for (li, &(s, e)) in labels.iter().enumerate() {
        if s == e {
            let at = if li + 1 == labels.len() && removed_root.is_some() {
                map_pos(c, removed_root.unwrap(), input_len)
            } else {
                map_pos(c, s, input_len)
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if e - s > 63 {
            return Err(EmailError::LabelTooLong {
                at: map_pos(c, s + 63, input_len),
            });
        }
        if c.bytes[s] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: map_pos(c, s, input_len),
            });
        }
        if c.bytes[e - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: map_pos(c, e - 1, input_len),
            });
        }
        for i in s..e {
            let b = c.bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: map_pos(c, i, input_len),
                });
            }
        }
    }
    if labels.len() < 2 && !allow_single {
        return Err(EmailError::DomainNeedsDot {
            at: map_pos(c, domain_start, input_len),
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if c.bytes[last_start..last_end].iter().all(|b| b.is_ascii_digit()) {
        return Err(EmailError::NumericTld {
            at: map_pos(c, last_start, input_len),
        });
    }

    let local_string = String::from_utf8(local).unwrap();
    let domain = String::from_utf8(
        c.bytes[domain_start..domain_end]
            .iter()
            .map(u8::to_ascii_lowercase)
            .collect(),
    )
    .unwrap();
    Ok((
        Address {
            local: local_string,
            domain,
            quoted,
        },
        origins.first().copied().unwrap_or(map_pos(c, 0, input_len)),
    ))
}

fn parse_with_policy(input: &str, policy: &Policy, allow_single: bool) -> Result<(Address, usize), EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }
    let cleaned = cleanup(raw)?;
    let (mut address, first_origin) = parse_cleaned(&cleaned, raw.len(), allow_single)?;

    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }
    if policy.strip_subaddress {
        if let Some(plus) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if plus == 0 {
                let at = if address.quoted {
                    let mut decoded_index = 0usize;
                    let mut source = 1usize;
                    while source < cleaned.bytes.len() && cleaned.bytes[source] != b'"' {
                        if cleaned.bytes[source] == b'\\' {
                            source += 2;
                        } else {
                            if decoded_index == 0 {
                                break;
                            }
                            decoded_index += 1;
                            source += 1;
                        }
                    }
                    map_pos(&cleaned, source, raw.len())
                } else {
                    first_origin
                };
                return Err(EmailError::EmptyNormalizedLocal { at });
            }
            address.local.truncate(plus);
        }
    }
    if dot_atom_valid(address.local.as_bytes()) {
        if policy.unquote_safe {
            address.quoted = false;
        }
    } else {
        address.quoted = true;
    }
    Ok((address, first_origin))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let policy = Policy::default();
    parse_with_policy(input, &policy, false).map(|x| x.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, policy, policy.allow_single_label).map(|x| x.0)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                if let Some((_, _, first)) = keys
                    .iter()
                    .find(|(local, domain, _)| local == &address.local && domain == &address.domain)
                {
                    duplicate_of[index] = Some(*first);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), index));
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}

struct Stream {
    data: Vec<u8>,
    pos: Vec<usize>,
    run: Vec<Option<usize>>,
    run_starts: Vec<(usize, usize)>,

fn horizontal(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn printable(b: u8) -> bool {
    b >= 0x20 && b <= 0x7e
}

fn fold_end(src: &[u8], i: usize) -> Option<usize> {
    if i + 2 >= src.len() || src[i] != b'\r' || src[i + 1] != b'\n' || !horizontal(src[i + 2]) {
        return None;
    }
    let mut j = i + 2;
    while j < src.len() && horizontal(src[j]) {
        j += 1;
    }
    Some(j)
}

fn scan_comment(src: &[u8], start: usize) -> Result<usize, EmailError> {
    let outer = start;
    let mut stack = vec![start];
    let mut i = start + 1;
    while i < src.len() {
        let b = src[i];
        if b == b'\\' {
            if i + 1 >= src.len() || !printable(src[i + 1]) {
                return Err(EmailError::CommentEscape { at: i });
            }
            i += 2;
        } else if b == b'\r' || b == b'\n' {
            if let Some(end) = fold_end(src, i) {
                i = end;
            } else {
                return Err(EmailError::BadFold { at: i });
            }
        } else if b == b'(' {
            if stack.len() == 3 {
                return Err(EmailError::CommentDepth { at: i });
            }
            stack.push(i);
            i += 1;
        } else if b == b')' {
            stack.pop();
            i += 1;
            if stack.is_empty() {
                return Ok(i);
            }
        } else {
            if !printable(b) && b != b'\t' {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
        }
    }
    Err(EmailError::UnclosedComment { at: outer })
}

fn cleanup(src: &[u8]) -> Result<Stream, EmailError> {
    let mut data = Vec::with_capacity(src.len());
    let mut pos = Vec::with_capacity(src.len());
    let mut run = Vec::with_capacity(src.len());
    let mut run_starts = Vec::new();
    let mut i = 0usize;
    let mut quoted = false;
    let mut current_run: Option<usize> = None;

    while i < src.len() {
        if quoted {
            data.push(src[i]);
            pos.push(i);
            run.push(None);
            if src[i] == b'\\' {
                i += 1;
                if i < src.len() {
                    data.push(src[i]);
                    pos.push(i);
                    run.push(None);
                    i += 1;
                }
            } else {
                if src[i] == b'"' {
                    quoted = false;
                }
                i += 1;
            }
            continue;
        }

        if data.is_empty() && src[i] == b'"' {
            quoted = true;
            data.push(src[i]);
            pos.push(i);
            run.push(None);
            i += 1;
            continue;
        }

        let start = i;
        let end = if horizontal(src[i]) {
            let mut j = i + 1;
            while j < src.len() && horizontal(src[j]) {
                j += 1;
            }
            Some(j)
        } else if src[i] == b'\r' || src[i] == b'\n' {
            fold_end(src, i).map(Some).unwrap_or(None)
        } else if src[i] == b'(' {
            Some(scan_comment(src, i)?)
        } else if src[i] == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        } else {
            None
        };

        if let Some(end) = end {
            let run_id = current_run.unwrap_or_else(|| {
                let n = run_starts.len();
                run_starts.push((start, end));
                n
            });
            if current_run.is_some() {
                run_starts[run_id].1 = end;
            }
            current_run = Some(run_id);
            i = end;
            continue;
        }

        if src[i] == b'\r' || src[i] == b'\n' {
            return Err(EmailError::BadFold { at: i });
        }

        current_run = None;
        data.push(src[i]);
        pos.push(i);
        run.push(None);
        i += 1;
    }

    let _ = run;
    Ok(Stream {
        data,
        pos,
        run: Vec::new(),
        run_starts,
    })
}

fn offset(s: &Stream, i: usize, n: usize) -> usize {
    s.pos.get(i).copied().unwrap_or(n)
}

fn atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() || bytes.len() > 64 {
        return false;
    }
    let mut last_dot = true;
    for &b in bytes {
        if b == b'.' {
            if last_dot {
                return false;
            }
            last_dot = true;
        } else if atom_char(b) {
            last_dot = false;
        } else {
            return false;
        }
    }
    !last_dot
}

fn parse(src: &[u8], policy: &Policy, allow_single: bool) -> Result<(Address, usize), EmailError> {
    let s = cleanup(src)?;
    for &(start, end) in &s.run_starts {
        let left = start == 0 || s.data.get(start.wrapping_sub(1)) == Some(&b'@');
        let right = end == src.len() || s.data.get(0).is_none() ||
            s.pos.iter().position(|&p| p >= end).map_or(true, |i| s.data.get(i) == Some(&b'@'));
        if !left && !right {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }
    if s.data.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut qend = None;
    if s.data[0] == b'"' {
        let mut i = 1;
        while i < s.data.len() {
            if s.data[i] == b'\\' {
                i += 2;
            } else if s.data[i] == b'"' {
                qend = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if qend.is_none() {
            return Err(EmailError::UnclosedQuote { at: offset(&s, 0, src.len()) });
        }
    }
    let mut ats = Vec::new();
    for i in 0..s.data.len() {
        if s.data[i] == b'@' && !(qend.is_some_and(|q| i <= q)) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt { at: offset(&s, ats[1], src.len()) });
    }
    let at = ats[0];
    if at == 0 {
        return Err(EmailError::EmptyLocal { at: offset(&s, at, src.len()) });
    }

    let (mut local, mut origins, mut quoted) = (Vec::new(), Vec::new(), false);
    if s.data[0] == b'"' {
        let end = qend.unwrap();
        if end + 1 != at {
            return Err(EmailError::TrailingLocal { at: offset(&s, end + 1, src.len()) });
        }
        quoted = true;
        let mut i = 1;
        while i < end {
            if s.data[i] == b'\\' {
                if i + 1 >= end || !matches!(s.data[i + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: offset(&s, i, src.len()) });
                }
                local.push(s.data[i + 1]);
                origins.push(offset(&s, i, src.len()));
                i += 2;
            } else {
                if !printable(s.data[i]) {
                    return Err(EmailError::QuotedChar { at: offset(&s, i, src.len()) });
                }
                local.push(s.data[i]);
                origins.push(offset(&s, i, src.len()));
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: offset(&s, end, src.len()) });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        if at > 64 {
            return Err(EmailError::LocalTooLong { at: offset(&s, 64, src.len()) });
        }
        for i in 0..at {
            let b = s.data[i];
            if b == b'.' {
                if i == 0 || i + 1 == at || s.data[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: offset(&s, i, src.len()) });
                }
            } else if !atom_char(b) {
                return Err(EmailError::LocalChar { at: offset(&s, i, src.len()) });
            }
            local.push(b);
            origins.push(offset(&s, i, src.len()));
        }
    }

    let ds = at + 1;
    if ds >= s.data.len() {
        return Err(EmailError::EmptyDomain { at: offset(&s, ds, src.len()) });
    }
    let mut de = s.data.len();
    let root = if s.data[de - 1] == b'.' { de -= 1; Some(de) } else { None };
    if de == ds {
        return Err(EmailError::EmptyDomain { at: offset(&s, ds, src.len()) });
    }
    if de - ds > 253 {
        return Err(EmailError::DomainTooLong { at: offset(&s, ds + 253, src.len()) });
    }
    let mut labels = Vec::new();
    let mut p = ds;
    for i in ds..=de {
        if i == de || s.data[i] == b'.' {
            labels.push((p, i));
            p = i + 1;
        }
    }
    for (li, &(a, b)) in labels.iter().enumerate() {
        if a == b {
            let x = if li + 1 == labels.len() && root.is_some() { root.unwrap() } else { a };
            return Err(EmailError::EmptyLabel { at: offset(&s, x, src.len()) });
        }
        if b - a > 63 {
            return Err(EmailError::LabelTooLong { at: offset(&s, a + 63, src.len()) });
        }
        if s.data[a] == b'-' {
            return Err(EmailError::LabelHyphen { at: offset(&s, a, src.len()) });
        }
        if s.data[b - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: offset(&s, b - 1, src.len()) });
        }
        for i in a..b {
            if !s.data[i].is_ascii_alphanumeric() && s.data[i] != b'-' {
                return Err(EmailError::DomainChar { at: offset(&s, i, src.len()) });
            }
        }
    }
    if labels.len() < 2 && !allow_single {
        return Err(EmailError::DomainNeedsDot { at: offset(&s, ds, src.len()) });
    }
    let (a, b) = *labels.last().unwrap();
    if s.data[a..b].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: offset(&s, a, src.len()) });
    }

    let mut address = Address {
        local: String::from_utf8(local).unwrap(),
        domain: String::from_utf8(s.data[ds..de].iter().map(u8::to_ascii_lowercase).collect()).unwrap(),
        quoted,
    };

    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }
    if policy.strip_subaddress {
        if let Some(p) = address.local.find('+') {
            if p == 0 {
                let source = if quoted {
                    let mut decoded = 0usize;
                    let mut i = 1usize;
                    let mut found = offset(&s, 0, src.len());
                    while i < qend.unwrap() {
                        if decoded == 0 {
                            found = offset(&s, i, src.len());
                            break;
                        }
                        if s.data[i] == b'\\' {
                            i += 2;
                        } else {
                            i += 1;
                            decoded += 1;
                        }
                    }
                    found
                } else {
                    origins.first().copied().unwrap_or(offset(&s, 0, src.len()))
                };
                return Err(EmailError::EmptyNormalizedLocal { at: source });
            }
            address.local.truncate(p);
        }
    }
    if atom(address.local.as_bytes()) {
        if policy.unquote_safe {
            address.quoted = false;
        }
    } else {
        address.quoted = true;
    }

    let first = origins.first().copied().unwrap_or(offset(&s, 0, src.len()));
    Ok((address, first))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse(input.as_bytes(), &Policy::default(), false).map(|x| x.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse(input.as_bytes(), policy, policy.allow_single_label).map(|x| x.0)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: Vec<(String, String, usize)> = Vec::new();

    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(e) => errors.push((i, e)),
            Ok(address) => {
                if let Some((_, _, first)) = seen.iter().find(|(l, d, _)| l == &address.local && d == &address.domain) {
                    duplicate_of[i] = Some(*first);
                } else {
                    seen.push((address.local.clone(), address.domain.clone(), i));
                    source_indices.push(i);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult { addresses, source_indices, duplicate_of, errors }
}
