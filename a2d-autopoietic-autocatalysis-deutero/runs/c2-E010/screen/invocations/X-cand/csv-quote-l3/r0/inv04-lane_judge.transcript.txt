fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig { component: ConfigPart::Delimiter });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig { component: ConfigPart::Quote });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig { component: ConfigPart::Quote });
    }
    Ok(())
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;

    let mut quoted = false;
    let mut after_quote = false;
    let mut opening_quote = 0usize;
    let mut field_started = false;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: i });
        }

        if quoted {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote as char);
                    i += 2;
                } else {
                    quoted = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if b == b'\r' && (i + 1 >= bytes.len() || bytes[i + 1] != b'\n') {
            return Err(CsvError::BareCr { record: record_index, byte: i });
        }

        if after_quote {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                field_started = false;
                record_started = true;
                after_quote = false;
                i += 1;
            } else if b == b'\n' || b == b'\r' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_started = false;
                field_started = false;
                after_quote = false;
                record_index += 1;
                i += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record: record_index, byte: i });
            }
            continue;
        }

        if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote { record: record_index, byte: i });
            }
            quoted = true;
            opening_quote = i;
            field_started = true;
            record_started = true;
            i += 1;
        } else if b == delimiter {
            record.push(std::mem::take(&mut field));
            field_started = false;
            record_started = true;
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_started = false;
            field_started = false;
            record_index += 1;
            i += if b == b'\r' { 2 } else { 1 };
        } else {
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            field_started = true;
            record_started = true;
            i += ch.len_utf8();
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: opening_quote });
    }

    if record_started || field_started || after_quote || !record.is_empty() {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table {
            header: None,
            rows: records,
            duplicates: config.duplicates,
        });
    }
    if records.is_empty() {
        return Ok(Table { header: None, rows: Vec::new(), duplicates: config.duplicates });
    }

    let mut iter = records.into_iter();
    let header = iter.next().unwrap();

    if config.duplicates == DuplicatePolicy::Reject {
        let mut seen = std::collections::HashMap::<String, usize>::new();
        for (second, name) in header.iter().enumerate() {
            if let Some(&first) = seen.get(name) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first,
                    second,
                });
            }
            seen.insert(name.clone(), second);
        }
    }

    let rows: Vec<Vec<String>> = iter.collect();
    for (offset, row) in rows.iter().enumerate() {
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth {
                record: offset + 1,
                expected: header.len(),
                actual: row.len(),
            });
        }
    }

    Ok(Table { header: Some(header), rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self { config, input: String::new(), cached: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let digits = if b[0] == b'-' { &b[1..] } else { b };
    if digits.is_empty() { return false; }
    if digits == b"0" { return b[0] != b'-'; }
    digits[0].is_ascii_digit() && digits[0] != b'0' && digits.iter().all(u8::is_ascii_digit)
}

fn decimal_parse(s: &str) -> Result<i64, CellErrorKind> {
    let b = s.as_bytes();
    let dot = match b.iter().position(|&x| x == b'.') {
        Some(p) => p,
        None => return Err(CellErrorKind::InvalidDecimal),
    };
    if dot + 3 != b.len() || dot == 0 {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let integral = &s[..dot];
    let frac = &b[dot + 1..];
    if frac.len() != 2 || !frac.iter().all(u8::is_ascii_digit) || !integer_syntax(integral) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = integral.starts_with('-');
    let whole = if negative { &integral[1..] } else { integral };
    let magnitude = whole.parse::<i128>().unwrap_or(i128::MAX);
    let cents = magnitude * 100 + ((frac[0] - b'0') as i128) * 10 + (frac[1] - b'0') as i128;
    if negative && cents == 0 {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let signed = if negative { -cents } else { cents };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn convert(value: &str, kind: ColumnType) -> Result<Value, CellErrorKind> {
    match kind {
        ColumnType::Text => Ok(Value::Text(value.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(value) { return Err(CellErrorKind::InvalidInteger); }
            value.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => decimal_parse(value).map(Value::Decimal),
        ColumnType::Boolean => match value {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    for (row_index, row) in table.rows.iter().enumerate() {
        let mut out = Vec::with_capacity(columns.len());
        let mut stop_row = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            if stop_row {
                out.push(None);
                continue;
            }

            let column = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let evaluated = if column.is_none() {
                Err((None, CellErrorKind::UnknownColumn))
            } else {
                let index = column.unwrap();
                match row.get(index) {
                    None => Err((Some(index), CellErrorKind::MissingCell)),
                    Some(cell) if cell.is_empty() && spec.nullable => Ok(Value::Null),
                    Some(cell) if cell.is_empty() && spec.kind != ColumnType::Text => {
                        Err((Some(index), CellErrorKind::Empty))
                    }
                    Some(cell) => convert(cell, spec.kind).map_err(|kind| (Some(index), kind)),
                }
            };

            match evaluated {
                Ok(value) => out.push(Some(value)),
                Err((column, kind)) => {
                    out.push(None);
                    result.errors.push(CellError { row: row_index, spec: spec_index, column, kind });
                    if policy != ErrorPolicy::Collect {
                        stop_row = true;
                    }
                    if policy == ErrorPolicy::Stop {
                        for _ in spec_index + 1..columns.len() { out.push(None); }
                        result.rows.push(out);
                        return result;
                    }
                }
            }
        }
        result.rows.push(out);
    }
    result
}

fn config_ok(c: Config) -> Result<(), CsvError> {
    if c.delimiter < 0x21 || c.delimiter > 0x7e {
        return Err(CsvError::InvalidConfig { component: ConfigPart::Delimiter });
    }
    if c.quote < 0x21 || c.quote > 0x7e {
        return Err(CsvError::InvalidConfig { component: ConfigPart::Quote });

        return Err(CsvError::InvalidConfig { component: ConfigPart::Quote });

fn lex(input: &str, d: u8, q: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let b = input.as_bytes();
    let mut p = if b.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    let mut out = Vec::new();
    if p == b.len() { return Ok(out); }

    let mut rec = 0usize;
    let mut row = Vec::new();
    let mut value = String::new();
    let mut quoted = false;
    let mut closed = false;
    let mut has_content = false;
    let mut opener = 0usize;

    while p < b.len() {
        let c = b[p];
        if c == 0 { return Err(CsvError::Nul { record: rec, byte: p }); }

        if quoted {
            if c == q {
                if p + 1 < b.len() && b[p + 1] == q {
                    value.push(q as char);
                    p += 2;
                } else {
                    quoted = false;
                    closed = true;
                    p += 1;
                }
            } else {
                let ch = input[p..].chars().next().unwrap();
                value.push(ch);
                p += ch.len_utf8();
            }
            continue;
        }

        if c == b'\r' && (p + 1 == b.len() || b[p + 1] != b'\n') {
            return Err(CsvError::BareCr { record: rec, byte: p });
        }
        if closed {
            if c == d {
                row.push(std::mem::take(&mut value));
                closed = false;
                has_content = false;
                p += 1;
            } else if c == b'\n' || c == b'\r' {
                row.push(std::mem::take(&mut value));
                out.push(std::mem::take(&mut row));
                rec += 1;
                closed = false;
                has_content = false;
                p += if c == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record: rec, byte: p });
            }
        } else if c == d {
            row.push(std::mem::take(&mut value));
            has_content = false;
            p += 1;
        } else if c == b'\n' || c == b'\r' {
            row.push(std::mem::take(&mut value));
            out.push(std::mem::take(&mut row));
            rec += 1;
            has_content = false;
            p += if c == b'\r' { 2 } else { 1 };
        } else if c == q {
            if has_content {
                return Err(CsvError::UnexpectedQuote { record: rec, byte: p });
            }
            quoted = true;
            opener = p;
            has_content = true;
            p += 1;
        } else {
            let ch = input[p..].chars().next().unwrap();
            value.push(ch);
            has_content = true;
            p += ch.len_utf8();
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote { record: rec, byte: opener });
    }
    if !row.is_empty() || !value.is_empty() || has_content || closed {
        row.push(value);
        out.push(row);
    }
    Ok(out)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {

}

fn make_table(input: &str, c: Config) -> Result<Table, CsvError> {
    config_ok(c)?;
    let records = lex(input, c.delimiter, c.quote)?;
    let (header, rows, first_data_record) = if c.headers && !records.is_empty() {
        (Some(records[0].clone()), records[1..].to_vec(), 1)
    } else if c.headers {
        (None, Vec::new(), 1)
    } else {
        (None, records, 0)
    };

    if c.duplicates == DuplicatePolicy::Reject {
        if let Some(h) = &header {
            let mut seen = std::collections::HashMap::<&str, usize>::new();
            for (i, name) in h.iter().enumerate() {
                if let Some(&first) = seen.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader { name: name.clone(), first, second: i });
                }
                seen.insert(name, i);
            }
        }
    }

    if let Some(h) = &header {
        for (i, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: first_data_record + i,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table { header, rows, duplicates: c.duplicates })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        config_ok(config)?;
        Ok(Self { config, bytes: Vec::new(), finished: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() { return Err(CsvError::ReaderFinished); }
        self.bytes.extend_from_slice(chunk.as_bytes());
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(r) = &self.finished { return r.clone(); }
        let text = std::str::from_utf8(&self.bytes).unwrap();
        let r = make_table(text, self.config);
        self.finished = Some(r.clone());
        r
    }

    pub fn reset(&mut self) {
        self.bytes.clear();
        self.finished = None;
    }
}

    let mut r = StreamReader::new(config)?;
    r.feed(input)?;
    r.finish()

fn int_grammar(s: &str) -> bool {
    let a = s.as_bytes();
    if a.is_empty() { return false; }
    let n = usize::from(a[0] == b'-');
    if n == a.len() { return false; }
    if a[n] == b'0' { return n == 0 && a.len() == 1; }
    (b'1'..=b'9').contains(&a[n]) && a[n+1..].iter().all(|x| x.is_ascii_digit())
}

fn dec_grammar(s: &str) -> bool {
    let a = s.as_bytes();
    if a.is_empty() { return false; }
    let n = usize::from(a[0] == b'-');
    if n >= a.len() { return false; }
    let dot = match a[n..].iter().position(|x| *x == b'.') { Some(x) => n+x, None => return false };
    if a.len() != dot+3 { return false; }
    let whole = &a[n..dot];
    let valid_whole = whole == b"0" ||
        (!whole.is_empty() && (b'1'..=b'9').contains(&whole[0]) &&
         whole[1..].iter().all(|x| x.is_ascii_digit()));
    valid_whole && a[dot+1..].iter().all(|x| x.is_ascii_digit()) &&
        !(n == 1 && whole == b"0" && a[dot+1..] == *b"00")
}

fn convert(s: &str, k: ColumnType, nullable: bool) -> Result<Value, CellErrorKind> {
    if nullable && s.is_empty() { return Ok(Value::Null); }
    if s.is_empty() && k != ColumnType::Text { return Err(CellErrorKind::Empty); }
    match k {
        ColumnType::Text => Ok(Value::Text(s.to_owned())),
        ColumnType::Integer => {
            if !int_grammar(s) { return Err(CellErrorKind::InvalidInteger); }
            s.parse().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !dec_grammar(s) { return Err(CellErrorKind::InvalidDecimal); }
            let neg = s.starts_with('-');
            let raw = if neg { &s[1..] } else { s };
            let mut it = raw.split('.');
            let whole = it.next().unwrap();
            let frac = it.next().unwrap();
            let mag: i128 = format!("{whole}{frac}").parse().map_err(|_| CellErrorKind::DecimalOverflow)?;
            let signed = if neg { -mag } else { mag };
            i64::try_from(signed).map(Value::Decimal).map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        }
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut output = TypedResult { rows: Vec::new(), errors: Vec::new() };
    'outer: for (ri, row) in table.rows.iter().enumerate() {
        let mut result_row = Vec::with_capacity(columns.len());
        for (si, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(n) => table.column_index(n),
            };
            let (cell_index, evaluated) = match resolved {
                None => (None, Err(CellErrorKind::UnknownColumn)),
                Some(i) => match row.get(i) {
                    None => (Some(i), Err(CellErrorKind::MissingCell)),
                    Some(s) => (Some(i), convert(s, spec.kind, spec.nullable)),
                }
            };
            match evaluated {
                Ok(v) => result_row.push(Some(v)),
                Err(kind) => {
                    output.errors.push(CellError { row: ri, spec: si, column: cell_index, kind });
                    result_row.push(None);
                    match policy {
                        ErrorPolicy::Collect => {}
                        ErrorPolicy::FirstPerRow => {
                            result_row.resize(columns.len(), None);
                            break;
                        }
                        ErrorPolicy::Stop => {
                            result_row.resize(columns.len(), None);
                            output.rows.push(result_row);
                            break 'outer;
                        }
                    }
                }
            }
        }
        output.rows.push(result_row);
    }
    output
}

        }

                None=>(None,None,Some(CellErrorKind::UnknownColumn)),
                Some(i)=>match row.get(i){
                    None=>(Some(i),None,Some(CellErrorKind::MissingCell)),
                    Some(v) if v.is_empty()&&s.nullable=>(Some(i),Some(Value::Null),None),
                    Some(v) if v.is_empty()&&s.kind!=ColumnType::Text=>(Some(i),None,Some(CellErrorKind::Empty)),
                    Some(v)=>{
                        let z=match s.kind{
                            ColumnType::Text=>Ok(Value::Text(v.clone())),
                            ColumnType::Integer=>if !int_grammar(v){Err(CellErrorKind::InvalidInteger)}else{v.parse::<i64>().map(Value::Integer).map_err(|_|CellErrorKind::IntegerOverflow)},
                            ColumnType::Decimal=>if !dec_grammar(v){Err(CellErrorKind::InvalidDecimal)}else{dec_value(v).map(Value::Decimal).ok_or(CellErrorKind::DecimalOverflow)},
                            ColumnType::Boolean=>match v.as_str(){"yes"|"1"=>Ok(Value::Boolean(true)),"no"|"0"=>Ok(Value::Boolean(false)),_=>Err(CellErrorKind::InvalidBoolean)}
                        };
                        match z{Ok(x)=>(Some(i),Some(x),None),Err(e)=>(Some(i),None,Some(e))}
                    }
                }
            };
            out.push(val);
            if let Some(kind)=err{
                result.errors.push(CellError{row:ri,spec:si,column:col,kind});
                if policy!=ErrorPolicy::Collect{
                    stopped=true;
                    if policy==ErrorPolicy::Stop{
                        for _ in si+1..columns.len(){out.push(None)}
                        result.rows.push(out); return result
                    }
                }
            }
        }
        result.rows.push(out);
    }
    result
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig { component: ConfigPart::Delimiter });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig { component: ConfigPart::Quote });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig { component: ConfigPart::Quote });
    }
    Ok(())
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut field_started = false;
    let mut quote_start = 0usize;
    let mut record_index = 0usize;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: i });
        }

        if in_quotes {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote as char);
                    i += 2;
                    continue;
                }
                in_quotes = false;
                after_quote = true;
                i += 1;
                continue;
            }
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            i += ch.len_utf8();
            continue;
        }

        if b == b'\r' && (i + 1 >= bytes.len() || bytes[i + 1] != b'\n') {
            return Err(CsvError::BareCr { record: record_index, byte: i });
        }

        if after_quote {
            if b == delimiter {
                row.push(std::mem::take(&mut field));
                after_quote = false;
                field_started = false;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                record_index += 1;
                after_quote = false;
                field_started = false;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte { record: record_index, byte: i });
        }

        if b == delimiter {
            row.push(std::mem::take(&mut field));
            field_started = false;
            i += 1;
            continue;
        }
        if b == b'\n' || b == b'\r' {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            record_index += 1;
            field_started = false;
            i += if b == b'\r' { 2 } else { 1 };
            continue;
        }
        if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote { record: record_index, byte: i });
            }
            in_quotes = true;
            quote_start = i;
            field_started = true;
            i += 1;
            continue;
        }

        let ch = input[i..].chars().next().unwrap();
        field.push(ch);
        field_started = true;
        i += ch.len_utf8();
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: quote_start });
    }

    if !row.is_empty() || !field.is_empty() || field_started || after_quote {
        row.push(field);
        records.push(row);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

fn build_table(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let records = parse_with(input, config.delimiter, config.quote)?;

    let (header, rows, header_record_index) = if config.headers {
        if records.is_empty() {
            (None, Vec::new(), None)
        } else {
            (Some(records[0].clone()), records[1..].to_vec(), Some(1usize))
        }
    } else {
        (None, records, None)
    };

    if let (Some(h), DuplicatePolicy::Reject) = (&header, config.duplicates) {
        let mut seen: std::collections::HashMap<&str, usize> = std::collections::HashMap::new();
        for (i, name) in h.iter().enumerate() {
            if let Some(&first) = seen.get(name.as_str()) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(), first, second: i,
                });
            }
            seen.insert(name.as_str(), i);
        }
    }

    if let Some(h) = &header {
        for (i, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: header_record_index.unwrap_or(0) + i,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table { header, rows, duplicates: config.duplicates })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self { config, input: String::new(), cached: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = build_table(&self.input, self.config);
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    if b[start] == b'0' {
        return start == 0 && b.len() == 1;
    }
    if !(b'1'..=b'9').contains(&b[start]) { return false; }
    b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    let dot = match b[start..].iter().position(|&x| x == b'.') {
        Some(p) => start + p,
        None => return false,
    };
    if b.len() != dot + 3 { return false; }
    let integral = &b[start..dot];
    if integral.is_empty() { return false; }
    if integral == b"0" {
        // allowed unless the full spelling is negative zero
    } else if !(b'1'..=b'9').contains(&integral[0])
        || !integral[1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    b[dot + 1..].iter().all(u8::is_ascii_digit)
        && !(start == 1 && integral == b"0" && b[dot + 1..] == *b"00")
}

fn eval_cell(s: &str, kind: ColumnType, nullable: bool) -> Result<Value, CellErrorKind> {
    if nullable && s.is_empty() {
        return Ok(Value::Null);
    }
    if s.is_empty() && kind != ColumnType::Text {
        return Err(CellErrorKind::Empty);
    }
    match kind {
        ColumnType::Text => Ok(Value::Text(s.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(s) { return Err(CellErrorKind::InvalidInteger); }
            s.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(s) { return Err(CellErrorKind::InvalidDecimal); }
            let negative = s.starts_with('-');
            let body = if negative { &s[1..] } else { s };
            let mut parts = body.split('.');
            let whole = parts.next().unwrap();
            let frac = parts.next().unwrap();
            let magnitude_text = format!("{}{}", whole, frac);
            let magnitude = magnitude_text.parse::<i128>().map_err(|_| CellErrorKind::DecimalOverflow)?;
            let signed = if negative { -magnitude } else { magnitude };
            if signed < i64::MIN as i128 || signed > i64::MAX as i128 {
                return Err(CellErrorKind::DecimalOverflow);
            }
            Ok(Value::Decimal(signed as i64))
        }
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut out = Vec::with_capacity(columns.len());
        for (spec_index, spec) in columns.iter().enumerate() {
            let column = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let failure = if column.is_none() {
                Some((None, CellErrorKind::UnknownColumn))
            } else {
                let index = column.unwrap();
                match row.get(index) {
                    None => Some((Some(index), CellErrorKind::MissingCell)),
                    Some(cell) => match eval_cell(cell, spec.kind, spec.nullable) {
                        Ok(value) => {
                            out.push(Some(value));
                            continue;
                        }
                        Err(kind) => Some((Some(index), kind)),
                    }
                }
            };
            let (error_column, kind) = failure.unwrap();
            result.errors.push(CellError {
                row: row_index, spec: spec_index, column: error_column, kind,
            });
            out.push(None);
            match policy {
                ErrorPolicy::Collect => {}
                ErrorPolicy::FirstPerRow => {
                    out.extend((spec_index + 1..columns.len()).map(|_| None));
                    break;
                }
                ErrorPolicy::Stop => {
                    out.extend((spec_index + 1..columns.len()).map(|_| None));
                    result.rows.push(out);
                    break 'rows;
                }
            }
        }
        result.rows.push(out);
    }
    result
}

fn parse_dialect(s: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = s.as_bytes();
    let mut pos = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if pos == bytes.len() { return Ok(Vec::new()); }

    let mut output = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0 = unquoted, 1 = quoted, 2 = after closing quote
    let mut opened = false;
    let mut opening = 0usize;
    let mut rec = 0usize;

    while pos < bytes.len() {
        let ch = bytes[pos];
        if ch == 0 {
            return Err(CsvError::Nul { record: rec, byte: pos });
        }
        if state != 1 && ch == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record: rec, byte: pos });
        }

        match state {
            1 => {
                if ch == quote {
                    if bytes.get(pos + 1) == Some(&quote) {
                        field.push(quote as char);
                        pos += 2;
                    } else {
                        state = 2;
                        pos += 1;
                    }
                } else {
                    field.push(ch as char);
                    pos += 1;
                }
            }
            2 => {
                if ch == delimiter {
                    row.push(std::mem::take(&mut field));
                    opened = false;
                    state = 0;
                    pos += 1;
                } else if ch == b'\n' || ch == b'\r' {
                    row.push(std::mem::take(&mut field));
                    output.push(std::mem::take(&mut row));
                    rec += 1;
                    opened = false;
                    state = 0;
                    pos += if ch == b'\r' { 2 } else { 1 };
                } else {
                    return Err(CsvError::TrailingByte { record: rec, byte: pos });
                }
            }
            _ => {
                if ch == delimiter {
                    row.push(std::mem::take(&mut field));
                    opened = false;
                    pos += 1;
                } else if ch == b'\n' || ch == b'\r' {
                    row.push(std::mem::take(&mut field));
                    output.push(std::mem::take(&mut row));
                    rec += 1;
                    opened = false;
                    pos += if ch == b'\r' { 2 } else { 1 };
                } else if ch == quote {
                    if opened {
                        return Err(CsvError::UnexpectedQuote { record: rec, byte: pos });
                    }
                    opened = true;
                    state = 1;
                    opening = pos;
                    pos += 1;
                } else {
                    field.push(ch as char);
                    opened = true;
                    pos += 1;
                }
            }
        }
    }

    if state == 1 {
        return Err(CsvError::UnclosedQuote { record: rec, byte: opening });
    }
    if !row.is_empty() || !field.is_empty() || opened || state == 2 {
        row.push(field);
        output.push(row);
    }
    Ok(output)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn make_table(input: &str, c: Config) -> Result<Table, CsvError> {
    check_config(c)?;
    let parsed = parse_dialect(input, c.delimiter, c.quote)?;
    let (header, rows) = if c.headers && !parsed.is_empty() {
        (Some(parsed[0].clone()), parsed[1..].to_vec())
    } else { (None, parsed) };

    if c.duplicates == DuplicatePolicy::Reject {
        if let Some(h) = &header {
            let mut seen = HashMap::<String, usize>::new();
            for (j, n) in h.iter().enumerate() {
                if let Some(&first) = seen.get(n) {
                    return Err(CsvError::DuplicateHeader { name: n.clone(), first, second: j });
                }
                seen.insert(n.clone(), j);
            }
        }
    }
    if let Some(h) = &header {
        for (i, r) in rows.iter().enumerate() {
            if r.len() != h.len() {
                return Err(CsvError::HeaderWidth { record: i + 1, expected: h.len(), actual: r.len() });
            }
        }
    }
    Ok(Table { header, rows, duplicates: c.duplicates })
}

}

fn int_grammar(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let p = usize::from(b[0] == b'-');
    if p == b.len() { return false; }
    if b[p] == b'0' { return p + 1 == b.len(); }
    (b'1'..=b'9').contains(&b[p]) && b[p+1..].iter().all(|x| x.is_ascii_digit())
}
fn dec_grammar(s: &str) -> bool {
    let b=s.as_bytes();
    if b.is_empty() { return false; }
    let p=usize::from(b[0]==b'-');
    let d=match b[p..].iter().position(|x| *x==b'.') { Some(x)=>p+x, None=>return false };
    if d==p || b.len()!=d+3 || !b[d+1..].iter().all(|x| x.is_ascii_digit()) { return false; }
    if b[p]==b'0' { d==p+1 } else { (b'1'..=b'9').contains(&b[p]) && b[p+1..d].iter().all(|x| x.is_ascii_digit()) }
}
fn typed_value(s: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if s.is_empty() {
        if spec.nullable { return Ok(Value::Null); }
        if spec.kind != ColumnType::Text { return Err(CellErrorKind::Empty); }
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(s.to_owned())),
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
        ColumnType::Integer => {
            if !int_grammar(s) { return Err(CellErrorKind::InvalidInteger); }
            s.parse().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !dec_grammar(s) || s == "-0.00" { return Err(CellErrorKind::InvalidDecimal); }
            let neg=s.starts_with('-');
            let raw=if neg { &s[1..] } else { s };
            let d=raw.find('.').unwrap();
            let whole: i128=raw[..d].parse().map_err(|_| CellErrorKind::DecimalOverflow)?;
            let frac=((raw.as_bytes()[d+1]-b'0') as i128)*10+(raw.as_bytes()[d+2]-b'0') as i128;
            let n=whole*100+frac;
            let n=if neg { -n } else { n };
            i64::try_from(n).map(Value::Decimal).map_err(|_| CellErrorKind::DecimalOverflow)
        }
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result=TypedResult { rows: Vec::new(), errors: Vec::new() };
    'outer: for (ri,row) in table.rows.iter().enumerate() {
        let mut values=vec![None; columns.len()];
        let mut failed=false;
        for (si,spec) in columns.iter().enumerate() {
            let ix=match &spec.column { ColumnRef::Index(i)=>Some(*i), ColumnRef::Name(n)=>table.column_index(n) };
            let evaluated=match ix {
                None => (None,Err(CellErrorKind::UnknownColumn)),
                Some(i)=>match row.get(i) {
                    None=>(Some(i),Err(CellErrorKind::MissingCell)),
                    Some(cell)=>(Some(i),typed_value(cell,spec)),
                }
            };
            match evaluated {
                (col,Ok(v))=>values[si]=Some(v),
                (col,Err(kind))=>{
                    result.errors.push(CellError { row:ri, spec:si, column:col, kind });
                    failed=true;
                    if policy != ErrorPolicy::Collect { break; }
                }
            }
        }
        result.rows.push(values);
        if failed && policy==ErrorPolicy::Stop { break 'outer; }
    }
    result
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut field_bytes = Vec::new();

    #[derive(Clone, Copy)]
    enum State {
        Start,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::Start;
    let mut i = start;
    let mut record_index = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        match state {
            State::Quoted { opening } => {
                if b == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field_bytes.push(quote);
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    field_bytes.push(b);
                    i += 1;
                }
                record_started = true;
            }
            State::AfterQuote => {
                if b == b'\r' {
                    if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                        return Err(CsvError::BareCr {
                            record: record_index,
                            byte: i,
                        });
                    }
                    field = String::from_utf8(field_bytes).unwrap();
                    record.push(field);
                    records.push(record);
                    record = Vec::new();
                    field_bytes = Vec::new();
                    state = State::Start;
                    record_index += 1;
                    record_started = false;
                    i += 2;
                } else if b == delimiter {
                    field = String::from_utf8(field_bytes).unwrap();
                    record.push(field);
                    field_bytes = Vec::new();
                    state = State::Start;
                    record_started = true;
                    i += 1;
                } else if b == b'\n' {
                    field = String::from_utf8(field_bytes).unwrap();
                    record.push(field);
                    records.push(record);
                    record = Vec::new();
                    field_bytes = Vec::new();
                    state = State::Start;
                    record_index += 1;
                    record_started = false;
                    i += 1;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
            State::Start => {
                if b == b'\r' {
                    if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                        return Err(CsvError::BareCr {
                            record: record_index,
                            byte: i,
                        });
                    }
                    record.push(String::new());
                    records.push(record);
                    record = Vec::new();
                    state = State::Start;
                    record_index += 1;
                    record_started = false;
                    i += 2;
                } else if b == b'\n' {
                    record.push(String::new());
                    records.push(record);
                    record = Vec::new();
                    state = State::Start;
                    record_index += 1;
                    record_started = false;
                    i += 1;
                } else if b == delimiter {
                    record.push(String::new());
                    record_started = true;
                    i += 1;
                } else if b == quote {
                    state = State::Quoted { opening: i };
                    record_started = true;
                    i += 1;
                } else {
                    field_bytes.push(b);
                    state = State::Unquoted;
                    record_started = true;
                    i += 1;
                }
            }
            State::Unquoted => {
                if b == b'\r' {
                    if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                        return Err(CsvError::BareCr {
                            record: record_index,
                            byte: i,
                        });
                    }
                    record.push(String::from_utf8(field_bytes).unwrap());
                    field_bytes = Vec::new();
                    records.push(record);
                    record = Vec::new();
                    state = State::Start;
                    record_index += 1;
                    record_started = false;
                    i += 2;
                } else if b == b'\n' {
                    record.push(String::from_utf8(field_bytes).unwrap());
                    field_bytes = Vec::new();
                    records.push(record);
                    record = Vec::new();
                    state = State::Start;
                    record_index += 1;
                    record_started = false;
                    i += 1;
                } else if b == delimiter {
                    record.push(String::from_utf8(field_bytes).unwrap());
                    field_bytes = Vec::new();
                    state = State::Start;
                    record_started = true;
                    i += 1;
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else {
                    field_bytes.push(b);
                    i += 1;
                }
            }
        }
    }

    match state {
        State::Quoted { opening } => {
            return Err(CsvError::UnclosedQuote {
                record: record_index,
                byte: opening,
            });
        }
        State::AfterQuote => {
            record.push(String::from_utf8(field_bytes).unwrap());
            records.push(record);
        }
        State::Unquoted => {
            record.push(String::from_utf8(field_bytes).unwrap());
            records.push(record);
        }
        State::Start => {
            if record_started || !record.is_empty() {
                record.push(String::new());
                records.push(record);
            }
        }
    }

    Ok(records)
}

}

fn make_table(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let mut parsed = parse_with(input, config.delimiter, config.quote)?;

    let header = if config.headers && !parsed.is_empty() {
        Some(parsed.remove(0))
    } else {
        None
    };

    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: Vec<(&str, usize)> = Vec::new();
            for (second, name) in h.iter().enumerate() {
                if let Some((_, first)) = seen.iter().find(|(prior, _)| *prior == name) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: *first,
                        second,
                    });
                }
                seen.push((name, second));
            }
        }

        for (record, row) in parsed.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: record + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: parsed,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    finished: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            finished: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.finished {
            return result.clone();
        }
        let result = make_table(&self.input, self.config);
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let mut i = 0;
    if b[0] == b'-' {
        i = 1;
    }
    if i >= b.len() {
        return false;
    }
    if b[i] == b'0' {
        return i == 0 && b.len() == 1;
    }
    if !(b'1'..=b'9').contains(&b[i]) {
        return false;
    }
    b[i + 1..].iter().all(|c| c.is_ascii_digit())
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let mut i = 0;
    if b[0] == b'-' {
        i = 1;
    }
    if i >= b.len() {
        return false;
    }
    let int_start = i;
    if b[i] == b'0' {
        i += 1;
        if i < b.len() && b[i].is_ascii_digit() {
            return false;
        }
    } else if (b'1'..=b'9').contains(&b[i]) {
        i += 1;
        while i < b.len() && b[i].is_ascii_digit() {
            i += 1;
        }
    } else {
        return false;
    }
    if i >= b.len() || b[i] != b'.' {
        return false;
    }
    let dot = i;
    if b.len() != dot + 3 || !b[dot + 1].is_ascii_digit() || !b[dot + 2].is_ascii_digit() {
        return false;
    }
    if b[0] == b'-' && b[int_start] == b'0' && b[dot + 1] == b'0' && b[dot + 2] == b'0' {
        return false;
    }
    true
}

fn parse_decimal(s: &str) -> Option<i64> {
    let negative = s.starts_with('-');
    let body = if negative { &s[1..] } else { s };
    let dot = body.find('.')?;
    let whole: i128 = body[..dot].parse().ok()?;
    let frac: i128 = body[dot + 1..].parse().ok()?;
    let magnitude = whole.checked_mul(100)?.checked_add(frac)?;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).ok()
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut stopped_row = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if stopped_row {
                output.push(None);
                continue;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let (column, value, error) = match resolved {
                None => (
                    None,
                    None,
                    Some(CellErrorKind::UnknownColumn),
                ),
                Some(index) => match row.get(index) {
                    None => (Some(index), None, Some(CellErrorKind::MissingCell)),
                    Some(cell) if cell.is_empty() && spec.nullable => {
                        let v = if spec.kind == ColumnType::Text {
                            Value::Null
                        } else {
                            Value::Null
                        };
                        (Some(index), Some(v), None)
                    }
                    Some(cell) if cell.is_empty() && spec.kind != ColumnType::Text => {
                        (Some(index), None, Some(CellErrorKind::Empty))
                    }
                    Some(cell) => {
                        let parsed = match spec.kind {
                            ColumnType::Text => Ok(Value::Text(cell.clone())),
                            ColumnType::Integer => {
                                if !integer_syntax(cell) {
                                    Err(CellErrorKind::InvalidInteger)
                                } else {
                                    match cell.parse::<i64>() {
                                        Ok(v) => Ok(Value::Integer(v)),
                                        Err(_) => Err(CellErrorKind::IntegerOverflow),
                                    }
                                }
                            }
                            ColumnType::Decimal => {
                                if !decimal_syntax(cell) {
                                    Err(CellErrorKind::InvalidDecimal)
                                } else {
                                    match parse_decimal(cell) {
                                        Some(v) => Ok(Value::Decimal(v)),
                                        None => Err(CellErrorKind::DecimalOverflow),
                                    }
                                }
                            }
                            ColumnType::Boolean => match cell.as_str() {
                                "yes" | "1" => Ok(Value::Boolean(true)),
                                "no" | "0" => Ok(Value::Boolean(false)),
                                _ => Err(CellErrorKind::InvalidBoolean),
                            },
                        };
                        match parsed {
                            Ok(v) => (Some(index), Some(v), None),
                            Err(e) => (Some(index), None, Some(e)),
                        }
                    }
                },
            };

            output.push(value);
            if let Some(kind) = error {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column,
                    kind,
                });
                if policy != ErrorPolicy::Collect {
                    stopped_row = true;
                    if policy == ErrorPolicy::Stop {
                        for _ in spec_index + 1..columns.len() {
                            output.push(None);
                        }
                        result.rows.push(output);
                        return result;
                    }
                }
            }
        }
        result.rows.push(output);
    }
    result
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    let mut records = Vec::new();
    if i == bytes.len() {
        return Ok(records);
    }

    let mut record = Vec::new();
    let mut field = String::new();
    let mut field_started = false;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_start = 0usize;
    let mut record_index = 0usize;

    while i < bytes.len() {
        let b = bytes[i];
        let absolute = i;

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: absolute,
            });
        }

        if !in_quotes && b == b'\r' {
            if bytes.get(i + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: absolute,
                });
            }
        }

        if in_quotes {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                    continue;
                }
                in_quotes = false;
                after_quote = true;
                i += 1;
                continue;
            }
            field.push(b as char);
            i += 1;
            continue;
        }

        if after_quote {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                field_started = false;
                after_quote = false;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                field_started = false;
                after_quote = false;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: absolute,
            });
        }

        if b == delimiter {
            record.push(std::mem::take(&mut field));
            field_started = false;
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_index += 1;
            field_started = false;
            i += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: absolute,
                });
            }
            in_quotes = true;
            field_started = true;
            quote_start = absolute;
            i += 1;
        } else {
            field.push(b as char);
            field_started = true;
            i += 1;
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if !record.is_empty() || !field.is_empty() || field_started || after_quote {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        valid_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = build_table(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

fn build_table(input: &str, config: Config) -> Result<Table, CsvError> {
    valid_config(config)?;
    let parsed = parse_with(input, config.delimiter, config.quote)?;
    let (header, rows) = if config.headers && !parsed.is_empty() {
        (Some(parsed[0].clone()), parsed[1..].to_vec())
    } else {
        (None, parsed)
    };

    if let (Some(header), DuplicatePolicy::Reject) = (&header, config.duplicates) {
        let mut seen = HashMap::<&str, usize>::new();
        for (second, name) in header.iter().enumerate() {
            if let Some(&first) = seen.get(name.as_str()) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first,
                    second,
                });
            }
            seen.insert(name, second);
        }
    }

    if let Some(header) = &header {
        for (offset, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: offset + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

    pub spec: usize,

    FirstPerRow,

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        start + 1 == b.len()
    } else {
        (b'1'..=b'9').contains(&b[start])
            && b[start + 1..].iter().all(u8::is_ascii_digit)
    }
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    let dot = match b[start..].iter().position(|&x| x == b'.') {
        Some(p) => start + p,
        None => return false,
    };
    if b.len() != dot + 3 || !b[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    if dot == start {
        return false;
    }
    if b[start] == b'0' {
        dot == start + 1
    } else {
        (b'1'..=b'9').contains(&b[start])
            && b[start + 1..dot].iter().all(u8::is_ascii_digit)
    }
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    if s == "-0.00" {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let number = if negative { &s[1..] } else { s };
    let dot = number.find('.').unwrap();
    let whole = number[..dot]
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let frac = ((number.as_bytes()[dot + 1] - b'0') as i128) * 10
        + (number.as_bytes()[dot + 2] - b'0') as i128;
    let signed = whole * 100 + frac;
    let signed = if negative { -signed } else { signed };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate_cell(
    cell: &str,
    spec: &ColumnSpec,
) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_owned())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut out = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut out_row = vec![None; columns.len()];
        let mut failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (column, value) = match resolved {
                None => (None, Err(CellErrorKind::UnknownColumn)),
                Some(index) => match row.get(index) {
                    None => (Some(index), Err(CellErrorKind::MissingCell)),
                    Some(cell) => (
                        Some(index),
                        evaluate_cell(cell, spec),
                    ),
                },
            };

            match value {
                Ok(v) => out_row[spec_index] = Some(v),
                Err(kind) => {
                    out.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column,
                        kind,
                    });
                    failed = true;
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                }
            }
        }

        out.rows.push(out_row);
        if failed && policy == ErrorPolicy::Stop {
            break 'rows;
        }
    }

    out
}
