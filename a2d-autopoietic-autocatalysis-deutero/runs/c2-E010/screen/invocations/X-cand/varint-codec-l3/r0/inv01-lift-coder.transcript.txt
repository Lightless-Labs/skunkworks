#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
    LengthOutOfBounds { offset: usize, length: u64 },
    InvalidTag { offset: usize, tag: u64 },
    InvalidWire { offset: usize, wire: u8 },
    OutOfOrder { offset: usize, previous: u32, tag: u32 },

fn put_varint(mut value: u64, out: &mut Vec<u8>) {
    while value >= 0x80 {
        out.push((value as u8) | 0x80);
        value >>= 7;
    }
    out.push(value as u8);
}

fn read_varint(input: &[u8], mut pos: usize, end: usize) -> Result<(u64, usize), Error> {
    let start = pos;
    let mut value = 0u64;

    for i in 0..10 {
        if pos >= end {
            return Err(Error::Truncated { offset: end, start });
        }
        let byte = input[pos];
        pos += 1;
        let payload = (byte & 0x7f) as u64;

        if i == 9 {
            if payload > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow { offset: pos - 1, start });
            }
            value |= payload << 63;
            return Ok((value, pos));
        }

        value |= payload << (7 * i);
        if byte & 0x80 == 0 {
            return Ok((value, pos));
        }
    }

    unreachable!()
}

pub fn encode_message(message: &Message) -> Result<Vec<u8>, Error> {
    let mut body = Vec::new();
    let mut previous: Option<(u32, bool)> = None;

    for (index, field) in message.fields.iter().enumerate() {
        let (tag, is_bytes) = match field {
            Field::UInt { tag, .. } => (*tag, false),
            Field::Bytes { tag, .. } => (*tag, true),
        };

        if !(1..=15).contains(&tag) {
            return Err(Error::InvalidTag {
                offset: index,
                tag: tag as u64,
            });
        }

        if let Some((prev_tag, prev_bytes)) = previous {
            if tag < prev_tag {
                return Err(Error::OutOfOrder {
                    offset: index,
                    previous: prev_tag,
                    tag,
                });
            }
            if tag == prev_tag && !(prev_bytes && is_bytes) {
                return Err(Error::Duplicate { offset: index, tag });
            }
        }

        let wire = if is_bytes { 2u64 } else { 0u64 };
        put_varint(((tag as u64) << 3) | wire, &mut body);
        match field {
            Field::UInt { value, .. } => put_varint(*value, &mut body),
            Field::Bytes { data, .. } => {
                put_varint(data.len() as u64, &mut body);
                body.extend_from_slice(data);
            }
        }
        previous = Some((tag, is_bytes));
    }

    let mut output = Vec::new();
    put_varint(body.len() as u64, &mut output);
    output.extend_from_slice(&body);
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Message {
    pub fields: Vec<Field>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Field {
    UInt { tag: u32, value: u64 },
    Bytes { tag: u32, data: Vec<u8> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
    LengthOutOfBounds { offset: usize, length: u64 },
    InvalidTag { offset: usize, tag: u64 },
    InvalidWire { offset: usize, wire: u8 },
    OutOfOrder { offset: usize, previous: u32, tag: u32 },
    Duplicate { offset: usize, tag: u32 },
}

fn write_varint(mut n: u64, out: &mut Vec<u8>) {
    loop {
        let mut byte = (n & 0x7f) as u8;
        n >>= 7;
        if n != 0 {
            byte |= 0x80;
        }
        out.push(byte);
        if n == 0 {
            break;
        }
    }
}

fn parse_varint(input: &[u8], start: usize, end: usize) -> Result<(u64, usize), Error> {
    let mut cursor = start;
    let mut result = 0u64;

    for byte_index in 0..10usize {
        if cursor == end {
            return Err(Error::Truncated {
                offset: end,
                start,
            });
        }

        let byte = input[cursor];
        let offset = cursor;
        cursor += 1;
        let bits = (byte & 0x7f) as u64;

        if byte_index == 9 {
            if bits > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow { offset, start });
            }
            result |= bits << 63;
            return Ok((result, cursor));
        }

        result |= bits << (7 * byte_index);
        if byte & 0x80 == 0 {
            return Ok((result, cursor));
        }
    }

    unreachable!()
}

pub fn encode_message(message: &Message) -> Result<Vec<u8>, Error> {
    let mut body = Vec::new();
    let mut previous_tag = None;
    let mut previous_bytes = false;

    for (index, field) in message.fields.iter().enumerate() {
        let (tag, bytes_field) = match field {
            Field::UInt { tag, .. } => (*tag, false),
            Field::Bytes { tag, .. } => (*tag, true),
        };

        if tag == 0 || tag > 15 {
            return Err(Error::InvalidTag {
                offset: index,
                tag: u64::from(tag),
            });
        }

        if let Some(prev) = previous_tag {
            if tag < prev {
                return Err(Error::OutOfOrder {
                    offset: index,
                    previous: prev,
                    tag,
                });
            }
            if tag == prev && !(previous_bytes && bytes_field) {
                return Err(Error::Duplicate { offset: index, tag });
            }
        }

        match field {
            Field::UInt { value, .. } => {
                write_varint((u64::from(tag) << 3), &mut body);
                write_varint(*value, &mut body);
            }
            Field::Bytes { data, .. } => {
                write_varint((u64::from(tag) << 3) | 2, &mut body);
                write_varint(data.len() as u64, &mut body);
                body.extend_from_slice(data);
            }
        }

        previous_tag = Some(tag);
        previous_bytes = bytes_field;
    }

    let mut encoded = Vec::new();
    write_varint(body.len() as u64, &mut encoded);
    encoded.extend_from_slice(&body);
    Ok(encoded)
}

pub fn decode_message(input: &[u8], start: usize) -> Result<(Message, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let (frame_length, mut cursor) = parse_varint(input, start, input.len())?;
    let available = input.len() - cursor;
    if frame_length > available as u64 {
        return Err(Error::LengthOutOfBounds {
            offset: start,
            length: frame_length,
        });
    }

    let frame_end = cursor + frame_length as usize;
    let mut fields = Vec::<Field>::new();
    let mut previous: Option<(u32, u8)> = None;

    while cursor < frame_end {
        let key_start = cursor;
        let (key, after_key) = parse_varint(input, cursor, frame_end)?;
        cursor = after_key;

        let tag64 = key >> 3;
        let wire = (key & 7) as u8;

        if tag64 == 0 || tag64 > 255 {
            return Err(Error::InvalidTag {
                offset: key_start,
                tag: tag64,
            });
        }
        let tag = tag64 as u32;

        match wire {
            0 | 1 | 2 | 5 => {}
            _ => return Err(Error::InvalidWire { offset: key_start, wire }),
        }

        if let Some((prior_tag, prior_wire)) = previous {
            if tag < prior_tag {
                return Err(Error::OutOfOrder {
                    offset: key_start,
                    previous: prior_tag,
                    tag,
                });
            }
            if tag == prior_tag && (prior_wire != 2 || wire != 2) {
                return Err(Error::Duplicate {
                    offset: key_start,
                    tag,
                });
            }
        }

        match wire {
            0 => {
                let (value, next) = parse_varint(input, cursor, frame_end)?;
                cursor = next;
                if tag <= 15 {
                    fields.push(Field::UInt { tag, value });
                }
            }
            1 | 5 => {
                let width = if wire == 1 { 8usize } else { 4usize };
                if frame_end - cursor < width {
                    return Err(Error::LengthOutOfBounds {
                        offset: key_start,
                        length: width as u64,
                    });
                }
                cursor += width;
            }
            2 => {
                let length_start = cursor;
                let (length, next) = parse_varint(input, cursor, frame_end)?;
                cursor = next;

                if length > (frame_end - cursor) as u64 {
                    return Err(Error::LengthOutOfBounds {
                        offset: length_start,
                        length,
                    });
                }
                let size = length as usize;
                if tag <= 15 {
                    match fields.last_mut() {
                        Some(Field::Bytes {
                            tag: previous_tag,
                            data,
                        }) if *previous_tag == tag => {
                            data.extend_from_slice(&input[cursor..cursor + size]);
                        }
                        _ => fields.push(Field::Bytes {
                            tag,
                            data: input[cursor..cursor + size].to_vec(),
                        }),
                    }
                }
                cursor += size;
            }
            _ => unreachable!(),
        }

        previous = Some((tag, wire));
    }

    Ok((Message { fields }, frame_end))
}
