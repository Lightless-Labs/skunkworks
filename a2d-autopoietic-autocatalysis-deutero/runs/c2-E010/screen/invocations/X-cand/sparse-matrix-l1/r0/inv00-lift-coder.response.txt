- Dimension checks run first: rows before columns; 4096 is allowed.
- COO entries are validated in original order, checking row bounds, column bounds, then finiteness.
- Duplicate values are added in original order starting from positive zero; overflow reports the index of the addition that overflowed.
- Coordinates are removed only after all additions succeed, when the final sum compares equal to zero.
- Canonical entries are sorted by row, then column, and retain exact finite nonzero sums.
- The prompt’s two-file format conflicts with its single-file output rule; each solution below is a standalone source file.

### Solution A

```rust
use std::collections::BTreeMap;

pub const MAX_DIM: usize = 4096;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Axis {
    Row,
    Column,
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Entry {
    pub row: usize,
    pub col: usize,
    pub value: f64,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Sparse {
    rows: usize,
    cols: usize,
    entries: Vec<Entry>,
}

#[derive(Clone, Debug, PartialEq)]
pub enum Error {
    Shape {
        axis: Axis,
        size: usize,
    },
    Bounds {
        entry: usize,
        axis: Axis,
        index: usize,
        limit: usize,
    },
    NonFinite {
        entry: usize,
    },
    SumOverflow {
        entry: usize,
    },
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        if rows > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Row,
                size: rows,
            });
        }
        if cols > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Column,
                size: cols,
            });
        }

        for (i, e) in entries.iter().enumerate() {
            if e.row >= rows {
                return Err(Error::Bounds {
                    entry: i,
                    axis: Axis::Row,
                    index: e.row,
                    limit: rows,
                });
            }
            if e.col >= cols {
                return Err(Error::Bounds {
                    entry: i,
                    axis: Axis::Column,
                    index: e.col,
                    limit: cols,
                });
            }
            if !e.value.is_finite() {
                return Err(Error::NonFinite { entry: i });
            }
        }

        let mut sums: BTreeMap<(usize, usize), f64> = BTreeMap::new();

        for (i, e) in entries.iter().enumerate() {
            let sum = sums.entry((e.row, e.col)).or_insert(0.0);
            *sum = *sum + e.value;
            if !sum.is_finite() {
                return Err(Error::SumOverflow { entry: i });
            }
        }

        let canonical = sums
            .into_iter()
            .filter_map(|((row, col), value)| {
                if value == 0.0 {
                    None
                } else {
                    Some(Entry { row, col, value })
                }
            })
            .collect();

        Ok(Self {
            rows,
            cols,
            entries: canonical,
        })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn entries(&self) -> &[Entry] {
        &self.entries
    }
}
```

### Solution B

```rust
pub const MAX_DIM: usize = 4096;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Axis {
    Row,
    Column,
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Entry {
    pub row: usize,
    pub col: usize,
    pub value: f64,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Sparse {
    rows: usize,
    cols: usize,
    entries: Vec<Entry>,
}

#[derive(Clone, Debug, PartialEq)]
pub enum Error {
    Shape {
        axis: Axis,
        size: usize,
    },
    Bounds {
        entry: usize,
        axis: Axis,
        index: usize,
        limit: usize,
    },
    NonFinite {
        entry: usize,
    },
    SumOverflow {
        entry: usize,
    },
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        if rows > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Row,
                size: rows,
            });
        }
        if cols > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Column,
                size: cols,
            });
        }

        for (i, entry) in entries.iter().enumerate() {
            if entry.row >= rows {
                return Err(Error::Bounds {
                    entry: i,
                    axis: Axis::Row,
                    index: entry.row,
                    limit: rows,
                });
            }
            if entry.col >= cols {
                return Err(Error::Bounds {
                    entry: i,
                    axis: Axis::Column,
                    index: entry.col,
                    limit: cols,
                });
            }
            if !entry.value.is_finite() {
                return Err(Error::NonFinite { entry: i });
            }
        }

        let mut ordered: Vec<(usize, usize, usize, f64)> = Vec::with_capacity(entries.len());
        for (i, entry) in entries.iter().enumerate() {
            ordered.push((entry.row, entry.col, i, entry.value));
        }
        ordered.sort_by_key(|&(row, col, index, _)| (row, col, index));

        let mut result = Vec::new();
        let mut pos = 0;

        while pos < ordered.len() {
            let row = ordered[pos].0;
            let col = ordered[pos].1;
            let mut sum = 0.0;

            while pos < ordered.len() && ordered[pos].0 == row && ordered[pos].1 == col {
                let (entry_row, entry_col, original_index, value) = ordered[pos];
                debug_assert_eq!(entry_row, row);
                debug_assert_eq!(entry_col, col);
                sum = sum + value;
                if !sum.is_finite() {
                    return Err(Error::SumOverflow {
                        entry: original_index,
                    });
                }
                pos += 1;
            }

            if sum != 0.0 {
                result.push(Entry { row, col, value: sum });
            }
        }

        Ok(Self {
            rows,
            cols,
            entries: result,
        })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn entries(&self) -> &[Entry] {
        &self.entries
    }
}
```
