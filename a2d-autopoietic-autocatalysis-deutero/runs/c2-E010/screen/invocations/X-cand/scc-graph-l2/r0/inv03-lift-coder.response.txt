- Check limits in the specified order: vertices, edge count, then endpoints.
- Treat `n == MAX_VERTICES` and `edges.len() == MAX_EDGES` as valid boundaries.
- For `n == 0`, validate edges normally; the first source endpoint is invalid.
- A singleton is cyclic only if it has a self-loop; parallel edges count individually.
- Arc witnesses refer to original input positions, including positions of internal edges.
- Use iterative SCC traversal to support long paths without recursion-depth risk; counts stay within the input limits.

### Solution A

```rust
pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adj = vec![Vec::new(); n];
    let mut rev = vec![Vec::new(); n];
    for &(from, to) in edges {
        adj[from].push(to);
        rev[to].push(from);
    }

    let mut seen = vec![false; n];
    let mut order = Vec::with_capacity(n);
    for start in 0..n {
        if seen[start] {
            continue;
        }
        seen[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((v, next)) = stack.last_mut() {
            if *next < adj[*v].len() {
                let w = adj[*v][*next];
                *next += 1;
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                let (v, _) = stack.pop().unwrap();
                order.push(v);
            }
        }
    }

    let mut assigned = vec![false; n];
    let mut raw_components = Vec::new();
    for &start in order.iter().rev() {
        if assigned[start] {
            continue;
        }
        assigned[start] = true;
        let mut component = Vec::new();
        let mut stack = vec![start];
        while let Some(v) = stack.pop() {
            component.push(v);
            for &w in &rev[v] {
                if !assigned[w] {
                    assigned[w] = true;
                    stack.push(w);
                }
            }
        }
        component.sort_unstable();
        raw_components.push(component);
    }

    raw_components.sort_by(|a, b| b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0])));

    let mut component_of = vec![0usize; n];
    for (id, component) in raw_components.iter().enumerate() {
        for &v in component {
            component_of[v] = id;
        }
    }

    let count = raw_components.len();
    let mut cyclic: Vec<bool> = raw_components.iter().map(|c| c.len() > 1).collect();
    let mut internal_edges = vec![0usize; count];
    let mut arc_data: Vec<Vec<(usize, usize, usize)>> = vec![Vec::new(); count];

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let from_id = component_of[from];
        let to_id = component_of[to];
        if from_id == to_id {
            internal_edges[from_id] += 1;
            if from == to {
                cyclic[from_id] = true;
            }
        } else {
            arc_data[from_id].push((to_id, edge_index, 1));
        }
    }

    let mut arcs = Vec::new();
    for from in 0..count {
        let entries = &mut arc_data[from];
        entries.sort_unstable_by_key(|entry| entry.0);
        let mut i = 0;
        while i < entries.len() {
            let to = entries[i].0;
            let first_edge = entries[i].1;
            let mut multiplicity = 0usize;
            while i < entries.len() && entries[i].0 == to {
                multiplicity += entries[i].2;
                i += 1;
            }
            arcs.push(Arc {
                from,
                to,
                multiplicity,
                first_edge,
            });
        }
    }
    arcs.sort_unstable_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    Ok(Graph {
        components: raw_components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
    })
}
```

### Solution B

```rust
pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (edge, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut outgoing = vec![Vec::<usize>::new(); n];
    let mut incoming = vec![Vec::<usize>::new(); n];
    for &(u, v) in edges {
        outgoing[u].push(v);
        incoming[v].push(u);
    }

    // Iterative Tarjan SCC decomposition.
    let mut index = vec![usize::MAX; n];
    let mut low = vec![0usize; n];
    let mut on_stack = vec![false; n];
    let mut tarjan_stack = Vec::with_capacity(n);
    let mut raw = Vec::<Vec<usize>>::new();
    let mut next_index = 0usize;

    for root in 0..n {
        if index[root] != usize::MAX {
            continue;
        }
        index[root] = next_index;
        low[root] = next_index;
        next_index += 1;
        tarjan_stack.push(root);
        on_stack[root] = true;

        let mut frames = vec![(root, 0usize, None::<usize>)];
        while let Some((v, next_edge, parent)) = frames.last_mut() {
            if *next_edge < outgoing[*v].len() {
                let w = outgoing[*v][*next_edge];
                *next_edge += 1;
                if index[w] == usize::MAX {
                    index[w] = next_index;
                    low[w] = next_index;
                    next_index += 1;
                    tarjan_stack.push(w);
                    on_stack[w] = true;
                    frames.push((w, 0, Some(*v)));
                } else if on_stack[w] {
                    low[*v] = low[*v].min(index[w]);
                }
            } else {
                let (finished, _, parent) = frames.pop().unwrap();
                if let Some(p) = parent {
                    low[p] = low[p].min(low[finished]);
                }
                if low[finished] == index[finished] {
                    let mut component = Vec::new();
                    loop {
                        let w = tarjan_stack.pop().unwrap();
                        on_stack[w] = false;
                        component.push(w);
                        if w == finished {
                            break;
                        }
                    }
                    component.sort_unstable();
                    raw.push(component);
                }
            }
        }
    }

    raw.sort_by(|a, b| b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0])));

    let mut component_of = vec![0usize; n];
    for (id, component) in raw.iter().enumerate() {
        for &v in component {
            component_of[v] = id;
        }
    }

    let mut cyclic: Vec<bool> = raw.iter().map(|component| component.len() >= 2).collect();
    let mut internal_edges = vec![0usize; raw.len()];
    let mut pair_counts = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (edge_index, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = pair_counts.entry((a, b)).or_insert((0, edge_index));
            entry.0 += 1;
        }
    }

    let mut arcs: Vec<Arc> = pair_counts
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_unstable_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    Ok(Graph {
        components: raw,
        component_of,
        cyclic,
        internal_edges,
        arcs,
    })
}
```
