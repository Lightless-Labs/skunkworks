pub enum Orientation { Clockwise, Degenerate, CounterClockwise }
#[derive(Debug, Clone, PartialEq, Eq)]

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}
fn dist2(a: Point, b: Point) -> u128 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    (dx * dx + dy * dy) as u128
}
fn isqrt(n: u128) -> u128 {
    let (mut lo, mut hi) = (0u128, 1u128 << 42);
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid <= n / mid { lo = mid; } else { hi = mid; }
    }
    lo
}

    let n = dist2(a, b) * 1_000_000;
    let r = isqrt(n);
    let rounded = if n - r * r >= r + 1 { r + 1 } else { r };
    rounded as u64
}

    (0..v.len()).map(|i| {
        let a = v[i]; let b = v[(i + 1) % v.len()];

    let signed = shoelace(&v);
    let mut lens = Vec::with_capacity(v.len());

        edge_milli: lens, perimeter_milli, longest_edge,
    })
}
fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a,b,q) == 0 && q.x >= a.x.min(b.x) && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y) && q.y <= a.y.max(b.y)
}
pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let v = prepare(points)?;
    if queries.len() > CAP { return Err(GeomError::TooManyQueries { len: queries.len() }); }
    for (i, &q) in queries.iter().enumerate() {
        if !valid(q) { return Err(GeomError::QueryOutOfRange { index: i }); }
    }
    let mut out = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        for i in 0..v.len() {
            if on_segment(v[i], v[(i+1)%v.len()], q) { boundary = Some(i); break; }
        }
        if let Some(edge) = boundary { out.push(Location::Boundary { edge }); continue; }
        let mut odd = false;
        for i in 0..v.len() {
            let a = v[i]; let b = v[(i+1)%v.len()];
            if (a.y > q.y) != (b.y > q.y) {
                let c = cross(a,b,q);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) { odd = !odd; }
            }
        }
        out.push(if odd { Location::Inside } else { Location::Outside });
    }
    Ok(out)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > CAP { return Err(GeomError::TooManyVertices { len: points.len() }); }
    for (i, &p) in points.iter().enumerate() {
        if !valid(p) { return Err(GeomError::OutOfRange { index: i }); }
    }
    let mut pairs: Vec<(Point, usize)> = Vec::new();
    for (i, &p) in points.iter().enumerate() {
        if let Some(k) = pairs.iter().position(|x| x.0 == p) { pairs[k].1 = pairs[k].1.min(i); }
        else { pairs.push((p, i)); }
    }
    pairs.sort_by_key(|x| x.0);
    let m = pairs.len();
    let mut verts: Vec<usize> = if m <= 1 { (0..m).collect() }
    else {
        let all_col = (2..m).all(|i| cross(pairs[0].0,pairs[m-1].0,pairs[i].0)==0);
        if all_col {
            match mode { Collinear::Include => (0..m).collect(), Collinear::Exclude => vec![0,m-1] }
        } else {
            let mut lower: Vec<usize> = Vec::new();
            for i in 0..m {
                while lower.len() >= 2 {
                    let n=lower.len();
                    if cross(pairs[lower[n-2]].0,pairs[lower[n-1]].0,pairs[i].0) > 0 { break; }
                    lower.pop();
                }
                lower.push(i);
            }
            let mut upper: Vec<usize> = Vec::new();
            for i in (0..m).rev() {
                while upper.len() >= 2 {
                    let n=upper.len();
                    if cross(pairs[upper[n-2]].0,pairs[upper[n-1]].0,pairs[i].0) > 0 { break; }
                    upper.pop();
                }
                upper.push(i);
            }
            lower.pop(); upper.pop();
            let mut hull=lower; hull.extend(upper);
            if mode == Collinear::Include {
                let corners = hull.clone();
                let mut expanded=Vec::new();
                for e in 0..corners.len() {
                    let a=corners[e]; let b=corners[(e+1)%corners.len()];
                    expanded.push(a);
                    for k in 0..m {
                        if k!=a && k!=b && on_segment(pairs[a].0,pairs[b].0,pairs[k].0) {
                            expanded.push(k);
                        }
                    }
                    // sort the edge's inserted points by distance from its first endpoint
                    let start=expanded.iter().position(|&x| x==a).unwrap();
                    let end=expanded.len();
                    expanded[start+1..end].sort_by_key(|&k| dist2(pairs[a].0,pairs[k].0));
                }
                hull=expanded;
            }
            hull
        }
    };
    let vertices: Vec<Point> = verts.drain(..).map(|i| pairs[i].0).collect();
    let source_indices: Vec<usize> = verts.iter().map(|&i| pairs[i].1).collect();
    // verts was drained above; recover provenance from the emitted vertices.
    let source_indices: Vec<usize> = vertices.iter().map(|p| pairs.iter().find(|x| x.0==*p).unwrap().1).collect();
    let area2=hull_area(&vertices);
    let perimeter_milli=perimeter(&vertices);
    Ok(Hull { vertices, source_indices, area2, perimeter_milli })
}

pub fn classify(points:&[Point],queries:&[Point])->Result<Vec<Location>,GeomError>{
    let v=prepare(points)?;
    if queries.len()>CAP{return Err(GeomError::TooManyQueries{len:queries.len()});}
    for (i,&q) in queries.iter().enumerate(){if !valid(q){return Err(GeomError::QueryOutOfRange{index:i});}}
    Ok(queries.iter().map(|&q|{
        if let Some(i)=(0..v.len()).find(|&i|on_segment(v[i],v[(i+1)%v.len()],q)){return Location::Boundary{edge:i};}

        for i in 0..v.len(){let a=v[i];let b=v[(i+1)%v.len()];if (a.y>q.y)!=(b.y>q.y){let c=cross(a,b,q);if (b.y>a.y&&c>0)||(b.y<a.y&&c<0){inside=!inside;}}}
        if inside{Location::Inside}else{Location::Outside}
    }).collect())
}

    for (i,&p) in points.iter().enumerate(){if !valid(p){return Err(GeomError::OutOfRange{index:i});}}
    let mut entries:Vec<(Point,usize)>=Vec::new();

    }

    if n<=1{indices=(0..n).collect();}

        indices=if mode==Collinear::Include{(0..n).collect()}else{vec![0,n-1]};
    }else{
        let mut lo=Vec::new();let mut hi=Vec::new();
        for i in 0..n{while lo.len()>=2&&cross(entries[lo[lo.len()-2]].0,entries[lo[lo.len()-1]].0,entries[i].0)<=0{lo.pop();}lo.push(i);}
