#[derive(Debug, Clone, PartialEq, Eq)]

    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },

pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,

const COORD_LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;
const MAX_QUERIES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if points.len() >= 2 && points[0] == points[points.len() - 1] {
        points.len() - 1
    } else {
        points.len()
    };
    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }
    Ok(points[..effective_len].to_vec())
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let acx = c.x as i128 - a.x as i128;
    let acy = c.y as i128 - a.y as i128;
    abx * acy - aby * acx
}

fn shoelace(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    if points.len() < 2 {
        return sum;
    }
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn isqrt(n: u128) -> u128 {
    if n < 2 {
        return n;
    }
    let mut x = 1u128 << ((128 - n.leading_zeros() as usize + 1) / 2);
    loop {
        let y = (x + n / x) / 2;
        if y >= x {
            return x;
        }
        x = y;
    }
}

fn edge_milli(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let d2 = (dx * dx + dy * dy) as u128;
    let scaled = d2 * 1_000_000u128;
    let floor = isqrt(scaled);
    let lhs = 4 * scaled;
    let midpoint_rhs = 4 * floor * floor + 4 * floor + 1;
    let rounded = if lhs >= midpoint_rhs { floor + 1 } else { floor };
    rounded as u64
}

fn perimeter_milli(points: &[Point]) -> u64 {
    if points.is_empty() {
        return 0;
    }
    let mut sum = 0u64;
    for i in 0..points.len() {
        sum += edge_milli(points[i], points[(i + 1) % points.len()]);
    }
    sum
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepare_polygon(points)?;
    let signed_area2 = shoelace(&polygon);
    let area2 = signed_area2.unsigned_abs();
    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    let mut edge_lengths = Vec::with_capacity(polygon.len());
    let mut perimeter = 0u64;
    let mut longest_edge = None;
    for i in 0..polygon.len() {
        let a = polygon[i];
        let b = polygon[(i + 1) % polygon.len()];
        let length = edge_milli(a, b);
        edge_lengths.push(length);
        perimeter += length;

        match longest_edge {
            None => longest_edge = Some(i),
            Some(best) => {
                let best_len = edge_lengths[best];
                if length > best_len {
                    longest_edge = Some(i);
                } else if length == best_len {
                    let pair = (a.min(b), a.max(b));
                    let best_a = polygon[best];
                    let best_b = polygon[(best + 1) % polygon.len()];
                    let best_pair = (best_a.min(best_b), best_a.max(best_b));
                    if pair < best_pair || (pair == best_pair && i < best) {
                        longest_edge = Some(i);
                    }
                }
            }
        }
    }

    Ok(Measures {
        signed_area2,
        area2,
        orientation,
        edge_milli: edge_lengths,
        perimeter_milli: perimeter,
        longest_edge,
    })
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && (a.x.min(b.x) as i128) <= q.x as i128
        && (q.x as i128) <= (a.x.max(b.x) as i128)
        && (a.y.min(b.y) as i128) <= q.y as i128
        && (q.y as i128) <= (a.y.max(b.y) as i128)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > MAX_QUERIES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut result = Vec::with_capacity(queries.len());
    for &query in queries {
        let mut boundary = None;
        let mut inside = false;
        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if on_segment(a, b, query) {
                boundary = Some(i);
                break;
            }
            if (a.y > query.y) != (b.y > query.y) {
                let turn = cross(a, b, query);
                if (b.y > a.y && turn > 0) || (b.y < a.y && turn < 0) {
                    inside = !inside;
                }
            }
        }
        result.push(match boundary {
            Some(edge) => Location::Boundary { edge },
            None if inside => Location::Inside,
            None => Location::Outside,
        });
    }
    Ok(result)
}

fn cloud_validated(points: &[Point]) -> Result<Vec<(Point, usize)>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if let Some(existing) = unique.iter_mut().find(|(p, _)| *p == point) {
            existing.1 = existing.1.min(index);
        } else {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|(point, _)| *point);
    Ok(unique)
}

fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    let unique = cloud_validated(points)?;
    if unique.len() <= 1 {
        let vertices: Vec<Point> = unique.iter().map(|entry| entry.0).collect();
        let source_indices: Vec<usize> = unique.iter().map(|entry| entry.1).collect();
        return Ok(Hull {
            vertices,
            source_indices,
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let all_collinear = unique
        .iter()
        .skip(2)
        .all(|entry| cross(unique[0].0, unique[unique.len() - 1].0, entry.0) == 0);

    if all_collinear {
        let selected: Vec<(Point, usize)> = match mode {
            Collinear::Include => unique,
            Collinear::Exclude => vec![unique[0], unique[unique.len() - 1]],
        };
        let vertices: Vec<Point> = selected.iter().map(|entry| entry.0).collect();
        let source_indices: Vec<usize> = selected.iter().map(|entry| entry.1).collect();
        return Ok(Hull {
            vertices,
            source_indices,
            area2: 0,
            perimeter_milli: perimeter_milli(&vertices),
        });
    }

    let mut lower: Vec<(Point, usize)> = Vec::new();
    for &entry in &unique {
        while lower.len() >= 2 {
            let turn = cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, entry.0);
            if turn > 0 || (mode == Collinear::Include && turn == 0) {
                break;
            }
            lower.pop();
        }
        lower.push(entry);
    }

    let mut upper: Vec<(Point, usize)> = Vec::new();
    for &entry in unique.iter().rev() {
        while upper.len() >= 2 {
            let turn = cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, entry.0);
            if turn > 0 || (mode == Collinear::Include && turn == 0) {
                break;
            }
            upper.pop();
        }
        upper.push(entry);
    }

    lower.pop();
    upper.pop();
    lower.extend(upper);

    let vertices: Vec<Point> = lower.iter().map(|entry| entry.0).collect();
    let source_indices: Vec<usize> = lower.iter().map(|entry| entry.1).collect();
    let area2 = shoelace(&vertices).unsigned_abs();
    let perimeter_milli = perimeter_milli(&vertices);
    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {

    }

            return Err(GeomError::OutOfRange { index });
        }

        - usize::from(points.len() >= 2 && points[0] == points[points.len() - 1]);
    if n < 3 {
        return Err(GeomError::TooFewVertices { len: n });

    Ok(points[..n].to_vec())

fn cross(a: Point, b: Point, q: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (q.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (q.x as i128 - a.x as i128)
}

fn signed_area(points: &[Point]) -> i128 {
    points
        .iter()
        .enumerate()
        .map(|(i, a)| {
            let b = points[(i + 1) % points.len()];
            a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128
        })
        .sum()
}

fn integer_sqrt(n: u128) -> u128 {
    let mut lo = 0u128;
    let mut hi = 1u128 << 64;
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid <= n / mid.max(1) {
            lo = mid;
        } else {
            hi = mid;
        }
    }
    lo
}

fn rounded_length(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let scaled = (dx * dx + dy * dy) as u128 * 1_000_000;
    let floor = integer_sqrt(scaled);
    let midpoint = 4 * floor * floor + 4 * floor + 1;
    (floor + u128::from(4 * scaled >= midpoint)) as u64
}

fn total_perimeter(vertices: &[Point]) -> u64 {
    if vertices.is_empty() {
        return 0;
    }
    (0..vertices.len())
        .map(|i| rounded_length(vertices[i], vertices[(i + 1) % vertices.len()]))
        .sum()
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepare_polygon(points)?;
    let signed_area2 = signed_area(&polygon);
    let area2 = signed_area2.unsigned_abs();
    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    let mut edge_milli = Vec::with_capacity(polygon.len());
    for i in 0..polygon.len() {
        edge_milli.push(rounded_length(polygon[i], polygon[(i + 1) % polygon.len()]));
    }

    let perimeter_milli = edge_milli.iter().copied().sum();
    let longest_edge = (0..polygon.len()).max_by(|&i, &j| {
        edge_milli[i].cmp(&edge_milli[j]).then_with(|| {
            let pair_i = (polygon[i].min(polygon[(i + 1) % polygon.len()]),
                          polygon[i].max(polygon[(i + 1) % polygon.len()]));
            let pair_j = (polygon[j].min(polygon[(j + 1) % polygon.len()]),
                          polygon[j].max(polygon[(j + 1) % polygon.len()]));
            pair_j.cmp(&pair_i).then_with(|| j.cmp(&i))
        })
    });

    Ok(Measures {
        signed_area2,
        area2,
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > MAX_QUERIES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &q) in queries.iter().enumerate() {
        if !in_range(q) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut locations = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut first_boundary = None;
        let mut parity = false;
        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if on_segment(a, b, q) {
                if first_boundary.is_none() {
                    first_boundary = Some(i);
                }
                continue;
            }
            if (a.y > q.y) != (b.y > q.y) {
                let c = cross(a, b, q);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                    parity = !parity;
                }
            }
        }
        locations.push(if let Some(edge) = first_boundary {
            Location::Boundary { edge }
        } else if parity {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(locations)
}

fn make_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut sorted: Vec<(Point, usize)> = points.iter().copied().enumerate().map(|(i, p)| (p, i)).collect();
    sorted.sort_by_key(|&(p, i)| (p, i));
    sorted.dedup_by(|a, b| a.0 == b.0);

    let n = sorted.len();
    let mut hull: Vec<(Point, usize)> = Vec::new();

    if n >= 2 {
        let mut lower = Vec::new();
        for &point in &sorted {
            while lower.len() >= 2 {
                let c = cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, point.0);
                if c > 0 || (mode == Collinear::Include && c == 0) {
                    break;
                }
                lower.pop();
            }
            lower.push(point);
        }

        let mut upper = Vec::new();
        for &point in sorted.iter().rev() {
            while upper.len() >= 2 {
                let c = cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, point.0);
                if c > 0 || (mode == Collinear::Include && c == 0) {
                    break;
                }
                upper.pop();
            }
            upper.push(point);
        }
        lower.pop();
        upper.pop();
        lower.extend(upper);
        hull = lower;
    } else {
        hull = sorted;
    }

    let vertices: Vec<Point> = hull.iter().map(|&(p, _)| p).collect();
    let source_indices: Vec<usize> = hull.iter().map(|&(_, i)| i).collect();
    let area2 = if vertices.len() < 3 {
        0
    } else {
        signed_area(&vertices).unsigned_abs()
    };
    let perimeter_milli = total_perimeter(&vertices);

    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    make_hull(points, mode)
}
