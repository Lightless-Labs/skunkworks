    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

const LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-LIMIT..=LIMIT).contains(&p.x) && (-LIMIT..=LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (i, &p) in points.iter().enumerate() {
        if !in_range(p) {
            return Err(GeomError::OutOfRange { index: i });
        }
    }
    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective[effective.len() - 1] == effective[0] {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn cross_vec(ax: i128, ay: i128, bx: i128, by: i128) -> i128 {
    ax * by - ay * bx
}

fn cross_points(a: Point, b: Point, q: Point) -> i128 {
    cross_vec(
        b.x as i128 - a.x as i128,
        b.y as i128 - a.y as i128,
        q.x as i128 - a.x as i128,
        q.y as i128 - a.y as i128,
    )
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    (dx * dx + dy * dy) as u128
}

fn isqrt(n: u128) -> u128 {
    if n < 2 {
        return n;
    }
    let mut lo = 1u128;
    let mut hi = 1u128 << 64;
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid <= n / mid {
            lo = mid;
        } else {
            hi = mid;
        }
    }
    lo
}

fn edge_length_milli(a: Point, b: Point) -> u64 {
    let scaled = squared_distance(a, b) * 1_000_000u128;
    let root = isqrt(scaled);
    let remainder = scaled - root * root;
    let rounded = if remainder >= root + 1 { root + 1 } else { root };
    rounded as u64
}

fn polygon_measures(points: &[Point]) -> Measures {
    let n = points.len();
    let mut signed_area2 = 0i128;
    let mut edge_milli = Vec::with_capacity(n);
    let mut perimeter_milli = 0u64;
    let mut longest_edge = None;

    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;

        let length = edge_length_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let replace = match longest_edge {
            None => true,
            Some(j) => {
                if length != edge_milli[j] {
                    length > edge_milli[j]
                } else {
                    let old_a = points[j];
                    let old_b = points[(j + 1) % n];
                    let (new_low, new_high) = if a <= b { (a, b) } else { (b, a) };
                    let (old_low, old_high) = if old_a <= old_b {
                        (old_a, old_b)
                    } else {
                        (old_b, old_a)
                    };
                    (new_low, new_high, i) < (old_low, old_high, j)
                }
            }
        };
        if replace {
            longest_edge = Some(i);
        }
    }

    let area2 = signed_area2.unsigned_abs();
    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    Measures {
        signed_area2,
        area2,
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {

    Ok(polygon_measures(&effective))

    Inside,

fn point_on_segment(a: Point, b: Point, q: Point) -> bool {
    cross_points(a, b, q) == 0
        && (q.x as i128) >= (a.x.min(b.x) as i128)
        && (q.x as i128) <= (a.x.max(b.x) as i128)
        && (q.y as i128) >= (a.y.min(b.y) as i128)
        && (q.y as i128) <= (a.y.max(b.y) as i128)
}

fn classify_one(points: &[Point], q: Point) -> Location {
    let n = points.len();

    for i in 0..n {
        if point_on_segment(points[i], points[(i + 1) % n], q) {
            return Location::Boundary { edge: i };
        }
    }

    let mut odd = false;
    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        if (a.y > q.y) != (b.y > q.y) {
            let cross = cross_points(a, b, q);
            if (b.y > a.y && cross > 0) || (b.y < a.y && cross < 0) {
                odd = !odd;
            }
        }
    }
    if odd {
        Location::Inside
    } else {
        Location::Outside
    }
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let effective = prepare_polygon(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (i, &q) in queries.iter().enumerate() {
        if !in_range(q) {
            return Err(GeomError::QueryOutOfRange { index: i });
        }
    }
    Ok(queries
        .iter()
        .copied()
        .map(|q| classify_one(&effective, q))
        .collect())
}

    Include,

fn turn(a: Point, b: Point, c: Point) -> i128 {
    cross_points(a, b, c)
}

fn all_collinear(sorted: &[Point]) -> bool {
    sorted.len() < 3 || sorted[0..1]
        .iter()
        .all(|_| turn(sorted[0], sorted[sorted.len() - 1], sorted[1]) == 0)
}

fn monotone_chain(sorted: &[Point], include_collinear: bool) -> Vec<Point> {
    if sorted.len() <= 1 {
        return sorted.to_vec();
    }
    let mut lower = Vec::new();
    for &p in sorted {
        while lower.len() >= 2 {
            let t = turn(lower[lower.len() - 2], lower[lower.len() - 1], p);
            if if include_collinear { t < 0 } else { t <= 0 } {
                lower.pop();
            } else {
                break;
            }
        }
        lower.push(p);
    }

    let mut upper = Vec::new();
    for &p in sorted.iter().rev() {
        while upper.len() >= 2 {
            let t = turn(upper[upper.len() - 2], upper[upper.len() - 1], p);
            if if include_collinear { t < 0 } else { t <= 0 } {
                upper.pop();
            } else {
                break;
            }
        }
        upper.push(p);
    }

    lower.pop();
    upper.pop();
    lower.extend(upper);
    lower
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (i, &p) in points.iter().enumerate() {
        if !in_range(p) {
            return Err(GeomError::OutOfRange { index: i });
        }
    }

    let mut first_seen = std::collections::BTreeMap::<Point, usize>::new();
    for (i, &p) in points.iter().enumerate() {
        first_seen.entry(p).or_insert(i);
    }
    let sorted: Vec<Point> = first_seen.keys().copied().collect();

    let vertices = if sorted.len() <= 2 {
        sorted.clone()
    } else if all_collinear(&sorted) {
        match mode {
            Collinear::Include => sorted.clone(),
            Collinear::Exclude => vec![sorted[0], sorted[sorted.len() - 1]],
        }
    } else {
        monotone_chain(&sorted, mode == Collinear::Include)
    };

    let source_indices = vertices.iter().map(|p| first_seen[p]).collect();
    let n = vertices.len();
    let mut signed_area = 0i128;
    let mut perimeter_milli = 0u64;
    if n >= 2 {
        for i in 0..n {
            let a = vertices[i];
            let b = vertices[(i + 1) % n];
            signed_area += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
            perimeter_milli += edge_length_milli(a, b);
        }
    }
    Ok(Hull {
        vertices,
        source_indices,
        area2: signed_area.unsigned_abs(),
        perimeter_milli,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

const COORD_BOUND: i64 = 1_000_000_000;
const POINT_CAP: usize = 4096;

fn valid_point(p: Point) -> bool {
    p.x >= -COORD_BOUND && p.x <= COORD_BOUND && p.y >= -COORD_BOUND && p.y <= COORD_BOUND
}

fn polygon_input(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > POINT_CAP {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    let mut i = 0;
    while i < points.len() {
        if !valid_point(points[i]) {
            return Err(GeomError::OutOfRange { index: i });
        }
        i += 1;
    }
    let n = points.len();
    let effective_len = if n >= 2 && points[n - 1] == points[0] {
        n - 1
    } else {
        n
    };
    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }
    Ok(points[..effective_len].to_vec())
}

fn cross_coords(ax: i128, ay: i128, bx: i128, by: i128) -> i128 {
    ax * by - ay * bx
}

fn orient(a: Point, b: Point, c: Point) -> i128 {
    cross_coords(
        b.x as i128 - a.x as i128,
        b.y as i128 - a.y as i128,
        c.x as i128 - a.x as i128,
        c.y as i128 - a.y as i128,
    )
}

fn dist_squared(a: Point, b: Point) -> u128 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    (dx * dx + dy * dy) as u128
}

fn integer_sqrt(n: u128) -> u128 {
    let mut low = 0u128;
    let mut high = 1u128 << 64;
    while low + 1 < high {
        let mid = (low + high) / 2;
        if mid <= n / mid {
            low = mid;
        } else {
            high = mid;
        }
    }
    low
}

fn rounded_milli(a: Point, b: Point) -> u64 {
    let n = dist_squared(a, b) * 1_000_000;
    let r = integer_sqrt(n);
    let up = n - r * r >= 2 * r + 1;
    (r + u128::from(up)) as u64
}

        let a = points[i];
        let b = points[(i + 1) % n];
        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
        let len = rounded_milli(a, b);
        edge_milli.push(len);
        perimeter_milli += len;

        if let Some(j) = longest_edge {
            let chosen = edge_milli[j];
            let pair_i = if a <= b { (a, b) } else { (b, a) };
            let old_a = points[j];
            let old_b = points[(j + 1) % n];
            let pair_j = if old_a <= old_b {
                (old_a, old_b)
            } else {
                (old_b, old_a)
            };
            if len > chosen || (len == chosen && (pair_i < pair_j || (pair_i == pair_j && i < j))) {
                longest_edge = Some(i);
            }
        } else {
            longest_edge = Some(i);
        }
    }

    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    Ok(build_measures(&polygon_input(points)?))
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    orient(a, b, q) == 0
        && a.x.min(b.x) <= q.x
        && q.x <= a.x.max(b.x)
        && a.y.min(b.y) <= q.y
        && q.y <= a.y.max(b.y)
}

fn locate(points: &[Point], q: Point) -> Location {
    let mut boundary = None;
    let mut inside = false;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        if on_segment(a, b, q) {
            boundary = Some(i);
            break;
        }
        if (a.y > q.y) != (b.y > q.y) {
            let c = orient(a, b, q);
            if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                inside = !inside;
            }
        }
    }
    if let Some(edge) = boundary {
        Location::Boundary { edge }
    } else if inside {
        Location::Inside
    } else {
        Location::Outside
    }
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = polygon_input(points)?;
    if queries.len() > POINT_CAP {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (i, &q) in queries.iter().enumerate() {
        if !valid_point(q) {
            return Err(GeomError::QueryOutOfRange { index: i });
        }
    }
    Ok(queries.iter().copied().map(|q| locate(&polygon, q)).collect())
}

    Exclude,
    Include,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,

fn monotone_hull(sorted: &[Point], keep_collinear: bool) -> Vec<Point> {
    if sorted.len() < 2 {
        return sorted.to_vec();
    }

    let mut chain = Vec::with_capacity(sorted.len() * 2);
    for &p in sorted.iter().chain(sorted.iter().rev()) {
        let lower_phase = chain.len() <= sorted.len();
        while chain.len() >= 2 {
            let c = orient(chain[chain.len() - 2], chain[chain.len() - 1], p);
            let remove = if keep_collinear { c < 0 } else { c <= 0 };
            if !remove {
                break;
            }
            chain.pop();
        }
        chain.push(p);
        if lower_phase && chain.len() == sorted.len() {
            // The second sweep is recognized by the fixed chain length below.
        }
    }

    let mut lower = Vec::new();
    for &p in sorted {
        while lower.len() >= 2 {
            let c = orient(lower[lower.len() - 2], lower[lower.len() - 1], p);
            if if keep_collinear { c < 0 } else { c <= 0 } {
                lower.pop();
            } else {
                break;
            }
        }
        lower.push(p);
    }
    let mut upper = Vec::new();
    for &p in sorted.iter().rev() {
        while upper.len() >= 2 {
            let c = orient(upper[upper.len() - 2], upper[upper.len() - 1], p);
            if if keep_collinear { c < 0 } else { c <= 0 } {
                upper.pop();
            } else {
                break;
            }
        }
        upper.push(p);
    }
    lower.pop();
    upper.pop();
    lower.extend(upper);
    lower
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > POINT_CAP {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (i, &p) in points.iter().enumerate() {
        if !valid_point(p) {
            return Err(GeomError::OutOfRange { index: i });
        }
    }

    let mut entries = std::collections::BTreeMap::<Point, usize>::new();
    for (i, &p) in points.iter().enumerate() {
        entries.entry(p).or_insert(i);
    }
    let sorted: Vec<Point> = entries.keys().copied().collect();

    let collinear = if sorted.len() <= 2 {
        true
    } else {
        orient(sorted[0], sorted[sorted.len() - 1], sorted[1]) == 0
            && sorted
                .iter()
                .all(|&p| orient(sorted[0], sorted[sorted.len() - 1], p) == 0)
    };

    let vertices = if sorted.is_empty() {
        Vec::new()
    } else if collinear {
        match mode {
            Collinear::Include => sorted.clone(),
            Collinear::Exclude if sorted.len() == 1 => sorted.clone(),
            Collinear::Exclude => vec![sorted[0], sorted[sorted.len() - 1]],
        }
    } else {
        monotone_hull(&sorted, mode == Collinear::Include)
    };

    let source_indices = vertices.iter().map(|p| entries[p]).collect();
    let mut signed_area2 = 0i128;
    let mut perimeter_milli = 0u64;
    if vertices.len() >= 2 {
        for i in 0..vertices.len() {
            let a = vertices[i];
            let b = vertices[(i + 1) % vertices.len()];
            signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
            perimeter_milli += rounded_milli(a, b);
        }
    }
    Ok(Hull {
        vertices,
        source_indices,
        area2: signed_area2.unsigned_abs(),
        perimeter_milli,
    })
}
