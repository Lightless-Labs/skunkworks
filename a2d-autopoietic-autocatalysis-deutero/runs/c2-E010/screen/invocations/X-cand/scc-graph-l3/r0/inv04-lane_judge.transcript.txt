fn postorder(adj: &[Vec<usize>]) -> Vec<usize> {
    let mut seen = vec![false; adj.len()];
    let mut result = Vec::with_capacity(adj.len());
    for root in 0..adj.len() {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while !stack.is_empty() {
            let last = stack.len() - 1;
            let v = stack[last].0;
            if stack[last].1 < adj[v].len() {
                let w = adj[v][stack[last].1];
                stack[last].1 += 1;
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                result.push(v);
                stack.pop();
            }
        }
    }
    result
}

                side: 1,

impl Graph {
    fn shortest(&self, start: usize, goal: usize) -> Option<Vec<usize>> {
        let k = self.components.len();
        let mut rev = vec![Vec::<usize>::new(); k];
        let mut out = vec![Vec::<usize>::new(); k];
        for arc in &self.arcs {
            out[arc.from].push(arc.to);
            rev[arc.to].push(arc.from);
        }
        let mut dist = vec![usize::MAX; k];
        let mut queue = VecDeque::new();
        dist[goal] = 0;
        queue.push_back(goal);
        while let Some(v) = queue.pop_front() {
            for &p in &rev[v] {
                if dist[p] == usize::MAX {
                    dist[p] = dist[v] + 1;
                    queue.push_back(p);
                }
            }
        }
        if dist[start] == usize::MAX {
            return None;
        }
        for neighbors in &mut out {
            neighbors.sort_unstable();
            neighbors.dedup();
        }
        let mut route = vec![start];
        let mut at = start;
        while at != goal {
            let next = out[at]
                .iter()
                .copied()
                .find(|&w| dist[w] != usize::MAX && dist[w] + 1 == dist[at])?;
            route.push(next);
            at = next;
        }
        Some(route)
    }

    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }
        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            return Ok(Query {
                kind: if self.cyclic[a] {
                    Relation::Cycle
                } else {
                    Relation::Trivial
                },
                route: if self.cyclic[a] {
                    vec![a, a]
                } else {
                    vec![a]
                },
            });
        }
        if let Some(route) = self.shortest(a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = self.shortest(b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

fn finish_order(adj: &[Vec<usize>]) -> Vec<usize> {
    let n = adj.len();

    let mut order = Vec::with_capacity(n);
    for start in 0..n {
        if seen[start] {
            continue;

        queue.push_back(goal);
        while let Some(v) = queue.pop_front() {
            for arc in &self.arcs {
                if arc.to == v && distance[arc.from] == usize::MAX {
                    distance[arc.from] = distance[v] + 1;
                    queue.push_back(arc.from);
                }
            }
        }
        if distance[start] == usize::MAX {
            return None;
        }

        let mut route = vec![start];
        let mut current = start;
        while current != goal {
            let next = out[current]
                .iter()
                .copied()
                .filter(|&v| distance[v] + 1 == distance[current])
                .min()?;
            route.push(next);
            current = next;
        }
        Some(route)
    }

                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }

    let mut distance = vec![usize::MAX; adj.len()];
    let mut q = std::collections::VecDeque::new();

    q.push_back(start);
    while let Some(v) = q.pop_front() {
        for &w in &adj[v] {
            if distance[w] == usize::MAX {

                q.push_back(w);

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices { count: n, limit: MAX_VERTICES });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (edge, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint { edge, side: 0, vertex: from });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint { edge, side: 1, vertex: to });
        }
    }

    let mut outgoing = vec![Vec::<usize>::new(); n];
    let mut incoming = vec![Vec::<usize>::new(); n];
    for &(from, to) in edges {
        outgoing[from].push(to);
        incoming[to].push(from);
    }

    let mut visited = vec![false; n];
    let mut order = Vec::with_capacity(n);
    for root in 0..n {
        if visited[root] {
            continue;
        }
        visited[root] = true;
        let mut todo = vec![(root, 0usize)];
        while let Some((v, next_edge)) = todo.last_mut() {
            if *next_edge == outgoing[*v].len() {
                let v = todo.pop().unwrap().0;
                order.push(v);
                continue;
            }
            let w = outgoing[*v][*next_edge];
            *next_edge += 1;
            if !visited[w] {
                visited[w] = true;
                todo.push((w, 0));
            }
        }
    }

    visited.fill(false);
    let mut groups = Vec::<Vec<usize>>::new();
    for &root in order.iter().rev() {
        if visited[root] {
            continue;
        }
        visited[root] = true;
        let mut group = Vec::new();
        let mut todo = vec![root];
        while let Some(v) = todo.pop() {
            group.push(v);
            for &w in &incoming[v] {
                if !visited[w] {
                    visited[w] = true;
                    todo.push(w);
                }
            }
        }
        group.sort_unstable();
        groups.push(group);
    }
    groups.sort_by(|x, y| y.len().cmp(&x.len()).then_with(|| x[0].cmp(&y[0])));

    let mut component_of = vec![0; n];
    for (id, group) in groups.iter().enumerate() {
        for &v in group {
            component_of[v] = id;
        }
    }
    let mut cyclic: Vec<bool> = groups.iter().map(|group| group.len() > 1).collect();
    let mut internal_edges = vec![0usize; groups.len()];
    let mut pairs = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (i, &(from, to)) in edges.iter().enumerate() {
        let a = component_of[from];
        let b = component_of[to];
        if a == b {
            internal_edges[a] += 1;
            if from == to {
                cyclic[a] = true;
            }
        } else {
            let pair = pairs.entry((a, b)).or_insert((0, i));
            pair.0 += 1;
            pair.1 = pair.1.min(i);
        }
    }

    let mut arcs: Vec<Arc> = pairs
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|x, y| x.from.cmp(&y.from).then_with(|| y.to.cmp(&x.to)));

    let mut dag = vec![Vec::new(); groups.len()];
    let mut indegree = vec![0usize; groups.len()];
    for edge in &arcs {
        dag[edge.from].push(edge.to);
        indegree[edge.to] += 1;
    }

    let mut candidates = std::collections::BinaryHeap::<(usize, usize, usize)>::new();
    for id in 0..groups.len() {
        if indegree[id] == 0 {
            candidates.push((internal_edges[id], groups[id][0], id));
        }
    }
    let mut topo = Vec::with_capacity(groups.len());
    let mut depth = vec![0usize; groups.len()];
    while let Some((_, _, id)) = candidates.pop() {
        topo.push(id);
        for &next in &dag[id] {
            depth[next] = depth[next].max(depth[id] + 1);
            indegree[next] -= 1;
            if indegree[next] == 0 {
                candidates.push((internal_edges[next], groups[next][0], next));
            }
        }
    }

    Ok(Graph {
        components: groups,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

fn reachable_route(adj: &[Vec<usize>], start: usize, goal: usize) -> Option<Vec<usize>> {
    let n = adj.len();
    let mut dist = vec![usize::MAX; n];
    let mut queue = std::collections::VecDeque::new();
    dist[start] = 0;
    queue.push_back(start);

    while let Some(v) = queue.pop_front() {
        if v == goal {
            break;
        }
        let next_dist = dist[v] + 1;
        for &w in &adj[v] {
            if dist[w] == usize::MAX {
                dist[w] = next_dist;
                queue.push_back(w);
            }
        }
    }
    if dist[goal] == usize::MAX {
        return None;
    }

    let mut route = vec![goal];
    let mut cur = goal;
    while cur != start {
        let want = dist[cur] - 1;
        let mut predecessor = None;
        for (p, outs) in adj.iter().enumerate() {
            if dist[p] == want && outs.binary_search(&cur).is_ok() {
                if predecessor.map_or(true, |old| p < old) {
                    predecessor = Some(p);
                }
            }
        }
        cur = predecessor?;
        route.push(cur);
    }
    route.reverse();
    Some(route)
}

        cyclic[cid] = members.len() > 1;
    }

    let mut internal_edges = vec![0usize; c];
    let mut arc_data: std::collections::BTreeMap<(usize, usize), (usize, usize)> =
        std::collections::BTreeMap::new();
    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_data.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            entry.1 = entry.1.min(i);
        }
    }

    let mut arcs = Vec::with_capacity(arc_data.len());
    for ((from, to), (multiplicity, first_edge)) in arc_data {
        arcs.push(Arc { from, to, multiplicity, first_edge });
    }
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut cadj = vec![Vec::new(); c];
    let mut indegree = vec![0usize; c];
    for arc in &arcs {
        cadj[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    let mut ready = std::collections::BinaryHeap::new();
    for cid in 0..c {
        if indegree[cid] == 0 {
            ready.push((internal_edges[cid], components[cid][0], cid));
        }
    }

    let mut topo = Vec::with_capacity(c);
    let mut depth = vec![0usize; c];
    while let Some((_, _, v)) = ready.pop() {
        topo.push(v);
        for &w in &cadj[v] {
            depth[w] = depth[w].max(depth[v] + 1);
            indegree[w] -= 1;
            if indegree[w] == 0 {
                ready.push((internal_edges[w], components[w][0], w));
            }
        }
    }

    Ok(Graph { components, component_of, cyclic, internal_edges, arcs, topo, depth })
}

use std::cmp::Reverse;
use std::collections::{BinaryHeap, HashMap, VecDeque};

pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }
    for (i, &(u, v)) in edges.iter().enumerate() {
        if u >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: u,
            });
        }
        if v >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: v,
            });
        }
    }

    let mut adj = vec![Vec::new(); n];
    let mut rev = vec![Vec::new(); n];
    for &(u, v) in edges {
        adj[u].push(v);
        rev[v].push(u);
    }

    // Iterative first pass of Kosaraju.
    let mut seen = vec![false; n];
    let mut finish = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((v, next)) = stack.last_mut() {
            if *next < adj[*v].len() {
                let w = adj[*v][*next];
                *next += 1;
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                let (v, _) = stack.pop().unwrap();
                finish.push(v);
            }
        }
    }

    let mut assigned = vec![false; n];
    let mut raw = Vec::<Vec<usize>>::new();
    for &root in finish.iter().rev() {
        if assigned[root] {
            continue;
        }
        assigned[root] = true;
        let mut component = Vec::new();
        let mut stack = vec![root];
        while let Some(v) = stack.pop() {
            component.push(v);
            for &w in &rev[v] {
                if !assigned[w] {
                    assigned[w] = true;
                    stack.push(w);
                }
            }
        }
        component.sort_unstable();
        raw.push(component);
    }

    raw.sort_by(|a, b| b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0])));
    let components = raw;
    let mut component_of = vec![0; n];
    for (id, c) in components.iter().enumerate() {
        for &v in c {
            component_of[v] = id;
        }
    }

    let k = components.len();
    let mut cyclic = vec![false; k];
    for (id, c) in components.iter().enumerate() {
        cyclic[id] = c.len() > 1;
    }
    let mut internal_edges = vec![0usize; k];
    let mut arc_map = HashMap::<(usize, usize), (usize, usize)>::new();
    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_map.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            entry.1 = entry.1.min(i);
        }
    }

    let mut arcs = Vec::with_capacity(arc_map.len());
    for ((from, to), (multiplicity, first_edge)) in arc_map {
        arcs.push(Arc {
            from,
            to,
            multiplicity,
            first_edge,
        });
    }
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut out = vec![Vec::new(); k];
    let mut indegree = vec![0usize; k];
    for arc in &arcs {
        out[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }
    let mut ready = BinaryHeap::new();
    for id in 0..k {
        if indegree[id] == 0 {
            ready.push((internal_edges[id], components[id][0], Reverse(id)));
        }
    }
    let mut topo = Vec::with_capacity(k);
    let mut depth = vec![0usize; k];
    while let Some((_, _, Reverse(v))) = ready.pop() {
        topo.push(v);
        for &w in &out[v] {
            depth[w] = depth[w].max(depth[v] + 1);
            indegree[w] -= 1;
            if indegree[w] == 0 {
                ready.push((internal_edges[w], components[w][0], Reverse(w)));
            }
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }
        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        let adjacency = self.query_adjacency();
        if let Some(route) = shortest_lex_route(&adjacency, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_lex_route(&adjacency, b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

impl Graph {
    fn query_adjacency(&self) -> Vec<Vec<usize>> {
        let mut adjacency = vec![Vec::new(); self.components.len()];
        for arc in &self.arcs {
            adjacency[arc.from].push(arc.to);
        }
        for list in &mut adjacency {
            list.sort_unstable();
            list.dedup();
        }
        adjacency
    }
}

fn shortest_lex_route(adjacency: &[Vec<usize>], start: usize, goal: usize) -> Option<Vec<usize>> {
    let mut distance = vec![usize::MAX; adjacency.len()];
    let mut queue = VecDeque::new();
    distance[goal] = 0;
    queue.push_back(goal);
    while let Some(v) = queue.pop_front() {
        for (u, neighbors) in adjacency.iter().enumerate() {
            if neighbors.binary_search(&v).is_ok() && distance[u] == usize::MAX {
                distance[u] = distance[v] + 1;
                queue.push_back(u);
            }
        }
    }
    if distance[start] == usize::MAX {
        return None;
    }

    let mut route = vec![start];
    let mut current = start;
    while current != goal {
        let next = *adjacency[current]
            .iter()
            .filter(|&&v| distance[v] + 1 == distance[current])
            .min()?;
        route.push(next);
        current = next;
    }
    Some(route)
}

use std::collections::{BinaryHeap, HashMap, VecDeque};

pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }
    for (i, &(u, v)) in edges.iter().enumerate() {
        if u >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: u,
            });
        }
        if v >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: v,
            });
        }
    }

    let mut adj = vec![Vec::new(); n];
    for &(u, v) in edges {
        adj[u].push(v);
    }

    // Iterative Tarjan SCC.
    let mut index = vec![usize::MAX; n];
    let mut low = vec![0usize; n];
    let mut on_stack = vec![false; n];
    let mut tarjan_stack = Vec::new();
    let mut raw = Vec::<Vec<usize>>::new();
    let mut next_index = 0usize;

    for root in 0..n {
        if index[root] != usize::MAX {
            continue;
        }
        index[root] = next_index;
        low[root] = next_index;
        next_index += 1;
        tarjan_stack.push(root);
        on_stack[root] = true;
        let mut frames = vec![(root, 0usize)];

        while let Some((v, next)) = frames.last_mut() {
            if *next < adj[*v].len() {
                let w = adj[*v][*next];
                *next += 1;
                if index[w] == usize::MAX {
                    index[w] = next_index;
                    low[w] = next_index;
                    next_index += 1;
                    tarjan_stack.push(w);
                    on_stack[w] = true;
                    frames.push((w, 0));
                } else if on_stack[w] {
                    low[*v] = low[*v].min(index[w]);
                }
            } else {
                let (finished, _) = frames.pop().unwrap();
                if let Some(&(parent, _)) = frames.last() {
                    low[parent] = low[parent].min(low[finished]);
                }
                if low[finished] == index[finished] {
                    let mut component = Vec::new();
                    loop {
                        let w = tarjan_stack.pop().unwrap();
                        on_stack[w] = false;
                        component.push(w);
                        if w == finished {
                            break;
                        }
                    }
                    component.sort_unstable();
                    raw.push(component);
                }
            }
        }
    }

    raw.sort_by(|a, b| b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0])));
    let components = raw;
    let mut component_of = vec![0; n];
    for (id, c) in components.iter().enumerate() {
        for &v in c {
            component_of[v] = id;
        }
    }

    let k = components.len();
    let mut cyclic = vec![false; k];
    for (id, c) in components.iter().enumerate() {
        cyclic[id] = c.len() > 1;
    }
    let mut internal_edges = vec![0usize; k];
    let mut map = HashMap::<(usize, usize), (usize, usize)>::new();
    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = map.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            entry.1 = entry.1.min(i);
        }
    }

    let mut arcs = map
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect::<Vec<_>>();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut out = vec![Vec::new(); k];
    let mut indegree = vec![0usize; k];
    for arc in &arcs {
        out[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    let mut ready = BinaryHeap::new();
    for id in 0..k {
        if indegree[id] == 0 {
            ready.push((internal_edges[id], components[id][0], Reverse(id)));
        }
    }
    let mut topo = Vec::with_capacity(k);
    let mut depth = vec![0usize; k];
    while let Some((_, _, Reverse(v))) = ready.pop() {
        topo.push(v);
        for &w in &out[v] {
            depth[w] = depth[w].max(depth[v] + 1);
            indegree[w] -= 1;
            if indegree[w] == 0 {
                ready.push((internal_edges[w], components[w][0], Reverse(w)));
            }
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }
        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        let mut adjacency = vec![Vec::new(); self.components.len()];
        for arc in &self.arcs {
            adjacency[arc.from].push(arc.to);
        }
        for neighbors in &mut adjacency {
            neighbors.sort_unstable();
            neighbors.dedup();
        }

        if let Some(route) = shortest_lex_route(&adjacency, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_lex_route(&adjacency, b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

fn shortest_lex_route(adjacency: &[Vec<usize>], start: usize, goal: usize) -> Option<Vec<usize>> {
    // Search forward from start; sorted neighbors and FIFO order preserve
    // lexicographic order among paths at each distance.
    let mut parent = vec![usize::MAX; adjacency.len()];
    let mut queue = VecDeque::new();
    parent[start] = start;
    queue.push_back(start);
    while let Some(v) = queue.pop_front() {
        if v == goal {
            break;
        }
        for &w in &adjacency[v] {
            if parent[w] == usize::MAX {
                parent[w] = v;
                queue.push_back(w);
            }
        }
    }
    if parent[goal] == usize::MAX {
        return None;
    }
    let mut route = Vec::new();
    let mut v = goal;
    loop {
        route.push(v);
        if v == start {
            break;
        }
        v = parent[v];
    }
    route.reverse();
    Some(route)
}
