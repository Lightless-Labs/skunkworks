- Validation order is significant: vertex limit, edge limit, then endpoints in original order.
- Component IDs follow size-descending, minimum-vertex-ascending order; route tie-breaking uses those numeric IDs.
- A singleton is cyclic only if it has a self-loop; duplicate edges count toward internal totals and arc multiplicity.
- Topological ordering uses one live priority queue, with internal-edge count then smallest vertex as descending priorities.
- Backward queries first choose the lexicographically smallest shortest route in the forward reachable direction, then reverse it.
- Both implementations use checked input bounds and iterative graph traversals to support maximum-size paths without recursion.

### Solution A

```rust
pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

fn reachable_route(adj: &[Vec<usize>], start: usize, goal: usize) -> Option<Vec<usize>> {
    let n = adj.len();
    let mut dist = vec![usize::MAX; n];
    let mut queue = std::collections::VecDeque::new();
    dist[start] = 0;
    queue.push_back(start);

    while let Some(v) = queue.pop_front() {
        if v == goal {
            break;
        }
        let next_dist = dist[v] + 1;
        for &w in &adj[v] {
            if dist[w] == usize::MAX {
                dist[w] = next_dist;
                queue.push_back(w);
            }
        }
    }
    if dist[goal] == usize::MAX {
        return None;
    }

    let mut route = vec![goal];
    let mut cur = goal;
    while cur != start {
        let want = dist[cur] - 1;
        let mut predecessor = None;
        for (p, outs) in adj.iter().enumerate() {
            if dist[p] == want && outs.binary_search(&cur).is_ok() {
                if predecessor.map_or(true, |old| p < old) {
                    predecessor = Some(p);
                }
            }
        }
        cur = predecessor?;
        route.push(cur);
    }
    route.reverse();
    Some(route)
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery { side: 0, vertex: from });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery { side: 1, vertex: to });
        }
        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query { kind: Relation::Cycle, route: vec![a, a] });
            }
            return Ok(Query { kind: Relation::Trivial, route: vec![a] });
        }

        let mut adj = vec![Vec::new(); self.components.len()];
        for arc in &self.arcs {
            adj[arc.from].push(arc.to);
        }
        for outs in &mut adj {
            outs.sort_unstable();
            outs.dedup();
        }

        if let Some(route) = reachable_route(&adj, a, b) {
            return Ok(Query { kind: Relation::Forward, route });
        }
        if let Some(mut route) = reachable_route(&adj, b, a) {
            route.reverse();
            return Ok(Query { kind: Relation::Backward, route });
        }
        Ok(Query { kind: Relation::Separate, route: Vec::new() })
    }
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices { count: n, limit: MAX_VERTICES });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(u, v)) in edges.iter().enumerate() {
        if u >= n {
            return Err(GraphError::BadEndpoint { edge: i, side: 0, vertex: u });
        }
        if v >= n {
            return Err(GraphError::BadEndpoint { edge: i, side: 1, vertex: v });
        }
    }

    let mut adj = vec![Vec::new(); n];
    let mut rev = vec![Vec::new(); n];
    for &(u, v) in edges {
        adj[u].push(v);
        rev[v].push(u);
    }

    let mut seen = vec![false; n];
    let mut finish = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((v, idx)) = stack.last_mut() {
            if *idx < adj[*v].len() {
                let w = adj[*v][*idx];
                *idx += 1;
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                let (done, _) = stack.pop().unwrap();
                finish.push(done);
            }
        }
    }

    seen.fill(false);
    let mut raw_components = Vec::new();
    for &root in finish.iter().rev() {
        if seen[root] {
            continue;
        }
        let mut members = Vec::new();
        let mut stack = vec![root];
        seen[root] = true;
        while let Some(v) = stack.pop() {
            members.push(v);
            for &w in &rev[v] {
                if !seen[w] {
                    seen[w] = true;
                    stack.push(w);
                }
            }
        }
        members.sort_unstable();
        raw_components.push(members);
    }

    raw_components.sort_by(|a, b| b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0])));
    let components = raw_components;
    let mut component_of = vec![0usize; n];
    for (cid, members) in components.iter().enumerate() {
        for &v in members {
            component_of[v] = cid;
        }
    }

    let c = components.len();
    let mut cyclic = vec![false; c];
    for (cid, members) in components.iter().enumerate() {
        cyclic[cid] = members.len() > 1;
    }

    let mut internal_edges = vec![0usize; c];
    let mut arc_data: std::collections::BTreeMap<(usize, usize), (usize, usize)> =
        std::collections::BTreeMap::new();
    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_data.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            entry.1 = entry.1.min(i);
        }
    }

    let mut arcs = Vec::with_capacity(arc_data.len());
    for ((from, to), (multiplicity, first_edge)) in arc_data {
        arcs.push(Arc { from, to, multiplicity, first_edge });
    }
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut cadj = vec![Vec::new(); c];
    let mut indegree = vec![0usize; c];
    for arc in &arcs {
        cadj[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    let mut ready = std::collections::BinaryHeap::new();
    for cid in 0..c {
        if indegree[cid] == 0 {
            ready.push((internal_edges[cid], components[cid][0], cid));
        }
    }

    let mut topo = Vec::with_capacity(c);
    let mut depth = vec![0usize; c];
    while let Some((_, _, v)) = ready.pop() {
        topo.push(v);
        for &w in &cadj[v] {
            depth[w] = depth[w].max(depth[v] + 1);
            indegree[w] -= 1;
            if indegree[w] == 0 {
                ready.push((internal_edges[w], components[w][0], w));
            }
        }
    }

    Ok(Graph { components, component_of, cyclic, internal_edges, arcs, topo, depth })
}
```

```rust
pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

fn query_path(adj: &[Vec<usize>], start: usize, goal: usize) -> Option<Vec<usize>> {
    let mut distance = vec![usize::MAX; adj.len()];
    let mut q = std::collections::VecDeque::new();
    distance[start] = 0;
    q.push_back(start);
    while let Some(v) = q.pop_front() {
        for &w in &adj[v] {
            if distance[w] == usize::MAX {
                distance[w] = distance[v] + 1;
                q.push_back(w);
            }
        }
    }
    if distance[goal] == usize::MAX {
        return None;
    }

    let mut path = vec![start];
    let mut at = start;
    while at != goal {
        let next_distance = distance[at] + 1;
        let next = adj[at]
            .iter()
            .copied()
            .filter(|&w| distance[w] == next_distance)
            .min()?;
        path.push(next);
        at = next;
    }
    Some(path)
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery { side: 0, vertex: from });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery { side: 1, vertex: to });
        }
        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            let (kind, route) = if self.cyclic[a] {
                (Relation::Cycle, vec![a, a])
            } else {
                (Relation::Trivial, vec![a])
            };
            return Ok(Query { kind, route });
        }

        let mut adj = vec![Vec::new(); self.components.len()];
        for arc in &self.arcs {
            adj[arc.from].push(arc.to);
        }
        for v in &mut adj {
            v.sort_unstable();
            v.dedup();
        }

        if let Some(route) = query_path(&adj, a, b) {
            return Ok(Query { kind: Relation::Forward, route });
        }
        if let Some(mut route) = query_path(&adj, b, a) {
            route.reverse();
            return Ok(Query { kind: Relation::Backward, route });
        }
        Ok(Query { kind: Relation::Separate, route: Vec::new() })
    }
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices { count: n, limit: MAX_VERTICES });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (edge, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint { edge, side: 0, vertex: from });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint { edge, side: 1, vertex: to });
        }
    }

    let mut outgoing = vec![Vec::<usize>::new(); n];
    let mut incoming = vec![Vec::<usize>::new(); n];
    for &(from, to) in edges {
        outgoing[from].push(to);
        incoming[to].push(from);
    }

    let mut visited = vec![false; n];
    let mut order = Vec::with_capacity(n);
    for root in 0..n {
        if visited[root] {
            continue;
        }
        visited[root] = true;
        let mut todo = vec![(root, 0usize)];
        while let Some((v, next_edge)) = todo.last_mut() {
            if *next_edge == outgoing[*v].len() {
                let v = todo.pop().unwrap().0;
                order.push(v);
                continue;
            }
            let w = outgoing[*v][*next_edge];
            *next_edge += 1;
            if !visited[w] {
                visited[w] = true;
                todo.push((w, 0));
            }
        }
    }

    visited.fill(false);
    let mut groups = Vec::<Vec<usize>>::new();
    for &root in order.iter().rev() {
        if visited[root] {
            continue;
        }
        visited[root] = true;
        let mut group = Vec::new();
        let mut todo = vec![root];
        while let Some(v) = todo.pop() {
            group.push(v);
            for &w in &incoming[v] {
                if !visited[w] {
                    visited[w] = true;
                    todo.push(w);
                }
            }
        }
        group.sort_unstable();
        groups.push(group);
    }
    groups.sort_by(|x, y| y.len().cmp(&x.len()).then_with(|| x[0].cmp(&y[0])));

    let mut component_of = vec![0; n];
    for (id, group) in groups.iter().enumerate() {
        for &v in group {
            component_of[v] = id;
        }
    }
    let mut cyclic: Vec<bool> = groups.iter().map(|group| group.len() > 1).collect();
    let mut internal_edges = vec![0usize; groups.len()];
    let mut pairs = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (i, &(from, to)) in edges.iter().enumerate() {
        let a = component_of[from];
        let b = component_of[to];
        if a == b {
            internal_edges[a] += 1;
            if from == to {
                cyclic[a] = true;
            }
        } else {
            let pair = pairs.entry((a, b)).or_insert((0, i));
            pair.0 += 1;
            pair.1 = pair.1.min(i);
        }
    }

    let mut arcs: Vec<Arc> = pairs
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|x, y| x.from.cmp(&y.from).then_with(|| y.to.cmp(&x.to)));

    let mut dag = vec![Vec::new(); groups.len()];
    let mut indegree = vec![0usize; groups.len()];
    for edge in &arcs {
        dag[edge.from].push(edge.to);
        indegree[edge.to] += 1;
    }

    let mut candidates = std::collections::BinaryHeap::<(usize, usize, usize)>::new();
    for id in 0..groups.len() {
        if indegree[id] == 0 {
            candidates.push((internal_edges[id], groups[id][0], id));
        }
    }
    let mut topo = Vec::with_capacity(groups.len());
    let mut depth = vec![0usize; groups.len()];
    while let Some((_, _, id)) = candidates.pop() {
        topo.push(id);
        for &next in &dag[id] {
            depth[next] = depth[next].max(depth[id] + 1);
            indegree[next] -= 1;
            if indegree[next] == 0 {
                candidates.push((internal_edges[next], groups[next][0], next));
            }
        }
    }

    Ok(Graph {
        components: groups,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}
```
