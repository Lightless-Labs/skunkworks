- Validation order is vertex count, edge count, then endpoints in input order, checking each source before its target.
- Component IDs follow the specified size and minimum-vertex ordering; arc witness indices refer to the original edge slice.
- Topological ties use the greatest internal-edge count, then greatest minimum vertex, with newly ready components immediately eligible.
- Shortest routes use lexicographically smallest component-ID sequences in the reachable direction; backward routes are reversed afterward.
- Limits fit `usize` counts without overflow for valid inputs; BFS-based queries keep memory bounded and handle long paths.
- There is no meaningful alternate reading of these rules, so Solution B is an independent SCC implementation.

### Solution A

```rust
pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

fn finish_order(adj: &[Vec<usize>]) -> Vec<usize> {
    let n = adj.len();
    let mut seen = vec![false; n];
    let mut order = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((v, next)) = stack.last_mut() {
            if *next < adj[*v].len() {
                let w = adj[*v][*next];
                *next += 1;
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                let (v, _) = stack.pop().unwrap();
                order.push(v);
            }
        }
    }
    order
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices { count: n, limit: MAX_VERTICES });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(u, v)) in edges.iter().enumerate() {
        if u >= n {
            return Err(GraphError::BadEndpoint { edge: i, side: 0, vertex: u });
        }
        if v >= n {
            return Err(GraphError::BadEndpoint { edge: i, side: 1, vertex: v });
        }
    }

    let mut adj = vec![Vec::new(); n];
    let mut rev = vec![Vec::new(); n];
    for &(u, v) in edges {
        adj[u].push(v);
        rev[v].push(u);
    }

    let order = finish_order(&adj);
    let mut assigned = vec![false; n];
    let mut raw = Vec::<Vec<usize>>::new();
    for &root in order.iter().rev() {
        if assigned[root] {
            continue;
        }
        assigned[root] = true;
        let mut comp = Vec::new();
        let mut stack = vec![root];
        while let Some(v) = stack.pop() {
            comp.push(v);
            for &w in &rev[v] {
                if !assigned[w] {
                    assigned[w] = true;
                    stack.push(w);
                }
            }
        }
        comp.sort_unstable();
        raw.push(comp);
    }
    raw.sort_by(|a, b| b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0])));

    let components = raw;
    let mut component_of = vec![0; n];
    for (cid, comp) in components.iter().enumerate() {
        for &v in comp {
            component_of[v] = cid;
        }
    }

    let k = components.len();
    let mut cyclic = vec![false; k];
    let mut internal_edges = vec![0usize; k];
    let mut pair_data = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();
    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = pair_data.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            if i < entry.1 {
                entry.1 = i;
            }
        }
    }
    for (cid, comp) in components.iter().enumerate() {
        if comp.len() >= 2 {
            cyclic[cid] = true;
        }
    }

    let mut arcs: Vec<Arc> = pair_data
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc { from, to, multiplicity, first_edge })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::new(); k];
    let mut indegree = vec![0usize; k];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }
    let mut ready = std::collections::BTreeSet::<(usize, usize, usize)>::new();
    for cid in 0..k {
        if indegree[cid] == 0 {
            ready.insert((internal_edges[cid], components[cid][0], cid));
        }
    }
    let mut topo = Vec::with_capacity(k);
    let mut depth = vec![0usize; k];
    while let Some(item) = ready.iter().next_back().copied() {
        ready.remove(&item);
        let cid = item.2;
        topo.push(cid);
        for &to in &outgoing[cid] {
            depth[to] = depth[to].max(depth[cid] + 1);
            indegree[to] -= 1;
            if indegree[to] == 0 {
                ready.insert((internal_edges[to], components[to][0], to));
            }
        }
    }

    Ok(Graph { components, component_of, cyclic, internal_edges, arcs, topo, depth })
}

impl Graph {
    fn shortest(&self, start: usize, goal: usize) -> Option<Vec<usize>> {
        let k = self.components.len();
        let mut outgoing = vec![Vec::new(); k];
        for arc in &self.arcs {
            outgoing[arc.from].push(arc.to);
        }
        for v in &mut outgoing {
            v.sort_unstable();
            v.dedup();
        }
        let mut dist = vec![usize::MAX; k];
        let mut parent = vec![usize::MAX; k];
        let mut queue = std::collections::VecDeque::new();
        dist[start] = 0;
        queue.push_back(start);
        while let Some(v) = queue.pop_front() {
            for &w in &outgoing[v] {
                if dist[w] == usize::MAX {
                    dist[w] = dist[v] + 1;
                    parent[w] = v;
                    queue.push_back(w);
                }
            }
        }
        if dist[goal] == usize::MAX {
            return None;
        }
        let mut route = vec![goal];
        let mut v = goal;
        while v != start {
            v = parent[v];
            route.push(v);
        }
        route.reverse();
        Some(route)
    }

    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery { side: 0, vertex: from });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery { side: 1, vertex: to });
        }
        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query { kind: Relation::Cycle, route: vec![a, a] });
            }
            return Ok(Query { kind: Relation::Trivial, route: vec![a] });
        }
        if let Some(route) = self.shortest(a, b) {
            return Ok(Query { kind: Relation::Forward, route });
        }
        if let Some(mut route) = self.shortest(b, a) {
            route.reverse();
            return Ok(Query { kind: Relation::Backward, route });
        }
        Ok(Query { kind: Relation::Separate, route: Vec::new() })
    }
}
```

```rust
pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices { count: n, limit: MAX_VERTICES });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(u, v)) in edges.iter().enumerate() {
        if u >= n {
            return Err(GraphError::BadEndpoint { edge: i, side: 0, vertex: u });
        }
        if v >= n {
            return Err(GraphError::BadEndpoint { edge: i, side: 1, vertex: v });
        }
    }

    let mut adj = vec![Vec::<usize>::new(); n];
    let mut rev = vec![Vec::<usize>::new(); n];
    for &(u, v) in edges {
        adj[u].push(v);
        rev[v].push(u);
    }

    // Iterative Tarjan SCC, using explicit DFS frames.
    let mut index = vec![usize::MAX; n];
    let mut low = vec![0usize; n];
    let mut on_stack = vec![false; n];
    let mut tarjan_stack = Vec::new();
    let mut raw = Vec::<Vec<usize>>::new();
    let mut next_index = 0usize;

    for root in 0..n {
        if index[root] != usize::MAX {
            continue;
        }
        index[root] = next_index;
        low[root] = next_index;
        next_index += 1;
        tarjan_stack.push(root);
        on_stack[root] = true;
        let mut frames = vec![(root, 0usize, None::<usize>)];

        while let Some((v, next_edge, parent)) = frames.last_mut() {
            if *next_edge < adj[*v].len() {
                let w = adj[*v][*next_edge];
                *next_edge += 1;
                if index[w] == usize::MAX {
                    index[w] = next_index;
                    low[w] = next_index;
                    next_index += 1;
                    tarjan_stack.push(w);
                    on_stack[w] = true;
                    frames.push((w, 0usize, Some(*v)));
                } else if on_stack[w] {
                    low[*v] = low[*v].min(index[w]);
                }
            } else {
                let (done, _, parent) = frames.pop().unwrap();
                if let Some(p) = parent {
                    low[p] = low[p].min(low[done]);
                }
                if low[done] == index[done] {
                    let mut comp = Vec::new();
                    loop {
                        let w = tarjan_stack.pop().unwrap();
                        on_stack[w] = false;
                        comp.push(w);
                        if w == done {
                            break;
                        }
                    }
                    comp.sort_unstable();
                    raw.push(comp);
                }
            }
        }
    }

    raw.sort_by(|a, b| b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0])));
    let components = raw;
    let mut component_of = vec![0usize; n];
    for (cid, comp) in components.iter().enumerate() {
        for &v in comp {
            component_of[v] = cid;
        }
    }

    let k = components.len();
    let mut cyclic = components.iter().map(|c| c.len() >= 2).collect::<Vec<_>>();
    let mut internal_edges = vec![0usize; k];
    let mut pairs = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();
    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = pairs.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            entry.1 = entry.1.min(i);
        }
    }
    let mut arcs = pairs.into_iter().map(|((from, to), (multiplicity, first_edge))| {
        Arc { from, to, multiplicity, first_edge }
    }).collect::<Vec<_>>();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::new(); k];
    let mut indegree = vec![0usize; k];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }
    let mut ready = std::collections::BTreeSet::<(usize, usize, usize)>::new();
    for cid in 0..k {
        if indegree[cid] == 0 {
            ready.insert((internal_edges[cid], components[cid][0], cid));
        }
    }
    let mut topo = Vec::with_capacity(k);
    let mut depth = vec![0usize; k];
    while let Some(entry) = ready.iter().next_back().copied() {
        ready.remove(&entry);
        let cid = entry.2;
        topo.push(cid);
        for &to in &outgoing[cid] {
            depth[to] = depth[to].max(depth[cid] + 1);
            indegree[to] -= 1;
            if indegree[to] == 0 {
                ready.insert((internal_edges[to], components[to][0], to));
            }
        }
    }

    Ok(Graph { components, component_of, cyclic, internal_edges, arcs, topo, depth })
}

impl Graph {
    fn shortest(&self, start: usize, goal: usize) -> Option<Vec<usize>> {
        let k = self.components.len();
        let mut outgoing = vec![Vec::<usize>::new(); k];
        for arc in &self.arcs {
            outgoing[arc.from].push(arc.to);
        }
        for list in &mut outgoing {
            list.sort_unstable();
        }

        // A FIFO BFS with ascending neighbors discovers lexicographically
        // smallest paths among shortest paths.
        let mut distance = vec![usize::MAX; k];
        let mut parent = vec![usize::MAX; k];
        let mut queue = std::collections::VecDeque::new();
        distance[start] = 0;
        queue.push_back(start);
        while let Some(v) = queue.pop_front() {
            for &w in &outgoing[v] {
                if distance[w] == usize::MAX {
                    distance[w] = distance[v] + 1;
                    parent[w] = v;
                    queue.push_back(w);
                }
            }
        }
        if distance[goal] == usize::MAX {
            return None;
        }
        let mut path = vec![goal];
        let mut at = goal;
        while at != start {
            at = parent[at];
            path.push(at);
        }
        path.reverse();
        Some(path)
    }

    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery { side: 0, vertex: from });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery { side: 1, vertex: to });
        }
        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            return Ok(if self.cyclic[a] {
                Query { kind: Relation::Cycle, route: vec![a, a] }
            } else {
                Query { kind: Relation::Trivial, route: vec![a] }
            });
        }
        if let Some(route) = self.shortest(a, b) {
            return Ok(Query { kind: Relation::Forward, route });
        }
        if let Some(mut route) = self.shortest(b, a) {
            route.reverse();
            return Ok(Query { kind: Relation::Backward, route });
        }
        Ok(Query { kind: Relation::Separate, route: Vec::new() })
    }
}
```
