fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();

        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,

            } else if v.minor > 0 {
                version.major == 0 && version.minor == v.minor

        }
    }

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let by_name: BTreeMap<&str, &Package> = catalog
        .iter()
        .map(|package| (package.name.as_str(), package))
        .collect();

    let mut active: BTreeMap<String, Vec<Constraint>> = BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut selected: BTreeMap<String, Version> = BTreeMap::new();

    loop {
        for package in active.keys() {
            if !by_name.contains_key(package.as_str()) {
                return Err(ResolveError::Missing {
                    package: package.clone(),
                });
            }
        }

        for (package, version) in &selected {
            let constraints = canonical(active.get(package).cloned().unwrap_or_default());
            if !constraints.iter().all(|c| permits(*c, *version)) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: constraints,
                });
            }
        }

        for (package, constraints) in &active {
            if selected.contains_key(package) {
                continue;
            }
            let canonical_constraints = canonical(constraints.clone());
            let catalog_package = by_name[package.as_str()];
            let has_match = catalog_package.releases.iter().any(|release| {
                canonical_constraints
                    .iter()
                    .all(|constraint| permits(*constraint, release.version))
            });
            if !has_match {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: canonical_constraints,
                });
            }
        }

        let next = active
            .keys()
            .find(|package| !selected.contains_key(*package))
            .cloned();
        let Some(package_name) = next else {
            return Ok(selected
                .into_iter()
                .map(|(package, version)| Selection { package, version })
                .collect());
        };

        let constraints = canonical(active[&package_name].clone());
        let package = by_name[package_name.as_str()];
        let version = package
            .releases
            .iter()
            .filter(|release| {
                constraints
                    .iter()
                    .all(|constraint| permits(*constraint, release.version))
            })
            .map(|release| release.version)
            .max()
            .expect("candidate existence was checked");

        selected.insert(package_name.clone(), version);
        let release = package
            .releases
            .iter()
            .find(|release| release.version == version)
            .expect("validated unique version");
        for dependency in &release.deps {
            active
                .entry(dependency.package.clone())
                .or_default()
                .push(dependency.constraint);
        }
    }
}

fn valid_name(name: &str) -> bool {

}

fn canonical(mut values: Vec<Constraint>) -> Vec<Constraint> {
    values.sort_by(|a, b| {
        let tag = |c: &Constraint| match c {
            Constraint::Exact(_) => 0,
            Constraint::AtLeast(_) => 1,
            Constraint::Before(_) => 2,
            Constraint::Caret(_) => 3,
        };
        tag(a).cmp(&tag(b)).then_with(|| match (a, b) {
            (Constraint::Exact(x), Constraint::Exact(y))
            | (Constraint::AtLeast(x), Constraint::AtLeast(y))
            | (Constraint::Before(x), Constraint::Before(y))
            | (Constraint::Caret(x), Constraint::Caret(y)) => x.cmp(y),
            _ => std::cmp::Ordering::Equal,
        })
    });
    values.dedup();
    values
}

fn allowed(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) if v < x => false,
        Constraint::Caret(x) if x.major != 0 => v.major == x.major,
        Constraint::Caret(x) if x.minor != 0 => v.major == 0 && v.minor == x.minor,
        Constraint::Caret(x) => v == x,
    }
}

fn version_string(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraints_string(values: &[Constraint]) -> String {
    canonical(values.to_vec())
        .iter()

            Constraint::Exact(v) => format!("={}", version_string(*v)),
            Constraint::AtLeast(v) => format!(">={}", version_string(*v)),
            Constraint::Before(v) => format!("<{}", version_string(*v)),
            Constraint::Caret(v) => format!("^{}", version_string(*v)),
        })
        .collect::<Vec<_>>()
        .join(",")
}

fn percent_encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        match b {
            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'.' | b'_' | b'-' => {

        }
    }
    out
}

    }
    for p in catalog {
        let mut versions = BTreeSet::new();
        for release in &p.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let package_map: BTreeMap<&str, &Package> = catalog
        .iter()
        .map(|p| (p.name.as_str(), p))
        .collect();

    let mut requirements: BTreeMap<String, Vec<Constraint>> = BTreeMap::new();
    for root in roots {
        requirements
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    let mut chosen: BTreeMap<String, Version> = BTreeMap::new();

    loop {
        if let Some(name) = requirements
            .keys()
            .find(|name| !package_map.contains_key(name.as_str()))
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in &chosen {
            let all = canonical(requirements.get(name).cloned().unwrap_or_default());
            if all.iter().any(|c| !allowed(*c, *version)) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: all,
                });
            }
        }

        for (name, all) in &requirements {
            if chosen.contains_key(name) {
                continue;
            }
            let normalized = canonical(all.clone());
            let package = package_map[name.as_str()];
            if !package.releases.iter().any(|release| {
                normalized
                    .iter()
                    .all(|constraint| allowed(*constraint, release.version))
            }) {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: normalized,
                });
            }
        }

        let Some(name) = requirements.keys().find(|name| !chosen.contains_key(*name)).cloned()
        else {
            return Ok(chosen
                .into_iter()
                .map(|(package, version)| Selection { package, version })
                .collect());
        };

        let normalized = canonical(requirements[&name].clone());
        let package = package_map[name.as_str()];
        let release = package
            .releases
            .iter()
            .filter(|release| {
                normalized
                    .iter()
                    .all(|constraint| allowed(*constraint, release.version))
            })
            .max_by_key(|release| release.version)
            .expect("matching release checked above");

        chosen.insert(name, release.version);
        for dep in &release.deps {
            requirements
                .entry(dep.package.clone())
                .or_default()
                .push(dep.constraint);
        }
    }
}
