pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;

    for i in 0..catalog.len() {
        for j in 0..i {
            if catalog[i].name == catalog[j].name {
                return Err(ResolveError::DuplicatePackage {
                    package: catalog[i].name.clone(),
                });
            }
        }
    }

    for package in catalog {
        for i in 0..package.releases.len() {
            for j in 0..i {
                if package.releases[i].version == package.releases[j].version {
                    return Err(ResolveError::DuplicateVersion {
                        package: package.name.clone(),
                        version: package.releases[i].version,
                    });
                }
            }
        }
    }

    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let packages: std::collections::BTreeMap<&str, &Package> = catalog
        .iter()
        .map(|p| (p.name.as_str(), p))
        .collect();
    let mut selected: std::collections::BTreeMap<String, Version> =
        std::collections::BTreeMap::new();

    loop {
        if let Some(name) = active.keys().find(|name| !packages.contains_key(name.as_str())) {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in &selected {
            if let Some(requirements) = active.get(name) {
                if !matches(*version, requirements) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: canonical_constraints(requirements),
                    });
                }
            }
        }

        for (name, requirements) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let package = packages[name.as_str()];
            if !package
                .releases
                .iter()
                .any(|release| matches(release.version, requirements))
            {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: canonical_constraints(requirements),
                });
            }
        }

        let next = active
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();
        let Some(name) = next else {
            break;
        };

        let package = packages[name.as_str()];
        let release = package
            .releases
            .iter()
            .filter(|release| matches(release.version, active.get(&name).unwrap()))
            .max_by_key(|release| release.version)
            .expect("candidate existence checked");
        selected.insert(name.clone(), release.version);
        for dep in &release.deps {
            active
                .entry(dep.package.clone())
                .or_default()
                .push(dep.constraint);
        }
    }

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}

        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut selected: std::collections::BTreeMap<String, Version> =
        std::collections::BTreeMap::new();
    let mut dependencies: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for requirement in roots {
        dependencies
            .entry(requirement.package.clone())
            .or_default()
            .push(requirement.constraint);
    }

    loop {
        let mut active = dependencies.clone();
        for (package_name, version) in &selected {
            let package = catalog
                .iter()
                .find(|package| package.name == *package_name)
                .expect("selected package was validated as present");
            let release = package
                .releases
                .iter()
                .find(|release| release.version == *version)
                .expect("selected release came from catalog");
            for dep in &release.deps {
                active
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
            }
        }

        if let Some(missing) = active
            .keys()
            .find(|name| !catalog.iter().any(|package| package.name == **name))
        {
            return Err(ResolveError::Missing {
                package: missing.clone(),
            });
        }

        for (package_name, version) in &selected {
            if let Some(constraints) = active.get(package_name) {
                if !satisfies(*version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: package_name.clone(),
                        selected: *version,
                        requirements: canonical_constraints(constraints),
                    });
                }
            }
        }

        for (package_name, constraints) in &active {
            if selected.contains_key(package_name) {
                continue;
            }
            let package = catalog
                .iter()
                .find(|package| package.name == *package_name)
                .expect("missing package was checked");
            if !package
                .releases
                .iter()
                .any(|release| satisfies(release.version, constraints))
            {
                return Err(ResolveError::NoMatch {
                    package: package_name.clone(),
                    requirements: canonical_constraints(constraints),
                });
            }
        }

        let next_name = active
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();
        let Some(package_name) = next_name else {
            return Ok(selected
                .into_iter()
                .map(|(package, version)| Selection { package, version })
                .collect());
        };

        let constraints = &active[&package_name];
        let package = catalog
            .iter()
            .find(|package| package.name == package_name)
            .expect("missing package was checked");
        let version = package
            .releases
            .iter()
            .filter(|release| satisfies(release.version, constraints))
            .map(|release| release.version)
            .max()
            .expect("candidate existence was checked");
        selected.insert(package_name, version);
    }
