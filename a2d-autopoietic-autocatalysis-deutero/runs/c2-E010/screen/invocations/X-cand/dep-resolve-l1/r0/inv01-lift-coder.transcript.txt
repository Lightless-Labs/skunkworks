fn validate_names(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        let package_site = format!("catalog[{i}]");
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: package_site,
                name: package.name.clone(),
            });

        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {

                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn check_duplicates(catalog: &[Package]) -> Result<(), ResolveError> {
    let mut seen_packages: Vec<&str> = Vec::new();
    for package in catalog {
        if seen_packages.iter().any(|name| *name == package.name) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
        seen_packages.push(&package.name);
    }

    for package in catalog {
        let mut seen_versions: Vec<Version> = Vec::new();
        for release in &package.releases {
            if seen_versions.contains(&release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
            seen_versions.push(release.version);
        }
    }
    Ok(())
}

fn resolve_inner(
    catalog: &[Package],
    active: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
    selected: &mut std::collections::BTreeMap<String, Version>,
) -> Result<(), ResolveError> {
    // R12: missing active packages take precedence over all other node errors.
    for package in active.keys() {
        if !catalog.iter().any(|entry| entry.name == *package) {
            return Err(ResolveError::Missing {
                package: package.clone(),
            });
        }
    }

    // R13: check selected packages in lexical order.
    for (package, version) in selected.iter() {
        if let Some(constraints) = active.get(package) {
            if constraints
                .iter()
                .any(|constraint| !matches(*version, *constraint))
            {
                let mut requirements = constraints.clone();
                canonicalize(&mut requirements);
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements,
                });
            }
        }
    }

    // R14: check all unselected active packages in lexical order before assigning.
    let mut chosen: Option<(String, Vec<(Version, Vec<Requirement>)>)> = None;
    for (package, constraints) in active.iter() {
        if selected.contains_key(package) {
            continue;
        }
        let entry = catalog.iter().find(|entry| entry.name == *package).unwrap();
        let mut matching: Vec<(Version, Vec<Requirement>)> = entry
            .releases
            .iter()
            .filter(|release| constraints.iter().all(|c| matches(release.version, *c)))
            .map(|release| (release.version, release.deps.clone()))
            .collect();
        if matching.is_empty() {
            let mut requirements = constraints.clone();
            canonicalize(&mut requirements);
            return Err(ResolveError::NoMatch {
                package: package.clone(),
                requirements,
            });
        }
        matching.sort_by_key(|(version, _)| *version);
        if chosen.is_none() {
            chosen = Some((package.clone(), matching));
        }
    }

    let Some((package, matching)) = chosen else {
        return Ok(());
    };

    let (version, deps) = matching.last().unwrap().clone();
    selected.insert(package, version);
    for dep in deps {
        active
            .entry(dep.package)
            .or_default()
            .push(dep.constraint);
    }
    resolve_inner(catalog, active, selected)
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;
    check_duplicates(catalog)?;

    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut selected = std::collections::BTreeMap::new();
    resolve_inner(catalog, &mut active, &mut selected)?;

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}

fn constraint_order(a: &Constraint, b: &Constraint) -> std::cmp::Ordering {
    fn rank(c: &Constraint) -> u8 {
        match c {
            Constraint::Exact(_) => 0,
            Constraint::AtLeast(_) => 1,
            Constraint::Before(_) => 2,
            Constraint::Caret(_) => 3,
        }
    }
    rank(a)
        .cmp(&rank(b))
        .then_with(|| {
            let version = |c: &Constraint| match *c {
                Constraint::Exact(v)
                | Constraint::AtLeast(v)
                | Constraint::Before(v)
                | Constraint::Caret(v) => v,
            };
            version(a).cmp(&version(b))
        })
}

    result.sort_by(constraint_order);
    result.dedup();
    result

fn matches(version: Version, c: Constraint) -> bool {

        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) if version < v => false,
        Constraint::Caret(v) if v.major > 0 => version.major == v.major,
        Constraint::Caret(v) if v.minor > 0 => version.major == 0 && version.minor == v.minor,
        Constraint::Caret(v) => version == v,

    out

            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
    }
}

fn validate(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    let invalid = |site: String, name: &String| ResolveError::InvalidName {
        site,
        name: name.clone(),
    };
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(invalid(format!("root[{i}]"), &root.package));
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(invalid(format!("catalog[{i}]"), &package.name));
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(invalid(
                        format!("catalog[{i}].release[{j}].dep[{k}]"),
                        &dep.package,
                    ));
                }
            }
        }
    }
    Ok(())
}

fn validate_duplicates(catalog: &[Package]) -> Result<(), ResolveError> {
    let mut names = std::collections::BTreeSet::new();
    for p in catalog {
        if !names.insert(p.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }

    for p in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }
    Ok(())
}

fn solve(
    catalog: &[Package],
    active: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
    selected: &mut std::collections::BTreeMap<String, Version>,
) -> Result<(), ResolveError> {
    if let Some(package) = active
        .keys()
        .find(|name| !catalog.iter().any(|entry| entry.name == ***name))
    {
        return Err(ResolveError::Missing {
            package: package.clone(),
        });
    }

    for (package, version) in selected.iter() {
        if let Some(constraints) = active.get(package) {
            if constraints.iter().any(|c| !matches(*version, *c)) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: canonical(constraints),
                });
            }
        }
    }

    let mut next: Option<(String, Version, Vec<Requirement>)> = None;
    for (package, constraints) in active.iter() {
        if selected.contains_key(package) {
            continue;
        }
        let entry = catalog.iter().find(|p| p.name == *package).unwrap();
        let best = entry
            .releases
            .iter()
            .filter(|r| constraints.iter().all(|c| matches(r.version, *c)))
            .max_by_key(|r| r.version);
        match best {
            None => {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: canonical(constraints),
                });
            }
            Some(release) if next.is_none() => {
                next = Some((package.clone(), release.version, release.deps.clone()));
            }
            Some(_) => {}
        }
    }

    let Some((package, version, deps)) = next else {
        return Ok(());
    };
    selected.insert(package, version);
    for dep in deps {
        active
            .entry(dep.package)
            .or_default()
            .push(dep.constraint);
    }
    solve(catalog, active, selected)
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    validate(catalog, roots)?;
    validate_duplicates(catalog)?;

    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    let mut selected = std::collections::BTreeMap::<String, Version>::new();
    solve(catalog, &mut active, &mut selected)?;

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}
