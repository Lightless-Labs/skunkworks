pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;

    for i in 0..catalog.len() {
        for j in 0..i {
            if catalog[i].name == catalog[j].name {
                return Err(ResolveError::DuplicatePackage {
                    package: catalog[i].name.clone(),
                });
            }
        }
    }

    for package in catalog {
        for i in 0..package.releases.len() {
            for j in 0..i {
                if package.releases[i].version == package.releases[j].version {
                    return Err(ResolveError::DuplicateVersion {
                        package: package.name.clone(),
                        version: package.releases[i].version,
                    });
                }
            }
        }
    }

    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let packages: std::collections::BTreeMap<&str, &Package> = catalog
        .iter()
        .map(|p| (p.name.as_str(), p))
        .collect();
    let mut selected: std::collections::BTreeMap<String, Version> =
        std::collections::BTreeMap::new();

    loop {
        if let Some(name) = active.keys().find(|name| !packages.contains_key(name.as_str())) {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in &selected {
            if let Some(requirements) = active.get(name) {
                if !matches(*version, requirements) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: canonical_constraints(requirements),
                    });
                }
            }
        }

        for (name, requirements) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let package = packages[name.as_str()];
            if !package
                .releases
                .iter()
                .any(|release| matches(release.version, requirements))
            {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: canonical_constraints(requirements),
                });
            }
        }

        let next = active
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();
        let Some(name) = next else {
            break;
        };

        let package = packages[name.as_str()];
        let release = package
            .releases
            .iter()
            .filter(|release| matches(release.version, active.get(&name).unwrap()))
            .max_by_key(|release| release.version)
            .expect("candidate existence checked");
        selected.insert(name.clone(), release.version);
        for dep in &release.deps {
            active
                .entry(dep.package.clone())
                .or_default()
                .push(dep.constraint);
        }
    }

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}

fn satisfies(version: Version, constraints: &[Constraint]) -> bool {
    for constraint in constraints {
        let accepted = match *constraint {
            Constraint::Exact(wanted) => version == wanted,
            Constraint::AtLeast(minimum) => version >= minimum,
            Constraint::Before(limit) => version < limit,
            Constraint::Caret(base) => {

                    version.major == 0 && version.minor == base.minor

        }
    }
    true

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut selected: std::collections::BTreeMap<String, Version> =
        std::collections::BTreeMap::new();
    let mut dependencies: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for requirement in roots {
        dependencies
            .entry(requirement.package.clone())
            .or_default()
            .push(requirement.constraint);
    }

    loop {
        let mut active = dependencies.clone();
        for (package_name, version) in &selected {
            let package = catalog
                .iter()
                .find(|package| package.name == *package_name)
                .expect("selected package was validated as present");
            let release = package
                .releases
                .iter()
                .find(|release| release.version == *version)
                .expect("selected release came from catalog");
            for dep in &release.deps {
                active
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
            }
        }

        if let Some(missing) = active
            .keys()
            .find(|name| !catalog.iter().any(|package| package.name == **name))
        {
            return Err(ResolveError::Missing {
                package: missing.clone(),
            });
        }

        for (package_name, version) in &selected {
            if let Some(constraints) = active.get(package_name) {
                if !satisfies(*version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: package_name.clone(),
                        selected: *version,
                        requirements: canonical_constraints(constraints),
                    });
                }
            }
        }

        for (package_name, constraints) in &active {
            if selected.contains_key(package_name) {
                continue;
            }
            let package = catalog
                .iter()
                .find(|package| package.name == *package_name)
                .expect("missing package was checked");
            if !package
                .releases
                .iter()
                .any(|release| satisfies(release.version, constraints))
            {
                return Err(ResolveError::NoMatch {
                    package: package_name.clone(),
                    requirements: canonical_constraints(constraints),
                });
            }
        }

        let next_name = active
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();
        let Some(package_name) = next_name else {
            return Ok(selected
                .into_iter()
                .map(|(package, version)| Selection { package, version })
                .collect());
        };

        let constraints = &active[&package_name];
        let package = catalog
            .iter()
            .find(|package| package.name == package_name)
            .expect("missing package was checked");
        let version = package
            .releases
            .iter()
            .filter(|release| satisfies(release.version, constraints))
            .map(|release| release.version)
            .max()
            .expect("candidate existence was checked");
        selected.insert(package_name, version);
    }
}

        if seen_packages.iter().any(|name| *name == package.name) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),

        seen_packages.push(&package.name);

        let mut seen_versions: Vec<Version> = Vec::new();

fn resolve_inner(
    catalog: &[Package],
    active: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
    selected: &mut std::collections::BTreeMap<String, Version>,
) -> Result<(), ResolveError> {
    // R12: missing active packages take precedence over all other node errors.
    for package in active.keys() {
        if !catalog.iter().any(|entry| entry.name == *package) {
            return Err(ResolveError::Missing {
                package: package.clone(),
            });
        }
    }

    // R13: check selected packages in lexical order.
    for (package, version) in selected.iter() {
        if let Some(constraints) = active.get(package) {
            if constraints
                .iter()
                .any(|constraint| !matches(*version, *constraint))
            {
                let mut requirements = constraints.clone();
                canonicalize(&mut requirements);
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements,
                });
            }
        }
    }

    // R14: check all unselected active packages in lexical order before assigning.
    let mut chosen: Option<(String, Vec<(Version, Vec<Requirement>)>)> = None;
    for (package, constraints) in active.iter() {
        if selected.contains_key(package) {
            continue;
        }
        let entry = catalog.iter().find(|entry| entry.name == *package).unwrap();
        let mut matching: Vec<(Version, Vec<Requirement>)> = entry
            .releases
            .iter()
            .filter(|release| constraints.iter().all(|c| matches(release.version, *c)))
            .map(|release| (release.version, release.deps.clone()))
            .collect();
        if matching.is_empty() {
            let mut requirements = constraints.clone();
            canonicalize(&mut requirements);
            return Err(ResolveError::NoMatch {
                package: package.clone(),
                requirements,
            });
        }
        matching.sort_by_key(|(version, _)| *version);
        if chosen.is_none() {
            chosen = Some((package.clone(), matching));
        }
    }

    let Some((package, matching)) = chosen else {
        return Ok(());
    };

    let (version, deps) = matching.last().unwrap().clone();
    selected.insert(package, version);
    for dep in deps {
        active
            .entry(dep.package)
            .or_default()
            .push(dep.constraint);
    }
    resolve_inner(catalog, active, selected)
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;
    check_duplicates(catalog)?;

    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut selected = std::collections::BTreeMap::new();
    resolve_inner(catalog, &mut active, &mut selected)?;

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}

        let tag = |c: &Constraint| match c {
            Constraint::Exact(_) => 0,
            Constraint::AtLeast(_) => 1,
            Constraint::Before(_) => 2,
            Constraint::Caret(_) => 3,

            | (Constraint::Caret(x), Constraint::Caret(y)) => x.cmp(y),
            _ => std::cmp::Ordering::Equal,
        })
    });
    values.dedup();
    values
}

            format!("duplicate-package|package={}", percent_encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            percent_encode(package),
            version_string(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", percent_encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            percent_encode(package),
            constraints_string(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            percent_encode(package),
            version_string(*selected),
            constraints_string(requirements)
        ),
    }
}

pub fn resolve(catalog: &[Package], roots: &[Requirement]) -> Result<Vec<Selection>, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: r.package.clone(),
            });
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: p.name.clone(),
            });
        }
        for (j, release) in p.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for p in catalog {
        if !package_names.insert(&p.name) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }
    for p in catalog {
        let mut versions = BTreeSet::new();
        for release in &p.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let package_map: BTreeMap<&str, &Package> = catalog
        .iter()
        .map(|p| (p.name.as_str(), p))
        .collect();

    let mut requirements: BTreeMap<String, Vec<Constraint>> = BTreeMap::new();
    for root in roots {
        requirements
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    let mut chosen: BTreeMap<String, Version> = BTreeMap::new();

    loop {
        if let Some(name) = requirements
            .keys()
            .find(|name| !package_map.contains_key(name.as_str()))
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in &chosen {
            let all = canonical(requirements.get(name).cloned().unwrap_or_default());
            if all.iter().any(|c| !allowed(*c, *version)) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: all,
                });
            }
        }

        for (name, all) in &requirements {
            if chosen.contains_key(name) {
                continue;
            }
            let normalized = canonical(all.clone());
            let package = package_map[name.as_str()];
            if !package.releases.iter().any(|release| {
                normalized
                    .iter()
                    .all(|constraint| allowed(*constraint, release.version))
            }) {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: normalized,
                });
            }
        }

        let Some(name) = requirements.keys().find(|name| !chosen.contains_key(*name)).cloned()
        else {
            return Ok(chosen
                .into_iter()
                .map(|(package, version)| Selection { package, version })
                .collect());
        };

        let normalized = canonical(requirements[&name].clone());
        let package = package_map[name.as_str()];
        let release = package
            .releases
            .iter()
            .filter(|release| {
                normalized
                    .iter()
                    .all(|constraint| allowed(*constraint, release.version))
            })
            .max_by_key(|release| release.version)
            .expect("matching release checked above");

        chosen.insert(name, release.version);
        for dep in &release.deps {
            requirements
                .entry(dep.package.clone())
                .or_default()
                .push(dep.constraint);
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },

        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn constraint_order(a: &Constraint, b: &Constraint) -> std::cmp::Ordering {
    fn rank(c: &Constraint) -> u8 {
        match c {
            Constraint::Exact(_) => 0,
            Constraint::AtLeast(_) => 1,
            Constraint::Before(_) => 2,
            Constraint::Caret(_) => 3,
        }
    }
    rank(a)
        .cmp(&rank(b))
        .then_with(|| {
            let version = |c: &Constraint| match *c {
                Constraint::Exact(v)
                | Constraint::AtLeast(v)
                | Constraint::Before(v)
                | Constraint::Caret(v) => v,
            };
            version(a).cmp(&version(b))
        })
}

fn canonical(constraints: &[Constraint]) -> Vec<Constraint> {
    let mut result = constraints.to_vec();
    result.sort_by(constraint_order);
    result.dedup();
    result
}

fn matches(version: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) if version < v => false,
        Constraint::Caret(v) if v.major > 0 => version.major == v.major,
        Constraint::Caret(v) if v.minor > 0 => version.major == 0 && version.minor == v.minor,
        Constraint::Caret(v) => version == v,
    }
}

fn encode(s: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::new();
    for &b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches_byte_passthrough(b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn matches_byte_passthrough(b: u8) -> bool {
    b == b'.' || b == b'_' || b == b'-'
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical(cs)
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn validate(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    let invalid = |site: String, name: &String| ResolveError::InvalidName {
        site,
        name: name.clone(),
    };
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(invalid(format!("root[{i}]"), &root.package));
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(invalid(format!("catalog[{i}]"), &package.name));
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(invalid(
                        format!("catalog[{i}].release[{j}].dep[{k}]"),
                        &dep.package,
                    ));
                }
            }
        }
    }
    Ok(())
}

fn validate_duplicates(catalog: &[Package]) -> Result<(), ResolveError> {
    let mut names = std::collections::BTreeSet::new();
    for p in catalog {
        if !names.insert(p.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }

    for p in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }
    Ok(())
}

fn solve(
    catalog: &[Package],
    active: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
    selected: &mut std::collections::BTreeMap<String, Version>,
) -> Result<(), ResolveError> {
    if let Some(package) = active
        .keys()
        .find(|name| !catalog.iter().any(|entry| entry.name == ***name))
    {
        return Err(ResolveError::Missing {
            package: package.clone(),
        });
    }

    for (package, version) in selected.iter() {
        if let Some(constraints) = active.get(package) {
            if constraints.iter().any(|c| !matches(*version, *c)) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: canonical(constraints),
                });
            }
        }
    }

    let mut next: Option<(String, Version, Vec<Requirement>)> = None;
    for (package, constraints) in active.iter() {
        if selected.contains_key(package) {
            continue;
        }
        let entry = catalog.iter().find(|p| p.name == *package).unwrap();
        let best = entry
            .releases
            .iter()
            .filter(|r| constraints.iter().all(|c| matches(r.version, *c)))
            .max_by_key(|r| r.version);
        match best {
            None => {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: canonical(constraints),
                });
            }
            Some(release) if next.is_none() => {
                next = Some((package.clone(), release.version, release.deps.clone()));
            }
            Some(_) => {}
        }
    }

    let Some((package, version, deps)) = next else {
        return Ok(());
    };
    selected.insert(package, version);
    for dep in deps {
        active
            .entry(dep.package)
            .or_default()
            .push(dep.constraint);
    }
    solve(catalog, active, selected)
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    validate(catalog, roots)?;
    validate_duplicates(catalog)?;

    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    let mut selected = std::collections::BTreeMap::<String, Version>::new();
    solve(catalog, &mut active, &mut selected)?;

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}
