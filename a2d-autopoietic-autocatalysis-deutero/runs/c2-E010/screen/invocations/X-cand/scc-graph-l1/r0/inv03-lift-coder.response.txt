### Solution A

```rust
pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }

    for (edge_index, &(source, target)) in edges.iter().enumerate() {
        if source >= n {
            return Err(GraphError::BadEndpoint {
                edge: edge_index,
                side: 0,
                vertex: source,
            });
        }
        if target >= n {
            return Err(GraphError::BadEndpoint {
                edge: edge_index,
                side: 1,
                vertex: target,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse_adjacency = vec![Vec::new(); n];
    for &(source, target) in edges {
        adjacency[source].push(target);
        reverse_adjacency[target].push(source);
    }

    // Iterative first pass of Kosaraju's algorithm.
    let mut visited = vec![false; n];
    let mut finish_order = Vec::with_capacity(n);

    for start in 0..n {
        if visited[start] {
            continue;
        }
        visited[start] = true;
        let mut stack = vec![(start, 0usize)];

        while let Some((vertex, next_edge)) = stack.last_mut() {
            if *next_edge < adjacency[*vertex].len() {
                let neighbor = adjacency[*vertex][*next_edge];
                *next_edge += 1;
                if !visited[neighbor] {
                    visited[neighbor] = true;
                    stack.push((neighbor, 0));
                }
            } else {
                let (finished_vertex, _) = stack.pop().unwrap();
                finish_order.push(finished_vertex);
            }
        }
    }

    // Second pass on the reversed graph.
    visited.fill(false);
    let mut components = Vec::new();

    for &start in finish_order.iter().rev() {
        if visited[start] {
            continue;
        }

        let mut component = Vec::new();
        let mut stack = vec![start];
        visited[start] = true;

        while let Some(vertex) = stack.pop() {
            component.push(vertex);
            for &neighbor in &reverse_adjacency[vertex] {
                if !visited[neighbor] {
                    visited[neighbor] = true;
                    stack.push(neighbor);
                }
            }
        }

        component.sort_unstable();
        components.push(component);
    }

    components.sort_unstable_by(|a, b| {
        b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0]))
    });

    let mut component_of = vec![0; n];
    let mut cyclic = Vec::with_capacity(components.len());

    for (component_id, component) in components.iter().enumerate() {
        for &vertex in component {
            component_of[vertex] = component_id;
        }
        cyclic.push(component.len() >= 2);
    }

    for &(source, target) in edges {
        if source == target {
            cyclic[component_of[source]] = true;
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
    })
}
```

### Solution B

```rust
pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }

    for (edge, &(source, target)) in edges.iter().enumerate() {
        if source >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 0,
                vertex: source,
            });
        }
        if target >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: target,
            });
        }
    }

    // Iterative Tarjan SCC algorithm.
    let mut adjacency = vec![Vec::new(); n];
    for &(source, target) in edges {
        adjacency[source].push(target);
    }

    let mut index = vec![usize::MAX; n];
    let mut lowlink = vec![0usize; n];
    let mut on_stack = vec![false; n];
    let mut tarjan_stack = Vec::with_capacity(n);
    let mut next_index = 0usize;
    let mut components = Vec::new();

    for root in 0..n {
        if index[root] != usize::MAX {
            continue;
        }

        index[root] = next_index;
        lowlink[root] = next_index;
        next_index += 1;
        tarjan_stack.push(root);
        on_stack[root] = true;

        let mut dfs = vec![(root, 0usize)];

        while !dfs.is_empty() {
            let frame_index = dfs.len() - 1;
            let vertex = dfs[frame_index].0;

            if dfs[frame_index].1 < adjacency[vertex].len() {
                let neighbor = adjacency[vertex][dfs[frame_index].1];
                dfs[frame_index].1 += 1;

                if index[neighbor] == usize::MAX {
                    index[neighbor] = next_index;
                    lowlink[neighbor] = next_index;
                    next_index += 1;
                    tarjan_stack.push(neighbor);
                    on_stack[neighbor] = true;
                    dfs.push((neighbor, 0));
                } else if on_stack[neighbor] {
                    lowlink[vertex] = lowlink[vertex].min(index[neighbor]);
                }
            } else {
                dfs.pop();

                if let Some(&(parent, _)) = dfs.last() {
                    lowlink[parent] = lowlink[parent].min(lowlink[vertex]);
                }

                if lowlink[vertex] == index[vertex] {
                    let mut component = Vec::new();
                    loop {
                        let member = tarjan_stack.pop().unwrap();
                        on_stack[member] = false;
                        component.push(member);
                        if member == vertex {
                            break;
                        }
                    }
                    component.sort_unstable();
                    components.push(component);
                }
            }
        }
    }

    components.sort_unstable_by(|a, b| {
        b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0]))
    });

    let mut component_of = vec![0usize; n];
    let mut cyclic = Vec::with_capacity(components.len());

    for (component_id, component) in components.iter().enumerate() {
        for &vertex in component {
            component_of[vertex] = component_id;
        }
        cyclic.push(component.len() >= 2);
    }

    for &(source, target) in edges {
        if source == target {
            cyclic[component_of[source]] = true;
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
    })
}
```
