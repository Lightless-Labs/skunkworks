    pub cyclic: Vec<bool>,

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {

        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(s, t)) in edges.iter().enumerate() {
        if s >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: s,
            });

        if t >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: t,
            });
        }
    }

    let mut rev = vec![Vec::new(); n];
    for &(s, t) in edges {
        adj[s].push(t);
        rev[t].push(s);
    }

    let mut seen = vec![false; n];
    let mut finish = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((v, next_edge)) = stack.last_mut() {
            if *next_edge < adj[*v].len() {
                let w = adj[*v][*next_edge];
                *next_edge += 1;
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                finish.push(*v);
                stack.pop();
            }
        }
    }

    seen.fill(false);
    let mut components = Vec::new();
    for &root in finish.iter().rev() {
        if seen[root] {
            continue;
        }
        seen[root] = true;

                    seen[w] = true;
                    stack.push(w);
                }
            }

        component.sort_unstable();
        components.push(component);
    }

        b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0]))

        cyclic,

                vertex: target,

            continue;

        next_index += 1;

}

    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (edge, &(source, target)) in edges.iter().enumerate() {
        if source >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 0,
                vertex: source,
            });
        }
        if target >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: target,
            });
        }
    }

    // Compute each vertex's reachability set using iterative graph searches.
    // Two vertices share an SCC exactly when each can reach the other.
    let mut adjacency = vec![Vec::new(); n];
    let mut self_loop = vec![false; n];
    for &(source, target) in edges {
        adjacency[source].push(target);
        if source == target {
            self_loop[source] = true;
        }
    }

    let mut reachable = vec![vec![false; n]; n];
    for start in 0..n {
        reachable[start][start] = true;
        let mut stack = vec![start];
        while let Some(vertex) = stack.pop() {
            for &neighbor in &adjacency[vertex] {
                if !reachable[start][neighbor] {
                    reachable[start][neighbor] = true;
                    stack.push(neighbor);
                }
            }
        }
    }

    let mut assigned = vec![false; n];
    let mut components = Vec::new();
    for vertex in 0..n {
        if assigned[vertex] {
            continue;
        }
        let mut component = Vec::new();
        for candidate in vertex..n {
            if !assigned[candidate]
                && reachable[vertex][candidate]
                && reachable[candidate][vertex]
            {
                assigned[candidate] = true;
                component.push(candidate);
            }
        }
        components.push(component);
    }

    components.sort_by(|a, b| b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0])));

    let mut component_of = vec![0; n];

            component_of[vertex] = id;
        }
        cyclic.push(component.len() > 1 || self_loop[component[0]]);

            count: n,
            limit: MAX_VERTICES,

        if source >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 0,
                vertex: source,
            });

            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: target,
            });
        }
    }

    let mut forward = vec![Vec::new(); n];
    let mut reverse = vec![Vec::new(); n];
    let mut self_loop = vec![false; n];

    for &(source, target) in edges {
        forward[source].push(target);
        reverse[target].push(source);
        if source == target {
            self_loop[source] = true;
        }
    }

    // Iterative depth-first search to obtain finishing order.
    let mut visited = vec![false; n];
    let mut finish = Vec::with_capacity(n);
    for start in 0..n {
        if visited[start] {
            continue;
        }
        visited[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((vertex, next_index)) = stack.pop() {
            if next_index < forward[vertex].len() {
                stack.push((vertex, next_index + 1));
                let next = forward[vertex][next_index];
                if !visited[next] {
                    visited[next] = true;
                    stack.push((next, 0));
                }
            } else {
                finish.push(vertex);
            }
        }
    }

    // Second pass of Kosaraju's algorithm.
    let mut assigned = vec![false; n];
    let mut raw_components: Vec<Vec<usize>> = Vec::new();
    for &start in finish.iter().rev() {
        if assigned[start] {
            continue;

        assigned[start] = true;
        let mut component = Vec::new();
        let mut stack = vec![start];

            component.push(vertex);

                    assigned[next] = true;
                    stack.push(next);
                }
            }

        index[start] = next_index;
        lowlink[start] = next_index;
        next_index += 1;
        tarjan_stack.push(start);
        on_stack[start] = true;
        let mut dfs = vec![(start, 0usize)];

        while !dfs.is_empty() {
            let frame_index = dfs.len() - 1;
            let vertex = dfs[frame_index].0;

            if dfs[frame_index].1 < graph[vertex].len() {
                let edge_index = dfs[frame_index].1;
                dfs[frame_index].1 += 1;
                let next = graph[vertex][edge_index];

                if index[next] == usize::MAX {
                    index[next] = next_index;
                    lowlink[next] = next_index;
                    next_index += 1;
                    tarjan_stack.push(next);
                    on_stack[next] = true;
                    dfs.push((next, 0));
                } else if on_stack[next] {
                    lowlink[vertex] = lowlink[vertex].min(index[next]);
                }
            } else {
                dfs.pop();

                if let Some(&(parent, _)) = dfs.last() {
                    lowlink[parent] = lowlink[parent].min(lowlink[vertex]);
                }

                if lowlink[vertex] == index[vertex] {
                    let mut component = Vec::new();
                    loop {
                        let member = tarjan_stack.pop().expect("active SCC vertex");
                        on_stack[member] = false;
                        component.push(member);
                        if member == vertex {
                            break;
                        }
                    }
                    component.sort_unstable();
                    components.push(component);
                }
            }
        }
    }

    components.sort_by(|a, b| {
        b.len()
            .cmp(&a.len())
            .then_with(|| a[0].cmp(&b[0]))
    });

    let mut component_of = vec![0; n];
    let mut cyclic = Vec::with_capacity(components.len());
    for (component_id, component) in components.iter().enumerate() {
        for &vertex in component {
            component_of[vertex] = component_id;
        }
        cyclic.push(component.len() >= 2 || self_loop[component[0]]);
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
    })
}

pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(source, target)) in edges.iter().enumerate() {
        if source >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: source,
            });
        }
        if target >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: target,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse = vec![Vec::new(); n];
    let mut self_loop = vec![false; n];
    for &(source, target) in edges {
        adjacency[source].push(target);
        reverse[target].push(source);
        if source == target {
            self_loop[source] = true;
        }
    }

    // Iterative DFS finishing order.
    let mut seen = vec![false; n];
    let mut order = Vec::with_capacity(n);
    for start in 0..n {
        if seen[start] {
            continue;
        }
        seen[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((vertex, next_edge)) = stack.last_mut() {
            if *next_edge < adjacency[*vertex].len() {
                let neighbor = adjacency[*vertex][*next_edge];
                *next_edge += 1;
                if !seen[neighbor] {
                    seen[neighbor] = true;
                    stack.push((neighbor, 0));
                }
            } else {
                let (finished, _) = stack.pop().unwrap();
                order.push(finished);
            }
        }
    }

    // Reverse-graph traversal in decreasing finishing order.
    let mut assigned = vec![false; n];
    let mut components = Vec::new();
    for &start in order.iter().rev() {
        if assigned[start] {
            continue;
        }
        assigned[start] = true;
        let mut component = Vec::new();
        let mut stack = vec![start];
        while let Some(vertex) = stack.pop() {
            component.push(vertex);
            for &neighbor in &reverse[vertex] {
                if !assigned[neighbor] {
                    assigned[neighbor] = true;
                    stack.push(neighbor);
                }
            }
        }
        component.sort_unstable();
        components.push(component);
    }

    components.sort_by(|a, b| b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0])));

    let mut component_of = vec![0; n];
    let mut cyclic = Vec::with_capacity(components.len());
    for (id, component) in components.iter().enumerate() {
        for &vertex in component {
            component_of[vertex] = id;
        }
        cyclic.push(component.len() > 1 || self_loop[component[0]]);
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
    })
}

pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }

    for (edge_index, &(source, target)) in edges.iter().enumerate() {
        if source >= n {
            return Err(GraphError::BadEndpoint {
                edge: edge_index,
                side: 0,
                vertex: source,
            });
        }
        if target >= n {
            return Err(GraphError::BadEndpoint {
                edge: edge_index,
                side: 1,
                vertex: target,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse_adjacency = vec![Vec::new(); n];
    for &(source, target) in edges {
        adjacency[source].push(target);
        reverse_adjacency[target].push(source);
    }

    // Iterative first pass of Kosaraju's algorithm.
    let mut visited = vec![false; n];
    let mut finish_order = Vec::with_capacity(n);

    for start in 0..n {
        if visited[start] {
            continue;
        }
        visited[start] = true;
        let mut stack = vec![(start, 0usize)];

        while let Some((vertex, next_edge)) = stack.last_mut() {
            if *next_edge < adjacency[*vertex].len() {
                let neighbor = adjacency[*vertex][*next_edge];
                *next_edge += 1;
                if !visited[neighbor] {
                    visited[neighbor] = true;
                    stack.push((neighbor, 0));
                }
            } else {
                let (finished_vertex, _) = stack.pop().unwrap();
                finish_order.push(finished_vertex);
            }
        }
    }

    // Second pass on the reversed graph.
    visited.fill(false);
    let mut components = Vec::new();

    for &start in finish_order.iter().rev() {
        if visited[start] {
            continue;
        }

        let mut component = Vec::new();
        let mut stack = vec![start];
        visited[start] = true;

        while let Some(vertex) = stack.pop() {
            component.push(vertex);
            for &neighbor in &reverse_adjacency[vertex] {
                if !visited[neighbor] {
                    visited[neighbor] = true;
                    stack.push(neighbor);
                }
            }
        }

        component.sort_unstable();
        components.push(component);
    }

    components.sort_unstable_by(|a, b| {
        b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0]))
    });

    let mut component_of = vec![0; n];
    let mut cyclic = Vec::with_capacity(components.len());

    for (component_id, component) in components.iter().enumerate() {
        for &vertex in component {
            component_of[vertex] = component_id;
        }
        cyclic.push(component.len() >= 2);
    }

    for &(source, target) in edges {
        if source == target {
            cyclic[component_of[source]] = true;
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
    })
}
