        return Ok(Charge {
            amount,
            bands: vec![Band {
                index,
                units: quantity,
                amount,
            }],
        });
    }

    let mut remaining = quantity;
    let mut total = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        let units = if let Some(bound) = tier.up_to {

                index,
                units,
                amount,

        }

}

fn make_band(
    sum: &mut u64,
    output: &mut Vec<Band>,
    index: usize,
    units: u64,
    tier: &Tier,
) -> Result<(), Error> {
    let amount = units
        .checked_mul(tier.unit)
        .and_then(|n| n.checked_add(tier.start))
        .ok_or(Error::Overflow { index })?;
    *sum = sum.checked_add(amount).ok_or(Error::Overflow { index })?;
    output.push(Band {
        index,
        units,
        amount,
    });
    Ok(())

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    let mut sum = 0u64;
    let mut bands = Vec::new();

    match price {
        Price::Flat { unit, start } => {
            if quantity != 0 {
                let amount = quantity
                    .checked_mul(*unit)
                    .and_then(|n| n.checked_add(*start))
                    .ok_or(Error::Overflow { index: 0 })?;
                sum = amount;
                bands.push(Band {
                    index: 0,
                    units: quantity,
                    amount,
                });
            }
        }
        Price::Tiers(tiers) => {
            validate(tiers)?;
            let mut left = quantity;
            let mut lower = 0u64;
            for (index, tier) in tiers.iter().enumerate() {
                if left == 0 {
                    break;
                }
                let allocated = match tier.up_to {
                    Some(upper) => {
                        let capacity = upper - lower;
                        lower = upper;
                        left.min(capacity)
                    }
                    None => left,
                };
                if allocated != 0 {
                    make_band(&mut sum, &mut bands, index, allocated, tier)?;
                    left -= allocated;
                }
            }
        }
    }

    Ok(Charge {
        amount: sum,
        bands,
    })
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = match (mode, price) {
        (_, Price::Flat { .. }) | (Mode::Graduated, _) => quote(quantity, price)?,
        (Mode::Volume, Price::Tiers(tiers)) => {
            validate(tiers)?;
            let mut charge = Charge {
                amount: 0,
                bands: Vec::new(),
            };
            if quantity != 0 {
                let selected = tiers
                    .iter()
                    .enumerate()
                    .find(|(_, t)| match t.up_to {
                        Some(bound) => bound >= quantity,
                        None => true,
                    });
                if let Some((index, tier)) = selected {
                    make_band(
                        &mut charge.amount,
                        &mut charge.bands,
                        index,
                        quantity,
                        tier,
                    )?;
                }
            }
            charge
        }
    };

    let minimum_adjustment = if quantity == 0 {
        0
    } else if minimum > charge.amount {
        minimum - charge.amount
    } else {
        0
    };
    let total = charge.amount + minimum_adjustment;
    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}

fn checked_band_amount(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    units

        .and_then(|product| product.checked_add(start))

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous_bound = 0;
    let last_index = tiers.len() - 1;

    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous_bound {
                    return Err(Error::Bound { index });
                }
                previous_bound = bound;
            }
            None => {
                if index != last_index {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers[last_index].up_to.is_some() {
        return Err(Error::MissingFinal);
    }

    Ok(())
}

fn graduated_tiers(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    let mut remaining = quantity;
    let mut previous_bound = 0;
    let mut total = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        if remaining == 0 {
            break;
        }

        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous_bound;
                previous_bound = bound;
                remaining.min(width)
            }
            None => remaining,

            let amount = checked_band_amount(units, tier.unit, tier.start, index)?;
            total = total
                .checked_add(amount)
                .ok_or(Error::Overflow { index })?;
            bands.push(Band {
                index,
                units,
                amount,

            remaining -= units;

        amount: total,
        bands,

fn add_band(total: &mut u64, bands: &mut Vec<Band>, index: usize, units: u64, amount: u64) -> Result<(), Error> {

fn quote_tiers(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    let mut cursor = 0u64;
    let mut total = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        if cursor == quantity {
            break;
        }
        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - cursor;
                let units = width.min(quantity - cursor);
                cursor += units;
                units
            }
            None => {
                let units = quantity - cursor;
                cursor = quantity;
                units
            }
        };
        if units != 0 {
            let amount = amount_for(units, tier, index)?;
            add_band(&mut total, &mut bands, index, units, amount)?;
        }
    }

    Ok(Charge {
        amount: total,
        bands,
    })

fn validate(price: &Price) -> Result<(), Error> {
    let tiers = match price {
        Price::Flat { .. } => return Ok(()),
        Price::Tiers(tiers) => tiers,
    };

    match tiers.len() {
        0 => return Err(Error::Empty),
        65.. => return Err(Error::TooMany),
        _ => {}
    }

    let final_index = tiers.len() - 1;
    let mut last_finite = 0u64;

    for (index, tier) in tiers.iter().enumerate() {
        if let Some(bound) = tier.up_to {
            if bound <= last_finite {
                return Err(Error::Bound { index });
            }
            last_finite = bound;
        } else if index != final_index {
            return Err(Error::Unbounded { index });
        }
    }

    if tiers[final_index].up_to.is_some() {
        Err(Error::MissingFinal)
    } else {
        Ok(())
    }
}

fn add_band(
    total: &mut u64,
    bands: &mut Vec<Band>,
    index: usize,
    units: u64,
    tier: &Tier,
) -> Result<(), Error> {
    let amount = units
        .checked_mul(tier.unit)
        .and_then(|value| value.checked_add(tier.start))
        .ok_or(Error::Overflow { index })?;
    *total = total
        .checked_add(amount)
        .ok_or(Error::Overflow { index })?;
    bands.push(Band {
        index,
        units,
        amount,
    });
    Ok(())

fn tier_charge(quantity: u64, tiers: &[Tier], mode: &Mode) -> Result<Charge, Error> {
    let mut charge = Charge {
        amount: 0,
        bands: Vec::new(),
    };
    if quantity == 0 {
        return Ok(charge);
    }

    match mode {
        Mode::Volume => {
            let index = tiers
                .iter()
                .position(|tier| tier.up_to.is_none_or(|bound| quantity <= bound))
                .expect("validated schedule has a final unbounded tier");
            add_band(
                &mut charge.amount,
                &mut charge.bands,
                index,
                quantity,
                &tiers[index],
            )?;
        }
        Mode::Graduated => {
            let mut remaining = quantity;
            let mut previous = 0u64;

            for (index, tier) in tiers.iter().enumerate() {
                if remaining == 0 {
                    break;
                }

                let allocated = if let Some(bound) = tier.up_to {
                    let width = bound - previous;
                    previous = bound;
                    remaining.min(width)
                } else {
                    remaining
                };

                if allocated != 0 {
                    add_band(
                        &mut charge.amount,
                        &mut charge.bands,
                        index,
                        allocated,
                        tier,
                    )?;
                    remaining -= allocated;
                }
            }
        }
    }

    Ok(charge)
}
