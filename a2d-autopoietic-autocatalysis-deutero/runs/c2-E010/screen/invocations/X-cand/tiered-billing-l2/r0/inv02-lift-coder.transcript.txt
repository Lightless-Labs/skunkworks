fn checked_band_amount(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    units
        .checked_mul(unit)
        .and_then(|product| product.checked_add(start))

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous_bound = 0;
    let last_index = tiers.len() - 1;

    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous_bound {
                    return Err(Error::Bound { index });
                }
                previous_bound = bound;
            }
            None => {
                if index != last_index {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers[last_index].up_to.is_some() {
        return Err(Error::MissingFinal);
    }

    Ok(())
}

fn graduated_tiers(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    let mut remaining = quantity;
    let mut previous_bound = 0;
    let mut total = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        if remaining == 0 {
            break;
        }

        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous_bound;
                previous_bound = bound;
                remaining.min(width)
            }
            None => remaining,
        };

        if units > 0 {
            let amount = checked_band_amount(units, tier.unit, tier.start, index)?;
            total = total
                .checked_add(amount)
                .ok_or(Error::Overflow { index })?;
            bands.push(Band {
                index,
                units,
                amount,
            });
            remaining -= units;
        }
    }

    Ok(Charge {
        amount: total,
        bands,
    })
}

- Schedule validation precedes all tier arithmetic; `TooMany` is checked before inspecting tiers.
- A finite bound includes its endpoint, and `None` is legal only on the final tier.
- A zero quantity has no bands and incurs no minimum adjustment.
- Overflow reports the original tier index; minimum adjustment is applied only after charge calculation succeeds.

    pub total: u64,

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn validate(price: &Price) -> Result<(), Error> {
    let tiers = match price {
        Price::Flat { .. } => return Ok(()),
        Price::Tiers(tiers) => tiers,
    };

    match tiers.len() {
        0 => return Err(Error::Empty),
        65.. => return Err(Error::TooMany),
        _ => {}
    }

    let final_index = tiers.len() - 1;
    let mut last_finite = 0u64;

    for (index, tier) in tiers.iter().enumerate() {
        if let Some(bound) = tier.up_to {
            if bound <= last_finite {
                return Err(Error::Bound { index });
            }
            last_finite = bound;
        } else if index != final_index {
            return Err(Error::Unbounded { index });
        }
    }

    if tiers[final_index].up_to.is_some() {
        Err(Error::MissingFinal)
    } else {
        Ok(())
    }
}

fn add_band(
    total: &mut u64,
    bands: &mut Vec<Band>,
    index: usize,
    units: u64,
    tier: &Tier,
) -> Result<(), Error> {
    let amount = units
        .checked_mul(tier.unit)
        .and_then(|value| value.checked_add(tier.start))
        .ok_or(Error::Overflow { index })?;
    *total = total
        .checked_add(amount)
        .ok_or(Error::Overflow { index })?;
    bands.push(Band {
        index,
        units,
        amount,
    });
    Ok(())
}

fn tier_charge(quantity: u64, tiers: &[Tier], mode: &Mode) -> Result<Charge, Error> {
    let mut charge = Charge {
        amount: 0,
        bands: Vec::new(),
    };
    if quantity == 0 {
        return Ok(charge);
    }

    match mode {
        Mode::Volume => {
            let index = tiers
                .iter()
                .position(|tier| tier.up_to.is_none_or(|bound| quantity <= bound))
                .expect("validated schedule has a final unbounded tier");
            add_band(
                &mut charge.amount,
                &mut charge.bands,
                index,
                quantity,
                &tiers[index],
            )?;
        }
        Mode::Graduated => {
            let mut remaining = quantity;
            let mut previous = 0u64;

            for (index, tier) in tiers.iter().enumerate() {
                if remaining == 0 {
                    break;
                }

                let allocated = if let Some(bound) = tier.up_to {
                    let width = bound - previous;
                    previous = bound;
                    remaining.min(width)
                } else {
                    remaining
                };

                if allocated != 0 {
                    add_band(
                        &mut charge.amount,
                        &mut charge.bands,
                        index,
                        allocated,
                        tier,
                    )?;
                    remaining -= allocated;
                }
            }
        }
    }

    Ok(charge)
}

fn flat_charge(quantity: u64, unit: u64, start: u64) -> Result<Charge, Error> {
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    let amount = quantity
        .checked_mul(unit)
        .and_then(|value| value.checked_add(start))
        .ok_or(Error::Overflow { index: 0 })?;
    Ok(Charge {
        amount,
        bands: vec![Band {
            index: 0,
            units: quantity,
            amount,
        }],
    })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    validate(price)?;
    match price {
        Price::Flat { unit, start } => flat_charge(quantity, *unit, *start),
        Price::Tiers(tiers) => tier_charge(quantity, tiers, &Mode::Graduated),
    }
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    validate(price)?;

    let charge = match price {
        Price::Flat { unit, start } => flat_charge(quantity, *unit, *start)?,
        Price::Tiers(tiers) => tier_charge(quantity, tiers, &mode)?,
    };

    let minimum_adjustment = if quantity == 0 {
        0
    } else {
        minimum.saturating_sub(charge.amount)
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .ok_or(Error::Overflow { index: 0 })?;

    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}
