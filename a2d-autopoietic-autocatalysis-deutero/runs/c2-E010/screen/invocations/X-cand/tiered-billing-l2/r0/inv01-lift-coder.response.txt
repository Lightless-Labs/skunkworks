- Validation completes before pricing; empty and oversized schedules take precedence.
- `None` is accepted only on the final tier; bounds are strictly increasing from zero.
- Finite bounds are inclusive, and a tier at its exact bound does not activate the next tier.
- Zero quantity yields no bands and skips the minimum, but still validates tier schedules.
- Overflow is attributed to the original tier index and is checked before minimum adjustment.
- Both solutions use the same interpretation; the requirements specify these behaviors directly, so there is no alternate reading to apply.

### Solution A

```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
    Graduated,
    Volume,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Bill {
    pub charge: Charge,
    pub minimum_adjustment: u64,
    pub total: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous {
                    return Err(Error::Bound { index });
                }
                previous = bound;
            }
            None => {
                if index + 1 != tiers.len() {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers.last().and_then(|tier| tier.up_to).is_some() {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn add_band(
    total: &mut u64,
    bands: &mut Vec<Band>,
    index: usize,
    units: u64,
    unit: u64,
    start: u64,
) -> Result<(), Error> {
    let amount = units
        .checked_mul(unit)
        .and_then(|product| product.checked_add(start))
        .ok_or(Error::Overflow { index })?;
    let new_total = total
        .checked_add(amount)
        .ok_or(Error::Overflow { index })?;
    *total = new_total;
    bands.push(Band {
        index,
        units,
        amount,
    });
    Ok(())
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                return Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                });
            }
            let mut amount = 0;
            let mut bands = Vec::new();
            add_band(&mut amount, &mut bands, 0, quantity, *unit, *start)?;
            Ok(Charge { amount, bands })
        }
        Price::Tiers(tiers) => {
            validate_tiers(tiers)?;
            let mut total = 0u64;
            let mut bands = Vec::new();
            let mut remaining = quantity;
            let mut previous = 0u64;

            for (index, tier) in tiers.iter().enumerate() {
                if remaining == 0 {
                    break;
                }
                let units = match tier.up_to {
                    Some(bound) => {
                        let width = bound - previous;
                        previous = bound;
                        remaining.min(width)
                    }
                    None => remaining,
                };
                if units > 0 {
                    add_band(
                        &mut total,
                        &mut bands,
                        index,
                        units,
                        tier.unit,
                        tier.start,
                    )?;
                    remaining -= units;
                }
            }

            Ok(Charge {
                amount: total,
                bands,
            })
        }
    }
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = match (&mode, price) {
        (_, Price::Flat { .. }) | (Mode::Graduated, _) => quote(quantity, price)?,
        (Mode::Volume, Price::Tiers(tiers)) => {
            validate_tiers(tiers)?;
            if quantity == 0 {
                Charge {
                    amount: 0,
                    bands: Vec::new(),
                }
            } else {
                let (index, tier) = tiers
                    .iter()
                    .enumerate()
                    .find(|(_, tier)| tier.up_to.map_or(true, |bound| bound >= quantity))
                    .expect("validated schedule has a final unbounded tier");
                let mut amount = 0;
                let mut bands = Vec::new();
                add_band(
                    &mut amount,
                    &mut bands,
                    index,
                    quantity,
                    tier.unit,
                    tier.start,
                )?;
                Charge { amount, bands }
            }
        }
    };

    let minimum_adjustment = if quantity > 0 {
        minimum.saturating_sub(charge.amount)
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .expect("minimum adjustment produces a total no greater than minimum or charge");
    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}
```

```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
    Graduated,
    Volume,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Bill {
    pub charge: Charge,
    pub minimum_adjustment: u64,
    pub total: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn validate(tiers: &[Tier]) -> Result<(), Error> {
    match tiers.len() {
        0 => return Err(Error::Empty),
        65.. => return Err(Error::TooMany),
        _ => {}
    }

    let mut last_bound = 0;
    for (i, tier) in tiers.iter().enumerate() {
        if let Some(bound) = tier.up_to {
            if bound <= last_bound {
                return Err(Error::Bound { index: i });
            }
            last_bound = bound;
        } else if i != tiers.len() - 1 {
            return Err(Error::Unbounded { index: i });
        }
    }

    if tiers[tiers.len() - 1].up_to.is_some() {
        Err(Error::MissingFinal)
    } else {
        Ok(())
    }
}

fn make_band(
    sum: &mut u64,
    output: &mut Vec<Band>,
    index: usize,
    units: u64,
    tier: &Tier,
) -> Result<(), Error> {
    let amount = units
        .checked_mul(tier.unit)
        .and_then(|n| n.checked_add(tier.start))
        .ok_or(Error::Overflow { index })?;
    *sum = sum.checked_add(amount).ok_or(Error::Overflow { index })?;
    output.push(Band {
        index,
        units,
        amount,
    });
    Ok(())
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    let mut sum = 0u64;
    let mut bands = Vec::new();

    match price {
        Price::Flat { unit, start } => {
            if quantity != 0 {
                let amount = quantity
                    .checked_mul(*unit)
                    .and_then(|n| n.checked_add(*start))
                    .ok_or(Error::Overflow { index: 0 })?;
                sum = amount;
                bands.push(Band {
                    index: 0,
                    units: quantity,
                    amount,
                });
            }
        }
        Price::Tiers(tiers) => {
            validate(tiers)?;
            let mut left = quantity;
            let mut lower = 0u64;
            for (index, tier) in tiers.iter().enumerate() {
                if left == 0 {
                    break;
                }
                let allocated = match tier.up_to {
                    Some(upper) => {
                        let capacity = upper - lower;
                        lower = upper;
                        left.min(capacity)
                    }
                    None => left,
                };
                if allocated != 0 {
                    make_band(&mut sum, &mut bands, index, allocated, tier)?;
                    left -= allocated;
                }
            }
        }
    }

    Ok(Charge {
        amount: sum,
        bands,
    })
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = match (mode, price) {
        (_, Price::Flat { .. }) | (Mode::Graduated, _) => quote(quantity, price)?,
        (Mode::Volume, Price::Tiers(tiers)) => {
            validate(tiers)?;
            let mut charge = Charge {
                amount: 0,
                bands: Vec::new(),
            };
            if quantity != 0 {
                let selected = tiers
                    .iter()
                    .enumerate()
                    .find(|(_, t)| match t.up_to {
                        Some(bound) => bound >= quantity,
                        None => true,
                    });
                if let Some((index, tier)) = selected {
                    make_band(
                        &mut charge.amount,
                        &mut charge.bands,
                        index,
                        quantity,
                        tier,
                    )?;
                }
            }
            charge
        }
    };

    let minimum_adjustment = if quantity == 0 {
        0
    } else if minimum > charge.amount {
        minimum - charge.amount
    } else {
        0
    };
    let total = charge.amount + minimum_adjustment;
    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}
```
