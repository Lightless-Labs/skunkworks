- Error precedence: I use Empty, SignOnly, InvalidDigit, Separator, LeadingZero, then TooManyDigits, as specified.
- Invalid bytes are scanned across the whole magnitude before separator checks; UTF-8 errors point to the first byte of the character.
- Separators must have digit bytes immediately on both sides; digit limits count digits only.
- Zero is canonical and unsigned; padding is applied before grouping, and neither affects the stored value.
- Magnitudes use little-endian decimal digits, so arithmetic has no machine-integer size bound.
- Both solutions use the same specified behavior; Solution B is an independent implementation using base-1,000,000,000 limbs.

### Solution A

```rust
use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    digits: Vec<u8>, // little-endian decimal digits
}

impl fmt::Debug for Big {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_struct("Big")
            .field("negative", &self.negative)
            .field("digits", &self.digits)
            .finish()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

impl Big {
    fn zero() -> Self {
        Self { negative: false, digits: vec![0] }
    }

    fn normalize(&mut self) {
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.len() == 1 && self.digits[0] == 0 {
            self.negative = false;
        }
    }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let sign_len = if bytes[0] == b'+' || bytes[0] == b'-' { 1 } else { 0 };
        if sign_len == 1 && bytes.len() == 1 {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for i in sign_len..bytes.len() {
            let b = bytes[i];
            if b != b'_' && !b.is_ascii_digit() {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut malformed_sep = None;
        for i in sign_len..bytes.len() {
            if bytes[i] == b'_' {
                let left_ok = i > sign_len && bytes[i - 1].is_ascii_digit();
                let right_ok = i + 1 < bytes.len() && bytes[i + 1].is_ascii_digit();
                if !left_ok || !right_ok {
                    malformed_sep = Some(i);
                    break;
                }
            }
        }
        if let Some(index) = malformed_sep {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[sign_len..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (sign_len..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero { index: first_digit_index });
        }
        if digit_count > max_digits {
            let excess = max_digits;
            let index = (sign_len..bytes.len())
                .filter(|&i| bytes[i] != b'_')
                .nth(excess)
                .unwrap();
            return Err(Error::TooManyDigits { limit: max_digits, index });
        }

        let mut digits = Vec::with_capacity(digit_count.max(1));
        for &b in bytes[sign_len..].iter().rev() {
            if b != b'_' {
                digits.push(b - b'0');
            }
        }
        let mut result = Self { negative: bytes[0] == b'-', digits };
        result.normalize();
        Ok(result)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude_len = self.digits.len().max(min_digits).max(1);
        let padding = magnitude_len - self.digits.len();
        let mut magnitude = String::with_capacity(magnitude_len + magnitude_len / group.max(1));
        for i in (0..magnitude_len).rev() {
            if group > 0 && i + 1 < magnitude_len && (i + 1) % group == 0 {
                magnitude.push('_');
            }
            let digit = if i >= self.digits.len() { 0 } else { self.digits[i] };
            let _ = padding;
            magnitude.push((b'0' + digit) as char);
        }
        if self.negative { format!("-{magnitude}") } else { magnitude }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => Self::cmp_mag(&self.digits, &rhs.digits),
            (true, true) => Self::cmp_mag(&rhs.digits, &self.digits),
        }
    }

    fn cmp_mag(a: &[u8], b: &[u8]) -> Ordering {
        a.len().cmp(&b.len()).then_with(|| {
            for (x, y) in a.iter().rev().zip(b.iter().rev()) {
                match x.cmp(y) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        })
    }

    fn add_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u8;
        for i in 0..a.len().max(b.len()) {
            let n = a.get(i).copied().unwrap_or(0)
                + b.get(i).copied().unwrap_or(0) + carry;
            out.push(n % 10);
            carry = n / 10;
        }
        if carry != 0 { out.push(carry); }
        out
    }

    fn sub_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
        // Requires a >= b.
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i8;
        for i in 0..a.len() {
            let mut n = a[i] as i8 - b.get(i).copied().unwrap_or(0) as i8 - borrow;
            if n < 0 {
                n += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(n as u8);
        }
        while out.len() > 1 && out.last() == Some(&0) { out.pop(); }
        out
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self { negative: self.negative, digits: Self::add_mag(&self.digits, &rhs.digits) };
            out.normalize();
            out
        } else {
            match Self::cmp_mag(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self { negative: self.negative, digits: Self::sub_mag(&self.digits, &rhs.digits) };
                    out.normalize();
                    out
                }
                Ordering::Less => {
                    let mut out = Self { negative: rhs.negative, digits: Self::sub_mag(&rhs.digits, &self.digits) };
                    out.normalize();
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !(out.digits.len() == 1 && out.digits[0] == 0) {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.digits.len() == 1 && self.digits[0] == 0 { 0 }
        else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.signum() == 0 || rhs.signum() == 0 { return Self::zero(); }
        let mut accum = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            let mut carry = 0u32;
            for (j, &b) in rhs.digits.iter().enumerate() {
                let k = i + j;
                let n = accum[k] + (a as u32) * (b as u32) + carry;
                accum[k] = n % 10;
                carry = n / 10;
            }
            let mut k = i + rhs.digits.len();
            while carry != 0 {
                let n = accum[k] + carry;
                accum[k] = n % 10;
                carry = n / 10;
                k += 1;
                if k == accum.len() && carry != 0 { accum.push(0); }
            }
        }
        let mut out = Self {
            negative: self.negative != rhs.negative,
            digits: accum.into_iter().map(|x| x as u8).collect(),
        };
        out.normalize();
        out
    }
}
```

### Solution B

```rust
use std::cmp::Ordering;
use std::fmt;

const BASE: u64 = 1_000_000_000;

#[derive(Clone, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    limbs: Vec<u32>, // little-endian base 10^9
}

impl fmt::Debug for Big {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_struct("Big")
            .field("negative", &self.negative)
            .field("limbs", &self.limbs)
            .finish()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

impl Big {
    fn zero() -> Self {
        Self { negative: false, limbs: vec![0] }
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.len() == 1 && self.limbs[0] == 0 {
            self.negative = false;
        }
    }

    fn cmp_mag(a: &[u32], b: &[u32]) -> Ordering {
        a.len().cmp(&b.len()).then_with(|| {
            for (x, y) in a.iter().rev().zip(b.iter().rev()) {
                match x.cmp(y) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        })
    }

    fn add_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let n = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64 + carry;
            out.push((n % BASE) as u32);
            carry = n / BASE;
        }
        if carry != 0 { out.push(carry as u32); }
        out
    }

    fn sub_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let mut n = a[i] as i64 - b.get(i).copied().unwrap_or(0) as i64 - borrow;
            if n < 0 {
                n += BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(n as u32);
        }
        while out.len() > 1 && out.last() == Some(&0) { out.pop(); }
        out
    }

    fn mul_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut accum = vec![0u128; a.len() + b.len()];
        for (i, &x) in a.iter().enumerate() {
            let mut carry = 0u128;
            for (j, &y) in b.iter().enumerate() {
                let k = i + j;
                let n = accum[k] + (x as u128) * (y as u128) + carry;
                accum[k] = n % BASE as u128;
                carry = n / BASE as u128;
            }
            let mut k = i + b.len();
            while carry != 0 {
                if k == accum.len() { accum.push(0); }
                let n = accum[k] + carry;
                accum[k] = n % BASE as u128;
                carry = n / BASE as u128;
                k += 1;
            }
        }
        let mut out: Vec<u32> = accum.into_iter().map(|x| x as u32).collect();
        while out.len() > 1 && out.last() == Some(&0) { out.pop(); }
        out
    }

    fn add_small(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() { self.limbs.push(0); }
            let n = self.limbs[i] as u64 + carry;
            self.limbs[i] = (n % BASE) as u32;
            carry = n / BASE;
            i += 1;
        }
    }

    fn mul_small_add(&mut self, multiplier: u32, add: u32) {
        let mut carry = add as u64;
        for limb in &mut self.limbs {
            let n = *limb as u64 * multiplier as u64 + carry;
            *limb = (n % BASE) as u32;
            carry = n / BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % BASE) as u32);
            carry /= BASE;
        }
    }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() { return Err(Error::Empty); }
        let sign_len = usize::from(bytes[0] == b'+' || bytes[0] == b'-');
        if sign_len == 1 && bytes.len() == 1 { return Err(Error::SignOnly); }

        for i in sign_len..bytes.len() {
            let b = bytes[i];
            if b != b'_' && !b.is_ascii_digit() {
                return Err(Error::InvalidDigit { index: i });
            }
        }

        for i in sign_len..bytes.len() {
            if bytes[i] == b'_' {
                let left = i > sign_len && bytes[i - 1].is_ascii_digit();
                let right = i + 1 < bytes.len() && bytes[i + 1].is_ascii_digit();
                if !left || !right { return Err(Error::Separator { index: i }); }
            }
        }

        let digit_count = bytes[sign_len..].iter().filter(|&&b| b != b'_').count();
        let first = (sign_len..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if digit_count > 1 && bytes[first] == b'0' {
            return Err(Error::LeadingZero { index: first });
        }
        if digit_count > max_digits {
            let index = (sign_len..bytes.len())
                .filter(|&i| bytes[i] != b'_')
                .nth(max_digits).unwrap();
            return Err(Error::TooManyDigits { limit: max_digits, index });
        }

        let mut result = Self::zero();
        result.limbs.clear();
        for &b in &bytes[sign_len..] {
            if b != b'_' {
                result.mul_small_add(10, (b - b'0') as u32);
            }
        }
        result.negative = bytes[0] == b'-';
        result.normalize();
        Ok(result)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let mut raw = if self.limbs.len() == 1 {
            format!("{}", self.limbs[0])
        } else {
            let mut s = self.limbs.last().unwrap().to_string();
            for limb in self.limbs.iter().rev().skip(1) {
                s.push_str(&format!("{:09}", limb));
            }
            s
        };
        if raw.len() < min_digits {
            let mut padded = String::with_capacity(min_digits);
            padded.extend(std::iter::repeat_n('0', min_digits - raw.len()));
            padded.push_str(&raw);
            raw = padded;
        }
        if group > 0 && raw.len() > group {
            let first_group = raw.len() % group;
            let first_group = if first_group == 0 { group } else { first_group };
            let mut grouped = String::with_capacity(raw.len() + (raw.len() - 1) / group);
            grouped.push_str(&raw[..first_group]);
            let mut i = first_group;
            while i < raw.len() {
                grouped.push('_');
                grouped.push_str(&raw[i..i + group]);
                i += group;
            }
            raw = grouped;
        }
        if self.negative { format!("-{raw}") } else { raw }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => Self::cmp_mag(&self.limbs, &rhs.limbs),
            (true, true) => Self::cmp_mag(&rhs.limbs, &self.limbs),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self { negative: self.negative, limbs: Self::add_mag(&self.limbs, &rhs.limbs) };
            out.normalize();
            out
        } else {
            match Self::cmp_mag(&self.limbs, &rhs.limbs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self { negative: self.negative, limbs: Self::sub_mag(&self.limbs, &rhs.limbs) };
                    out.normalize();
                    out
                }
                Ordering::Less => {
                    let mut out = Self { negative: rhs.negative, limbs: Self::sub_mag(&rhs.limbs, &self.limbs) };
                    out.normalize();
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self { self.add(&rhs.neg()) }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !(out.limbs.len() == 1 && out.limbs[0] == 0) { out.negative = !out.negative; }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.limbs.len() == 1 && self.limbs[0] == 0 { 0 }
        else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.signum() == 0 || rhs.signum() == 0 { return Self::zero(); }
        let mut out = Self {
            negative: self.negative != rhs.negative,
            limbs: Self::mul_mag(&self.limbs, &rhs.limbs),
        };
        out.normalize();
        out
    }
}
```
