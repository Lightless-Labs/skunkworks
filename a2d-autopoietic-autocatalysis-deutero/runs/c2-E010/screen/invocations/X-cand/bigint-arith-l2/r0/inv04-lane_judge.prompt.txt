use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    digits: Vec<u8>, // Big-endian decimal digits; zero is [0].
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' || bytes[0] == b'-' { 1 } else { 0 };
        if start == bytes.len() && start == 1 {
            return Err(Error::SignOnly);
        }

        for i in start..bytes.len() {
            if bytes[i] != b'_' && !bytes[i].is_ascii_digit() {
                return Err(Error::InvalidDigit { index: i });
            }
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index: i });
            }
        }

        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero { index: first_digit_index });
        }

        if digit_count > max_digits {
            let wanted = max_digits.saturating_add(1);
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen == wanted {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
            // Covers usize::MAX limits without relying on addition overflow.
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: bytes.len() - 1,
            });
        }

        let digits = bytes[start..]
            .iter()
            .filter(|&&b| b != b'_')
            .map(|&b| b - b'0')
            .collect::<Vec<_>>();

        let is_zero = digits.iter().all(|&d| d == 0);
        Ok(Self {
            negative: bytes[0] == b'-' && !is_zero,
            digits,
        })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = self.digits.len().max(1).max(min_digits);
        let pad = width - self.digits.len();
        let mut out = String::with_capacity(width + width / group.max(1) + usize::from(self.negative));
        if self.negative {
            out.push('-');
        }

        for i in 0..width {
            if group > 0 && i > 0 && (width - i) % group == 0 {
                out.push('_');
            }
            let digit = if i < pad { 0 } else { self.digits[i - pad] };
            out.push((b'0' + digit) as char);
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_mag(&self.digits, &rhs.digits),
            (true, true) => cmp_mag(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::from_parts(self.negative, add_mag(&self.digits, &rhs.digits))
        } else {
            match cmp_mag(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => Self::from_parts(self.negative, sub_mag(&self.digits, &rhs.digits)),
                Ordering::Less => Self::from_parts(rhs.negative, sub_mag(&rhs.digits, &self.digits)),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            Self::zero()
        } else {
            Self { negative: !self.negative, digits: self.digits.clone() }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        Self::from_parts(self.negative ^ rhs.negative, mul_mag(&self.digits, &rhs.digits))
    }

    fn zero() -> Self {
        Self { negative: false, digits: vec![0] }
    }

    fn is_zero(&self) -> bool {
        self.digits.iter().all(|&d| d == 0)
    }

    fn from_parts(negative: bool, digits: Vec<u8>) -> Self {
        let first = digits.iter().position(|&d| d != 0).unwrap_or(digits.len().saturating_sub(1));
        let digits = digits[first..].to_vec();
        let zero = digits.len() == 1 && digits[0] == 0;
        Self { negative: negative && !zero, digits }
    }
}

fn cmp_mag(a: &[u8], b: &[u8]) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.cmp(b))
}

fn add_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
    let (mut i, mut j, mut carry) = (a.len(), b.len(), 0u8);
    while i > 0 || j > 0 || carry > 0 {
        let x = if i > 0 { i -= 1; a[i] } else { 0 };
        let y = if j > 0 { j -= 1; b[j] } else { 0 };
        let sum = x + y + carry;
        out.push(sum % 10);
        carry = sum / 10;
    }
    out.reverse();
    out
}

// Requires a >= b.
fn sub_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len());
    let (mut i, mut j, mut borrow) = (a.len(), b.len(), 0i8);
    while i > 0 {
        i -= 1;
        let x = a[i] as i8 - borrow;
        let y = if j > 0 { j -= 1; b[j] as i8 } else { 0 };
        if x < y {
            out.push((x + 10 - y) as u8);
            borrow = 1;
        } else {
            out.push((x - y) as u8);
            borrow = 0;
        }
    }
    out.reverse();
    let first = out.iter().position(|&d| d != 0).unwrap_or(out.len().saturating_sub(1));
    out[first..].to_vec()
}

fn mul_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut acc = vec![0u32; a.len() + b.len()];
    for i in (0..a.len()).rev() {
        for j in (0..b.len()).rev() {
            acc[i + j + 1] += (a[i] as u32) * (b[j] as u32);
        }
    }
    for k in (1..acc.len()).rev() {
        let carry = acc[k] / 10;
        acc[k] %= 10;
        acc[k - 1] += carry;
    }
    let digits = acc.into_iter().map(|x| x as u8).collect::<Vec<_>>();
    let first = digits.iter().position(|&d| d != 0).unwrap_or(digits.len() - 1);
    digits[first..].to_vec()
}

use std::fmt;

#[derive(Clone, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Big-endian decimal digits; zero is represented as [0].
    digits: Vec<u8>,
}

impl fmt::Debug for Big {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_struct("Big")
            .field("negative", &self.negative)
            .field("digits", &self.digits)
            .finish()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = if bytes[0] == b'-' {
            (true, 1)
        } else if bytes[0] == b'+' {
            (false, 1)
        } else {
            (false, 0)
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut positions = Vec::new();
        for (index, byte) in bytes.iter().copied().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            if !(b'0'..=b'9').contains(&byte) {
                return Err(Error::InvalidDigit { index });
            }
            positions.push(index);
        }

        for index in start..bytes.len() {
            if bytes[index] == b'_' {
                let valid_left = index > start && (b'0'..=b'9').contains(&bytes[index - 1]);
                let valid_right = index + 1 < bytes.len()
                    && (b'0'..=b'9').contains(&bytes[index + 1]);
                if !valid_left || !valid_right {
                    return Err(Error::Separator { index });
                }
            }
        }

        if positions.len() > 1 && bytes[positions[0]] == b'0' {
            return Err(Error::LeadingZero {
                index: positions[0],
            });
        }

        if positions.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: positions[max_digits],
            });
        }

        let digits: Vec<u8> = positions
            .into_iter()
            .map(|index| bytes[index] - b'0')
            .collect();
        let is_zero = digits.iter().all(|&digit| digit == 0);

        Ok(Self {
            negative: negative && !is_zero,
            digits: if digits.is_empty() { vec![0] } else { digits },
        })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude_len = self.digits.len().max(min_digits).max(1);
        let zero_padding = magnitude_len - self.digits.len();
        let separators = if group == 0 {
            0
        } else {
            (magnitude_len - 1) / group
        };
        let mut output = String::with_capacity(
            magnitude_len + separators + usize::from(self.negative),
        );

        if self.negative {
            output.push('-');
        }

        for index in 0..magnitude_len {
            if group > 0 && index > 0 && (magnitude_len - index) % group == 0 {
                output.push('_');
            }
            let digit = if index < zero_padding {
                0
            } else {
                self.digits[index - zero_padding]
            };
            output.push(char::from(b'0' + digit));
        }

        output
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        if self.negative != rhs.negative {
            return if self.negative {
                Ordering::Less
            } else {
                Ordering::Greater
            };
        }

        let magnitude_order = cmp_big_endian(&self.digits, &rhs.digits);
        if self.negative {
            magnitude_order.reverse()
        } else {
            magnitude_order
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::from_parts(
                self.negative,
                add_big_endian(&self.digits, &rhs.digits),
            )
        } else {
            match cmp_big_endian(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => Self::from_parts(
                    self.negative,
                    sub_big_endian(&self.digits, &rhs.digits),
                ),
                Ordering::Less => Self::from_parts(
                    rhs.negative,
                    sub_big_endian(&rhs.digits, &self.digits),
                ),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        let mut flipped = rhs.clone();
        if !flipped.is_zero() {
            flipped.negative = !flipped.negative;
        }
        self.add(&flipped)
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let n = self.digits.len();
        let m = rhs.digits.len();
        let mut result = vec![0u8; n + m];

        for i in (0..n).rev() {
            let mut carry = 0u32;
            for j in (0..m).rev() {
                let slot = i + j + 1;
                let total = u32::from(self.digits[i]) * u32::from(rhs.digits[j])
                    + u32::from(result[slot])
                    + carry;
                result[slot] = (total % 10) as u8;
                carry = total / 10;
            }
            result[i] = (u32::from(result[i]) + carry) as u8;
        }

        Self::from_parts(self.negative != rhs.negative, result)
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn from_parts(negative: bool, mut digits: Vec<u8>) -> Self {
        while digits.len() > 1 && digits[0] == 0 {
            digits.remove(0);
        }
        if digits.is_empty() {
            digits.push(0);
        }
        let is_zero = digits.len() == 1 && digits[0] == 0;
        Self {
            negative: negative && !is_zero,
            digits,
        }
    }
}

fn cmp_big_endian(a: &[u8], b: &[u8]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => a.cmp(b),
        other => other,
    }
}

fn add_big_endian(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut i = a.len();
    let mut j = b.len();
    let mut carry = 0u16;
    let mut reversed = Vec::with_capacity(a.len().max(b.len()) + 1);

    while i > 0 || j > 0 || carry > 0 {
        let x = if i > 0 {
            i -= 1;
            u16::from(a[i])
        } else {
            0
        };
        let y = if j > 0 {
            j -= 1;
            u16::from(b[j])
        } else {
            0
        };
        let sum = x + y + carry;
        reversed.push((sum % 10) as u8);
        carry = sum / 10;
    }

    reversed.reverse();
    reversed
}

fn sub_big_endian(a: &[u8], b: &[u8]) -> Vec<u8> {
    // Requires a >= b.
    let mut i = a.len();
    let mut j = b.len();
    let mut borrow = 0i16;
    let mut reversed = Vec::with_capacity(a.len());

    while i > 0 {
        i -= 1;
        let x = i16::from(a[i]);
        let y = if j > 0 {
            j -= 1;
            i16::from(b[j])
        } else {
            0
        };

        let mut digit = x - y - borrow;
        if digit < 0 {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        reversed.push(digit as u8);
    }

    while reversed.len() > 1 && reversed.last() == Some(&0) {
        reversed.pop();
    }
    reversed.reverse();
    reversed
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    digits: Vec<u8>, // little-endian, canonical: [0] or no high zero
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let begin = usize::from(bytes[0] == b'+' || bytes[0] == b'-');
        if begin == bytes.len() {
            return Err(Error::SignOnly);
        }
        let negative = begin == 1 && bytes[0] == b'-';

        for (i, &byte) in bytes.iter().enumerate().skip(begin) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index: i });
            }
        }

        for i in begin..bytes.len() {
            if bytes[i] == b'_'
                && (i == begin
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index: i });
            }
        }

        let count = bytes[begin..].iter().filter(|&&b| b != b'_').count();
        let first_digit = (begin..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero { index: first_digit });
        }
        if count > max_digits {
            let target = max_digits.checked_add(1);
            let mut seen = 0usize;
            for i in begin..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if Some(seen) == target {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut digits: Vec<u8> = bytes[begin..]
            .iter()
            .filter(|&&b| b != b'_')
            .map(|&b| b - b'0')
            .collect();
        digits.reverse();
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        let zero = digits.len() == 1 && digits[0] == 0;
        Ok(Self { negative: negative && !zero, digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = min_digits.max(self.digits.len()).max(1);
        let pad = width - self.digits.len();
        let mut out = String::new();
        if self.negative {
            out.push('-');
        }
        for i in (0..width).rev() {
            let remaining = i + 1;
            if group > 0 && remaining < width && remaining % group == 0 {
                out.push('_');
            }
            if i >= self.digits.len() {
                out.push('0');
            } else {
                out.push((b'0' + self.digits[i]) as char);
            }
        }
        let _ = pad;
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_le(&self.digits, &rhs.digits),
            (true, true) => cmp_le(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::make(add_le(&self.digits, &rhs.digits), self.negative)
        } else {
            match cmp_le(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => Self::make(sub_le(&self.digits, &rhs.digits), self.negative),
                Ordering::Less => Self::make(sub_le(&rhs.digits, &self.digits), rhs.negative),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut answer = self.clone();
        if !answer.is_zero() {
            answer.negative = !answer.negative;
        }
        answer
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut acc = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &x) in self.digits.iter().enumerate() {
            for (j, &y) in rhs.digits.iter().enumerate() {
                acc[i + j] += u32::from(x) * u32::from(y);
            }
        }
        for i in 0..acc.len() - 1 {
            acc[i + 1] += acc[i] / 10;
            acc[i] %= 10;
        }
        let digits = acc.into_iter().map(|v| (v % 10) as u8).collect();
        Self::make(digits, self.negative ^ rhs.negative)
    }

    fn zero() -> Self {
        Self { negative: false, digits: vec![0] }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn make(mut digits: Vec<u8>, negative: bool) -> Self {
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        if digits.is_empty() {
            digits.push(0);
        }
        let zero = digits.len() == 1 && digits[0] == 0;
        Self { negative: negative && !zero, digits }
    }
}

fn cmp_le(a: &[u8], b: &[u8]) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.iter().rev().cmp(b.iter().rev()))
}

fn add_le(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u8;
    for i in 0..a.len().max(b.len()) {
        let sum = a.get(i).copied().unwrap_or(0) + b.get(i).copied().unwrap_or(0) + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }
    if carry != 0 {
        result.push(carry);
    }
    result
}

fn sub_le(a: &[u8], b: &[u8]) -> Vec<u8> {
    // Caller guarantees a >= b.
    let mut result = Vec::with_capacity(a.len());
    let mut borrow = 0i16;
    for i in 0..a.len() {
        let mut d = i16::from(a[i]) - i16::from(b.get(i).copied().unwrap_or(0)) - borrow;
        if d < 0 {
            d += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push(d as u8);
    }
    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

#[derive(Clone, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    digits: Vec<u8>, // little-endian decimal digits
}

impl fmt::Debug for Big {

            .field("digits", &self.digits)

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

impl Big {
    fn zero() -> Self {
        Self { negative: false, digits: vec![0] }
    }

    fn normalize(&mut self) {
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.len() == 1 && self.digits[0] == 0 {
            self.negative = false;
        }
    }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let sign_len = if bytes[0] == b'+' || bytes[0] == b'-' { 1 } else { 0 };
        if sign_len == 1 && bytes.len() == 1 {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for i in sign_len..bytes.len() {
            let b = bytes[i];
            if b != b'_' && !b.is_ascii_digit() {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut malformed_sep = None;
        for i in sign_len..bytes.len() {
            if bytes[i] == b'_' {
                let left_ok = i > sign_len && bytes[i - 1].is_ascii_digit();
                let right_ok = i + 1 < bytes.len() && bytes[i + 1].is_ascii_digit();
                if !left_ok || !right_ok {
                    malformed_sep = Some(i);
                    break;
                }
            }
        }
        if let Some(index) = malformed_sep {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[sign_len..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (sign_len..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero { index: first_digit_index });
        }
        if digit_count > max_digits {
            let excess = max_digits;
            let index = (sign_len..bytes.len())
                .filter(|&i| bytes[i] != b'_')
                .nth(excess)
                .unwrap();
            return Err(Error::TooManyDigits { limit: max_digits, index });
        }

        let mut digits = Vec::with_capacity(digit_count.max(1));
        for &b in bytes[sign_len..].iter().rev() {
            if b != b'_' {
                digits.push(b - b'0');
            }
        }
        let mut result = Self { negative: bytes[0] == b'-', digits };
        result.normalize();
        Ok(result)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude_len = self.digits.len().max(min_digits).max(1);
        let padding = magnitude_len - self.digits.len();
        let mut magnitude = String::with_capacity(magnitude_len + magnitude_len / group.max(1));
        for i in (0..magnitude_len).rev() {
            if group > 0 && i + 1 < magnitude_len && (i + 1) % group == 0 {
                magnitude.push('_');
            }
            let digit = if i >= self.digits.len() { 0 } else { self.digits[i] };
            let _ = padding;
            magnitude.push((b'0' + digit) as char);
        }
        if self.negative { format!("-{magnitude}") } else { magnitude }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => Self::cmp_mag(&self.digits, &rhs.digits),
            (true, true) => Self::cmp_mag(&rhs.digits, &self.digits),
        }
    }

    fn cmp_mag(a: &[u8], b: &[u8]) -> Ordering {
        a.len().cmp(&b.len()).then_with(|| {
            for (x, y) in a.iter().rev().zip(b.iter().rev()) {
                match x.cmp(y) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        })
    }

    fn add_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u8;
        for i in 0..a.len().max(b.len()) {
            let n = a.get(i).copied().unwrap_or(0)
                + b.get(i).copied().unwrap_or(0) + carry;
            out.push(n % 10);
            carry = n / 10;
        }
        if carry != 0 { out.push(carry); }
        out
    }

    fn sub_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
        // Requires a >= b.
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i8;
        for i in 0..a.len() {
            let mut n = a[i] as i8 - b.get(i).copied().unwrap_or(0) as i8 - borrow;
            if n < 0 {
                n += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(n as u8);
        }
        while out.len() > 1 && out.last() == Some(&0) { out.pop(); }
        out
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self { negative: self.negative, digits: Self::add_mag(&self.digits, &rhs.digits) };
            out.normalize();
            out
        } else {
            match Self::cmp_mag(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self { negative: self.negative, digits: Self::sub_mag(&self.digits, &rhs.digits) };
                    out.normalize();
                    out
                }
                Ordering::Less => {
                    let mut out = Self { negative: rhs.negative, digits: Self::sub_mag(&rhs.digits, &self.digits) };
                    out.normalize();
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !(out.digits.len() == 1 && out.digits[0] == 0) {
            out.negative = !out.negative;
        }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.digits.len() == 1 && self.digits[0] == 0 { 0 }
        else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.signum() == 0 || rhs.signum() == 0 { return Self::zero(); }
        let mut accum = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            let mut carry = 0u32;
            for (j, &b) in rhs.digits.iter().enumerate() {
                let k = i + j;
                let n = accum[k] + (a as u32) * (b as u32) + carry;
                accum[k] = n % 10;
                carry = n / 10;
            }
            let mut k = i + rhs.digits.len();
            while carry != 0 {
                let n = accum[k] + carry;
                accum[k] = n % 10;
                carry = n / 10;
                k += 1;
                if k == accum.len() && carry != 0 { accum.push(0); }
            }
        }
        let mut out = Self {
            negative: self.negative != rhs.negative,
            digits: accum.into_iter().map(|x| x as u8).collect(),
        };
        out.normalize();
        out
    }
}

pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    digits: Vec<u8>, // big-endian, canonical: [0] or no leading zero
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' || bytes[0] == b'-' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }
        let negative = start == 1 && bytes[0] == b'-';

        if let Some(index) = (start..bytes.len())
            .find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())
        {
            return Err(Error::InvalidDigit { index });
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index: i });
            }
        }

        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero { index: first_digit_index });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen == excess_number {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut digits: Vec<u8> = bytes[start..]
            .iter()
            .filter(|&&b| b != b'_')
            .map(|&b| b - b'0')
            .collect();
        let first = digits.iter().position(|&d| d != 0).unwrap_or(digits.len() - 1);
        digits.drain(..first);
        let is_zero = digits.len() == 1 && digits[0] == 0;

        Ok(Self { negative: negative && !is_zero, digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = min_digits.max(self.digits.len()).max(1);
        let padding = width - self.digits.len();
        let mut out = String::with_capacity(width + width.saturating_sub(1) / group.max(1) + usize::from(self.negative));
        if self.negative {
            out.push('-');
        }
        for pos in 0..width {
            if group > 0 && pos > 0 && (width - pos) % group == 0 {
                out.push('_');
            }
            if pos < padding {
                out.push('0');
            } else {
                out.push((b'0' + self.digits[pos - padding]) as char);
            }
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {

            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_mag(&self.digits, &rhs.digits),
            (true, true) => cmp_mag(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        match (self.negative, rhs.negative) {
            (a, b) if a == b => Self::from_mag(add_mag(&self.digits, &rhs.digits), a),
            _ => match cmp_mag(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => Self::from_mag(sub_mag(&self.digits, &rhs.digits), self.negative),
                Ordering::Less => Self::from_mag(sub_mag(&rhs.digits, &self.digits), rhs.negative),

    pub fn neg(&self) -> Self {

    pub fn signum(&self) -> i8 {

            return Self::zero();
        }
        let mut accum = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().rev().enumerate() {
            for (j, &b) in rhs.digits.iter().rev().enumerate() {
                accum[i + j] += u32::from(a) * u32::from(b);
            }
        }
        for i in 0..accum.len() - 1 {
            let carry = accum[i] / 10;
            accum[i] %= 10;
            accum[i + 1] += carry;
        }
        let mut digits: Vec<u8> = accum.into_iter().map(|x| (x % 10) as u8).collect();
        digits.reverse();
        Self::from_mag(digits, self.negative ^ rhs.negative)
    }

    fn zero() -> Self {
        Self { negative: false, digits: vec![0] }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

        let canonical = digits[first..].to_vec();
        let zero = canonical.len() == 1 && canonical[0] == 0;

fn add_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
    let (mut i, mut j, mut carry) = (a.len(), b.len(), 0u8);
    while i > 0 || j > 0 || carry > 0 {
        let x = if i > 0 { i -= 1; a[i] } else { 0 };
        let y = if j > 0 { j -= 1; b[j] } else { 0 };
        let sum = x + y + carry;
        out.push(sum % 10);
        carry = sum / 10;
    }
    out.reverse();
    out
}

    // Caller guarantees a >= b.
    let mut out = Vec::with_capacity(a.len());
    let (mut i, mut j, mut borrow) = (a.len(), b.len(), 0i8);
    while i > 0 {
        i -= 1;
        let x = i16::from(a[i]) - i16::from(borrow as u8);
        let y = if j > 0 { j -= 1; i16::from(b[j]) } else { 0 };
        let mut d = x - y;
        borrow = 0;
        if d < 0 {
            d += 10;
            borrow = 1;
        }
        out.push(d as u8);
    }
    while out.len() > 1 && out.last() == Some(&0) {
        out.pop();
    }
    out.reverse();

            return Err(Error::Empty);
        }

            return Err(Error::SignOnly);

            (false, false) => cmp_mag(&self.digits, &rhs.digits),
            (true, true) => cmp_mag(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::from_parts(self.negative, add_mag(&self.digits, &rhs.digits))
        } else {
            match cmp_mag(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => Self::from_parts(self.negative, sub_mag(&self.digits, &rhs.digits)),
                Ordering::Less => Self::from_parts(rhs.negative, sub_mag(&rhs.digits, &self.digits)),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            Self::zero()
        } else {
            Self { negative: !self.negative, digits: self.digits.clone() }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        Self::from_parts(self.negative ^ rhs.negative, mul_mag(&self.digits, &rhs.digits))
    }

    fn zero() -> Self {
        Self { negative: false, digits: vec![0] }
    }

    fn is_zero(&self) -> bool {
        self.digits.iter().all(|&d| d == 0)
    }

    fn from_parts(negative: bool, digits: Vec<u8>) -> Self {
        let last_nonzero = digits.iter().rposition(|&d| d != 0);
        let digits = match last_nonzero {
            Some(i) => digits[..=i].to_vec(),
            None => vec![0],
        };
        let zero = digits.len() == 1 && digits[0] == 0;
        Self { negative: negative && !zero, digits }
    }
}

fn cmp_mag(a: &[u8], b: &[u8]) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| {
        for i in (0..a.len()).rev() {
            match a[i].cmp(&b[i]) {
                Ordering::Equal => {}
                other => return other,
            }
        }
        Ordering::Equal
    })
}

fn add_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u8;
    for i in 0..a.len().max(b.len()) {
        let sum = a.get(i).copied().unwrap_or(0) + b.get(i).copied().unwrap_or(0) + carry;
        out.push(sum % 10);
        carry = sum / 10;
    }
    if carry != 0 {
        out.push(carry);
    }
    if out.is_empty() {
        out.push(0);
    }
    out
}

// Requires a >= b.
fn sub_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len());
    let mut borrow = 0i8;
    for i in 0..a.len() {
        let x = a[i] as i8 - borrow;
        let y = b.get(i).copied().unwrap_or(0) as i8;
        if x < y {
            out.push((x + 10 - y) as u8);
            borrow = 1;
        } else {
            out.push((x - y) as u8);
            borrow = 0;
        }
    }
    while out.len() > 1 && out.last() == Some(&0) {
        out.pop();
    }
    out
}

fn mul_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = vec![0u8; a.len() + b.len()];
    for i in 0..a.len() {
        let mut carry = 0u32;
        for j in 0..b.len() {
            let k = i + j;
            let total = out[k] as u32 + (a[i] as u32) * (b[j] as u32) + carry;
            out[k] = (total % 10) as u8;
            carry = total / 10;
        }
        let mut k = i + b.len();
        while carry > 0 {
            if k == out.len() {
                out.push(0);
            }
            let total = out[k] as u32 + carry;
            out[k] = (total % 10) as u8;
            carry = total / 10;
            k += 1;
        }
    }
    while out.len() > 1 && out.last() == Some(&0) {
        out.pop();
    }
    out
}

use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian decimal digits; zero is represented as [0].
    digits: Vec<u8>,
}

impl fmt::Debug for Big {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_struct("Big")
            .field("negative", &self.negative)
            .field("digits", &self.digits)
            .finish()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut digit_positions = Vec::new();

        // Invalid bytes take precedence over malformed separators.
        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b != b'_' {
                if !b.is_ascii_digit() {
                    return Err(Error::InvalidDigit { index: i });
                }
                digit_positions.push(i);
            }
        }

        // Check underscores only after the entire invalid-byte scan succeeds.
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let left_ok = i > start && bytes[i - 1].is_ascii_digit();
                let right_ok = i + 1 < bytes.len() && bytes[i + 1].is_ascii_digit();
                if !left_ok || !right_ok {
                    return Err(Error::Separator { index: i });
                }
            }
        }

        if digit_positions.len() > 1 && bytes[digit_positions[0]] == b'0' {
            return Err(Error::LeadingZero {
                index: digit_positions[0],
            });
        }

        if digit_positions.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_positions[max_digits],
            });
        }

        let mut digits: Vec<u8> = digit_positions
            .iter()
            .rev()
            .map(|&i| bytes[i] - b'0')
            .collect();

        if digits.is_empty() {
            digits.push(0);
        }

        let is_zero = digits.len() == 1 && digits[0] == 0;
        Ok(Self {
            negative: negative && !is_zero,
            digits,
        })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = self.digits.len().max(min_digits).max(1);
        let padding = width - self.digits.len();
        let mut out = String::new();

        if self.negative {
            out.push('-');
        }

        for i in 0..width {
            if group > 0 && i > 0 && (width - i) % group == 0 {
                out.push('_');
            }
            let digit = if i < padding {
                0
            } else {
                self.digits[width - 1 - i]
            };
            out.push(char::from(b'0' + digit));
        }

        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_magnitude(&self.digits, &rhs.digits),
            (true, true) => cmp_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let digits = add_magnitude(&self.digits, &rhs.digits);
            Self::from_parts(self.negative, digits)
        } else {
            match cmp_magnitude(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    Self::from_parts(self.negative, sub_magnitude(&self.digits, &rhs.digits))
                }
                Ordering::Less => {
                    Self::from_parts(rhs.negative, sub_magnitude(&rhs.digits, &self.digits))
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut accum = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            for (j, &b) in rhs.digits.iter().enumerate() {
                accum[i + j] += u32::from(a) * u32::from(b);
            }
        }

        let mut digits = Vec::with_capacity(accum.len() + 1);
        let mut carry = 0u32;
        for value in accum {
            let total = value + carry;
            digits.push((total % 10) as u8);
            carry = total / 10;
        }
        while carry > 0 {
            digits.push((carry % 10) as u8);
            carry /= 10;
        }

        Self::from_parts(self.negative != rhs.negative, digits)
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn from_parts(negative: bool, mut digits: Vec<u8>) -> Self {
        trim_zeros(&mut digits);
        let is_zero = digits.len() == 1 && digits[0] == 0;
        Self {
            negative: negative && !is_zero,
            digits,
        }
    }
}

fn trim_zeros(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    if digits.is_empty() {
        digits.push(0);
    }
}

fn cmp_magnitude(a: &[u8], b: &[u8]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => {
            for (x, y) in a.iter().rev().zip(b.iter().rev()) {
                match x.cmp(y) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let n = a.len().max(b.len());
    let mut result = Vec::with_capacity(n + 1);
    let mut carry = 0u8;

    for i in 0..n {
        let x = a.get(i).copied().unwrap_or(0);
        let y = b.get(i).copied().unwrap_or(0);
        let sum = x + y + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }

    if carry != 0 {
        result.push(carry);
    }
    result
}

fn sub_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    // Requires a >= b.
    let mut result = Vec::with_capacity(a.len());
    let mut borrow = 0i16;

    for (i, &x) in a.iter().enumerate() {
        let y = b.get(i).copied().unwrap_or(0);
        let mut value = i16::from(x) - i16::from(y) - borrow;
        if value < 0 {
            value += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push(value as u8);
    }

    trim_zeros(&mut result);
    result
}

use std::cmp::Ordering;
use std::fmt;

const BASE: u64 = 1_000_000_000;

#[derive(Clone, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    limbs: Vec<u32>, // little-endian base 10^9
}

impl fmt::Debug for Big {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_struct("Big")
            .field("negative", &self.negative)
            .field("limbs", &self.limbs)
            .finish()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

impl Big {
    fn zero() -> Self {
        Self { negative: false, limbs: vec![0] }
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.len() == 1 && self.limbs[0] == 0 {
            self.negative = false;
        }
    }

    fn cmp_mag(a: &[u32], b: &[u32]) -> Ordering {
        a.len().cmp(&b.len()).then_with(|| {
            for (x, y) in a.iter().rev().zip(b.iter().rev()) {
                match x.cmp(y) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        })
    }

    fn add_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let n = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64 + carry;
            out.push((n % BASE) as u32);
            carry = n / BASE;
        }
        if carry != 0 { out.push(carry as u32); }
        out
    }

    fn sub_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let mut n = a[i] as i64 - b.get(i).copied().unwrap_or(0) as i64 - borrow;
            if n < 0 {
                n += BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(n as u32);
        }
        while out.len() > 1 && out.last() == Some(&0) { out.pop(); }
        out
    }

    fn mul_mag(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut accum = vec![0u128; a.len() + b.len()];
        for (i, &x) in a.iter().enumerate() {
            let mut carry = 0u128;
            for (j, &y) in b.iter().enumerate() {
                let k = i + j;
                let n = accum[k] + (x as u128) * (y as u128) + carry;
                accum[k] = n % BASE as u128;
                carry = n / BASE as u128;
            }
            let mut k = i + b.len();
            while carry != 0 {
                if k == accum.len() { accum.push(0); }
                let n = accum[k] + carry;
                accum[k] = n % BASE as u128;
                carry = n / BASE as u128;
                k += 1;
            }
        }
        let mut out: Vec<u32> = accum.into_iter().map(|x| x as u32).collect();
        while out.len() > 1 && out.last() == Some(&0) { out.pop(); }
        out
    }

    fn add_small(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() { self.limbs.push(0); }
            let n = self.limbs[i] as u64 + carry;
            self.limbs[i] = (n % BASE) as u32;
            carry = n / BASE;
            i += 1;
        }
    }

    fn mul_small_add(&mut self, multiplier: u32, add: u32) {
        let mut carry = add as u64;
        for limb in &mut self.limbs {
            let n = *limb as u64 * multiplier as u64 + carry;
            *limb = (n % BASE) as u32;
            carry = n / BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % BASE) as u32);
            carry /= BASE;
        }
    }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() { return Err(Error::Empty); }
        let sign_len = usize::from(bytes[0] == b'+' || bytes[0] == b'-');
        if sign_len == 1 && bytes.len() == 1 { return Err(Error::SignOnly); }

        for i in sign_len..bytes.len() {
            let b = bytes[i];
            if b != b'_' && !b.is_ascii_digit() {
                return Err(Error::InvalidDigit { index: i });
            }
        }

        for i in sign_len..bytes.len() {
            if bytes[i] == b'_' {
                let left = i > sign_len && bytes[i - 1].is_ascii_digit();
                let right = i + 1 < bytes.len() && bytes[i + 1].is_ascii_digit();
                if !left || !right { return Err(Error::Separator { index: i }); }
            }
        }

        let digit_count = bytes[sign_len..].iter().filter(|&&b| b != b'_').count();
        let first = (sign_len..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if digit_count > 1 && bytes[first] == b'0' {
            return Err(Error::LeadingZero { index: first });
        }
        if digit_count > max_digits {
            let index = (sign_len..bytes.len())
                .filter(|&i| bytes[i] != b'_')
                .nth(max_digits).unwrap();
            return Err(Error::TooManyDigits { limit: max_digits, index });
        }

        let mut result = Self::zero();
        result.limbs.clear();
        for &b in &bytes[sign_len..] {
            if b != b'_' {
                result.mul_small_add(10, (b - b'0') as u32);
            }
        }
        result.negative = bytes[0] == b'-';
        result.normalize();
        Ok(result)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let mut raw = if self.limbs.len() == 1 {
            format!("{}", self.limbs[0])
        } else {
            let mut s = self.limbs.last().unwrap().to_string();
            for limb in self.limbs.iter().rev().skip(1) {
                s.push_str(&format!("{:09}", limb));
            }
            s
        };
        if raw.len() < min_digits {
            let mut padded = String::with_capacity(min_digits);
            padded.extend(std::iter::repeat_n('0', min_digits - raw.len()));
            padded.push_str(&raw);
            raw = padded;
        }
        if group > 0 && raw.len() > group {
            let first_group = raw.len() % group;
            let first_group = if first_group == 0 { group } else { first_group };
            let mut grouped = String::with_capacity(raw.len() + (raw.len() - 1) / group);
            grouped.push_str(&raw[..first_group]);
            let mut i = first_group;
            while i < raw.len() {
                grouped.push('_');
                grouped.push_str(&raw[i..i + group]);
                i += group;
            }
            raw = grouped;
        }
        if self.negative { format!("-{raw}") } else { raw }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => Self::cmp_mag(&self.limbs, &rhs.limbs),
            (true, true) => Self::cmp_mag(&rhs.limbs, &self.limbs),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self { negative: self.negative, limbs: Self::add_mag(&self.limbs, &rhs.limbs) };
            out.normalize();
            out
        } else {
            match Self::cmp_mag(&self.limbs, &rhs.limbs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self { negative: self.negative, limbs: Self::sub_mag(&self.limbs, &rhs.limbs) };
                    out.normalize();
                    out
                }
                Ordering::Less => {
                    let mut out = Self { negative: rhs.negative, limbs: Self::sub_mag(&rhs.limbs, &self.limbs) };
                    out.normalize();
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self { self.add(&rhs.neg()) }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !(out.limbs.len() == 1 && out.limbs[0] == 0) { out.negative = !out.negative; }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.limbs.len() == 1 && self.limbs[0] == 0 { 0 }
        else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.signum() == 0 || rhs.signum() == 0 { return Self::zero(); }
        let mut out = Self {
            negative: self.negative != rhs.negative,
            limbs: Self::mul_mag(&self.limbs, &rhs.limbs),
        };
        out.normalize();
        out
    }
}
