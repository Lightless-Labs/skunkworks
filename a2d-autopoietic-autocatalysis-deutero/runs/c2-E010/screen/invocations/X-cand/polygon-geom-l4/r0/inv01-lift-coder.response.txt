- Validation follows the stated precedence: length, coordinate range, one closing-copy removal, minimum length, then geometry.
- Boundary classification checks every closed segment first and reports the smallest effective edge index; parity uses strict exact crossings.
- Hull collinearity modes affect only boundary points; all-collinear `Include` returns sorted distinct points, while `Exclude` returns endpoints.
- Lengths use integer square root with half-up rounding per edge; perimeter sums those rounded lengths.
- Coordinate bounds make `i128` sufficient for geometric predicates and area; raw length caps keep pairwise checks bounded.
- The requirements specify one interpretation for these cases, so Solution B uses an independent algorithm with the same semantics.

### Solution A

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point { pub x: i64, pub y: i64 }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation { Clockwise, Degenerate, CounterClockwise }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128, pub area2: u128, pub orientation: Orientation,
    pub edge_milli: Vec<u64>, pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
    Duplicate { first: usize, second: usize },
    ZeroArea,
    AdjacentOverlap { vertex: usize },
    Intersection { edge_a: usize, edge_b: usize },
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location { Outside, Inside, Boundary { edge: usize } }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear { Exclude, Include }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull { pub vertices: Vec<Point>, pub source_indices: Vec<usize>, pub area2: u128, pub perimeter_milli: u64 }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Validated { pub vertices: Vec<Point>, pub source_indices: Vec<usize>, pub measures: Measures }

const LIMIT: i64 = 1_000_000_000;

fn in_range(p: Point) -> bool { p.x >= -LIMIT && p.x <= LIMIT && p.y >= -LIMIT && p.y <= LIMIT }
fn prepare(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > 4096 { return Err(GeomError::TooManyVertices { len: points.len() }); }
    for (i, &p) in points.iter().enumerate() {
        if !in_range(p) { return Err(GeomError::OutOfRange { index: i }); }
    }
    let mut v = points.to_vec();
    if v.len() >= 2 && v[0] == *v.last().unwrap() { v.pop(); }
    if v.len() < 3 { return Err(GeomError::TooFewVertices { len: v.len() }); }
    Ok(v)
}
fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128-a.x as i128)*(c.y as i128-a.y as i128)
        -(b.y as i128-a.y as i128)*(c.x as i128-a.x as i128)
}
fn area_signed(v: &[Point]) -> i128 {
    let mut s=0i128;
    for i in 0..v.len() {
        let a=v[i]; let b=v[(i+1)%v.len()];
        s += a.x as i128*b.y as i128-a.y as i128*b.x as i128;
    }
    s
}
fn isqrt(n: u128) -> u128 {
    if n < 2 { return n; }
    let mut x=1u128 << ((128-n.leading_zeros() as usize+1)/2);
    loop {
        let y=(x+n/x)/2;
        if y>=x { return x; }
        x=y;
    }
}
fn edge_len(a: Point,b: Point)->u64 {
    let dx=(b.x as i128-a.x as i128) as i128;
    let dy=(b.y as i128-a.y as i128) as i128;
    let n=(dx*dx+dy*dy) as u128 * 1_000_000;
    let r=isqrt(n);
    let rounded=if 4*n >= (2*r+1)*(2*r+1) {r+1} else {r};
    rounded as u64
}
fn measures(v: &[Point])->Measures {
    let signed=area_signed(v);
    let mut lengths=Vec::with_capacity(v.len());
    let mut perimeter=0u64;
    let mut longest=None;
    for i in 0..v.len() {
        let l=edge_len(v[i],v[(i+1)%v.len()]);
        lengths.push(l); perimeter+=l;
        let better=match longest {
            None=>true,
            Some(j) if l>lengths[j]=>true,
            Some(j) if l==lengths[j]=>{
                let pair=|k:usize| { let a=v[k]; let b=v[(k+1)%v.len()]; if a<=b {(a,b)} else {(b,a)} };
                pair(i)<pair(j) || (pair(i)==pair(j) && i<j)
            },
            _=>false,
        };
        if better { longest=Some(i); }
    }
    Measures { signed_area2:signed, area2:signed.unsigned_abs(), orientation:if signed>0 {Orientation::CounterClockwise} else if signed<0 {Orientation::Clockwise} else {Orientation::Degenerate}, edge_milli:lengths, perimeter_milli:perimeter, longest_edge:longest }
}
pub fn measure(points:&[Point])->Result<Measures,GeomError> { let v=prepare(points)?; Ok(measures(&v)) }

fn on_segment(a:Point,b:Point,q:Point)->bool {
    cross(a,b,q)==0 && q.x>=a.x.min(b.x) && q.x<=a.x.max(b.x) && q.y>=a.y.min(b.y) && q.y<=a.y.max(b.y)
}
pub fn classify(points:&[Point],queries:&[Point])->Result<Vec<Location>,GeomError> {
    let v=prepare(points)?;
    if queries.len()>4096 { return Err(GeomError::TooManyQueries{len:queries.len()}); }
    for (i,&q) in queries.iter().enumerate() { if !in_range(q) { return Err(GeomError::QueryOutOfRange{index:i}); } }
    let mut out=Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary=None;
        for i in 0..v.len() { if on_segment(v[i],v[(i+1)%v.len()],q) { boundary=Some(i); break; } }
        if let Some(edge)=boundary { out.push(Location::Boundary{edge}); continue; }
        let mut odd=false;
        for i in 0..v.len() {
            let a=v[i]; let b=v[(i+1)%v.len()];
            if (a.y>q.y)!=(b.y>q.y) {
                let c=cross(a,b,q);
                if (b.y>a.y && c>0)||(b.y<a.y && c<0) { odd=!odd; }
            }
        }
        out.push(if odd {Location::Inside} else {Location::Outside});
    }
    Ok(out)
}
fn turn(a:Point,b:Point,c:Point)->i128 { cross(a,b,c) }
fn chain(points:&[(Point,usize)],include:bool)->Vec<(Point,usize)> {
    let mut lower:Vec<(Point,usize)>=Vec::new();
    for &p in points {
        while lower.len()>=2 {
            let n=lower.len();
            let c=turn(lower[n-2].0,lower[n-1].0,p.0);
            if c>0 || (include && c==0) { break; }
            lower.pop();
        }
        lower.push(p);
    }
    lower
}
pub fn convex_hull(points:&[Point],mode:Collinear)->Result<Hull,GeomError> {
    if points.len()>4096 { return Err(GeomError::TooManyVertices{len:points.len()}); }
    for (i,&p) in points.iter().enumerate() { if !in_range(p) { return Err(GeomError::OutOfRange{index:i}); } }
    let mut first=BTreeMap::<Point,usize>::new();
    for (i,&p) in points.iter().enumerate() { first.entry(p).or_insert(i); }
    let unique:Vec<(Point,usize)>=first.into_iter().collect();
    let mut hull=if unique.len()<=1 { unique.clone() } else {
        let inc=mode==Collinear::Include;
        let mut lo=chain(&unique,inc);
        let mut rev=unique.clone(); rev.reverse();
        let mut hi=chain(&rev,inc);
        lo.pop(); hi.pop(); lo.extend(hi); lo
    };
    if hull.len()>=3 && area_signed(&hull.iter().map(|x|x.0).collect::<Vec<_>>())<0 { hull.reverse(); }
    if hull.len()>1 {
        let min=hull.iter().enumerate().min_by_key(|(_,x)|x.0).unwrap().0.0;
        while hull[0].0!=min { hull.rotate_left(1); }
    }
    let vertices:Vec<Point>=hull.iter().map(|x|x.0).collect();
    let source_indices=hull.iter().map(|x|x.1).collect();
    let m=measures(&vertices);
    Ok(Hull{vertices,source_indices,area2:m.area2,perimeter_milli:m.perimeter_milli})
}
fn orient_sign(a:Point,b:Point,c:Point)->i8 { let x=cross(a,b,c); if x>0 {1} else if x<0 {-1} else {0} }
fn intersects(a:Point,b:Point,c:Point,d:Point)->bool {
    let o1=orient_sign(a,b,c); let o2=orient_sign(a,b,d);
    let o3=orient_sign(c,d,a); let o4=orient_sign(c,d,b);
    if o1==0 && on_segment(a,b,c) { return true; }
    if o2==0 && on_segment(a,b,d) { return true; }
    if o3==0 && on_segment(c,d,a) { return true; }
    if o4==0 && on_segment(c,d,b) { return true; }
    o1!=o2 && o3!=o4
}
pub fn validate(points:&[Point])->Result<Validated,GeomError> {
    let v=prepare(points)?; let n=v.len();
    let mut seen=BTreeMap::<Point,usize>::new();
    let mut dup=None;
    for (j,&p) in v.iter().enumerate() {
        if let Some(&i)=seen.get(&p) { let pair=(i,j); if dup.map_or(true,|d|pair<d) {dup=Some(pair);} }
        else {seen.insert(p,j);}
    }
    if let Some((first,second))=dup { return Err(GeomError::Duplicate{first,second}); }
    let signed=area_signed(&v);
    if signed==0 { return Err(GeomError::ZeroArea); }
    for i in 0..n {
        let prev=v[(i+n-1)%n]; let cur=v[i]; let next=v[(i+1)%n];
        if cross(prev,cur,next)==0 {
            let dot=(prev.x as i128-cur.x as i128)*(next.x as i128-cur.x as i128)
                +(prev.y as i128-cur.y as i128)*(next.y as i128-cur.y as i128);
            if dot>0 { return Err(GeomError::AdjacentOverlap{vertex:i}); }
        }
    }
    for i in 0..n { for j in i+1..n {
        if j==i+1 || (i==0 && j==n-1) { continue; }
        if intersects(v[i],v[(i+1)%n],v[j],v[(j+1)%n]) { return Err(GeomError::Intersection{edge_a:i,edge_b:j}); }
    }}
    let mut indices:Vec<usize>=(0..n).collect();
    let min=*v.iter().min().unwrap();
    let start=v.iter().position(|&p|p==min).unwrap();
    if signed>0 { indices=(0..n).map(|k|(start+k)%n).collect(); }
    else { indices=(0..n).map(|k|(start+n-k)%n).collect(); }
    let vertices:Vec<Point>=indices.iter().map(|&i|v[i]).collect();
    let source_indices=indices;
    let m=measures(&vertices);
    Ok(Validated{vertices,source_indices,measures:m})
}
```
