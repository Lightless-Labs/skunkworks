- Validate `start` before any other condition, including `count == 0`; both solutions do.
- A missing byte is truncation at `input.len()`, while byte-ten overflow takes precedence, even if that byte also continues; both solutions do.
- Accept nonminimal encodings, but reject any tenth-byte payload above 1 or continuation bit; both solutions do.
- Zigzag handles `i64::MIN` without signed overflow; both solutions use bitwise mapping.
- Stream `position` counts committed bytes, including pending bytes; a failed push rolls back the entire chunk.
- The prompt specifies no alternate semantics, so Solution B uses a different decoding implementation while preserving the same requirements.

### Solution A
```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut out = Vec::with_capacity(10);
    loop {
        let payload = (value & 0x7f) as u8;
        value >>= 7;
        if value == 0 {
            out.push(payload);
            break;
        }
        out.push(payload | 0x80);
    }
    out
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut value = 0u64;
    for group in 0..10usize {
        let offset = start + group;
        if offset >= input.len() {
            return Err(Error::Truncated { offset: input.len(), start });
        }

        let byte = input[offset];
        let payload = (byte & 0x7f) as u64;

        if group == 9 && (payload > 1 || byte & 0x80 != 0) {
            return Err(Error::Overflow { offset, start });
        }

        value |= payload << (group * 7);
        if byte & 0x80 == 0 {
            return Ok((value, offset + 1));
        }
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut values = Vec::new();
    let mut cursor = start;
    for _ in 0..count {
        let (value, next) = decode_u64(input, cursor)?;
        values.push(value);
        cursor = next;
    }
    Ok((values, cursor))
}

pub fn encode_i64(value: i64) -> Vec<u8> {
    let bits = value as u64;
    let zigzag = (bits << 1) ^ ((value >> 63) as u64);
    encode_u64(zigzag)
}

pub fn decode_i64(input: &[u8], start: usize) -> Result<(i64, usize), Error> {
    let (value, next) = decode_u64(input, start)?;
    let decoded = ((value >> 1) as i64) ^ (-((value & 1) as i64));
    Ok((decoded, next))
}

#[derive(Debug, Clone)]
pub struct StreamDecoder {
    position: usize,
    pending: Vec<u8>,
    pending_start: usize,
}

impl StreamDecoder {
    pub fn new() -> Self {
        Self {
            position: 0,
            pending: Vec::new(),
            pending_start: 0,
        }
    }

    pub fn push(&mut self, chunk: &[u8]) -> Result<Vec<u64>, Error> {
        let mut position = self.position;
        let mut pending = self.pending.clone();
        let mut pending_start = self.pending_start;
        let mut output = Vec::new();

        for &byte in chunk {
            let absolute = position;
            position += 1;
            pending.push(byte);

            let group = pending.len() - 1;
            let payload = (byte & 0x7f) as u64;
            if group == 9 && (payload > 1 || byte & 0x80 != 0) {
                return Err(Error::Overflow {
                    offset: absolute,
                    start: pending_start,
                });
            }

            if byte & 0x80 == 0 {
                let (value, _) = decode_u64(&pending, 0)
                    .expect("validated stream integer");
                output.push(value);
                pending.clear();
                pending_start = position;
            }
        }

        self.position = position;
        self.pending = pending;
        self.pending_start = pending_start;
        Ok(output)
    }

    pub fn finish(&self) -> Result<(), Error> {
        if self.pending.is_empty() {
            Ok(())
        } else {
            Err(Error::Truncated {
                offset: self.position,
                start: self.position - self.pending.len(),
            })
        }
    }

    pub fn position(&self) -> usize {
        self.position
    }

    pub fn pending_len(&self) -> usize {
        self.pending.len()
    }

    pub fn reset(&mut self) {
        self.position = 0;
        self.pending.clear();
        self.pending_start = 0;
    }
}

impl Default for StreamDecoder {
    fn default() -> Self {
        Self::new()
    }
}
```

### Solution B
```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut result = Vec::new();
    while value >= 0x80 {
        result.push((value as u8) | 0x80);
        value >>= 7;
    }
    result.push(value as u8);
    result
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }
    let mut value = 0u64;
    let mut index = start;
    let mut shift = 0u32;

    loop {
        if index == input.len() {
            return Err(Error::Truncated { offset: input.len(), start });
        }
        let byte = input[index];
        let payload = byte & 0x7f;

        if shift == 63 && (payload > 1 || byte & 0x80 != 0) {
            return Err(Error::Overflow { offset: index, start });
        }
        value |= (payload as u64) << shift;

        if byte & 0x80 == 0 {
            return Ok((value, index + 1));
        }
        if shift == 63 {
            return Err(Error::Overflow { offset: index, start });
        }
        shift += 7;
        index += 1;
    }
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }
    let mut output = Vec::new();
    let mut cursor = start;
    for _ in 0..count {
        let decoded = decode_u64(input, cursor)?;
        output.push(decoded.0);
        cursor = decoded.1;
    }
    Ok((output, cursor))
}

pub fn encode_i64(value: i64) -> Vec<u8> {
    let zigzag = ((value as u64) << 1) ^ ((value >> 63) as u64);
    encode_u64(zigzag)
}

pub fn decode_i64(input: &[u8], start: usize) -> Result<(i64, usize), Error> {
    let (value, cursor) = decode_u64(input, start)?;
    let signed = if value & 1 == 0 {
        (value >> 1) as i64
    } else {
        -((value >> 1) as i64) - 1
    };
    Ok((signed, cursor))
}

#[derive(Debug, Clone)]
pub struct StreamDecoder {
    position: usize,
    value: u64,
    groups: usize,
    pending_start: usize,
}

impl StreamDecoder {
    pub fn new() -> Self {
        Self {
            position: 0,
            value: 0,
            groups: 0,
            pending_start: 0,
        }
    }

    pub fn push(&mut self, chunk: &[u8]) -> Result<Vec<u64>, Error> {
        let mut position = self.position;
        let mut value = self.value;
        let mut groups = self.groups;
        let mut pending_start = self.pending_start;
        let mut output = Vec::new();

        for &byte in chunk {
            let offset = position;
            position += 1;
            let payload = byte & 0x7f;

            if groups == 9 && (payload > 1 || byte & 0x80 != 0) {
                return Err(Error::Overflow {
                    offset,
                    start: pending_start,
                });
            }
            value |= (payload as u64) << (groups * 7);
            groups += 1;

            if byte & 0x80 == 0 {
                output.push(value);
                value = 0;
                groups = 0;
                pending_start = position;
            }
        }

        self.position = position;
        self.value = value;
        self.groups = groups;
        self.pending_start = pending_start;
        Ok(output)
    }

    pub fn finish(&self) -> Result<(), Error> {
        if self.groups == 0 {
            Ok(())
        } else {
            Err(Error::Truncated {
                offset: self.position,
                start: self.pending_start,
            })
        }
    }

    pub fn position(&self) -> usize {
        self.position
    }

    pub fn pending_len(&self) -> usize {
        self.groups
    }

    pub fn reset(&mut self) {
        self.position = 0;
        self.value = 0;
        self.groups = 0;
        self.pending_start = 0;
    }
}

impl Default for StreamDecoder {
    fn default() -> Self {
        Self::new()
    }
}
```
