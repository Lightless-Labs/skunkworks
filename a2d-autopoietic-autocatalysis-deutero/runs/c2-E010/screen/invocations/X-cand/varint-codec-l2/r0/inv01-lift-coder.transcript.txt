pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut out = Vec::with_capacity(10);
    loop {
        let mut byte = (value & 0x7f) as u8;
        value >>= 7;
        if value != 0 {
            byte |= 0x80;
        }
        out.push(byte);
        if value == 0 {
            return out;
        }
    }
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut value = 0u64;
    for group in 0..10usize {
        let offset = start + group;
        if offset >= input.len() {
            return Err(Error::Truncated { offset: input.len(), start });
        }
        let byte = input[offset];
        let payload = (byte & 0x7f) as u64;

        if group == 9 && (payload > 1 || byte & 0x80 != 0) {
            return Err(Error::Overflow { offset, start });
        }

        value |= payload << (group * 7);
        if byte & 0x80 == 0 {
            return Ok((value, offset + 1));
        }
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut values = Vec::with_capacity(count.min(input.len().saturating_sub(start)));
    let mut cursor = start;
    for _ in 0..count {
        let (value, next) = decode_u64(input, cursor)?;
        values.push(value);
        cursor = next;
    }
    Ok((values, cursor))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(value: u64) -> Vec<u8> {
    let mut n = value;
    let mut bytes = Vec::new();
    loop {
        let payload = (n % 128) as u8;
        n /= 128;
        bytes.push(if n == 0 { payload } else { payload | 0x80 });
        if n == 0 {
            break;
        }
    }
    bytes
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut acc = 0u64;
    let mut cursor = start;
    let mut shift = 0u32;
    let mut count = 0usize;

    while count < 10 {
        if cursor >= input.len() {
            return Err(Error::Truncated { offset: input.len(), start });
        }

        let byte = input[cursor];
        let payload = (byte & 0x7f) as u64;
        if count == 9 && (payload > 1 || byte & 0x80 != 0) {
            return Err(Error::Overflow { offset: cursor, start });
        }

        acc |= payload << shift;
        cursor += 1;
        count += 1;
        if byte & 0x80 == 0 {
            return Ok((acc, cursor));
        }
        shift += 7;
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }
    let mut out = Vec::new();
    let mut cursor = start;
    for _ in 0..count {
        let (v, next) = decode_u64(input, cursor)?;
        out.push(v);
        cursor = next;
    }
    Ok((out, cursor))
}

        let mut absolute = self.absolute;
        let mut partial = self.partial;
        let mut groups = self.groups;
        let mut integer_start = self.integer_start;
        let mut completed = Vec::new();

            if groups == 9 && (payload > 1 || byte & 0x80 != 0) {
                return Err(Error::Overflow { offset, start: integer_start });

                partial = 0;

                integer_start = absolute;
            }

        self.partial = partial;

    }
