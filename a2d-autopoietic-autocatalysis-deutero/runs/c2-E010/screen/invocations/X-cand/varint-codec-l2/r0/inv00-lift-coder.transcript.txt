    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut output = Vec::with_capacity(10);
    loop {
        let payload = (value & 0x7f) as u8;
        value >>= 7;
        if value == 0 {
            output.push(payload);
            break;
        }
        output.push(payload | 0x80);
    }
    output
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut value = 0u64;
    for group in 0..10usize {
        let index = match start.checked_add(group) {
            Some(index) if index < input.len() => index,
            _ => return Err(Error::Truncated { offset: input.len(), start }),
        };
        let byte = input[index];
        let payload = (byte & 0x7f) as u64;

        if group == 9 && (payload > 1 || byte & 0x80 != 0) {
            return Err(Error::Overflow { offset: index, start });
        }

        value |= payload << (group * 7);
        if byte & 0x80 == 0 {
            return Ok((value, index + 1));
        }
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],

    count: usize,
) -> Result<(Vec<u64>, usize), Error> {

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut bytes = Vec::new();
    while value >= 128 {
        bytes.push(((value & 127) as u8) | 128);
        value >>= 7;
    }
    bytes.push(value as u8);
    bytes
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut cursor = start;
    let mut value = 0u64;
    let mut shift = 0u32;

    for group in 0..10usize {
        if cursor >= input.len() {
            return Err(Error::Truncated {
                offset: input.len(),
                start,
            });
        }

        let byte = input[cursor];
        let payload = byte & 0x7f;
        if group == 9 && (payload > 1 || byte & 0x80 != 0) {
            return Err(Error::Overflow {
                offset: cursor,
                start,
            });
        }

        value |= (payload as u64) << shift;
        cursor += 1;
        if byte & 0x80 == 0 {
            return Ok((value, cursor));
        }
        shift += 7;
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }
    let mut cursor = start;
    let mut output = Vec::new();
    for _ in 0..count {
        let (value, next) = decode_u64(input, cursor)?;
        output.push(value);
        cursor = next;
    }
    Ok((output, cursor))
}

pub fn encode_i64(value: i64) -> Vec<u8> {
    let encoded = if value < 0 {
        value.unsigned_abs().wrapping_mul(2).wrapping_sub(1)
    } else {
        (value as u64).wrapping_mul(2)
    };
    encode_u64(encoded)
}

pub fn decode_i64(input: &[u8], start: usize) -> Result<(i64, usize), Error> {

    let decoded = if encoded & 1 == 0 {
        (encoded >> 1) as i64
    } else {
        -((encoded >> 1) as i64) - 1
    };
    Ok((decoded, cursor))
}
