- `decode_many` validates `start` even when `count == 0`; with a valid cursor and zero count it reads nothing.
- On byte ten, invalid payload or continuation is overflow before any truncation check; no eleventh byte is examined.
- Nonminimal encodings are accepted as long as they satisfy the ten-byte limit.
- Stream positions include buffered bytes only after a successful push; an overflowing push rolls back its entire chunk.
- `finish` reports a pending integer as truncated but leaves the decoder open for later pushes.
- For Solution B, the independent decoder uses a bounded byte loop; the edge-case readings are otherwise the same.

### Solution A
```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut out = Vec::with_capacity(10);
    loop {
        let mut byte = (value & 0x7f) as u8;
        value >>= 7;
        if value != 0 {
            byte |= 0x80;
        }
        out.push(byte);
        if value == 0 {
            return out;
        }
    }
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut value = 0u64;
    for group in 0..10usize {
        let offset = start + group;
        if offset >= input.len() {
            return Err(Error::Truncated { offset: input.len(), start });
        }
        let byte = input[offset];
        let payload = (byte & 0x7f) as u64;

        if group == 9 && (payload > 1 || byte & 0x80 != 0) {
            return Err(Error::Overflow { offset, start });
        }

        value |= payload << (group * 7);
        if byte & 0x80 == 0 {
            return Ok((value, offset + 1));
        }
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut values = Vec::with_capacity(count.min(input.len().saturating_sub(start)));
    let mut cursor = start;
    for _ in 0..count {
        let (value, next) = decode_u64(input, cursor)?;
        values.push(value);
        cursor = next;
    }
    Ok((values, cursor))
}

pub fn encode_i64(value: i64) -> Vec<u8> {
    let zigzag = ((value as u64) << 1) ^ ((value >> 63) as u64);
    encode_u64(zigzag)
}

pub fn decode_i64(input: &[u8], start: usize) -> Result<(i64, usize), Error> {
    let (value, next) = decode_u64(input, start)?;
    let decoded = if value & 1 == 0 {
        (value >> 1) as i64
    } else {
        -((value >> 1) as i64) - 1
    };
    Ok((decoded, next))
}

pub struct StreamDecoder {
    position: usize,
    pending: Vec<u8>,
    pending_start: usize,
}

impl StreamDecoder {
    pub fn new() -> Self {
        Self { position: 0, pending: Vec::new(), pending_start: 0 }
    }

    pub fn push(&mut self, chunk: &[u8]) -> Result<Vec<u64>, Error> {
        let mut values = Vec::new();
        let mut pending = self.pending.clone();
        let mut pending_start = self.pending_start;
        let mut position = self.position;

        for &byte in chunk {
            if pending.is_empty() {
                pending_start = position;
            }
            pending.push(byte);
            position += 1;

            if pending.len() == 10 {
                let payload = pending[9] & 0x7f;
                if payload > 1 || pending[9] & 0x80 != 0 {
                    return Err(Error::Overflow {
                        offset: position - 1,
                        start: pending_start,
                    });
                }
            }

            if byte & 0x80 == 0 {
                let (value, _) = decode_u64(&pending, 0)?;
                values.push(value);
                pending.clear();
            }
        }

        self.position = position;
        self.pending = pending;
        self.pending_start = if self.pending.is_empty() { position } else { pending_start };
        Ok(values)
    }

    pub fn finish(&self) -> Result<(), Error> {
        if self.pending.is_empty() {
            Ok(())
        } else {
            Err(Error::Truncated {
                offset: self.position,
                start: self.position - self.pending.len(),
            })
        }
    }

    pub fn position(&self) -> usize {
        self.position
    }

    pub fn pending_len(&self) -> usize {
        self.pending.len()
    }

    pub fn reset(&mut self) {
        self.position = 0;
        self.pending.clear();
        self.pending_start = 0;
    }
}

impl Default for StreamDecoder {
    fn default() -> Self {
        Self::new()
    }
}
```

### Solution B
```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(value: u64) -> Vec<u8> {
    let mut n = value;
    let mut bytes = Vec::new();
    loop {
        let payload = (n % 128) as u8;
        n /= 128;
        bytes.push(if n == 0 { payload } else { payload | 0x80 });
        if n == 0 {
            break;
        }
    }
    bytes
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut acc = 0u64;
    let mut cursor = start;
    let mut shift = 0u32;
    let mut count = 0usize;

    while count < 10 {
        if cursor >= input.len() {
            return Err(Error::Truncated { offset: input.len(), start });
        }

        let byte = input[cursor];
        let payload = (byte & 0x7f) as u64;
        if count == 9 && (payload > 1 || byte & 0x80 != 0) {
            return Err(Error::Overflow { offset: cursor, start });
        }

        acc |= payload << shift;
        cursor += 1;
        count += 1;
        if byte & 0x80 == 0 {
            return Ok((acc, cursor));
        }
        shift += 7;
    }

    unreachable!()
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }
    let mut out = Vec::new();
    let mut cursor = start;
    for _ in 0..count {
        let (v, next) = decode_u64(input, cursor)?;
        out.push(v);
        cursor = next;
    }
    Ok((out, cursor))
}

pub fn encode_i64(value: i64) -> Vec<u8> {
    let zigzag = ((value as u64) << 1) ^ ((value >> 63) as u64);
    encode_u64(zigzag)
}

pub fn decode_i64(input: &[u8], start: usize) -> Result<(i64, usize), Error> {
    let (u, cursor) = decode_u64(input, start)?;
    let v = if u & 1 == 0 {
        (u >> 1) as i64
    } else {
        -((u >> 1) as i64) - 1
    };
    Ok((v, cursor))
}

pub struct StreamDecoder {
    absolute: usize,
    partial: u64,
    groups: usize,
    integer_start: usize,
}

impl StreamDecoder {
    pub fn new() -> Self {
        Self { absolute: 0, partial: 0, groups: 0, integer_start: 0 }
    }

    pub fn push(&mut self, chunk: &[u8]) -> Result<Vec<u64>, Error> {
        let mut absolute = self.absolute;
        let mut partial = self.partial;
        let mut groups = self.groups;
        let mut integer_start = self.integer_start;
        let mut completed = Vec::new();

        for &byte in chunk {
            let offset = absolute;
            let payload = (byte & 0x7f) as u64;
            if groups == 9 && (payload > 1 || byte & 0x80 != 0) {
                return Err(Error::Overflow { offset, start: integer_start });
            }

            partial |= payload << (groups * 7);
            groups += 1;
            absolute += 1;

            if byte & 0x80 == 0 {
                completed.push(partial);
                partial = 0;
                groups = 0;
                integer_start = absolute;
            }
        }

        self.absolute = absolute;
        self.partial = partial;
        self.groups = groups;
        self.integer_start = integer_start;
        Ok(completed)
    }

    pub fn finish(&self) -> Result<(), Error> {
        if self.groups == 0 {
            Ok(())
        } else {
            Err(Error::Truncated {
                offset: self.absolute,
                start: self.absolute - self.groups,
            })
        }
    }

    pub fn position(&self) -> usize {
        self.absolute
    }

    pub fn pending_len(&self) -> usize {
        self.groups
    }

    pub fn reset(&mut self) {
        self.absolute = 0;
        self.partial = 0;
        self.groups = 0;
        self.integer_start = 0;
    }
}

impl Default for StreamDecoder {
    fn default() -> Self {
        Self::new()
    }
}
```
