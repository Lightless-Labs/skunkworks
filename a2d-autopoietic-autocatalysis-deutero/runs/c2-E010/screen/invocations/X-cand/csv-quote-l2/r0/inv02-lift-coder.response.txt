- A lone final delimiter creates a trailing empty field; a terminal newline creates no extra record.
- A pair of quotes at EOF is treated as an escape, so the quoted field remains unclosed.
- Errors use absolute UTF-8 byte offsets, and only outside-quote record terminators advance the error record index.
- NUL and bare CR checks take precedence over quote-state errors where specified.
- Lexical errors anywhere in the input take precedence over header errors.
- Solution A scans bytes directly; Solution B uses an independent character-iterator parser with byte offsets.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|s| s == name)
            }
        }
    }
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    let mut row: Vec<String> = Vec::new();
    let mut field = String::new();

    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quoted_open = 0usize;
    let mut record = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !in_quotes && b == b'\r' && (i + 1 == bytes.len() || bytes[i + 1] != b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if in_quotes {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote as char);
                    i += 2;
                    record_started = true;
                    continue;
                }
                in_quotes = false;
                after_quote = true;
                i += 1;
                record_started = true;
                continue;
            }
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            i += ch.len_utf8();
            record_started = true;
            continue;
        }

        if after_quote {
            if b == delimiter {
                row.push(std::mem::take(&mut field));
                after_quote = false;
                record_started = true;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                record += 1;
                record_started = false;
                after_quote = false;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte { record, byte: i });
        }

        if b == delimiter {
            row.push(std::mem::take(&mut field));
            record_started = true;
            i += 1;
            continue;
        }

        if b == b'\n' || b == b'\r' {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            record += 1;
            record_started = false;
            i += if b == b'\r' { 2 } else { 1 };
            continue;
        }

        if b == quote {
            if field.is_empty() && row.is_empty() || field.is_empty() && !record_started {
                in_quotes = true;
                quoted_open = i;
                record_started = true;
                i += 1;
                continue;
            }
            return Err(CsvError::UnexpectedQuote { record, byte: i });
        }

        let ch = input[i..].chars().next().unwrap();
        field.push(ch);
        record_started = true;
        i += ch.len_utf8();
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quoted_open,
        });
    }

    if record_started || !row.is_empty() || !field.is_empty() || after_quote {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

fn make_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let mut header = None;
    let mut rows = records;

    if config.headers && !rows.is_empty() {
        let h = rows.remove(0);

        if config.duplicates == DuplicatePolicy::Reject {
            for second in 0..h.len() {
                if let Some(first) = h[..second].iter().position(|name| name == &h[second]) {
                    return Err(CsvError::DuplicateHeader {
                        name: h[second].clone(),
                        first,
                        second,
                    });
                }
            }
        }

        for (data_index, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: data_index + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }

        header = Some(h);
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| make_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let h = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => h.iter().enumerate().filter(|(_, n)| n.as_str() == name).map(|(i, _)| i).last(),
            _ => h.iter().position(|n| n == name),
        }
    }
}

fn check_config(c: Config) -> Result<(), CsvError> {
    if c.delimiter < 0x21 || c.delimiter > 0x7e {
        return Err(CsvError::InvalidConfig { component: ConfigPart::Delimiter });
    }
    if c.quote < 0x21 || c.quote > 0x7e {
        return Err(CsvError::InvalidConfig { component: ConfigPart::Quote });
    }
    if c.delimiter == c.quote {
        return Err(CsvError::InvalidConfig { component: ConfigPart::Quote });
    }
    Ok(())
}

fn parse_dialect(s: &str, delim: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let b = s.as_bytes();
    let start = if b.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if start == b.len() {
        return Ok(vec![]);
    }

    let mut out = Vec::new();
    let mut row = Vec::new();
    let mut value = String::new();
    let mut idx = start;
    let mut rec = 0usize;
    let mut quoted = false;
    let mut closed = false;
    let mut opening = 0usize;
    let mut started = false;

    while idx < b.len() {
        let byte = b[idx];
        if byte == 0 {
            return Err(CsvError::Nul { record: rec, byte: idx });
        }
        if !quoted && byte == b'\r' && (idx + 1 >= b.len() || b[idx + 1] != b'\n') {
            return Err(CsvError::BareCr { record: rec, byte: idx });
        }

        if quoted {
            if byte == quote {
                if idx + 1 < b.len() && b[idx + 1] == quote {
                    value.push(quote as char);
                    idx += 2;
                    started = true;
                } else {
                    quoted = false;
                    closed = true;
                    idx += 1;
                    started = true;
                }
            } else {
                let ch = s[idx..].chars().next().unwrap();
                value.push(ch);
                idx += ch.len_utf8();
                started = true;
            }
            continue;
        }

        if closed {
            match byte {
                x if x == delim => {
                    row.push(std::mem::take(&mut value));
                    closed = false;
                    idx += 1;
                    started = true;
                }
                b'\n' | b'\r' => {
                    row.push(std::mem::take(&mut value));
                    out.push(std::mem::take(&mut row));
                    rec += 1;
                    closed = false;
                    started = false;
                    idx += if byte == b'\r' { 2 } else { 1 };
                }
                _ => return Err(CsvError::TrailingByte { record: rec, byte: idx }),
            }
            continue;
        }

        if byte == delim {
            row.push(std::mem::take(&mut value));
            started = true;
            idx += 1;
        } else if byte == b'\n' || byte == b'\r' {
            row.push(std::mem::take(&mut value));
            out.push(std::mem::take(&mut row));
            rec += 1;
            started = false;
            idx += if byte == b'\r' { 2 } else { 1 };
        } else if byte == quote {
            if value.is_empty() && !started && row.is_empty() {
                quoted = true;
                opening = idx;
                started = true;
                idx += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record: rec, byte: idx });
            }
        } else {
            let ch = s[idx..].chars().next().unwrap();
            value.push(ch);
            started = true;
            idx += ch.len_utf8();
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote { record: rec, byte: opening });
    }
    if started || !row.is_empty() || !value.is_empty() || closed {
        row.push(value);
        out.push(row);
    }
    Ok(out)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn assemble(records: Vec<Vec<String>>, c: Config) -> Result<Table, CsvError> {
    let mut header = None;
    let mut rows = records;
    if c.headers && !rows.is_empty() {
        let h = rows.remove(0);
        if c.duplicates == DuplicatePolicy::Reject {
            for second in 0..h.len() {
                for first in 0..second {
                    if h[first] == h[second] {
                        return Err(CsvError::DuplicateHeader { name: h[second].clone(), first, second });
                    }
                }
            }
        }
        for (offset, r) in rows.iter().enumerate() {
            if r.len() != h.len() {
                return Err(CsvError::HeaderWidth { record: offset + 1, expected: h.len(), actual: r.len() });
            }
        }
        header = Some(h);
    }
    Ok(Table { header, rows, duplicates: c.duplicates })
}

pub struct StreamReader {
    config: Config,
    bytes: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        check_config(config)?;
        Ok(Self { config, bytes: String::new(), result: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.bytes.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(r) = &self.result {
            return r.clone();
        }
        let r = parse_dialect(&self.bytes, self.config.delimiter, self.config.quote)
            .and_then(|parsed| assemble(parsed, self.config));
        self.result = Some(r.clone());
        r
    }

    pub fn reset(&mut self) {
        self.bytes.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}
```
