- Scan errors are returned immediately; EOF checks run only after framing and scanning finish.
- Empty unpadded input is valid; a one-symbol residual is truncated.
- A pending third-position `=` takes precedence over EOF truncation; strict tail-bit checks run only after a complete padded group or valid unpadded EOF.
- Wrapped rows require a separator only when another byte follows a full row; a complete separator must be followed by a symbol.
- The symbol limit counts every non-skipped symbol, including invalid bytes and `=`, and is checked before padding or alphabet classification.
- Offsets count physical bytes across chunks, including skipped whitespace and line endings.

    pub padding: Padding,
    pub whitespace: Whitespace,
    pub strict: bool,
    pub max_symbols: Option<usize>,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
    NonCanonical { offset: usize },
    MixedAlphabet { offset: usize },
    Overlong { offset: usize, limit: usize },
}

fn validate(o: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = o.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}
fn digit(b: u8, a: Alphabet) -> Option<u8> {
    match b {
        b'A'..=b'Z' => Some(b - b'A'),
        b'a'..=b'z' => Some(b - b'a' + 26),
        b'0'..=b'9' => Some(b - b'0' + 52),
        b'+' if a == Alphabet::Standard => Some(62),
        b'/' if a == Alphabet::Standard => Some(63),
        b'-' if a == Alphabet::UrlSafe => Some(62),
        b'_' if a == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}
fn foreign(b: u8, a: Alphabet) -> bool {
    match a { Alphabet::Standard => b == b'-' || b == b'_', Alphabet::UrlSafe => b == b'+' || b == b'/' }
}
fn encode_digits(input: &[u8], a: Alphabet, padding: Padding) -> Vec<u8> {
    let alphabet = match a {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/",
        Alphabet::UrlSafe => b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_",
    };
    let mut out = Vec::new();
    let mut i = 0;
    while i + 3 <= input.len() {
        let n = ((input[i] as u32) << 16) | ((input[i+1] as u32) << 8) | input[i+2] as u32;
        for s in [18,12,6,0] { out.push(alphabet[((n >> s) & 63) as usize]); }
        i += 3;
    }
    let rem = input.len() - i;
    if rem == 1 {
        let n = input[i];
        out.push(alphabet[(n >> 2) as usize]);
        out.push(alphabet[((n & 3) << 4) as usize]);
        if padding != Padding::Forbidden { out.extend_from_slice(b"=="); }
    } else if rem == 2 {
        let n = ((input[i] as u16) << 8) | input[i+1] as u16;
        out.push(alphabet[(n >> 10) as usize]);
        out.push(alphabet[((n >> 4) & 63) as usize]);
        out.push(alphabet[((n & 15) << 2) as usize]);
        if padding != Padding::Forbidden { out.push(b'='); }
    }
    out
}
fn wrap_symbols(raw: &[u8], whitespace: Whitespace) -> Vec<u8> {
    let Whitespace::Wrapped { width, ending } = whitespace else { return raw.to_vec(); };
    let sep: &[u8] = match ending { LineEnding::Lf => b"\n", LineEnding::CrLf => b"\r\n" };
    let mut out = Vec::new();
    for (i, &b) in raw.iter().enumerate() {
        if i > 0 && i % width == 0 { out.extend_from_slice(sep); }
        out.push(b);
    }
    out
}
pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;

    Ok(String::from_utf8(wrap_symbols(&raw, options.whitespace)).unwrap())
}
pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    let mut d = Decoder::new(options)?;
    let mut out = d.push(input)?;
    out.extend(d.finish()?);
    Ok(out)
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    closed: bool,
    symbols: usize,
}
impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self { options, tail: Vec::new(), closed: false, symbols: 0 })
    }
    fn emit(&mut self, bytes: &[u8]) -> String {
        let raw = encode_digits(bytes, self.options.alphabet, self.options.padding);
        let mut out = Vec::new();
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            let sep: &[u8] = match ending { LineEnding::Lf => b"\n", LineEnding::CrLf => b"\r\n" };
            for b in raw {
                if self.symbols > 0 && self.symbols % width == 0 { out.extend_from_slice(sep); }
                out.push(b);
                self.symbols += 1;
            }
        } else { out = raw; }
        String::from_utf8(out).unwrap()
    }
    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed { return Err(Error::AlreadyFinished); }
        if input.is_empty() { return Ok(String::new()); }
        let mut data = core::mem::take(&mut self.tail);
        data.extend_from_slice(input);
        let n = data.len() / 3 * 3;
        self.tail.extend_from_slice(&data[n..]);
        Ok(self.emit(&data[..n]))
    }
    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed { return Err(Error::AlreadyFinished); }
        self.closed = true;
        let tail = core::mem::take(&mut self.tail);
        Ok(self.emit(&tail))
    }
}

pub struct Decoder {
    options: Options,
    offset: usize,
    count: usize,
    group: Vec<(u8, usize)>,
    pending: Option<usize>,
    terminal: bool,
    row: usize,
    frame: u8,
    frame_start: usize,
    after_sep: bool,
    closed: bool,
    sticky: Option<Error>,
}
impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self { options, offset: 0, count: 0, group: Vec::new(), pending: None, terminal: false,
            row: 0, frame: 0, frame_start: 0, after_sep: false, closed: false, sticky: None })
    }
    fn fail(&mut self, e: Error) -> Error { self.sticky = Some(e.clone()); e }
    fn scan_symbol(&mut self, b: u8, pos: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if let Some(limit) = self.options.max_symbols {
            if self.count >= limit { return Err(Error::Overlong { offset: pos, limit }); }
        }
        self.count += 1;
        if self.terminal { return Err(Error::BadPadding { offset: pos }); }
        if let Some(p) = self.pending {
            if b != b'=' { return Err(Error::BadPadding { offset: pos }); }
            self.pending = None;
            self.group.push((b, pos));
            self.terminal = true;
            out.push((digit(self.group[0].0, self.options.alphabet).unwrap() << 2)
                | (digit(self.group[1].0, self.options.alphabet).unwrap() >> 4));
            if self.options.strict && digit(self.group[1].0, self.options.alphabet).unwrap() & 15 != 0 {
                return Err(Error::NonCanonical { offset: self.group[1].1 });
            }
            self.group.clear();
            return Ok(());
        }
        if b == b'=' {
            if self.options.padding == Padding::Forbidden { return Err(Error::BadPadding { offset: pos }); }
            match self.group.len() {
                0 | 1 => return Err(Error::BadPadding { offset: pos }),
                2 => { self.pending = Some(pos); self.group.push((b,pos)); return Ok(()); }
                3 => {
                    let d0 = digit(self.group[0].0, self.options.alphabet).unwrap();
                    let d1 = digit(self.group[1].0, self.options.alphabet).unwrap();
                    let d2 = digit(self.group[2].0, self.options.alphabet).unwrap();
                    out.extend_from_slice(&[(d0 << 2) | (d1 >> 4), (d1 << 4) | (d2 >> 2)]);
                    if self.options.strict && d2 & 3 != 0 { return Err(Error::NonCanonical { offset: self.group[2].1 }); }
                    self.group.clear();
                    self.terminal = true;
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }
        let d = match digit(b, self.options.alphabet) {
            Some(d) => d,
            None if self.options.strict && foreign(b, self.options.alphabet) => return Err(Error::MixedAlphabet { offset: pos }),
            None => return Err(Error::InvalidByte { offset: pos, byte: b }),
        };
        self.group.push((b,pos));
        if self.group.len() == 4 {
            let a = digit(self.group[0].0, self.options.alphabet).unwrap();
            let b = digit(self.group[1].0, self.options.alphabet).unwrap();
            let c = digit(self.group[2].0, self.options.alphabet).unwrap();
            let d = digit(self.group[3].0, self.options.alphabet).unwrap();
            out.extend_from_slice(&[(a<<2)|(b>>4), (b<<4)|(c>>2), (c<<6)|d]);
            self.group.clear();
        }
        Ok(())
    }
    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(e) = &self.sticky { return Err(e.clone()); }
        if self.closed { return Err(Error::AlreadyFinished); }
        let mut out = Vec::new();
        for &b in input {
            let pos = self.offset;
            self.offset += 1;
            match self.options.whitespace {
                Whitespace::IgnoreAscii if matches!(b, 9|10|11|12|13|32) => continue,
                Whitespace::Wrapped { width, ending } => {
                    if self.frame == 1 {
                        if b != b'\n' { return Err(self.fail(Error::BadWrap { offset: pos })); }
                        self.frame = 0; self.row = 0; self.after_sep = true;
                        continue;
                    }
                    if self.row == width {
                        let expected = match ending { LineEnding::Lf => b'\n', LineEnding::CrLf => b'\r' };
                        if b != expected { return Err(self.fail(Error::BadWrap { offset: pos })); }
                        self.frame_start = pos;
                        if ending == LineEnding::CrLf { self.frame = 1; } else { self.row = 0; self.after_sep = true; }
                        continue;
                    }
                    if matches!(b, 9|10|11|12|13|32) {
                        return Err(self.fail(Error::BadWrap { offset: pos }));
                    }
                    if self.after_sep { self.after_sep = false; }
                    self.row += 1;
                }
                _ => {}
            }
            if let Err(e) = self.scan_symbol(b, pos, &mut out) { return Err(self.fail(e)); }
        }
        Ok(out)
    }
    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(e) = &self.sticky { return Err(e.clone()); }
        if self.closed { return Err(Error::AlreadyFinished); }
        if self.frame == 1 { return Err(self.fail(Error::BadWrap { offset: self.frame_start })); }
        if self.after_sep { return Err(self.fail(Error::BadWrap { offset: self.frame_start })); }
        if self.pending.is_some() { return Err(self.fail(Error::BadPadding { offset: self.pending.unwrap() })); }
        let mut out = Vec::new();
        if self.group.is_empty() { self.closed = true; return Ok(out); }
        if self.group.len() == 1 { return Err(self.fail(Error::Truncated { offset: self.group[0].1 })); }
        if self.options.padding == Padding::Required {
            return Err(self.fail(Error::Truncated { offset: self.group[0].1 }));
        }
        if self.terminal { self.closed = true; return Ok(out); }
        let vals: Vec<u8> = self.group.iter().map(|x| digit(x.0, self.options.alphabet).unwrap()).collect();
        if vals.len() == 2 {
            if self.options.strict && vals[1] & 15 != 0 { return Err(self.fail(Error::NonCanonical { offset: self.group[1].1 })); }
            out.push((vals[0]<<2)|(vals[1]>>4));
        } else if vals.len() == 3 {
            if self.options.strict && vals[2] & 3 != 0 { return Err(self.fail(Error::NonCanonical { offset: self.group[2].1 })); }
            out.extend_from_slice(&[(vals[0]<<2)|(vals[1]>>4),(vals[1]<<4)|(vals[2]>>2)]);
        }
        self.closed = true;
        Ok(out)
    }
}
