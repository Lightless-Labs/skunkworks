- `put` on a disabled cache returns `Disabled` before counting or aging.
- A counted operation increments `pending`; when it reaches `period`, aging happens immediately and `pending` resets to zero.
- Aging uses ceiling division, so frequency 3 becomes 2.
- Snapshot and eviction ties follow global oldest-to-newest recency, never key order.
- Explicit aging leaves recency and pending unchanged.
- Both implementations use straightforward scans; ordering cost is proportional to the number of resident entries.

### Solution A

```rust
use std::collections::{BTreeMap, VecDeque};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroPeriod,
    Disabled,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Entry {
    pub key: u64,
    pub value: i64,
    pub frequency: u8,
}

#[derive(Debug, Clone)]
pub struct Cache {
    capacity: usize,
    period: u32,
    pending: u32,
    entries: BTreeMap<u64, Entry>,
    recency: VecDeque<u64>,
}

impl Cache {
    pub fn new(capacity: usize, period: u32) -> Result<Self, Error> {
        if period == 0 {
            return Err(Error::ZeroPeriod);
        }
        Ok(Self {
            capacity,
            period,
            pending: 0,
            entries: BTreeMap::new(),
            recency: VecDeque::new(),
        })
    }

    pub fn capacity(&self) -> usize {
        self.capacity
    }

    pub fn pending(&self) -> u32 {
        self.pending
    }

    pub fn snapshot(&self) -> Vec<Entry> {
        let mut entries: Vec<Entry> = self
            .recency
            .iter()
            .filter_map(|key| self.entries.get(key).cloned())
            .collect();
        entries.sort_by_key(|entry| {
            let position = self.recency.iter().position(|key| *key == entry.key).unwrap_or(0);
            (entry.frequency, position)
        });
        entries
    }

    fn age_one(&mut self) {
        for entry in self.entries.values_mut() {
            entry.frequency = (entry.frequency + 1) / 2;
        }
    }

    fn begin(&mut self) {
        self.pending += 1;
        if self.pending == self.period {
            self.age_one();
            self.pending = 0;
        }
    }

    fn touch(&mut self, key: u64) {
        self.recency.retain(|&existing| existing != key);
        self.recency.push_back(key);
    }

    fn credit(&mut self, key: u64) {
        if let Some(entry) = self.entries.get_mut(&key) {
            entry.frequency = entry.frequency.saturating_add(1).min(7);
        }
        self.touch(key);
    }

    fn victim(&self) -> Option<u64> {
        self.recency.iter().copied().min_by_key(|key| {
            let frequency = self.entries.get(key).map_or(u8::MAX, |entry| entry.frequency);
            let position = self.recency.iter().position(|&candidate| candidate == *key).unwrap_or(0);
            (frequency, position)
        })
    }

    fn erase(&mut self, key: u64) -> Option<Entry> {
        self.recency.retain(|&existing| existing != key);
        self.entries.remove(&key)
    }

    pub fn get(&mut self, key: u64) -> Option<i64> {
        self.begin();
        let value = self.entries.get(&key).map(|entry| entry.value);
        if value.is_some() {
            self.credit(key);
        }
        value
    }

    pub fn put(&mut self, key: u64, value: i64) -> Result<Vec<u64>, Error> {
        if self.capacity == 0 {
            return Err(Error::Disabled);
        }
        self.begin();

        if let Some(entry) = self.entries.get_mut(&key) {
            entry.value = value;
            entry.frequency = entry.frequency.saturating_add(1).min(7);
            self.touch(key);
            return Ok(Vec::new());
        }

        let mut removed = Vec::new();
        if self.entries.len() >= self.capacity {
            if let Some(victim) = self.victim() {
                self.erase(victim);
                removed.push(victim);
            }
        }
        self.entries.insert(
            key,
            Entry {
                key,
                value,
                frequency: 1,
            },
        );
        self.touch(key);
        Ok(removed)
    }

    pub fn remove(&mut self, key: u64) -> Option<i64> {
        self.begin();
        self.erase(key).map(|entry| entry.value)
    }

    pub fn age(&mut self, rounds: u8) {
        for _ in 0..rounds {
            self.age_one();
        }
    }

    pub fn resize(&mut self, capacity: usize) -> Vec<u64> {
        self.capacity = capacity;
        let mut removed = Vec::new();
        while self.entries.len() > capacity {
            if let Some(victim) = self.victim() {
                self.erase(victim);
                removed.push(victim);
            } else {
                break;
            }
        }
        removed
    }
}
```

### Solution B

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroPeriod,
    Disabled,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Entry {
    pub key: u64,
    pub value: i64,
    pub frequency: u8,
}

#[derive(Debug, Clone)]
struct Record {
    value: i64,
    frequency: u8,
    recency: u128,
}

#[derive(Debug, Clone)]
pub struct Cache {
    capacity: usize,
    period: u32,
    pending: u32,
    next_recency: u128,
    entries: BTreeMap<u64, Record>,
}

impl Cache {
    pub fn new(capacity: usize, period: u32) -> Result<Self, Error> {
        if period == 0 {
            return Err(Error::ZeroPeriod);
        }
        Ok(Self {
            capacity,
            period,
            pending: 0,
            next_recency: 0,
            entries: BTreeMap::new(),
        })
    }

    pub fn capacity(&self) -> usize {
        self.capacity
    }

    pub fn pending(&self) -> u32 {
        self.pending
    }

    fn ordered_keys(&self) -> Vec<u64> {
        let mut keys: Vec<u64> = self.entries.keys().copied().collect();
        keys.sort_by_key(|key| self.entries.get(key).map(|record| record.recency).unwrap_or(0));
        keys
    }

    pub fn snapshot(&self) -> Vec<Entry> {
        let mut entries: Vec<Entry> = self.entries.iter().map(|(&key, record)| Entry {
            key,
            value: record.value,
            frequency: record.frequency,
        }).collect();
        entries.sort_by_key(|entry| {
            let recency = self.entries.get(&entry.key).map(|record| record.recency).unwrap_or(0);
            (entry.frequency, recency)
        });
        entries
    }

    fn age_one(&mut self) {
        for record in self.entries.values_mut() {
            record.frequency = (record.frequency + 1) / 2;
        }
    }

    fn begin(&mut self) {
        self.pending += 1;
        if self.pending == self.period {
            self.age_one();
            self.pending = 0;
        }
    }

    fn touch(&mut self, key: u64) {
        let recency = self.next_recency;
        self.next_recency = self.next_recency.wrapping_add(1);
        if let Some(record) = self.entries.get_mut(&key) {
            record.recency = recency;
        }
    }

    fn credit(&mut self, key: u64) {
        if let Some(record) = self.entries.get_mut(&key) {
            record.frequency = record.frequency.saturating_add(1).min(7);
        }
        self.touch(key);
    }

    fn victim(&self) -> Option<u64> {
        self.entries.iter().min_by_key(|(_, record)| {
            (record.frequency, record.recency)
        }).map(|(&key, _)| key)
    }

    pub fn get(&mut self, key: u64) -> Option<i64> {
        self.begin();
        let value = self.entries.get(&key).map(|record| record.value);
        if value.is_some() {
            self.credit(key);
        }
        value
    }

    pub fn put(&mut self, key: u64, value: i64) -> Result<Vec<u64>, Error> {
        if self.capacity == 0 {
            return Err(Error::Disabled);
        }
        self.begin();

        if let Some(record) = self.entries.get_mut(&key) {
            record.value = value;
            record.frequency = record.frequency.saturating_add(1).min(7);
            self.touch(key);
            return Ok(Vec::new());
        }

        let mut removed = Vec::new();
        if self.entries.len() >= self.capacity {
            if let Some(victim) = self.victim() {
                self.entries.remove(&victim);
                removed.push(victim);
            }
        }

        let recency = self.next_recency;
        self.next_recency = self.next_recency.wrapping_add(1);
        self.entries.insert(key, Record {
            value,
            frequency: 1,
            recency,
        });
        Ok(removed)
    }

    pub fn remove(&mut self, key: u64) -> Option<i64> {
        self.begin();
        self.entries.remove(&key).map(|record| record.value)
    }

    pub fn age(&mut self, rounds: u8) {
        for _ in 0..rounds {
            self.age_one();
        }
    }

    pub fn resize(&mut self, capacity: usize) -> Vec<u64> {
        self.capacity = capacity;
        let mut removed = Vec::new();
        while self.entries.len() > capacity {
            if let Some(victim) = self.victim() {
                self.entries.remove(&victim);
                removed.push(victim);
            } else {
                break;
            }
        }
        removed
    }
}
```
