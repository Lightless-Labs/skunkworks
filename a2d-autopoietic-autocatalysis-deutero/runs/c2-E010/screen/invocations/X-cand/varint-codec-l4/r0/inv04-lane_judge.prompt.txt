#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Message {
    pub fields: Vec<Field>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Field {
    UInt { tag: u32, value: u64 },
    Bytes { tag: u32, data: Vec<u8> },
    Nested { tag: u32, message: Message },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
    LengthOutOfBounds { offset: usize, length: u64 },
    InvalidTag { offset: usize, tag: u64 },
    InvalidWire { offset: usize, wire: u8 },
    OutOfOrder { offset: usize, previous: u32, tag: u32 },
    Duplicate { offset: usize, tag: u32 },
    NonCanonical { offset: usize },
    DepthExceeded { offset: usize, depth: usize },
    InvalidDepthLimit { limit: usize },
}

fn put_varint(mut value: u64, out: &mut Vec<u8>) {
    while value >= 0x80 {
        out.push((value as u8) | 0x80);
        value >>= 7;
    }
    out.push(value as u8);
}

fn read_varint(
    input: &[u8],
    mut pos: usize,
    end: usize,
) -> Result<(u64, usize), Error> {
    let start = pos;
    let mut value = 0u64;

    for index in 0..10 {
        if pos >= end {
            return Err(Error::Truncated { offset: end, start });
        }
        let byte = input[pos];
        if index == 9 && ((byte & 0x7f) > 1 || (byte & 0x80) != 0) {
            return Err(Error::Overflow { offset: pos, start });
        }
        value |= ((byte & 0x7f) as u64) << (index * 7);
        pos += 1;

        if (byte & 0x80) == 0 {
            if index > 0 && (byte & 0x7f) == 0 {
                return Err(Error::NonCanonical { offset: start });
            }
            return Ok((value, pos));
        }
    }

    unreachable!()
}

fn encode_body(message: &Message, max_depth: usize, depth: usize) -> Result<Vec<u8>, Error> {
    let mut body = Vec::new();
    let mut previous: Option<(u32, u8)> = None;

    for (index, field) in message.fields.iter().enumerate() {
        let (tag, wire) = match field {
            Field::UInt { tag, .. } => (*tag, 0),
            Field::Bytes { tag, .. } => (*tag, 2),
            Field::Nested { tag, .. } => (*tag, 3),
        };

        if !(1..=15).contains(&tag) {
            return Err(Error::InvalidTag { offset: index, tag: tag as u64 });
        }
        if let Some((prev_tag, prev_wire)) = previous {
            if tag < prev_tag {
                return Err(Error::OutOfOrder { offset: index, previous: prev_tag, tag });
            }
            if tag == prev_tag && !(prev_wire == 2 && wire == 2) {
                return Err(Error::Duplicate { offset: index, tag });
            }
        }

        let payload = match field {
            Field::UInt { value, .. } => {
                let mut v = Vec::new();
                put_varint(*value, &mut v);
                v
            }
            Field::Bytes { data, .. } => {
                let mut v = Vec::new();
                put_varint(data.len() as u64, &mut v);
                v.extend_from_slice(data);
                v
            }
            Field::Nested { message, .. } => {
                let child_depth = depth + 1;
                if child_depth > max_depth {
                    return Err(Error::DepthExceeded { offset: index, depth: child_depth });
                }
                let child = encode_body(message, max_depth, child_depth)?;
                let mut v = Vec::new();
                put_varint(child.len() as u64, &mut v);
                v.extend_from_slice(&child);
                v
            }
        };

        put_varint(((tag as u64) << 3) | wire as u64, &mut body);
        body.extend_from_slice(&payload);
        previous = Some((tag, wire));
    }

    Ok(body)
}

pub fn encode_message(message: &Message, max_depth: usize) -> Result<Vec<u8>, Error> {
    if max_depth > 32 {
        return Err(Error::InvalidDepthLimit { limit: max_depth });
    }
    let body = encode_body(message, max_depth, 0)?;
    let mut result = Vec::new();
    put_varint(body.len() as u64, &mut result);
    result.extend_from_slice(&body);
    Ok(result)
}

fn decode_body(
    input: &[u8],
    start: usize,
    end: usize,
    max_depth: usize,
    depth: usize,
) -> Result<Message, Error> {

    let mut pos = start;
    let mut previous: Option<(u32, u8)> = None;

    while pos < end {
        let key_start = pos;
        let (key, next) = read_varint(input, pos, end)?;
        pos = next;

        let tag64 = key >> 3;
        let wire = (key & 7) as u8;
        if tag64 > 255 || tag64 == 0 {
            return Err(Error::InvalidTag { offset: key_start, tag: tag64 });
        }
        let tag = tag64 as u32;

        if !matches!(wire, 0 | 1 | 2 | 3 | 5) {
            return Err(Error::InvalidWire { offset: key_start, wire });
        }

        if let Some((prev_tag, prev_wire)) = previous {
            if tag < prev_tag {
                return Err(Error::OutOfOrder { offset: key_start, previous: prev_tag, tag });
            }
            if tag == prev_tag && !(prev_wire == 2 && wire == 2) {
                return Err(Error::Duplicate { offset: key_start, tag });
            }
        }

        let retained = tag <= 15;

            0 => {
                let payload_start = pos;
                let (value, next) = read_varint(input, pos, end)?;
                pos = next;

                    fields.push(Field::UInt { tag, value });

                    return Err(Error::LengthOutOfBounds {
                        offset: key_start,
                        length: length as u64,
                    });
                }
                pos += length;
            }
            2 => {
                let length_start = pos;
                let (length, next) = read_varint(input, pos, end)?;
                pos = next;
                let remaining = end - pos;
                if length > remaining as u64 {
                    return Err(Error::LengthOutOfBounds { offset: length_start, length });
                }
                let length_usize = usize::try_from(length).map_err(|_| Error::LengthOutOfBounds {
                    offset: length_start,
                    length,
                })?;
                let data_end = pos + length_usize;
                if retained {
                    if let Some(Field::Bytes { tag: last_tag, data }) = fields.last_mut() {
                        if *last_tag == tag {
                            data.extend_from_slice(&input[pos..data_end]);
                        } else {
                            fields.push(Field::Bytes { tag, data: input[pos..data_end].to_vec() });
                        }
                    } else {
                        fields.push(Field::Bytes { tag, data: input[pos..data_end].to_vec() });
                    }
                }
                pos = data_end;
            }
            3 => {
                let length_start = pos;
                let (length, next) = read_varint(input, pos, end)?;
                pos = next;
                let remaining = end - pos;
                if length > remaining as u64 {
                    return Err(Error::LengthOutOfBounds { offset: length_start, length });
                }
                let length_usize = usize::try_from(length).map_err(|_| Error::LengthOutOfBounds {
                    offset: length_start,
                    length,
                })?;
                let child_end = pos + length_usize;
                let child_depth = depth + 1;
                if child_depth > max_depth {
                    return Err(Error::DepthExceeded { offset: key_start, depth: child_depth });
                }
                let child = decode_body(input, pos, child_end, max_depth, child_depth)?;
                if retained {
                    fields.push(Field::Nested { tag, message: child });
                }
                pos = child_end;
            }
            _ => unreachable!(),
        }

        previous = Some((tag, wire));
    }

    Ok(Message { fields })
}

pub fn decode_message(
    input: &[u8],
    start: usize,
    max_depth: usize,
) -> Result<(Message, usize), Error> {
    if max_depth > 32 {
        return Err(Error::InvalidDepthLimit { limit: max_depth });
    }
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let (length, body_start) = read_varint(input, start, input.len())?;
    let remaining = input.len() - body_start;
    if length > remaining as u64 {
        return Err(Error::LengthOutOfBounds { offset: start, length });
    }
    let body_len = usize::try_from(length).map_err(|_| Error::LengthOutOfBounds {
        offset: start,
        length,
    })?;
    let body_end = body_start + body_len;
    let message = decode_body(input, body_start, body_end, max_depth, 0)?;
    Ok((message, body_end))
}

pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
    LengthOutOfBounds { offset: usize, length: u64 },
    InvalidTag { offset: usize, tag: u64 },
    InvalidWire { offset: usize, wire: u8 },
    OutOfOrder { offset: usize, previous: u32, tag: u32 },
    Duplicate { offset: usize, tag: u32 },
    NonCanonical { offset: usize },
    DepthExceeded { offset: usize, depth: usize },
    InvalidDepthLimit { limit: usize },
}

fn b_put_varint(mut value: u64, output: &mut Vec<u8>) {
    while value > 0x7f {
        output.push(((value & 0x7f) as u8) | 0x80);
        value >>= 7;
    }
    output.push(value as u8);
}

fn b_read_varint(input: &[u8], position: &mut usize, end: usize) -> Result<u64, Error> {
    let origin = *position;
    let mut value = 0u64;
    let mut shift = 0;
    for byte_index in 0..10 {
        if *position == end {
            return Err(Error::Truncated { offset: end, start: origin });
        }
        let byte_offset = *position;
        let byte = input[*position];
        *position += 1;
        let low = (byte & 0x7f) as u64;
        if byte_index == 9 && (low > 1 || byte & 0x80 != 0) {
            return Err(Error::Overflow { offset: byte_offset, start: origin });
        }
        value |= low << shift;
        if byte & 0x80 == 0 {
            if byte_index != 0 && low == 0 {
                return Err(Error::NonCanonical { offset: origin });
            }
            return Ok(value);
        }
        shift += 7;
    }
    unreachable!()
}

fn b_encode_body(message: &Message, depth: usize, limit: usize, body: &mut Vec<u8>) -> Result<(), Error> {
    let mut prior: Option<(u32, u8)> = None;
    for (i, field) in message.fields.iter().enumerate() {
        let (tag, wire) = match field {
            Field::UInt { tag, .. } => (*tag, 0),
            Field::Bytes { tag, .. } => (*tag, 2),
            Field::Nested { tag, .. } => (*tag, 3),
        };
        if tag == 0 || tag > 15 {
            return Err(Error::InvalidTag { offset: i, tag: tag as u64 });
        }
        if let Some((previous, previous_wire)) = prior {
            if tag < previous {
                return Err(Error::OutOfOrder { offset: i, previous, tag });
            }
            if tag == previous && (wire != 2 || previous_wire != 2) {
                return Err(Error::Duplicate { offset: i, tag });
            }
        }

        b_put_varint(((tag as u64) << 3) | wire as u64, body);
        match field {
            Field::UInt { value, .. } => b_put_varint(*value, body),
            Field::Bytes { data, .. } => {
                b_put_varint(data.len() as u64, body);
                body.extend_from_slice(data);
            }
            Field::Nested { message, .. } => {
                let next_depth = depth + 1;
                if next_depth > limit {
                    return Err(Error::DepthExceeded { offset: i, depth: next_depth });
                }
                let mut nested_body = Vec::new();
                b_encode_body(message, next_depth, limit, &mut nested_body)?;
                b_put_varint(nested_body.len() as u64, body);
                body.extend_from_slice(&nested_body);
            }
        }
        prior = Some((tag, wire));
    }
    Ok(())
}

    framed.extend_from_slice(&body);
    Ok(framed)
}

                        }
                    } else {

                    }

    }

fn put_varint(mut value: u64, out: &mut Vec<u8>) {
    while value >= 0x80 {
        out.push((value as u8) | 0x80);
        value >>= 7;
    }
    out.push(value as u8);
}

fn encode_body(message: &Message, max_depth: usize, depth: usize) -> Result<Vec<u8>, Error> {
    let mut out = Vec::new();
    let mut previous: Option<(u32, u8)> = None;

    for (index, field) in message.fields.iter().enumerate() {
        let (tag, wire) = match field {
            Field::UInt { tag, .. } => (*tag, 0),
            Field::Bytes { tag, .. } => (*tag, 2),
            Field::Nested { tag, .. } => (*tag, 3),
        };

        if !(1..=15).contains(&tag) {
            return Err(Error::InvalidTag {
                offset: index,
                tag: tag as u64,
            });
        }

        if let Some((prev_tag, prev_wire)) = previous {
            if tag < prev_tag {
                return Err(Error::OutOfOrder {
                    offset: index,
                    previous: prev_tag,
                    tag,
                });
            }
            if tag == prev_tag && !(prev_wire == 2 && wire == 2) {
                return Err(Error::Duplicate { offset: index, tag });
            }
        }

        previous = Some((tag, wire));
        put_varint(((tag as u64) << 3) | wire as u64, &mut out);

        match field {
            Field::UInt { value, .. } => put_varint(*value, &mut out),
            Field::Bytes { data, .. } => {
                put_varint(data.len() as u64, &mut out);
                out.extend_from_slice(data);
            }
            Field::Nested { message, .. } => {
                let child_depth = depth + 1;
                if child_depth > max_depth {
                    return Err(Error::DepthExceeded {
                        offset: index,
                        depth: child_depth,
                    });

                let child = encode_body(message, max_depth, child_depth)?;
                put_varint(child.len() as u64, &mut out);
                out.extend_from_slice(&child);
            }
        }
    }
    Ok(out)

struct Decoder<'a> {
    input: &'a [u8],

impl<'a> Decoder<'a> {
    fn read_varint(&self, pos: &mut usize, end: usize) -> Result<u64, Error> {
        let start = *pos;
        let mut value = 0u64;
        for i in 0..10 {
            if *pos >= end {
                return Err(Error::Truncated {
                    offset: end,
                    start,
                });
            }
            let byte = self.input[*pos];
            let offset = *pos;
            *pos += 1;
            let payload = (byte & 0x7f) as u64;
            if i == 9 && (payload > 1 || (byte & 0x80) != 0) {
                return Err(Error::Overflow { offset, start });
            }
            value |= payload << (i * 7);
            if byte & 0x80 == 0 {
                if i > 0 && payload == 0 {
                    return Err(Error::NonCanonical { offset: start });
                }
                return Ok(value);
            }
        }
        unreachable!()
    }

    fn parse_body(
        &self,
        mut pos: usize,
        end: usize,
        max_depth: usize,
        depth: usize,
        retain: bool,
    ) -> Result<(Message, usize), Error> {
        let mut fields: Vec<Field> = Vec::new();
        let mut previous: Option<(u32, u8)> = None;

        while pos < end {
            let key_start = pos;
            let key = self.read_varint(&mut pos, end)?;
            let tag64 = key >> 3;
            let wire = (key & 7) as u8;

            if !(1..=255).contains(&tag64) {
                return Err(Error::InvalidTag {
                    offset: key_start,
                    tag: tag64,
                });
            }
            let tag = tag64 as u32;
            if !matches!(wire, 0 | 1 | 2 | 3 | 5) {
                return Err(Error::InvalidWire {
                    offset: key_start,
                    wire,
                });
            }

            if let Some((prev_tag, prev_wire)) = previous {
                if tag < prev_tag {
                    return Err(Error::OutOfOrder {
                        offset: key_start,
                        previous: prev_tag,
                        tag,
                    });
                }
                if tag == prev_tag && !(prev_wire == 2 && wire == 2) {
                    return Err(Error::Duplicate {
                        offset: key_start,
                        tag,
                    });
                }
            }
            previous = Some((tag, wire));

            let known = retain && tag <= 15;
            match wire {
                0 => {
                    let value = self.read_varint(&mut pos, end)?;
                    if known {
                        fields.push(Field::UInt { tag, value });
                    }
                }
                1 | 5 => {
                    let len = if wire == 1 { 8usize } else { 4usize };
                    if end.saturating_sub(pos) < len {
                        return Err(Error::LengthOutOfBounds {
                            offset: key_start,
                            length: len as u64,
                        });
                    }
                    pos += len;
                }
                2 => {
                    let length_start = pos;
                    let len = self.read_varint(&mut pos, end)?;
                    let remaining = end.saturating_sub(pos);
                    if len > remaining as u64 {
                        return Err(Error::LengthOutOfBounds {
                            offset: length_start,
                            length: len,
                        });
                    }
                    let len = usize::try_from(len).map_err(|_| Error::LengthOutOfBounds {
                        offset: length_start,
                        length: len,
                    })?;
                    if known {
                        let data = &self.input[pos..pos + len];
                        if let Some(Field::Bytes { data: accumulated, .. }) = fields.last_mut() {
                            if matches!(previous, Some((t, 2)) if t == tag) {
                                accumulated.extend_from_slice(data);
                            } else {
                                fields.push(Field::Bytes {
                                    tag,
                                    data: data.to_vec(),
                                });
                            }
                        } else {
                            fields.push(Field::Bytes {
                                tag,
                                data: data.to_vec(),
                            });
                        }
                    }
                    pos += len;
                }
                3 => {
                    let length_start = pos;
                    let len = self.read_varint(&mut pos, end)?;
                    let remaining = end.saturating_sub(pos);
                    if len > remaining as u64 {
                        return Err(Error::LengthOutOfBounds {
                            offset: length_start,
                            length: len,
                        });
                    }
                    let len_usize = usize::try_from(len).map_err(|_| Error::LengthOutOfBounds {
                        offset: length_start,
                        length: len,
                    })?;
                    let child_depth = depth + 1;
                    if child_depth > max_depth {
                        return Err(Error::DepthExceeded {
                            offset: key_start,
                            depth: child_depth,
                        });
                    }
                    let child_end = pos + len_usize;
                    let (child, _) =
                        self.parse_body(pos, child_end, max_depth, child_depth, known)?;
                    pos = child_end;
                    if known {
                        fields.push(Field::Nested {
                            tag,
                            message: child,
                        });
                    }
                }
                _ => unreachable!(),
            }
        }

        Ok((Message { fields }, pos))
    }
}

pub fn decode_message(
    input: &[u8],
    start: usize,
    max_depth: usize,
) -> Result<(Message, usize), Error> {
    if max_depth > 32 {
        return Err(Error::InvalidDepthLimit { limit: max_depth });
    }
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }
    let decoder = Decoder { input };
    let mut pos = start;
    let header_start = pos;
    let length = decoder.read_varint(&mut pos, input.len())?;
    let remaining = input.len().saturating_sub(pos);
    if length > remaining as u64 {
        return Err(Error::LengthOutOfBounds {
            offset: header_start,
            length,
        });
    }
    let body_len = usize::try_from(length).map_err(|_| Error::LengthOutOfBounds {
        offset: header_start,
        length,
    })?;
    let end = pos + body_len;
    let (message, cursor) = decoder.parse_body(pos, end, max_depth, 0, true)?;
    Ok((message, cursor))
}

fn enc_varint(value: u64, out: &mut Vec<u8>) {
    let mut v = value;
    loop {
        let mut b = (v & 0x7f) as u8;
        v >>= 7;
        if v != 0 {
            b |= 0x80;
        }
        out.push(b);
        if v == 0 {
            break;
        }
    }
}

fn dec_varint(data: &[u8], cursor: &mut usize, limit: usize) -> Result<u64, Error> {
    let begin = *cursor;
    let mut acc = 0u64;
    let mut shift = 0;
    let mut count = 0usize;
    loop {
        if *cursor == limit {
            return Err(Error::Truncated { offset: limit, start: begin });
        }
        let at = *cursor;
        let b = data[at];
        if count == 9 && ((b & 0x7f) > 1 || b & 0x80 != 0) {
            return Err(Error::Overflow { offset: at, start: begin });
        }
        acc |= ((b & 127) as u64) << shift;
        *cursor += 1;
        count += 1;
        if b & 128 == 0 {
            if count > 1 && b & 127 == 0 {
                return Err(Error::NonCanonical { offset: begin });
            }
            return Ok(acc);
        }
        shift += 7;
    }
}

fn encode_content(msg: &Message, limit: usize, depth: usize) -> Result<Vec<u8>, Error> {

        let (tag, wire) = match f {
            Field::UInt { tag, .. } => (*tag, 0u8),
            Field::Bytes { tag, .. } => (*tag, 2u8),
            Field::Nested { tag, .. } => (*tag, 3u8),

            return Err(Error::InvalidTag { offset: i, tag: tag as u64 });
        }
        if let Some((p, pw)) = last {
            if tag < p {
                return Err(Error::OutOfOrder { offset: i, previous: p, tag });
            }
            if tag == p && (pw != 2 || wire != 2) {
                return Err(Error::Duplicate { offset: i, tag });
            }
        }

        enc_varint(((tag as u64) << 3) | wire as u64, &mut out);
        match f {
            Field::UInt { value, .. } => enc_varint(*value, &mut out),
            Field::Bytes { data, .. } => {
                enc_varint(data.len() as u64, &mut out);
                out.extend_from_slice(data);

            Field::Nested { message, .. } => {
                let child_depth = depth + 1;
                if child_depth > limit {

            }
        }

    }
    Ok(out)

            if tag64 < 1 || tag64 > 255 {
                return Err(Error::InvalidTag {
                    offset: key_at,
                    tag: tag64,
                });
            }
            let tag = tag64 as u32;
            if wire != 0 && wire != 1 && wire != 2 && wire != 3 && wire != 5 {
                return Err(Error::InvalidWire {
                    offset: key_at,
                    wire,
                });
            }
            if let Some((prev, prev_wire)) = last {
                if tag < prev {
                    return Err(Error::OutOfOrder {
                        offset: key_at,
                        previous: prev,
                        tag,
                    });
                }
                if tag == prev && !(prev_wire == 2 && wire == 2) {
                    return Err(Error::Duplicate { offset: key_at, tag });
                }
            }
            last = Some((tag, wire));
            let keep = retain && tag <= 15;

            match wire {
                0 => {
                    let value = self.varint(&mut cursor, end)?;
                    if keep {
                        result.fields.push(Field::UInt { tag, value });
                    }
                }
                1 | 5 => {
                    let width = if wire == 1 { 8usize } else { 4usize };
                    if end - cursor < width {
                        return Err(Error::LengthOutOfBounds {
                            offset: key_at,
                            length: width as u64,
                        });
                    }
                    cursor += width;
                }
                2 => {
                    let len_at = cursor;
                    let length = self.varint(&mut cursor, end)?;
                    if length > (end - cursor) as u64 {
                        return Err(Error::LengthOutOfBounds {
                            offset: len_at,
                            length,
                        });
                    }
                    let length = usize::try_from(length).map_err(|_| {
                        Error::LengthOutOfBounds {
                            offset: len_at,
                            length,
                        }
                    })?;
                    if keep {
                        if let Some(Field::Bytes { tag: prev_tag, data }) =
                            result.fields.last_mut()
                        {
                            if *prev_tag == tag {
                                data.extend_from_slice(&self.bytes[cursor..cursor + length]);
                            } else {
                                result.fields.push(Field::Bytes {
                                    tag,
                                    data: self.bytes[cursor..cursor + length].to_vec(),
                                });
                            }
                        } else {
                            result.fields.push(Field::Bytes {
                                tag,
                                data: self.bytes[cursor..cursor + length].to_vec(),
                            });
                        }
                    }
                    cursor += length;
                }
                3 => {
                    let len_at = cursor;
                    let length = self.varint(&mut cursor, end)?;
                    if length > (end - cursor) as u64 {
                        return Err(Error::LengthOutOfBounds {
                            offset: len_at,
                            length,
                        });
                    }
                    let length_usize = usize::try_from(length).map_err(|_| {
                        Error::LengthOutOfBounds {
                            offset: len_at,
                            length,
                        }
                    })?;
                    let child_depth = depth + 1;
                    if child_depth > max_depth {
                        return Err(Error::DepthExceeded {
                            offset: key_at,
                            depth: child_depth,
                        });
                    }
                    let child_end = cursor + length_usize;
                    let child = self.body(cursor, child_end, max_depth, child_depth, keep)?;
                    cursor = child_end;
                    if keep {
                        result.fields.push(Field::Nested {
                            tag,
                            message: child,
                        });
                    }
                }
                _ => unreachable!(),
            }
        }
        Ok(result)
    }
}

pub fn decode_message(
    input: &[u8],
    start: usize,
    max_depth: usize,
) -> Result<(Message, usize), Error> {
    if max_depth > 32 {
        return Err(Error::InvalidDepthLimit { limit: max_depth });
    }
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let parser = Parser { bytes: input };
    let mut cursor = start;
    let header = cursor;
    let frame_len = parser.varint(&mut cursor, input.len())?;
    if frame_len > (input.len() - cursor) as u64 {
        return Err(Error::LengthOutOfBounds {
            offset: header,
            length: frame_len,
        });
    }
    let frame_len = usize::try_from(frame_len).map_err(|_| Error::LengthOutOfBounds {
        offset: header,
        length: frame_len,
    })?;
    let frame_end = cursor + frame_len;
    let message = parser.body(cursor, frame_end, max_depth, 0, true)?;
    Ok((message, frame_end))
}

fn put_varint(mut value: u64, out: &mut Vec<u8>) {
    while value >= 0x80 {
        out.push((value as u8) | 0x80);
        value >>= 7;
    }
    out.push(value as u8);
}

fn read_varint(input: &[u8], pos: &mut usize, end: usize) -> Result<u64, Error> {
    let start = *pos;
    let mut value = 0u64;
    for i in 0..10 {
        if *pos >= end {
            return Err(Error::Truncated { offset: end, start });
        }
        let b = input[*pos];
        let at = *pos;
        *pos += 1;
        let payload = (b & 0x7f) as u64;
        if i == 9 && (payload > 1 || b & 0x80 != 0) {
            return Err(Error::Overflow { offset: at, start });
        }
        value |= payload << (7 * i);
        if b & 0x80 == 0 {
            if i > 0 && payload == 0 {
                return Err(Error::NonCanonical { offset: start });
            }
            return Ok(value);
        }
    }
    unreachable!()
}

fn encode_body(message: &Message, depth: usize, max_depth: usize, out: &mut Vec<u8>) -> Result<(), Error> {
    let mut previous: Option<(u32, u8)> = None;
    for (index, field) in message.fields.iter().enumerate() {
        let (tag, wire) = match field {
            Field::UInt { tag, .. } => (*tag, 0),
            Field::Bytes { tag, .. } => (*tag, 2),
            Field::Nested { tag, .. } => (*tag, 3),
        };
        if !(1..=15).contains(&tag) {
            return Err(Error::InvalidTag { offset: index, tag: tag as u64 });
        }
        if let Some((prev, prev_wire)) = previous {
            if tag < prev {
                return Err(Error::OutOfOrder { offset: index, previous: prev, tag });
            }
            if tag == prev && (wire != 2 || prev_wire != 2) {
                return Err(Error::Duplicate { offset: index, tag });
            }
        }

        let key = ((tag as u64) << 3) | wire as u64;
        put_varint(key, out);
        match field {
            Field::UInt { value, .. } => put_varint(*value, out),
            Field::Bytes { data, .. } => {
                put_varint(data.len() as u64, out);
                out.extend_from_slice(data);
            }
            Field::Nested { message, .. } => {
                let child_depth = depth + 1;
                if child_depth > max_depth {
                    return Err(Error::DepthExceeded { offset: index, depth: child_depth });
                }
                let mut child = Vec::new();
                encode_body(message, child_depth, max_depth, &mut child)?;
                put_varint(child.len() as u64, out);
                out.extend_from_slice(&child);
            }
        }
        previous = Some((tag, wire));
    }
    Ok(())
}

    let mut shift = 0u32;

fn emit_body(message: &Message, depth: usize, limit: usize) -> Result<Vec<u8>, Error> {

    let mut prior_tag = 0u32;
    let mut prior_wire = 255u8;
    let mut has_prior = false;

    for (i, field) in message.fields.iter().enumerate() {
        let (tag, wire) = match field {
            Field::UInt { tag, .. } => (*tag, 0u8),
            Field::Bytes { tag, .. } => (*tag, 2u8),
            Field::Nested { tag, .. } => (*tag, 3u8),
        };

            return Err(Error::InvalidTag { offset: i, tag: tag as u64 });
        }
        if has_prior {
            if tag < prior_tag {
                return Err(Error::OutOfOrder { offset: i, previous: prior_tag, tag });
            }
            if tag == prior_tag && !(prior_wire == 2 && wire == 2) {
                return Err(Error::Duplicate { offset: i, tag });
            }
        }
        prior_tag = tag;
        prior_wire = wire;
        has_prior = true;

        let key = ((tag as u64) << 3) | wire as u64;
        enc_varint(key, &mut bytes);
        match field {
            Field::UInt { value, .. } => enc_varint(*value, &mut bytes),
            Field::Bytes { data, .. } => {
                enc_varint(data.len() as u64, &mut bytes);
                bytes.extend_from_slice(data);
            }
            Field::Nested { message, .. } => {
                let child_depth = depth + 1;
                if child_depth > limit {
                    return Err(Error::DepthExceeded { offset: i, depth: child_depth });
                }
                let body = emit_body(message, child_depth, limit)?;
                enc_varint(body.len() as u64, &mut bytes);
                bytes.extend_from_slice(&body);
            }
        }
    }
    Ok(bytes)
}

pub fn encode_message(message: &Message, max_depth: usize) -> Result<Vec<u8>, Error> {
    depth_limit(max_depth)?;
    let body = emit_body(message, 0, max_depth)?;
    let mut framed = Vec::new();
    enc_varint(body.len() as u64, &mut framed);
    framed.extend_from_slice(&body);
    Ok(framed)
}

fn decode_body(input: &[u8], begin: usize, finish: usize, depth: usize, limit: usize) -> Result<Message, Error> {
    let mut cursor = begin;
    let mut result = Message { fields: Vec::new() };
    let mut prev_tag: Option<u32> = None;
    let mut prev_wire = 0u8;

    while cursor < finish {
        let key_position = cursor;
        let key = dec_varint(input, &mut cursor, finish)?;
        let raw_tag = key >> 3;
        let wire = (key & 7) as u8;
        if raw_tag == 0 || raw_tag > 255 {
            return Err(Error::InvalidTag { offset: key_position, tag: raw_tag });
        }
        if wire != 0 && wire != 1 && wire != 2 && wire != 3 && wire != 5 {
            return Err(Error::InvalidWire { offset: key_position, wire });
        }
        let tag = raw_tag as u32;
        if let Some(previous) = prev_tag {
            if tag < previous {
                return Err(Error::OutOfOrder { offset: key_position, previous, tag });
            }
            if tag == previous && !(prev_wire == 2 && wire == 2) {
                return Err(Error::Duplicate { offset: key_position, tag });
            }
        }
        prev_tag = Some(tag);
        prev_wire = wire;

        match wire {
            0 => {
                let value = dec_varint(input, &mut cursor, finish)?;
                if tag <= 15 {
                    result.fields.push(Field::UInt { tag, value });
                }
            }
            1 | 5 => {
                let width = if wire == 1 { 8usize } else { 4usize };
                if finish - cursor < width {
                    return Err(Error::LengthOutOfBounds { offset: key_position, length: width as u64 });
                }
                cursor += width;
            }
            2 | 3 => {
                let length_position = cursor;
                let length = dec_varint(input, &mut cursor, finish)?;
                if length > (finish - cursor) as u64 {
                    return Err(Error::LengthOutOfBounds { offset: length_position, length });
                }
                let count = usize::try_from(length).map_err(|_| Error::LengthOutOfBounds {
                    offset: length_position,
                    length,
                })?;
                let payload_start = cursor;
                let payload_end = cursor + count;
                if wire == 2 {
                    if tag <= 15 {
                        let chunk = input[payload_start..payload_end].to_vec();
                        match result.fields.last_mut() {
                            Some(Field::Bytes { tag: previous, data }) if *previous == tag => {
                                data.extend_from_slice(&chunk);
                            }
                            _ => result.fields.push(Field::Bytes { tag, data: chunk }),
                        }
                    }
                } else {
                    let next_depth = depth + 1;
                    if next_depth > limit {
                        return Err(Error::DepthExceeded { offset: key_position, depth: next_depth });
                    }
                    let child = decode_body(input, payload_start, payload_end, next_depth, limit)?;
                    if tag <= 15 {
                        result.fields.push(Field::Nested { tag, message: child });
                    }
                }
                cursor = payload_end;
            }
            _ => unreachable!(),
        }
    }
    Ok(result)
}

pub fn decode_message(input: &[u8], start: usize, max_depth: usize) -> Result<(Message, usize), Error> {
    depth_limit(max_depth)?;
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }
    let frame_start = start;
    let mut cursor = start;
    let frame_size = dec_varint(input, &mut cursor, input.len())?;
    if frame_size > (input.len() - cursor) as u64 {
        return Err(Error::LengthOutOfBounds { offset: frame_start, length: frame_size });
    }
    let body_size = usize::try_from(frame_size).map_err(|_| Error::LengthOutOfBounds {
        offset: frame_start,
        length: frame_size,
    })?;
    let frame_end = cursor + body_size;
    let decoded = decode_body(input, cursor, frame_end, 0, max_depth)?;
    Ok((decoded, frame_end))
}
