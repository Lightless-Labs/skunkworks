- Validation order: time admission comes first; transaction lookup precedes key/range checks, and range checks precede cursor/numeric checks.
- A transaction sees the newest committed version at or before its begin revision; its own write replaces that view, including tombstones and expired values.
- TTL deadlines are absolute and expire at `deadline <= now`; TTL zero writes a tombstone.
- Scan offsets apply only without a cursor; a valid cursor resumes strictly after `last`, with no offset reapplied.
- GC keeps each key’s newest version plus the newest version visible to each active transaction, regardless of liveness.
- Restore considers the union of current live keys and unexpired saved keys; an empty union does not advance revision.

### Solution A

use std::collections::{BTreeMap, BTreeSet, HashMap};

#[derive(Clone)]
struct Entry {
    value: String,
    deadline: Option<u64>,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    entry: Option<Entry>,
}

#[derive(Clone)]
struct Write {
    entry: Option<Entry>,
}

struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Write>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: HashMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Entry>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: HashMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards { previous: self.clock, now });
        }
        self.clock = now;
        Ok(())
    }

    fn tx_check(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn selected_entry(&self, key: &str, tx: Option<u64>) -> Option<Entry> {
        if let Some(id) = tx {
            let t = self.transactions.get(&id)?;
            if let Some(w) = t.writes.get(key) {
                return w.entry.clone();
            }
            self.data.get(key)?
                .iter()
                .rev()
                .find(|v| v.revision <= t.begin_revision)
                .and_then(|v| v.entry.clone())
        } else {
            self.data.get(key)?.last()?.entry.clone()
        }
    }

    fn visible(&self, key: &str, tx: Option<u64>, now: u64) -> Option<String> {
        self.selected_entry(key, tx)
            .filter(|e| e.deadline.map_or(true, |d| d > now))
            .map(|e| e.value)
    }

    fn append(&mut self, key: String, entry: Option<Entry>, revision: u64) {
        self.data.entry(key).or_default().push(Version { revision, entry });
    }

    fn validate_key(key: &str) -> Result<(), Error> {
        if key.is_empty() { Err(Error::KeyEmpty) } else { Ok(()) }
    }

    fn validate_name(name: &str) -> Result<(), Error> {
        if name.is_empty() { return Err(Error::NameEmpty); }
        for (at, ch) in name.char_indices() {
            let valid = if at == 0 {
                ch.is_ascii_lowercase()
            } else {
                ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
            };
            if !valid { return Err(Error::NameInvalid { at }); }
        }
        Ok(())
    }

    fn range_check(start: &str, end: Option<&str>) -> Result<(), Error> {
        if end.map_or(false, |e| e <= start) { Err(Error::InvalidRange) } else { Ok(()) }
    }

    fn matching_keys(
        &self,
        tx: Option<u64>,
        now: u64,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Vec<(String, String)> {
        let mut out = Vec::new();
        let mut keys = BTreeSet::new();
        for key in self.data.keys() { keys.insert(key.as_str()); }
        if let Some(id) = tx {
            if let Some(t) = self.transactions.get(&id) {
                for key in t.writes.keys() { keys.insert(key.as_str()); }
            }
        }
        for key in keys {
            if key < start || end.map_or(false, |e| key >= e) || prefix.map_or(false, |p| !key.starts_with(p)) {
                continue;
            }
            if let Some(value) = self.visible(key, tx, now) {
                out.push((key.to_string(), value));
            }
        }
        out
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) { return Err(Error::TransactionExists(id)); }
        self.transactions.insert(id, Transaction {
            begin_revision: self.revision,
            read_only,
            writes: BTreeMap::new(),
        });
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let t = self.transactions.get(&id).ok_or(Error::UnknownTransaction(id))?;
        for key in t.writes.keys() {
            if self.data.get(key).and_then(|v| v.last()).map_or(false, |v| v.revision > t.begin_revision) {
                return Err(Error::Conflict { tx: id, key: key.clone() });
            }
        }
        if t.writes.is_empty() {
            self.transactions.remove(&id);
            return Ok(self.revision);
        }
        let writes = t.writes.clone();
        self.revision += 1;
        let rev = self.revision;
        for (key, write) in writes { self.append(key, write.entry, rev); }
        self.transactions.remove(&id);
        Ok(rev)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.remove(&id).is_none() { return Err(Error::UnknownTransaction(id)); }
        Ok(())
    }

    pub fn set(&mut self, now: u64, tx: Option<u64>, key: &str, value: &str, ttl: Option<u64>) -> Result<(), Error> {
        self.admit(now)?;
        self.tx_check(tx)?;
        Self::validate_key(key)?;
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only { return Err(Error::ReadOnly(id)); }
        }
        let entry = match ttl {
            Some(0) => None,
            Some(d) => Some(Entry {
                value: value.to_string(),
                deadline: Some(now.checked_add(d).ok_or(Error::DeadlineOverflow)?),
            }),
            None => Some(Entry { value: value.to_string(), deadline: None }),
        };
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_string(), Write { entry });
        } else {
            self.revision += 1;
            self.append(key.to_string(), entry, self.revision);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.tx_check(tx)?;
        Self::validate_key(key)?;
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only { return Err(Error::ReadOnly(id)); }
        }
        let existed = self.visible(key, tx, now).is_some();
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_string(), Write { entry: None });
        } else {
            self.revision += 1;
            self.append(key.to_string(), None, self.revision);
        }
        Ok(existed)
    }

    pub fn get(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.tx_check(tx)?;
        Self::validate_key(key)?;
        Ok(self.visible(key, tx, now))
    }

    pub fn count(&mut self, now: u64, tx: Option<u64>, value: &str) -> Result<usize, Error> {
        self.admit(now)?;
        self.tx_check(tx)?;
        let keys: Vec<String> = self.data.keys().cloned().collect();
        let mut n = 0;
        for key in keys {
            if self.visible(&key, tx, now).as_deref() == Some(value) { n += 1; }
        }
        if let Some(id) = tx {
            for key in self.transactions.get(&id).unwrap().writes.keys() {
                if !self.data.contains_key(key) && self.visible(key, tx, now).as_deref() == Some(value) { n += 1; }
            }
        }
        Ok(n)
    }

    pub fn scan(&mut self, now: u64, tx: Option<u64>, start: &str, end: Option<&str>, prefix: Option<&str>, offset: usize, limit: usize, cursor: Option<&Cursor>) -> Result<Page, Error> {
        self.admit(now)?;
        self.tx_check(tx)?;
        Self::range_check(start, end)?;
        if let Some(c) = cursor {
            if c.start != start || c.end.as_deref() != end || c.prefix.as_deref() != prefix || c.offset != offset ||
                c.last.is_empty() || c.last < start ||
                end.map_or(false, |e| c.last.as_str() >= e) ||
                prefix.map_or(false, |p| !c.last.starts_with(p)) {
                return Err(Error::CursorMismatch);
            }
        }
        if limit == 0 { return Ok(Page { items: Vec::new(), next: None }); }
        let all = self.matching_keys(tx, now, start, end, prefix);
        let after: Vec<(String, String)> = if let Some(c) = cursor {
            all.into_iter().filter(|(k, _)| k > &c.last).collect()
        } else {
            all.into_iter().skip(offset).collect()
        };
        let has_more = after.len() > limit;
        let items: Vec<_> = after.into_iter().take(limit).collect();
        let next = if has_more {
            items.last().map(|(last, _)| Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: last.clone(),
            })
        } else { None };
        Ok(Page { items, next })
    }

    pub fn aggregate(&mut self, now: u64, tx: Option<u64>, start: &str, end: Option<&str>, prefix: Option<&str>) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.tx_check(tx)?;
        Self::range_check(start, end)?;
        let items = self.matching_keys(tx, now, start, end, prefix);
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in items {
            let parsed = parse_canonical(&value).ok_or_else(|| Error::InvalidNumber { key: key.clone(), value: value.clone() })?;
            sum = sum.checked_add(parsed).ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        let exists = self.snapshots.contains_key(name);
        if !exists && self.snapshots.len() >= 3 { return Err(Error::SnapshotLimit); }
        let mut copy = BTreeMap::new();
        for (key, versions) in &self.data {
            if let Some(entry) = versions.last().and_then(|v| v.entry.clone()) {
                if entry.deadline.map_or(true, |d| d > now) { copy.insert(key.clone(), entry); }
            }
        }
        self.snapshots.insert(name.to_string(), copy);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        let saved = self.snapshots.get(name).cloned().ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() { return Err(Error::ActiveTransactions); }
        let mut restored = BTreeMap::new();
        for (key, entry) in saved {
            if entry.deadline.map_or(true, |d| d > now) { restored.insert(key, entry); }
        }
        let mut union = BTreeSet::new();
        for (key, versions) in &self.data {
            if versions.last().and_then(|v| v.entry.as_ref()).map_or(false, |e| e.deadline.map_or(true, |d| d > now)) {
                union.insert(key.clone());
            }
        }
        for key in restored.keys() { union.insert(key.clone()); }
        if union.is_empty() { return Ok(self.revision); }
        self.revision += 1;
        let rev = self.revision;
        for key in union {
            self.append(key.clone(), restored.get(&key).cloned(), rev);
        }
        Ok(rev)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        if self.snapshots.remove(name).is_none() { return Err(Error::SnapshotMissing(name.to_string())); }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let mut removed = 0usize;
        let ids: Vec<String> = self.data.keys().cloned().collect();
        for key in ids {
            let versions = self.data.get_mut(&key).unwrap();
            let mut keep = BTreeSet::new();
            if !versions.is_empty() { keep.insert(versions.len() - 1); }
            for tx in self.transactions.values() {
                if let Some(i) = versions.iter().rposition(|v| v.revision <= tx.begin_revision) { keep.insert(i); }
            }
            let old = std::mem::take(versions);
            for (i, v) in old.into_iter().enumerate() {
                if keep.contains(&i) { versions.push(v); } else { removed += 1; }
            }
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let keys = self.data.len();
        let versions: usize = self.data.values().map(Vec::len).sum();
        let tombstones = self.data.values().flat_map(|v| v.iter()).filter(|v| v.entry.is_none()).count();
        let live_keys = self.data.iter().filter(|(_, vs)| {
            vs.last().and_then(|v| v.entry.as_ref()).map_or(false, |e| e.deadline.map_or(true, |d| d > now))
        }).count();
        Stats {
            clock: self.clock,
            revision: self.revision,
            keys,
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions: self.transactions.values().filter(|t| t.read_only).count(),
            staged_writes: self.transactions.values().map(|t| t.writes.len()).sum(),
            snapshots: self.snapshots.len(),
        }.pipe(Ok)
    }
}

trait Pipe: Sized {
    fn pipe<T>(self, f: impl FnOnce(Self) -> T) -> T { f(self) }
}
impl<T> Pipe for T {}

fn parse_canonical(s: &str) -> Option<i64> {
    if s.is_empty() || s == "-0" { return None; }
    let digits = if let Some(rest) = s.strip_prefix('-') { rest } else { s };
    if digits.is_empty() || (digits.len() > 1 && digits.starts_with('0')) ||
        !digits.bytes().all(|b| b.is_ascii_digit()) ||
        (s.starts_with('0') && s.len() > 1) {
        return None;
    }
    s.parse::<i64>().ok()
}

use std::collections::{BTreeMap, BTreeSet, HashMap};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),

    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {

    pub sum: i64,

#[derive(Clone)]
struct Record {
    rev: u64,
    payload: Option<(String, Option<u64>)>,
}

#[derive(Clone)]
struct Pending {
    payload: Option<(String, Option<u64>)>,
}

struct TxState {
    at: u64,
    ro: bool,
    pending: BTreeMap<String, Pending>,
}

pub struct Store {
    time: u64,
    rev: u64,
    records: BTreeMap<String, Vec<Record>>,
    txs: BTreeMap<u64, TxState>,
    saved: BTreeMap<String, BTreeMap<String, (String, Option<u64>)>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            time: 0,
            rev: 0,
            records: BTreeMap::new(),
            txs: BTreeMap::new(),
            saved: BTreeMap::new(),
        }
    }

    fn time(&mut self, now: u64) -> Result<(), Error> {
        if now < self.time {
            return Err(Error::TimeBackwards { previous: self.time, now });
        }
        self.time = now;
        Ok(())
    }

    fn check_tx(&self, id: Option<u64>) -> Result<(), Error> {
        if let Some(id) = id {
            if !self.txs.contains_key(&id) { return Err(Error::UnknownTransaction(id)); }
        }
        Ok(())
    }

    fn selected(&self, key: &str, id: Option<u64>) -> Option<(String, Option<u64>)> {
        if let Some(id) = id {
            let tx = self.txs.get(&id)?;
            if let Some(w) = tx.pending.get(key) { return w.payload.clone(); }
            self.records.get(key)?.iter().rev()
                .find(|r| r.rev <= tx.at).and_then(|r| r.payload.clone())
        } else {
            self.records.get(key)?.last()?.payload.clone()
        }
    }

    fn value(&self, key: &str, id: Option<u64>, now: u64) -> Option<String> {
        self.selected(key, id).and_then(|(v, deadline)| {
            if deadline.map_or(true, |d| d > now) { Some(v) } else { None }
        })
    }

    fn put(&mut self, key: String, payload: Option<(String, Option<u64>)>, rev: u64) {
        self.records.entry(key).or_default().push(Record { rev, payload });
    }

    fn key_ok(key: &str) -> Result<(), Error> {
        if key.is_empty() { Err(Error::KeyEmpty) } else { Ok(()) }
    }

    fn name_ok(name: &str) -> Result<(), Error> {
        if name.is_empty() { return Err(Error::NameEmpty); }
        for (pos, c) in name.char_indices() {
            let okay = if pos == 0 { c.is_ascii_lowercase() }
                else { c.is_ascii_lowercase() || c.is_ascii_digit() || c == '_' };
            if !okay { return Err(Error::NameInvalid { at: pos }); }
        }
        Ok(())
    }

    fn range_ok(start: &str, end: Option<&str>) -> Result<(), Error> {
        if end.is_some_and(|e| e <= start) { Err(Error::InvalidRange) } else { Ok(()) }
    }

    fn collect(&self, id: Option<u64>, now: u64, start: &str, end: Option<&str>, prefix: Option<&str>) -> Vec<(String, String)> {
        let mut keys = BTreeSet::<String>::new();
        keys.extend(self.records.keys().cloned());
        if let Some(id) = id {
            if let Some(tx) = self.txs.get(&id) {
                keys.extend(tx.pending.keys().cloned());
            }
        }
        keys.into_iter().filter_map(|key| {
            if key.as_str() < start || end.is_some_and(|e| key.as_str() >= e) ||
                prefix.is_some_and(|p| !key.starts_with(p)) { return None; }
            self.value(&key, id, now).map(|v| (key, v))
        }).collect()
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.time(now)?;
        if self.txs.contains_key(&id) { return Err(Error::TransactionExists(id)); }
        self.txs.insert(id, TxState { at: self.rev, ro: read_only, pending: BTreeMap::new() });
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.time(now)?;
        let tx = self.txs.get(&id).ok_or(Error::UnknownTransaction(id))?;
        for key in tx.pending.keys() {
            if self.records.get(key).and_then(|v| v.last()).is_some_and(|r| r.rev > tx.at) {
                return Err(Error::Conflict { tx: id, key: key.clone() });
            }
        }
        if tx.pending.is_empty() {
            self.txs.remove(&id);
            return Ok(self.rev);
        }
        let pending = tx.pending.clone();
        self.rev += 1;
        let new_rev = self.rev;
        for (key, value) in pending { self.put(key, value.payload, new_rev); }
        self.txs.remove(&id);
        Ok(new_rev)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.time(now)?;
        if self.txs.remove(&id).is_none() { return Err(Error::UnknownTransaction(id)); }
        Ok(())
    }

    pub fn set(&mut self, now: u64, tx: Option<u64>, key: &str, value: &str, ttl: Option<u64>) -> Result<(), Error> {
        self.time(now)?;
        self.check_tx(tx)?;
        Self::key_ok(key)?;
        if let Some(id) = tx {
            if self.txs.get(&id).unwrap().ro { return Err(Error::ReadOnly(id)); }
        }
        let payload = match ttl {
            Some(0) => None,
            Some(d) => Some((value.to_string(), Some(now.checked_add(d).ok_or(Error::DeadlineOverflow)?))),
            None => Some((value.to_string(), None)),
        };
        if let Some(id) = tx {
            self.txs.get_mut(&id).unwrap().pending.insert(key.to_owned(), Pending { payload });
        } else {
            self.rev += 1;
            self.put(key.to_owned(), payload, self.rev);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.time(now)?;
        self.check_tx(tx)?;
        Self::key_ok(key)?;
        if let Some(id) = tx {
            if self.txs.get(&id).unwrap().ro { return Err(Error::ReadOnly(id)); }
        }
        let was_live = self.value(key, tx, now).is_some();
        if let Some(id) = tx {
            self.txs.get_mut(&id).unwrap().pending.insert(key.to_owned(), Pending { payload: None });
        } else {
            self.rev += 1;
            self.put(key.to_owned(), None, self.rev);
        }
        Ok(was_live)
    }

    pub fn get(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<Option<String>, Error> {
        self.time(now)?;
        self.check_tx(tx)?;
        Self::key_ok(key)?;
        Ok(self.value(key, tx, now))
    }

    pub fn count(&mut self, now: u64, tx: Option<u64>, value: &str) -> Result<usize, Error> {
        self.time(now)?;
        self.check_tx(tx)?;
        Ok(self.collect(tx, now, "", None, None).iter().filter(|(_, v)| v == value).count())
    }

    pub fn scan(&mut self, now: u64, tx: Option<u64>, start: &str, end: Option<&str>, prefix: Option<&str>, offset: usize, limit: usize, cursor: Option<&Cursor>) -> Result<Page, Error> {
        self.time(now)?;
        self.check_tx(tx)?;
        Self::range_ok(start, end)?;
        if let Some(c) = cursor {
            if c.start != start || c.end.as_deref() != end || c.prefix.as_deref() != prefix ||
                c.offset != offset || c.last.is_empty() || c.last.as_str() < start ||
                end.is_some_and(|e| c.last.as_str() >= e) ||
                prefix.is_some_and(|p| !c.last.starts_with(p)) {
                return Err(Error::CursorMismatch);
            }
        }
        if limit == 0 { return Ok(Page { items: Vec::new(), next: None }); }
        let all = self.collect(tx, now, start, end, prefix);
        let selected: Vec<_> = if let Some(c) = cursor {
            all.into_iter().filter(|(k, _)| k.as_str() > c.last.as_str()).collect()
        } else {
            all.into_iter().skip(offset).collect()
        };
        let more = selected.len() > limit;
        let items: Vec<_> = selected.into_iter().take(limit).collect();
        let next = if more {
            items.last().map(|(last, _)| Cursor {
                start: start.to_owned(),
                end: end.map(str::to_owned),
                prefix: prefix.map(str::to_owned),
                offset,
                last: last.clone(),
            })
        } else { None };
        Ok(Page { items, next })
    }

    pub fn aggregate(&mut self, now: u64, tx: Option<u64>, start: &str, end: Option<&str>, prefix: Option<&str>) -> Result<Aggregate, Error> {
        self.time(now)?;
        self.check_tx(tx)?;
        Self::range_ok(start, end)?;
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in self.collect(tx, now, start, end, prefix) {
            let n = parse_number(&value).ok_or_else(|| Error::InvalidNumber { key: key.clone(), value: value.clone() })?;
            sum = sum.checked_add(n).ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.time(now)?;
        Self::name_ok(name)?;
        if !self.saved.contains_key(name) && self.saved.len() == 3 { return Err(Error::SnapshotLimit); }
        let mut snapshot = BTreeMap::new();
        for (key, records) in &self.records {
            if let Some((value, deadline)) = records.last().and_then(|r| r.payload.clone()) {
                if deadline.map_or(true, |d| d > now) { snapshot.insert(key.clone(), (value, deadline)); }
            }
        }
        self.saved.insert(name.to_owned(), snapshot);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.time(now)?;
        Self::name_ok(name)?;
        let snapshot = self.saved.get(name).cloned().ok_or_else(|| Error::SnapshotMissing(name.to_owned()))?;
        if !self.txs.is_empty() { return Err(Error::ActiveTransactions); }
        let snapshot: BTreeMap<_, _> = snapshot.into_iter().filter(|(_, (_, deadline))| deadline.map_or(true, |d| d > now)).collect();
        let current: BTreeSet<_> = self.records.iter().filter_map(|(k, vs)| {
            vs.last().and_then(|r| r.payload.as_ref()).and_then(|(_, d)| {
                if d.map_or(true, |x| x > now) { Some(k.clone()) } else { None }
            })
        }).collect();
        let union: BTreeSet<_> = current.union(&snapshot.keys().cloned().collect()).cloned().collect();
        if union.is_empty() { return Ok(self.rev); }
        self.rev += 1;
        let revision = self.rev;
        for key in union {
            let payload = snapshot.get(&key).map(|(v, d)| (v.clone(), *d));
            self.put(key, payload, revision);
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.time(now)?;
        Self::name_ok(name)?;
        if self.saved.remove(name).is_none() { return Err(Error::SnapshotMissing(name.to_owned())); }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.time(now)?;
        let mut removed = 0;
        for versions in self.records.values_mut() {
            let mut keep = BTreeSet::new();
            if !versions.is_empty() { keep.insert(versions.len() - 1); }
            for tx in self.txs.values() {
                if let Some(i) = versions.iter().rposition(|r| r.rev <= tx.at) { keep.insert(i); }
            }
            let old = std::mem::take(versions);
            for (i, record) in old.into_iter().enumerate() {
                if keep.contains(&i) { versions.push(record); } else { removed += 1; }
            }
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.time(now)?;
        let versions = self.records.values().map(Vec::len).sum();
        let tombstones = self.records.values().flatten().filter(|r| r.payload.is_none()).count();
        let live_keys = self.records.values().filter(|vs| {
            vs.last().and_then(|r| r.payload.as_ref()).is_some_and(|(_, d)| d.map_or(true, |x| x > now))
        }).count();
        Ok(Stats {
            clock: self.time,
            revision: self.rev,
            keys: self.records.len(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.txs.len(),
            read_only_transactions: self.txs.values().filter(|t| t.ro).count(),
            staged_writes: self.txs.values().map(|t| t.pending.len()).sum(),
            snapshots: self.saved.len(),
        })
    }
}

fn parse_number(s: &str) -> Option<i64> {
    if s.is_empty() || s == "-0" { return None; }
    let digits = s.strip_prefix('-').unwrap_or(s);
    if digits.is_empty() || !digits.bytes().all(|b| b.is_ascii_digit()) ||
        (digits.len() > 1 && digits.starts_with('0')) {
        return None;
    }
    s.parse().ok()
}
