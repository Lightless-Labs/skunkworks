    overhead: u64,
    opened: bool,

        return Err(Error::NoServers);

    if servers.len() > 32 {
        return Err(Error::TooManyServers { count: servers.len() });
    }

        if s.id == 0 {
            return Err(Error::ZeroServerId { index: i });

            return Err(Error::DuplicateServerId { index: i, id: s.id });

            return Err(Error::ZeroCustomerId { index: i });

            return Err(Error::DuplicateCustomerId { index: i, id: c.id });
        }
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| a.arrival.cmp(&b.arrival).then(b.id.cmp(&a.id)));

    let mut states: Vec<State> = servers
        .iter()
        .map(|s| State {
            id: s.id,
            opens: s.opens,
            overhead: s.overhead,
            opened: false,
            idle_since: s.opens,
            running: None,
        })
        .collect();

    let mut waiting: Vec<&Customer> = Vec::new();

            None => break,

            .iter()
            .enumerate()
            .filter_map(|(i, s)| s.running.filter(|r| r.0 == time).map(|r| (s.id, i, r.1)))
            .collect();

        for (_, i, customer) in done {
            events.push(Event::Finish {
                time,
                customer,
                server: states[i].id,
            });
            states[i].running = None;
            states[i].idle_since = time;
        }

            .iter()
            .enumerate()

            .collect();

        for i in opens {
            states[i].opened = true;
            events.push(Event::Open { time, server: states[i].id });

            ai += 1;
        }

            events.push(Event::Arrive { time, customer: c.id });
            waiting.push(*c);
        }

        while !waiting.is_empty() {
            let server_index = states
                .iter()
                .enumerate()
                .filter(|(_, s)| s.opened && s.running.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since.cmp(&b.idle_since)
                        .then(a.overhead.cmp(&b.overhead))
                        .then(b.id.cmp(&a.id))
                })
                .map(|(i, _)| i);
            let Some(si) = server_index else { break };
            let c = waiting.remove(0);
            let server_id = states[si].id;
            let overhead = states[si].overhead;
            let finish = c.arrival
                .checked_sub(c.arrival)
                .and_then(|_| time.checked_add(c.service))
                .and_then(|x| x.checked_add(overhead))

                    customer: c.id,
                    server: server_id,
                    time,
                })?;

                time,
                customer: c.id,
                server: server_id,
                finish,
            });
            if finish == time {
                events.push(Event::Finish { time, customer: c.id, server: server_id });
                states[si].idle_since = time;
            } else {
                states[si].running = Some((finish, c.id));
            }
        }
    }

    Ok(events)
}

                states[server_index].customer = Some(customer.id);
            }
        }

        if next_customer == ordered_customers.len()
            && waiting.is_empty()
            && states.iter().all(|state| !state.occupied)
            && states.iter().all(|state| state.opened)
        {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now && next_time.map_or(true, |current| time < current) {
                next_time = Some(time);
            }
        };

        if next_customer < ordered_customers.len() {
            consider(ordered_customers[next_customer].arrival);
        }
        for state in &states {
            if !state.opened {
                consider(state.server.opens);
            }
            if let Some(finish) = state.finish_at {
                consider(finish);
            }
        }

        match next_time {
            Some(time) => now = time,
            None => break,
        }
    }

    Ok(events)
}

pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start { time: u64, customer: u32, server: u32, finish: u64 },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    idle_since: u64,
    busy_until: Option<u64>,
    busy_customer: u32,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers { count: servers.len() });
    }

    for i in 0..servers.len() {
        if servers[i].id == 0 {
            return Err(Error::ZeroServerId { index: i });
        }
        if servers[..i].iter().any(|s| s.id == servers[i].id) {
            return Err(Error::DuplicateServerId { index: i, id: servers[i].id });
        }
    }

    for i in 0..customers.len() {
        if customers[i].id == 0 {
            return Err(Error::ZeroCustomerId { index: i });
        }
        if customers[..i].iter().any(|c| c.id == customers[i].id) {
            return Err(Error::DuplicateCustomerId { index: i, id: customers[i].id });
        }
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| a.arrival.cmp(&b.arrival).then_with(|| b.id.cmp(&a.id)));

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|s| ServerState {
            id: s.id,
            opens: s.opens,
            overhead: s.overhead,
            opened: false,
            idle_since: s.opens,
            busy_until: None,
            busy_customer: 0,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<Customer> = Vec::new();
    let mut arrival_index = 0usize;
    let mut now = 0u64;

    loop {
        let mut had_event_at_time = false;

        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (s.busy_until == Some(now)).then_some(i))
            .collect();
        finishing.sort_by_key(|&i| states[i].id);
        for i in finishing {
            let customer = states[i].busy_customer;
            events.push(Event::Finish { time: now, customer, server: states[i].id });
            states[i].busy_until = None;
            states[i].idle_since = now;
            had_event_at_time = true;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (!s.opened && s.opens == now).then_some(i))
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].id));
        for i in opening {
            events.push(Event::Open { time: now, server: states[i].id });
            states[i].opened = true;
            had_event_at_time = true;
        }

        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == now {
            let c = arrivals[arrival_index].clone();
            events.push(Event::Arrive { time: now, customer: c.id });
            waiting.push(c);
            arrival_index += 1;
            had_event_at_time = true;
        }

        while !waiting.is_empty() {
            let best = states
                .iter()
                .enumerate()
                .filter(|(_, s)| s.opened && s.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(i, _)| i);

            let Some(si) = best else { break };
            let c = waiting.remove(0);
            let s = &states[si];
            let first = now.checked_add(c.service).ok_or(Error::TimeOverflow {
                customer: c.id,
                server: s.id,
                time: now,
            })?;
            let finish = first.checked_add(s.overhead).ok_or(Error::TimeOverflow {
                customer: c.id,
                server: s.id,
                time: now,
            })?;

            events.push(Event::Start { time: now, customer: c.id, server: s.id, finish });
            had_event_at_time = true;

            if finish == now {
                events.push(Event::Finish { time: now, customer: c.id, server: s.id });
                states[si].idle_since = now;
            } else {
                states[si].busy_until = Some(finish);
                states[si].busy_customer = c.id;
            }
        }

        if arrival_index == arrivals.len()
            && waiting.is_empty()
            && states.iter().all(|s| s.opened && s.busy_until.is_none())
        {
            break;
        }

        let next_arrival = arrivals.get(arrival_index).map(|c| c.arrival);
        let next_open = states.iter().filter(|s| !s.opened).map(|s| s.opens).min();
        let next_finish = states.iter().filter_map(|s| s.busy_until).min();
        let next = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&t| t > now || !had_event_at_time)
            .min();

        match next {
            Some(t) => now = t,
            None => break,
        }
    }

    Ok(events)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    open: bool,
    busy_until: Option<u64>,
    active_customer: Option<u32>,
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    let mut seen_servers = Vec::new();
    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if seen_servers.contains(&server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
        seen_servers.push(server.id);
    }

    let mut seen_customers = Vec::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if seen_customers.contains(&customer.id) {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
        seen_customers.push(customer.id);
    }

    let mut queue = customers.to_vec();
    queue.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            open: false,
            busy_until: None,
            active_customer: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting = std::collections::VecDeque::<usize>::new();
    let mut next_arrival = 0usize;
    let mut now = 0u64;

    loop {
        let mut completions: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                if state.busy_until == Some(now) {
                    Some(i)
                } else {
                    None
                }
            })
            .collect();
        completions.sort_by_key(|&i| states[i].server.id);

        for i in completions {
            let customer = states[i].active_customer.take().unwrap();
            events.push(Event::Finish {
                time: now,
                customer,
                server: states[i].server.id,
            });
            states[i].busy_until = None;
            states[i].idle_since = now;
        }

        let mut openings: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                if !state.open && state.server.opens == now {
                    Some(i)
                } else {
                    None
                }
            })
            .collect();
        openings.sort_by_key(|&i| std::cmp::Reverse(states[i].server.id));

        for i in openings {
            states[i].open = true;
            events.push(Event::Open {
                time: now,
                server: states[i].server.id,
            });
        }

        let cohort_begin = next_arrival;
        while next_arrival < queue.len() && queue[next_arrival].arrival == now {
            next_arrival += 1;
        }
        for index in cohort_begin..next_arrival {
            events.push(Event::Arrive {
                time: now,
                customer: queue[index].id,
            });
            waiting.push_back(index);
        }

        loop {
            let Some(&customer_index) = waiting.front() else {
                break;
            };
            let server_index = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.open && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = server_index else {
                break;
            };

            waiting.pop_front();
            let customer = &queue[customer_index];
            let server = &states[server_index].server;
            let finish = now
                .checked_add(customer.service)
                .and_then(|time| time.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                })?;

            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = now;
            } else {
                states[server_index].busy_until = Some(finish);
                states[server_index].active_customer = Some(customer.id);
            }
        }

        if next_arrival == queue.len()
            && waiting.is_empty()
            && states.iter().all(|state| state.busy_until.is_none())
            && states.iter().all(|state| state.open)
        {
            break;
        }

        let mut next = None;
        let mut add_candidate = |candidate: u64| {
            if candidate > now && next.map_or(true, |current| candidate < current) {
                next = Some(candidate);
            }
        };

        if next_arrival < queue.len() {
            add_candidate(queue[next_arrival].arrival);
        }
        for state in &states {
            if !state.open {
                add_candidate(state.server.opens);
            }
            if let Some(time) = state.busy_until {
                add_candidate(time);
            }
        }

        if let Some(time) = next {
            now = time;
        } else {
            break;
        }
    }

    Ok(events)
}
