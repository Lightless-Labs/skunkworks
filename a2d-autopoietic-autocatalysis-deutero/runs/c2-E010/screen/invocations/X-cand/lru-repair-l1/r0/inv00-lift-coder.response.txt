- Validation order: time reversal precedes all insert checks; weight checks precede TTL overflow.
- Expiry boundary: entries expire when `expires_at <= now`.
- Purge order: expired keys are returned in ascending numeric order.
- `Some(0)` still runs every insert validation, then purges and removes a surviving same-key entry without storing or evicting.
- Eviction uses subtraction-based capacity checks to avoid `u64` overflow; exact-capacity entries are allowed.
- These requirements specify a single corrected behavior, so Solution B uses an independent recency representation with the same semantics.

### Solution A

```rust
use std::collections::{BTreeMap, VecDeque};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BackwardTime { last: u64, now: u64 },
    ZeroWeight,
    TooHeavy { weight: u64, capacity: u64 },
    DeadlineOverflow,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Entry {
    pub key: u64,
    pub value: i64,
    pub weight: u64,
    pub expires_at: Option<u64>,
}

#[derive(Debug, Clone)]
pub struct Cache {
    capacity: u64,
    used: u64,
    clock: u64,
    entries: BTreeMap<u64, Entry>,
    order: VecDeque<u64>,
}

impl Cache {
    pub fn new(capacity: u64) -> Self {
        Self {
            capacity,
            used: 0,
            clock: 0,
            entries: BTreeMap::new(),
            order: VecDeque::new(),
        }
    }

    pub fn capacity(&self) -> u64 {
        self.capacity
    }

    pub fn used(&self) -> u64 {
        self.used
    }

    pub fn clock(&self) -> u64 {
        self.clock
    }

    pub fn snapshot(&self) -> Vec<Entry> {
        self.order
            .iter()
            .filter_map(|key| self.entries.get(key).cloned())
            .collect()
    }

    fn validate_time(&self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            Err(Error::BackwardTime {
                last: self.clock,
                now,
            })
        } else {
            Ok(())
        }
    }

    fn erase(&mut self, key: u64) -> Option<Entry> {
        let entry = self.entries.remove(&key)?;
        self.used -= entry.weight;
        self.order.retain(|&k| k != key);
        Some(entry)
    }

    fn touch(&mut self, key: u64) {
        self.order.retain(|&k| k != key);
        self.order.push_back(key);
    }

    fn expire(&mut self, now: u64) -> Vec<u64> {
        let expired: Vec<u64> = self
            .entries
            .iter()
            .filter_map(|(&key, entry)| {
                entry
                    .expires_at
                    .is_some_and(|deadline| deadline <= now)
                    .then_some(key)
            })
            .collect();
        for &key in &expired {
            self.erase(key);
        }
        self.clock = now;
        expired
    }

    pub fn advance(&mut self, now: u64) -> Result<Vec<u64>, Error> {
        self.validate_time(now)?;
        Ok(self.expire(now))
    }

    pub fn get(&mut self, key: u64, now: u64) -> Result<Option<i64>, Error> {
        self.validate_time(now)?;
        self.expire(now);
        let value = self.entries.get(&key).map(|entry| entry.value);
        if value.is_some() {
            self.touch(key);
        }
        Ok(value)
    }

    pub fn remove(&mut self, key: u64, now: u64) -> Result<Option<i64>, Error> {
        self.validate_time(now)?;
        self.expire(now);
        Ok(self.erase(key).map(|entry| entry.value))
    }

    pub fn insert(
        &mut self,
        key: u64,
        value: i64,
        weight: u64,
        ttl: Option<u64>,
        now: u64,
    ) -> Result<Vec<u64>, Error> {
        self.validate_time(now)?;
        if weight == 0 {
            return Err(Error::ZeroWeight);
        }
        if weight > self.capacity {
            return Err(Error::TooHeavy {
                weight,
                capacity: self.capacity,
            });
        }
        let expires_at = match ttl {
            Some(duration) => Some(now.checked_add(duration).ok_or(Error::DeadlineOverflow)?),
            None => None,
        };

        let mut removed = self.expire(now);

        if ttl == Some(0) {
            self.erase(key);
            return Ok(removed);
        }

        self.erase(key);
        while self.used > self.capacity - weight {
            let victim = self.order.front().copied().expect("used weight has a resident entry");
            self.erase(victim);
            removed.push(victim);
        }

        self.used += weight;
        self.entries.insert(
            key,
            Entry {
                key,
                value,
                weight,
                expires_at,
            },
        );
        self.touch(key);
        Ok(removed)
    }

    pub fn resize(&mut self, capacity: u64, now: u64) -> Result<Vec<u64>, Error> {
        self.validate_time(now)?;
        let mut removed = self.expire(now);
        self.capacity = capacity;

        while self.used > capacity {
            let key = self.order.front().copied().expect("used weight has a resident entry");
            self.erase(key);
            removed.push(key);
        }
        Ok(removed)
    }
}
```

### Solution B

```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BackwardTime { last: u64, now: u64 },
    ZeroWeight,
    TooHeavy { weight: u64, capacity: u64 },
    DeadlineOverflow,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Entry {
    pub key: u64,
    pub value: i64,
    pub weight: u64,
    pub expires_at: Option<u64>,
}

#[derive(Debug, Clone)]
pub struct Cache {
    capacity: u64,
    used: u64,
    clock: u64,
    entries: BTreeMap<u64, Entry>,
    // Monotonically increasing sequence numbers encode call-order recency.
    recency: BTreeMap<u64, u64>,
    by_recency: BTreeSet<(u64, u64)>,
    next_sequence: u64,
}

impl Cache {
    pub fn new(capacity: u64) -> Self {
        Self {
            capacity,
            used: 0,
            clock: 0,
            entries: BTreeMap::new(),
            recency: BTreeMap::new(),
            by_recency: BTreeSet::new(),
            next_sequence: 0,
        }
    }

    pub fn capacity(&self) -> u64 {
        self.capacity
    }

    pub fn used(&self) -> u64 {
        self.used
    }

    pub fn clock(&self) -> u64 {
        self.clock
    }

    pub fn snapshot(&self) -> Vec<Entry> {
        self.by_recency
            .iter()
            .filter_map(|&(_, key)| self.entries.get(&key).cloned())
            .collect()
    }

    fn validate_time(&self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            Err(Error::BackwardTime {
                last: self.clock,
                now,
            })
        } else {
            Ok(())
        }
    }

    fn touch(&mut self, key: u64) {
        if let Some(old) = self.recency.remove(&key) {
            self.by_recency.remove(&(old, key));
        }
        let sequence = self.next_sequence;
        self.next_sequence = self.next_sequence.wrapping_add(1);
        self.recency.insert(key, sequence);
        self.by_recency.insert((sequence, key));
    }

    fn erase(&mut self, key: u64) -> Option<Entry> {
        let entry = self.entries.remove(&key)?;
        self.used -= entry.weight;
        if let Some(sequence) = self.recency.remove(&key) {
            self.by_recency.remove(&(sequence, key));
        }
        Some(entry)
    }

    fn expire(&mut self, now: u64) -> Vec<u64> {
        let keys: Vec<u64> = self
            .entries
            .iter()
            .filter_map(|(&key, entry)| {
                entry
                    .expires_at
                    .is_some_and(|deadline| deadline <= now)
                    .then_some(key)
            })
            .collect();
        for &key in &keys {
            self.erase(key);
        }
        self.clock = now;
        keys
    }

    pub fn advance(&mut self, now: u64) -> Result<Vec<u64>, Error> {
        self.validate_time(now)?;
        Ok(self.expire(now))
    }

    pub fn get(&mut self, key: u64, now: u64) -> Result<Option<i64>, Error> {
        self.validate_time(now)?;
        self.expire(now);
        let value = self.entries.get(&key).map(|entry| entry.value);
        if value.is_some() {
            self.touch(key);
        }
        Ok(value)
    }

    pub fn remove(&mut self, key: u64, now: u64) -> Result<Option<i64>, Error> {
        self.validate_time(now)?;
        self.expire(now);
        Ok(self.erase(key).map(|entry| entry.value))
    }

    pub fn insert(
        &mut self,
        key: u64,
        value: i64,
        weight: u64,
        ttl: Option<u64>,
        now: u64,
    ) -> Result<Vec<u64>, Error> {
        self.validate_time(now)?;
        if weight == 0 {
            return Err(Error::ZeroWeight);
        }
        if weight > self.capacity {
            return Err(Error::TooHeavy {
                weight,
                capacity: self.capacity,
            });
        }
        let expires_at = match ttl {
            Some(duration) => Some(now.checked_add(duration).ok_or(Error::DeadlineOverflow)?),
            None => None,
        };

        let mut removed = self.expire(now);
        if ttl == Some(0) {
            self.erase(key);
            return Ok(removed);
        }

        self.erase(key);
        while self.used > self.capacity - weight {
            let (_, victim) = self
                .by_recency
                .iter()
                .next()
                .copied()
                .expect("used weight has a resident entry");
            self.erase(victim);
            removed.push(victim);
        }

        self.used += weight;
        self.entries.insert(
            key,
            Entry {
                key,
                value,
                weight,
                expires_at,
            },
        );
        self.touch(key);
        Ok(removed)
    }

    pub fn resize(&mut self, capacity: u64, now: u64) -> Result<Vec<u64>, Error> {
        self.validate_time(now)?;
        let mut removed = self.expire(now);
        self.capacity = capacity;
        while self.used > capacity {
            let (_, key) = self
                .by_recency
                .iter()
                .next()
                .copied()
                .expect("used weight has a resident entry");
            self.erase(key);
            removed.push(key);
        }
        Ok(removed)
    }
}
```
