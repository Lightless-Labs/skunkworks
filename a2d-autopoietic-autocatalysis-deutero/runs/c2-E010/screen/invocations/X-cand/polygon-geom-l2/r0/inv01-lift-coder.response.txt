- Validation order: vertex cap, coordinate scan, one closure removal, then minimum effective length.
- Query cap and coordinate checks happen only after successful polygon preparation; all queries are checked before classification.
- Boundary includes zero-length edges and overlapping segments; the lowest effective edge index owns a boundary point.
- Square roots are rounded per edge to nearest milli-unit, with exact half ties upward; perimeter sums those rounded values.
- “Longest” compares rounded lengths first, then lexicographically sorted endpoint pairs, then edge index.
- Coordinates are validated before arithmetic; `i128`/`u128` intermediates keep all accepted geometry exact.

### Solution A

```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

const LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;
const SCALE2: u128 = 1_000_000;

fn in_range(p: Point) -> bool {
    (-LIMIT..=LIMIT).contains(&p.x) && (-LIMIT..=LIMIT).contains(&p.y)
}

fn prepare(points: &[Point]) -> Result<&[Point], GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = points.len()
        - usize::from(points.len() >= 2 && points[0] == points[points.len() - 1]);
    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }
    Ok(&points[..effective_len])
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = (b.x as i128 - a.x as i128) as u128;
    let dy = (b.y as i128 - a.y as i128) as u128;
    dx * dx + dy * dy
}

fn rounded_sqrt(value: u128) -> u128 {
    // Find floor(sqrt(value)) without overflowing a square.
    let mut lo = 0u128;
    let mut hi = value.min(u64::MAX as u128) + 1;
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid <= value / mid {
            lo = mid;
        } else {
            hi = mid;
        }
    }
    let remainder = value - lo * lo;
    // value is an integer. It is at least halfway to (lo+1)^2 exactly
    // when remainder >= lo + 1.
    if remainder >= lo + 1 { lo + 1 } else { lo }
}

fn edge_milli(a: Point, b: Point) -> u64 {
    rounded_sqrt(squared_distance(a, b) * SCALE2) as u64
}

fn cross(a: Point, b: Point, q: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let aqx = q.x as i128 - a.x as i128;
    let aqy = q.y as i128 - a.y as i128;
    abx * aqy - aby * aqx
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepare(points)?;
    let n = polygon.len();

    let mut signed_area2 = 0i128;
    let mut edge_milli_values = Vec::with_capacity(n);
    let mut perimeter_milli = 0u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..n {
        let a = polygon[i];
        let b = polygon[(i + 1) % n];

        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;

        let length = edge_milli(a, b);
        edge_milli_values.push(length);
        perimeter_milli += length;

        let replace = match longest_edge {
            None => true,
            Some(current) => {
                if length != edge_milli_values[current] {
                    length > edge_milli_values[current]
                } else {
                    let old_a = polygon[current];
                    let old_b = polygon[(current + 1) % n];
                    let old_pair = if old_a <= old_b { (old_a, old_b) } else { (old_b, old_a) };
                    let new_pair = if a <= b { (a, b) } else { (b, a) };
                    (new_pair, i) < (old_pair, current)
                }
            }
        };
        if replace {
            longest_edge = Some(i);
        }
    }

    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli: edge_milli_values,
        perimeter_milli,
        longest_edge,
    })
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

fn classify_one(polygon: &[Point], q: Point) -> Location {
    let n = polygon.len();

    for i in 0..n {
        if on_segment(polygon[i], polygon[(i + 1) % n], q) {
            return Location::Boundary { edge: i };
        }
    }

    let mut inside = false;
    for i in 0..n {
        let a = polygon[i];
        let b = polygon[(i + 1) % n];
        if (a.y > q.y) != (b.y > q.y) {
            let c = cross(a, b, q);
            if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                inside = !inside;
            }
        }
    }

    if inside { Location::Inside } else { Location::Outside }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare(points)?;

    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    Ok(queries.iter().map(|&q| classify_one(polygon, q)).collect())
}
```

### Solution B

```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

const LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn valid(p: Point) -> bool {
    p.x >= -LIMIT && p.x <= LIMIT && p.y >= -LIMIT && p.y <= LIMIT
}

fn prepare(points: &[Point]) -> Result<&[Point], GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (i, &p) in points.iter().enumerate() {
        if !valid(p) {
            return Err(GeomError::OutOfRange { index: i });
        }
    }
    let n = points.len() - usize::from(points.len() >= 2 && points[0] == points[points.len() - 1]);
    if n < 3 {
        return Err(GeomError::TooFewVertices { len: n });
    }
    Ok(&points[..n])
}

fn cross(a: Point, b: Point, q: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (q.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (q.x as i128 - a.x as i128)
}

fn length_milli(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let radicand = (dx * dx + dy * dy) as u128 * 1_000_000;
    let mut low = 0u128;
    let mut high = 3_000_000_000u128;
    while low < high {
        let mid = low + (high - low) / 2;
        if mid * mid <= radicand {
            low = mid + 1;
        } else {
            high = mid;
        }
    }
    let floor = low - 1;
    let rem = radicand - floor * floor;
    if rem >= floor + 1 { (floor + 1) as u64 } else { floor as u64 }
}

fn pair(a: Point, b: Point) -> (Point, Point) {
    if a <= b { (a, b) } else { (b, a) }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let p = prepare(points)?;
    let n = p.len();
    let mut signed_area2 = 0i128;
    let mut edge_milli = Vec::with_capacity(n);
    for i in 0..n {
        let a = p[i];
        let b = p[(i + 1) % n];
        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
        edge_milli.push(length_milli(a, b));
    }

    let perimeter_milli = edge_milli.iter().copied().sum();
    let longest_edge = (0..n).max_by(|&i, &j| {
        edge_milli[i]
            .cmp(&edge_milli[j])
            .then_with(|| pair(p[j], p[(j + 1) % n]).cmp(&pair(p[i], p[(i + 1) % n])))
            .then_with(|| j.cmp(&i))
    });

    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

fn classify_one(p: &[Point], q: Point) -> Location {
    let n = p.len();
    let mut owner = None;
    for i in 0..n {
        if on_segment(p[i], p[(i + 1) % n], q) {
            owner = Some(i);
            break;
        }
    }
    if let Some(edge) = owner {
        return Location::Boundary { edge };
    }

    let mut parity = false;
    for i in 0..n {
        let a = p[i];
        let b = p[(i + 1) % n];
        if (a.y > q.y) != (b.y > q.y) {
            let side = cross(a, b, q);
            if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                parity = !parity;
            }
        }
    }
    if parity { Location::Inside } else { Location::Outside }
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (i, &q) in queries.iter().enumerate() {
        if !valid(q) {
            return Err(GeomError::QueryOutOfRange { index: i });
        }
    }
    Ok(queries.iter().map(|&q| classify_one(polygon, q)).collect())
}
```
