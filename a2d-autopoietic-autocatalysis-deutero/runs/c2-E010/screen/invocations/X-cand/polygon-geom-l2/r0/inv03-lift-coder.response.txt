- Polygon length is checked before coordinates; coordinates are scanned before closure removal and minimum length.
- Exactly one repeated closing vertex is removed; other repeated vertices and zero-length edges remain.
- Query count and coordinate checks happen only after polygon preparation, and all queries are checked before classification.
- Boundary points take precedence; if multiple edges contain the point, the smallest effective edge index owns it.
- Lengths use integer square root with half ties rounded upward; longest edge ties use sorted endpoints, then index.
- All arithmetic uses sufficiently wide integers, including for extreme `i64` inputs passed to classification.

### Solution A

```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]

    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

const COORD_LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare(points: &[Point]) -> Result<&[Point], GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }
    let effective = if points.len() >= 2 && points[points.len() - 1] == points[0] {
        &points[..points.len() - 1]
    } else {
        points
    };
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices { len: effective.len() });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, q: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let aqx = q.x as i128 - a.x as i128;
    let aqy = q.y as i128 - a.y as i128;
    abx * aqy - aby * aqx
}

fn is_on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && (q.x as i128) >= (a.x.min(b.x) as i128)
        && (q.x as i128) <= (a.x.max(b.x) as i128)
        && (q.y as i128) >= (a.y.min(b.y) as i128)
        && (q.y as i128) <= (a.y.max(b.y) as i128)
}

fn isqrt(n: u128) -> u128 {
    if n < 2 {
        return n;
    }
    let mut x = n;
    let mut y = (x + 1) / 2;
    while y < x {
        x = y;
        y = (x + n / x) / 2;
    }
    x
}

fn rounded_length_milli(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let squared = (dx * dx + dy * dy) as u128;
    let scaled = squared * 1_000_000;
    let floor = isqrt(scaled);
    let remainder = scaled - floor * floor;
    let rounded = if remainder >= floor + 1 { floor + 1 } else { floor };
    rounded as u64
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let points = prepare(points)?;
    let n = points.len();

    let mut signed_area2 = 0i128;
    let mut edge_milli = Vec::with_capacity(n);
    let mut perimeter_milli = 0u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;

        let length = rounded_length_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let replace = match longest_edge {
            None => true,
            Some(old) => {
                let old_length = edge_milli[old];
                if length > old_length {
                    true
                } else if length < old_length {
                    false
                } else {
                    let pair = (a.min(b), a.max(b));
                    let old_a = points[old];
                    let old_b = points[(old + 1) % n];
                    pair < (old_a.min(old_b), old_a.max(old_b))
                        || (pair == (old_a.min(old_b), old_a.max(old_b)) && i < old)
                }
            }
        };
        if replace {
            longest_edge = Some(i);
        }
    }

    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare(points)?;

    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if is_on_segment(a, b, q) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            results.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if (a.y > q.y) != (b.y > q.y) {
                let c = cross(a, b, q);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                    inside = !inside;
                }
            }
        }
        results.push(if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(results)
}
```

### Solution B

```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

const LIMIT: i64 = 1_000_000_000;
const CAP: usize = 4096;

fn valid(p: Point) -> bool {
    p.x >= -LIMIT && p.x <= LIMIT && p.y >= -LIMIT && p.y <= LIMIT
}

fn prepared(points: &[Point]) -> Result<&[Point], GeomError> {
    if points.len() > CAP {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    if let Some(index) = points.iter().position(|&p| !valid(p)) {
        return Err(GeomError::OutOfRange { index });
    }
    let len = points.len()
        - usize::from(points.len() >= 2 && points[0] == points[points.len() - 1]);
    if len < 3 {
        return Err(GeomError::TooFewVertices { len });
    }
    Ok(&points[..len])
}

fn cross(a: Point, b: Point, q: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (q.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (q.x as i128 - a.x as i128)
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && a.x.min(b.x) <= q.x
        && q.x <= a.x.max(b.x)
        && a.y.min(b.y) <= q.y
        && q.y <= a.y.max(b.y)
}

fn integer_sqrt(n: u128) -> u128 {
    if n == 0 {
        return 0;
    }
    let mut lo = 1u128;
    let mut hi = 1u128 << 64;
    while lo < hi {
        let mid = lo + (hi - lo) / 2;
        if mid <= n / mid {
            lo = mid + 1;
        } else {
            hi = mid;
        }
    }
    lo - 1
}

fn edge_length(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let squared = (dx * dx + dy * dy) as u128;
    let scaled = squared * 1_000_000;
    let root = integer_sqrt(scaled);
    let next_square = (root + 1) * (root + 1);
    if scaled - root * root >= next_square - scaled {
        (root + 1) as u64
    } else {
        root as u64
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepared(points)?;
    let n = polygon.len();
    let mut signed_area2 = 0i128;
    let mut edge_milli = Vec::with_capacity(n);

    for i in 0..n {
        let a = polygon[i];
        let b = polygon[(i + 1) % n];
        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
        edge_milli.push(edge_length(a, b));
    }

    let perimeter_milli: u64 = edge_milli.iter().copied().sum();
    let mut longest_edge = None;
    for i in 0..n {
        let should_select = match longest_edge {
            None => true,
            Some(j) => {
                if edge_milli[i] != edge_milli[j] {
                    edge_milli[i] > edge_milli[j]
                } else {
                    let pair_i = (
                        polygon[i].min(polygon[(i + 1) % n]),
                        polygon[i].max(polygon[(i + 1) % n]),
                    );
                    let pair_j = (
                        polygon[j].min(polygon[(j + 1) % n]),
                        polygon[j].max(polygon[(j + 1) % n]),
                    );
                    pair_i < pair_j || (pair_i == pair_j && i < j)
                }
            }
        };
        if should_select {
            longest_edge = Some(i);
        }
    }

    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepared(points)?;

    if queries.len() > CAP {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    if let Some(index) = queries.iter().position(|&q| !valid(q)) {
        return Err(GeomError::QueryOutOfRange { index });
    }

    let mut answer = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut result = None;
        for i in 0..polygon.len() {
            if on_segment(polygon[i], polygon[(i + 1) % polygon.len()], q) {
                result = Some(Location::Boundary { edge: i });
                break;
            }
        }

        if let Some(location) = result {
            answer.push(location);
            continue;
        }

        let mut parity = false;
        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if (a.y > q.y) != (b.y > q.y) {
                let orientation = cross(a, b, q);
                if (a.y < b.y && orientation > 0) || (a.y > b.y && orientation < 0) {
                    parity = !parity;
                }
            }
        }
        answer.push(if parity {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(answer)
}
```
