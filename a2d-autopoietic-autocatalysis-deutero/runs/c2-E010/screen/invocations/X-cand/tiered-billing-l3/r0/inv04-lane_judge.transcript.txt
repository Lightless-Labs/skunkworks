fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous {
                    return Err(Error::Bound { index });
                }
                previous = bound;
            }
            None => {
                if index + 1 != tiers.len() {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers.last().and_then(|tier| tier.up_to).is_some() {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn graduated(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                return Ok(Charge { amount: 0, bands: Vec::new() });
            }
            let amount = checked_band_amount(quantity, *unit, *start, 0)?;
            Ok(Charge {
                amount,
                bands: vec![Band { index: 0, units: quantity, amount }],
            })
        }
        Price::Tiers(tiers) => {
            validate_tiers(tiers)?;
            if quantity == 0 {
                return Ok(Charge { amount: 0, bands: Vec::new() });
            }

            let mut remaining = quantity;
            let mut previous = 0u64;
            let mut total = 0u64;
            let mut bands = Vec::new();

            for (index, tier) in tiers.iter().enumerate() {
                if remaining == 0 {
                    break;
                }
                let allocated = match tier.up_to {
                    Some(bound) => {
                        let width = bound - previous;
                        previous = bound;
                        remaining.min(width)
                    }
                    None => remaining,
                };
                if allocated > 0 {
                    let amount = checked_band_amount(allocated, tier.unit, tier.start, index)?;
                    total = add_band(total, amount, index)?;
                    bands.push(Band { index, units: allocated, amount });
                    remaining -= allocated;
                }
            }
            Ok(Charge { amount: total, bands })
        }
    }
}

    graduated(quantity, price)
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Tiers(_), Mode::Volume) => {
            if let Price::Tiers(tiers) = price {
                validate_tiers(tiers)?;
                if quantity == 0 {
                    Charge { amount: 0, bands: Vec::new() }
                } else {
                    let index = tiers.iter().position(|tier| {
                        tier.up_to.map_or(true, |bound| bound >= quantity)
                    }).expect("validated tiers have a final unbounded tier");
                    let tier = &tiers[index];
                    let amount = checked_band_amount(quantity, tier.unit, tier.start, index)?;
                    Charge {
                        amount,
                        bands: vec![Band { index, units: quantity, amount }],
                    }
                }
            } else {
                unreachable!()
            }
        }
        (_, Mode::Graduated) | (Price::Flat { .. }, Mode::Volume) => graduated(quantity, price)?,
    };

    let minimum_adjustment = if quantity > 0 && minimum > charge.amount {
        minimum - charge.amount
    } else {
        0
    };
    let total = charge.amount.checked_add(minimum_adjustment)
        .ok_or(Error::Overflow { index: 0 })?;
    Ok(Bill { charge, minimum_adjustment, total })
}

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {
    if !(28..=31).contains(&days) {
        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);
    }

    let mut previous_end = start;
    let mut credited_days = 0u32;
    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < previous_end {
            return Err(Error::CreditOrder { index });
        }
        credited_days += credit.end - credit.start;
        previous_end = credit.end;
    }

    let active_days = (end - start) - credited_days;
    let numerator = (fee as u128) * (active_days as u128);
    let denominator = days as u128;
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;

    let rounded = match rounding {
        Rounding::Down => quotient,
        Rounding::Up => quotient + u128::from(remainder != 0),
        Rounding::HalfEven => {
            let twice = remainder * 2;
            if twice > denominator || (twice == denominator && quotient % 2 == 1) {
                quotient + 1
            } else {
                quotient
            }
        }
    };

    Ok(Proration { active_days, amount: rounded as u64 })
}

fn check_schedule(tiers: &[Tier]) -> Result<(), Error> {
    match tiers.len() {
        0 => return Err(Error::Empty),
        65.. => return Err(Error::TooMany),
        _ => {}
    }

    let mut last_finite = 0u64;
    for (i, tier) in tiers.iter().enumerate() {
        if let Some(bound) = tier.up_to {
            if bound <= last_finite {
                return Err(Error::Bound { index: i });
            }
            last_finite = bound;
        } else if i != tiers.len() - 1 {
            return Err(Error::Unbounded { index: i });
        }
    }

    if tiers[tiers.len() - 1].up_to.is_some() {
        Err(Error::MissingFinal)
    } else {
        Ok(())
    }
}

fn graduated_charge(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    check_schedule(tiers)?;
    let mut total = 0u64;
    let mut bands = Vec::new();
    let mut lower = 0u64;

    for (i, tier) in tiers.iter().enumerate() {
        let upper = tier.up_to.unwrap_or(u64::MAX);
        if quantity > lower {

        }
        lower = upper;

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {
    if !matches!(days, 28 | 29 | 30 | 31) {
        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);
    }

    let mut cursor = start;
    let mut active_days = end - start;
    for (i, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index: i });
        }
        if credit.start < cursor {
            return Err(Error::CreditOrder { index: i });
        }
        active_days -= credit.end - credit.start;
        cursor = credit.end;
    }

    let numerator = u128::from(fee) * u128::from(active_days);
    let divisor = u128::from(days);
    let whole = numerator / divisor;
    let remainder = numerator % divisor;
    let amount = match rounding {
        Rounding::Down => whole,
        Rounding::Up => whole + u128::from(remainder != 0),
        Rounding::HalfEven => {

                std::cmp::Ordering::Less => whole,
                std::cmp::Ordering::Greater => whole + 1,
                std::cmp::Ordering::Equal if whole % 2 == 1 => whole + 1,
                std::cmp::Ordering::Equal => whole,
            }
        }
    };
    Ok(Proration { active_days, amount: amount as u64 })
}

        };

        if units > 0 {
            let amount = checked_band(units, tier.unit, tier.start, index)?;
            total = total
                .checked_add(amount)
                .ok_or(Error::Overflow { index })?;
            bands.push(Band {
                index,
                units,
                amount,
            });
            remaining -= units;
        }
    }

    Ok(Charge {
        amount: total,
        bands,
    })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                })
            } else {
                let amount = checked_band(quantity, *unit, *start, 0)?;
                Ok(Charge {
                    amount,
                    bands: vec![Band {
                        index: 0,
                        units: quantity,
                        amount,
                    }],
                })
            }
        }
        Price::Tiers(tiers) => graduated(quantity, tiers),
    }
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Tiers(tiers), Mode::Volume) => {
            validate_tiers(tiers)?;
            if quantity == 0 {
                Charge {
                    amount: 0,
                    bands: Vec::new(),
                }
            } else {
                let mut previous = 0u64;
                let mut selected = None;
                for (index, tier) in tiers.iter().enumerate() {
                    match tier.up_to {
                        Some(bound) => {
                            if quantity <= bound {
                                selected = Some((index, tier));
                                break;
                            }
                            previous = bound;
                        }
                        None => {
                            selected = Some((index, tier));
                            break;
                        }
                    }
                }
                let (index, tier) = selected.expect("validated tiers have a final unbounded tier");
                let amount = checked_band(quantity, tier.unit, tier.start, index)?;
                Charge {
                    amount,
                    bands: vec![Band {
                        index,
                        units: quantity,
                        amount,
                    }],
                }
            }
        }
        (_, Mode::Graduated) | (Price::Flat { .. }, Mode::Volume) => quote(quantity, price)?,
    };

    let minimum_adjustment = if quantity > 0 {
        minimum.saturating_sub(charge.amount)
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .expect("minimum adjustment cannot make total exceed minimum or u64::MAX");

    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {
    if !matches!(days, 28 | 29 | 30 | 31) {
        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);
    }

    let mut previous_credit_end = start;
    let mut credited_days = 0u32;
    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < previous_credit_end {
            return Err(Error::CreditOrder { index });
        }
        credited_days += credit.end - credit.start;
        previous_credit_end = credit.end;
    }

    let active_days = end - start - credited_days;
    let numerator = (fee as u128) * (active_days as u128);
    let denominator = days as u128;
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;

    let rounded = match rounding {
        Rounding::Down => quotient,
        Rounding::Up => quotient + u128::from(remainder != 0),
        Rounding::HalfEven => {
            let twice_remainder = remainder * 2;
            if twice_remainder > denominator
                || (twice_remainder == denominator && quotient % 2 == 1)
            {
                quotient + 1
            } else {
                quotient
            }
        }
    };

    Ok(Proration {
        active_days,
        amount: rounded as u64,
    })
}

fn flat_charge(quantity: u64, unit: u64, start: u64) -> Result<Charge, Error> {
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }
    let value = u128::from(quantity) * u128::from(unit) + u128::from(start);

            amount,

fn graduated_charge(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;
    let mut used = 0u64;
    let mut prior_limit = 0u64;
    let mut total = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        let units = match tier.up_to {
            Some(limit) => {
                let capacity = limit - prior_limit;
                prior_limit = limit;
                quantity.saturating_sub(used).min(capacity)
            }
            None => quantity - used,
        };

        if units == 0 {
            continue;
        }

        let amount = band_amount(units, tier, index)?;
        total = u64::try_from(u128::from(total) + u128::from(amount))
            .map_err(|_| Error::Overflow { index })?;
        bands.push(Band {
            index,
            units,
            amount,
        });
        used += units;
    }

    Ok(Charge { amount: total, bands })
}

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {
    if days != 28 && days != 29 && days != 30 && days != 31 {
        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);
    }

    let mut prev_end = start;
    let mut active = end - start;
    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < prev_end {
            return Err(Error::CreditOrder { index });
        }
        active -= credit.end - credit.start;
        prev_end = credit.end;
    }

    let product = u128::from(fee) * u128::from(active);
    let denominator = u128::from(days);
    let floor = product / denominator;
    let remainder = product % denominator;

    let amount = match rounding {
        Rounding::Down => floor,
        Rounding::Up if remainder != 0 => floor + 1,
        Rounding::Up => floor,
        Rounding::HalfEven => {
            match (remainder * 2).cmp(&denominator) {
                std::cmp::Ordering::Less => floor,
                std::cmp::Ordering::Greater => floor + 1,
                std::cmp::Ordering::Equal if floor % 2 == 0 => floor,
                std::cmp::Ordering::Equal => floor + 1,
            }
        }
    };

    Ok(Proration {
        active_days: active,
        amount: amount as u64,
    })
}

    for (index, tier) in tiers.iter().enumerate() {
        if remaining == 0 {
            break;

        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous_bound;
                previous_bound = bound;
                remaining.min(width)

            total = total
                .checked_add(band.amount)
                .ok_or(Error::Overflow { index })?;
            bands.push(band);
            remaining -= units;

        amount: total,
        bands,

            return Err(Error::CreditOrder { index });
        }
        credited += credit.end - credit.start;
        previous_end = credit.end;
    }

    let active_days = (end - start) - credited;
    let numerator = u128::from(fee) * u128::from(active_days);
    let denominator = u128::from(days);
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;

    let rounded = match rounding {
        Rounding::Down => quotient,
        Rounding::Up => quotient + u128::from(remainder != 0),
        Rounding::HalfEven => {
            let doubled = remainder * 2;
            if doubled < denominator {
                quotient
            } else if doubled > denominator || quotient % 2 == 1 {
                quotient + 1
            } else {
                quotient
            }
        }
    };

    Ok(Proration { active_days, amount: rounded as u64 })
}

fn validate(tiers: &[Tier]) -> Result<(), Error> {
    match tiers.len() {
        0 => return Err(Error::Empty),
        65.. => return Err(Error::TooMany),
        _ => {}
    }

    let mut prior = 0;
    let final_index = tiers.len() - 1;
    for (index, tier) in tiers.iter().enumerate() {
        if let Some(bound) = tier.up_to {
            if bound <= prior {
                return Err(Error::Bound { index });
            }
            prior = bound;
        } else if index != final_index {
            return Err(Error::Unbounded { index });
        }

        return Err(Error::MissingFinal);

    Ok(())

fn graduated_charge(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate(tiers)?;
    let mut charge = 0u64;
    let mut bands = Vec::new();
    let mut lower = 0u64;

    for (index, tier) in tiers.iter().enumerate() {
        if quantity <= lower {
            break;
        }

        let upper = tier.up_to.unwrap_or(quantity);
        let units = quantity.min(upper) - lower;
        if units != 0 {
            let band = make_band(index, units, tier)?;
            charge = charge
                .checked_add(band.amount)
                .ok_or(Error::Overflow { index })?;
            bands.push(band);
        }

        if tier.up_to.is_none() || quantity <= upper {
            break;
        }
        lower = upper;
    }

    Ok(Charge {
        amount: charge,
        bands,
    })
}

    validate(tiers)?;
    if quantity == 0 {

            amount: 0,
            bands: Vec::new(),
        });
    }

    let index = tiers
        .iter()
        .position(|tier| tier.up_to.is_none_or(|bound| quantity <= bound))
        .expect("the validated last tier is unbounded");
    let band = make_band(index, quantity, &tiers[index])?;
    Ok(Charge {
        amount: band.amount,
        bands: vec![band],
    })
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Flat { unit, start }, _) => flat(quantity, *unit, *start)?,
        (Price::Tiers(tiers), Mode::Graduated) => graduated_charge(quantity, tiers)?,
        (Price::Tiers(tiers), Mode::Volume) => volume_charge(quantity, tiers)?,
    };

    let minimum_adjustment = if quantity == 0 || charge.amount >= minimum {
        0
    } else {
        minimum - charge.amount
    };
    let total = charge.amount + minimum_adjustment;

    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {
    if days != 28 && days != 29 && days != 30 && days != 31 {
        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);
    }

    let mut credit_total = 0u32;
    let mut last_end = start;
    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < last_end {
            return Err(Error::CreditOrder { index });
        }
        credit_total += credit.end - credit.start;
        last_end = credit.end;
    }

    let active_days = end - start - credit_total;
    let product = u128::from(fee) * u128::from(active_days);
    let divisor = u128::from(days);
    let q = product / divisor;
    let r = product % divisor;

    let amount = match rounding {
        Rounding::Down => q,
        Rounding::Up if r > 0 => q + 1,
        Rounding::Up => q,
        Rounding::HalfEven => {
            match (2 * r).cmp(&divisor) {
                std::cmp::Ordering::Less => q,
                std::cmp::Ordering::Greater => q + 1,
                std::cmp::Ordering::Equal if q % 2 == 1 => q + 1,
                std::cmp::Ordering::Equal => q,
            }
        }
    };

    Ok(Proration {
        active_days,
        amount: amount as u64,
    })
}

fn validate_schedule(schedule: &[Tier]) -> Result<(), Error> {
    match schedule.len() {
        0 => return Err(Error::Empty),
        65.. => return Err(Error::TooMany),
        _ => {}
    }

    let mut last_bound = 0;
    for (i, tier) in schedule.iter().enumerate() {
        if let Some(bound) = tier.up_to {
            if bound <= last_bound {
                return Err(Error::Bound { index: i });
            }
            last_bound = bound;
        } else if i != schedule.len() - 1 {
            return Err(Error::Unbounded { index: i });
        }
    }

    if schedule[schedule.len() - 1].up_to.is_some() {
        Err(Error::MissingFinal)
    } else {
        Ok(())
    }
}

fn band_charge(units: u64, tier: &Tier, index: usize) -> Result<u64, Error> {
    units.checked_mul(tier.unit)
        .and_then(|x| x.checked_add(tier.start))
        .ok_or(Error::Overflow { index })
}

fn finish_band(sum: u64, band: u64, index: usize) -> Result<u64, Error> {
    match sum.checked_add(band) {
        Some(v) => Ok(v),
        None => Err(Error::Overflow { index }),
    }
}

fn make_flat(quantity: u64, unit: u64, start: u64) -> Result<Charge, Error> {
    if quantity == 0 {
        return Ok(Charge { amount: 0, bands: vec![] });
    }
    let amount = quantity.checked_mul(unit)
        .and_then(|x| x.checked_add(start))
        .ok_or(Error::Overflow { index: 0 })?;
    Ok(Charge {
        amount,
        bands: vec![Band { index: 0, units: quantity, amount }],
    })
}

fn graduated_charge(quantity: u64, schedule: &[Tier]) -> Result<Charge, Error> {
    validate_schedule(schedule)?;
    if quantity == 0 {
        return Ok(Charge { amount: 0, bands: Vec::new() });
    }

    let mut left = quantity;
    let mut lower = 0;
    let mut sum = 0;
    let mut bands = Vec::new();

    for (i, tier) in schedule.iter().enumerate() {
        if left == 0 {
            break;
        }
        let take = if let Some(upper) = tier.up_to {
            let width = upper - lower;
            lower = upper;
            left.min(width)
        } else {
            left
        };

        if take != 0 {
            let amount = band_charge(take, tier, i)?;
            sum = finish_band(sum, amount, i)?;
            bands.push(Band { index: i, units: take, amount });
            left -= take;
        }
    }

    Ok(Charge { amount: sum, bands })
}

fn price_charge(quantity: u64, price: &Price, mode: Mode) -> Result<Charge, Error> {
    match (price, mode) {
        (Price::Flat { unit, start }, _) => make_flat(quantity, *unit, *start),
        (Price::Tiers(schedule), Mode::Graduated) => graduated_charge(quantity, schedule),
        (Price::Tiers(schedule), Mode::Volume) => {
            validate_schedule(schedule)?;
            if quantity == 0 {
                return Ok(Charge { amount: 0, bands: Vec::new() });
            }
            let selected = schedule.iter().position(|tier| {
                match tier.up_to {
                    Some(upper) => quantity <= upper,
                    None => true,
                }
            }).unwrap();
            let amount = band_charge(quantity, &schedule[selected], selected)?;
            Ok(Charge {
                amount,
                bands: vec![Band { index: selected, units: quantity, amount }],
            })
        }
    }
}

    match price {
        Price::Flat { unit, start } => make_flat(quantity, *unit, *start),
        Price::Tiers(schedule) => graduated_charge(quantity, schedule),
    }
}

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {

        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);

        prior_end = range.end;

    let numerator = u128::from(fee) * u128::from(active_days);
    let denominator = u128::from(days);
    let whole = numerator / denominator;
    let remainder = numerator % denominator;

                || (doubled_remainder == denominator && whole & 1 == 1)

            } else {
                whole
            }
        }
    };

}
