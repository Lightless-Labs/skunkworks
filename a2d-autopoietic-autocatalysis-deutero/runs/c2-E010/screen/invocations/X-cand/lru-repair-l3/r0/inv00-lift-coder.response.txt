- Validation order: insertion checks zero weight, then weight versus capacity, then pinned feasibility.
- Shard routing uses `key % shard_count` in `u64` before converting to `usize`.
- Usage is stored per shard as `u64`; global sums and pinned feasibility use `u128`.
- Eviction selects the oldest eligible entry, and equality with a limit never triggers another eviction.
- Replacement excludes the old same-key entry from pinned feasibility and usage; it is never reported as a victim.
- For Solution B, I use a separate `Vec<Entry>` recency store instead of a key map plus key queue.

### Solution A

```rust
use std::collections::{BTreeMap, VecDeque};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyShards,
    ZeroWeight,
    TooHeavy { weight: u64, capacity: u64 },
    Blocked { shard: usize },
    Missing { key: u64 },
    Pinned { key: u64 },
    BadShard { shard: usize },
    PinnedCapacity { minimum: u128, target: u128 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Entry {
    pub key: u64,
    pub value: i64,
    pub weight: u64,
    pub pinned: bool,
    pub shard: usize,
}

#[derive(Debug, Clone)]
pub struct Cache {
    capacities: Vec<u64>,
    usage: Vec<u64>,
    entries: BTreeMap<u64, Entry>,
    order: VecDeque<u64>,
}

impl Cache {
    pub fn new(capacities: Vec<u64>) -> Result<Self, Error> {
        if capacities.is_empty() {
            return Err(Error::EmptyShards);
        }
        let count = capacities.len();
        Ok(Self {
            capacities,
            usage: vec![0; count],
            entries: BTreeMap::new(),
            order: VecDeque::new(),
        })
    }

    pub fn capacities(&self) -> Vec<u64> {
        self.capacities.clone()
    }

    pub fn usage(&self) -> Vec<u64> {
        self.usage.clone()
    }

    pub fn used(&self) -> u128 {
        self.usage.iter().map(|&n| n as u128).sum()
    }

    pub fn snapshot(&self) -> Vec<Entry> {
        self.order
            .iter()
            .filter_map(|key| self.entries.get(key).cloned())
            .collect()
    }

    fn shard(&self, key: u64) -> usize {
        (key % self.capacities.len() as u64) as usize
    }

    fn touch(&mut self, key: u64) {
        self.order.retain(|&k| k != key);
        self.order.push_back(key);
    }

    fn erase(&mut self, key: u64) -> Option<Entry> {
        let entry = self.entries.remove(&key)?;
        self.usage[entry.shard] -= entry.weight;
        self.order.retain(|&k| k != key);
        Some(entry)
    }

    fn local_victim(&self, shard: usize) -> Option<u64> {
        self.order
            .iter()
            .find(|&&key| {
                self.entries
                    .get(&key)
                    .is_some_and(|entry| entry.shard == shard && !entry.pinned)
            })
            .copied()
    }

    fn global_victim(&self) -> Option<u64> {
        self.order
            .iter()
            .find(|&&key| self.entries.get(&key).is_some_and(|entry| !entry.pinned))
            .copied()
    }

    fn pinned_weight(&self, shard: Option<usize>, exclude: Option<u64>) -> u128 {
        self.entries
            .values()
            .filter(|entry| {
                entry.pinned
                    && shard.is_none_or(|s| entry.shard == s)
                    && Some(entry.key) != exclude
            })
            .map(|entry| entry.weight as u128)
            .sum()
    }

    pub fn get(&mut self, key: u64) -> Option<i64> {
        let value = self.entries.get(&key).map(|entry| entry.value);
        if value.is_some() {
            self.touch(key);
        }
        value
    }

    pub fn set_pin(&mut self, key: u64, pinned: bool) -> Result<(), Error> {
        let entry = self.entries.get_mut(&key).ok_or(Error::Missing { key })?;
        entry.pinned = pinned;
        Ok(())
    }

    pub fn remove(&mut self, key: u64) -> Result<i64, Error> {
        let entry = self.entries.get(&key).ok_or(Error::Missing { key })?;
        if entry.pinned {
            return Err(Error::Pinned { key });
        }
        Ok(self.erase(key).expect("entry checked above").value)
    }

    pub fn insert(
        &mut self,
        key: u64,
        value: i64,
        weight: u64,
        pinned: bool,
    ) -> Result<Vec<u64>, Error> {
        let shard = self.shard(key);
        let capacity = self.capacities[shard];

        if weight == 0 {
            return Err(Error::ZeroWeight);
        }
        if weight > capacity {
            return Err(Error::TooHeavy { weight, capacity });
        }
        if self.pinned_weight(Some(shard), Some(key)) > (capacity - weight) as u128 {
            return Err(Error::Blocked { shard });
        }

        let old_weight = self.entries.get(&key).map_or(0, |entry| entry.weight);
        if self.entries.contains_key(&key) {
            self.erase(key);
        }

        let mut removed = Vec::new();
        while self.usage[shard] > capacity - weight {
            let victim = self.local_victim(shard)
                .expect("pinned feasibility guarantees an unpinned victim");
            self.erase(victim);
            removed.push(victim);
        }

        self.usage[shard] = self.usage[shard]
            .checked_add(weight)
            .expect("validated shard capacity bounds usage");
        let _ = old_weight;
        self.entries.insert(
            key,
            Entry { key, value, weight, pinned, shard },
        );
        self.touch(key);
        Ok(removed)
    }

    pub fn resize(&mut self, shard: usize, capacity: u64) -> Result<Vec<u64>, Error> {
        if shard >= self.capacities.len() {
            return Err(Error::BadShard { shard });
        }
        let minimum = self.pinned_weight(Some(shard), None);
        if minimum > capacity as u128 {
            return Err(Error::PinnedCapacity {
                minimum,
                target: capacity as u128,
            });
        }

        self.capacities[shard] = capacity;
        let mut removed = Vec::new();
        while self.usage[shard] > capacity {
            let victim = self.local_victim(shard)
                .expect("pinned weight is within the requested capacity");
            self.erase(victim);
            removed.push(victim);
        }
        Ok(removed)
    }

    pub fn trim(&mut self, target: u128) -> Result<Vec<u64>, Error> {
        let minimum = self.pinned_weight(None, None);
        if minimum > target {
            return Err(Error::PinnedCapacity { minimum, target });
        }

        let mut removed = Vec::new();
        while self.used() > target {
            let victim = self.global_victim()
                .expect("pinned weight is within the target");
            self.erase(victim);
            removed.push(victim);
        }
        Ok(removed)
    }
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyShards,
    ZeroWeight,
    TooHeavy { weight: u64, capacity: u64 },
    Blocked { shard: usize },
    Missing { key: u64 },
    Pinned { key: u64 },
    BadShard { shard: usize },
    PinnedCapacity { minimum: u128, target: u128 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Entry {
    pub key: u64,
    pub value: i64,
    pub weight: u64,
    pub pinned: bool,
    pub shard: usize,
}

#[derive(Debug, Clone)]
pub struct Cache {
    capacities: Vec<u64>,
    usage: Vec<u64>,
    entries: Vec<Entry>,
}

impl Cache {
    pub fn new(capacities: Vec<u64>) -> Result<Self, Error> {
        if capacities.is_empty() {
            return Err(Error::EmptyShards);
        }
        let usage = vec![0; capacities.len()];
        Ok(Self { capacities, usage, entries: Vec::new() })
    }

    pub fn capacities(&self) -> Vec<u64> {
        self.capacities.clone()
    }

    pub fn usage(&self) -> Vec<u64> {
        self.usage.clone()
    }

    pub fn used(&self) -> u128 {
        self.usage.iter().map(|&weight| weight as u128).sum()
    }

    pub fn snapshot(&self) -> Vec<Entry> {
        self.entries.clone()
    }

    fn shard(&self, key: u64) -> usize {
        (key % self.capacities.len() as u64) as usize
    }

    fn position(&self, key: u64) -> Option<usize> {
        self.entries.iter().position(|entry| entry.key == key)
    }

    fn erase_at(&mut self, index: usize) -> Entry {
        let entry = self.entries.remove(index);
        self.usage[entry.shard] -= entry.weight;
        entry
    }

    fn pinned_weight(&self, shard: Option<usize>, exclude: Option<u64>) -> u128 {
        self.entries
            .iter()
            .filter(|entry| {
                entry.pinned
                    && shard.is_none_or(|s| entry.shard == s)
                    && Some(entry.key) != exclude
            })
            .map(|entry| entry.weight as u128)
            .sum()
    }

    pub fn get(&mut self, key: u64) -> Option<i64> {
        let index = self.position(key)?;
        let value = self.entries[index].value;
        let entry = self.entries.remove(index);
        self.entries.push(entry);
        Some(value)
    }

    pub fn set_pin(&mut self, key: u64, pinned: bool) -> Result<(), Error> {
        let index = self.position(key).ok_or(Error::Missing { key })?;
        self.entries[index].pinned = pinned;
        Ok(())
    }

    pub fn remove(&mut self, key: u64) -> Result<i64, Error> {
        let index = self.position(key).ok_or(Error::Missing { key })?;
        if self.entries[index].pinned {
            return Err(Error::Pinned { key });
        }
        Ok(self.erase_at(index).value)
    }

    pub fn insert(
        &mut self,
        key: u64,
        value: i64,
        weight: u64,
        pinned: bool,
    ) -> Result<Vec<u64>, Error> {
        let shard = self.shard(key);
        let capacity = self.capacities[shard];

        if weight == 0 {
            return Err(Error::ZeroWeight);
        }
        if weight > capacity {
            return Err(Error::TooHeavy { weight, capacity });
        }
        if self.pinned_weight(Some(shard), Some(key)) > (capacity - weight) as u128 {
            return Err(Error::Blocked { shard });
        }

        if let Some(index) = self.position(key) {
            self.erase_at(index);
        }

        let mut removed = Vec::new();
        while self.usage[shard] > capacity - weight {
            let index = self.entries.iter().position(|entry| {
                entry.shard == shard && !entry.pinned
            }).expect("pinned feasibility guarantees an unpinned victim");
            let victim = self.erase_at(index);
            removed.push(victim.key);
        }

        self.usage[shard] = self.usage[shard]
            .checked_add(weight)
            .expect("validated shard capacity bounds usage");
        self.entries.push(Entry { key, value, weight, pinned, shard });
        Ok(removed)
    }

    pub fn resize(&mut self, shard: usize, capacity: u64) -> Result<Vec<u64>, Error> {
        if shard >= self.capacities.len() {
            return Err(Error::BadShard { shard });
        }
        let minimum = self.pinned_weight(Some(shard), None);
        if minimum > capacity as u128 {
            return Err(Error::PinnedCapacity { minimum, target: capacity as u128 });
        }
        self.capacities[shard] = capacity;

        let mut removed = Vec::new();
        while self.usage[shard] > capacity {
            let index = self.entries.iter().position(|entry| {
                entry.shard == shard && !entry.pinned
            }).expect("pinned weight is within requested capacity");
            removed.push(self.erase_at(index).key);
        }
        Ok(removed)
    }

    pub fn trim(&mut self, target: u128) -> Result<Vec<u64>, Error> {
        let minimum = self.pinned_weight(None, None);
        if minimum > target {
            return Err(Error::PinnedCapacity { minimum, target });
        }

        let mut removed = Vec::new();
        while self.used() > target {
            let index = self.entries.iter().position(|entry| !entry.pinned)
                .expect("pinned weight is within target");
            removed.push(self.erase_at(index).key);
        }
        Ok(removed)
    }
}
```
