```rust
use std::collections::{BTreeMap, VecDeque};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyShards,
    ZeroWeight,
    TooHeavy { weight: u64, capacity: u64 },
    Blocked { shard: usize },
    Missing { key: u64 },
    Pinned { key: u64 },
    BadShard { shard: usize },
    PinnedCapacity { minimum: u128, target: u128 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Entry {
    pub key: u64,
    pub value: i64,
    pub weight: u64,
    pub pinned: bool,
    pub shard: usize,
}

#[derive(Debug, Clone)]
pub struct Cache {
    capacities: Vec<u64>,
    usage: Vec<u64>,
    entries: BTreeMap<u64, Entry>,
    order: VecDeque<u64>,
}

impl Cache {
    pub fn new(capacities: Vec<u64>) -> Result<Self, Error> {
        if capacities.is_empty() {
            return Err(Error::EmptyShards);
        }
        let usage = vec![0; capacities.len()];
        Ok(Self {
            capacities,
            usage,
            entries: BTreeMap::new(),
            order: VecDeque::new(),
        })
    }

    pub fn capacities(&self) -> Vec<u64> {
        self.capacities.clone()
    }

    pub fn usage(&self) -> Vec<u64> {
        self.usage.clone()
    }

    pub fn used(&self) -> u128 {
        self.usage.iter().map(|&n| n as u128).sum()
    }

    pub fn snapshot(&self) -> Vec<Entry> {
        self.order
            .iter()
            .map(|key| self.entries[key].clone())
            .collect()
    }

    fn shard(&self, key: u64) -> usize {
        (key % self.capacities.len() as u64) as usize
    }

    fn touch(&mut self, key: u64) {
        self.order.retain(|&k| k != key);
        self.order.push_back(key);
    }

    fn erase(&mut self, key: u64) -> Option<Entry> {
        let entry = self.entries.remove(&key)?;
        self.usage[entry.shard] -= entry.weight;
        self.order.retain(|&k| k != key);
        Some(entry)
    }

    fn local_victim(&self, shard: usize) -> Option<u64> {
        self.order
            .iter()
            .find(|&&key| {
                let entry = &self.entries[&key];
                entry.shard == shard && !entry.pinned
            })
            .copied()
    }

    fn global_victim(&self) -> Option<u64> {
        self.order
            .iter()
            .find(|&&key| !self.entries[&key].pinned)
            .copied()
    }

    fn pinned_weight(&self, shard: Option<usize>, exclude: Option<u64>) -> u128 {
        self.entries
            .values()
            .filter(|entry| {
                entry.pinned
                    && Some(entry.key) != exclude
                    && shard.is_none_or(|s| entry.shard == s)
            })
            .map(|entry| entry.weight as u128)
            .sum()
    }

    pub fn get(&mut self, key: u64) -> Option<i64> {
        let value = self.entries.get(&key).map(|entry| entry.value);
        if value.is_some() {
            self.touch(key);
        }
        value
    }

    pub fn set_pin(&mut self, key: u64, pinned: bool) -> Result<(), Error> {
        let entry = self.entries.get_mut(&key).ok_or(Error::Missing { key })?;
        entry.pinned = pinned;
        Ok(())
    }

    pub fn remove(&mut self, key: u64) -> Result<i64, Error> {
        let entry = self.entries.get(&key).ok_or(Error::Missing { key })?;
        if entry.pinned {
            return Err(Error::Pinned { key });
        }
        Ok(self.erase(key).unwrap().value)
    }

    pub fn insert(
        &mut self,
        key: u64,
        value: i64,
        weight: u64,
        pinned: bool,
    ) -> Result<Vec<u64>, Error> {
        let shard = self.shard(key);
        let capacity = self.capacities[shard];

        if weight == 0 {
            return Err(Error::ZeroWeight);
        }
        if weight > capacity {
            return Err(Error::TooHeavy { weight, capacity });
        }

        let fixed = self.pinned_weight(Some(shard), Some(key));
        if fixed > (capacity - weight) as u128 {
            return Err(Error::Blocked { shard });
        }

        self.erase(key);
        let mut removed = Vec::new();
        while self.usage[shard] > capacity - weight {
            let victim = self.local_victim(shard).expect("feasibility guarantees a victim");
            self.erase(victim);
            removed.push(victim);
        }

        self.usage[shard] += weight;
        self.entries.insert(
            key,
            Entry {
                key,
                value,
                weight,
                pinned,
                shard,
            },
        );
        self.touch(key);
        Ok(removed)
    }

    pub fn resize(&mut self, shard: usize, capacity: u64) -> Result<Vec<u64>, Error> {
        if shard >= self.capacities.len() {
            return Err(Error::BadShard { shard });
        }
        let minimum = self.pinned_weight(Some(shard), None);
        if minimum > capacity as u128 {
            return Err(Error::PinnedCapacity {
                minimum,
                target: capacity as u128,
            });
        }

        self.capacities[shard] = capacity;
        let mut removed = Vec::new();
        while self.usage[shard] > capacity {
            let victim = self.local_victim(shard).expect("pinned weight fits capacity");
            self.erase(victim);
            removed.push(victim);
        }
        Ok(removed)
    }

    pub fn trim(&mut self, target: u128) -> Result<Vec<u64>, Error> {
        let minimum = self.pinned_weight(None, None);
        if minimum > target {
            return Err(Error::PinnedCapacity { minimum, target });
        }

        let mut removed = Vec::new();
        while self.used() > target {
            let victim = self.global_victim().expect("pinned weight fits target");
            self.erase(victim);
            removed.push(victim);
        }
        Ok(removed)
    }
}
```

```rust
use std::collections::{BTreeMap, VecDeque};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyShards,
    ZeroWeight,
    TooHeavy { weight: u64, capacity: u64 },
    Blocked { shard: usize },
    Missing { key: u64 },
    Pinned { key: u64 },
    BadShard { shard: usize },
    PinnedCapacity { minimum: u128, target: u128 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Entry {
    pub key: u64,
    pub value: i64,
    pub weight: u64,
    pub pinned: bool,
    pub shard: usize,
}

#[derive(Debug, Clone)]
pub struct Cache {
    capacities: Vec<u64>,
    usage: Vec<u64>,
    entries: BTreeMap<u64, Entry>,
    order: VecDeque<u64>,
}

impl Cache {
    pub fn new(capacities: Vec<u64>) -> Result<Self, Error> {
        if capacities.is_empty() {
            return Err(Error::EmptyShards);
        }
        let shard_count = capacities.len();
        Ok(Self {
            capacities,
            usage: vec![0; shard_count],
            entries: BTreeMap::new(),
            order: VecDeque::new(),
        })
    }

    pub fn capacities(&self) -> Vec<u64> {
        self.capacities.iter().copied().collect()
    }

    pub fn usage(&self) -> Vec<u64> {
        self.usage.iter().copied().collect()
    }

    pub fn used(&self) -> u128 {
        self.usage
            .iter()
            .fold(0u128, |sum, &weight| sum + weight as u128)
    }

    pub fn snapshot(&self) -> Vec<Entry> {
        self.order
            .iter()
            .filter_map(|key| self.entries.get(key).cloned())
            .collect()
    }

    fn shard(&self, key: u64) -> usize {
        (key % self.capacities.len() as u64) as usize
    }

    fn touch(&mut self, key: u64) {
        if let Some(position) = self.order.iter().position(|&stored| stored == key) {
            self.order.remove(position);
        }
        self.order.push_back(key);
    }

    fn erase(&mut self, key: u64) -> Option<Entry> {
        let entry = self.entries.remove(&key)?;
        self.usage[entry.shard] -= entry.weight;
        if let Some(position) = self.order.iter().position(|&stored| stored == key) {
            self.order.remove(position);
        }
        Some(entry)
    }

    fn oldest_eligible(&self, shard: Option<usize>) -> Option<u64> {
        self.order.iter().find_map(|&key| {
            let entry = self.entries.get(&key)?;
            (!entry.pinned && shard.is_none_or(|s| entry.shard == s)).then_some(key)
        })
    }

    fn pinned_weight(&self, shard: Option<usize>, except: Option<u64>) -> u128 {
        self.entries
            .iter()
            .filter(|(key, entry)| {
                Some(**key) != except
                    && entry.pinned
                    && shard.map_or(true, |s| entry.shard == s)
            })
            .map(|(_, entry)| entry.weight as u128)
            .sum()
    }

    pub fn get(&mut self, key: u64) -> Option<i64> {
        let value = self.entries.get(&key).map(|entry| entry.value)?;
        self.touch(key);
        Some(value)
    }

    pub fn set_pin(&mut self, key: u64, pinned: bool) -> Result<(), Error> {
        match self.entries.get_mut(&key) {
            Some(entry) => {
                entry.pinned = pinned;
                Ok(())
            }
            None => Err(Error::Missing { key }),
        }
    }

    pub fn remove(&mut self, key: u64) -> Result<i64, Error> {
        match self.entries.get(&key) {
            None => Err(Error::Missing { key }),
            Some(entry) if entry.pinned => Err(Error::Pinned { key }),
            Some(_) => Ok(self.erase(key).unwrap().value),
        }
    }

    pub fn insert(
        &mut self,
        key: u64,
        value: i64,
        weight: u64,
        pinned: bool,
    ) -> Result<Vec<u64>, Error> {
        let shard = self.shard(key);
        let capacity = self.capacities[shard];
        if weight == 0 {
            return Err(Error::ZeroWeight);
        }
        if weight > capacity {
            return Err(Error::TooHeavy { weight, capacity });
        }
        if self.pinned_weight(Some(shard), Some(key)) > (capacity - weight) as u128 {
            return Err(Error::Blocked { shard });
        }

        let mut removed = Vec::new();
        self.erase(key);
        while self.usage[shard] > capacity - weight {
            let victim = self.oldest_eligible(Some(shard)).unwrap();
            self.erase(victim);
            removed.push(victim);
        }
        self.usage[shard] += weight;
        self.entries.insert(
            key,
            Entry {
                key,
                value,
                weight,
                pinned,
                shard,
            },
        );
        self.order.push_back(key);
        Ok(removed)
    }

    pub fn resize(&mut self, shard: usize, capacity: u64) -> Result<Vec<u64>, Error> {
        if shard >= self.capacities.len() {
            return Err(Error::BadShard { shard });
        }
        let minimum = self.pinned_weight(Some(shard), None);
        if minimum > capacity as u128 {
            return Err(Error::PinnedCapacity {
                minimum,
                target: capacity as u128,
            });
        }
        self.capacities[shard] = capacity;
        let mut removed = Vec::new();
        while self.usage[shard] > capacity {
            let victim = self.oldest_eligible(Some(shard)).unwrap();
            self.erase(victim);
            removed.push(victim);
        }
        Ok(removed)
    }

    pub fn trim(&mut self, target: u128) -> Result<Vec<u64>, Error> {
        let minimum = self.pinned_weight(None, None);
        if minimum > target {
            return Err(Error::PinnedCapacity { minimum, target });
        }
        let mut removed = Vec::new();
        while self.used() > target {
            let victim = self.oldest_eligible(None).unwrap();
            self.erase(victim);
            removed.push(victim);
        }
        Ok(removed)
    }
}
```
