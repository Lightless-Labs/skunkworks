- A final quote pair at EOF is an escaped quote, so the quoted field remains unclosed.
- CR is checked before quote-state errors only outside quotes; inside quotes it is field data.
- The initial BOM is skipped only at byte zero; a field BOM is quoted by the writer only when it is the first output field’s prefix.
- Writer NUL preflight covers the whole original table before width and zero-column checks.
- Decimal values are signed hundredths; syntax is validated before range.
- Canonical output always uses LF and one final LF when it has records.

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if start == bytes.len() { return Ok(Vec::new()); }
    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut recno = 0usize;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_at = 0usize;
    let mut touched = false;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 { return Err(CsvError::Nul { record: recno, byte: i }); }
        if !in_quotes && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record: recno, byte: i });
        }
        if in_quotes {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                    touched = true;
                    continue;
                }
                in_quotes = false;
                after_quote = true;
                i += 1;
                continue;
            }
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            i += ch.len_utf8();
            touched = true;
            continue;
        }
        if after_quote {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                after_quote = false;
                touched = true;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                recno += 1;
                after_quote = false;
                touched = false;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte { record: recno, byte: i });
        }
        if b == quote {
            if field.is_empty() {
                in_quotes = true;
                quote_at = i;
                touched = true;
                i += 1;
                continue;
            }
            return Err(CsvError::UnexpectedQuote { record: recno, byte: i });
        }
        if b == delimiter {
            record.push(std::mem::take(&mut field));
            touched = true;
            i += 1;
            continue;
        }
        if b == b'\n' || b == b'\r' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            recno += 1;
            touched = false;
            i += if b == b'\r' { 2 } else { 1 };
            continue;
        }
        let ch = input[i..].chars().next().unwrap();
        field.push(ch);
        touched = true;
        i += ch.len_utf8();
    }
    if in_quotes { return Err(CsvError::UnclosedQuote { record: recno, byte: quote_at }); }
    if touched || after_quote || !record.is_empty() || !field.is_empty() {
        record.push(field);
        records.push(record);
    }
    Ok(records)
}

fn build_table(input: &str, config: Config) -> Result<Table, CsvError> {

        .map_err(|component| CsvError::InvalidConfig { component })?;

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    if b[start] == b'0' { return start == 0 && b.len() == 1; }
    if !(b'1'..=b'9').contains(&b[start]) { return false; }
    b[start+1..].iter().all(u8::is_ascii_digit)
}
fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if b.len() < start + 4 || b[b.len()-3] != b'.' { return false; }
    let whole = &b[start..b.len()-3];
    let frac = &b[b.len()-2..];
    if !frac.iter().all(u8::is_ascii_digit) || whole.is_empty() { return false; }
    if whole == b"0" { return true; }
    (b'1'..=b'9').contains(&whole[0]) && whole[1..].iter().all(u8::is_ascii_digit)
}
fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) { return Err(CellErrorKind::InvalidInteger); }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}
fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) { return Err(CellErrorKind::InvalidDecimal); }
    let neg = s.starts_with('-');
    let body = if neg { &s[1..] } else { s };
    let dot = body.len() - 3;
    let whole: i128 = body[..dot].parse().unwrap();
    let frac: i128 = body[dot+1..].parse().unwrap();
    let n = whole * 100 + frac;
    let signed = if neg { -n } else { n };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}
fn evaluate(cell: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if spec.nullable { return Ok(Value::Null); }
        if spec.kind != ColumnType::Text { return Err(CellErrorKind::Empty); }
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}
pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    'rows: for (ri, row) in table.rows.iter().enumerate() {
        let mut out = Vec::with_capacity(columns.len());
        let mut failed = false;
        for (si, spec) in columns.iter().enumerate() {
            if failed {
                out.push(None);
                continue;
            }
            let col = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(n) => table.column_index(n),
            };
            let evaluated = match col {
                None => Err(CellErrorKind::UnknownColumn),
                Some(i) => match row.get(i) {
                    None => Err(CellErrorKind::MissingCell),
                    Some(cell) => evaluate(cell, spec),
                },
            };
            match evaluated {
                Ok(v) => out.push(Some(v)),
                Err(kind) => {
                    out.push(None);
                    result.errors.push(CellError { row: ri, spec: si, column: col, kind });
                    failed = true;
                    if policy == ErrorPolicy::Stop {
                        result.rows.push(out);
                        break 'rows;
                    }
                    if policy == ErrorPolicy::FirstPerRow { continue; }
                }
            }
        }
        result.rows.push(out);
    }
    result
}

fn cmp_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b) {
        let c = x.as_bytes().cmp(y.as_bytes());
        if c != std::cmp::Ordering::Equal { return c; }
    }
    a.len().cmp(&b.len())

fn encode_field(s: &str, delimiter: u8, quote: u8, first_record: bool, first_field: bool, sole: bool) -> String {
    let b = s.as_bytes();
    let edge_space = b.first().is_some_and(|x| *x == b' ' || *x == b'\t')
        || b.last().is_some_and(|x| *x == b' ' || *x == b'\t');
    let bom = first_record && first_field && s.starts_with('\u{feff}');
    let must = s.as_bytes().contains(&delimiter) || s.as_bytes().contains(&quote)
        || s.contains('\r') || s.contains('\n') || edge_space || bom || (sole && s.is_empty());
    if !must { return s.to_string(); }
    let mut out = String::new();
    out.push(quote as char);
    for ch in s.chars() {
        if ch as u32 == quote as u32 { out.push(quote as char); }
        out.push(ch);
    }
    out.push(quote as char);
    out
}
fn encode_records(records: &[Vec<String>], d: u8, q: u8, ending: &str, final_term: bool) -> String {
    let mut out = String::new();
    for (ri, row) in records.iter().enumerate() {
        if ri > 0 { out.push_str(ending); }
        let sole = row.len() == 1;
        for (ci, field) in row.iter().enumerate() {
            if ci > 0 { out.push(d as char); }
            out.push_str(&encode_field(field, d, q, ri == 0, ci == 0, sole));
        }
    }
    if final_term && !records.is_empty() { out.push_str(ending); }
    out
}
pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    valid_config(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;
    let mut originals: Vec<&Vec<String>> = Vec::new();
    if let Some(h) = &table.header { originals.push(h); }
    originals.extend(table.rows.iter());
    for (ri, row) in originals.iter().enumerate() {
        for (ci, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|b| *b == 0) {
                return Err(WriteError::Nul { record: ri, column: ci, byte });
            }
        }
    }
    if let Some(h) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(WriteError::Width { record: i + 1, expected: h.len(), actual: row.len() });
            }
        }
    }
    for (ri, row) in originals.iter().enumerate() {
        if row.is_empty() { return Err(WriteError::ZeroColumns { record: ri }); }
    }
    if config.mode == WriteMode::Canonical {
        if let Some(h) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(e) = writer_duplicate(h) { return Err(e); }
            }
        }
    }
    let mut records: Vec<Vec<String>> = Vec::new();
    if config.mode == WriteMode::Preserve {
        if let Some(h) = &table.header { records.push(h.clone()); }
        records.extend(table.rows.clone());
        let ending = match config.line_ending { LineEnding::Lf => "\n", LineEnding::CrLf => "\r\n" };
        return Ok(encode_records(&records, config.delimiter, config.quote, ending, config.final_terminator));
    }

    if let Some(h) = &table.header {
        let mut chosen: HashMap<&str, usize> = HashMap::new();
        for (i, name) in h.iter().enumerate() {
            match table.duplicates {
                DuplicatePolicy::KeepLast => { chosen.insert(name, i); }
                DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => { chosen.entry(name).or_insert(i); }
            }
        }
        let mut cols: Vec<(String, usize)> = chosen.into_iter().map(|(n, i)| (n.to_string(), i)).collect();
        cols.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
        let indexes: Vec<usize> = cols.iter().map(|x| x.1).collect();
        records.push(cols.into_iter().map(|x| x.0).collect());
        let mut rows: Vec<Vec<String>> = table.rows.iter().map(|r| indexes.iter().map(|i| r[*i].clone()).collect()).collect();
        rows.sort_by(|a, b| cmp_rows(a, b));
        records.extend(rows);
    } else {
        let mut rows = table.rows.clone();
        rows.sort_by(|a, b| cmp_rows(a, b));
        records = rows;
    }
    Ok(encode_records(&records, config.delimiter, config.quote, "\n", true))
}

fn parse_dialect(s: &str, d: u8, q: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let b = s.as_bytes();
    let offset = if b.starts_with(&[239, 187, 191]) { 3 } else { 0 };
    let mut out = Vec::new();
    if offset == b.len() { return Ok(out); }
    let mut rows: Vec<Vec<String>> = Vec::new();
    let mut row = Vec::new();
    let mut cell = String::new();
    let mut pos = offset;
    let mut record = 0usize;
    let mut quoted = false;
    let mut closed = false;
    let mut opening = 0usize;
    let mut active = false;
    while pos < b.len() {
        let c = b[pos];
        if c == 0 { return Err(CsvError::Nul { record, byte: pos }); }
        if !quoted && c == 13 && b.get(pos + 1) != Some(&10) {
            return Err(CsvError::BareCr { record, byte: pos });
        }
        if quoted {
            if c == q {
                if b.get(pos + 1) == Some(&q) {
                    cell.push(q as char);
                    active = true;
                    pos += 2;
                } else {
                    quoted = false;
                    closed = true;
                    pos += 1;
                }
            } else {
                let ch = s[pos..].chars().next().unwrap();
                cell.push(ch);
                active = true;
                pos += ch.len_utf8();
            }
            continue;
        }
        if closed {
            if c == d {
                row.push(std::mem::take(&mut cell));
                closed = false;
                active = true;
                pos += 1;
            } else if c == 10 || c == 13 {
                row.push(std::mem::take(&mut cell));
                rows.push(std::mem::take(&mut row));
                record += 1;
                active = false;
                closed = false;
                pos += if c == 13 { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record, byte: pos });
            }
            continue;
        }
        if c == q {
            if cell.is_empty() {
                quoted = true;
                opening = pos;
                active = true;
                pos += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record, byte: pos });
            }
        } else if c == d {
            row.push(std::mem::take(&mut cell));
            active = true;
            pos += 1;
        } else if c == 10 || c == 13 {
            row.push(std::mem::take(&mut cell));
            rows.push(std::mem::take(&mut row));
            record += 1;
            active = false;
            pos += if c == 13 { 2 } else { 1 };

            let ch = s[pos..].chars().next().unwrap();
            cell.push(ch);
            active = true;
            pos += ch.len_utf8();
        }
    }
    if quoted { return Err(CsvError::UnclosedQuote { record, byte: opening }); }
    if active || closed || !row.is_empty() || !cell.is_empty() {
        row.push(cell);
        rows.push(row);
    }
    Ok(rows)
}

                    return Err(CsvError::DuplicateHeader { name: n.clone(), first, second: i });
                }
                seen.insert(n, i);
            }
        }
        for (i, row) in parsed.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth { record: i + 1, expected: h.len(), actual: row.len() });
            }
        }
    }
    Ok(Table { header, rows: parsed, duplicates: cfg.duplicates })
}
impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate(config.delimiter, config.quote).map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { cfg: config, buf: String::new(), cached: None })
    }
    pub fn feed(&mut self, s: &str) -> Result<(), CsvError> {
        if self.cached.is_some() { return Err(CsvError::ReaderFinished); }
        self.buf.push_str(s);
        Ok(())
    }
    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if self.cached.is_none() { self.cached = Some(get_table(&self.buf, self.cfg)); }
        self.cached.as_ref().unwrap().clone()
    }
    pub fn reset(&mut self) { self.buf.clear(); self.cached = None; }
}
pub fn read_csv(s: &str, config: Config) -> Result<Table, CsvError> {
    let mut r = StreamReader::new(config)?;
    r.feed(s)?;
    r.finish()
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ColumnRef { Index(usize), Name(String) }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType { Text, Integer, Decimal, Boolean }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec { pub column: ColumnRef, pub kind: ColumnType, pub nullable: bool }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value { Null, Text(String), Integer(i64), Decimal(i64), Boolean(bool) }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind { UnknownColumn, MissingCell, Empty, InvalidInteger, IntegerOverflow, InvalidDecimal, DecimalOverflow, InvalidBoolean }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError { pub row: usize, pub spec: usize, pub column: Option<usize>, pub kind: CellErrorKind }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy { Collect, FirstPerRow, Stop }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult { pub rows: Vec<Vec<Option<Value>>>, pub errors: Vec<CellError> }
fn int_ok(s: &str) -> bool {
    let b=s.as_bytes(); let k=usize::from(b.first()==Some(&b'-'));
    if k==b.len() { return false; }
    if b[k]==b'0' { return k==0 && b.len()==1; }
    (b'1'..=b'9').contains(&b[k]) && b[k+1..].iter().all(u8::is_ascii_digit)
}
fn dec_ok(s: &str) -> bool {
    let b=s.as_bytes(); let k=usize::from(b.first()==Some(&b'-'));
    if b.len()<k+4 || b[b.len()-3]!=b'.' { return false; }
    let w=&b[k..b.len()-3]; let f=&b[b.len()-2..];
    if w==b"0" { f.iter().all(u8::is_ascii_digit) }
    else { !w.is_empty() && (b'1'..=b'9').contains(&w[0]) && w[1..].iter().all(u8::is_ascii_digit) && f.iter().all(u8::is_ascii_digit) }
}
fn value(s: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if s.is_empty() {
        if spec.nullable { return Ok(Value::Null); }
        if spec.kind != ColumnType::Text { return Err(CellErrorKind::Empty); }
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(s.to_owned())),
        ColumnType::Integer => {
            if !int_ok(s) { Err(CellErrorKind::InvalidInteger) }
            else { s.parse().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow) }
        }
        ColumnType::Decimal => {
            if !dec_ok(s) { return Err(CellErrorKind::InvalidDecimal); }
            let neg=s.starts_with('-'); let t=if neg { &s[1..] } else { s };
            let dot=t.len()-3;
            let n: i128=t[..dot].parse::<i128>().unwrap()*100+t[dot+1..].parse::<i128>().unwrap();
            let n=if neg {-n} else { n };
            i64::try_from(n).map(Value::Decimal).map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match s { "yes"|"1"=>Ok(Value::Boolean(true)), "no"|"0"=>Ok(Value::Boolean(false)), _=>Err(CellErrorKind::InvalidBoolean) }
    }
}
pub fn extract_columns(t: &Table, specs: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result=TypedResult { rows:Vec::new(), errors:Vec::new() };
    'outer: for (ri,row) in t.rows.iter().enumerate() {
        let mut out=Vec::with_capacity(specs.len()); let mut stopped=false;
        for (si,spec) in specs.iter().enumerate() {
            if stopped { out.push(None); continue; }
            let col=match &spec.column { ColumnRef::Index(i)=>Some(*i), ColumnRef::Name(n)=>t.column_index(n) };
            let got=match col { None=>Err(CellErrorKind::UnknownColumn), Some(i)=>row.get(i).ok_or(CellErrorKind::MissingCell).and_then(|s| value(s,spec)) };
            match got {
                Ok(v)=>out.push(Some(v)),
                Err(kind)=>{
                    out.push(None); result.errors.push(CellError { row:ri,spec:si,column:col,kind });
                    if policy==ErrorPolicy::Stop { result.rows.push(out); break 'outer; }
                    if policy==ErrorPolicy::FirstPerRow { stopped=true; }
                }
            }
        }
        result.rows.push(out);
    }
    result
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding { Lf, CrLf }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum WriteMode { Preserve, Canonical }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig { pub delimiter:u8,pub quote:u8,pub line_ending:LineEnding,pub final_terminator:bool,pub mode:WriteMode }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError { InvalidConfig{component:ConfigPart},Nul{record:usize,column:usize,byte:usize},Width{record:usize,expected:usize,actual:usize},ZeroColumns{record:usize},DuplicateHeader{name:String,first:usize,second:usize} }
fn compare_rows(a:&[String],b:&[String])->std::cmp::Ordering {
    for (x,y) in a.iter().zip(b) { let c=x.as_bytes().cmp(y.as_bytes()); if c!=std::cmp::Ordering::Equal{return c;} }
    a.len().cmp(&b.len())
}
fn emit_field(s:&str,d:u8,q:u8,bom:bool,sole:bool)->String {
    let bytes=s.as_bytes();
    let edge=bytes.first().is_some_and(|x|*x==32||*x==9)||bytes.last().is_some_and(|x|*x==32||*x==9);
    let quote=bytes.contains(&d)||bytes.contains(&q)||s.contains('\r')||s.contains('\n')||edge||bom||(sole&&s.is_empty());
    if !quote { return s.to_string(); }
    let mut o=String::new(); o.push(q as char);
    for c in s.chars() { if c as u32==q as u32 {o.push(q as char);} o.push(c); }
    o.push(q as char); o
}
fn emit(records:&[Vec<String>],d:u8,q:u8,sep:&str,final_term:bool)->String {
    let mut out=String::new();
    for (ri,r) in records.iter().enumerate() {
        if ri>0 {out.push_str(sep);}
        for (ci,s) in r.iter().enumerate() {
            if ci>0 {out.push(d as char);}
            out.push_str(&emit_field(s,d,q,ri==0&&ci==0&&s.starts_with('\u{feff}'),r.len()==1));
        }
    }
    if final_term&&!records.is_empty(){out.push_str(sep);}
    out
}
pub fn write_csv(t:&Table,c:WriteConfig)->Result<String,WriteError>{
    validate(c.delimiter,c.quote).map_err(|component|WriteError::InvalidConfig{component})?;
    let mut all:Vec<&Vec<String>>=Vec::new(); if let Some(h)=&t.header{all.push(h);} all.extend(t.rows.iter());
    for (ri,r) in all.iter().enumerate(){for (ci,s) in r.iter().enumerate(){if let Some(byte)=s.as_bytes().iter().position(|b|*b==0){return Err(WriteError::Nul{record:ri,column:ci,byte});}}}
    if let Some(h)=&t.header {for (i,r) in t.rows.iter().enumerate(){if r.len()!=h.len(){return Err(WriteError::Width{record:i+1,expected:h.len(),actual:r.len()});}}}
    for (i,r) in all.iter().enumerate(){if r.is_empty(){return Err(WriteError::ZeroColumns{record:i});}}
    if c.mode==WriteMode::Canonical&&t.header.is_some()&&t.duplicates==DuplicatePolicy::Reject {
        let h=t.header.as_ref().unwrap(); let mut seen:HashMap<&str,usize>=HashMap::new();
        for (i,n) in h.iter().enumerate(){if let Some(&first)=seen.get(n.as_str()){return Err(WriteError::DuplicateHeader{name:n.clone(),first,second:i});} seen.insert(n,i);}
    }
    let mut records=Vec::new();
    if c.mode==WriteMode::Preserve {
        if let Some(h)=&t.header{records.push(h.clone());} records.extend(t.rows.clone());
        let sep=match c.line_ending{LineEnding::Lf=>"\n",LineEnding::CrLf=>"\r\n"};
        return Ok(emit(&records,c.delimiter,c.quote,sep,c.final_terminator));
    }
    if let Some(h)=&t.header {
        let mut map:HashMap<String,usize>=HashMap::new();
        for (i,n) in h.iter().enumerate(){match t.duplicates {DuplicatePolicy::KeepLast=>{map.insert(n.clone(),i);},_=>{map.entry(n.clone()).or_insert(i);}}}
        let mut pairs:Vec<(String,usize)>=map.into_iter().collect(); pairs.sort_by(|a,b|a.0.as_bytes().cmp(b.0.as_bytes()));
        let idx:Vec<usize>=pairs.iter().map(|x|x.1).collect();
        records.push(pairs.iter().map(|x|x.0.clone()).collect());
        let mut rows:Vec<Vec<String>>=t.rows.iter().map(|r|idx.iter().map(|i|r[*i].clone()).collect()).collect();
        rows.sort_by(|a,b|compare_rows(a,b)); records.extend(rows);
    } else {
        let mut rows=t.rows.clone(); rows.sort_by(|a,b|compare_rows(a,b)); records=rows;
    }
    Ok(emit(&records,c.delimiter,c.quote,"\n",true))
}
