- An empty input has no records, but each terminator creates a record; a final terminator creates no extra record.
- CSV error record indexes count only terminators outside quotes, while byte offsets count UTF-8 bytes from the original input, including an ignored initial BOM.
- NUL and bare CR checks take precedence over state-specific errors; unclosed quotes are reported only after scanning all bytes.
- Reader lexical validation completes before duplicate-header and width validation; writer preflight checks NULs across the original table before structural errors.
- Integer and decimal spelling is validated in full before range checking; decimal values are signed hundredths.
- Canonical writing sorts decoded field strings by UTF-8 bytes, and Preserve uses the configured terminator and final-terminator setting.

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart { Delimiter, Quote }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy { Reject, KeepFirst, KeepLast }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]

    pub delimiter: u8, pub quote: u8,
    pub headers: bool, pub duplicates: DuplicatePolicy,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>, pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}
impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => header.iter().position(|s| s == name),
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    let mut records = Vec::new();
    if start == bytes.len() {
        return Ok(records);
    }

    let mut fields = Vec::new();
    let mut field = Vec::new();
    let mut i = start;
    let mut record = 0usize;
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: closed quote
    let mut quote_start = 0usize;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if state != 2 && b == b'\r' && (i + 1 >= bytes.len() || bytes[i + 1] != b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if state == 2 {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote);
                    i += 2;
                } else {
                    state = 3;
                    i += 1;
                }
            } else {
                field.push(b);
                i += 1;
            }
            continue;
        }

        if state == 3 {
            if b == delimiter {
                fields.push(String::from_utf8(field).unwrap());
                field = Vec::new();
                state = 0;
                i += 1;
            } else if b == b'\n' || b == b'\r' {
                fields.push(String::from_utf8(field).unwrap());
                records.push(fields);
                fields = Vec::new();
                field = Vec::new();
                state = 0;
                record += 1;
                i += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record, byte: i });
            }
            continue;
        }

        if b == b'\r' || b == b'\n' {
            fields.push(String::from_utf8(field).unwrap());
            records.push(fields);
            fields = Vec::new();
            field = Vec::new();
            state = 0;
            record += 1;
            i += if b == b'\r' { 2 } else { 1 };
        } else if b == delimiter {
            fields.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            state = 0;
            i += 1;
        } else if b == quote {
            if state == 0 && field.is_empty() {
                state = 2;
                quote_start = i;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record, byte: i });
            }
        } else {
            field.push(b);
            state = 1;
            i += 1;
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote { record, byte: quote_start });
    }
    if !fields.is_empty() || !field.is_empty() || state != 0 {
        fields.push(String::from_utf8(field).unwrap());
        records.push(fields);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table { header: None, rows: records, duplicates: config.duplicates });
    }
    if records.is_empty() {
        return Ok(Table { header: None, rows: Vec::new(), duplicates: config.duplicates });
    }

    let mut iter = records.into_iter();
    let header = iter.next().unwrap();
    if config.duplicates == DuplicatePolicy::Reject {
        let mut seen: HashMap<String, usize> = HashMap::new();
        for (second, name) in header.iter().enumerate() {
            if let Some(&first) = seen.get(name) {
                return Err(CsvError::DuplicateHeader { name: name.clone(), first, second });
            }
            seen.insert(name.clone(), second);
        }
    }

    let rows: Vec<Vec<String>> = iter.collect();
    for (idx, row) in rows.iter().enumerate() {
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth {
                record: idx + 1,
                expected: header.len(),
                actual: row.len(),
            });
        }
    }
    Ok(Table { header: Some(header), rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}
impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), cached: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

    pub row: usize, pub spec: usize, pub column: Option<usize>,
    pub kind: CellErrorKind,
}

pub enum ErrorPolicy { Collect, FirstPerRow, Stop }

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    let b = s.as_bytes();
    if b.is_empty() { return Err(CellErrorKind::InvalidInteger); }
    let digits = if b[0] == b'-' { &b[1..] } else { b };
    if digits.is_empty() || digits[0] == b'0' && digits.len() > 1
        || digits[0] == b'0' && b[0] == b'-'
        || digits[0] == b'0' && b[0] != b'-' && digits.len() > 1
        || digits[0] != b'0' && !(b'1'..=b'9').contains(&digits[0])
        || digits.iter().any(|c| !c.is_ascii_digit()) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    let b = s.as_bytes();
    let mut i = 0usize;
    if b.first() == Some(&b'-') { i = 1; }
    let integral_start = i;
    while i < b.len() && b[i].is_ascii_digit() { i += 1; }
    let integral = &b[integral_start..i];
    if integral.is_empty() || (integral.len() > 1 && integral[0] == b'0')
        || i >= b.len() || b[i] != b'.' || b.len() - i != 3
        || !b[i + 1..].iter().all(|c| c.is_ascii_digit()) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    if b[0] == b'-' && integral == b"0" && b[i + 1..] == *b"00" {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let mut digits = String::new();
    if b[0] == b'-' { digits.push('-'); }
    digits.push_str(std::str::from_utf8(integral).unwrap());
    digits.push_str(std::str::from_utf8(&b[i + 1..]).unwrap());
    digits.parse::<i64>().map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate_cell(s: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if spec.nullable && s.is_empty() { return Ok(Value::Null); }
    if s.is_empty() {
        if spec.kind == ColumnType::Text { return Ok(Value::Text(String::new())); }
        return Err(CellErrorKind::Empty);
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(s.to_owned())),
        ColumnType::Integer => parse_integer(s).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(s).map(Value::Decimal),
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    'rows: for (row_idx, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut failed = false;
        for (spec_idx, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(k) => Some(*k),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (column, value_or_error) = match resolved {
                None => (None, Err(CellErrorKind::UnknownColumn)),
                Some(k) => match row.get(k) {
                    None => (Some(k), Err(CellErrorKind::MissingCell)),
                    Some(cell) => (Some(k), evaluate_cell(cell, spec)),
                },
            };
            match value_or_error {
                Ok(value) => output.push(Some(value)),
                Err(kind) => {
                    output.push(None);
                    result.errors.push(CellError { row: row_idx, spec: spec_idx, column, kind });
                    failed = true;
                    if policy == ErrorPolicy::FirstPerRow || policy == ErrorPolicy::Stop {
                        for _ in spec_idx + 1..columns.len() { output.push(None); }
                        result.rows.push(output);
                        if policy == ErrorPolicy::Stop { break 'rows; }
                        break;
                    }
                }
            }
        }
        result.rows.push(output);
        let _ = failed;
    }
    result
}

    ZeroColumns { record: usize },

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header { original.push(header); }
    original.extend(table.rows.iter());

    for (record, row) in original.iter().enumerate() {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul { record, column, byte });
            }
        }
    }

    if let Some(header) = &table.header {
        for (idx, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: idx + 1, expected: header.len(), actual: row.len(),
                });
            }
        }
    }

    for (record, row) in original.iter().enumerate() {
        if row.is_empty() { return Err(WriteError::ZeroColumns { record }); }
    }

    if config.mode == WriteMode::Canonical
        && table.header.is_some()
        && table.duplicates == DuplicatePolicy::Reject
    {
        let header = table.header.as_ref().unwrap();
        let mut seen: HashMap<String, usize> = HashMap::new();
        for (second, name) in header.iter().enumerate() {
            if let Some(&first) = seen.get(name) {
                return Err(WriteError::DuplicateHeader { name: name.clone(), first, second });
            }
            seen.insert(name.clone(), second);
        }
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(h) = &table.header { records.push(h.clone()); }
            records.extend(table.rows.clone());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut keep: HashMap<String, usize> = HashMap::new();
                for (idx, name) in header.iter().enumerate() {
                    match table.duplicates {
                        DuplicatePolicy::KeepLast => { keep.insert(name.clone(), idx); }
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                            keep.entry(name.clone()).or_insert(idx);
                        }
                    }
                }
                let mut retained: Vec<(String, usize)> = keep.into_iter().collect();
                retained.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
                records.push(retained.iter().map(|(name, _)| name.clone()).collect());
                let mut projected: Vec<Vec<String>> = table.rows.iter().map(|row| {
                    retained.iter().map(|(_, idx)| row[*idx].clone()).collect()
                }).collect();
                projected.sort_by(|a, b| compare_rows(a, b));
                records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| compare_rows(a, b));
                records = rows;
            }
        }
    }

    if records.is_empty() { return Ok(String::new()); }
    let (ending, final_terminator) = match config.mode {
        WriteMode::Preserve => (
            match config.line_ending { LineEnding::Lf => "\n", LineEnding::CrLf => "\r\n" },
            config.final_terminator,
        ),
        WriteMode::Canonical => ("\n", true),
    };

    let mut out = String::new();
    for (ri, row) in records.iter().enumerate() {
        if ri > 0 { out.push_str(ending); }
        for (ci, field) in row.iter().enumerate() {
            if ci > 0 { out.push(config.delimiter as char); }
            let starts_bom = ri == 0 && ci == 0 && field.as_bytes().starts_with(&[0xef, 0xbb, 0xbf]);
            let sole_empty = row.len() == 1 && field.is_empty();
            let edge_space = field.starts_with(' ') || field.starts_with('\t')
                || field.ends_with(' ') || field.ends_with('\t');
            let needs_quote = field.as_bytes().contains(&config.delimiter)
                || field.as_bytes().contains(&config.quote)
                || field.contains('\r') || field.contains('\n')
                || edge_space || starts_bom || sole_empty;
            if needs_quote {
                out.push(config.quote as char);
                for ch in field.chars() {
                    if ch == config.quote as char { out.push(config.quote as char); }
                    out.push(ch);
                }
                out.push(config.quote as char);
            } else {
                out.push_str(field);
            }
        }
    }
    if final_terminator { out.push_str(ending); }
    Ok(out)
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let ord = x.as_bytes().cmp(y.as_bytes());
        if ord != std::cmp::Ordering::Equal { return ord; }
    }
    a.len().cmp(&b.len())
}

fn dialect_ok(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if delimiter < 0x21 || delimiter > 0x7e { Err(ConfigPart::Delimiter) }
    else if quote < 0x21 || quote > 0x7e { Err(ConfigPart::Quote) }
    else if delimiter == quote { Err(ConfigPart::Quote) }
    else { Ok(()) }
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let offset = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if offset == bytes.len() { return Ok(Vec::new()); }

    let mut output = Vec::new();
    let mut row = Vec::new();
    let mut cell = Vec::new();
    let mut idx = offset;
    let mut rec = 0usize;
    let mut mode = 0u8;
    let mut opening = 0usize;
    let mut any = false;

    while idx < bytes.len() {
        let b = bytes[idx];
        if b == 0 { return Err(CsvError::Nul { record: rec, byte: idx }); }
        if mode != 2 && b == b'\r' && (idx + 1 == bytes.len() || bytes[idx + 1] != b'\n') {
            return Err(CsvError::BareCr { record: rec, byte: idx });
        }

        if mode == 2 {
            if b == quote {
                if idx + 1 < bytes.len() && bytes[idx + 1] == quote {
                    cell.push(quote);
                    idx += 2;
                } else {
                    mode = 3;
                    idx += 1;
                }
            } else {
                cell.push(b);
                idx += 1;
            }
            any = true;
            continue;
        }

        if mode == 3 {
            if b == delimiter {
                row.push(String::from_utf8(cell).unwrap());
                cell = Vec::new();
                mode = 0;
                idx += 1;
                any = true;
            } else if b == b'\n' || b == b'\r' {
                row.push(String::from_utf8(cell).unwrap());
                output.push(row);
                row = Vec::new();
                cell = Vec::new();
                mode = 0;
                rec += 1;
                idx += if b == b'\r' { 2 } else { 1 };
                any = false;
            } else {
                return Err(CsvError::TrailingByte { record: rec, byte: idx });
            }
            continue;
        }

        if b == b'\n' || b == b'\r' {
            row.push(String::from_utf8(cell).unwrap());
            output.push(row);
            row = Vec::new();
            cell = Vec::new();
            mode = 0;
            rec += 1;
            idx += if b == b'\r' { 2 } else { 1 };
            any = false;
        } else if b == delimiter {
            row.push(String::from_utf8(cell).unwrap());
            cell = Vec::new();
            mode = 0;
            idx += 1;
            any = true;
        } else if b == quote {
            if mode == 0 && cell.is_empty() {
                mode = 2;
                opening = idx;
                idx += 1;
                any = true;
            } else {
                return Err(CsvError::UnexpectedQuote { record: rec, byte: idx });
            }
        } else {
            cell.push(b);
            mode = 1;
            idx += 1;
            any = true;
        }
    }

    if mode == 2 { return Err(CsvError::UnclosedQuote { record: rec, byte: opening }); }
    if any || !row.is_empty() || !cell.is_empty() || mode != 0 {
        row.push(String::from_utf8(cell).unwrap());
        output.push(row);
    }
    Ok(output)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {

fn assemble(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table { header: None, rows: records, duplicates: config.duplicates });
    }
    if records.is_empty() {
        return Ok(Table { header: None, rows: Vec::new(), duplicates: config.duplicates });
    }
    let mut it = records.into_iter();
    let header = it.next().unwrap();
    if config.duplicates == DuplicatePolicy::Reject {
        let mut seen = HashMap::<&str, usize>::new();
        for (col, name) in header.iter().enumerate() {
            if let Some(&first) = seen.get(name.as_str()) {
                return Err(CsvError::DuplicateHeader { name: name.clone(), first, second: col });
            }
            seen.insert(name.as_str(), col);
        }
    }
    let rows: Vec<_> = it.collect();
    for (i, row) in rows.iter().enumerate() {
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth { record: i + 1, expected: header.len(), actual: row.len() });
        }
    }
    Ok(Table { header: Some(header), rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    bytes: Vec<u8>,
    result: Option<Result<Table, CsvError>>,
}
impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        dialect_ok(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, bytes: Vec::new(), result: None })
    }
    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() { return Err(CsvError::ReaderFinished); }
        self.bytes.extend_from_slice(chunk.as_bytes());
        Ok(())
    }
    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(r) = &self.result { return r.clone(); }
        let text = std::str::from_utf8(&self.bytes).unwrap();
        let r = parse_with(text, self.config.delimiter, self.config.quote)
            .and_then(|records| assemble(records, self.config));
        self.result = Some(r.clone());
        r
    }
    pub fn reset(&mut self) {
        self.bytes.clear();
        self.result = None;
    }
}
pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut r = StreamReader::new(config)?;
    r.feed(input)?;
    r.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ColumnRef { Index(usize), Name(String) }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType { Text, Integer, Decimal, Boolean }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef, pub kind: ColumnType, pub nullable: bool,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value { Null, Text(String), Integer(i64), Decimal(i64), Boolean(bool) }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn, MissingCell, Empty, InvalidInteger, IntegerOverflow,
    InvalidDecimal, DecimalOverflow, InvalidBoolean,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize, pub spec: usize, pub column: Option<usize>,
    pub kind: CellErrorKind,
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy { Collect, FirstPerRow, Stop }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>, pub errors: Vec<CellError>,
}

fn integer_value(s: &str) -> Result<i64, CellErrorKind> {
    let b = s.as_bytes();
    if b.is_empty() { return Err(CellErrorKind::InvalidInteger); }
    let d = if b[0] == b'-' { &b[1..] } else { b };
    if d.is_empty() || !d.iter().all(|x| x.is_ascii_digit())
        || (d.len() > 1 && d[0] == b'0')
        || (b[0] == b'-' && d[0] == b'0')
    {
        return Err(CellErrorKind::InvalidInteger);
    }
    if d[0] != b'0' && !(b'1'..=b'9').contains(&d[0]) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}
fn decimal_value(s: &str) -> Result<i64, CellErrorKind> {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    let dot = match b[start..].iter().position(|&x| x == b'.') {
        Some(p) => start + p,
        None => return Err(CellErrorKind::InvalidDecimal),
    };
    let whole = &b[start..dot];
    let frac = &b[dot + 1..];
    if whole.is_empty() || (whole.len() > 1 && whole[0] == b'0')
        || !whole.iter().all(|x| x.is_ascii_digit())
        || (whole[0] != b'0' && !(b'1'..=b'9').contains(&whole[0]))
        || frac.len() != 2 || !frac.iter().all(|x| x.is_ascii_digit())
        || (start == 1 && whole == b"0" && frac == b"00")
    {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let mut magnitude = String::from_utf8(whole.to_vec()).unwrap();
    magnitude.push_str(std::str::from_utf8(frac).unwrap());
    if start == 1 { magnitude.insert(0, '-'); }
    magnitude.parse::<i64>().map_err(|_| CellErrorKind::DecimalOverflow)
}
fn cell_value(s: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if spec.nullable && s.is_empty() { return Ok(Value::Null); }
    if s.is_empty() && spec.kind != ColumnType::Text { return Err(CellErrorKind::Empty); }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(s.to_owned())),
        ColumnType::Integer => integer_value(s).map(Value::Integer),
        ColumnType::Decimal => decimal_value(s).map(Value::Decimal),
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}
pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut out = TypedResult { rows: Vec::new(), errors: Vec::new() };
    'outer: for (ri, row) in table.rows.iter().enumerate() {
        let mut values = Vec::with_capacity(columns.len());
        for (si, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(n) => table.column_index(n),
            };
            let (col, val) = match resolved {
                None => (None, Err(CellErrorKind::UnknownColumn)),
                Some(i) => match row.get(i) {
                    None => (Some(i), Err(CellErrorKind::MissingCell)),
                    Some(s) => (Some(i), cell_value(s, spec)),
                },
            };
            match val {
                Ok(v) => values.push(Some(v)),
                Err(kind) => {
                    values.push(None);
                    out.errors.push(CellError { row: ri, spec: si, column: col, kind });
                    if policy != ErrorPolicy::Collect {
                        values.resize(columns.len(), None);
                        out.rows.push(values);
                        if policy == ErrorPolicy::Stop { break 'outer; }
                        break;
                    }
                }
            }
        }
        out.rows.push(values);
    }
    out
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding { Lf, CrLf }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum WriteMode { Preserve, Canonical }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {
    pub delimiter: u8, pub quote: u8, pub line_ending: LineEnding,
    pub final_terminator: bool, pub mode: WriteMode,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn lex_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b) {
        let c = x.as_bytes().cmp(y.as_bytes());
        if c != std::cmp::Ordering::Equal { return c; }
    }
    a.len().cmp(&b.len())
}
fn quote_field(s: &str, delim: u8, quote: u8, first_record: bool, first_field: bool, sole: bool) -> String {
    let bom = first_record && first_field && s.as_bytes().starts_with(&[0xef,0xbb,0xbf]);
    let edge = s.starts_with(' ') || s.starts_with('\t') || s.ends_with(' ') || s.ends_with('\t');
    let must = s.as_bytes().contains(&delim) || s.as_bytes().contains(&quote)
        || s.contains('\r') || s.contains('\n') || edge || bom || (sole && s.is_empty());
    if !must { return s.to_owned(); }
    let mut out = String::new();
    out.push(quote as char);
    for ch in s.chars() {
        if ch == quote as char { out.push(quote as char); }
        out.push(ch);
    }
    out.push(quote as char);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    dialect_ok(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;
    let mut orig: Vec<&Vec<String>> = Vec::new();
    if let Some(h) = &table.header { orig.push(h); }
    orig.extend(table.rows.iter());
    for (r, row) in orig.iter().enumerate() {
        for (c, field) in row.iter().enumerate() {
            if let Some(b) = field.as_bytes().iter().position(|&x| x == 0) {
                return Err(WriteError::Nul { record: r, column: c, byte: b });
            }
        }
    }
    if let Some(h) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(WriteError::Width { record: i + 1, expected: h.len(), actual: row.len() });
            }
        }
    }
    for (r, row) in orig.iter().enumerate() {
        if row.is_empty() { return Err(WriteError::ZeroColumns { record: r }); }
    }
    if config.mode == WriteMode::Canonical && table.header.is_some()
        && table.duplicates == DuplicatePolicy::Reject
    {
        let h = table.header.as_ref().unwrap();
        let mut seen: HashMap<&str, usize> = HashMap::new();
        for (second, name) in h.iter().enumerate() {
            if let Some(&first) = seen.get(name.as_str()) {
                return Err(WriteError::DuplicateHeader { name: name.clone(), first, second });
            }
            seen.insert(name.as_str(), second);
        }
    }

    let mut recs: Vec<Vec<String>> = match config.mode {
        WriteMode::Preserve => {
            let mut v = Vec::new();
            if let Some(h) = &table.header { v.push(h.clone()); }
            v.extend(table.rows.clone());
            v
        }
        WriteMode::Canonical => {
            if let Some(h) = &table.header {
                let mut chosen: HashMap<String, usize> = HashMap::new();
                for (i, name) in h.iter().enumerate() {
                    match table.duplicates {
                        DuplicatePolicy::KeepLast => { chosen.insert(name.clone(), i); }
                        _ => { chosen.entry(name.clone()).or_insert(i); }
                    }
                }
                let mut cols: Vec<(String, usize)> = chosen.into_iter().collect();
                cols.sort_by(|a,b| a.0.as_bytes().cmp(b.0.as_bytes()));
                let mut v = vec![cols.iter().map(|x| x.0.clone()).collect()];
                let mut projected: Vec<Vec<String>> = table.rows.iter()
                    .map(|r| cols.iter().map(|x| r[x.1].clone()).collect()).collect();
                projected.sort_by(|a,b| lex_rows(a,b));
                v.extend(projected);
                v
            } else {
                let mut r = table.rows.clone();
                r.sort_by(|a,b| lex_rows(a,b));
                r
            }
        }
    };

    if recs.is_empty() { return Ok(String::new()); }
    let ending = match config.mode {
        WriteMode::Preserve => match config.line_ending { LineEnding::Lf => "\n", LineEnding::CrLf => "\r\n" },
        WriteMode::Canonical => "\n",
    };
    let final_term = config.mode == WriteMode::Canonical || config.final_terminator;
    let mut out = String::new();
    for (ri, row) in recs.iter().enumerate() {
        if ri != 0 { out.push_str(ending); }
        for (ci, field) in row.iter().enumerate() {
            if ci != 0 { out.push(config.delimiter as char); }
            out.push_str(&quote_field(field, config.delimiter, config.quote, ri == 0, ci == 0, row.len() == 1));
        }
    }
    if final_term { out.push_str(ending); }
    Ok(out)
}
