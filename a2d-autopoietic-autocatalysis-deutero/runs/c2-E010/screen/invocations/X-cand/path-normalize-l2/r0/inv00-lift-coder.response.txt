- NUL, backslash, and control checks apply to the full raw path in that precedence order; offsets are UTF-8 byte offsets.
- Absolute paths have one root, and retained `..` components never cancel each other.
- Directory intent comes from the original request’s trailing slash or final nonempty `.`/`..` component.
- `resolve` validates cwd, then optional home, then the request; context errors hide their underlying lexical errors.
- A leading `~` expands only when it is exactly `~` or followed by `/`; other leading tilde forms fail at byte 1.
- Both implementations use checked indexing and linear-time component reduction; B uses a separate stack representation.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<(usize, &str)> {
    let mut out = Vec::new();
    let mut start = None;
    for (i, b) in path.bytes().enumerate() {
        if b == b'/' {
            if let Some(s) = start.take() {
                out.push((s, &path[s..i]));
            }
        } else if start.is_none() {
            start = Some(i);
        }
    }
    if let Some(s) = start {
        out.push((s, &path[s..]));
    }
    out
}

fn reduce(
    stack: &mut Vec<(String, bool)>,
    path: &str,
    skip_initial: usize,
    absolute: bool,
) -> Result<(), PathError> {
    for (at, part) in components(path) {
        if at < skip_initial {
            continue;
        }
        match part {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|(_, ordinary)| *ordinary) {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(("..".to_owned(), false));
                }
            }
            _ => stack.push((part.to_owned(), true)),
        }
    }
    Ok(())
}

fn render(stack: &[(String, bool)], absolute: bool, directory: bool) -> String {
    let mut result = String::new();
    if absolute {
        result.push('/');
    }
    for (i, (part, _)) in stack.iter().enumerate() {
        if i != 0 {
            result.push('/');
        }
        result.push_str(part);
    }

    if result.is_empty() {
        result.push('.');
    }
    if directory && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let directory = directory_intent(path);
    let mut stack = Vec::new();
    reduce(&mut stack, path, 0, absolute)?;
    Ok(render(&stack, absolute, directory))
}

fn validated_absolute_context(path: &str, error: PathError) -> Result<Vec<(String, bool)>, PathError> {
    if path.as_bytes().first() != Some(&b'/') {
        return Err(error);
    }
    match normalize(path) {
        Ok(normalized) => {
            let mut stack = Vec::new();
            for (_, part) in components(&normalized) {
                stack.push((part.to_owned(), part != ".."));
            }
            Ok(stack)
        }
        Err(_) => Err(error),
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_stack = validated_absolute_context(cwd, PathError::InvalidCwd)?;
    let home_stack = match home {
        Some(value) => Some(validated_absolute_context(value, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde = bytes.first() == Some(&b'~');
    let expanded = if tilde {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        Some(home_stack.as_ref().ok_or(PathError::MissingHome)?)
    } else {
        None
    };

    let directory = directory_intent(path);
    let absolute = !tilde && bytes.first() == Some(&b'/');
    let mut stack = if absolute {
        Vec::new()
    } else if let Some(home) = expanded {
        home.clone()
    } else {
        cwd_stack
    };

    if tilde {
        let skip = if bytes.len() > 1 && bytes[1] == b'/' {
            2
        } else {
            1
        };
        reduce(&mut stack, path, skip, true)?;
    } else {
        reduce(&mut stack, path, 0, true)?;
    }

    Ok(render(&stack, true, directory))
}
``` 

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    for (i, byte) in path.bytes().enumerate() {
        if byte == 0 {
            return Err(PathError::Nul { at: i });
        }
    }
    for (i, byte) in path.bytes().enumerate() {
        if byte == b'\\' {
            return Err(PathError::Backslash { at: i });
        }
    }
    for (i, byte) in path.bytes().enumerate() {
        if (1..=31).contains(&byte) || byte == 127 {
            return Err(PathError::Control { at: i });
        }
    }
    Ok(())
}

fn final_component_is_dir(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    let end = path.rsplit('/').find(|s| !s.is_empty());
    matches!(end, Some(".") | Some(".."))
}

fn split_components(path: &str) -> Vec<(usize, &str)> {
    let mut result = Vec::new();
    let mut offset = 0usize;
    for part in path.split('/') {
        if !part.is_empty() {
            result.push((offset, part));
        }
        offset += part.len() + 1;
    }
    result
}

fn apply(
    stack: &mut Vec<String>,
    src: &str,
    offset_floor: usize,
    can_escape: bool,
) -> Result<(), PathError> {
    for (offset, part) in split_components(src) {
        if offset < offset_floor {
            continue;
        }
        if part == "." {
            continue;
        }
        if part == ".." {
            if stack.last().is_some_and(|s| s != "..") {
                stack.pop();
            } else if can_escape {
                return Err(PathError::Escape { at: offset });
            } else {
                stack.push("..".to_string());
            }
        } else {
            stack.push(part.to_string());
        }
    }
    Ok(())
}

fn output(stack: &[String], absolute: bool, directory: bool) -> String {
    let mut pieces = Vec::with_capacity(stack.len() + usize::from(absolute));
    if absolute {
        pieces.push(String::new());
    }
    pieces.extend(stack.iter().cloned());

    let mut result = pieces.join("/");
    if result.is_empty() {
        result = ".".to_string();
    }
    if directory && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn context_stack(path: &str, invalid: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(invalid);
    }
    let normalized = normalize(path).map_err(|_| invalid)?;
    Ok(split_components(&normalized)
        .into_iter()
        .map(|(_, part)| part.to_string())
        .collect())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate(path)?;
    let absolute = path.starts_with('/');
    let directory = final_component_is_dir(path);
    let mut stack = Vec::new();
    apply(&mut stack, path, 0, absolute)?;
    Ok(output(&stack, absolute, directory))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let mut stack = context_stack(cwd, PathError::InvalidCwd)?;
    let home_stack = match home {
        Some(h) => Some(context_stack(h, PathError::InvalidHome)?),
        None => None,
    };

    validate(path)?;
    let bytes = path.as_bytes();
    let is_tilde = bytes.first() == Some(&b'~');
    if is_tilde && bytes.len() > 1 && bytes[1] != b'/' {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    let absolute;
    let start;
    let mut floor = 0;
    if is_tilde {
        stack = home_stack.ok_or(PathError::MissingHome)?;
        absolute = true;
        floor = if bytes.len() > 1 { 2 } else { 1 };
    } else if bytes.first() == Some(&b'/') {
        stack.clear();
        absolute = true;
    } else {
        absolute = true;
    }

    let directory = final_component_is_dir(path);
    apply(&mut stack, path, floor, false)?;
    Ok(output(&stack, absolute, directory))
}
```
