#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.split('/')
        .filter(|part| !part.is_empty())
        .last()
        .is_some_and(|part| part == "." || part == "..")
}

fn components(path: &str, base: &[String], absolute: bool) -> Result<Vec<String>, PathError> {
    let mut out = base.to_vec();
    let mut offset = 0usize;

    for part in path.split('/') {
        let len = part.len();
        if !part.is_empty() {
            if part == "." {
                // Nothing to do.
            } else if part == ".." {
                if out.last().is_some_and(|last| last != "..") {
                    out.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: offset });
                } else {
                    out.push("..".to_string());
                }
            } else {
                out.push(part.to_string());
            }
        }
        offset = offset.saturating_add(len).saturating_add(1);
    }
    Ok(out)
}

fn render(parts: &[String], absolute: bool, dir: bool) -> String {
    let mut result = if absolute {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    } else if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    };

    if dir && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let dir = directory_intent(path);
    let parts = components(path, &[], absolute)?;
    Ok(render(&parts, absolute, dir))
}

fn normalized_context(path: &str, invalid: PathError) -> Result<Vec<String>, PathError> {
    if path.as_bytes().first() != Some(&b'/') {
        return Err(invalid);
    }
    match normalize(path) {
        Ok(value) => {
            let trimmed = value.trim_matches('/');
            if trimmed.is_empty() {
                Ok(Vec::new())
            } else {
                Ok(trimmed.split('/').map(str::to_string).collect())
            }
        }
        Err(_) => Err(invalid),
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = normalized_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(normalized_context(value, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde = bytes.first() == Some(&b'~');
    let expanded = if tilde {
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            let base = home_parts.ok_or(PathError::MissingHome)?;
            let remainder = if bytes.len() > 1 { &path[2..] } else { "" };
            Some((base, remainder, true))
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else {
        None
    };

    let dir = if tilde {
        bytes.len() > 1 && bytes.get(1) == Some(&b'/') || directory_intent(path)
    } else {
        directory_intent(path)
    };

    let (base, remainder, absolute) = if let Some((base, remainder, _)) = expanded {
        (base, remainder, true)
    } else if bytes.first() == Some(&b'/') {
        (Vec::new(), path, true)
    } else {
        (cwd_parts, path, true)
    };

    // Offsets in the remainder are translated back to the original request.
    let prefix = if tilde { 2 } else { 0 };
    let mut result = base;
    let mut offset = prefix;
    for part in remainder.split('/') {
        let len = part.len();
        if !part.is_empty() {
            if part == "." {
            } else if part == ".." {
                if result.is_empty() {
                    return Err(PathError::Escape {
                        at: offset,
                    });
                }
                result.pop();

                result.push(part.to_string());
            }
        }
        offset = offset.saturating_add(len).saturating_add(1);

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    let b = path.as_bytes();
    if let Some(i) = b.iter().position(|&x| x == 0) {
        return Err(PathError::Nul { at: i });
    }
    if let Some(i) = b.iter().position(|&x| x == b'\\') {
        return Err(PathError::Backslash { at: i });
    }
    if let Some(i) = b.iter().position(|&x| (1..=31).contains(&x) || x == 127) {
        return Err(PathError::Control { at: i });
    }
    Ok(())
}

fn wants_directory(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|s| !s.is_empty())
            .next_back()
            .is_some_and(|s| s == "." || s == "..")
}

fn split_normalized_absolute(path: &str, invalid: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(invalid);
    }
    validate_raw(path).map_err(|_| invalid.clone())?;
    let mut stack: Vec<String> = Vec::new();
    let mut at = 0usize;
    for part in path.split('/') {
        let len = part.len();
        match part {
            "" | "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(invalid);
                }
            }
            other => stack.push(other.to_owned()),
        }
        at = at.saturating_add(len).saturating_add(1);
    }
    let _ = at;
    Ok(stack)
}

fn reduce_relative_or_absolute(
    path: &str,
    mut stack: Vec<String>,
    offset_base: usize,
) -> Result<Vec<String>, PathError> {
    let mut at = offset_base;
    for part in path.split('/') {
        let len = part.len();
        match part {
            "" | "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            other => stack.push(other.to_owned()),
        }
        at = at.saturating_add(len).saturating_add(1);
    }
    Ok(stack)
}

fn emit(stack: &[String], dir: bool) -> String {
    let mut out = if stack.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", stack.join("/"))
    };
    if dir && out != "/" {
        out.push('/');
    }
    out
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let mut stack = Vec::new();
    let mut at = 0usize;
    for part in path.split('/') {
        let len = part.len();
        match part {
            "" | "." => {}
            ".." => {
                if stack.last().is_some_and(|s: &String| s != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            other => stack.push(other.to_owned()),
        }
        at = at.saturating_add(len).saturating_add(1);
    }

    let mut out = if absolute {
        if stack.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", stack.join("/"))
        }
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };
    if wants_directory(path) && out != "/" && out != "." {
        out.push('/');
    }
    Ok(out)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd = split_normalized_absolute(cwd, PathError::InvalidCwd)?;
    let home = match home {
        Some(p) => Some(split_normalized_absolute(p, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let bytes = path.as_bytes();
    let (base, rest, offset) = if bytes.first() == Some(&b'~') {
        if bytes.len() == 1 || bytes[1] == b'/' {
            let base = home.ok_or(PathError::MissingHome)?;
            let rest = if bytes.len() > 1 { &path[2..] } else { "" };
            (base, rest, 2)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if bytes.first() == Some(&b'/') {
        (Vec::new(), path, 0)
    } else {
        (cwd, path, 0)
    };

    let stack = reduce_relative_or_absolute(rest, base, offset)?;
    Ok(emit(&stack, wants_directory(path)))
}
