- Validation precedence is global: empty, then NUL, backslash, and control bytes, before component reduction.
- A leading slash alone makes a path absolute; repeated slashes collapse to one.
- Directory intent uses the raw path’s final slash or final nonempty `.`/`..` component.
- `resolve` validates cwd, then optional home, then the request; invalid context errors hide their underlying causes.
- Tilde expansion applies only to exactly `~` or a leading `~/`; other leading tilde forms fail at byte offset 1.
- Both implementations use byte offsets from the original input and treat UTF-8 names literally.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn reduce_components<'a, I>(
    components: I,
    absolute: bool,
) -> Result<Vec<&'a str>, PathError>
where
    I: IntoIterator<Item = (&'a str, usize)>,
{
    let mut kept: Vec<&'a str> = Vec::new();

    for (component, at) in components {
        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if kept.last().is_some_and(|last| *last != "..") {
                kept.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                kept.push("..");
            }
        } else {
            kept.push(component);
        }
    }

    Ok(kept)
}

fn join_absolute(components: &[&str]) -> String {
    if components.is_empty() {
        return "/".to_owned();
    }
    let mut result = String::new();
    for component in components {
        result.push('/');
        result.push_str(component);
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let components = path
        .split('/')
        .scan(0usize, |offset, component| {
            let at = *offset;
            *offset += component.len() + 1;
            Some((component, at))
        });
    let kept = reduce_components(components, absolute)?;
    let mut result = if absolute {
        join_absolute(&kept)
    } else if kept.is_empty() {
        ".".to_owned()
    } else {
        kept.join("/")
    };

    if directory_intent(path) && result != "/" && result != "." {
        result.push('/');
    }
    Ok(result)
}

fn validated_absolute(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(normalized
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validated_absolute(cwd)?;

    let home_components = match home {
        Some(home_path) => Some(
            validated_absolute(home_path).map_err(|_| PathError::InvalidHome)?,
        ),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde = bytes[0] == b'~';
    let mut component_start = 0usize;
    let (mut kept, absolute) = if tilde {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let base = home_components.ok_or(PathError::MissingHome)?;
        component_start = if bytes.len() > 1 { 2 } else { 1 };
        (base, true)
    } else if bytes[0] == b'/' {
        (Vec::new(), true)
    } else {
        (cwd_components, true)
    };

    let tail = &path[component_start..];
    let components = tail.split('/').scan(component_start, |offset, component| {
        let at = *offset;
        *offset += component.len() + 1;
        Some((component, at))
    });

    for (component, at) in components {
        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if kept.pop().is_none() {
                return Err(PathError::Escape { at });
            }
        } else {
            kept.push(component.to_owned());
        }
    }

    let refs: Vec<&str> = kept.iter().map(String::as_str).collect();
    let mut result = if absolute {
        join_absolute(&refs)
    } else {
        unreachable!()
    };

    if directory_intent(path) && result != "/" {
        result.push('/');
    }
    Ok(result)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    let bytes = path.as_bytes();
    if let Some(at) = bytes.iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = bytes.iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = bytes
        .iter()
        .position(|&b| (b >= 1 && b <= 31) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str, base: usize) -> impl Iterator<Item = (&str, usize)> {
    path.split('/').scan(base, |offset, component| {
        let at = *offset;
        *offset += component.len() + 1;
        Some((component, at))
    })
}

fn intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|part| !part.is_empty())
            .last()
            .is_some_and(|part| part == "." || part == "..")
}

fn append_component(stack: &mut Vec<String>, part: &str, absolute: bool, at: usize) -> Result<(), PathError> {
    match part {
        "" | "." => Ok(()),
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
                Ok(())
            } else if absolute {
                Err(PathError::Escape { at })
            } else {
                stack.push("..".to_owned());
                Ok(())
            }
        }
        _ => {
            stack.push(part.to_owned());
            Ok(())
        }
    }
}

fn normalized_stack(path: &str) -> Result<Vec<String>, PathError> {
    validate_raw(path)?;
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let mut stack = Vec::new();
    for (part, at) in components(path, 0) {
        append_component(&mut stack, part, true, at)?;
    }
    Ok(stack)
}

fn format_absolute(stack: &[String], trailing: bool) -> String {
    let mut out = String::from("/");
    out.push_str(&stack.join("/"));
    if trailing && !stack.is_empty() {
        out.push('/');
    }
    out
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let mut stack: Vec<String> = Vec::new();
    for (part, at) in components(path, 0) {
        append_component(&mut stack, part, absolute, at)?;
    }

    if absolute {
        Ok(format_absolute(&stack, intent(path)))
    } else if stack.is_empty() {
        Ok(".".to_owned())
    } else {
        let mut out = stack.join("/");
        if intent(path) {
            out.push('/');
        }
        Ok(out)
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_stack = normalized_stack(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_stack = match home {
        Some(value) => Some(normalized_stack(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let bytes = path.as_bytes();
    let mut stack = if bytes[0] == b'~' {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home = home_stack.ok_or(PathError::MissingHome)?;
        let start = if bytes.len() > 1 { 2 } else { 1 };
        for (part, at) in components(&path[start..], start) {
            append_component(&mut stack, part, true, at)?;
        }
        stack = home;
        // Reapply the tail against the home stack.
        for (part, at) in components(&path[start..], start) {
            append_component(&mut stack, part, true, at)?;
        }
    } else if bytes[0] == b'/' {
        for (part, at) in components(path, 0) {
            append_component(&mut stack, part, true, at)?;
        }
    } else {
        stack = cwd_stack;
        for (part, at) in components(path, 0) {
            append_component(&mut stack, part, true, at)?;
        }
    }

    Ok(format_absolute(&stack, intent(path)))
}
```
