fn band_charge(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    units
        .checked_mul(unit)
        .and_then(|product| product.checked_add(start))
        .ok_or(Error::Overflow { index })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                return Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                });
            }

            let amount = band_charge(quantity, *unit, *start, 0)?;
            Ok(Charge {
                amount,
                bands: vec![Band {
                    index: 0,
                    units: quantity,
                    amount,
                }],
            })
        }
        Price::Tiers(tiers) => {
            if tiers.is_empty() {
                return Err(Error::Empty);
            }
            if tiers.len() > 64 {
                return Err(Error::TooMany);
            }

            let mut previous = 0u64;
            for (index, tier) in tiers.iter().enumerate() {
                match tier.up_to {
                    Some(bound) => {
                        if bound <= previous {
                            return Err(Error::Bound { index });
                        }
                        previous = bound;
                    }
                    None => {
                        if index != tiers.len() - 1 {
                            return Err(Error::Unbounded { index });
                        }
                    }
                }
            }

            if tiers.last().and_then(|tier| tier.up_to).is_some() {
                return Err(Error::MissingFinal);
            }

            let mut remaining = quantity;
            let mut previous_bound = 0u64;
            let mut total = 0u64;
            let mut bands = Vec::new();

            for (index, tier) in tiers.iter().enumerate() {
                let units = match tier.up_to {
                    Some(bound) => {
                        let width = bound - previous_bound;
                        previous_bound = bound;
                        remaining.min(width)
                    }
                    None => remaining,
                };

                    let amount = band_charge(units, tier.unit, tier.start, index)?;
                    total = total
                        .checked_add(amount)
                        .ok_or(Error::Overflow { index })?;
                    bands.push(Band {
                        index,
                        units,
                        amount,
                    });
                    remaining -= units;
                }

                if remaining == 0 {
                    break;
                }
            }

            Ok(Charge {
                amount: total,
                bands,
            })
        }
    }

            if quantity == 0 {
                (0, Vec::new())
            } else {
                let amount = quantity
                    .checked_mul(*unit)

                    vec![Band {
                        index: 0,

                        amount,

                )
            }
        }

            match tiers.len() {
                0 => return Err(Error::Empty),
                65.. => return Err(Error::TooMany),
                _ => {}
            }

            let mut last_finite = 0u64;
            for (index, tier) in tiers.iter().enumerate() {
                if let Some(bound) = tier.up_to {
                    if bound <= last_finite {
                        return Err(Error::Bound { index });
                    }
                    last_finite = bound;
                } else if index != tiers.len() - 1 {
                    return Err(Error::Unbounded { index });
                }
            }

            if tiers[tiers.len() - 1].up_to.is_some() {
                return Err(Error::MissingFinal);
            }

            let mut remaining = quantity;
            let mut lower = 0u64;
            let mut sum = 0u64;
            let mut result_bands = Vec::new();

            for (index, tier) in tiers.iter().enumerate() {
                let units = if remaining == 0 {
                    0
                } else if let Some(bound) = tier.up_to {
                    let capacity = bound - lower;
                    lower = bound;
                    remaining.min(capacity)
                } else {
                    remaining
                };

                if units == 0 {
                    continue;
                }
                remaining -= units;

                let charge = units
                    .checked_mul(tier.unit)
                    .and_then(|value| value.checked_add(tier.start))
                    .ok_or(Error::Overflow { index })?;

                    .checked_add(charge)

                    index,
                    units,
                    amount: charge,
                });
            }

        }
    };

    Ok(Charge {
        amount: total,
        bands,
    })

    index: usize,
    units: u64,
    unit: u64,
    start: u64,
) -> Result<(), Error> {
    if units == 0 {
        return Ok(());
    }
    let amount = checked_band_amount(units, unit, start, index)?;
    *total = total.checked_add(amount).ok_or(Error::Overflow { index })?;
    bands.push(Band {
        index,
        units,
        amount,
    });
    Ok(())
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    let mut total = 0u64;
    let mut bands = Vec::new();

    match price {
        Price::Flat { unit, start } => {
            add_band(&mut bands, &mut total, 0, quantity, *unit, *start)?;
        }
        Price::Tiers(tiers) => {
            let count = tiers.len();
            if count == 0 {
                return Err(Error::Empty);
            }
            if count > 64 {
                return Err(Error::TooMany);
            }

            let mut finite = 0u64;
            for (i, tier) in tiers.iter().enumerate() {
                if let Some(bound) = tier.up_to {
                    if bound <= finite {
                        return Err(Error::Bound { index: i });
                    }
                    finite = bound;
                } else if i != count - 1 {
                    return Err(Error::Unbounded { index: i });
                }
            }
            if tiers[count - 1].up_to.is_some() {
                return Err(Error::MissingFinal);
            }

            let mut remaining = quantity;
            let mut lower = 0u64;
            for (i, tier) in tiers.iter().enumerate() {
                let allocated = if let Some(bound) = tier.up_to {
                    let width = bound - lower;
                    lower = bound;
                    remaining.min(width)
                } else {
                    remaining
                };
                remaining -= allocated;
                add_band(
                    &mut bands,
                    &mut total,
                    i,
                    allocated,
                    tier.unit,
                    tier.start,
                )?;
            }
        }
    }

    Ok(Charge {
        amount: total,
        bands,
    })
}
