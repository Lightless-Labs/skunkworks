#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn band_charge(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    units
        .checked_mul(unit)
        .and_then(|product| product.checked_add(start))
        .ok_or(Error::Overflow { index })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                return Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                });
            }

            let amount = band_charge(quantity, *unit, *start, 0)?;
            Ok(Charge {
                amount,
                bands: vec![Band {
                    index: 0,
                    units: quantity,
                    amount,
                }],
            })
        }
        Price::Tiers(tiers) => {
            if tiers.is_empty() {
                return Err(Error::Empty);
            }
            if tiers.len() > 64 {
                return Err(Error::TooMany);
            }

            let mut previous = 0u64;
            for (index, tier) in tiers.iter().enumerate() {
                match tier.up_to {
                    Some(bound) => {
                        if bound <= previous {
                            return Err(Error::Bound { index });
                        }
                        previous = bound;
                    }
                    None => {
                        if index != tiers.len() - 1 {
                            return Err(Error::Unbounded { index });
                        }
                    }
                }
            }

            if tiers.last().and_then(|tier| tier.up_to).is_some() {
                return Err(Error::MissingFinal);
            }

            let mut remaining = quantity;
            let mut previous_bound = 0u64;
            let mut total = 0u64;
            let mut bands = Vec::new();

            for (index, tier) in tiers.iter().enumerate() {
                let units = match tier.up_to {
                    Some(bound) => {
                        let width = bound - previous_bound;
                        previous_bound = bound;
                        remaining.min(width)
                    }
                    None => remaining,
                };

                if units > 0 {
                    let amount = band_charge(units, tier.unit, tier.start, index)?;
                    total = total
                        .checked_add(amount)
                        .ok_or(Error::Overflow { index })?;
                    bands.push(Band {
                        index,
                        units,
                        amount,
                    });
                    remaining -= units;
                }

                if remaining == 0 {
                    break;
                }
            }

            Ok(Charge {
                amount: total,
                bands,
            })
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn checked_band(units: u64, tier: &Tier, index: usize) -> Result<u64, Error> {
    match units.checked_mul(tier.unit) {
        Some(product) => product
            .checked_add(tier.start)
            .ok_or(Error::Overflow { index }),
        None => Err(Error::Overflow { index }),
    }
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    let tiers = match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                return Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                });
            }
            let amount = quantity
                .checked_mul(*unit)
                .and_then(|product| product.checked_add(*start))
                .ok_or(Error::Overflow { index: 0 })?;
            return Ok(Charge {
                amount,
                bands: vec![Band {
                    index: 0,
                    units: quantity,
                    amount,
                }],
            });
        }
        Price::Tiers(tiers) => tiers,
    };

    match tiers.len() {
        0 => return Err(Error::Empty),
        1..=64 => {}
        _ => return Err(Error::TooMany),
    }

    let mut previous_bound = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        if let Some(bound) = tier.up_to {
            if bound <= previous_bound {
                return Err(Error::Bound { index });
            }
            previous_bound = bound;
        } else if index + 1 != tiers.len() {
            return Err(Error::Unbounded { index });
        }
    }

    if tiers[tiers.len() - 1].up_to.is_some() {
        return Err(Error::MissingFinal);
    }

    let mut bands = Vec::new();
    let mut total = 0u64;
    let mut previous_bound = 0u64;

    for (index, tier) in tiers.iter().enumerate() {
        let available = match tier.up_to {
            Some(bound) => {
                let width = bound - previous_bound;
                previous_bound = bound;
                width
            }
            None => quantity,
        };

        let already_allocated: u64 = bands.iter().map(|band| band.units).sum();
        let units = available.min(quantity - already_allocated);

        if units != 0 {
            let amount = checked_band(units, tier, index)?;
            total = total
                .checked_add(amount)
                .ok_or(Error::Overflow { index })?;
            bands.push(Band {
                index,
                units,
                amount,
            });
        }

        if bands.iter().map(|band| band.units).sum::<u64>() == quantity {
            break;
        }
    }

    Ok(Charge {
        amount: total,
        bands,
    })
}
