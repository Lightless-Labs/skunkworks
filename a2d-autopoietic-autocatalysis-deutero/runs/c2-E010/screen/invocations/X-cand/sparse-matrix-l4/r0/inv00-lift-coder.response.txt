- Constructor dimension checks run before all other checks; COO validates the full input before summing.
- COO duplicate sums use original input order, start at positive zero, and report overflow at the index being added.
- CSR validates endpoints before pointer monotonicity, then checks each row’s slots in column, order, finite-value, zero order.
- Text parsing checks the final LF first, then header and counts before records; record lexical checks precede semantic checks, which precede coordinate order.
- Empty dimensions preserve shape and allocate only the required row pointer; all index and overflow checks still apply.
- I see no meaningful ambiguity in the specified behavior, so Solution B uses an independent COO-backed implementation for construction and a separate CSR conversion path.

### Solution A

            if !e.value.is_finite() {
                return Err(Error::NonFinite { entry: i });

        for (i, e) in entries.iter().enumerate() {
            let sum = sums.entry((e.row, e.col)).or_insert(0.0);
            *sum = *sum + e.value;
            if !sum.is_finite() {
                return Err(Error::SumOverflow { entry: i });
            }
        }

        let canonical = sums.into_iter()
            .filter_map(|((row, col), value)| {
                if value == 0.0 { None } else { Some(Entry { row, col, value }) }

        Ok(Self { rows, cols, entries: canonical })

    pub fn from_parts(
        rows: usize,
        cols: usize,
        row_ptr: Vec<usize>,
        col_idx: Vec<usize>,
        values: Vec<f64>,
    ) -> Result<Self, Error> {
        check_shape(rows, cols)?;
        if row_ptr.len() != rows + 1 {
            return Err(Error::RowPtrLength { expected: rows + 1, actual: row_ptr.len() });
        }
        if col_idx.len() != values.len() {
            return Err(Error::ColumnValueLength { columns: col_idx.len(), values: values.len() });
        }
        if row_ptr[0] != 0 {
            return Err(Error::RowPtrStart { found: row_ptr[0] });
        }
        if row_ptr[rows] != values.len() {
            return Err(Error::RowPtrEnd { found: row_ptr[rows], expected: values.len() });
        }
        for r in 0..rows {
            if row_ptr[r] > row_ptr[r + 1] {
                return Err(Error::RowPtrOrder { row: r });
            }
        }
        for r in 0..rows {
            let start = row_ptr[r];
            let end = row_ptr[r + 1];
            let mut previous: Option<usize> = None;
            for slot in start..end {
                let column = col_idx[slot];
                if column >= cols {
                    return Err(Error::ColumnBounds { slot, column, limit: cols });
                }
                if previous.is_some_and(|p| column <= p) {
                    return Err(Error::ColumnOrder { slot });
                }
                let value = values[slot];
                if !value.is_finite() {
                    return Err(Error::NonFiniteValue { slot });
                }
                if value == 0.0 {
                    return Err(Error::ExplicitZero { slot });
                }
                previous = Some(column);
            }
        }
        Ok(Self { rows, cols, row_ptr, col_idx, values })
    }

    pub fn row_ptr(&self) -> &[usize] {

    pub fn to_coo(&self) -> Sparse {

        for r in 0..self.rows {
            for slot in self.row_ptr[r]..self.row_ptr[r + 1] {
                entries.push(Entry { row: r, col: self.col_idx[slot], value: self.values[slot] });
            }
        }
        Sparse { rows: self.rows, cols: self.cols, entries }
    }

    pub fn mul_vector(&self, x: &[f64]) -> Result<Vec<f64>, Error> {
        if x.len() != self.cols {
            return Err(Error::VectorLength { expected: self.cols, actual: x.len() });
        }
        for (index, value) in x.iter().enumerate() {
            if !value.is_finite() {
                return Err(Error::NonFiniteVector { index });
            }
        }
        let mut result = Vec::with_capacity(self.rows);
        for row in 0..self.rows {
            let mut acc = 0.0;
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let column = self.col_idx[slot];
                let product = self.values[slot] * x[column];
                if !product.is_finite() {
                    return Err(Error::ProductOverflow { row, column });
                }
                acc = acc + product;
                if !acc.is_finite() {
                    return Err(Error::AccumulationOverflow { row, column });
                }
            }
            if acc == 0.0 {
                acc = 0.0;
            }
            result.push(acc);
        }
        Ok(result)
    }

    pub fn transpose(&self) -> Csr {
        let mut counts = vec![0usize; self.cols];
        for &column in &self.col_idx {
            counts[column] += 1;
        }
        let mut row_ptr = vec![0usize; self.cols + 1];
        for r in 0..self.cols {
            row_ptr[r + 1] = row_ptr[r] + counts[r];
        }
        let mut next = row_ptr[..self.cols].to_vec();
        let mut col_idx = vec![0usize; self.values.len()];
        let mut values = vec![0.0; self.values.len()];
        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let column = self.col_idx[slot];
                let target = next[column];
                col_idx[target] = row;
                values[target] = self.values[slot];
                next[column] += 1;
            }
        }
        Csr { rows: self.cols, cols: self.rows, row_ptr, col_idx, values }
    }

    pub fn multiply(&self, rhs: &Csr) -> Result<Csr, Error> {
        if self.cols != rhs.rows {
            return Err(Error::DimensionMismatch { left_cols: self.cols, right_rows: rhs.rows });
        }
        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);

        for row in 0..self.rows {
            let mut accumulators: BTreeMap<usize, f64> = BTreeMap::new();
            for left_slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let inner = self.col_idx[left_slot];
                let left_value = self.values[left_slot];
                for right_slot in rhs.row_ptr[inner]..rhs.row_ptr[inner + 1] {
                    let column = rhs.col_idx[right_slot];
                    let product = left_value * rhs.values[right_slot];
                    if !product.is_finite() {
                        return Err(Error::MultiplyOverflow { row, inner, column });
                    }
                    let sum = accumulators.entry(column).or_insert(0.0);
                    *sum = *sum + product;
                    if !sum.is_finite() {
                        return Err(Error::MultiplySumOverflow { row, inner, column });
                    }
                }
            }
            for (column, value) in accumulators {
                if value != 0.0 {
                    col_idx.push(column);
                    values.push(value);
                }
            }
            row_ptr.push(values.len());
        }

        Ok(Csr { rows: self.rows, cols: rhs.cols, row_ptr, col_idx, values })
    }

    pub fn prune(&self, tolerance: f64) -> Result<Csr, Error> {
        if !tolerance.is_finite() || tolerance < 0.0 {
            return Err(Error::Tolerance);
        }
        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);
        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let value = self.values[slot];
                if value.abs() > tolerance {
                    col_idx.push(self.col_idx[slot]);
                    values.push(value);
                }
            }
            row_ptr.push(values.len());
        }
        Ok(Csr { rows: self.rows, cols: self.cols, row_ptr, col_idx, values })
    }

    pub fn write_text(&self) -> String {
        let mut out = format!("SM1 {} {} {}\n", self.rows, self.cols, self.values.len());
        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                out.push_str(&format!("{} {} {:016x}\n", row, self.col_idx[slot], self.values[slot].to_bits()));
            }
        }
        out
    }

    pub fn parse_text(text: &str) -> Result<Csr, Error> {
        let fail = |line, kind| Error::Text { line, kind };
        if !text.ends_with('\n') {
            let line = 1 + text.bytes().filter(|&b| b == b'\n').count();
            return Err(fail(line, TextError::Newline));
        }

        let body = &text[..text.len() - 1];
        let lines: Vec<&str> = body.split('\n').collect();
        let header = lines[0];
        let fields: Vec<&str> = header.split(' ').collect();
        if fields.len() != 4 || fields.iter().any(|s| s.is_empty()) || fields[0] != "SM1" {
            return Err(fail(1, TextError::Header));
        }

        fn decimal(s: &str) -> Option<usize> {
            if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
                return None;
            }
            if s.len() > 1 && s.as_bytes()[0] == b'0' {
                return None;
            }
            s.parse::<usize>().ok()
        }

        let rows = decimal(fields[1]).ok_or_else(|| fail(1, TextError::Decimal))?;
        let cols = decimal(fields[2]).ok_or_else(|| fail(1, TextError::Decimal))?;
        let nnz = decimal(fields[3]).ok_or_else(|| fail(1, TextError::Decimal))?;
        if rows > MAX_DIM || cols > MAX_DIM {
            return Err(fail(1, TextError::Dimension));
        }
        if nnz != lines.len() - 1 {
            return Err(fail(1, TextError::Count));
        }

        let mut row_ptr = vec![0usize; rows + 1];
        let mut col_idx = Vec::with_capacity(nnz);
        let mut values = Vec::with_capacity(nnz);
        let mut previous: Option<(usize, usize)> = None;

        for (i, line) in lines.iter().enumerate().skip(1) {
            let line_no = i + 1;
            let fields: Vec<&str> = line.split(' ').collect();
            if fields.len() != 3 || fields.iter().any(|s| s.is_empty()) {
                return Err(fail(line_no, TextError::Record));
            }
            let row = decimal(fields[0]).ok_or_else(|| fail(line_no, TextError::Decimal))?;
            let col = decimal(fields[1]).ok_or_else(|| fail(line_no, TextError::Decimal))?;
            let hex = fields[2].as_bytes();
            if hex.len() != 16 || !hex.iter().all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(b)) {
                return Err(fail(line_no, TextError::Hex));
            }
            let bits = u64::from_str_radix(fields[2], 16).map_err(|_| fail(line_no, TextError::Hex))?;
            let value = f64::from_bits(bits);

            if row >= rows {
                return Err(fail(line_no, TextError::RowBounds));
            }
            if col >= cols {
                return Err(fail(line_no, TextError::ColumnBounds));
            }
            if !value.is_finite() {
                return Err(fail(line_no, TextError::NonFinite));
            }
            if value == 0.0 {
                return Err(fail(line_no, TextError::Zero));
            }
            if previous.is_some_and(|p| (row, col) <= p) {
                return Err(fail(line_no, TextError::Order));
            }
            previous = Some((row, col));
            row_ptr[row + 1] += 1;
            col_idx.push(col);
            values.push(value);
        }

        for r in 0..rows {
            row_ptr[r + 1] += row_ptr[r];
        }
        Ok(Csr { rows, cols, row_ptr, col_idx, values })
    }
}

    pub fn transpose(&self) -> Csr {
        let mut row_ptr = vec![0usize; self.cols + 1];
        for &col in &self.col_idx {
            row_ptr[col + 1] += 1;
        }
        for row in 0..self.cols {
            row_ptr[row + 1] += row_ptr[row];
        }
        let mut positions = row_ptr[..self.cols].to_vec();
        let mut col_idx = vec![0usize; self.values.len()];
        let mut values = vec![0.0; self.values.len()];

        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {

                col_idx[target] = row;

            }
        }

        Csr { rows: self.cols, cols: self.rows, row_ptr, col_idx, values }
    }

    pub fn multiply(&self, rhs: &Csr) -> Result<Csr, Error> {
        if self.cols != rhs.rows {
            return Err(Error::DimensionMismatch { left_cols: self.cols, right_rows: rhs.rows });
        }

        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);

        for row in 0..self.rows {
            let mut sums = BTreeMap::<usize, f64>::new();
            for left_slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let inner = self.col_idx[left_slot];
                for right_slot in rhs.row_ptr[inner]..rhs.row_ptr[inner + 1] {
                    let col = rhs.col_idx[right_slot];
                    let product = self.values[left_slot] * rhs.values[right_slot];
                    if !product.is_finite() {
                        return Err(Error::MultiplyOverflow { row, inner, column: col });
                    }
                    let acc = sums.entry(col).or_insert(0.0);
                    let next = *acc + product;
                    if !next.is_finite() {
                        return Err(Error::MultiplySumOverflow { row, inner, column: col });
                    }
                    *acc = next;
                }
            }
            for (col, value) in sums {
                if value != 0.0 {
                    col_idx.push(col);
                    values.push(value);
                }
            }
            row_ptr.push(values.len());
        }

        Ok(Csr { rows: self.rows, cols: rhs.cols, row_ptr, col_idx, values })
    }

    pub fn prune(&self, tolerance: f64) -> Result<Csr, Error> {
        if !tolerance.is_finite() || tolerance < 0.0 {
            return Err(Error::Tolerance);
        }

        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);
        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let value = self.values[slot];
                if value.abs() > tolerance {
                    col_idx.push(self.col_idx[slot]);
                    values.push(value);
                }
            }
            row_ptr.push(values.len());
        }
        Ok(Csr { rows: self.rows, cols: self.cols, row_ptr, col_idx, values })
    }

    pub fn write_text(&self) -> String {
        let mut text = format!("SM1 {} {} {}\n", self.rows, self.cols, self.values.len());
        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                text.push_str(&format!("{row} {} {:016x}\n", self.col_idx[slot], self.values[slot].to_bits()));
            }
        }
        text
    }

    pub fn parse_text(text: &str) -> Result<Csr, Error> {
        let text_error = |line, kind| Error::Text { line, kind };
        if !text.ends_with('\n') {
            return Err(text_error(1 + text.bytes().filter(|&b| b == b'\n').count(), TextError::Newline));
        }

        let lines: Vec<&str> = text[..text.len() - 1].split('\n').collect();
        let header_fields: Vec<&str> = lines[0].split(' ').collect();
        if header_fields.len() != 4
            || header_fields.iter().any(|field| field.is_empty())
            || header_fields[0] != "SM1"
        {
            return Err(text_error(1, TextError::Header));
        }

        fn parse_decimal(token: &str) -> Option<usize> {
            if token.is_empty() || !token.bytes().all(|b| b.is_ascii_digit()) {
                return None;
            }
            if token.len() > 1 && token.as_bytes()[0] == b'0' {
                return None;
            }
            token.parse().ok()
        }

        let rows = parse_decimal(header_fields[1]).ok_or_else(|| text_error(1, TextError::Decimal))?;
        let cols = parse_decimal(header_fields[2]).ok_or_else(|| text_error(1, TextError::Decimal))?;
        let nnz = parse_decimal(header_fields[3]).ok_or_else(|| text_error(1, TextError::Decimal))?;
        if rows > MAX_DIM || cols > MAX_DIM {
            return Err(text_error(1, TextError::Dimension));
        }
        if nnz != lines.len() - 1 {
            return Err(text_error(1, TextError::Count));
        }

        let mut row_ptr = vec![0usize; rows + 1];
        let mut col_idx = Vec::with_capacity(nnz);
        let mut values = Vec::with_capacity(nnz);
        let mut previous = None;

        for (offset, line) in lines.iter().enumerate().skip(1) {
            let line_no = offset + 1;
            let fields: Vec<&str> = line.split(' ').collect();
            if fields.len() != 3 || fields.iter().any(|field| field.is_empty()) {
                return Err(text_error(line_no, TextError::Record));
            }

            let row = parse_decimal(fields[0]).ok_or_else(|| text_error(line_no, TextError::Decimal))?;
            let col = parse_decimal(fields[1]).ok_or_else(|| text_error(line_no, TextError::Decimal))?;
            let hex = fields[2].as_bytes();
            if hex.len() != 16
                || !hex.iter().all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(b))
            {
                return Err(text_error(line_no, TextError::Hex));
            }
            let bits = u64::from_str_radix(fields[2], 16).map_err(|_| text_error(line_no, TextError::Hex))?;
            let value = f64::from_bits(bits);

            if row >= rows {
                return Err(text_error(line_no, TextError::RowBounds));
            }
            if col >= cols {
                return Err(text_error(line_no, TextError::ColumnBounds));
            }
            if !value.is_finite() {
                return Err(text_error(line_no, TextError::NonFinite));
            }
            if value == 0.0 {
                return Err(text_error(line_no, TextError::Zero));
            }
            if previous.map_or(false, |p| (row, col) <= p) {
                return Err(text_error(line_no, TextError::Order));
            }
            previous = Some((row, col));
            row_ptr[row + 1] += 1;
            col_idx.push(col);
            values.push(value);
        }

        for row in 0..rows {
            row_ptr[row + 1] += row_ptr[row];
        }
        Ok(Csr { rows, cols, row_ptr, col_idx, values })
    }
}
