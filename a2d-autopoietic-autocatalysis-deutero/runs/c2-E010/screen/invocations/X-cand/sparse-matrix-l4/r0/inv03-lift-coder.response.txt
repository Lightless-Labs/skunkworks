- Shape checks run before all other constructor checks; exactly `MAX_DIM` is allowed.
- COO validates every input entry before summing, then adds in original order, including zero contributions.
- CSR validates endpoints before pointer monotonicity, then checks each slot in the specified order.
- Text parsing checks the final LF before any content, and count before inspecting records.
- Multiplication follows the required nested slot visit order and reports the first nonfinite product or sum.
- For Solution B, I use a different COO accumulation structure; the validation and arithmetic order remain the same.

### Solution A

```rust
use std::collections::BTreeMap;

pub const MAX_DIM: usize = 4096;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Axis { Row, Column }

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Entry { pub row: usize, pub col: usize, pub value: f64 }

#[derive(Clone, Debug, PartialEq)]
pub struct Sparse { rows: usize, cols: usize, entries: Vec<Entry> }

#[derive(Clone, Debug, PartialEq)]
pub struct Csr { rows: usize, cols: usize, row_ptr: Vec<usize>, col_idx: Vec<usize>, values: Vec<f64> }

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum TextError { Newline, Header, Decimal, Dimension, Count, Record, Hex, RowBounds, ColumnBounds, NonFinite, Zero, Order }

#[derive(Clone, Debug, PartialEq)]
pub enum Error {
    Shape { axis: Axis, size: usize },
    Bounds { entry: usize, axis: Axis, index: usize, limit: usize },
    NonFinite { entry: usize },
    SumOverflow { entry: usize },
    RowPtrLength { expected: usize, actual: usize },
    ColumnValueLength { columns: usize, values: usize },
    RowPtrStart { found: usize },
    RowPtrEnd { found: usize, expected: usize },
    RowPtrOrder { row: usize },
    ColumnBounds { slot: usize, column: usize, limit: usize },
    ColumnOrder { slot: usize },
    NonFiniteValue { slot: usize },
    ExplicitZero { slot: usize },
    VectorLength { expected: usize, actual: usize },
    NonFiniteVector { index: usize },
    ProductOverflow { row: usize, column: usize },
    AccumulationOverflow { row: usize, column: usize },
    DimensionMismatch { left_cols: usize, right_rows: usize },
    MultiplyOverflow { row: usize, inner: usize, column: usize },
    MultiplySumOverflow { row: usize, inner: usize, column: usize },
    Tolerance,
    Text { line: usize, kind: TextError },
}

fn check_shape(rows: usize, cols: usize) -> Result<(), Error> {
    if rows > MAX_DIM { return Err(Error::Shape { axis: Axis::Row, size: rows }); }
    if cols > MAX_DIM { return Err(Error::Shape { axis: Axis::Column, size: cols }); }
    Ok(())
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        check_shape(rows, cols)?;
        for (i, e) in entries.iter().enumerate() {
            if e.row >= rows { return Err(Error::Bounds { entry: i, axis: Axis::Row, index: e.row, limit: rows }); }
            if e.col >= cols { return Err(Error::Bounds { entry: i, axis: Axis::Column, index: e.col, limit: cols }); }
            if !e.value.is_finite() { return Err(Error::NonFinite { entry: i }); }
        }
        let mut sums: BTreeMap<(usize, usize), f64> = BTreeMap::new();
        for (i, e) in entries.iter().enumerate() {
            let s = sums.entry((e.row, e.col)).or_insert(0.0);
            *s = *s + e.value;
            if !s.is_finite() { return Err(Error::SumOverflow { entry: i }); }
        }
        let canonical = sums.into_iter().filter_map(|((row,col),value)| {
            if value == 0.0 { None } else { Some(Entry { row, col, value }) }
        }).collect();
        Ok(Self { rows, cols, entries: canonical })
    }

    pub fn shape(&self) -> (usize, usize) { (self.rows, self.cols) }
    pub fn entries(&self) -> &[Entry] { &self.entries }

    pub fn to_csr(&self) -> Csr {
        let mut row_ptr = vec![0; self.rows + 1];
        for e in &self.entries { row_ptr[e.row + 1] += 1; }
        for r in 0..self.rows { row_ptr[r + 1] += row_ptr[r]; }
        let col_idx = self.entries.iter().map(|e| e.col).collect();
        let values = self.entries.iter().map(|e| e.value).collect();
        Csr { rows: self.rows, cols: self.cols, row_ptr, col_idx, values }
    }
}

impl Csr {
    pub fn from_parts(rows: usize, cols: usize, row_ptr: Vec<usize>, col_idx: Vec<usize>, values: Vec<f64>) -> Result<Self, Error> {
        check_shape(rows, cols)?;
        if row_ptr.len() != rows + 1 { return Err(Error::RowPtrLength { expected: rows + 1, actual: row_ptr.len() }); }
        if col_idx.len() != values.len() { return Err(Error::ColumnValueLength { columns: col_idx.len(), values: values.len() }); }
        if row_ptr[0] != 0 { return Err(Error::RowPtrStart { found: row_ptr[0] }); }
        if row_ptr[rows] != values.len() { return Err(Error::RowPtrEnd { found: row_ptr[rows], expected: values.len() }); }
        for r in 0..rows {
            if row_ptr[r] > row_ptr[r + 1] { return Err(Error::RowPtrOrder { row: r }); }
        }
        for r in 0..rows {
            let mut previous = None;
            for s in row_ptr[r]..row_ptr[r + 1] {
                let c = col_idx[s];
                if c >= cols { return Err(Error::ColumnBounds { slot: s, column: c, limit: cols }); }
                if let Some(p) = previous { if c <= p { return Err(Error::ColumnOrder { slot: s }); } }
                if !values[s].is_finite() { return Err(Error::NonFiniteValue { slot: s }); }
                if values[s] == 0.0 { return Err(Error::ExplicitZero { slot: s }); }
                previous = Some(c);
            }
        }
        Ok(Self { rows, cols, row_ptr, col_idx, values })
    }

    pub fn shape(&self) -> (usize, usize) { (self.rows, self.cols) }
    pub fn row_ptr(&self) -> &[usize] { &self.row_ptr }
    pub fn col_indices(&self) -> &[usize] { &self.col_idx }
    pub fn values(&self) -> &[f64] { &self.values }

    pub fn to_coo(&self) -> Sparse {
        let mut entries = Vec::with_capacity(self.values.len());
        for r in 0..self.rows {
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                entries.push(Entry { row: r, col: self.col_idx[s], value: self.values[s] });
            }
        }
        Sparse { rows: self.rows, cols: self.cols, entries }
    }

    pub fn mul_vector(&self, x: &[f64]) -> Result<Vec<f64>, Error> {
        if x.len() != self.cols { return Err(Error::VectorLength { expected: self.cols, actual: x.len() }); }
        for (i, v) in x.iter().enumerate() {
            if !v.is_finite() { return Err(Error::NonFiniteVector { index: i }); }
        }
        let mut out = Vec::with_capacity(self.rows);
        for r in 0..self.rows {
            let mut acc = 0.0;
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                let c = self.col_idx[s];
                let p = self.values[s] * x[c];
                if !p.is_finite() { return Err(Error::ProductOverflow { row: r, column: c }); }
                acc = acc + p;
                if !acc.is_finite() { return Err(Error::AccumulationOverflow { row: r, column: c }); }
            }
            out.push(if acc == 0.0 { 0.0 } else { acc });
        }
        Ok(out)
    }

    pub fn transpose(&self) -> Csr {
        let mut counts = vec![0usize; self.cols];
        for &c in &self.col_idx { counts[c] += 1; }
        let mut row_ptr = vec![0usize; self.cols + 1];
        for r in 0..self.cols { row_ptr[r + 1] = row_ptr[r] + counts[r]; }
        let mut next = row_ptr[..self.cols].to_vec();
        let mut col_idx = vec![0usize; self.values.len()];
        let mut values = vec![0.0; self.values.len()];
        for r in 0..self.rows {
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                let tr = self.col_idx[s];
                let dst = next[tr];
                next[tr] += 1;
                col_idx[dst] = r;
                values[dst] = self.values[s];
            }
        }
        Csr { rows: self.cols, cols: self.rows, row_ptr, col_idx, values }
    }

    pub fn multiply(&self, rhs: &Csr) -> Result<Csr, Error> {
        if self.cols != rhs.rows {
            return Err(Error::DimensionMismatch { left_cols: self.cols, right_rows: rhs.rows });
        }
        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);
        for r in 0..self.rows {
            let mut sums: BTreeMap<usize, f64> = BTreeMap::new();
            for ls in self.row_ptr[r]..self.row_ptr[r + 1] {
                let k = self.col_idx[ls];
                for rs in rhs.row_ptr[k]..rhs.row_ptr[k + 1] {
                    let c = rhs.col_idx[rs];
                    let p = self.values[ls] * rhs.values[rs];
                    if !p.is_finite() { return Err(Error::MultiplyOverflow { row: r, inner: k, column: c }); }
                    let a = sums.entry(c).or_insert(0.0);
                    *a = *a + p;
                    if !a.is_finite() { return Err(Error::MultiplySumOverflow { row: r, inner: k, column: c }); }
                }
            }
            for (c, v) in sums {
                if v != 0.0 { col_idx.push(c); values.push(v); }
            }
            row_ptr.push(values.len());
        }
        Ok(Csr { rows: self.rows, cols: rhs.cols, row_ptr, col_idx, values })
    }

    pub fn prune(&self, tolerance: f64) -> Result<Csr, Error> {
        if !tolerance.is_finite() || tolerance < 0.0 { return Err(Error::Tolerance); }
        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);
        for r in 0..self.rows {
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                if self.values[s].abs() > tolerance {
                    col_idx.push(self.col_idx[s]);
                    values.push(self.values[s]);
                }
            }
            row_ptr.push(values.len());
        }
        Ok(Csr { rows: self.rows, cols: self.cols, row_ptr, col_idx, values })
    }

    pub fn write_text(&self) -> String {
        let mut out = format!("SM1 {} {} {}\n", self.rows, self.cols, self.values.len());
        for r in 0..self.rows {
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                out.push_str(&format!("{} {} {:016x}\n", r, self.col_idx[s], self.values[s].to_bits()));
            }
        }
        out
    }

    pub fn parse_text(text: &str) -> Result<Csr, Error> {
        fn err(line: usize, kind: TextError) -> Error { Error::Text { line, kind } }
        fn decimal(s: &str) -> Option<usize> {
            if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) { return None; }
            if s.len() > 1 && s.as_bytes()[0] == b'0' { return None; }
            s.parse::<usize>().ok()
        }
        if !text.ends_with('\n') {
            return Err(err(1 + text.bytes().filter(|&b| b == b'\n').count(), TextError::Newline));
        }
        let body = &text[..text.len() - 1];
        let lines: Vec<&str> = body.split('\n').collect();
        let header: Vec<&str> = lines[0].split(' ').collect();
        if header.len() != 4 || header.iter().any(|s| s.is_empty()) || header[0] != "SM1" {
            return Err(err(1, TextError::Header));
        }
        let rows = decimal(header[1]).ok_or_else(|| err(1, TextError::Decimal))?;
        let cols = decimal(header[2]).ok_or_else(|| err(1, TextError::Decimal))?;
        let nnz = decimal(header[3]).ok_or_else(|| err(1, TextError::Decimal))?;
        if rows > MAX_DIM || cols > MAX_DIM { return Err(err(1, TextError::Dimension)); }
        if nnz != lines.len() - 1 { return Err(err(1, TextError::Count)); }
        let mut row_ptr = vec![0usize; rows + 1];
        let mut col_idx = Vec::with_capacity(nnz);
        let mut values = Vec::with_capacity(nnz);
        let mut prev: Option<(usize,usize)> = None;
        let mut current_row = 0usize;
        for (i, line) in lines.iter().enumerate().skip(1) {
            let ln = i + 1;
            let fields: Vec<&str> = line.split(' ').collect();
            if fields.len() != 3 || fields.iter().any(|s| s.is_empty()) { return Err(err(ln, TextError::Record)); }
            let row = decimal(fields[0]).ok_or_else(|| err(ln, TextError::Decimal))?;
            let col = decimal(fields[1]).ok_or_else(|| err(ln, TextError::Decimal))?;
            let h = fields[2];
            if h.len() != 16 || !h.bytes().all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b)) {
                return Err(err(ln, TextError::Hex));
            }
            let bits = u64::from_str_radix(h, 16).unwrap();
            let value = f64::from_bits(bits);
            if row >= rows { return Err(err(ln, TextError::RowBounds)); }
            if col >= cols { return Err(err(ln, TextError::ColumnBounds)); }
            if !value.is_finite() { return Err(err(ln, TextError::NonFinite)); }
            if value == 0.0 { return Err(err(ln, TextError::Zero)); }
            if let Some(p) = prev { if (row,col) <= p { return Err(err(ln, TextError::Order)); } }
            while current_row < row { row_ptr[current_row + 1] = col_idx.len(); current_row += 1; }
            col_idx.push(col);
            values.push(value);
            prev = Some((row,col));
        }
        while current_row < rows { row_ptr[current_row + 1] = col_idx.len(); current_row += 1; }
        Ok(Csr { rows, cols, row_ptr, col_idx, values })
    }
}
```

```rust
        let lines:Vec<&str>=text[..text.len()-1].split('\n').collect();
        let h:Vec<&str>=lines[0].split(' ').collect();
        if h.len()!=4||h.iter().any(|x|x.is_empty())||h[0]!="SM1"{return Err(e(1,TextError::Header))}
        let rows=dec(h[1]).ok_or_else(||e(1,TextError::Decimal))?;
        let cols=dec(h[2]).ok_or_else(||e(1,TextError::Decimal))?;
        let n=dec(h[3]).ok_or_else(||e(1,TextError::Decimal))?;
        if rows>MAX_DIM||cols>MAX_DIM{return Err(e(1,TextError::Dimension))}
        if n!=lines.len()-1{return Err(e(1,TextError::Count))}
        let mut rp=vec![0usize;rows+1];let mut ci=Vec::with_capacity(n);let mut vs=Vec::with_capacity(n);
        let(mut cr,mut prev)=(0usize,None);
        for(i,line)in lines.iter().enumerate().skip(1){
            let ln=i+1;let f:Vec<&str>=line.split(' ').collect();
            if f.len()!=3||f.iter().any(|x|x.is_empty()){return Err(e(ln,TextError::Record))}
            let r=dec(f[0]).ok_or_else(||e(ln,TextError::Decimal))?;let c=dec(f[1]).ok_or_else(||e(ln,TextError::Decimal))?;
            if f[2].len()!=16||!f[2].bytes().all(|b|b.is_ascii_digit()||(b'a'..=b'f').contains(&b)){return Err(e(ln,TextError::Hex))}
            let v=f64::from_bits(u64::from_str_radix(f[2],16).unwrap());
            if r>=rows{return Err(e(ln,TextError::RowBounds))}
            if c>=cols{return Err(e(ln,TextError::ColumnBounds))}
            if !v.is_finite(){return Err(e(ln,TextError::NonFinite))}
            if v==0.0{return Err(e(ln,TextError::Zero))}
            if let Some(p)=prev{if(r,c)<=p{return Err(e(ln,TextError::Order))}}
            while cr<r{rp[cr+1]=ci.len();cr+=1}
            ci.push(c);vs.push(v);prev=Some((r,c))
        }
        while cr<rows{rp[cr+1]=ci.len();cr+=1}
        Ok(Csr{rows,cols,row_ptr:rp,col_idx:ci,values:vs})
    }
}
```
