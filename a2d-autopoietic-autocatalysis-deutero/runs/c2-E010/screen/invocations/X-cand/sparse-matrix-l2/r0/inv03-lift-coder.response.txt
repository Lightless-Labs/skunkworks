- Dimension checks happen before every other constructor check; `MAX_DIM` itself is allowed.
- COO validation completes in input order before summing; each coordinate then accumulates in input order from `+0.0`.
- CSR endpoint and pointer-order checks precede slot access, so malformed pointers cannot trigger indexing panics.
- Column ordering is checked only within each row, after column bounds and before value checks.
- Vector inputs are fully validated before multiplication; product overflow precedes accumulation overflow.
- I treat the two requested solutions as independently implemented variants with the same specified behavior.

### Solution A

```rust
pub const MAX_DIM: usize = 4096;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Axis {
    Row,
    Column,
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Entry {
    pub row: usize,
    pub col: usize,
    pub value: f64,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Sparse {
    rows: usize,
    cols: usize,
    entries: Vec<Entry>,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Csr {
    rows: usize,
    cols: usize,
    row_ptr: Vec<usize>,
    col_idx: Vec<usize>,
    values: Vec<f64>,
}

#[derive(Clone, Debug, PartialEq)]
pub enum Error {
    Shape { axis: Axis, size: usize },
    Bounds { entry: usize, axis: Axis, index: usize, limit: usize },
    NonFinite { entry: usize },
    SumOverflow { entry: usize },
    RowPtrLength { expected: usize, actual: usize },
    ColumnValueLength { columns: usize, values: usize },
    RowPtrStart { found: usize },
    RowPtrEnd { found: usize, expected: usize },
    RowPtrOrder { row: usize },
    ColumnBounds { slot: usize, column: usize, limit: usize },
    ColumnOrder { slot: usize },
    NonFiniteValue { slot: usize },
    ExplicitZero { slot: usize },
    VectorLength { expected: usize, actual: usize },
    NonFiniteVector { index: usize },
    ProductOverflow { row: usize, column: usize },
    AccumulationOverflow { row: usize, column: usize },
}

fn check_shape(rows: usize, cols: usize) -> Result<(), Error> {
    if rows > MAX_DIM {
        return Err(Error::Shape { axis: Axis::Row, size: rows });
    }
    if cols > MAX_DIM {
        return Err(Error::Shape { axis: Axis::Column, size: cols });
    }
    Ok(())
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        check_shape(rows, cols)?;

        for (i, e) in entries.iter().enumerate() {
            if e.row >= rows {
                return Err(Error::Bounds {
                    entry: i,
                    axis: Axis::Row,
                    index: e.row,
                    limit: rows,
                });
            }
            if e.col >= cols {
                return Err(Error::Bounds {
                    entry: i,
                    axis: Axis::Column,
                    index: e.col,
                    limit: cols,
                });
            }
            if !e.value.is_finite() {
                return Err(Error::NonFinite { entry: i });
            }
        }

        let mut sums: Vec<(usize, usize, f64)> = Vec::new();
        for (i, e) in entries.iter().enumerate() {
            let position = sums.iter().position(|&(r, c, _)| r == e.row && c == e.col);
            let index = match position {
                Some(index) => index,
                None => {
                    sums.push((e.row, e.col, 0.0));
                    sums.len() - 1
                }
            };
            let sum = sums[index].2 + e.value;
            if !sum.is_finite() {
                return Err(Error::SumOverflow { entry: i });
            }
            sums[index].2 = sum;
        }

        let mut canonical: Vec<Entry> = sums
            .into_iter()
            .filter_map(|(row, col, value)| {
                if value == 0.0 {
                    None
                } else {
                    Some(Entry { row, col, value })
                }
            })
            .collect();
        canonical.sort_by_key(|e| (e.row, e.col));

        Ok(Self { rows, cols, entries: canonical })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn entries(&self) -> &[Entry] {
        &self.entries
    }

    pub fn to_csr(&self) -> Csr {
        let mut row_ptr = vec![0; self.rows + 1];
        for e in &self.entries {
            row_ptr[e.row + 1] += 1;
        }
        for row in 0..self.rows {
            row_ptr[row + 1] += row_ptr[row];
        }

        Csr {
            rows: self.rows,
            cols: self.cols,
            row_ptr,
            col_idx: self.entries.iter().map(|e| e.col).collect(),
            values: self.entries.iter().map(|e| e.value).collect(),
        }
    }
}

impl Csr {
    pub fn from_parts(
        rows: usize,
        cols: usize,
        row_ptr: Vec<usize>,
        col_idx: Vec<usize>,
        values: Vec<f64>,
    ) -> Result<Self, Error> {
        check_shape(rows, cols)?;

        let expected = rows + 1;
        if row_ptr.len() != expected {
            return Err(Error::RowPtrLength { expected, actual: row_ptr.len() });
        }
        if col_idx.len() != values.len() {
            return Err(Error::ColumnValueLength {
                columns: col_idx.len(),
                values: values.len(),
            });
        }
        if row_ptr[0] != 0 {
            return Err(Error::RowPtrStart { found: row_ptr[0] });
        }
        if row_ptr[rows] != values.len() {
            return Err(Error::RowPtrEnd {
                found: row_ptr[rows],
                expected: values.len(),
            });
        }

        for row in 0..rows {
            if row_ptr[row] > row_ptr[row + 1] {
                return Err(Error::RowPtrOrder { row });
            }
        }

        for row in 0..rows {
            let start = row_ptr[row];
            let end = row_ptr[row + 1];
            let mut previous: Option<usize> = None;
            for slot in start..end {
                let column = col_idx[slot];
                if column >= cols {
                    return Err(Error::ColumnBounds { slot, column, limit: cols });
                }
                if previous.is_some_and(|p| column <= p) {
                    return Err(Error::ColumnOrder { slot });
                }
                let value = values[slot];
                if !value.is_finite() {
                    return Err(Error::NonFiniteValue { slot });
                }
                if value == 0.0 {
                    return Err(Error::ExplicitZero { slot });
                }
                previous = Some(column);
            }
        }

        Ok(Self { rows, cols, row_ptr, col_idx, values })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn row_ptr(&self) -> &[usize] {
        &self.row_ptr
    }

    pub fn col_indices(&self) -> &[usize] {
        &self.col_idx
    }

    pub fn values(&self) -> &[f64] {
        &self.values
    }

    pub fn to_coo(&self) -> Sparse {
        let mut entries = Vec::with_capacity(self.values.len());
        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                entries.push(Entry {
                    row,
                    col: self.col_idx[slot],
                    value: self.values[slot],
                });
            }
        }
        Sparse { rows: self.rows, cols: self.cols, entries }
    }

    pub fn mul_vector(&self, x: &[f64]) -> Result<Vec<f64>, Error> {
        if x.len() != self.cols {
            return Err(Error::VectorLength { expected: self.cols, actual: x.len() });
        }
        for (index, value) in x.iter().enumerate() {
            if !value.is_finite() {
                return Err(Error::NonFiniteVector { index });
            }
        }

        let mut result = Vec::with_capacity(self.rows);
        for row in 0..self.rows {
            let mut accumulator = 0.0;
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let column = self.col_idx[slot];
                let product = self.values[slot] * x[column];
                if !product.is_finite() {
                    return Err(Error::ProductOverflow { row, column });
                }
                let next = accumulator + product;
                if !next.is_finite() {
                    return Err(Error::AccumulationOverflow { row, column });
                }
                accumulator = next;
            }
            if accumulator == 0.0 {
                accumulator = 0.0;
            }
            result.push(accumulator);
        }
        Ok(result)
    }
}
```

### Solution B

```rust
pub const MAX_DIM: usize = 4096;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Axis {
    Row,
    Column,
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Entry {
    pub row: usize,
    pub col: usize,
    pub value: f64,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Sparse {
    rows: usize,
    cols: usize,
    entries: Vec<Entry>,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Csr {
    rows: usize,
    cols: usize,
    row_ptr: Vec<usize>,
    col_idx: Vec<usize>,
    values: Vec<f64>,
}

#[derive(Clone, Debug, PartialEq)]
pub enum Error {
    Shape { axis: Axis, size: usize },
    Bounds { entry: usize, axis: Axis, index: usize, limit: usize },
    NonFinite { entry: usize },
    SumOverflow { entry: usize },
    RowPtrLength { expected: usize, actual: usize },
    ColumnValueLength { columns: usize, values: usize },
    RowPtrStart { found: usize },
    RowPtrEnd { found: usize, expected: usize },
    RowPtrOrder { row: usize },
    ColumnBounds { slot: usize, column: usize, limit: usize },
    ColumnOrder { slot: usize },
    NonFiniteValue { slot: usize },
    ExplicitZero { slot: usize },
    VectorLength { expected: usize, actual: usize },
    NonFiniteVector { index: usize },
    ProductOverflow { row: usize, column: usize },
    AccumulationOverflow { row: usize, column: usize },
}

fn validate_shape(rows: usize, cols: usize) -> Result<(), Error> {
    match () {
        _ if rows > MAX_DIM => Err(Error::Shape { axis: Axis::Row, size: rows }),
        _ if cols > MAX_DIM => Err(Error::Shape { axis: Axis::Column, size: cols }),
        _ => Ok(()),
    }
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        validate_shape(rows, cols)?;

        for (entry, item) in entries.iter().enumerate() {
            if item.row >= rows {
                return Err(Error::Bounds {
                    entry,
                    axis: Axis::Row,
                    index: item.row,
                    limit: rows,
                });
            }
            if item.col >= cols {
                return Err(Error::Bounds {
                    entry,
                    axis: Axis::Column,
                    index: item.col,
                    limit: cols,
                });
            }
            if !item.value.is_finite() {
                return Err(Error::NonFinite { entry });
            }
        }

        // Coordinate-indexed sums preserve each coordinate's occurrence order.
        let mut sums: Vec<(usize, usize, f64)> = Vec::new();
        for (entry, item) in entries.iter().enumerate() {
            let index = match sums.iter().position(|(r, c, _)| *r == item.row && *c == item.col) {
                Some(index) => index,
                None => {
                    sums.push((item.row, item.col, 0.0));
                    sums.len() - 1
                }
            };

            let updated = sums[index].2 + item.value;
            if !updated.is_finite() {
                return Err(Error::SumOverflow { entry });
            }
            sums[index].2 = updated;
        }

        sums.sort_by_key(|(row, col, _)| (*row, *col));
        let entries = sums
            .into_iter()
            .filter(|(_, _, value)| *value != 0.0)
            .map(|(row, col, value)| Entry { row, col, value })
            .collect();

        Ok(Self { rows, cols, entries })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn entries(&self) -> &[Entry] {
        &self.entries
    }

    pub fn to_csr(&self) -> Csr {
        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        row_ptr.push(0);
        let mut cursor = 0;
        for row in 0..self.rows {
            while cursor < self.entries.len() && self.entries[cursor].row == row {
                cursor += 1;
            }
            row_ptr.push(cursor);
        }

        let mut col_idx = Vec::with_capacity(self.entries.len());
        let mut values = Vec::with_capacity(self.entries.len());
        for item in &self.entries {
            col_idx.push(item.col);
            values.push(item.value);
        }

        Csr { rows: self.rows, cols: self.cols, row_ptr, col_idx, values }
    }
}

impl Csr {
    pub fn from_parts(
        rows: usize,
        cols: usize,
        row_ptr: Vec<usize>,
        col_idx: Vec<usize>,
        values: Vec<f64>,
    ) -> Result<Self, Error> {
        validate_shape(rows, cols)?;

        if row_ptr.len() != rows + 1 {
            return Err(Error::RowPtrLength { expected: rows + 1, actual: row_ptr.len() });
        }
        if col_idx.len() != values.len() {
            return Err(Error::ColumnValueLength {
                columns: col_idx.len(),
                values: values.len(),
            });
        }
        if row_ptr[0] != 0 {
            return Err(Error::RowPtrStart { found: row_ptr[0] });
        }
        if row_ptr[rows] != values.len() {
            return Err(Error::RowPtrEnd { found: row_ptr[rows], expected: values.len() });
        }
        for row in 0..rows {
            if row_ptr[row + 1] < row_ptr[row] {
                return Err(Error::RowPtrOrder { row });
            }
        }

        for row in 0..rows {
            let mut previous: Option<usize> = None;
            for slot in row_ptr[row]..row_ptr[row + 1] {
                let column = col_idx[slot];
                if column >= cols {
                    return Err(Error::ColumnBounds { slot, column, limit: cols });
                }
                if let Some(prior) = previous {
                    if column <= prior {
                        return Err(Error::ColumnOrder { slot });
                    }
                }
                let value = values[slot];
                if !value.is_finite() {
                    return Err(Error::NonFiniteValue { slot });
                }
                if value == 0.0 {
                    return Err(Error::ExplicitZero { slot });
                }
                previous = Some(column);
            }
        }

        Ok(Self { rows, cols, row_ptr, col_idx, values })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn row_ptr(&self) -> &[usize] {
        &self.row_ptr
    }

    pub fn col_indices(&self) -> &[usize] {
        &self.col_idx
    }

    pub fn values(&self) -> &[f64] {
        &self.values
    }

    pub fn to_coo(&self) -> Sparse {
        let mut entries = Vec::with_capacity(self.values.len());
        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                entries.push(Entry { row, col: self.col_idx[slot], value: self.values[slot] });
            }
        }
        Sparse { rows: self.rows, cols: self.cols, entries }
    }

    pub fn mul_vector(&self, x: &[f64]) -> Result<Vec<f64>, Error> {
        if x.len() != self.cols {
            return Err(Error::VectorLength { expected: self.cols, actual: x.len() });
        }

        for index in 0..x.len() {
            if !x[index].is_finite() {
                return Err(Error::NonFiniteVector { index });
            }
        }

        let mut output = Vec::with_capacity(self.rows);
        for row in 0..self.rows {
            let mut total = 0.0;
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                let column = self.col_idx[slot];
                let product = self.values[slot] * x[column];
                if !product.is_finite() {
                    return Err(Error::ProductOverflow { row, column });
                }
                total += product;
                if !total.is_finite() {
                    return Err(Error::AccumulationOverflow { row, column });
                }
            }
            output.push(if total == 0.0 { 0.0 } else { total });
        }
        Ok(output)
    }
}
```
