- **Trailing slash:** I treat only `/` as the valid zero-segment root; every other path ending in `/` is invalid after its preceding segments are checked.
- **Query and fragment:** For requests, I strip at the first raw `?` or `#` before validation; patterns receive no such stripping.
- **Capture parsing:** A segment beginning `:` or `*` is parsed as a capture, and capture-name errors are reported at the segment start before any later segment is checked.
- **Wildcard ranking and matching:** A wildcard consumes zero or more remaining segments, joins them with `/`, and ends its rank vector; a wildcard route ties another route only when those vectors are equal.
- **Percent escapes:** I accept uppercase `%HH` only when it decodes to printable ASCII that is neither `/` nor an unreserved byte; comparison and captures use the once-decoded value.
- **Registration ties:** I keep routes in registration order and replace a winner only when a later matching route has a strictly higher rank.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(s: &str, base: usize) -> Result<String, usize> {
    let bytes = s.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || unreserved(value) {
                return Err(base + i);
            }
            out.push(value);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_head<'a>(input: &'a str, is_pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if is_pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();

    let mut start = 1;
    while start < end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(invalid(start));
        }

        let raw = &input[start..stop];
        if is_pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_name(name) || names.iter().any(|n| n == name) {
                return Err(invalid(start));
            }
            names.push(name.to_owned());
            if marker == b'*' && stop != end {
                return Err(invalid(start));
            }
            segments.push(if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            let decoded = decode_segment(raw, start).map_err(invalid)?;
            segments.push(Segment::Static(decoded));
        }

        if stop == end {
            break;
        }
        start = stop + 1;
        if start == end {
            break;
        }
    }

    if trailing {
        return Err(invalid(bytes.len() - 1));
    }

    Ok((segments, false))
}

fn parse_request(input: &str) -> Result<Vec<String>, Error> {
    let cut = input
        .bytes()
        .position(|b| b == b'?' || b == b'#')
        .unwrap_or(input.len());
    let head = &input[..cut];
    let (segments, _) = parse_head(head, false)?;
    Ok(segments
        .into_iter()
        .map(|s| match s {
            Segment::Static(v) => v,
            _ => unreachable!(),
        })
        .collect())
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let (segments, _) = parse_head(pattern, true)?;
        let id = self.routes.len();
        let mut rank = Vec::new();
        for segment in &segments {
            match segment {
                Segment::Static(_) => rank.push(2),
                Segment::Param(_) => rank.push(1),
                Segment::Wildcard(_) => {
                    rank.push(0);
                    break;
                }
            }
        }
        if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
            rank.push(3);
        }
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let request = parse_request(path)?;
        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            let mut params = Vec::new();
            let mut ri = 0usize;
            let mut pi = 0usize;
            let mut matched = true;

            while ri < route.segments.len() {
                match &route.segments[ri] {
                    Segment::Static(value) => {
                        if request.get(pi) != Some(value) {
                            matched = false;
                            break;
                        }
                        ri += 1;
                        pi += 1;
                    }
                    Segment::Param(name) => {
                        if let Some(value) = request.get(pi) {
                            params.push((name.clone(), value.clone()));
                            ri += 1;
                            pi += 1;
                        } else {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Wildcard(name) => {
                        let value = request[pi..].join("/");
                        params.push((name.clone(), value));
                        pi = request.len();
                        ri += 1;
                        break;
                    }
                }
            }

            if matched && pi == request.len() {
                let replace = best
                    .as_ref()
                    .map(|(current, _)| route.rank > current.rank)
                    .unwrap_or(true);
                if replace {
                    best = Some((route, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

impl Default for Router {
    fn default() -> Self {
        Self::new()
    }
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Part {
    Literal(String),
    Capture { name: String, wildcard: bool },
}

#[derive(Clone)]
struct Entry {
    id: usize,
    parts: Vec<Part>,
    rank: Vec<u8>,
}

pub struct Router {
    entries: Vec<Entry>,
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'.' || b == b'_' || b == b'~' || b == b'-'
}

fn nibble(b: u8) -> Option<u8> {
    if b.is_ascii_digit() {
        Some(b - b'0')
    } else if (b'A'..=b'F').contains(&b) {
        Some(b - b'A' + 10)
    } else {
        None
    }
}

fn decode(raw: &str, offset: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::new();
    let mut index = 0;
    while index < bytes.len() {
        if bytes[index] == b'%' {
            if index + 2 >= bytes.len() {
                return Err(offset + index);
            }
            let high = nibble(bytes[index + 1]).ok_or(offset + index)?;
            let low = nibble(bytes[index + 2]).ok_or(offset + index)?;
            let value = high * 16 + low;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || is_unreserved(value) {
                return Err(offset + index);
            }
            decoded.push(value);
            index += 3;
        } else {
            let value = bytes[index];
            if !is_unreserved(value) {
                return Err(offset + index);
            }
            decoded.push(value);
            index += 1;
        }
    }
    Ok(String::from_utf8(decoded).unwrap())
}

fn good_name(name: &str) -> bool {
    let bs = name.as_bytes();
    if bs.is_empty() || !bs[0].is_ascii_lowercase() {
        return false;
    }
    bs[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse(input: &str, pattern: bool) -> Result<Vec<Part>, Error> {
    let err = |at| if pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    };

    if !input.starts_with('/') {
        return Err(err(0));
    }
    if input == "/" {
        return Ok(vec![]);
    }

    let bytes = input.as_bytes();
    let trailing = bytes[bytes.len() - 1] == b'/';
    let limit = bytes.len() - usize::from(trailing);
    let mut parts = Vec::new();
    let mut names = Vec::new();
    let mut pos = 1;

    while pos < limit {
        let mut next = pos;
        while next < limit && bytes[next] != b'/' {
            next += 1;
        }
        if next == pos {
            return Err(err(pos));
        }
        let token = &input[pos..next];

        if pattern && (token.starts_with(':') || token.starts_with('*')) {
            let wildcard = token.as_bytes()[0] == b'*';
            let name = &token[1..];
            if !good_name(name) || names.iter().any(|known: &String| known == name) {
                return Err(err(pos));
            }
            if wildcard && next < limit {
                return Err(err(pos));
            }
            names.push(name.to_owned());
            parts.push(Part::Capture {
                name: name.to_owned(),
                wildcard,
            });
        } else {
            parts.push(Part::Literal(decode(token, pos).map_err(err)?));
        }

        if next == limit {
            break;
        }
        pos = next + 1;
    }

    if trailing {
        return Err(err(bytes.len() - 1));
    }
    Ok(parts)
}

fn request_parts(path: &str) -> Result<Vec<String>, Error> {
    let cut = path
        .as_bytes()
        .iter()
        .position(|b| *b == b'?' || *b == b'#')
        .unwrap_or(path.len());
    parse(&path[..cut], false).map(|parts| {
        parts
            .into_iter()
            .map(|p| match p {
                Part::Literal(v) => v,
                Part::Capture { .. } => unreachable!(),
            })
            .collect()
    })
}

impl Router {
    pub fn new() -> Self {
        Self { entries: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let parts = parse(pattern, true)?;
        let id = self.entries.len();
        let mut rank = Vec::new();
        let mut has_wildcard = false;
        for part in &parts {
            match part {
                Part::Literal(_) => rank.push(2),
                Part::Capture {
                    wildcard: false, ..
                } => rank.push(1),
                Part::Capture {
                    wildcard: true, ..
                } => {
                    rank.push(0);
                    has_wildcard = true;
                    break;
                }
            }
        }
        if !has_wildcard {
            rank.push(3);
        }
        self.entries.push(Entry { id, parts, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let values = request_parts(path)?;
        let mut chosen: Option<(&Entry, Vec<(String, String)>)> = None;

        for entry in &self.entries {
            let mut captures = Vec::new();
            let mut index = 0;
            let mut ok = true;
            for part in &entry.parts {
                match part {
                    Part::Literal(literal) => {
                        if values.get(index) != Some(literal) {
                            ok = false;
                            break;
                        }
                        index += 1;
                    }
                    Part::Capture {
                        name,
                        wildcard: false,
                    } => {
                        if let Some(value) = values.get(index) {
                            captures.push((name.clone(), value.clone()));
                            index += 1;
                        } else {
                            ok = false;
                            break;
                        }
                    }
                    Part::Capture {
                        name,
                        wildcard: true,
                    } => {
                        captures.push((name.clone(), values[index..].join("/")));
                        index = values.len();
                        break;
                    }
                }
            }

            if ok && index == values.len() {
                if chosen
                    .as_ref()
                    .map(|(old, _)| entry.rank > old.rank)
                    .unwrap_or(true)
                {
                    chosen = Some((entry, captures));
                }
            }
        }

        chosen
            .map(|(entry, params)| Match {
                route: entry.id,
                params,
            })
            .ok_or(Error::NotFound)
    }
}

impl Default for Router {
    fn default() -> Self {
        Self::new()
    }
}
```
