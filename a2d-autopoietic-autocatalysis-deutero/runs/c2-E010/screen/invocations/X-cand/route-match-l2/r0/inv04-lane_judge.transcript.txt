    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    segments: Vec<Segment>,
    has_wildcard: bool,
    id: usize,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let id = self.routes.len();
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));
        self.routes.push(Route {
            segments,
            has_wildcard,
            id,
        });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, ch)| if ch == '?' || ch == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parts = parse_path(head)?;

        let mut best: Option<(Vec<u8>, usize, Vec<(String, String)>)> = None;

        for route in &self.routes {
            let Some(params) = match_route(route, &parts) else {
                continue;
            };
            let rank = route_rank(route);
            let replace = match &best {
                None => true,
                Some((best_rank, best_id, _)) => {
                    rank > *best_rank || (rank == *best_rank && route.id < *best_id)
                }
            };
            if replace {
                best = Some((rank, route.id, params));
            }
        }

        match best {
            Some((_, route, params)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();
    let bytes = pattern.as_bytes();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'/')
            .map(|offset| start + offset)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..end];
        let first = bytes[start];
        if first == b':' || first == b'*' {
            let name = &raw[1..];
            if first == b'*' && end < bytes.len() && end + 1 < bytes.len() {
                return Err(Error::InvalidPattern { at: start });
            }
            if !valid_name(name) || names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_string());
            if first == b':' {
                segments.push(Segment::Param(name.to_string()));
            } else {
                segments.push(Segment::Wildcard(name.to_string()));
            }
        } else {
            let decoded = decode_segment(raw, start).map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }

        if end == bytes.len() {
            break;
        }
        if end + 1 == bytes.len() {
            return Err(Error::InvalidPattern { at: end });
        }
        start = end + 1;
    }

    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'/')
            .map(|offset| start + offset)
            .unwrap_or(bytes.len());

        if end == start {
            return Err(Error::InvalidPath { at: start });
        }

        let decoded = decode_segment(&path[start..end], start)
            .map_err(|at| Error::InvalidPath { at })?;
        parts.push(decoded);

        if end == bytes.len() {
            break;
        }
        if end + 1 == bytes.len() {
            return Err(Error::InvalidPath { at: end });
        }
        start = end + 1;
    }

    Ok(parts)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let Some(hi) = upper_hex(bytes[i + 1]) else {
                return Err(base + i);
            };
            let Some(lo) = upper_hex(bytes[i + 2]) else {
                return Err(base + i);
            };
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded segments are ASCII"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in &route.segments {
        match segment {
            Segment::Static(expected) => {
                if path.get(path_index)? != expected {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                let value = path.get(path_index)?;
                params.push((name.clone(), value.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                path_index = path.len();
                break;
            }
        }
    }

    if path_index == path.len() {
        Some(params)
    } else {
        None
    }
}

fn route_rank(route: &Route) -> Vec<u8> {
    let mut rank = Vec::with_capacity(route.segments.len() + 1);
    for segment in &route.segments {
        rank.push(match segment {
            Segment::Static(_) => 2,
            Segment::Param(_) => 1,
            Segment::Wildcard(_) => 0,
        });
        if matches!(segment, Segment::Wildcard(_)) {
            return rank;
        }
    }
    if !route.has_wildcard {
        rank.push(3);
    }
    rank
}

            .as_bytes()

            .unwrap_or(path.len());

            let mut vi = 0usize;
            let mut captures = Vec::new();
            let mut trailing_ok = false;

            for part in &entry.parts {
                match part {
                    Part::Text(expected) => {
                        if values.get(vi) != Some(expected) {
                            break;
                        }
                        vi += 1;
                        pi += 1;
                    }
                    Part::One(name) => {
                        if vi == values.len() {
                            break;
                        }
                        captures.push((name.clone(), values[vi].clone()));
                        vi += 1;
                        pi += 1;
                    }
                    Part::Rest(name) => {
                        captures.push((name.clone(), values[vi..].join("/")));
                        vi = values.len();
                        pi += 1;
                    }
                }
            }

            let has_rest = entry.parts.iter().any(|p| matches!(p, Part::Rest(_)));
            if has_rest {
                trailing_ok = !trailing;
            } else {
                trailing_ok = trailing == (entry.parts.is_empty() && trailing);
            }

            if vi != values.len() || pi != entry.parts.len() || !trailing_ok {
                continue;
            }

            let better = match &winner {
                None => true,
                Some((best_id, _)) => {
                    entry.rank > self.entries[*best_id].rank
                        || (entry.rank == self.entries[*best_id].rank && id < *best_id)
                }
            };
            if better {
                winner = Some((id, captures));
            }
        }

        winner
            .map(|(route, params)| Match { route, params })
            .ok_or(Error::NotFound)
    }
}

fn compile(pattern: &str) -> Result<Entry, Error> {
    let (pieces, trailing) = split(pattern, true)?;
    let mut parts = Vec::new();
    let mut seen = std::collections::HashSet::new();

    for (piece, offset) in pieces.iter().copied() {
        if piece.starts_with(':') || piece.starts_with('*') {
            let marker = piece.as_bytes()[0];
            let name = &piece[1..];
            if !name_ok(name) || !seen.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: offset });
            }
            if marker == b'*' {
                if offset_piece_is_not_last(&pieces, offset) {
                    return Err(Error::InvalidPattern { at: offset });
                }
                parts.push(Part::Rest(name.to_string()));
            } else {
                parts.push(Part::One(name.to_string()));
            }
        } else {
            parts.push(Part::Text(decode(piece, offset, true)?));
        }
    }

    let mut rank = Vec::new();
    let mut wildcard = false;
    for p in &parts {
        match p {
            Part::Text(_) => rank.push(2),
            Part::One(_) => rank.push(1),
            Part::Rest(_) => {
                rank.push(0);
                wildcard = true;
                break;
            }
        }
    }
    if !wildcard {
        rank.push(3);
    }

    let _ = trailing;
    Ok(Entry { parts, rank })
}

fn offset_piece_is_not_last(pieces: &[(&str, usize)], offset: usize) -> bool {
    pieces
        .iter()
        .position(|(_, at)| *at == offset)
        .map(|i| i + 1 != pieces.len())
        .unwrap_or(true)
}

fn request_segments(path: &str) -> Result<(Vec<String>, bool), Error> {
    let (pieces, trailing) = split(path, false)?;
    let mut values = Vec::new();
    for (piece, offset) in pieces {
        values.push(decode(piece, offset, false)?);
    }
    Ok((values, trailing))
}

fn split(input: &str, pattern: bool) -> Result<(Vec<(&str, usize)>, bool), Error> {
    let fail = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if input.as_bytes().first() != Some(&b'/') {
        return Err(fail(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let mut result = Vec::new();
    let mut begin = 1;
    let mut trailing = false;

    for i in 1..bytes.len() {
        if bytes[i] != b'/' {
            continue;
        }
        if i == bytes.len() - 1 {
            trailing = true;
            break;
        }
        if i == begin {
            return Err(fail(begin));
        }
        result.push((&input[begin..i], begin));
        begin = i + 1;
    }

    if !trailing {
        if begin == bytes.len() {
            return Err(fail(begin));
        }
        result.push((&input[begin..], begin));
    }
    Ok((result, trailing))
}

fn name_ok(name: &str) -> bool {

        && b[0].is_ascii_lowercase()
        && b[1..]

fn decode(piece: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let fail = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    let b = piece.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        if b[i] == b'%' {
            if i + 2 >= b.len() {
                return Err(fail(offset + i));
            }
            let h = upper_hex(b[i + 1]).ok_or_else(|| fail(offset + i))?;
            let l = upper_hex(b[i + 2]).ok_or_else(|| fail(offset + i))?;

                return Err(fail(offset + i));
            }
            out.push(decoded);
            i += 3;
        } else if is_plain(b[i]) {
            out.push(b[i]);
            i += 1;
        } else {
            return Err(fail(offset + i));
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|b| *b == b'/')
            .map_or(bytes.len(), |n| start + n);
        if start == end {
            return Err(Error::InvalidPattern { at: start });
        }

        let token = &s[start..end];
        match bytes[start] {
            b':' | b'*' => {
                let marker = bytes[start];
                if marker == b'*' && end < bytes.len() && end + 1 < bytes.len() {
                    return Err(Error::InvalidPattern { at: start });
                }
                let name = &token[1..];
                if !name_ok(name) || seen.iter().any(|v: &String| v == name) {
                    return Err(Error::InvalidPattern { at: start });
                }
                seen.push(name.to_owned());
                if marker == b':' {
                    out.push(Part::Capture(name.to_owned()));
                } else {
                    out.push(Part::Rest(name.to_owned()));
                }
            }
            _ => {
                let literal =
                    decode(token, start).map_err(|at| Error::InvalidPattern { at })?;
                out.push(Part::Literal(literal));
            }
        }

        if end == bytes.len() {
            break;
        }
        if end + 1 == bytes.len() {
            return Err(Error::InvalidPattern { at: end });
        }
        start = end + 1;
    }

    Ok(out)
}

fn parse_path(s: &str) -> Result<Vec<String>, Error> {
    if !s.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if s == "/" {
        return Ok(Vec::new());
    }

    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut start = 1;

    while start < bytes.len() {
        let end = bytes[start..]
            .iter()
            .position(|b| *b == b'/')
            .map_or(bytes.len(), |n| start + n);
        if start == end {
            return Err(Error::InvalidPath { at: start });
        }
        out.push(decode(&s[start..end], start).map_err(|at| Error::InvalidPath { at })?);

        if end == bytes.len() {
            break;
        }
        if end + 1 == bytes.len() {
            return Err(Error::InvalidPath { at: end });
        }
        start = end + 1;
    }

    Ok(out)
}

fn name_ok(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b[0].is_ascii_lowercase()
        && b[1..]
            .iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_')
}

fn decode(s: &str, offset: usize) -> Result<String, usize> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        if b[i] == b'%' {
            if i + 2 >= b.len() {
                return Err(offset + i);
            }
            let a = hex(b[i + 1]).ok_or(offset + i)?;
            let c = hex(b[i + 2]).ok_or(offset + i)?;
            let v = a * 16 + c;
            if !(0x20..=0x7e).contains(&v) || v == b'/' || unreserved(v) {
                return Err(offset + i);
            }
            out.push(v);
            i += 3;
        } else if unreserved(b[i]) {
            out.push(b[i]);
            i += 1;
        } else {
            return Err(offset + i);
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b'.' == b || b'_' == b || b'~' == b || b'-' == b
}

fn try_match(entry: &Entry, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut out = Vec::new();
    let mut i = 0;
    for part in &entry.parts {
        match part {
            Part::Literal(v) => {
                if path.get(i)? != v {
                    return None;
                }
                i += 1;
            }
            Part::Capture(name) => {
                out.push((name.clone(), path.get(i)?.clone()));
                i += 1;
            }
            Part::Rest(name) => {
                out.push((name.clone(), path[i..].join("/")));
                i = path.len();
                break;
            }
        }
    }
    (i == path.len()).then_some(out)
}

fn rank(entry: &Entry) -> Vec<u8> {
    let mut out = Vec::new();
    for part in &entry.parts {
        match part {
            Part::Literal(_) => out.push(2),
            Part::Capture(_) => out.push(1),
            Part::Rest(_) => {
                out.push(0);
                return out;
            }
        }
    }
    out.push(3);
    out
}

    segments: Vec<Segment>,
    rank: Vec<u8>,

pub struct Router {

}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let (segments, rank) = parse_pattern(pattern)?;
        let id = self.routes.len();
        id.checked_add(1).ok_or(Error::InvalidPattern {
            at: pattern.len(),
        })?;
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request_segments = parse_path(head)?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            let Some(params) = match_route(&route.segments, &request_segments) else {
                continue;
            };
            let replace = match &best {
                None => true,
                Some((current, _)) => {
                    route.rank > current.rank
                        || (route.rank == current.rank && route.id < current.id)
                }
            };
            if replace {
                best = Some((route, params));
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

    fn default() -> Self {

    }

fn parse_path(input: &str) -> Result<Vec<String>, Error> {

        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());

    let mut result = Vec::new();
    let mut start = 1;

                if i == bytes.len() {

                return Err(Error::InvalidPath { at: i });

            result.push(decode_segment(input, start, i, false)?);
            start = i + 1;
        }
    }
    Ok(result)

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, Vec<u8>), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok((Vec::new(), vec![3]));
    }

    let bytes = input.as_bytes();
    let mut raw_segments: Vec<(usize, usize)> = Vec::new();
    let mut start = 1;
    for i in 1..=bytes.len() {
        if i == bytes.len() || bytes[i] == b'/' {
            if i == start {
                if i == bytes.len() {
                    return Err(Error::InvalidPattern { at: i - 1 });
                }
                return Err(Error::InvalidPattern { at: i });
            }
            raw_segments.push((start, i));
            start = i + 1;
        }
    }

    let mut segments = Vec::with_capacity(raw_segments.len());
    let mut rank = Vec::with_capacity(raw_segments.len() + 1);
    let mut names = std::collections::HashSet::new();

    for (index, &(start, end)) in raw_segments.iter().enumerate() {
        let raw = &input[start..end];
        if let Some(name) = raw.strip_prefix(':') {
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(Segment::Param(name.to_owned()));
            rank.push(1);
        } else if let Some(name) = raw.strip_prefix('*') {
            if index + 1 != raw_segments.len() {
                return Err(Error::InvalidPattern { at: start });
            }
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(Segment::Wildcard(name.to_owned()));
            rank.push(0);
        } else {
            let decoded = decode_segment(input, start, end, true)?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }
    }

    rank.push(3);
    Ok((segments, rank))
}

}

fn decode_segment(
    input: &str,
    start: usize,
    end: usize,
    pattern: bool,
) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut out = Vec::with_capacity(end - start);
    let mut i = start;

    while i < end {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= end {
                return Err(if pattern {
                    Error::InvalidPattern { at: i }
                } else {
                    Error::InvalidPath { at: i }
                });
            }
            let (Some(hi), Some(lo)) = (hex_upper(bytes[i + 1]), hex_upper(bytes[i + 2])) else {
                return Err(if pattern {
                    Error::InvalidPattern { at: i }
                } else {
                    Error::InvalidPath { at: i }
                });
            };
            let decoded = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(if pattern {
                    Error::InvalidPattern { at: i }
                } else {
                    Error::InvalidPath { at: i }
                });
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: i }
            } else {
                Error::InvalidPath { at: i }
            });
        }
    }

    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn match_route(
    pattern: &[Segment],
    request: &[String],
) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut pi = 0;
    let mut ri = 0;

    while pi < pattern.len() {
        match &pattern[pi] {
            Segment::Wildcard(name) => {
                let value = request[ri..].join("/");
                params.push((name.clone(), value));
                return Some(params);
            }
            Segment::Static(value) => {
                if request.get(ri)? != value {
                    return None;
                }
                pi += 1;
                ri += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), request.get(ri)?.clone()));
                pi += 1;
                ri += 1;
            }
        }
    }

    if ri == request.len() {
        Some(params)
    } else {
        None
    }
}

fn parse(input: &str, pattern: bool) -> Result<Vec<Part>, Error> {
    let err = |at| if pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    };

    if !input.starts_with('/') {
        return Err(err(0));
    }
    if input == "/" {
        return Ok(vec![]);
    }

    let bytes = input.as_bytes();
    let trailing = bytes[bytes.len() - 1] == b'/';
    let limit = bytes.len() - usize::from(trailing);
    let mut parts = Vec::new();
    let mut names = Vec::new();
    let mut pos = 1;

    while pos < limit {
        let mut next = pos;
        while next < limit && bytes[next] != b'/' {
            next += 1;
        }
        if next == pos {
            return Err(err(pos));
        }
        let token = &input[pos..next];

        if pattern && (token.starts_with(':') || token.starts_with('*')) {
            let wildcard = token.as_bytes()[0] == b'*';
            let name = &token[1..];
            if !good_name(name) || names.iter().any(|known: &String| known == name) {
                return Err(err(pos));
            }
            if wildcard && next < limit {
                return Err(err(pos));
            }
            names.push(name.to_owned());
            parts.push(Part::Capture {
                name: name.to_owned(),
                wildcard,
            });
        } else {
            parts.push(Part::Literal(decode(token, pos).map_err(err)?));
        }

        if next == limit {
            break;
        }
        pos = next + 1;
    }

    if trailing {
        return Err(err(bytes.len() - 1));
    }
    Ok(parts)
}

fn request_parts(path: &str) -> Result<Vec<String>, Error> {
    let cut = path
        .as_bytes()
        .iter()
        .position(|b| *b == b'?' || *b == b'#')
        .unwrap_or(path.len());
    parse(&path[..cut], false).map(|parts| {
        parts
            .into_iter()
            .map(|p| match p {
                Part::Literal(v) => v,
                Part::Capture { .. } => unreachable!(),
            })
            .collect()
    })
}

impl Router {
    pub fn new() -> Self {
        Self { entries: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let parts = parse(pattern, true)?;
        let id = self.entries.len();
        let mut rank = Vec::new();
        let mut has_wildcard = false;
        for part in &parts {
            match part {
                Part::Literal(_) => rank.push(2),
                Part::Capture {
                    wildcard: false, ..
                } => rank.push(1),
                Part::Capture {
                    wildcard: true, ..
                } => {
                    rank.push(0);
                    has_wildcard = true;
                    break;
                }
            }
        }
        if !has_wildcard {
            rank.push(3);
        }
        self.entries.push(Entry { id, parts, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let values = request_parts(path)?;
        let mut chosen: Option<(&Entry, Vec<(String, String)>)> = None;

        for entry in &self.entries {
            let mut captures = Vec::new();
            let mut index = 0;
            let mut ok = true;
            for part in &entry.parts {
                match part {
                    Part::Literal(literal) => {
                        if values.get(index) != Some(literal) {
                            ok = false;
                            break;
                        }
                        index += 1;
                    }
                    Part::Capture {
                        name,
                        wildcard: false,
                    } => {
                        if let Some(value) = values.get(index) {
                            captures.push((name.clone(), value.clone()));
                            index += 1;
                        } else {
                            ok = false;
                            break;
                        }
                    }
                    Part::Capture {
                        name,
                        wildcard: true,
                    } => {
                        captures.push((name.clone(), values[index..].join("/")));
                        index = values.len();
                        break;
                    }
                }
            }

            if ok && index == values.len() {
                if chosen
                    .as_ref()
                    .map(|(old, _)| entry.rank > old.rank)
                    .unwrap_or(true)
                {
                    chosen = Some((entry, captures));
                }
            }
        }

        chosen
            .map(|(entry, params)| Match {
                route: entry.id,
                params,
            })
            .ok_or(Error::NotFound)
    }
}

fn parse_head<'a>(input: &'a str, is_pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if is_pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();

    let mut start = 1;
    while start < end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(invalid(start));
        }

        let raw = &input[start..stop];
        if is_pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_name(name) || names.iter().any(|n| n == name) {
                return Err(invalid(start));
            }
            names.push(name.to_owned());
            if marker == b'*' && stop != end {
                return Err(invalid(start));
            }
            segments.push(if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            let decoded = decode_segment(raw, start).map_err(invalid)?;
            segments.push(Segment::Static(decoded));
        }

        if stop == end {
            break;
        }
        start = stop + 1;
        if start == end {
            break;
        }
    }

    if trailing {
        return Err(invalid(bytes.len() - 1));
    }

    Ok((segments, false))
}

fn parse_request(input: &str) -> Result<Vec<String>, Error> {
    let cut = input
        .bytes()
        .position(|b| b == b'?' || b == b'#')
        .unwrap_or(input.len());
    let head = &input[..cut];
    let (segments, _) = parse_head(head, false)?;
    Ok(segments
        .into_iter()
        .map(|s| match s {
            Segment::Static(v) => v,
            _ => unreachable!(),
        })
        .collect())
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let (segments, _) = parse_head(pattern, true)?;
        let id = self.routes.len();
        let mut rank = Vec::new();
        for segment in &segments {
            match segment {
                Segment::Static(_) => rank.push(2),
                Segment::Param(_) => rank.push(1),
                Segment::Wildcard(_) => {
                    rank.push(0);
                    break;
                }
            }
        }
        if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
            rank.push(3);
        }
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let request = parse_request(path)?;
        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            let mut params = Vec::new();
            let mut ri = 0usize;
            let mut pi = 0usize;
            let mut matched = true;

            while ri < route.segments.len() {
                match &route.segments[ri] {
                    Segment::Static(value) => {
                        if request.get(pi) != Some(value) {
                            matched = false;
                            break;
                        }
                        ri += 1;
                        pi += 1;
                    }
                    Segment::Param(name) => {
                        if let Some(value) = request.get(pi) {
                            params.push((name.clone(), value.clone()));
                            ri += 1;
                            pi += 1;
                        } else {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Wildcard(name) => {
                        let value = request[pi..].join("/");
                        params.push((name.clone(), value));
                        pi = request.len();
                        ri += 1;
                        break;
                    }
                }
            }

            if matched && pi == request.len() {
                let replace = best
                    .as_ref()
                    .map(|(current, _)| route.rank > current.rank)
                    .unwrap_or(true);
                if replace {
                    best = Some((route, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

impl Default for Router {
    fn default() -> Self {
        Self::new()
    }
}

    Capture(String),
    Rest(String),

struct Entry {
    id: usize,
    parts: Vec<Part>,
    rank: Vec<u8>,

pub struct Router {
    entries: Vec<Entry>,
}

impl Router {
    pub fn new() -> Self {
        Self { entries: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let (parts, rank) = compile(pattern)?;
        let id = self.entries.len();
        id.checked_add(1).ok_or(Error::InvalidPattern {
            at: pattern.len(),
        })?;
        self.entries.push(Entry { id, parts, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let cut = path
            .as_bytes()
            .iter()
            .position(|b| *b == b'?' || *b == b'#')
            .unwrap_or(path.len());
        let values = read_path(&path[..cut])?;

        let mut winner: Option<(&Entry, Vec<(String, String)>)> = None;
        for entry in &self.entries {
            if let Some(captures) = run(&entry.parts, &values) {
                let better = winner.as_ref().map_or(true, |(old, _)| {
                    entry.rank > old.rank
                        || (entry.rank == old.rank && entry.id < old.id)
                });
                if better {
                    winner = Some((entry, captures));
                }
            }
        }

        winner
            .map(|(entry, params)| Match {
                route: entry.id,
                params,
            })
            .ok_or(Error::NotFound)
    }

fn compile(pattern: &str) -> Result<(Vec<Part>, Vec<u8>), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), vec![3]));
    }

    let bytes = pattern.as_bytes();
    let mut ranges = Vec::new();
    let mut begin = 1;
    for pos in 1..=bytes.len() {
        if pos == bytes.len() || bytes[pos] == b'/' {
            if pos == begin {
                return Err(Error::InvalidPattern {
                    at: if pos == bytes.len() { pos - 1 } else { pos },
                });
            }
            ranges.push((begin, pos));
            begin = pos + 1;
        }
    }

    let mut parts = Vec::with_capacity(ranges.len());
    let mut rank = Vec::with_capacity(ranges.len() + 1);
    let mut names = std::collections::HashSet::new();

    for (i, (begin, end)) in ranges.iter().copied().enumerate() {
        let raw = &pattern[begin..end];
        if let Some(name) = raw.strip_prefix('*') {
            if i + 1 != ranges.len() || !name_ok(name) || !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: begin });
            }
            parts.push(Part::Rest(name.to_string()));
            rank.push(0);
        } else if let Some(name) = raw.strip_prefix(':') {
            if !name_ok(name) || !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: begin });
            }
            parts.push(Part::Capture(name.to_string()));
            rank.push(1);
        } else {
            parts.push(Part::Literal(decode(pattern, begin, end, true)?));
            rank.push(2);
        }
    }

    if !matches!(parts.last(), Some(Part::Rest(_))) {
        rank.push(3);
    }
    Ok((parts, rank))
}

fn read_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut values = Vec::new();
    let mut begin = 1;
    for pos in 1..=bytes.len() {
        if pos == bytes.len() || bytes[pos] == b'/' {
            if pos == begin {
                return Err(Error::InvalidPath {
                    at: if pos == bytes.len() { pos - 1 } else { pos },
                });
            }
            values.push(decode(path, begin, pos, false)?);
            begin = pos + 1;
        }
    }
    Ok(values)
}

fn name_ok(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b[0].is_ascii_lowercase()
        && b[1..]
            .iter()
            .all(|x| x.is_ascii_lowercase() || x.is_ascii_digit() || *x == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'.' || b == b'_' || b == b'~' || b == b'-'
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode(s: &str, start: usize, end: usize, is_pattern: bool) -> Result<String, Error> {
    let bytes = s.as_bytes();
    let mut decoded = Vec::with_capacity(end - start);
    let mut pos = start;
    while pos < end {
        let byte = bytes[pos];
        if byte == b'%' {
            let value = if pos + 2 < end {
                upper_hex(bytes[pos + 1])
                    .zip(upper_hex(bytes[pos + 2]))
                    .map(|(a, b)| a * 16 + b)
            } else {
                None
            };
            match value {
                Some(v)
                    if (0x20..=0x7e).contains(&v)
                        && v != b'/'
                        && !unreserved(v) =>
                {
                    decoded.push(v);
                    pos += 3;
                }
                _ => return Err(error_at(is_pattern, pos)),
            }
        } else if unreserved(byte) {
            decoded.push(byte);
            pos += 1;
        } else {
            return Err(error_at(is_pattern, pos));
        }
    }
    Ok(String::from_utf8(decoded).expect("output is ASCII"))
}

fn error_at(is_pattern: bool, at: usize) -> Error {
    if is_pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    }
}

fn run(parts: &[Part], values: &[String]) -> Option<Vec<(String, String)>> {
    let mut captures = Vec::new();
    let mut index = 0;

    for part in parts {
        match part {
            Part::Rest(name) => {
                captures.push((name.clone(), values[index..].join("/")));
                return Some(captures);
            }
            Part::Literal(literal) => {
                if values.get(index)? != literal {
                    return None;
                }
                index += 1;
            }
            Part::Capture(name) => {
                captures.push((name.clone(), values.get(index)?.clone()));
                index += 1;
            }
        }
    }

    (index == values.len()).then_some(captures)
}
