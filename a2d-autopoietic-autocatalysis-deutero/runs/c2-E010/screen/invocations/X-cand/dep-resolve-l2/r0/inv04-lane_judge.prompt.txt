struct Search<'a> {
    catalog: &'a [Package],
    indices: std::collections::BTreeMap<String, usize>,
    attempts: usize,
    max_attempts: usize,
}

impl Search<'_> {
    fn inspect(
        &mut self,
        assigned: &mut std::collections::BTreeMap<String, Version>,
        root_constraints: &std::collections::BTreeMap<String, Vec<Constraint>>,
    ) -> Result<Vec<Selection>, ResolveError> {
        let mut active = root_constraints.clone();
        for (name, version) in assigned.iter() {
            if let Some(&pi) = self.indices.get(name) {
                if let Some(release) = self.catalog[pi].releases.iter().find(|r| r.version == *version) {
                    for dep in &release.deps {
                        active.entry(dep.package.clone()).or_default().push(dep.constraint);
                    }
                }
            }
        }

        if let Some(name) = active.keys().filter(|n| !self.indices.contains_key(*n)).min() {
            return Err(ResolveError::Missing { package: name.clone() });
        }

        for (name, version) in assigned.iter() {
            if let Some(cs) = active.get(name) {
                if !cs.iter().all(|c| matches(*version, *c)) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: canonical(cs.clone()),
                    });
                }
            }
        }

        for (name, cs) in &active {
            if !assigned.contains_key(name) {
                let pi = self.indices[name];
                if !self.catalog[pi].releases.iter().any(|r| cs.iter().all(|c| matches(r.version, *c))) {
                    return Err(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: canonical(cs.clone()),
                    });
                }
            }
        }

        let next = active.keys().find(|n| !assigned.contains_key(*n)).cloned();
        if let Some(name) = next {
            let pi = self.indices[&name];
            let mut candidates: Vec<Version> = self.catalog[pi].releases.iter()
                .map(|r| r.version)
                .filter(|v| active[&name].iter().all(|c| matches(*v, *c)))
                .collect();
            candidates.sort_by(|a, b| b.cmp(a));
            let mut first_error = None;
            for version in candidates {
                if self.attempts == self.max_attempts {
                    return Err(ResolveError::Limit { attempts: self.max_attempts });
                }
                self.attempts += 1;
                assigned.insert(name.clone(), version);
                let result = self.inspect(assigned, root_constraints);
                assigned.remove(&name);
                match result {
                    Ok(solution) => return Ok(solution),
                    Err(e @ ResolveError::Limit { .. }) => return Err(e),
                    Err(e) => {
                        if first_error.is_none() {
                            first_error = Some(e);
                        }
                    }
                }
            }
            return Err(first_error.unwrap_or_else(|| ResolveError::NoMatch {
                package: name.clone(),
                requirements: canonical(active[&name].clone()),
            }));
        }

        Ok(assigned.iter().map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        }).collect())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate(catalog, roots)?;
    let mut indices = std::collections::BTreeMap::new();
    for (i, p) in catalog.iter().enumerate() {
        indices.insert(p.name.clone(), i);
    }
    let mut root_constraints: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for r in roots {
        root_constraints.entry(r.package.clone()).or_default().push(r.constraint);
    }
    let mut search = Search { catalog, indices, attempts: 0, max_attempts };
    search.inspect(&mut std::collections::BTreeMap::new(), &root_constraints)
}

impl Search<'_> {
    fn inspect(
        &mut self,
        selected: &mut std::collections::BTreeMap<String, Version>,
        roots: &[Requirement],
    ) -> Result<Option<Vec<Selection>>, ResolveError> {
        let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
            std::collections::BTreeMap::new();

        for root in roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }
        for (name, version) in selected.iter() {
            if let Some(&index) = self.by_name.get(name.as_str()) {
                if let Some(release) = self.packages[index]
                    .releases
                    .iter()
                    .find(|release| release.version == *version)
                {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }

        for constraints in active.values_mut() {
            *constraints = canonical(std::mem::take(constraints));
        }

        if let Some(name) = active
            .keys()
            .filter(|name| !self.by_name.contains_key(name.as_str()))
            .next()
        {
            return self.fail(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in selected.iter() {
            if let Some(constraints) = active.get(name) {
                if constraints.iter().any(|c| !matches(*version, *c)) {
                    return self.fail(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let index = self.by_name[name.as_str()];
            let has_match = self.packages[index].releases.iter().any(|release| {
                constraints
                    .iter()
                    .all(|constraint| matches(release.version, *constraint))
            });
            if !has_match {
                return self.fail(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
        let Some(name) = next else {
            return Ok(Some(
                selected
                    .iter()
                    .map(|(package, version)| Selection {
                        package: package.clone(),
                        version: *version,
                    })
                    .collect(),
            ));
        };

        let index = self.by_name[name.as_str()];
        let constraints = &active[&name];
        let mut candidates: Vec<Version> = self.packages[index]
            .releases
            .iter()
            .filter(|release| {
                constraints
                    .iter()
                    .all(|constraint| matches(release.version, *constraint))
            })
            .map(|release| release.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            match self.inspect(selected, roots) {
                Ok(Some(solution)) => return Ok(Some(solution)),
                Ok(None) => {}
                Err(error @ ResolveError::Limit { .. }) => return Err(error),
                Err(error) => {
                    if self.first_error.is_none() {
                        self.first_error = Some(error);
                    }
                }
            }
            selected.remove(&name);
        }

        Ok(None)
    }

    fn fail(
        &mut self,
        error: ResolveError,
    ) -> Result<Option<Vec<Selection>>, ResolveError> {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
        Ok(None)
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let by_name = catalog
        .iter()
        .enumerate()
        .map(|(i, package)| (package.name.as_str(), i))
        .collect();

    let mut search = Search {
        packages: catalog,
        by_name,
        max_attempts,
        attempts: 0,
        first_error: None,
    };
    let mut selected = std::collections::BTreeMap::new();
    match search.inspect(&mut selected, roots)? {
        Some(solution) => Ok(solution),
        None => Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        })),
    }
}

fn encode(s: &str) -> String {
    let mut result = String::with_capacity(s.len());
    for byte in s.bytes() {
        match byte {
            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'.' | b'_' | b'-' => {
                result.push(byte as char)

                result.push('%');
                result.push(HEX[(byte >> 4) as usize] as char);
                result.push(HEX[(byte & 0x0f) as usize] as char);
            }
        }
    }
    result

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(required) => version == required,
        Constraint::AtLeast(required) => version >= required,
        Constraint::Before(bound) => version < bound,
        Constraint::Caret(base) => {
            version >= base
                && if base.major != 0 {
                    version.major == base.major
                } else if base.minor != 0 {
                    version.major == 0 && version.minor == base.minor
                } else {
                    version == base
                }
        }
    }
}

    constraints.iter().all(|c| matches_constraint(version, *c))

fn duplicate_checks(catalog: &[Package]) -> Result<(), ResolveError> {
    let mut package_names = std::collections::BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage { package: package.name.clone() });
        }
    }
    for package in catalog {
        let mut seen = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    Ok(())

fn active_constraints(
    roots: &[Requirement],
    assignments: &[(String, Version)],
    packages: &[Package],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();

    for root in roots {
        active.entry(root.package.clone()).or_default().push(root.constraint);
    }
    for (name, version) in assignments {
        if let Some(package) = packages.iter().find(|p| p.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active.entry(dep.package.clone()).or_default().push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        *constraints = canonical(constraints);
    }
    active
}

struct Engine<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    attempts: usize,
    budget: usize,
}

impl Engine<'_> {
    fn search(&mut self, chosen: &mut Vec<(String, Version)>) -> Result<Vec<Selection>, ResolveError> {
        let active = active_constraints(self.roots, chosen, self.catalog);

        for name in active.keys() {
            if !self.catalog.iter().any(|p| p.name == *name) {
                return Err(ResolveError::Missing { package: name.clone() });
            }
        }

        let mut selected: Vec<_> = chosen.iter().collect();
        selected.sort_by(|a, b| a.0.cmp(&b.0));
        for (name, version) in selected {
            if let Some(constraints) = active.get(name) {
                if !satisfies(*version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if !chosen.iter().any(|(n, _)| n == name) {
                let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
                if !package.releases.iter().any(|r| satisfies(r.version, constraints)) {
                    return Err(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        let next_name = active
            .keys()
            .find(|name| !chosen.iter().any(|(n, _)| n == *name))
            .cloned();

        let Some(name) = next_name else {
            let mut result: Vec<Selection> = chosen
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            result.sort_by(|a, b| a.package.cmp(&b.package));
            return Ok(result);
        };

        let constraints = &active[&name];
        let package = self.catalog.iter().find(|p| p.name == name).unwrap();
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| satisfies(*v, constraints))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        let mut earliest = None;
        for version in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit { attempts: self.budget });
            }
            self.attempts += 1;
            chosen.push((name.clone(), version));
            let child = self.search(chosen);
            chosen.pop();

            match child {
                Ok(solution) => return Ok(solution),
                Err(e @ ResolveError::Limit { .. }) => return Err(e),
                Err(error) => {
                    if earliest.is_none() {
                        earliest = Some(error);
                    }
                }
            }
        }

        Err(earliest.unwrap_or_else(|| ResolveError::NoMatch {
            package: name,
            requirements: constraints.clone(),
        }))
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate(catalog, roots)?;
    duplicate_checks(catalog)?;

    let mut engine = Engine {
        catalog,
        roots,
        attempts: 0,
        budget: max_attempts,
    };
    engine.search(&mut Vec::new())
}

fn collect_active(
    roots: &[Requirement],
    chosen: &std::collections::BTreeMap<String, Version>,
    catalog: &std::collections::BTreeMap<String, &Package>,

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: std::collections::BTreeMap<String, &'a Package>,
    attempts: usize,
    max_attempts: usize,
}

impl Search<'_> {
    fn visit(
        &mut self,
        chosen: &mut std::collections::BTreeMap<String, Version>,
    ) -> Result<Vec<Selection>, ResolveError> {
        let active = collect_active(self.roots, chosen, &self.catalog);

        if let Some(name) = active.keys().find(|n| !self.catalog.contains_key(*n)) {
            return Err(ResolveError::Missing { package: name.clone() });
        }

        for (name, version) in chosen.iter() {
            if let Some(cs) = active.get(name) {
                if !matches_all(*version, cs) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: cs.clone(),
                    });
                }
            }
        }

        for (name, cs) in &active {
            if !chosen.contains_key(name) {
                let p = self.catalog.get(name).unwrap();
                if !p.releases.iter().any(|r| matches_all(r.version, cs)) {
                    return Err(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: cs.clone(),
                    });
                }
            }
        }

        let next = active.keys().find(|n| !chosen.contains_key(*n)).cloned();
        let Some(name) = next else {
            return Ok(chosen
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect());
        };

        let cs = active.get(&name).unwrap();
        let p = self.catalog.get(&name).unwrap();
        let mut candidates: Vec<Version> = p
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| matches_all(*v, cs))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        let mut first_error = None;
        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit { attempts: self.max_attempts });
            }
            self.attempts += 1;
            chosen.insert(name.clone(), version);
            let result = self.visit(chosen);
            chosen.remove(&name);
            match result {
                Ok(solution) => return Ok(solution),
                Err(e @ ResolveError::Limit { .. }) => return Err(e),
                Err(e) => {
                    if first_error.is_none() {
                        first_error = Some(e);
                    }
                }
            }
        }
        Err(first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: name,
            requirements: cs.clone(),
        }))
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate(catalog, roots)?;
    prepare(catalog)?;
    let map = catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let mut search = Search {
        roots,
        catalog: map,
        attempts: 0,
        max_attempts,
    };
    let mut chosen = std::collections::BTreeMap::new();
    search.visit(&mut chosen)
}

fn good_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && *b.last().unwrap() != b'-'
        && b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')

fn matches_constraint(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) if x.major != 0 => v >= x && v.major == x.major,
        Constraint::Caret(x) if x.minor != 0 => v >= x && v.major == 0 && v.minor == x.minor,
        Constraint::Caret(x) => v == x,
    }
}

fn canon(v: &mut Vec<Constraint>) {
    v.sort();
    v.dedup();
}

    format!("{}.{}.{}", v.major, v.minor, v.patch)

fn cstr(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", vstr(v)),
        Constraint::AtLeast(v) => format!(">={}", vstr(v)),
        Constraint::Before(v) => format!("<{}", vstr(v)),
        Constraint::Caret(v) => format!("^{}", vstr(v)),
    }
}

fn esc(s: &str) -> String {
    let mut o = String::new();
    for b in s.bytes() {
        match b {
            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'.' | b'_' | b'-' => o.push(b as char),
            _ => o.push_str(&format!("%{:02X}", b)),
        }
    }
    o
}

fn cstrs(cs: &[Constraint]) -> String {
    let mut cs = cs.to_vec();
    canon(&mut cs);
    cs.iter().map(|c| cstr(*c)).collect::<Vec<_>>().join(",")
}

struct Engine<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    attempts: usize,
    budget: usize,
}

impl<'a> Engine<'a> {
    fn node(&mut self, chosen: &mut Vec<(String, Version, usize)>) -> Result<Vec<Selection>, ResolveError> {
        let mut constraints = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
        for r in self.roots {
            constraints.entry(r.package.clone()).or_default().push(r.constraint);
        }
        for (name, _, ri) in chosen.iter() {
            let pi = self.catalog.iter().position(|p| p.name == *name).unwrap();
            for d in &self.catalog[pi].releases[*ri].deps {
                constraints.entry(d.package.clone()).or_default().push(d.constraint);
            }
        }

        if let Some(name) = constraints.keys().find(|n| !self.catalog.iter().any(|p| p.name == ***n)) {
            return Err(ResolveError::Missing { package: name.clone() });
        }

        let mut chosen_names: Vec<&String> = chosen.iter().map(|x| &x.0).collect();
        chosen_names.sort();
        for name in chosen_names {
            let pi = self.catalog.iter().position(|p| p.name == *name).unwrap();
            let v = chosen.iter().find(|x| x.0 == *name).unwrap().1;
            let mut cs = constraints.get(name).cloned().unwrap_or_default();
            canon(&mut cs);
            if !cs.iter().all(|c| matches_constraint(*c, v)) {
                return Err(ResolveError::Conflict { package: name.clone(), selected: v, requirements: cs });
            }
            let _ = pi;
        }

        let mut pending: Option<(String, Vec<usize>)> = None;
        for (name, cs0) in &constraints {
            if chosen.iter().any(|x| x.0 == *name) { continue; }
            let pi = self.catalog.iter().position(|p| p.name == *name).unwrap();
            let mut cs = cs0.clone();
            canon(&mut cs);
            let mut candidates: Vec<usize> = self.catalog[pi].releases.iter().enumerate()
                .filter(|(_, r)| cs.iter().all(|c| matches_constraint(*c, r.version)))
                .map(|(i, _)| i).collect();
            candidates.sort_by_key(|i| std::cmp::Reverse(self.catalog[pi].releases[*i].version));
            if candidates.is_empty() {
                return Err(ResolveError::NoMatch { package: name.clone(), requirements: cs });
            }
            if pending.is_none() { pending = Some((name.clone(), candidates)); }
        }

        if let Some((name, candidates)) = pending {
            let pi = self.catalog.iter().position(|p| p.name == name).unwrap();
            let mut first = None;
            for ri in candidates {
                if self.attempts == self.budget {
                    return Err(ResolveError::Limit { attempts: self.budget });
                }
                self.attempts += 1;
                let v = self.catalog[pi].releases[ri].version;
                chosen.push((name.clone(), v, ri));
                let res = self.node(chosen);
                chosen.pop();
                match res {
                    Ok(x) => return Ok(x),
                    Err(x @ ResolveError::Limit { .. }) => return Err(x),
                    Err(x) => if first.is_none() { first = Some(x); },
                }
            }
            return Err(first.unwrap());
        }

        let mut out: Vec<Selection> = chosen.iter().map(|(n, v, _)| Selection { package: n.clone(), version: *v }).collect();
        out.sort_by(|a,b| a.package.cmp(&b.package));
        Ok(out)
    }
}

pub fn resolve(catalog: &[Package], roots: &[Requirement], max_attempts: usize) -> Result<Vec<Selection>, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !good_name(&r.package) {
            return Err(ResolveError::InvalidName { site: format!("root[{}]", i), name: r.package.clone() });
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        if !good_name(&p.name) {
            return Err(ResolveError::InvalidName { site: format!("catalog[{}]", i), name: p.name.clone() });
        }
        for (j, r) in p.releases.iter().enumerate() {
            for (k, d) in r.deps.iter().enumerate() {
                if !good_name(&d.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: d.package.clone(),
                    });
                }
            }
        }
    }
    let mut packages = std::collections::BTreeSet::new();
    for p in catalog {
        if !packages.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage { package: p.name.clone() });
        }
    }
    for p in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion { package: p.name.clone(), version: r.version });
            }
        }
    }
    Engine { catalog, roots, attempts: 0, budget: max_attempts }.node(&mut Vec::new())
}

fn valid_name(name: &str) -> bool {

    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'

fn canonical(xs: &[Constraint]) -> Vec<Constraint> {
    let mut out = xs.to_vec();
    out.sort();
    out.dedup();
    out
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || b == b'.' || b == b'_' || b == b'-' {
            out.push(b as char);
        } else {
            let hex = b"0123456789ABCDEF";
            out.push('%');
            out.push(hex[(b >> 4) as usize] as char);
            out.push(hex[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn error_text(e: &ResolveError) -> String {

        ResolveError::DuplicatePackage { package } => {

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x && if x.major != 0 { v.major == x.major }
            else if x.minor != 0 { v.major == 0 && v.minor == x.minor }
            else { v == x }
        }
    }
}

struct Solver<'a> {
    catalog: &'a [Package],
    attempts: usize,
    limit: usize,
    first: Option<ResolveError>,
}

impl Solver<'_> {
    fn search(
        &mut self,
        roots: &[Requirement],
        selected: &mut std::collections::BTreeMap<String, Version>,
    ) -> Result<Option<Vec<Selection>>, ResolveError> {
        let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
            std::collections::BTreeMap::new();
        for r in roots {
            active.entry(r.package.clone()).or_default().push(r.constraint);
        }
        for (name, version) in selected.iter() {
            let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
            let release = package.releases.iter().find(|r| r.version == *version).unwrap();
            for d in &release.deps {
                active.entry(d.package.clone()).or_default().push(d.constraint);
            }
        }
        for cs in active.values_mut() {
            *cs = canonical(cs);
        }

        if let Some(name) = active.keys().find(|name| !self.catalog.iter().any(|p| p.name == ***name)) {
            return self.record(ResolveError::Missing { package: name.clone() });
        }

        for (name, version) in selected.iter() {
            if let Some(cs) = active.get(name) {
                if cs.iter().any(|c| !matches(*version, *c)) {
                    return self.record(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: cs.clone(),
                    });
                }
            }
        }

        for (name, cs) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let p = self.catalog.iter().find(|p| p.name == *name).unwrap();
            if !p.releases.iter().any(|r| cs.iter().all(|c| matches(r.version, *c))) {
                return self.record(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: cs.clone(),
                });
            }
        }

        let next = active.keys().find(|n| !selected.contains_key(*n)).cloned();
        let Some(name) = next else {
            return Ok(Some(selected.iter().map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            }).collect()));
        };

        let p = self.catalog.iter().find(|p| p.name == name).unwrap();
        let mut candidates: Vec<Version> = p.releases.iter()
            .filter(|r| active[&name].iter().all(|c| matches(r.version, *c)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        for version in candidates {
            if self.attempts == self.limit {
                return Err(ResolveError::Limit { attempts: self.limit });
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            match self.search(roots, selected)? {
                Some(solution) => return Ok(Some(solution)),
                None => {}
            }
            selected.remove(&name);
        }

        Ok(None)
    }

    fn record(&mut self, e: ResolveError) -> Result<Option<Vec<Selection>>, ResolveError> {
        if self.first.is_none() {
            self.first = Some(e);
        }
        Ok(None)
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: p.name.clone(),
            });
        }
        for (j, r) in p.releases.iter().enumerate() {
            for (k, d) in r.deps.iter().enumerate() {
                if !valid_name(&d.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: d.package.clone(),
                    });
                }
            }
        }
    }

    let mut names = std::collections::BTreeSet::new();
    for p in catalog {
        if !names.insert(p.name.as_str()) {
            return Err(ResolveError::DuplicatePackage { package: p.name.clone() });
        }
    }
    for p in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }

    let mut solver = Solver { catalog, attempts: 0, limit: max_attempts, first: None };
    let mut selected = std::collections::BTreeMap::new();
    match solver.search(roots, &mut selected)? {
        Some(solution) => Ok(solution),
        None => Err(solver.first.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        })),
    }
}
