    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = Vec::new();
    let mut state = State::FieldStart;
    let mut record_index = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        let outside_quotes = !matches!(state, State::Quoted { .. });
        if outside_quotes && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });

            State::FieldStart => match b {

                    record.push(String::new());
                    i += 1;

                    i += 1;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    #[derive(Clone, Copy, PartialEq, Eq)]
    enum State {
        Start,
        Plain,
        Quoted,
        Closed,
    }

    let src = input.as_bytes();
    let mut pos = usize::from(input.starts_with('\u{feff}')) * 3;
    let mut rows: Vec<Vec<String>> = Vec::new();
    let mut row: Vec<String> = Vec::new();
    let mut buf: Vec<u8> = Vec::new();
    let mut state = State::Start;
    let mut row_no = 0usize;
    let mut opening = 0usize;
    let mut had_input = pos < src.len();

    while pos < src.len() {
        let ch = src[pos];

        if ch == 0 {
            return Err(CsvError::Nul {
                record: row_no,
                byte: pos,
            });
        }

        if state != State::Quoted && ch == b'\r' && src.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: row_no,
                byte: pos,
            });
        }

        match state {
            State::Start => {
                match ch {
                    b',' => {
                        row.push(String::new());
                        pos += 1;
                    }
                    b'"' => {
                        opening = pos;
                        state = State::Quoted;
                        pos += 1;
                    }
                    b'\n' | b'\r' => {
                        row.push(String::new());
                        rows.push(std::mem::take(&mut row));
                        row_no += 1;
                        pos += if ch == b'\r' { 2 } else { 1 };
                        had_input = false;
                    }
                    _ => {
                        buf.push(ch);
                        state = State::Plain;
                        pos += 1;
                    }
                }
            }
            State::Plain => {
                match ch {
                    b',' => {
                        row.push(String::from_utf8(std::mem::take(&mut buf)).unwrap());
                        state = State::Start;
                        pos += 1;
                    }
                    b'\n' | b'\r' => {
                        row.push(String::from_utf8(std::mem::take(&mut buf)).unwrap());
                        rows.push(std::mem::take(&mut row));
                        row_no += 1;
                        pos += if ch == b'\r' { 2 } else { 1 };
                        state = State::Start;
                        had_input = false;
                    }
                    b'"' => {
                        return Err(CsvError::UnexpectedQuote {
                            record: row_no,
                            byte: pos,
                        });
                    }
                    _ => {
                        buf.push(ch);
                        pos += 1;
                    }
                }
            }
            State::Quoted => {
                if ch == b'"' {
                    if src.get(pos + 1) == Some(&b'"') {
                        buf.push(b'"');
                        pos += 2;
                    } else {
                        state = State::Closed;
                        pos += 1;
                    }
                } else {
                    buf.push(ch);
                    pos += 1;
                }
            }
            State::Closed => {
                match ch {
                    b',' => {
                        row.push(String::from_utf8(std::mem::take(&mut buf)).unwrap());
                        state = State::Start;
                        pos += 1;
                    }
                    b'\n' | b'\r' => {
                        row.push(String::from_utf8(std::mem::take(&mut buf)).unwrap());
                        rows.push(std::mem::take(&mut row));
                        row_no += 1;
                        pos += if ch == b'\r' { 2 } else { 1 };
                        state = State::Start;
                        had_input = false;
                    }
                    _ => {
                        return Err(CsvError::TrailingByte {
                            record: row_no,
                            byte: pos,
                        });
                    }
                }
            }
        }

        if pos < src.len() {
            had_input = true;
        }
    }

    if state == State::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: row_no,
            byte: opening,
        });
    }

    if had_input {
        row.push(String::from_utf8(buf).unwrap());
        rows.push(row);
    }

    Ok(rows)
}
