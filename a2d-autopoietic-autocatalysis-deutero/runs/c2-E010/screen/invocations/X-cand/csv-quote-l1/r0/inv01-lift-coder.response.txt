- Record indexes count only outside-quote LF or CRLF terminators already consumed.
- The initial BOM is ignored, but its three UTF-8 bytes still count in error offsets.
- NUL is checked first at each byte; outside quotes, bare CR is checked before trailing-byte rules.
- A pair of quotes inside a quoted field is always an escape, even at EOF; an unmatched opening quote reports its opening byte.
- Empty input yields no records; a terminator creates a record, but a final terminator creates no extra record.
- Both versions preserve field bytes exactly and use checked UTF-8 slices; B uses a byte-state machine with field buffers.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = Vec::new();
    let mut state = State::FieldStart;
    let mut record_index = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        let outside_quotes = !matches!(state, State::Quoted { .. });
        if outside_quotes && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        match state {
            State::FieldStart => match b {
                b',' => {
                    record.push(String::new());
                    i += 1;
                }
                b'"' => {
                    state = State::Quoted { opening: i };
                    i += 1;
                }
                b'\n' => {
                    record.push(String::new());
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    i += 1;
                }
                b'\r' => {
                    record.push(String::new());
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    i += 2;
                }
                _ => {
                    field.push(b);
                    state = State::Unquoted;
                    i += 1;
                }
            },
            State::Unquoted => match b {
                b',' => {
                    record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    state = State::FieldStart;
                    i += 1;
                }
                b'\n' => {
                    record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    i += 1;
                }
                b'\r' => {
                    record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    i += 2;
                }
                b'"' => {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                }
                _ => {
                    field.push(b);
                    i += 1;
                }
            },
            State::Quoted { opening } => {
                if b == b'"' {
                    if bytes.get(i + 1) == Some(&b'"') {
                        field.push(b'"');
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    field.push(b);
                    i += 1;
                }
                let _ = opening;
            }
            State::AfterQuote => match b {
                b',' => {
                    record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    state = State::FieldStart;
                    i += 1;
                }
                b'\n' => {
                    record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    i += 1;
                }
                b'\r' => {
                    record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    i += 2;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            },
        }
    }

    match state {
        State::Quoted { opening } => {
            return Err(CsvError::UnclosedQuote {
                record: record_index,
                byte: opening,
            });
        }
        State::Unquoted | State::AfterQuote => {
            record.push(String::from_utf8(field).unwrap());
            records.push(record);
        }
        State::FieldStart => {
            if !record.is_empty() {
                record.push(String::new());
                records.push(record);
            }
        }
    }

    Ok(records)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    #[derive(Clone, Copy, PartialEq, Eq)]
    enum State {
        Start,
        Plain,
        Quoted,
        Closed,
    }

    let src = input.as_bytes();
    let mut pos = usize::from(input.starts_with('\u{feff}')) * 3;
    let mut rows: Vec<Vec<String>> = Vec::new();
    let mut row: Vec<String> = Vec::new();
    let mut buf: Vec<u8> = Vec::new();
    let mut state = State::Start;
    let mut row_no = 0usize;
    let mut opening = 0usize;
    let mut had_input = pos < src.len();

    while pos < src.len() {
        let ch = src[pos];

        if ch == 0 {
            return Err(CsvError::Nul {
                record: row_no,
                byte: pos,
            });
        }

        if state != State::Quoted && ch == b'\r' && src.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: row_no,
                byte: pos,
            });
        }

        match state {
            State::Start => {
                match ch {
                    b',' => {
                        row.push(String::new());
                        pos += 1;
                    }
                    b'"' => {
                        opening = pos;
                        state = State::Quoted;
                        pos += 1;
                    }
                    b'\n' | b'\r' => {
                        row.push(String::new());
                        rows.push(std::mem::take(&mut row));
                        row_no += 1;
                        pos += if ch == b'\r' { 2 } else { 1 };
                        had_input = false;
                    }
                    _ => {
                        buf.push(ch);
                        state = State::Plain;
                        pos += 1;
                    }
                }
            }
            State::Plain => {
                match ch {
                    b',' => {
                        row.push(String::from_utf8(std::mem::take(&mut buf)).unwrap());
                        state = State::Start;
                        pos += 1;
                    }
                    b'\n' | b'\r' => {
                        row.push(String::from_utf8(std::mem::take(&mut buf)).unwrap());
                        rows.push(std::mem::take(&mut row));
                        row_no += 1;
                        pos += if ch == b'\r' { 2 } else { 1 };
                        state = State::Start;
                        had_input = false;
                    }
                    b'"' => {
                        return Err(CsvError::UnexpectedQuote {
                            record: row_no,
                            byte: pos,
                        });
                    }
                    _ => {
                        buf.push(ch);
                        pos += 1;
                    }
                }
            }
            State::Quoted => {
                if ch == b'"' {
                    if src.get(pos + 1) == Some(&b'"') {
                        buf.push(b'"');
                        pos += 2;
                    } else {
                        state = State::Closed;
                        pos += 1;
                    }
                } else {
                    buf.push(ch);
                    pos += 1;
                }
            }
            State::Closed => {
                match ch {
                    b',' => {
                        row.push(String::from_utf8(std::mem::take(&mut buf)).unwrap());
                        state = State::Start;
                        pos += 1;
                    }
                    b'\n' | b'\r' => {
                        row.push(String::from_utf8(std::mem::take(&mut buf)).unwrap());
                        rows.push(std::mem::take(&mut row));
                        row_no += 1;
                        pos += if ch == b'\r' { 2 } else { 1 };
                        state = State::Start;
                        had_input = false;
                    }
                    _ => {
                        return Err(CsvError::TrailingByte {
                            record: row_no,
                            byte: pos,
                        });
                    }
                }
            }
        }

        if pos < src.len() {
            had_input = true;
        }
    }

    if state == State::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: row_no,
            byte: opening,
        });
    }

    if had_input {
        row.push(String::from_utf8(buf).unwrap());
        rows.push(row);
    }

    Ok(rows)
}
```
