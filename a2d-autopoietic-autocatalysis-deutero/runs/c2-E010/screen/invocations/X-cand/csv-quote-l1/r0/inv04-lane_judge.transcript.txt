#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    #[derive(Clone, Copy, PartialEq, Eq)]
    enum State {
        Start,
        Plain,
        Quoted,
        Closed,
    }

    let bytes = input.as_bytes();
    let mut pos = usize::from(input.starts_with('\u{feff}')) * 3;
    let mut rows = Vec::new();
    let mut row = Vec::new();
    let mut value = String::new();
    let mut state = State::Start;
    let mut opening = 0usize;

    if pos == bytes.len() {
        return Ok(rows);
    }

    while pos < bytes.len() {
        let byte = bytes[pos];
        let row_index = rows.len();

        if byte == 0 {
            return Err(CsvError::Nul { record: row_index, byte: pos });
        }

        if state != State::Quoted && byte == b'\r' {
            if bytes.get(pos + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr { record: row_index, byte: pos });
            }
            row.push(std::mem::take(&mut value));
            rows.push(std::mem::take(&mut row));
            state = State::Start;
            pos += 2;
            continue;
        }

        match state {
            State::Start => match byte {
                b'"' => {
                    opening = pos;
                    state = State::Quoted;
                    pos += 1;
                }
                b',' => {
                    row.push(String::new());
                    pos += 1;
                }
                b'\n' => {
                    row.push(String::new());
                    rows.push(std::mem::take(&mut row));
                    pos += 1;
                }
                _ => {
                    let c = input[pos..].chars().next().unwrap();
                    value.push(c);
                    state = State::Plain;
                    pos += c.len_utf8();
                }
            },
            State::Plain => match byte {
                b',' => {
                    row.push(std::mem::take(&mut value));
                    state = State::Start;
                    pos += 1;
                }
                b'\n' => {
                    row.push(std::mem::take(&mut value));
                    rows.push(std::mem::take(&mut row));
                    state = State::Start;
                    pos += 1;
                }
                b'"' => {
                    return Err(CsvError::UnexpectedQuote { record: row_index, byte: pos });
                }
                _ => {
                    let c = input[pos..].chars().next().unwrap();
                    value.push(c);
                    pos += c.len_utf8();
                }
            },
            State::Quoted => {
                if byte == b'"' {
                    if bytes.get(pos + 1) == Some(&b'"') {
                        value.push('"');
                        pos += 2;
                    } else {
                        state = State::Closed;
                        pos += 1;
                    }
                } else {
                    let c = input[pos..].chars().next().unwrap();
                    value.push(c);
                    pos += c.len_utf8();
                }
            }
            State::Closed => match byte {
                b',' => {
                    row.push(std::mem::take(&mut value));
                    state = State::Start;
                    pos += 1;
                }
                b'\n' => {
                    row.push(std::mem::take(&mut value));
                    rows.push(std::mem::take(&mut row));
                    state = State::Start;
                    pos += 1;
                }
                _ => {
                    return Err(CsvError::TrailingByte { record: row_index, byte: pos });
                }
            },
        }
    }

    if state == State::Quoted {
        return Err(CsvError::UnclosedQuote { record: rows.len(), byte: opening });
    }

    if !row.is_empty() || !value.is_empty() || state != State::Start {
        row.push(value);
        rows.push(row);
    }

    Ok(rows)
}

                }
            }
            State::Quoted => {
                if ch == b'"' {
                    if src.get(pos + 1) == Some(&b'"') {
                        buf.push(b'"');
                        pos += 2;
                    } else {
                        state = State::Closed;
                        pos += 1;
                    }
                } else {
                    buf.push(ch);
                    pos += 1;
                }
            }
            State::Closed => {
                match ch {
                    b',' => {
                        row.push(String::from_utf8(std::mem::take(&mut buf)).unwrap());
                        state = State::Start;
                        pos += 1;
                    }
                    b'\n' | b'\r' => {
                        row.push(String::from_utf8(std::mem::take(&mut buf)).unwrap());
                        rows.push(std::mem::take(&mut row));
                        row_no += 1;
                        pos += if ch == b'\r' { 2 } else { 1 };
                        state = State::Start;
                        had_input = false;
                    }
                    _ => {
                        return Err(CsvError::TrailingByte {
                            record: row_no,
                            byte: pos,
                        });
                    }
                }
            }
        }

        if pos < src.len() {
            had_input = true;
        }
    }

    if state == State::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: row_no,
            byte: opening,
        });
    }

    if had_input {
        row.push(String::from_utf8(buf).unwrap());
        rows.push(row);
    }

    Ok(rows)
}
