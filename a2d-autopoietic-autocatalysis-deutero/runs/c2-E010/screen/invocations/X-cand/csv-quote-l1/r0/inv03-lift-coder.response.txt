- Errors use absolute UTF-8 byte offsets; the ignored initial BOM still counts toward offsets.
- Error record numbers count only terminators outside quotes already consumed.
- A quote is legal only at the first byte of a field; after a closing quote, only delimiter, record ending, or EOF is legal.
- NUL is checked before other byte-specific errors; outside quotes, bare CR is checked before trailing-byte errors.
- A doubled quote at EOF is an escape, so the quoted field remains unclosed.
- Empty input yields no records, while each consumed terminator yields a record and EOF adds a record only for unfinished input.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) { 3 } else { 0 };

    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut record = 0usize;
    let mut field_start = true;
    let mut quoted = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;
    let mut touched = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !quoted && b == b'\r' && (i + 1 >= bytes.len() || bytes[i + 1] != b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if quoted {
            if b == b'"' {
                if i + 1 < bytes.len() && bytes[i + 1] == b'"' {
                    field.push('"');
                    i += 2;
                } else {
                    quoted = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                i += ch.len_utf8();
            }
            touched = true;
            continue;
        }

        if after_quote {
            match b {
                b',' => {
                    row.push(std::mem::take(&mut field));
                    field_start = true;
                    after_quote = false;
                    touched = true;
                    i += 1;
                }
                b'\n' => {
                    row.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut row));
                    record += 1;
                    field_start = true;
                    after_quote = false;
                    touched = false;
                    i += 1;
                }
                b'\r' => {
                    row.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut row));
                    record += 1;
                    field_start = true;
                    after_quote = false;
                    touched = false;
                    i += 2;
                }
                _ => return Err(CsvError::TrailingByte { record, byte: i }),
            }
            continue;
        }

        match b {
            b',' => {
                row.push(std::mem::take(&mut field));
                field_start = true;
                touched = true;
                i += 1;
            }
            b'\n' => {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                record += 1;
                field_start = true;
                touched = false;
                i += 1;
            }
            b'\r' => {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                record += 1;
                field_start = true;
                touched = false;
                i += 2;
            }
            b'"' if field_start => {
                quoted = true;
                quote_open = i;
                field_start = false;
                touched = true;
                i += 1;
            }
            b'"' => return Err(CsvError::UnexpectedQuote { record, byte: i }),
            _ => {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                field_start = false;
                touched = true;
                i += ch.len_utf8();
            }
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_open,
        });
    }

    if touched || !row.is_empty() || !field.is_empty() || after_quote {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) { 3 } else { 0 };
    if pos == bytes.len() {
        return Ok(Vec::new());
    }

    let mut result = Vec::new();
    let mut current_row = Vec::new();
    let mut current_field = String::new();
    let mut record = 0usize;
    let mut in_quotes = false;
    let mut just_closed_quote = false;
    let mut at_field_start = true;
    let mut opening_quote = 0usize;
    let mut record_has_content = false;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == 0 {
            return Err(CsvError::Nul { record, byte: pos });
        }

        if !in_quotes
            && byte == b'\r'
            && (pos + 1 == bytes.len() || bytes[pos + 1] != b'\n')
        {
            return Err(CsvError::BareCr { record, byte: pos });
        }

        if in_quotes {
            if byte == b'"' {
                if bytes.get(pos + 1) == Some(&b'"') {
                    current_field.push('"');
                    pos += 2;
                } else {
                    in_quotes = false;
                    just_closed_quote = true;
                    pos += 1;
                }
            } else {
                let ch = input[pos..].chars().next().unwrap();
                current_field.push(ch);
                pos += ch.len_utf8();
            }
            record_has_content = true;
            continue;
        }

        if just_closed_quote {
            match byte {
                b',' => {
                    current_row.push(std::mem::take(&mut current_field));
                    just_closed_quote = false;
                    at_field_start = true;
                    record_has_content = true;
                    pos += 1;
                    continue;
                }
                b'\n' | b'\r' => {
                    current_row.push(std::mem::take(&mut current_field));
                    result.push(std::mem::take(&mut current_row));
                    record += 1;
                    just_closed_quote = false;
                    at_field_start = true;
                    record_has_content = false;
                    pos += if byte == b'\r' { 2 } else { 1 };
                    continue;
                }
                _ => return Err(CsvError::TrailingByte { record, byte: pos }),
            }
        }

        match byte {
            b',' => {
                current_row.push(std::mem::take(&mut current_field));
                at_field_start = true;
                record_has_content = true;
                pos += 1;
            }
            b'\n' | b'\r' => {
                current_row.push(std::mem::take(&mut current_field));
                result.push(std::mem::take(&mut current_row));
                record += 1;
                at_field_start = true;
                record_has_content = false;
                pos += if byte == b'\r' { 2 } else { 1 };
            }
            b'"' if at_field_start => {
                in_quotes = true;
                opening_quote = pos;
                at_field_start = false;
                record_has_content = true;
                pos += 1;
            }
            b'"' => return Err(CsvError::UnexpectedQuote { record, byte: pos }),
            _ => {
                let ch = input[pos..].chars().next().unwrap();
                current_field.push(ch);
                at_field_start = false;
                record_has_content = true;
                pos += ch.len_utf8();
            }
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: opening_quote,
        });
    }

    if record_has_content || !current_row.is_empty() || !current_field.is_empty() || just_closed_quote {
        current_row.push(current_field);
        result.push(current_row);
    }

    Ok(result)
}
```
