#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        3
    } else {
        0
    };

    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut output = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut row_number = 0usize;
    let mut at_field_start = true;
    let mut quoted = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: row_number,
                byte: i,
            });
        }

        if !quoted && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: row_number,
                byte: i,
            });
        }

        if quoted {
            if b == b'"' {
                if bytes.get(i + 1) == Some(&b'"') {
                    field.push('"');
                    i += 2;
                } else {
                    quoted = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            match b {
                b',' => {
                    row.push(std::mem::take(&mut field));
                    at_field_start = true;
                    after_quote = false;
                    i += 1;
                }
                b'\n' => {
                    row.push(std::mem::take(&mut field));
                    output.push(std::mem::take(&mut row));
                    row_number += 1;
                    at_field_start = true;
                    after_quote = false;
                    i += 1;
                }
                b'\r' => {
                    row.push(std::mem::take(&mut field));
                    output.push(std::mem::take(&mut row));
                    row_number += 1;
                    at_field_start = true;
                    after_quote = false;
                    i += 2;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: row_number,
                        byte: i,
                    });
                }
            }
            continue;
        }

        match b {
            b',' => {
                row.push(std::mem::take(&mut field));
                at_field_start = true;
                i += 1;
            }
            b'\n' => {
                row.push(std::mem::take(&mut field));
                output.push(std::mem::take(&mut row));
                row_number += 1;
                at_field_start = true;
                i += 1;
            }
            b'\r' => {
                row.push(std::mem::take(&mut field));
                output.push(std::mem::take(&mut row));
                row_number += 1;
                at_field_start = true;
                i += 2;
            }
            b'"' if at_field_start => {
                quoted = true;
                quote_open = i;
                at_field_start = false;
                i += 1;
            }
            b'"' => {
                return Err(CsvError::UnexpectedQuote {
                    record: row_number,
                    byte: i,
                });
            }
            _ => {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                at_field_start = false;
                i += ch.len_utf8();
            }
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote {
            record: row_number,
            byte: quote_open,
        });
    }

    if after_quote || !at_field_start {
        row.push(field);
        output.push(row);
    } else if i > 0 && bytes[i - 1] == b',' {
        row.push(String::new());
        output.push(row);
    }

    Ok(output)
}
