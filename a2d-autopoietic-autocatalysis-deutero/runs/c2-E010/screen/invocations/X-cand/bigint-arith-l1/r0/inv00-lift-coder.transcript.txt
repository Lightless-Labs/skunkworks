### Solution A

- Error precedence: invalid bytes first, then malformed separators, leading zero, and digit limit.
- Error offsets are UTF-8 byte offsets in the original input, including a leading `+`.
- Underscores are valid only between ASCII digits; digit limits exclude signs and underscores.
- Zero has one digit, and `max_digits = 0` rejects it.
- Formatting pads before grouping; grouping counts digits from the right.
- Big integers use little-endian base-10 digits, with zero represented canonically.

```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b != b'_' && !b.is_ascii_digit() {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let prev_digit = i > start && bytes[i - 1].is_ascii_digit();
                let next_digit = i + 1 < bytes.len() && bytes[i + 1].is_ascii_digit();
                if !prev_digit || !next_digit {
                    separator = Some(i);
                    break;
                }
            }
        }
        if let Some(index) = separator {
            return Err(Error::Separator { index });
        }

        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero { index: first_digit_index });
        }

        if digit_count > max_digits {
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen > max_digits {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &b in bytes[start..].iter().rev() {
            if b != b'_' {
                digits.push(b - b'0');
            }
        }
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let natural_len = self.digits.len().max(1);
        let width = natural_len.max(min_digits);
        let pad = width - natural_len;
        let mut out = String::with_capacity(width + if group > 0 { width / group } else { 0 });

        for pos in (0..width).rev() {
            if group > 0 && pos < width && pos != width - 1 && (pos + 1) % group == 0 {
                out.push('_');
            }
            let digit = if pos < pad { 0 } else { self.digits[pos - pad] };
            out.push((b'0' + digit) as char);
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => {
                for i in (0..self.digits.len()).rev() {
                    match self.digits[i].cmp(&rhs.digits[i]) {
                        Ordering::Equal => {}
                        ord => return ord,
                    }
                }
                Ordering::Equal
            }
            ord => ord,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let n = self.digits.len().max(rhs.digits.len());
        let mut digits = Vec::with_capacity(n + 1);
        let mut carry = 0u8;
        for i in 0..n {
            let a = self.digits.get(i).copied().unwrap_or(0);
            let b = rhs.digits.get(i).copied().unwrap_or(0);
            let sum = a + b + carry;
            digits.push(sum % 10);
            carry = sum / 10;
        }
        if carry != 0 {
            digits.push(carry);
        }
        Self { digits }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut digits = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i8;
        for i in 0..self.digits.len() {
            let a = self.digits[i] as i8 - borrow;
            let b = rhs.digits.get(i).copied().unwrap_or(0) as i8;
            if a < b {
                digits.push((a + 10 - b) as u8);
                borrow = 1;
            } else {
                digits.push((a - b) as u8);
                borrow = 0;
            }
        }
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        Ok(Self { digits })
    }
```

### Solution B

- Error precedence and byte-offset rules follow the same explicit order as Solution A.
- Parsing accepts `+` only at byte zero; a lone `+` is `SignOnly`.
- Zero is canonical and counts as one digit for the limit.
- Padding is applied before right-counted grouping.
- Big integers use little-endian base-1,000,000,000 limbs in this independent implementation.

```rust
use std::cmp::Ordering;

const BASE: u64 = 1_000_000_000;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    limbs: Vec<u32>,
}

impl Big {
    fn zero() -> Self {
        Self { limbs: vec![0] }
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
    }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = usize::from(bytes[0] == b'+');
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b != b'_' && !b.is_ascii_digit() {
                return Err(Error::InvalidDigit { index: i });
            }
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                if i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit()
                {
                    return Err(Error::Separator { index: i });
                }
            }
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if digit_count > 1 && bytes[first] == b'0' {
            return Err(Error::LeadingZero { index: first });
        }
        if digit_count > max_digits {
            let mut count = 0;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    count += 1;
                    if count > max_digits {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut result = Self::zero();
        for &b in &bytes[start..] {
            if b == b'_' {
                continue;
            }
            let mut carry = (b - b'0') as u64;
            for limb in &mut result.limbs {
                let value = (*limb as u64) * 10 + carry;
                *limb = (value % BASE) as u32;
                carry = value / BASE;
            }
            if carry != 0 {
                result.limbs.push(carry as u32);
            }
        }
        result.normalize();
        Ok(result)
    }

    fn decimal_digits(&self) -> Vec<u8> {
        let mut n = self.clone();
        if n.limbs.iter().all(|&x| x == 0) {
            return vec![0];
        }
        let mut reversed = Vec::new();
        while n.limbs.iter().any(|&x| x != 0) {
            let mut rem = 0u64;
            for limb in n.limbs.iter_mut().rev() {
                let value = rem * BASE + *limb as u64;
                *limb = (value / 10) as u32;
                rem = value % 10;
            }
            reversed.push(rem as u8);
            n.normalize();
        }
        reversed
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let digits = self.decimal_digits();
        let width = digits.len().max(min_digits);
        let padding = width - digits.len();
        let mut out = String::with_capacity(width + if group > 0 { width / group } else { 0 });
        for pos in 0..width {
            if group > 0 && pos > 0 && (width - pos) % group == 0 {
                out.push('_');
            }
            if pos < padding {
                out.push('0');
            } else {
                out.push((b'0' + digits[digits.len() - 1 - (pos - padding)]) as char);
            }
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        ord => return ord,
                    }
                }
                Ordering::Equal
            }
            ord => ord,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let n = self.limbs.len().max(rhs.limbs.len());
        let mut limbs = Vec::with_capacity(n + 1);
        let mut carry = 0u64;
        for i in 0..n {
            let sum = self.limbs.get(i).copied().unwrap_or(0) as u64
                + rhs.limbs.get(i).copied().unwrap_or(0) as u64
                + carry;
            limbs.push((sum % BASE) as u32);
            carry = sum / BASE;
        }
        if carry != 0 {
            limbs.push(carry as u32);
        }
        Self { limbs }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }
        let mut limbs = Vec::with_capacity(self.limbs.len());
        let mut borrow = 0i64;
        for i in 0..self.limbs.len() {
            let a = self.limbs[i] as i64 - borrow;
            let b = rhs.limbs.get(i).copied().unwrap_or(0) as i64;
            if a < b {
                limbs.push((a + BASE as i64 - b) as u32);
                borrow = 1;
            } else {
                limbs.push((a - b) as u32);
                borrow = 0;
            }
        }
        let mut result = Self { limbs };
        result.normalize();
        Ok(result)
    }
}
```
