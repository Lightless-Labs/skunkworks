    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    limbs: Vec<u32>,
}

impl Big {
    fn zero() -> Self {
        Self { limbs: vec![0] }
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
    }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = usize::from(bytes[0] == b'+');
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b != b'_' && !b.is_ascii_digit() {
                return Err(Error::InvalidDigit { index: i });
            }
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                if i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit()
                {
                    return Err(Error::Separator { index: i });
                }
            }
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if digit_count > 1 && bytes[first] == b'0' {
            return Err(Error::LeadingZero { index: first });
        }
        if digit_count > max_digits {
            let mut count = 0;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    count += 1;
                    if count > max_digits {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut result = Self::zero();
        for &b in &bytes[start..] {
            if b == b'_' {
                continue;
            }
            let mut carry = (b - b'0') as u64;
            for limb in &mut result.limbs {
                let value = (*limb as u64) * 10 + carry;
                *limb = (value % BASE) as u32;
                carry = value / BASE;
            }
            if carry != 0 {
                result.limbs.push(carry as u32);
            }
        }
        result.normalize();
        Ok(result)
    }

            return vec![0];

                rem = value % 10;
            }
            reversed.push(rem as u8);
            n.normalize();
        }
        reversed
    }

        let padding = width - digits.len();

                out.push('0');
            } else {
                out.push((b'0' + digits[digits.len() - 1 - (pos - padding)]) as char);

        }

                    match a.cmp(b) {

                        ord => return ord,
                    }
                }
                Ordering::Equal
            }
            ord => ord,
        }
    }

        let mut carry = 0u64;
        for i in 0..n {
            let sum = self.limbs.get(i).copied().unwrap_or(0) as u64
                + rhs.limbs.get(i).copied().unwrap_or(0) as u64
                + carry;
            limbs.push((sum % BASE) as u32);
            carry = sum / BASE;

        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for (i, &b) in bytes.iter().enumerate().skip(start) {
            if b != b'_' && !b.is_ascii_digit() {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let prev_digit = i > start && bytes[i - 1].is_ascii_digit();
                let next_digit = i + 1 < bytes.len() && bytes[i + 1].is_ascii_digit();
                if !prev_digit || !next_digit {
                    separator = Some(i);
                    break;
                }
            }
        }
        if let Some(index) = separator {
            return Err(Error::Separator { index });
        }

        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero { index: first_digit_index });
        }

        if digit_count > max_digits {
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    seen += 1;
                    if seen > max_digits {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &b in bytes[start..].iter().rev() {
            if b != b'_' {
                digits.push(b - b'0');
            }
        }
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        Ok(Self { digits })

            out.push((b'0' + digit) as char);

pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Little-endian base-10 digits. Zero is represented by [0].
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = usize::from(bytes[0] == b'+');
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes have priority over every other lexical error.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        // Once all bytes are known to be digits or underscores, check separators.
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index });
            }
        }

        let first_digit_index = (start..bytes.len())
            .find(|&index| bytes[index].is_ascii_digit())
            .expect("nonempty magnitude with valid separators");

        let digit_count = bytes[start..]
            .iter()
            .filter(|&&byte| byte.is_ascii_digit())
            .count();

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_digit_number = max_digits;
            let index = bytes[start..]
                .iter()
                .enumerate()
                .filter(|(_, byte)| byte.is_ascii_digit())
                .nth(excess_digit_number)
                .map(|(offset, _)| start + offset)
                .expect("digit count exceeds limit");
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes[start..].iter().rev() {
            if byte.is_ascii_digit() {
                digits.push(byte - b'0');
            }
        }

        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let natural_len = self.digits.len().max(1);
        let width = natural_len.max(min_digits);
        let padding = width - self.digits.len();

        let mut out = String::with_capacity(
            width + if group > 0 { width.saturating_sub(1) / group } else { 0 },
        );

        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                out.push('_');
            }

            if position < padding {
                out.push('0');
            } else {
                let digit_index = self.digits.len() - 1 - (position - padding);
                out.push(char::from(b'0' + self.digits[digit_index]));
            }
        }

        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> std::cmp::Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            std::cmp::Ordering::Equal => {
                for (left, right) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match left.cmp(right) {
                        std::cmp::Ordering::Equal => {}
                        ordering => return ordering,
                    }
                }
                std::cmp::Ordering::Equal
            }
            ordering => ordering,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let max_len = self.digits.len().max(rhs.digits.len());
        let mut result = Vec::with_capacity(max_len.saturating_add(1));
        let mut carry = 0u8;

        for index in 0..max_len {
            let left = self.digits.get(index).copied().unwrap_or(0);
            let right = rhs.digits.get(index).copied().unwrap_or(0);
            let sum = left + right + carry;
            result.push(sum % 10);
            carry = sum / 10;
        }

        if carry != 0 {
            result.push(carry);
        }

        Self { digits: result }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == std::cmp::Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut result = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i16;

        for index in 0..self.digits.len() {
            let left = i16::from(self.digits[index]) - borrow;
            let right = i16::from(rhs.digits.get(index).copied().unwrap_or(0));

            if left < right {
                result.push((left + 10 - right) as u8);
                borrow = 1;
            } else {
                result.push((left - right) as u8);
                borrow = 0;
            }
        }

        while result.len() > 1 && result.last() == Some(&0) {
            result.pop();
        }

        Ok(Self { digits: result })
    }
}

            limbs.push(value as u32);
        }

        // Parsing consumed digits left-to-right into big-endian limbs. If the
        // first limb was full, the remaining partial limb belongs at the end.
        // The loop above already preserves that order.
        if limbs.is_empty() {
            limbs.push(0);
        }

        Ok(Self { limbs })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let mut raw = String::new();
        if let Some(first) = self.limbs.first() {
            raw.push_str(&first.to_string());
            for limb in self.limbs.iter().skip(1) {
                raw.push_str(&format!("{limb:09}"));
            }
        } else {
            raw.push('0');
        }

        let width = raw.len().max(min_digits).max(1);
        let padding = width - raw.len();
        let mut out = String::with_capacity(
            width + if group > 0 { width.saturating_sub(1) / group } else { 0 },
        );

        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                out.push('_');
            }
            if position < padding {
                out.push('0');
            } else {
                out.push(raw.as_bytes()[position - padding] as char);
            }
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => self.limbs.cmp(&rhs.limbs),
            ordering => ordering,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut result = Vec::with_capacity(self.limbs.len().max(rhs.limbs.len()) + 1);
        let mut carry = 0u64;
        let mut left = self.limbs.iter().rev();
        let mut right = rhs.limbs.iter().rev();

        loop {
            let a = left.next().copied();
            let b = right.next().copied();
            if a.is_none() && b.is_none() {
                break;
            }

            let sum = u64::from(a.unwrap_or(0)) + u64::from(b.unwrap_or(0)) + carry;
            result.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }

        if carry != 0 {
            result.push(carry as u32);
        }
        result.reverse();
        Self { limbs: result }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut result = Vec::with_capacity(self.limbs.len());
        let mut borrow = 0i64;
        let mut right = rhs.limbs.iter().rev();

        for &left_limb in self.limbs.iter().rev() {
            let right_limb = i64::from(right.next().copied().unwrap_or(0));
            let mut difference = i64::from(left_limb) - right_limb - borrow;
            if difference < 0 {
                difference += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            result.push(difference as u32);
        }

        result.reverse();
        let first_nonzero = result.iter().position(|&limb| limb != 0);
        let limbs = match first_nonzero {
            Some(index) => result.split_off(index),
            None => vec![0],
        };

        Ok(Self { limbs })
    }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Big-endian decimal digits; zero is [0].
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let b = text.as_bytes();
        if b.is_empty() {
            return Err(Error::Empty);
        }
        let offset = usize::from(b[0] == b'+');
        if offset == b.len() {
            return Err(Error::SignOnly);
        }

        let invalid = (offset..b.len())
            .find(|&i| b[i] != b'_' && !(b'0'..=b'9').contains(&b[i]));
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for i in offset..b.len() {
            if b[i] == b'_'
                && (i == offset
                    || i + 1 == b.len()
                    || !(b'0'..=b'9').contains(&b[i - 1])
                    || !(b'0'..=b'9').contains(&b[i + 1]))
            {
                return Err(Error::Separator { index: i });
            }
        }

        let count = b[offset..].iter().filter(|&&x| x != b'_').count();
        let first = (offset..b.len()).find(|&i| b[i] != b'_').unwrap();
        if count > 1 && b[first] == b'0' {
            return Err(Error::LeadingZero { index: first });
        }
        if count > max_digits {
            let index = (offset..b.len())
                .filter(|&i| b[i] != b'_')
                .nth(max_digits)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let digits = b[offset..]
            .iter()
            .copied()
            .filter(|&x| x != b'_')
            .map(|x| x - b'0')
            .collect();
        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let natural = self.digits.len().max(1);
        let width = natural.max(min_digits);
        let mut out = String::with_capacity(width.saturating_add(width.saturating_sub(1) / group.max(1)));
        let padding = width - self.digits.len();
        for i in 0..width {
            if group > 0 && i > 0 && (width - i) % group == 0 {
                out.push('_');
            }
            if i < padding {
                out.push('0');
            } else {
                out.push(char::from(b'0' + self.digits[i - padding]));
            }
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        self.digits.len().cmp(&rhs.digits.len()).then_with(|| self.digits.cmp(&rhs.digits))
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut i = self.digits.len();
        let mut j = rhs.digits.len();
        let mut carry = 0u8;
        let mut rev = Vec::with_capacity(i.max(j) + 1);
        while i > 0 || j > 0 || carry > 0 {
            let a = if i > 0 { i -= 1; self.digits[i] } else { 0 };
            let b = if j > 0 { j -= 1; rhs.digits[j] } else { 0 };
            let sum = a + b + carry;
            rev.push(sum % 10);
            carry = sum / 10;
        }
        rev.reverse();
        Self { digits: rev }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }
        let mut i = self.digits.len();
        let mut j = rhs.digits.len();
        let mut borrow = 0i8;
        let mut rev = Vec::with_capacity(i);
        while i > 0 {
            i -= 1;
            let a = self.digits[i] as i8 - borrow;
            let b = if j > 0 { j -= 1; rhs.digits[j] as i8 } else { 0 };
            if a < b {
                rev.push((a + 10 - b) as u8);
                borrow = 1;
            } else {
                rev.push((a - b) as u8);
                borrow = 0;
            }
        }
        while rev.len() > 1 && rev.last() == Some(&0) {
            rev.pop();
        }
        rev.reverse();
        while rev.len() > 1 && rev[0] == 0 {
            rev.remove(0);
        }
        Ok(Self { digits: rev })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index });
            }
        }

        let first_digit_index = if start < bytes.len() && bytes[start] == b'_' {
            start + 1
        } else {
            start
        };
        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let target = max_digits + 1;
            let mut seen = 0usize;
            for (index, &byte) in bytes.iter().enumerate().skip(start) {
                if byte.is_ascii_digit() {
                    seen += 1;
                    if seen == target {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index,
                        });
                    }
                }
            }
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes[start..].iter().rev() {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }

        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let actual = self.digits.len().max(1);
        let width = actual.max(min_digits);
        let padding = width - actual;
        let mut out = String::with_capacity(width + if group > 0 { width / group } else { 0 });

        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                out.push('_');
            }

            if position < padding {
                out.push('0');
            } else {
                let from_left = position - padding;
                let digit_index = actual - 1 - from_left;
                out.push((b'0' + self.digits[digit_index]) as char);
            }
        }

        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> std::cmp::Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            std::cmp::Ordering::Equal => {
                for (left, right) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match left.cmp(right) {
                        std::cmp::Ordering::Equal => {}
                        other => return other,
                    }
                }
                std::cmp::Ordering::Equal
            }
            other => other,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut digits = Vec::with_capacity(self.digits.len().max(rhs.digits.len()) + 1);
        let mut carry = 0u8;
        let length = self.digits.len().max(rhs.digits.len());

        for index in 0..length {
            let left = self.digits.get(index).copied().unwrap_or(0);
            let right = rhs.digits.get(index).copied().unwrap_or(0);
            let sum = left + right + carry;
            digits.push(sum % 10);
            carry = sum / 10;
        }

        if carry != 0 {
            digits.push(carry);
        }

        Self { digits }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == std::cmp::Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut digits = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i16;

        for index in 0..self.digits.len() {
            let left = self.digits[index] as i16 - borrow;
            let right = rhs.digits.get(index).copied().unwrap_or(0) as i16;
            if left < right {
                digits.push((left + 10 - right) as u8);
                borrow = 1;
            } else {
                digits.push((left - right) as u8);
                borrow = 0;
            }
        }

        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }

        Ok(Self { digits })
    }
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let magnitude_start = usize::from(bytes[0] == b'+');
        if magnitude_start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut first_invalid = None;
        for (i, &b) in bytes.iter().enumerate().skip(magnitude_start) {
            if b != b'_' && !(b'0'..=b'9').contains(&b) {
                first_invalid = Some(i);
                break;
            }
        }
        if let Some(index) = first_invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut first_digit = None;
        let mut last_was_digit = false;
        let mut digit_count = 0usize;
        let mut first_malformed_separator = None;

        for (i, &b) in bytes.iter().enumerate().skip(magnitude_start) {
            if b == b'_' {
                let next_is_digit = bytes
                    .get(i + 1)
                    .map(|n| n.is_ascii_digit())
                    .unwrap_or(false);
                if !last_was_digit || !next_is_digit {
                    if first_malformed_separator.is_none() {
                        first_malformed_separator = Some(i);
                    }
                }
                last_was_digit = false;
            } else {
                if first_digit.is_none() {
                    first_digit = Some((i, b));
                }
                digit_count += 1;
                last_was_digit = true;
            }
        }

        if let Some(index) = first_malformed_separator {
            return Err(Error::Separator { index });
        }

        let (first_digit_index, first_digit_byte) = first_digit.unwrap();
        if digit_count > 1 && first_digit_byte == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let target = max_digits + 1;
            let mut count = 0usize;
            for (i, &b) in bytes.iter().enumerate().skip(magnitude_start) {
                if b.is_ascii_digit() {
                    count += 1;
                    if count == target {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &b in bytes[magnitude_start..].iter().rev() {
            if b != b'_' {
                digits.push(b - b'0');
            }
        }

        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let digit_count = self.digits.len().max(1);
        let width = digit_count.max(min_digits);
        let pad = width - digit_count;
        let separators = if group == 0 { 0 } else { (width - 1) / group };
        let mut result = String::with_capacity(width + separators);

        for i in 0..width {
            if group != 0 && i != 0 && (width - i) % group == 0 {
                result.push('_');
            }

            let digit = if i < pad {
                0
            } else {
                self.digits[digit_count - 1 - (i - pad)]
            };
            result.push(char::from(b'0' + digit));
        }

        result
    }

    pub fn cmp_value(&self, rhs: &Self) -> std::cmp::Ordering {
        let left_len = self.digits.len();
        let right_len = rhs.digits.len();

        if left_len != right_len {
            return left_len.cmp(&right_len);
        }

        for i in (0..left_len).rev() {
            if self.digits[i] != rhs.digits[i] {
                return self.digits[i].cmp(&rhs.digits[i]);
            }
        }

        std::cmp::Ordering::Equal
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let size = self.digits.len().max(rhs.digits.len());
        let mut out = Vec::with_capacity(size + 1);
        let mut carry = 0u8;

        for i in 0..size {
            let a = self.digits.get(i).copied().unwrap_or(0);
            let b = rhs.digits.get(i).copied().unwrap_or(0);
            let sum = a + b + carry;
            out.push(sum % 10);
            carry = sum / 10;
        }

        if carry > 0 {
            out.push(carry);
        }

        Self { digits: out }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == std::cmp::Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut out = Vec::with_capacity(self.digits.len());
        let mut borrow = 0u8;

        for i in 0..self.digits.len() {
            let a = self.digits[i];
            let b = rhs.digits.get(i).copied().unwrap_or(0);
            let subtrahend = b + borrow;

            if a < subtrahend {
                out.push(a + 10 - subtrahend);
                borrow = 1;
            } else {
                out.push(a - subtrahend);
                borrow = 0;
            }
        }

        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }

        Ok(Self { digits: out })
    }

        if start == bytes.len() {

            .find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())

            return Err(Error::InvalidDigit { index: i });
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index: i });
            }
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();

        if digit_count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero { index: first_digit });
        }

        if digit_count > max_digits {
            let excess_number = max_digits;
            let index = (start..bytes.len())
                .filter(|&i| bytes[i] != b'_')
                .nth(excess_number)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &b in bytes[start..].iter().rev() {
            if b != b'_' {
                digits.push(b - b'0');
            }
        }

        let mut raw = String::with_capacity(width);
        for _ in natural_len..width {
            raw.push('0');
        }
        for &d in self.digits.iter().rev() {
            raw.push(char::from(b'0' + d));
        }

        if group == 0 || raw.len() <= group {
            return raw;
        }

        let extra = (raw.len() - 1) / group;
        let mut out = String::with_capacity(raw.len() + extra);
        for (i, ch) in raw.chars().enumerate() {

                out.push('_');

            out.push(ch);

            Ordering::Equal => {
                for (a, b) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}

                }
                Ordering::Equal

        let mut out = Vec::with_capacity(self.digits.len());

        for i in 0..self.digits.len() {
            let mut d = self.digits[i] as i16
                - rhs.digits.get(i).copied().unwrap_or(0) as i16
                - borrow;
            if d < 0 {
                d += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(d as u8);
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
