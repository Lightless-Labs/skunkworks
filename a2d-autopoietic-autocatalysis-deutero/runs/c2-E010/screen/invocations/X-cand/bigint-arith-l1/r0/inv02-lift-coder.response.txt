- Error checks follow the specified precedence: invalid bytes, then malformed separators, then leading zero, then digit limit.
- Offsets are UTF-8 byte offsets into the original input, including an optional leading `+`.
- A leading `+` alone is `SignOnly`; a leading `-` is an invalid digit.
- Zero has one canonical representation; `max_digits = 0` rejects it, while formatting still emits at least one digit.
- Grouping and padding apply to the formatted magnitude, with groups counted from the right.
- Both implementations use base-10 digits and exact carry/borrow arithmetic; Solution B uses a separate vector-based representation.

### Solution A

```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    digits: Vec<u8>, // little-endian; zero is [0]
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        if let Some(i) = (start..bytes.len())
            .find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())
        {
            return Err(Error::InvalidDigit { index: i });
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index: i });
            }
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();

        if digit_count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero { index: first_digit });
        }

        if digit_count > max_digits {
            let excess_number = max_digits;
            let index = (start..bytes.len())
                .filter(|&i| bytes[i] != b'_')
                .nth(excess_number)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &b in bytes[start..].iter().rev() {
            if b != b'_' {
                digits.push(b - b'0');
            }
        }
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let natural_len = self.digits.len().max(1);
        let width = natural_len.max(min_digits);
        let mut raw = String::with_capacity(width);
        for _ in natural_len..width {
            raw.push('0');
        }
        for &d in self.digits.iter().rev() {
            raw.push(char::from(b'0' + d));
        }

        if group == 0 || raw.len() <= group {
            return raw;
        }

        let extra = (raw.len() - 1) / group;
        let mut out = String::with_capacity(raw.len() + extra);
        for (i, ch) in raw.chars().enumerate() {
            if i > 0 && (raw.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(ch);
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => {
                for (a, b) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let n = self.digits.len().max(rhs.digits.len());
        let mut out = Vec::with_capacity(n + 1);
        let mut carry = 0u8;
        for i in 0..n {
            let sum = self.digits.get(i).copied().unwrap_or(0)
                + rhs.digits.get(i).copied().unwrap_or(0)
                + carry;
            out.push(sum % 10);
            carry = sum / 10;
        }
        if carry != 0 {
            out.push(carry);
        }
        Self { digits: out }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }
        let mut out = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i16;
        for i in 0..self.digits.len() {
            let mut d = self.digits[i] as i16
                - rhs.digits.get(i).copied().unwrap_or(0) as i16
                - borrow;
            if d < 0 {
                d += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(d as u8);
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        Ok(Self { digits: out })
    }
}
```

### Solution B

```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Big-endian decimal digits; zero is [0].
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let b = text.as_bytes();
        if b.is_empty() {
            return Err(Error::Empty);
        }
        let offset = usize::from(b[0] == b'+');
        if offset == b.len() {
            return Err(Error::SignOnly);
        }

        let invalid = (offset..b.len())
            .find(|&i| b[i] != b'_' && !(b'0'..=b'9').contains(&b[i]));
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for i in offset..b.len() {
            if b[i] == b'_'
                && (i == offset
                    || i + 1 == b.len()
                    || !(b'0'..=b'9').contains(&b[i - 1])
                    || !(b'0'..=b'9').contains(&b[i + 1]))
            {
                return Err(Error::Separator { index: i });
            }
        }

        let count = b[offset..].iter().filter(|&&x| x != b'_').count();
        let first = (offset..b.len()).find(|&i| b[i] != b'_').unwrap();
        if count > 1 && b[first] == b'0' {
            return Err(Error::LeadingZero { index: first });
        }
        if count > max_digits {
            let index = (offset..b.len())
                .filter(|&i| b[i] != b'_')
                .nth(max_digits)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let digits = b[offset..]
            .iter()
            .copied()
            .filter(|&x| x != b'_')
            .map(|x| x - b'0')
            .collect();
        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let natural = self.digits.len().max(1);
        let width = natural.max(min_digits);
        let mut out = String::with_capacity(width.saturating_add(width.saturating_sub(1) / group.max(1)));
        let padding = width - self.digits.len();
        for i in 0..width {
            if group > 0 && i > 0 && (width - i) % group == 0 {
                out.push('_');
            }
            if i < padding {
                out.push('0');
            } else {
                out.push(char::from(b'0' + self.digits[i - padding]));
            }
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        self.digits.len().cmp(&rhs.digits.len()).then_with(|| self.digits.cmp(&rhs.digits))
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut i = self.digits.len();
        let mut j = rhs.digits.len();
        let mut carry = 0u8;
        let mut rev = Vec::with_capacity(i.max(j) + 1);
        while i > 0 || j > 0 || carry > 0 {
            let a = if i > 0 { i -= 1; self.digits[i] } else { 0 };
            let b = if j > 0 { j -= 1; rhs.digits[j] } else { 0 };
            let sum = a + b + carry;
            rev.push(sum % 10);
            carry = sum / 10;
        }
        rev.reverse();
        Self { digits: rev }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }
        let mut i = self.digits.len();
        let mut j = rhs.digits.len();
        let mut borrow = 0i8;
        let mut rev = Vec::with_capacity(i);
        while i > 0 {
            i -= 1;
            let a = self.digits[i] as i8 - borrow;
            let b = if j > 0 { j -= 1; rhs.digits[j] as i8 } else { 0 };
            if a < b {
                rev.push((a + 10 - b) as u8);
                borrow = 1;
            } else {
                rev.push((a - b) as u8);
                borrow = 0;
            }
        }
        while rev.len() > 1 && rev.last() == Some(&0) {
            rev.pop();
        }
        rev.reverse();
        while rev.len() > 1 && rev[0] == 0 {
            rev.remove(0);
        }
        Ok(Self { digits: rev })
    }
}
```
