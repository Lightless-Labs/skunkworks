#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    for (index, byte) in input.bytes().enumerate() {
        if !(32..=126).contains(&byte) && !matches!(byte, b'\t' | b'\n' | b'\r') {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();

    for word in input.split(|c: char| matches!(c, ' ' | '\t' | '\n' | '\r')).filter(|s| !s.is_empty()) {
        let mut remaining = word;

        while !remaining.is_empty() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            if remaining.len() <= width {
                let needed = remaining.len() + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(remaining);
                    break;
                }

                if !line.is_empty() {
                    lines.push(std::mem::take(&mut line));
                    continue;
                }
            }

            if line.is_empty() {
                let (chunk, rest) = remaining.split_at(width);
                lines.push(chunk.to_string());
                remaining = rest;
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(&remaining[..available]);
                    lines.push(std::mem::take(&mut line));
                    remaining = &remaining[available..];
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}
