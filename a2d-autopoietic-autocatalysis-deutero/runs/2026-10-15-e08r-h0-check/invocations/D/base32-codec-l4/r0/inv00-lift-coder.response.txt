#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut all = Vec::with_capacity(self.pending.len() + input.len());
        all.extend_from_slice(&self.pending);
        all.extend_from_slice(input);

        let complete_len = all.len() / 5 * 5;
        let mut output = String::new();
        for chunk in all[..complete_len].chunks_exact(5) {
            encode_symbols(chunk, self.config.alphabet, &mut output);
        }

        self.pending.clear();
        self.pending.extend_from_slice(&all[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        if !self.pending.is_empty() {
            let symbol_count = (self.pending.len() * 8 + 4) / 5;
            encode_symbols(&self.pending, self.config.alphabet, &mut output);
            if self.config.padding == Padding::Required {
                while output.len() % 8 != 0 {
                    output.push('=');
                }
            }
            debug_assert_eq!(output.len() - if self.config.padding == Padding::Required {
                output.len() - symbol_count
            } else {
                0
            }, symbol_count);
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_byte(alphabet: Alphabet, value: u8) -> u8 {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567"[value as usize],
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV"[value as usize],
    }
}

fn encode_symbols(bytes: &[u8], alphabet: Alphabet, output: &mut String) {
    let bit_len = bytes.len() * 8;
    let symbol_count = (bit_len + 4) / 5;

    for symbol_index in 0..symbol_count {
        let mut value = 0u8;
        for bit_index in 0..5 {
            let bit_position = symbol_index * 5 + bit_index;
            value <<= 1;
            if bit_position < bit_len {
                let byte = bytes[bit_position / 8];
                value |= (byte >> (7 - bit_position % 8)) & 1;
            }
        }
        output.push(alphabet_byte(alphabet, value) as char);
    }
}

pub struct Decoder {
    config: Config,
    position: usize,
    data_count: usize,
    pending_value: u8,
    pending_bits: u8,
    last_data_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    expected_padding: usize,
    padding_count: usize,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            position: 0,
            data_count: 0,
            pending_value: 0,
            pending_bits: 0,
            last_data_offset: None,
            first_padding_offset: None,
            expected_padding: 0,
            padding_count: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            let recognized = if byte == b'=' {
                None
            } else {
                match decode_value(next.config.alphabet, byte) {
                    Some((value, canonical)) => {
                        if byte.is_ascii_lowercase() && next.config.canonical {
                            return Err(StreamError::NonCanonical {
                                offset,
                                expected: canonical,
                            });
                        }
                        Some(value)
                    }
                    None => return Err(StreamError::InvalidByte { offset, byte }),
                }
            };

            if byte == b'=' {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }

                if next.first_padding_offset.is_none() {
                    let residue = next.data_count % 8;
                    let expected = match residue {
                        2 => 6,
                        4 => 4,
                        5 => 3,
                        7 => 1,
                        _ => return Err(StreamError::Padding { offset }),
                    };
                    next.first_padding_offset = Some(offset);
                    next.expected_padding = expected;
                }

                if next.padding_count >= next.expected_padding {
                    return Err(StreamError::Padding { offset });
                }
                next.padding_count += 1;
            } else if let Some(value) = recognized {
                if next.first_padding_offset.is_some() {
                    return Err(StreamError::Padding { offset });
                }

                next.data_count += 1;
                next.last_data_offset = Some(offset);
                next.pending_value = (next.pending_value << 5) | value;
                next.pending_bits += 5;
                while next.pending_bits >= 8 {
                    next.pending_bits -= 8;
                    output.push((next.pending_value >> next.pending_bits) as u8);
                    if next.pending_bits == 0 {
                        next.pending_value = 0;
                    } else {
                        next.pending_value &= (1u8 << next.pending_bits) - 1;
                    }
                }
            }
            next.position += 1;
        }

        self.copy_state_from(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        let expected = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => {
                return Err(StreamError::InvalidLength {
                    offset: self.position,
                })
            }
        };

        if self.config.padding == Padding::Required && self.padding_count != expected {
            return Err(StreamError::Padding {
                offset: self.first_padding_offset.unwrap_or(self.position),
            });
        }

        let unused = match residue {
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused != 0 && self.pending_value != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(self.position),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> DecoderState {
        DecoderState {
            config: self.config,
            position: self.position,
            data_count: self.data_count,
            pending_value: self.pending_value,
            pending_bits: self.pending_bits,
            last_data_offset: self.last_data_offset,
            first_padding_offset: self.first_padding_offset,
            expected_padding: self.expected_padding,
            padding_count: self.padding_count,
            closed: self.closed,
        }
    }

    fn copy_state_from(&mut self, state: DecoderState) {
        self.position = state.position;
        self.data_count = state.data_count;
        self.pending_value = state.pending_value;
        self.pending_bits = state.pending_bits;
        self.last_data_offset = state.last_data_offset;
        self.first_padding_offset = state.first_padding_offset;
        self.expected_padding = state.expected_padding;
        self.padding_count = state.padding_count;
        self.closed = state.closed;
    }
}

struct DecoderState {
    config: Config,
    position: usize,
    data_count: usize,
    pending_value: u8,
    pending_bits: u8,
    last_data_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    expected_padding: usize,
    padding_count: usize,
    closed: bool,
}

fn decode_value(alphabet: Alphabet, byte: u8) -> Option<(u8, u8)> {
    let upper = byte.to_ascii_uppercase();
    let value = match alphabet {
        Alphabet::Standard => match upper {
            b'A'..=b'Z' => upper - b'A',
            b'2'..=b'7' => upper - b'2' + 26,
            _ => return None,
        },
        Alphabet::Hex => match upper {
            b'0'..=b'9' => upper - b'0',
            b'A'..=b'V' => upper - b'A' + 10,
            _ => return None,
        },
    };
    Some((value, upper))
}
