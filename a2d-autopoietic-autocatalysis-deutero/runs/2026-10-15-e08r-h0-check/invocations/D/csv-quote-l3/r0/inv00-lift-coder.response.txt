#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_with_config(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_records(input, b',', b'"')
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_config(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let records = parse_records(input, config.delimiter, config.quote)?;

    if !config.headers {
        return Ok(Table {
            header: None,
            rows: records,
            duplicates: config.duplicates,
        });
    }

    if records.is_empty() {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    }

    let header = records[0].clone();
    if config.duplicates == DuplicatePolicy::Reject {
        let mut first_positions = std::collections::HashMap::<&str, usize>::new();
        for (second, name) in header.iter().enumerate() {
            if let Some(&first) = first_positions.get(name.as_str()) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first,
                    second,
                });
            }
            first_positions.insert(name.as_str(), second);
        }
    }

    for (index, row) in records.iter().enumerate().skip(1) {
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth {
                record: index,
                expected: header.len(),
                actual: row.len(),
            });
        }
    }

    Ok(Table {
        header: Some(header),
        rows: records.into_iter().skip(1).collect(),
        duplicates: config.duplicates,
    })
}

#[derive(Clone, Copy)]
enum ParseState {
    FieldStart,
    Unquoted,
    Quoted,
    AfterQuote,
}

fn parse_records(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let offset = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    if offset == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = Vec::new();
    let mut state = ParseState::FieldStart;
    let mut record_index = 0usize;
    let mut opening_quote = 0usize;
    let mut position = offset;

    while position < bytes.len() {
        let byte = bytes[position];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: position,
            });
        }

        let outside = !matches!(state, ParseState::Quoted);
        if outside && byte == b'\r' && bytes.get(position + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: position,
            });
        }

        match state {
            ParseState::FieldStart => {
                if byte == delimiter {
                    row.push(String::new());
                    position += 1;
                } else if byte == quote {
                    opening_quote = position;
                    state = ParseState::Quoted;
                    position += 1;
                } else if byte == b'\n' || byte == b'\r' {
                    row.push(String::new());
                    records.push(std::mem::take(&mut row));
                    record_index += 1;
                    position += if byte == b'\r' { 2 } else { 1 };
                } else {
                    field.push(byte);
                    state = ParseState::Unquoted;
                    position += 1;
                }
            }
            ParseState::Unquoted => {
                if byte == delimiter {
                    row.push(bytes_to_string(std::mem::take(&mut field)));
                    state = ParseState::FieldStart;
                    position += 1;
                } else if byte == b'\n' || byte == b'\r' {
                    row.push(bytes_to_string(std::mem::take(&mut field)));
                    records.push(std::mem::take(&mut row));
                    record_index += 1;
                    state = ParseState::FieldStart;
                    position += if byte == b'\r' { 2 } else { 1 };
                } else if byte == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: position,
                    });
                } else {
                    field.push(byte);
                    position += 1;
                }
            }
            ParseState::Quoted => {
                if byte == quote {
                    if bytes.get(position + 1) == Some(&quote) {
                        field.push(quote);
                        position += 2;
                    } else {
                        state = ParseState::AfterQuote;
                        position += 1;
                    }
                } else {
                    field.push(byte);
                    position += 1;
                }
            }
            ParseState::AfterQuote => {
                if byte == delimiter {
                    row.push(bytes_to_string(std::mem::take(&mut field)));
                    state = ParseState::FieldStart;
                    position += 1;
                } else if byte == b'\n' || byte == b'\r' {
                    row.push(bytes_to_string(std::mem::take(&mut field)));
                    records.push(std::mem::take(&mut row));
                    record_index += 1;
                    state = ParseState::FieldStart;
                    position += if byte == b'\r' { 2 } else { 1 };
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: position,
                    });
                }
            }
        }
    }

    if matches!(state, ParseState::Quoted) {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if !row.is_empty() || !field.is_empty() || !matches!(state, ParseState::FieldStart) {
        row.push(bytes_to_string(field));
        records.push(row);
    }

    Ok(records)
}

fn bytes_to_string(bytes: Vec<u8>) -> String {
    String::from_utf8(bytes).expect("CSV parsing preserves valid UTF-8")
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ColumnRef {
    Index(usize),
    Name(String),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType {
    Text,
    Integer,
    Decimal,
    Boolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef,
    pub kind: ColumnType,
    pub nullable: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Null,
    Text(String),
    Integer(i64),
    Decimal(i64),
    Boolean(bool),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if row_failed && policy == ErrorPolicy::FirstPerRow {
                output.push(None);
                continue;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let (value, error) = match resolved {
                None => (
                    None,
                    Some(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: None,
                        kind: CellErrorKind::UnknownColumn,
                    }),
                ),
                Some(column_index) => match row.get(column_index) {
                    None => (
                        None,
                        Some(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: Some(column_index),
                            kind: CellErrorKind::MissingCell,
                        }),
                    ),
                    Some(cell) => evaluate_cell(cell, spec, row_index, spec_index, column_index),
                },
            };

            output.push(value);

            if let Some(error) = error {
                result.errors.push(error);
                row_failed = true;
                if policy == ErrorPolicy::Stop {
                    result.rows.push(output);
                    break 'rows;
                }
            }
        }

        result.rows.push(output);
    }

    result
}

fn evaluate_cell(
    cell: &str,
    spec: &ColumnSpec,
    row: usize,
    spec_index: usize,
    column_index: usize,
) -> (Option<Value>, Option<CellError>) {
    if cell.is_empty() {
        if spec.nullable {
            return (Some(Value::Null), None);
        }
        if spec.kind != ColumnType::Text {
            return (
                None,
                Some(CellError {
                    row,
                    spec: spec_index,
                    column: Some(column_index),
                    kind: CellErrorKind::Empty,
                }),
            );
        }
    }

    let result = match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Integer => parse_integer(cell),
        ColumnType::Decimal => parse_decimal(cell),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    };

    match result {
        Ok(value) => (Some(value), None),
        Err(kind) => (
            None,
            Some(CellError {
                row,
                spec: spec_index,
                column: Some(column_index),
                kind,
            }),
        ),
    }
}

fn parse_integer(value: &str) -> Result<Value, CellErrorKind> {
    if !valid_integer_syntax(value) {
        return Err(CellErrorKind::InvalidInteger);
    }
    value
        .parse::<i64>()
        .map(Value::Integer)
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn valid_integer_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    if bytes.is_empty() {
        return false;
    }

    let start = usize::from(bytes[0] == b'-');
    if start == bytes.len() {
        return false;
    }

    if bytes[start] == b'0' {
        return start + 1 == bytes.len() && start == 0;
    }

    if !(b'1'..=b'9').contains(&bytes[start]) {
        return false;
    }

    bytes[start + 1..].iter().all(u8::is_ascii_digit)
}

fn parse_decimal(value: &str) -> Result<Value, CellErrorKind> {
    if !valid_decimal_syntax(value) {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let negative = value.starts_with('-');
    let unsigned = if negative { &value[1..] } else { value };
    let (whole, fractional) = unsigned.split_once('.').expect("validated decimal");
    let whole_value = whole.parse::<i128>().expect("validated digits");
    let fraction_value = fractional.parse::<i128>().expect("two validated digits");
    let magnitude = whole_value * 100 + fraction_value;
    let signed = if negative { -magnitude } else { magnitude };

    i64::try_from(signed)
        .map(Value::Decimal)
        .map_err(|_| CellErrorKind::DecimalOverflow)
}

fn valid_decimal_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    if bytes.is_empty() {
        return false;
    }

    let negative = bytes[0] == b'-';
    let start = usize::from(negative);
    let Some(dot_relative) = bytes[start..].iter().position(|byte| *byte == b'.') else {
        return false;
    };
    let dot = start + dot_relative;

    if bytes.len() != dot + 3 {
        return false;
    }
    let integral = &bytes[start..dot];
    if integral.is_empty() {
        return false;
    }
    if integral == b"0" {
        if negative && bytes[dot + 1..].iter().all(|byte| *byte == b'0') {
            return false;
        }
    } else if !(b'1'..=b'9').contains(&integral[0])
        || !integral[1..].iter().all(u8::is_ascii_digit)
    {
        return false;
    }

    bytes[dot + 1..].iter().all(u8::is_ascii_digit)
}
