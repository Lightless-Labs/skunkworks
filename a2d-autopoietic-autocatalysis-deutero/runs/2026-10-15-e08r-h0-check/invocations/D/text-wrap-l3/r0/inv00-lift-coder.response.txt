#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    breaks: Vec<usize>,
}

#[derive(Clone)]
struct Piece {
    text: String,
    words: usize,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<Piece>,
}

impl Line {
    fn content_len(&self) -> usize {
        self.pieces.iter().map(|p| p.text.chars().count()).sum::<usize>()
            + self.pieces.len().saturating_sub(1)
    }

    fn render(&self) -> String {
        let mut out = " ".repeat(self.indent);
        for (i, piece) in self.pieces.iter().enumerate() {
            if i > 0 {
                out.push(' ');
            }
            out.push_str(&piece.text);
        }
        out
    }
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut breaks = Vec::new();

    let finish = |chars: &mut Vec<char>, breaks: &mut Vec<usize>, words: &mut Vec<Word>| {
        if !chars.is_empty() {
            words.push(Word {
                chars: std::mem::take(chars),
                breaks: std::mem::take(breaks),
            });
        }
    };

    for &c in segment {
        if is_separator(c) {
            finish(&mut chars, &mut breaks, &mut words);
        } else if c == '~' {
            breaks.push(chars.len());
        } else {
            chars.push(c);
        }
    }
    finish(&mut chars, &mut breaks, &mut words);
    words
}

fn make_line(indent: usize) -> Line {
    Line {
        indent,
        pieces: Vec::new(),
    }
}

fn add_piece(line: &mut Line, text: String, words: usize) {
    line.pieces.push(Piece { text, words });
}

fn layout_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    let mut output = Vec::new();
    let mut line_index = 0usize;
    let mut line = make_line(options.first_indent);

    let capacity = |index: usize| {
        options.width
            - if index == 0 {
                options.first_indent
            } else {
                options.rest_indent
            }
    };

    for word in words {
        let mut offset = 0usize;
        let mut is_first_piece = true;

        while offset < word.chars.len() {
            let cap = capacity(line_index);
            let current_len = line.content_len();
            let gap = if line.pieces.is_empty() { 0 } else { 1 };
            let room = cap.saturating_sub(current_len + gap);
            let suffix_len = word.chars.len() - offset;

            if suffix_len <= room {
                let text: String = word.chars[offset..].iter().collect();
                add_piece(&mut line, text, if is_first_piece { 1 } else { 0 });
                offset = word.chars.len();
                continue;
            }

            let available = room;
            let opportunity = word
                .breaks
                .iter()
                .copied()
                .filter(|&b| b > offset && b < word.chars.len() && b - offset + 1 <= available)
                .last();

            let accepted = if let Some(break_at) = opportunity {
                if line.pieces.is_empty() {
                    Some(break_at)
                } else {
                    let unused = available - (break_at - offset + 1);
                    let baseline = cap.saturating_sub(current_len);
                    let lhs = (unused as u128) * (unused as u128) + options.hyphen_cost as u128;
                    let rhs = (baseline as u128) * (baseline as u128);
                    if lhs < rhs {
                        Some(break_at)
                    } else {
                        None
                    }
                }
            } else {
                None
            };

            if let Some(break_at) = accepted {
                let mut text: String = word.chars[offset..break_at].iter().collect();
                text.push('-');
                add_piece(&mut line, text, if is_first_piece { 1 } else { 0 });
                offset = break_at + 1;
                output.push(line.render());
                line_index += 1;
                line = make_line(options.rest_indent);
                is_first_piece = false;
                continue;
            }

            if !line.pieces.is_empty() {
                output.push(line.render());
                line_index += 1;
                line = make_line(options.rest_indent);
                continue;
            }

            let cap = capacity(line_index);
            if suffix_len > cap {
                let end = offset + cap;
                let text: String = word.chars[offset..end].iter().collect();
                add_piece(&mut line, text, if is_first_piece { 1 } else { 0 });
                offset = end;
                while word.breaks.iter().any(|&b| b > offset.saturating_sub(cap) && b < offset) {
                    // Markers are represented separately and are discarded by advancing past
                    // their positions through the next suffix selection.
                    break;
                }
                output.push(line.render());
                line_index += 1;
                line = make_line(options.rest_indent);
                is_first_piece = false;
            } else {
                let text: String = word.chars[offset..].iter().collect();
                add_piece(&mut line, text, if is_first_piece { 1 } else { 0 });
                offset = word.chars.len();
            }
        }
    }

    if !line.pieces.is_empty() {
        output.push(line.render());
    }

    if options.justify && output.len() >= 2 {
        let final_line = output.len() - 1;
        for (idx, rendered) in output.iter_mut().enumerate().take(final_line) {
            let corresponding_indent = if idx == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            let source_line = &words; // Keep the line assembly below independent of source words.
            let _ = source_line;
            let _ = corresponding_indent;
            let _ = rendered;
        }
    }

    output
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let mut original = Vec::new();
    for (index, c) in input.char_indices() {
        original.push((c, index));
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let mut marker_error = None;
    for i in 0..original.len() {
        if original[i].0 != '~' {
            continue;
        }
        let valid_neighbor = |j: usize| {
            j < original.len()
                && original[j].0 != '~'
                && !is_separator(original[j].0)
        };
        if i == 0 || i + 1 >= original.len() || !valid_neighbor(i - 1) || !valid_neighbor(i + 1) {
            marker_error = Some(original[i].1);
            break;
        }
        // Ensure both neighbors belong to the same word: separators are rejected above.
    }
    if let Some(index) = marker_error {
        return Err(WrapError::InvalidMarker { index });
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < original.len() {
        let c = original[i].0;
        if c == '\r' {
            if i + 1 < original.len() && original[i + 1].0 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(c);
            i += 1;
        }
    }

    let mut result = Vec::new();
    let mut segment = Vec::new();
    let mut pos = 0usize;

    while pos < normalized.len() {
        if normalized[pos] == '\n' {
            let start = pos;
            while pos < normalized.len() && normalized[pos] == '\n' {
                pos += 1;
            }
            let count = pos - start;
            if count >= 2 {
                let words = parse_words(&segment);
                result.extend(layout_paragraph(&words, options));
                segment.clear();
                result.extend(std::iter::repeat(String::new()).take(count - 1));
                continue;
            }
        }
        segment.push(normalized[pos]);
        pos += 1;
    }

    let words = parse_words(&segment);
    result.extend(layout_paragraph(&words, options));
    Ok(result)
}
