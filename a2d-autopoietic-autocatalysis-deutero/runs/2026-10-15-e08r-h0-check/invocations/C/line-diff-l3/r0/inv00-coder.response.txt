#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<(u8, String)>,
}

fn parse_num(s: &str) -> Option<usize> {
    if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    if s.len() > 1 && s.starts_with('0') {
        return None;
    }
    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(s: &str) -> Option<(usize, usize, usize, usize)> {
    let rest = s.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(" +")?;
    let (new, suffix) = rest.split_once(" @@")?;
    if !suffix.is_empty() {
        return None;
    }
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    let (a, b, c, d) = (parse_num(a)?, parse_num(b)?, parse_num(c)?, parse_num(d)?);
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    Some((a, b, c, d))
}

fn patch_lines(patch: &str) -> Vec<&str> {
    if patch.is_empty() {
        return Vec::new();
    }
    patch[..patch.len() - 1].split('\n').collect()
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }

    let lines = patch_lines(patch);
    for (i, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(i).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() <= 2 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_old_end: i128 = 0;
    let mut net: i128 = 0;

    while i < lines.len() {
        let header_line = i + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[i]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        let old_start = if b > 0 { a - 1 } else { a };
        let new_start = if d > 0 { c - 1 } else { c };

        i += 1;
        let body_start = i;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let row = lines[i];
            let Some((&prefix, payload)) = row.as_bytes().split_first() else {
                return Err(ApplyError::BadBody { line: i + 1 });
            };
            if !matches!(prefix, b' ' | b'-' | b'+') || payload.contains(&b'\r') {
                return Err(ApplyError::BadBody { line: i + 1 });
            }
            let payload = std::str::from_utf8(payload).unwrap().to_owned();
            if prefix == b' ' || prefix == b'-' {
                old_seen += 1;
            }
            if prefix == b' ' || prefix == b'+' {
                new_seen += 1;
            }
            rows.push((prefix, payload));
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|(p, _)| *p == b'+' || *p == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if (old_start as i128) < previous_old_end
            || (old_start as i128 + net) < 0
            || new_start as i128 != old_start as i128 + net
        {
            return Err(ApplyError::Misordered { line: header_line });
        }
        previous_old_end = old_start as i128 + b as i128;
        net += d as i128 - b as i128;

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });
        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }
        if i == body_start {
            // Empty bodies are structurally invalid above; keep this guard for progress.
            return Err(ApplyError::BadBody { line: header_line });
        }
    }

    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let lines: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source[..source.len() - 1]
            .split('\n')
            .map(str::to_owned)
            .collect()
    };
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;
    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|(p, _)| *p == b' ').count();
        let trail = hunk.rows.iter().rev().take_while(|(p, _)| *p == b' ').count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, i128)> = None;
        let mut best_by_start: Vec<(usize, usize, usize)> = Vec::new();

        for dropped_total in 0..=max_l + max_r {
            for l in 0..=max_l {
                let r = dropped_total.checked_sub(l);
                let Some(r) = r else { continue };
                if r > max_r {
                    continue;
                }
                for s in 0..=working.len().saturating_sub(hunk.old_count) {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = (s as i128 - expected).abs();
                    if distance > options.max_offset as i128 {
                        continue;
                    }

                    let mut old_payloads = Vec::new();
                    for (idx, (prefix, payload)) in hunk.rows.iter().enumerate() {
                        if idx < l || idx >= hunk.rows.len().saturating_sub(r) {
                            continue;
                        }
                        if *prefix == b' ' || *prefix == b'-' {
                            old_payloads.push(payload.as_str());
                        }
                    }
                    let actual_start = s + l;
                    let actual_end = s + hunk.old_count - r;
                    if actual_start > actual_end || actual_end > working.len() {
                        continue;
                    }
                    if old_payloads.len() != actual_end - actual_start
                        || !old_payloads
                            .iter()
                            .zip(&working[actual_start..actual_end])
                            .all(|(a, b)| *a == b)
                    {
                        continue;
                    }

                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_by_start.clear();
                            best_by_start.push((s, l, r));
                        }
                        Some(old_rank) if rank < old_rank => {
                            best_rank = Some(rank);
                            best_by_start.clear();
                            best_by_start.push((s, l, r));
                        }
                        Some(old_rank) if rank == old_rank => {
                            best_by_start.push((s, l, r));
                        }
                        _ => {}
                    }
                }
            }
        }

        if best_by_start.is_empty() {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        }

        best_by_start.sort_by_key(|&(s, l, r)| (s, l, r));
        best_by_start.dedup_by_key(|entry| entry.0);
        if best_by_start.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: best_by_start[0].0,
                second: best_by_start[1].0,
            });
        }

        let (s, l, r) = best_by_start[0];
        let actual_start = s + l;
        let actual_end = s + hunk.old_count - r;
        let mut replacement = Vec::new();
        for (idx, (prefix, payload)) in hunk.rows.iter().enumerate() {
            if idx < l || idx >= hunk.rows.len().saturating_sub(r) {
                continue;
            }
            if *prefix == b' ' || *prefix == b'+' {
                replacement.push(payload.clone());
            }
        }
        working.splice(actual_start..actual_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut result = String::new();
    for line in working {
        result.push_str(&line);
        result.push('\n');
    }
    Ok(result)
}

fn main() {}
