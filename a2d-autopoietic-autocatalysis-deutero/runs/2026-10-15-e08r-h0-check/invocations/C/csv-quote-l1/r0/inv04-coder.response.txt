#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if input.starts_with('\u{feff}') { 3 } else { 0 };

    if pos == bytes.len() {
        return Ok(Vec::new());
    }

    #[derive(Clone, Copy)]
    enum State {
        Start,
        Plain,
        Quoted(usize),
        Closed,
    }

    let mut state = State::Start;
    let mut rows = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        let in_quotes = matches!(state, State::Quoted(_));
        if !in_quotes && byte == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: pos,
            });
        }

        match state {
            State::Start => match byte {
                b'"' => {
                    state = State::Quoted(pos);
                    pos += 1;
                }
                b',' => {
                    row.push(String::new());
                    pos += 1;
                }
                b'\n' | b'\r' => {
                    row.push(String::new());
                    rows.push(std::mem::take(&mut row));
                    record_index += 1;
                    pos += if byte == b'\r' { 2 } else { 1 };
                }
                _ => {
                    let ch = input[pos..].chars().next().unwrap();
                    field.push(ch);
                    state = State::Plain;
                    pos += ch.len_utf8();
                }
            },
            State::Plain => match byte {
                b'"' => {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: pos,
                    });
                }
                b',' => {
                    row.push(std::mem::take(&mut field));
                    state = State::Start;
                    pos += 1;
                }
                b'\n' | b'\r' => {
                    row.push(std::mem::take(&mut field));
                    rows.push(std::mem::take(&mut row));
                    record_index += 1;
                    state = State::Start;
                    pos += if byte == b'\r' { 2 } else { 1 };
                }
                _ => {
                    let ch = input[pos..].chars().next().unwrap();
                    field.push(ch);
                    pos += ch.len_utf8();
                }
            },
            State::Quoted(_) => {
                if byte == b'"' {
                    if bytes.get(pos + 1) == Some(&b'"') {
                        field.push('"');
                        pos += 2;
                    } else {
                        state = State::Closed;
                        pos += 1;
                    }
                } else {
                    let ch = input[pos..].chars().next().unwrap();
                    field.push(ch);
                    pos += ch.len_utf8();
                }
            }
            State::Closed => match byte {
                b',' => {
                    row.push(std::mem::take(&mut field));
                    state = State::Start;
                    pos += 1;
                }
                b'\n' | b'\r' => {
                    row.push(std::mem::take(&mut field));
                    rows.push(std::mem::take(&mut row));
                    record_index += 1;
                    state = State::Start;
                    pos += if byte == b'\r' { 2 } else { 1 };
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: pos,
                    });
                }
            },
        }
    }

    if let State::Quoted(opening) = state {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    if !matches!(state, State::Start) || !row.is_empty() {
        row.push(field);
        rows.push(row);
    }

    Ok(rows)
}

fn main() {}
