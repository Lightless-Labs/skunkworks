```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: Vec<String>,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: Vec::new(),
        }
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            self.cwd.clone()
        };

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                other => components.push(other.to_string()),
            }
        }
        Ok(components)
    }

    fn lookup(&self, components: &[String]) -> Result<&Entry, Error> {
        let mut prefix = Vec::new();
        for (i, component) in components.iter().enumerate() {
            match self.entries.get(&Self::canonical(&prefix)) {
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => {
                    return Err(Error::NotDir(Self::canonical(&prefix)));
                }
                None => {
                    return Err(Error::Missing(Self::canonical(&prefix)));
                }
            }

            prefix.push(component.clone());
            let path = Self::canonical(&prefix);
            match self.entries.get(&path) {
                Some(entry) if i + 1 == components.len() => return Ok(entry),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(path)),
                None => return Err(Error::Missing(path)),
            }
        }
        self.entries.get("/").ok_or_else(|| Error::Missing("/".to_string()))
    }

    fn parent_components(components: &[String]) -> &[String] {
        &components[..components.len() - 1]
    }

    fn check_parent(&self, components: &[String]) -> Result<(), Error> {
        let parent = Self::canonical(Self::parent_components(components));
        match self.entries.get(&parent) {
            Some(Entry::Dir) => Ok(()),
            Some(Entry::File(_)) => Err(Error::NotDir(parent)),
            None => Err(Error::Missing(parent)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }

        self.check_parent(&components)?;
        let canonical = Self::canonical(&components);
        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }
        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = self.normalize(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }

        self.check_parent(&components)?;
        let canonical = Self::canonical(&components);
        match self.entries.get_mut(&canonical) {
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                *contents = data.to_vec();
                Ok(())
            }
            None => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = self.normalize(path)?;
        match self.lookup(&components)? {
            Entry::Dir => Err(Error::IsDir(Self::canonical(&components))),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = self.normalize(path)?;
        let canonical = Self::canonical(&components);
        match self.lookup(&components)? {
            Entry::File(_) => return Err(Error::NotDir(canonical.clone())),
            Entry::Dir => {}
        }

        let mut children = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == "/" {
                continue;
            }
            let Some((parent, basename)) = entry_path.rsplit_once('/') else {
                continue;
            };
            let parent = if parent.is_empty() { "/" } else { parent };
            if parent == canonical {
                let kind = match entry {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                };
                children.push((basename.to_string(), kind));
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        Self::canonical(&self.cwd)
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        match self.lookup(&components)? {
            Entry::Dir => {
                self.cwd = components;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(Self::canonical(&components))),
        }
    }
}

fn main() {}
```
