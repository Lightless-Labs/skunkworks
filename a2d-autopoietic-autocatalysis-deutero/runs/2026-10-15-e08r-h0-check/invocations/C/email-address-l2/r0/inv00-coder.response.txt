#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_prefix_end = if bytes[0] == b'"' {
        let mut i = 1;
        let mut closing = None;

        while i < bytes.len() {
            match bytes[i] {
                b'\\' => {
                    i += 2;
                }
                b'"' => {
                    closing = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }

        match closing {
            Some(i) => Some(i),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i] == b'@'
            && !matches!(quoted_prefix_end, Some(end) if i <= end)
        {
            separators.push(i);
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let mut local = String::new();
    let is_quoted = bytes[0] == b'"';

    if is_quoted {
        let closing = quoted_prefix_end.expect("quoted prefix has closing quote");

        if closing + 1 != sep {
            return Err(EmailError::TrailingLocal { at: closing + 1 });
        }

        let mut origins = Vec::new();
        let mut i = 1;
        while i < closing {
            if bytes[i] == b'\\' {
                let escaped = bytes[i + 1];
                match escaped {
                    b'"' | b'\\' | b'@' => {
                        local.push(escaped as char);
                        origins.push(i);
                        i += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: i }),
                }
            } else {
                let b = bytes[i];
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: i });
                }
                local.push(b as char);
                origins.push(i);
                i += 1;
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: closing });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        let local_bytes = &bytes[..sep];

        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for (i, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || local_bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: i });
                }
            } else if !(b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)) {
                return Err(EmailError::LocalChar { at: i });
            }
        }

        local = input[..sep].to_string();
    }

    let domain_start = sep + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_end = if has_root_dot {
        bytes.len() - 1
    } else {
        bytes.len()
    };

    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_bytes = &bytes[domain_start..domain_end];

    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for i in domain_start..domain_end {
        if bytes[i] == b'.' {
            labels.push((label_start, i));
            label_start = i + 1;
        }
    }
    labels.push((label_start, domain_end));

    for (idx, &(start, end)) in labels.iter().enumerate() {
        if start == end {
            let at = if idx + 1 == labels.len() && has_root_dot {
                domain_end
            } else {
                start
            };
            return Err(EmailError::EmptyLabel { at });
        }

        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong { at: start + 63 });
        }

        if bytes[start] == b'-' {
            return Err(EmailError::LabelHyphen { at: start });
        }
        if bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: end - 1 });
        }

        for i in start..end {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar { at: i });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let domain = input[domain_start..domain_end].to_ascii_lowercase();

    Ok(Address {
        local,
        domain,
        quoted: is_quoted,
    })
}

fn main() {}
