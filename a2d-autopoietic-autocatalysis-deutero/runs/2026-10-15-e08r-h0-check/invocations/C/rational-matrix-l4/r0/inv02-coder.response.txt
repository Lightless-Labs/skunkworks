use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_wide(num as i128, den as i128)
    }

    fn from_wide(mut num: i128, mut den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        if den < 0 {
            num = -num;
            den = -den;
        }
        let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
        num /= g;
        den /= g;
        if num < i64::MIN as i128 || num > i64::MAX as i128 || den > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: num as i64, den: den as i64 })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_wide(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_wide(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_wide(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::from_wide(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: -self.num, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        let fail = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() {
            return Err(fail(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = b[0] == b'-';
        if negative {
            i = 1;
        }
        let num_start = i;
        while i < b.len() && b[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(fail(i, ParseKind::InvalidByte));
        }
        let num_end = i;
        let (den_start, den_end) = if i < b.len() {
            if b[i] != b'/' {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            i += 1;
            let start = i;
            while i < b.len() && b[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            if i < b.len() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            (Some(start), Some(i))
        } else {
            (None, None)
        };

        let digit_start = if negative { 1 } else { 0 };
        let num_digits = num_end - digit_start;
        if num_digits >= 2 && b[digit_start] == b'0' {
            return Err(fail(digit_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == 1 && b[1] == b'0' {
            return Err(fail(0, ParseKind::NonCanonical));
        }
        if let (Some(s), Some(e)) = (den_start, den_end) {
            if e - s >= 2 && b[s] == b'0' {
                return Err(fail(s + 1, ParseKind::NonCanonical));
            }
        }

        let mut mag: u128 = 0;
        for &c in &b[digit_start..num_end] {
            mag = mag.saturating_mul(10).saturating_add((c - b'0') as u128);
        }
        let max_num = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if mag > max_num {
            return Err(fail(0, ParseKind::OutOfRange));
        }

        let mut den_mag = 1u128;
        if let (Some(s), Some(e)) = (den_start, den_end) {
            den_mag = 0;
            for &c in &b[s..e] {
                den_mag = den_mag.saturating_mul(10).saturating_add((c - b'0') as u128);
            }
            if den_mag > i64::MAX as u128 {
                return Err(fail(s, ParseKind::OutOfRange));
            }
            if den_mag == 0 {
                return Err(fail(s, ParseKind::ZeroDenominator));
            }
            if den_mag == 1 {
                return Err(fail(s - 1, ParseKind::NonCanonical));
            }
            if mag == 0 {
                return Err(fail(s - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(mag, den_mag) > 1 {
                return Err(fail(s - 1, ParseKind::NonCanonical));
            }
        }

        let signed = if negative {
            if mag == (1u128 << 63) { i64::MIN } else { -(mag as i64) }
        } else {
            mag as i64
        };
        Ok(Self { num: signed, den: den_mag as i64 })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    fn eliminate(&self, candidates: usize) -> Result<Reduction, MatrixError> {
        let mut m = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0;
        for col in 0..candidates {
            if p == m.rows { break; }
            let mut best: Option<usize> = None;
            for r in p..m.rows {
                let x = m.data[r * m.cols + col];
                if x.num == 0 { continue; }
                let key = (x.den, x.num.unsigned_abs(), r);
                if best.map_or(true, |br| {
                    let y = m.data[br * m.cols + col];
                    key < (y.den, y.num.unsigned_abs(), br)
                }) {
                    best = Some(r);
                }
            }
            let q = match best { Some(q) => q, None => continue };
            if q != p {
                for c in 0..m.cols {
                    m.data.swap(p * m.cols + c, q * m.cols + c);
                }
                swaps.push((p, q));
            }
            let pivot = m.data[p * m.cols + col];
            for c in 0..m.cols {
                let original = m.data[p * m.cols + c];
                m.data[p * m.cols + c] = original.div(&pivot).map_err(|source|
                    MatrixError::Arithmetic { row: p, col: c, step: p, source })?;
            }
            for r in 0..m.rows {
                if r == p { continue; }
                let factor = m.data[r * m.cols + col];
                if factor.num == 0 { continue; }
                for c in 0..m.cols {
                    let product = factor.mul(&m.data[p * m.cols + c]).map_err(|source|
                        MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
                    m.data[r * m.cols + c] = m.data[r * m.cols + c].sub(&product).map_err(|source|
                        MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction { matrix: m, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }
        let aug = Matrix { rows: n, cols: width, data };
        let red = aug.eliminate(n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank: red.pivots.len() });
        }
        let mut out = Vec::with_capacity(n.checked_mul(n).ok_or(MatrixError::ShapeOverflow)?);
        for r in 0..n {
            out.extend_from_slice(&red.matrix.data[r * width + n..(r + 1) * width]);
        }
        Ok(Matrix { rows: n, cols: n, data: out })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients: self.rows, rhs: rhs.rows });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            data.extend_from_slice(&rhs.data[r * k..(r + 1) * k]);
        }
        let aug = Matrix { rows: self.rows, cols: width, data };
        let red = aug.eliminate(n).map_err(SolveError::Matrix)?;

        for r in 0..self.rows {
            if red.matrix.data[r * width..r * width + n].iter().all(|x| x.num == 0) {
                if let Some(j) = (0..k).find(|&j| red.matrix.data[r * width + n + j].num != 0) {
                    return Ok(Solution::None {
                        row: r,
                        rhs_col: j,
                        witness: red.matrix.data[r * width..(r + 1) * width].to_vec(),
                    });
                }
            }
        }

        let rank = red.pivots.len();
        let particular_len = n.checked_mul(k)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let zero = Rational { num: 0, den: 1 };
        let mut particular = vec![zero; particular_len];
        for (i, &pc) in red.pivots.iter().enumerate() {
            for j in 0..k {
                particular[pc * k + j] = red.matrix.data[i * width + n + j];
            }
        }
        let particular = Matrix { rows: n, cols: k, data: particular };
        if rank == n {
            return Ok(Solution::Unique { values: particular, pivots: red.pivots });
        }

        let free: Vec<usize> = (0..n).filter(|c| !red.pivots.contains(c)).collect();
        let f = free.len();
        let basis_len = n.checked_mul(f)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![zero; basis_len];
        for (j, &fc) in free.iter().enumerate() {
            basis[fc * f + j] = Rational { num: 1, den: 1 };
            for (i, &pc) in red.pivots.iter().enumerate() {
                let coeff = red.matrix.data[i * width + fc];
                let neg = coeff.neg().map_err(|source| SolveError::Matrix(
                    MatrixError::Arithmetic { row: pc, col: j, step: rank, source },
                ))?;
                basis[pc * f + j] = neg;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix { rows: n, cols: f, data: basis },
            pivots: red.pivots,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}

fn main() {}
