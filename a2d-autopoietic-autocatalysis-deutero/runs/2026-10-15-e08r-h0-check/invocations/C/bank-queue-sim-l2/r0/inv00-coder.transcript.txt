                index,
                id: customer.id,

            })
            .collect();
        opening.sort_by(|&a, &b| states[b].server.id.cmp(&states[a].server.id));
        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time,
                server: states[i].server.id,
            });
        }

        // Arrival phase, already sorted by arrival ascending and ID descending.
        while next_customer < customer_order.len()
            && customers[customer_order[next_customer]].arrival == time
        {
            let index = customer_order[next_customer];
            let customer = &customers[index];
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            waiting.push_back(index);
            next_customer += 1;
        }

        // Dispatch phase.
        while let Some(&customer_index) = waiting.front() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.occupied_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };

            let customer = &customers[customer_index];
            let server = &states[server_index].server;
            let finish = time
                .checked_add(customer.service)
                .and_then(|v| v.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time,
                })?;

            waiting.pop_front();
            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = time;
                finished_customers += 1;
            } else {
                states[server_index].occupied_until = Some(finish);
            }
        }

        if finished_customers == customers.len()
            && next_customer == customer_order.len()
            && states.iter().all(|state| state.opened)
        {
            return Ok(events);
        }

        let next_time = customer_order
            .get(next_customer)
            .map(|&i| customers[i].arrival)
            .into_iter()
            .chain(
                states
                    .iter()
                    .filter(|state| !state.opened)
                    .map(|state| state.server.opens),
            )
            .chain(states.iter().filter_map(|state| state.occupied_until))
            .filter(|&candidate| candidate > time)
            .min();

        let Some(next_time) = next_time else {
            return Ok(events);
        };
        time = next_time;
    }
}

fn main() {}
