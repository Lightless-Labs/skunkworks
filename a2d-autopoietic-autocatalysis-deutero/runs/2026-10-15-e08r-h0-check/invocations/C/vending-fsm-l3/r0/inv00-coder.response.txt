```rust
use std::collections::HashMap;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    events: Vec<Event>,
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = HashMap::<String, ()>::new();
        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.contains_key(&product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.insert(product.id.clone(), ());
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        let denominations = [1u16, 3, 4, 10];
        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: denominations[i],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            events: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    fn log(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }

    fn product_index(&self, id: &str) -> Option<usize> {
        self.products.iter().position(|p| p.id == id)
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if ![1, 3, 4, 10].contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        let new_credit = self.credit as u32 + coin as u32;
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit = new_credit as u16;
        self.escrow.push(coin);
        self.log(EventKind::Inserted {
            coin,
            credit: self.credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.product_index(id).is_none() {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        self.log(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
        if amount == 0 {
            return Some(Vec::new());
        }

        let denom = [1u16, 3, 4, 10];
        let mut remaining = amount as u32;
        let mut greedy_counts = [0u16; 4];

        for i in (0..4).rev() {
            let take = (tubes[i] as u32).min(remaining / denom[i] as u32);
            greedy_counts[i] = take as u16;
            remaining -= take * denom[i] as u32;
        }

        if remaining == 0 {
            return Some(Self::counts_to_descending(greedy_counts));
        }

        let target = amount as usize;
        let mut best: Vec<Option<[u16; 4]>> = vec![None; target + 1];
        best[0] = Some([0; 4]);

        for value in 0..=target {
            let Some(counts) = best[value] else {
                continue;
            };
            for i in 0..4 {
                if counts[i] >= tubes[i] {
                    continue;
                }
                let next = value + denom[i] as usize;
                if next > target {
                    continue;
                }
                let mut candidate = counts;
                candidate[i] += 1;

                let replace = match best[next] {
                    None => true,
                    Some(existing) => {
                        let candidate_total: u32 = candidate.iter().map(|&x| x as u32).sum();
                        let existing_total: u32 = existing.iter().map(|&x| x as u32).sum();
                        candidate_total < existing_total
                            || (candidate_total == existing_total
                                && Self::lex_greater_desc(candidate, existing))
                    }
                };
                if replace {
                    best[next] = Some(candidate);
                }
            }
        }

        best[target].map(Self::counts_to_descending)
    }

    fn lex_greater_desc(a: [u16; 4], b: [u16; 4]) -> bool {
        for i in (0..4).rev() {
            if a[i] != b[i] {
                return a[i] > b[i];
            }
        }
        false
    }

    fn counts_to_descending(counts: [u16; 4]) -> Vec<u16> {
        let denom = [1u16, 3, 4, 10];
        let mut result = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..counts[i] {
                result.push(denom[i]);
            }
        }
        result
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = match &self.selected {
            Some(id) => id.clone(),
            None => return Err(Error::NoSelection),
        };
        let index = self.product_index(&id).expect("selected product exists");
        let product = &self.products[index];

        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let price = product.price;
        let amount = self.credit - price;
        let change = Self::choose_change(amount, self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        for coin in &change {
            let i = match coin {
                1 => 0,
                3 => 1,
                4 => 2,
                10 => 3,
                _ => unreachable!(),
            };
            proposed[i] -= 1;
        }
        for coin in &self.escrow {
            let i = match coin {
                1 => 0,
                3 => 1,
                4 => 2,
                10 => 3,
                _ => unreachable!(),
            };
            proposed[i] += 1;
        }

        let denominations = [1u16, 3, 4, 10];
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: denominations[i],
                });
            }
        }

        self.products[index].stock -= 1;
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        let receipt = Receipt {
            id: id.clone(),
            price,
            change: change.clone(),
        };
        self.log(EventKind::Vended {
            id,
            price,
            change,
        });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let mut coins = self.escrow.clone();
        coins.reverse();

        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }

        let new_stock = self.products[index].stock as u32 + count as u32;
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        self.products[index].stock = new_stock as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: new_stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }
}

fn main() {}
```
