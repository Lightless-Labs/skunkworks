#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    values: [Vec<u8>; 5],
    day_of_month_star: bool,
    day_of_week_star: bool,
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn name_value(field: Field, atom: &str) -> Option<u32> {
    let upper = atom.to_ascii_uppercase();
    match field {
        Field::Month => match upper.as_str() {
            "JAN" => Some(1),
            "FEB" => Some(2),
            "MAR" => Some(3),
            "APR" => Some(4),
            "MAY" => Some(5),
            "JUN" => Some(6),
            "JUL" => Some(7),
            "AUG" => Some(8),
            "SEP" => Some(9),
            "OCT" => Some(10),
            "NOV" => Some(11),
            "DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_str() {
            "SUN" => Some(0),
            "MON" => Some(1),
            "TUE" => Some(2),
            "WED" => Some(3),
            "THU" => Some(4),
            "FRI" => Some(5),
            "SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

fn is_digits(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn valid_atom(field: Field, s: &str) -> bool {
    is_digits(s) || name_value(field, s).is_some()
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    if s.len() > 1 && s.as_bytes()[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for byte in s.bytes() {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn parse_base_atom(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    if is_digits(s) {
        let value = parse_number(s, field, offset)?;
        let (min, max) = bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
        Ok(value)
    } else {
        name_value(field, s).ok_or_else(|| invalid_syntax(field, offset))
    }
}

fn parse_step(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    let value = parse_number(s, field, offset)?;
    let (min, max) = bounds(field);
    let width = max - min + 1;
    if value == 0 || value > width {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    for (offset, byte) in text.bytes().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = bounds(field);
    let mut selected = vec![false; (max + 1) as usize];
    let mut item_offset = 0usize;

    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_offset,
            });
        }

        if item.bytes().filter(|&b| b == b'/').count() > 1 {
            return Err(invalid_syntax(field, item_offset));
        }

        let (base, step_text) = if let Some(pos) = item.find('/') {
            (&item[..pos], Some((&item[pos + 1..], item_offset + pos + 1)))
        } else {
            (item, None)
        };

        if base.is_empty() || step_text.is_some_and(|(step, _)| step.is_empty()) {
            return Err(invalid_syntax(field, item_offset));
        }

        let is_star = base == "*";
        let hyphen_count = base.bytes().filter(|&b| b == b'-').count();

        let base_kind = if is_star {
            0
        } else if hyphen_count == 0 {
            if !valid_atom(field, base) {
                return Err(invalid_syntax(field, item_offset));
            }
            1
        } else if hyphen_count == 1 {
            let pos = base.find('-').unwrap();
            if !valid_atom(field, &base[..pos]) || !valid_atom(field, &base[pos + 1..]) {
                return Err(invalid_syntax(field, item_offset));
            }
            2
        } else {
            return Err(invalid_syntax(field, item_offset));
        };

        let (range_start, range_end) = match base_kind {
            0 => (min, max),
            1 => {
                let value = parse_base_atom(base, field, item_offset)?;
                (value, if step_text.is_some() { max } else { value })
            }
            _ => {
                let pos = base.find('-').unwrap();
                let start = parse_base_atom(&base[..pos], field, item_offset)?;
                let end = parse_base_atom(&base[pos + 1..], field, item_offset + pos + 1)?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_offset + pos + 1,
                    });
                }
                (start, end)
            }
        };

        let step = if let Some((step, offset)) = step_text {
            if !is_digits(step) {
                return Err(invalid_syntax(field, item_offset));
            }
            Some(parse_step(step, field, offset)?)
        } else {
            None
        };

        if let Some(step) = step {
            let mut value = range_start;
            while value <= range_end {
                selected[value as usize] = true;
                match value.checked_add(step) {
                    Some(next) => value = next,
                    None => break,
                }
            }
        } else {
            for value in range_start..=range_end {
                selected[value as usize] = true;
            }
        }

        item_offset += item.len() + 1;
    }

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(value, &included)| included.then_some(value as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut values: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    for index in 0..5 {
        values[index] = parse_field(pieces[index], fields[index])?;
    }

    Ok(Cron {
        values,
        day_of_month_star: pieces[2] == "*",
        day_of_week_star: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.values[field_index(field)].clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if !advance_minute(&mut current) {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.values[0].contains(&at.minute)
            || !self.values[1].contains(&at.hour)
            || !self.values[3].contains(&at.month)
        {
            return false;
        }

        let dom = self.values[2].contains(&at.day);
        let dow = self.values[4].contains(&weekday(at.year, at.month, at.day));

        match (self.day_of_month_star, self.day_of_week_star) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn advance_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;

    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;

    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;

    if at.month < 12 {
        at.month += 1;
        return true;
    }
    at.month = 1;

    if at.year < 2399 {
        at.year += 1;
        return true;
    }
    false
}
