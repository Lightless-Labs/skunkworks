```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn symbol_value(byte: u8, alphabet: Alphabet) -> Option<u8> {
    alphabet_bytes(alphabet)
        .iter()
        .position(|&symbol| symbol == byte)
        .map(|value| value as u8)
}

pub fn encode(input: &[u8], alphabet: Alphabet, padding: Padding) -> String {
    let symbols = alphabet_bytes(alphabet);
    let mut output = String::new();
    let mut accumulator = 0u32;
    let mut bits = 0u32;

    for &byte in input {
        accumulator = (accumulator << 8) | byte as u32;
        bits += 8;

        while bits >= 5 {
            bits -= 5;
            output.push(symbols[((accumulator >> bits) & 0x1f) as usize] as char);
        }
    }

    if bits > 0 {
        let value = ((accumulator << (5 - bits)) & 0x1f) as usize;
        output.push(symbols[value] as char);
    }

    if padding == Padding::Required {
        while output.len() % 8 != 0 {
            output.push('=');
        }
    }

    output
}

pub fn decode(
    input: &str,
    alphabet: Alphabet,
    padding: Padding,
) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let n = bytes.len();

    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && symbol_value(byte, alphabet).is_none() {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let p = bytes.iter().position(|&byte| byte == b'=').unwrap_or(n);

    if padding == Padding::Omitted && p < n {
        return Err(DecodeError::Padding { offset: p });
    }

    if padding == Padding::Required && p < n {
        if let Some(offset) = (p + 1..n).find(|&i| bytes[i] != b'=') {
            return Err(DecodeError::Padding { offset });
        }
    }

    let d = p;
    let residue = d % 8;

    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength { offset: n });
    }

    if padding == Padding::Required {
        let expected_padding = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => unreachable!(),
        };

        if n - d != expected_padding || n % 8 != 0 {
            return Err(DecodeError::Padding { offset: p });
        }
    }

    let unused_bits = match residue {
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => 0,
    };

    if unused_bits > 0 {
        let last_value = symbol_value(bytes[d - 1], alphabet).unwrap();
        let mask = (1u8 << unused_bits) - 1;
        if last_value & mask != 0 {
            return Err(DecodeError::TrailingBits { offset: d - 1 });
        }
    }

    let mut output = Vec::with_capacity(d * 5 / 8);
    let mut accumulator = 0u32;
    let mut bits = 0u32;

    for &byte in &bytes[..d] {
        accumulator = (accumulator << 5) | symbol_value(byte, alphabet).unwrap() as u32;
        bits += 5;

        if bits >= 8 {
            bits -= 8;
            output.push(((accumulator >> bits) & 0xff) as u8);
        }
    }

    Ok(output)
}

fn main() {}
```
