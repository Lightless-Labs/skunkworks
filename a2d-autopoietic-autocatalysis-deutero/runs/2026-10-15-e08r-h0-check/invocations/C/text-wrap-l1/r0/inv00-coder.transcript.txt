#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(matches!(byte, b' '|b'\t'|b'\n'|b'\r') || (32..=126).contains(&byte)) {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && matches!(bytes[i], b' '|b'\t'|b'\n'|b'\r') {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && !matches!(bytes[i], b' '|b'\t'|b'\n'|b'\r') {
            i += 1;
        }
        let mut pos = start;

        loop {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            let remaining = i - pos;
            if remaining <= width {
                let needed = remaining + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(std::str::from_utf8(&bytes[pos..i]).unwrap());
                    break;
                } else {
                    if !line.is_empty() {
                        lines.push(std::mem::take(&mut line));
                    }
                    continue;
                }
            }

            if line.is_empty() {
                line.push_str(std::str::from_utf8(&bytes[pos..pos + width]).unwrap());
                lines.push(std::mem::take(&mut line));
                pos += width;
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(std::str::from_utf8(&bytes[pos..pos + available]).unwrap());
                    lines.push(std::mem::take(&mut line));
                    pos += available;
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }
