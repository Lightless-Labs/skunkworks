#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

#[derive(Clone)]
struct Piece {
    text: String,
}

struct Line {
    pieces: Vec<Piece>,
    indent: usize,
}

impl Line {
    fn is_empty(&self) -> bool {
        self.pieces.is_empty()
    }

    fn content_len(&self) -> usize {
        self.pieces
            .iter()
            .map(|piece| piece.text.chars().count())
            .sum::<usize>()
            + self.pieces.len().saturating_sub(1)
    }

    fn push_piece(&mut self, text: String) {
        self.pieces.push(Piece { text });
    }

    fn render(&self, justify: bool, width: usize, line_index: usize, is_final: bool) -> String {
        let gaps = self.pieces.len().saturating_sub(1);
        let pre_width = self.indent + self.content_len();
        let extra = if justify && !is_final && gaps > 0 {
            width.saturating_sub(pre_width)
        } else {
            0
        };
        let base = if gaps > 0 { extra / gaps } else { 0 };
        let remainder = if gaps > 0 { extra % gaps } else { 0 };
        let start = if gaps > 0 { line_index % gaps } else { 0 };

        let mut out = " ".repeat(self.indent);
        for (i, piece) in self.pieces.iter().enumerate() {
            out.push_str(&piece.text);
            if i < gaps {
                let extra_here = base + usize::from((i + gaps - start) % gaps < remainder);
                out.push_str(&" ".repeat(1 + extra_here));
            }
        }
        out
    }
}

fn emit_line(lines: &mut Vec<Line>, current: &mut Line) {
    if !current.is_empty() {
        lines.push(Line {
            pieces: std::mem::take(&mut current.pieces),
            indent: current.indent,
        });
    }
}

fn layout_paragraph(words: &[Vec<char>], options: &WrapOptions) -> Vec<String> {
    let mut lines = Vec::<Line>::new();
    let mut current = Line {
        pieces: Vec::new(),
        indent: options.first_indent,
    };

    for word in words {
        let mut offset = 0usize;

        while offset < word.len() {
            let capacity = options.width - current.indent;
            let content_len = current.content_len();
            let room = capacity - content_len;
            let suffix_len = word.len() - offset;
            let separator = usize::from(!current.is_empty());

            if suffix_len + separator <= room {
                let text = word[offset..]
                    .iter()
                    .copied()
                    .filter(|&c| c != '~')
                    .collect();
                current.push_piece(text);
                offset = word.len();
                continue;
            }

            let available = if current.is_empty() {
                capacity
            } else {
                capacity - content_len
            };
            let mut rightmost = None;
            for i in (offset + 1)..word.len() {
                if word[i] == '~' && i < word.len() - 1 {
                    let prefix_len = i - offset;
                    if prefix_len + 1 <= available {
                        rightmost = Some((i, available - prefix_len - 1));
                    }
                }
            }

            if let Some((marker, unused)) = rightmost {
                let accepted = current.is_empty()
                    || unused * unused + options.hyphen_cost
                        < available * available;
                if accepted {
                    let mut text: String = word[offset..marker]
                        .iter()
                        .copied()
                        .filter(|&c| c != '~')
                        .collect();
                    text.push('-');
                    current.push_piece(text);
                    offset = marker + 1;
                    emit_line(&mut lines, &mut current);
                    current.indent = options.rest_indent;
                    continue;
                }
            }

            if !current.is_empty() {
                emit_line(&mut lines, &mut current);
                current.indent = options.rest_indent;
                continue;
            }

            let visible: Vec<char> = word[offset..]
                .iter()
                .copied()
                .filter(|&c| c != '~')
                .collect();
            if visible.len() <= capacity {
                current.push_piece(visible.into_iter().collect());
                offset = word.len();
            } else {
                current.push_piece(visible[..capacity].iter().collect());

                let mut consumed = 0usize;
                let mut next_offset = offset;
                while next_offset < word.len() && consumed < capacity {
                    if word[next_offset] != '~' {
                        consumed += 1;
                    }
                    next_offset += 1;
                }
                offset = next_offset;
                emit_line(&mut lines, &mut current);
                current.indent = options.rest_indent;
            }
        }
    }

    emit_line(&mut lines, &mut current);
    let count = lines.len();
    lines
        .into_iter()
        .enumerate()
        .map(|(i, line)| line.render(options.justify, options.width, i, i + 1 == count))
        .collect()
}

fn split_words(segment: &[char]) -> Vec<Vec<char>> {
    let mut words = Vec::new();
    let mut current = Vec::new();

    for &c in segment {
        if is_separator(c) {
            if !current.is_empty() {
                words.push(std::mem::take(&mut current));
            }
        } else {
            current.push(c);
        }
    }
    if !current.is_empty() {
        words.push(current);
    }
    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let valid_side = |j: isize| {
                if j < 0 || j as usize >= chars.len() {
                    return false;
                }
                let c = chars[j as usize].1;
                c != '~' && !is_separator(c)
            };
            if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut input_chars = input.chars().peekable();
    while let Some(c) = input_chars.next() {
        if c == '\r' {
            if input_chars.peek() == Some(&'\n') {
                input_chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    if normalized.is_empty() {
        return Ok(Vec::new());
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0usize;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut end = i;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            if end - i >= 2 {
                output.extend(layout_paragraph(&split_words(&segment), options));
                segment.clear();
                for _ in 1..(end - i) {
                    output.push(String::new());
                }
                i = end;
                continue;
            }
        }
        segment.push(normalized[i]);
        i += 1;
    }
    output.extend(layout_paragraph(&split_words(&segment), options));
    Ok(output)
}

fn main() {}
