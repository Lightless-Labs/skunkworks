#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(ParseError::Empty);
    }

    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(index) => index,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map_or(bytes.len(), |offset| authority_start + offset);

    let authority = &input[authority_start..authority_end];
    let auth_bytes = authority.as_bytes();

    let mut at_sign = None;
    for (offset, &byte) in auth_bytes.iter().enumerate() {
        if byte == b'@' {
            if at_sign.is_some() {
                return Err(ParseError::InvalidUserInfo {
                    at: authority_start + offset,
                });
            }
            at_sign = Some(offset);
        }
    }

    let (userinfo, host_port, host_port_start) = match at_sign {
        Some(0) => {
            return Err(ParseError::InvalidUserInfo { at: authority_start });
        }
        Some(offset) => (
            Some(authority[..offset].to_owned()),
            &authority[offset + 1..],
            authority_start + offset + 1,
        ),
        None => (None, authority, authority_start),
    };

    let host_port_bytes = host_port.as_bytes();
    let host_colon = host_port_bytes.iter().position(|&byte| byte == b':');
    let host_end = host_colon.unwrap_or(host_port.len());
    let host = &host_port[..host_end];

    if host.is_empty() {
        return Err(ParseError::InvalidHost { at: host_port_start });
    }
    for (offset, &byte) in host.as_bytes().iter().enumerate() {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost {
                at: host_port_start + offset,
            });
        }
    }

    let port = match host_colon {
        None => None,
        Some(offset) => {
            let port_start = host_port_start + offset + 1;
            let port_text = &host_port[offset + 1..];
            let port_bytes = port_text.as_bytes();

            if port_bytes.is_empty() {
                return Err(ParseError::InvalidPort { at: port_start });
            }

            for (index, &byte) in port_bytes.iter().enumerate() {
                if !byte.is_ascii_digit() {
                    return Err(ParseError::InvalidPort { at: port_start + index });
                }
            }

            if port_bytes.len() >= 2 && port_bytes[0] == b'0' {
                return Err(ParseError::InvalidPort { at: port_start });
            }

            let mut value: u32 = 0;
            for &digit in port_bytes {
                value = match value
                    .checked_mul(10)
                    .and_then(|number| number.checked_add((digit - b'0') as u32))
                {
                    Some(number) => number,
                    None => return Err(ParseError::InvalidPort { at: port_start }),
                };
            }

            if !(1..=65535).contains(&value) {
                return Err(ParseError::InvalidPort { at: port_start });
            }
            Some(value as u16)
        }
    };

    let remainder = &input[authority_end..];
    let fragment_marker = remainder.find('#');
    let before_fragment = fragment_marker
        .map_or(remainder, |index| &remainder[..index]);
    let fragment = fragment_marker.map(|index| remainder[index + 1..].to_owned());

    let query_marker = before_fragment.find('?');
    let path_text = query_marker
        .map_or(before_fragment, |index| &before_fragment[..index]);
    let query = query_marker.map(|index| before_fragment[index + 1..].to_owned());

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host: host.to_ascii_lowercase(),
        port,
        path: if path_text.is_empty() {
            "/".to_owned()
        } else {
            path_text.to_owned()
        },
        query,
        fragment,
    })
}

fn main() {}
