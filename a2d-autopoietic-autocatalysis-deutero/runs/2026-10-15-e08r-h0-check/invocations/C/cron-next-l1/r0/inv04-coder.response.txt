```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    values: [Vec<u8>; 5],
    day_stars: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.values[field_index(field)].clone()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn domain(field: Field) -> (u32, u32, u32) {
    match field {
        Field::Minute => (0, 59, 60),
        Field::Hour => (0, 23, 24),
        Field::DayOfMonth => (1, 31, 31),
        Field::Month => (1, 12, 12),
        Field::DayOfWeek => (0, 6, 7),
    }
}

fn number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }

    let mut value = 0u32;
    for &b in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(u32::from(b - b'0')))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }

    let (min, max, width) = domain(field);
    if step {
        if value == 0 || value > width {
            return Err(ParseError::BadStep { field, offset: start });
        }
    } else if value < min || value > max {
        return Err(ParseError::OutOfRange { field, offset: start });
    }
    Ok(value)
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_field(text: &str, field: Field) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit() || matches!(b, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max, _) = domain(field);
    let mut selected = vec![false; (max + 1) as usize];
    let mut item_start = 0usize;
    let mut exact_day_star = false;

    for item_end in (0..=bytes.len()).filter(|&i| i == bytes.len() || bytes[i] == b',') {
        if item_start == item_end {
            return Err(ParseError::EmptyItem { field, offset: item_start });
        }

        let item = &bytes[item_start..item_end];
        let slashes: Vec<usize> = item.iter().enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slashes.len() > 1 {
            return Err(invalid_syntax(field, item_start));
        }

        let base_end = slashes.first().copied().unwrap_or(item.len());
        let step_span = slashes.first().map(|&i| (i + 1, item.len()));
        if let Some((s, e)) = step_span {
            if s == e {
                return Err(invalid_syntax(field, item_start));
            }
        }

        let base = &item[..base_end];
        let hyphens: Vec<usize> = base.iter().enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let (kind, start_span, end_span) = if base == b"*" {
            (0u8, None, None)
        } else if hyphens.is_empty() && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            (1, Some((0, base.len())), None)
        } else if hyphens.len() == 1 {
            let h = hyphens[0];
            if h == 0 || h + 1 == base.len()
                || !base[..h].iter().all(u8::is_ascii_digit)
                || !base[h + 1..].iter().all(u8::is_ascii_digit)
            {
                return Err(invalid_syntax(field, item_start));
            }
            (2, Some((0, h)), Some((h + 1, base.len())))
        } else {
            return Err(invalid_syntax(field, item_start));
        };

        let start_value = if let Some((s, e)) = start_span {
            Some(number(bytes, item_start + s, item_start + e, field, false)?)
        } else {
            None
        };
        let end_value = if let Some((s, e)) = end_span {
            Some(number(bytes, item_start + s, item_start + e, field, false)?)
        } else {
            None
        };

        let range_start = start_value.unwrap_or(min);
        let range_end = match kind {
            0 => max,
            1 if step_span.is_some() => max,
            1 => range_start,
            2 => {
                let end = end_value.unwrap();
                if range_start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + end_span.unwrap().0,
                    });
                }
                end
            }
            _ => unreachable!(),
        };

        let step = if let Some((s, e)) = step_span {
            Some(number(bytes, item_start + s, item_start + e, field, true)?)
        } else {
            None
        };

        if (field == Field::DayOfMonth || field == Field::DayOfWeek) && item == b"*" {
            exact_day_star = true;
        }

        let increment = step.unwrap_or(1);
        let mut value = range_start;
        while value <= range_end {
            selected[value as usize] = true;
            value += increment;
        }

        item_start = item_end + 1;
    }

    let values = selected.into_iter().enumerate()
        .filter_map(|(v, yes)| (yes && (v as u32) >= min).then_some(v as u8))
        .collect();
    Ok((values, exact_day_star))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount { found: fields.len() });
    }

    let kinds = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut values: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut day_stars = [false; 2];

    for i in 0..5 {
        let (parsed, exact_star) = parse_field(fields[i], kinds[i])?;
        values[i] = parsed;
        if i == 2 {
            day_stars[0] = exact_star;
        } else if i == 4 {
            day_stars[1] = exact_star;
        }
    }

    Ok(Cron { values, day_stars })
}

fn main() {}
```
