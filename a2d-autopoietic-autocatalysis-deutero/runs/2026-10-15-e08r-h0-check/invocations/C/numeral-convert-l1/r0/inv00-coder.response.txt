#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanCase {
    Upper,
    Lower,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanError {
    Empty,
    InvalidByte { index: usize },
    MixedCase { index: usize },
    NonCanonical { index: usize },
    Range,
}

pub fn roman_encode(value: u32, case: RomanCase) -> Result<String, RomanError> {
    if value > 3999 {
        return Err(RomanError::Range);
    }

    let lowercase = matches!(case, RomanCase::Lower);
    if value == 0 {
        return Ok(if lowercase { "n" } else { "N" }.to_string());
    }

    let tokens: &[(u32, &str)] = &[
        (1000, "M"),
        (900, "CM"),
        (500, "D"),
        (400, "CD"),
        (100, "C"),
        (90, "XC"),
        (50, "L"),
        (40, "XL"),
        (10, "X"),
        (9, "IX"),
        (5, "V"),
        (4, "IV"),
        (1, "I"),
    ];

    let mut remaining = value;
    let mut result = String::new();
    for &(amount, token) in tokens {
        while remaining >= amount {
            remaining -= amount;
            if lowercase {
                result.push_str(&token.to_ascii_lowercase());
            } else {
                result.push_str(token);
            }
        }
    }
    Ok(result)
}

fn roman_symbol_value(byte: u8) -> u32 {
    match byte.to_ascii_uppercase() {
        b'I' => 1,
        b'V' => 5,
        b'X' => 10,
        b'L' => 50,
        b'C' => 100,
        b'D' => 500,
        b'M' => 1000,
        _ => 0,
    }
}

pub fn roman_decode(input: &str) -> Result<u32, RomanError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(RomanError::Empty);
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if !matches!(
            byte,
            b'I' | b'V' | b'X' | b'L' | b'C' | b'D' | b'M' | b'N'
                | b'i' | b'v' | b'x' | b'l' | b'c' | b'd' | b'm' | b'n'
        ) {
            return Err(RomanError::InvalidByte { index });
        }
    }

    let first_is_lower = bytes[0].is_ascii_lowercase();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte.is_ascii_alphabetic() && byte.is_ascii_lowercase() != first_is_lower {
            return Err(RomanError::MixedCase { index });
        }
    }

    if bytes.len() == 1 && matches!(bytes[0], b'N' | b'n') {
        return Ok(0);
    }
    if let Some(index) = bytes.iter().position(|&byte| matches!(byte, b'N' | b'n')) {
        let _ = index;
        return Err(RomanError::NonCanonical { index: 0 });
    }

    let mut total = 0u32;
    let mut position = 0usize;
    while position < bytes.len() {
        let current = roman_symbol_value(bytes[position]);
        if position + 1 < bytes.len() {
            let next = roman_symbol_value(bytes[position + 1]);
            if next > current {
                total = total.saturating_add(next - current).min(4000);
                position += 2;
                continue;
            }
        }
        total = total.saturating_add(current).min(4000);
        position += 1;
    }

    if total > 3999 {
        return Err(RomanError::Range);
    }

    let case = if first_is_lower {
        RomanCase::Lower
    } else {
        RomanCase::Upper
    };
    let canonical = roman_encode(total, case)?;
    if input != canonical {
        let differing_index = input
            .bytes()
            .zip(canonical.bytes())
            .position(|(actual, expected)| actual != expected)
            .unwrap_or_else(|| input.len().min(canonical.len()));
        return Err(RomanError::NonCanonical {
            index: differing_index,
        });
    }

    Ok(total)
}

fn main() {}
