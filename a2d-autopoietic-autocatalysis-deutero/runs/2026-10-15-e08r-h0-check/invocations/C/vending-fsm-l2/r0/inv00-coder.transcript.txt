    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut catalog = HashMap::new();
        for product in products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId { id: product.id });
            }
            if catalog.contains_key(&product.id) {
                return Err(Error::DuplicateId { id: product.id });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice { id: product.id });
            }
            catalog.insert(product.id, product.price);
        }

        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        Ok(Self {
            products: catalog,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let index = match coin {
            1 => 0,
            3 => 1,
            4 => 2,
            10 => 3,
            _ => return Err(Error::BadCoin { coin }),
        };

        let _ = index;
        if self.credit + coin > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let price_exists = self.products.contains_key(id);
        if !price_exists {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = match self.selected.as_ref() {
            Some(id) => id,
            None => return Err(Error::NoSelection),
        };
        let price = *self.products.get(selected_id).unwrap();
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = match choose_change(amount, self.tubes) {
            Some(change) => change,
            None => return Err(Error::NoChange { amount }),
        };

        let mut proposed = self.tubes;
        for (i, count) in change.iter().enumerate() {
            proposed[i] -= count;
        }
        for &coin in &self.escrow {
            let i = match coin {
                1 => 0,
                3 => 1,
                4 => 2,
                10 => 3,
                _ => unreachable!(),
            };
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        let mut change_coins = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..change[i] {
                change_coins.push(DENOMINATIONS[i]);
            }
        }

        let receipt = Receipt {
            id: selected_id.clone(),
            price,
            change: change_coins,
        };
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }
}

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    if amount == 0 {
        return Some([0; 4]);
    }

    let mut greedy = [0u16; 4];
    let mut remaining = amount;
    for i in (0..4).rev() {
        let coin = DENOMINATIONS[i];
        let take = tubes[i].min(remaining / coin);
        greedy[i] = take;
        remaining -= take * coin;
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    search_change(0, amount, tubes, [0; 4], &mut best);
    best
}

fn search_change(
    index: usize,
    remaining: u16,
    tubes: [u16; 4],
    current: [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if index == 4 {
        if remaining != 0 {
            return;
        }
        let replace = match best {
            None => true,
            Some(old) => {
                let total: u32 = current.iter().map(|&x| x as u32).sum();
                let old_total: u32 = old.iter().map(|&x| x as u32).sum();
                total < old_total || (total == old_total && lex_greater_desc(&current, old))
            }
        };
        if replace {
            *best = Some(current);
        }
        return;
    }

    let coin = DENOMINATIONS[index];
    let max_by_value = remaining / coin;
    let max_count = tubes[index].min(max_by_value);
    for count in 0..=max_count {
        let used = count * coin;
        let next_remaining = remaining - used;

        if let Some(old) = best.as_ref() {
            let current_total: u32 = current.iter().map(|&x| x as u32).sum();
            if current_total + count as u32 > old.iter().map(|&x| x as u32).sum() {
                continue;
            }
        }

        let mut next = current;
        next[index] = count;
        search_change(index + 1, next_remaining, tubes, next, best);
    }
}

fn lex_greater_desc(a: &[u16; 4], b: &[u16; 4]) -> bool {
    for i in (0..4).rev() {
        if a[i] != b[i] {
            return a[i] > b[i];
        }
    }
    false
}

fn main() {}
