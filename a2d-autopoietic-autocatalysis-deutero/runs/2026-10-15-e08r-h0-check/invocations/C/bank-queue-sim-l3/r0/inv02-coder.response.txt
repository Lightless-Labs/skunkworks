```rust
use std::collections::HashSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    remaining: u64,
    paused_at: Option<u64>,
}

struct Running {
    customer: Customer,
    work_at: u64,
    finish: u64,
}

fn waiting_cmp(a: &Waiting, b: &Waiting) -> std::cmp::Ordering {
    use std::cmp::Ordering;

    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.paused_at, b.paused_at) {
            (Some(x), Some(y)) => x.cmp(&y).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
        ids.insert(customer.id);
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut next_arrival = 0usize;
    let mut running: Option<Running> = None;
    let mut started = HashSet::new();
    let mut last_dispatched: Option<u32> = None;
    let mut finished_count = 0usize;
    let mut time = arrivals[0].arrival;

    while finished_count < customers.len() {
        if running.as_ref().is_some_and(|active| active.finish == time) {
            let done = running.take().unwrap();
            events.push(Event {
                time,
                customer: done.customer.id,
                kind: Kind::Finish,
            });
            finished_count += 1;
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                remaining: customer.service,
                customer,
                paused_at: None,
            });
            next_arrival += 1;
        }

        if let Some(active) = running.as_ref() {
            if time >= active.work_at {
                waiting.sort_by(waiting_cmp);
                if let Some(best) = waiting.first() {
                    if best.customer.priority < active.customer.priority {
                        let by = best.customer.id;
                        let paused = running.take().unwrap();
                        let remaining = paused.finish - time;
                        events.push(Event {
                            time,
                            customer: paused.customer.id,
                            kind: Kind::Preempt { remaining, by },
                        });
                        waiting.push(Waiting {
                            customer: paused.customer,
                            remaining,
                            paused_at: Some(time),
                        });
                    }
                }
            }
        }

        if running.is_none() {
            loop {
                if waiting.is_empty() {
                    break;
                }

                waiting.sort_by(waiting_cmp);
                let selected = waiting.remove(0);
                let id = selected.customer.id;
                let cost = match last_dispatched {
                    Some(last) if last != id => switch_cost,
                    _ => 0,
                };

                let work_at = time.checked_add(cost).ok_or(Error::TimeOverflow {
                    customer: id,
                    time,
                })?;
                let finish = work_at
                    .checked_add(selected.remaining)
                    .ok_or(Error::TimeOverflow {
                        customer: id,
                        time,
                    })?;

                let kind = if started.insert(id) {
                    Kind::Start {
                        remaining: selected.remaining,
                        work_at,
                        finish,
                    }
                } else {
                    Kind::Resume {
                        remaining: selected.remaining,
                        work_at,
                        finish,
                    }
                };

                events.push(Event {
                    time,
                    customer: id,
                    kind,
                });
                last_dispatched = Some(id);
                running = Some(Running {
                    customer: selected.customer,
                    work_at,
                    finish,
                });

                if finish != time {
                    break;
                }

                running = None;
                events.push(Event {
                    time,
                    customer: id,
                    kind: Kind::Finish,
                });
                finished_count += 1;
            }
        }

        if finished_count == customers.len() {
            break;
        }

        let mut next_time = None;
        if next_arrival < arrivals.len() {
            next_time = Some(arrivals[next_arrival].arrival);
        }
        if let Some(active) = running.as_ref() {
            for candidate in [
                Some(active.finish),
                (active.work_at > time).then_some(active.work_at),
            ]
            .into_iter()
            .flatten()
            {
                next_time = Some(next_time.map_or(candidate, |t: u64| t.min(candidate)));
            }
        }

        if let Some(t) = next_time {
            time = t;
        } else {
            break;
        }
    }

    Ok(events)
}

fn main() {}
```
