#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
}

fn split_lines(text: &str) -> Vec<String> {
    if text.is_empty() {
        return Vec::new();
    }
    text[..text.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn make_changes(base: &[String], other: &[String]) -> Vec<Change> {
    let n = base.len();
    let m = other.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if base[i] == other[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j, mut base_pos) = (0usize, 0usize, 0usize);
    let mut changes = Vec::new();
    let mut active_start: Option<usize> = None;
    let mut active_end = 0usize;
    let mut replacement = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 active_start: &mut Option<usize>,
                 active_end: &mut usize,
                 replacement: &mut Vec<String>| {
        if let Some(start) = active_start.take() {
            changes.push(Change {
                start,
                end: *active_end,
                replacement: std::mem::take(replacement),
            });
        }
    };

    while i < n || j < m {
        if i < n && j < m && base[i] == other[j] {
            flush(
                &mut changes,
                &mut active_start,
                &mut active_end,
                &mut replacement,
            );
            i += 1;
            j += 1;
            base_pos += 1;
        } else if i == n {
            if active_start.is_none() {
                active_start = Some(base_pos);
                active_end = base_pos;
            }
            replacement.push(other[j].clone());
            j += 1;
        } else if j == m {
            if active_start.is_none() {
                active_start = Some(base_pos);
                active_end = base_pos;
            }
            i += 1;
            base_pos += 1;
            active_end = base_pos;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            let delete = if delete_score > insert_score {
                true
            } else if insert_score > delete_score {
                false
            } else {
                (i + j) % 2 == 0
            };

            if active_start.is_none() {
                active_start = Some(base_pos);
                active_end = base_pos;
            }
            if delete {
                i += 1;
                base_pos += 1;
                active_end = base_pos;
            } else {
                replacement.push(other[j].clone());
                j += 1;
            }
        }
    }

    flush(
        &mut changes,
        &mut active_start,
        &mut active_end,
        &mut replacement,
    );
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_empty = a.start == a.end;
    let b_empty = b.start == b.end;
    match (a_empty, b_empty) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

struct Component {
    start: usize,
    end: usize,
    left: Vec<usize>,
    right: Vec<usize>,
}

fn project(
    base: &[String],
    start: usize,
    end: usize,
    changes: &[Change],
    indices: &[usize],
) -> Vec<String> {
    if indices.is_empty() {
        return base[start..end].to_vec();
    }

    let mut ordered = indices.to_vec();
    ordered.sort_by_key(|&idx| (changes[idx].start, changes[idx].end));

    let mut out = Vec::new();
    let mut cursor = start;
    for idx in ordered {
        let change = &changes[idx];
        if change.start > cursor {
            out.extend_from_slice(&base[cursor..change.start]);
        }
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    if cursor < end {
        out.extend_from_slice(&base[cursor..end]);
    }
    out
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = make_changes(&base_lines, &left_lines);
    let right_changes = make_changes(&base_lines, &right_lines);

    let total = left_changes.len() + right_changes.len();
    let mut parent: Vec<usize> = (0..total).collect();

    fn find(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = find(parent, parent[x]);
        }
        parent[x]
    }

    fn union(parent: &mut [usize], a: usize, b: usize) {
        let root_a = find(parent, a);
        let root_b = find(parent, b);
        if root_a != root_b {
            parent[root_b] = root_a;
        }
    }

    for i in 0..left_changes.len() {
        for j in 0..right_changes.len() {
            if interacts(&left_changes[i], &right_changes[j]) {
                union(&mut parent, i, left_changes.len() + j);
            }
        }
    }

    let mut components: Vec<Component> = Vec::new();
    let mut roots: Vec<usize> = Vec::new();

    for i in 0..total {
        let root = find(&mut parent, i);
        let (change, is_left, idx) = if i < left_changes.len() {
            (&left_changes[i], true, i)
        } else {
            (&right_changes[i - left_changes.len()], false, i - left_changes.len())
        };

        if let Some(component_index) = roots.iter().position(|&r| r == root) {
            let component = &mut components[component_index];
            component.start = component.start.min(change.start);
            component.end = component.end.max(change.end);
            if is_left {
                component.left.push(idx);
            } else {
                component.right.push(idx);
            }
        } else {
            roots.push(root);
            components.push(Component {
                start: change.start,
                end: change.end,
                left: if is_left { vec![idx] } else { Vec::new() },
                right: if is_left { Vec::new() } else { vec![idx] },
            });
        }
    }

    components.sort_by_key(|c| (c.start, usize::from(c.start != c.end), c.end));

    let mut output_lines: Vec<String> = Vec::new();
    let mut cursor = 0usize;
    let mut conflicts = Vec::new();

    for component in components {
        if cursor < component.start {
            output_lines.extend_from_slice(&base_lines[cursor..component.start]);
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(
            &base_lines,
            component.start,
            component.end,
            &left_changes,
            &component.left,
        );
        let right_projection = project(
            &base_lines,
            component.start,
            component.end,
            &right_changes,
            &component.right,
        );

        if left_projection == right_projection {
            output_lines.extend(left_projection);
        } else if left_projection == base_projection {
            output_lines.extend(right_projection);
        } else if right_projection == base_projection {
            output_lines.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output_lines.extend(left_projection),
                Resolution::PreferRight => output_lines.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output_lines.len();
                    let base_len = component.end - component.start;
                    let base_marker_start = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    let output_len =
                        4 + left_projection.len() + base_projection.len() + right_projection.len();

                    output_lines.push(format!("<<<<<<< LEFT {}", ordinal));
                    output_lines.extend(left_projection);
                    output_lines.push(format!(
                        "||||||| BASE {},{}",
                        base_marker_start, base_len
                    ));
                    output_lines.extend(base_projection);
                    output_lines.push("=======".to_owned());
                    output_lines.extend(right_projection);
                    output_lines.push(format!(">>>>>>> RIGHT {}", ordinal));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len,
                    });
                }
            }
        }

        cursor = component.end;
    }

    if cursor < base_lines.len() {
        output_lines.extend_from_slice(&base_lines[cursor..]);
    }

    let mut text = String::new();
    for line in &output_lines {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

fn main() {}
