fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut quoted = false;
    let mut after_quote = false;
    let mut field_started = false;
    let mut opening_quote = 0usize;
    let mut record_index = 0usize;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }
        if !quoted && byte == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        if quoted {
            if byte == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                } else {
                    quoted = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            if byte == delimiter {
                fields.push(std::mem::take(&mut field));
                after_quote = false;
                field_started = false;
                i += 1;
                continue;
            }
            if byte == b'\n' || byte == b'\r' {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                after_quote = false;
                field_started = false;
                record_index += 1;
                i += if byte == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: i,
            });
        }

        if byte == delimiter {
            fields.push(std::mem::take(&mut field));
            field_started = false;
            i += 1;
            continue;
        }
        if byte == b'\n' || byte == b'\r' {
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            field_started = false;
            record_index += 1;
            i += if byte == b'\r' { 2 } else { 1 };
            continue;
        }
        if byte == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: i,
                });
            }
            quoted = true;
            opening_quote = i;
            field_started = true;
            i += 1;
            continue;
        }

        let ch = input[i..].chars().next().unwrap();
        field.push(ch);
        field_started = true;
        i += ch.len_utf8();
    }

    if quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if !fields.is_empty() || !field.is_empty() || field_started || after_quote {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let mut header = None;
    let mut rows = records;

    if config.headers && !rows.is_empty() {
        let first = rows.remove(0);

        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen = std::collections::HashMap::<String, usize>::new();
            for (second, name) in first.iter().enumerate() {
                if let Some(&first_index) = seen.get(name) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: first_index,
                        second,
                    });
                }
                seen.insert(name.clone(), second);
            }
        }

        for (data_index, row) in rows.iter().enumerate() {
            if row.len() != first.len() {
                return Err(CsvError::HeaderWidth {
                    record: data_index + 1,
                    expected: first.len(),
                    actual: row.len(),
                });
            }
        }
        header = Some(first);
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    finished: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        valid_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            finished: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.finished {
            return result.clone();
        }

        let result = parse_with(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()

fn integer_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    if bytes == b"0" {
        return true;
    }
    let start = usize::from(bytes.first() == Some(&b'-'));
    if start >= bytes.len() || !(b'1'..=b'9').contains(&bytes[start]) {
        return false;
    }
    bytes[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    let negative = bytes.first() == Some(&b'-');
    let start = usize::from(negative);
    if start >= bytes.len() {
        return false;
    }
    let Some(relative_dot) = bytes[start..].iter().position(|&byte| byte == b'.') else {
        return false;
    };
    let dot = start + relative_dot;

    if bytes.len() != dot + 3 || !bytes[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }

    let integral = &bytes[start..dot];
    if integral == b"0" {
        return !negative || &bytes[dot + 1..] != b"00";
    }

    !integral.is_empty()
        && (b'1'..=b'9').contains(&integral[0])
        && integral[1..].iter().all(u8::is_ascii_digit)
}

fn decimal_value(value: &str) -> Option<i64> {
    let negative = value.starts_with('-');
    let digits = if negative { &value[1..] } else { value };
    let mut magnitude: i128 = 0;
    for byte in digits.bytes().filter(|&byte| byte != b'.') {
        magnitude = magnitude.checked_mul(10)?.checked_add((byte - b'0') as i128)?;
    }
    if negative {
        magnitude = -magnitude;
    }
    i64::try_from(magnitude).ok()
}

fn evaluate_cell(
    table: &Table,
    row_index: usize,
    row: &[String],
    spec_index: usize,
    spec: &ColumnSpec,
) -> Result<Value, CellError> {
    let column = match &spec.column {
        ColumnRef::Index(index) => Some(*index),
        ColumnRef::Name(name) => table.column_index(name),
    };

    let Some(column_index) = column else {
        return Err(CellError {
            row: row_index,
            spec: spec_index,
            column: None,
            kind: CellErrorKind::UnknownColumn,
        });
    };

    let Some(value) = row.get(column_index) else {
        return Err(CellError {
            row: row_index,
            spec: spec_index,
            column: Some(column_index),
            kind: CellErrorKind::MissingCell,
        });
    };

    if value.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(column_index),
                kind: CellErrorKind::Empty,
            });
        }
    }

    let make_error = |kind| CellError {
        row: row_index,
        spec: spec_index,
        column: Some(column_index),
        kind,
    };

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(value.clone())),
        ColumnType::Integer => {
            if !integer_syntax(value) {
                return Err(make_error(CellErrorKind::InvalidInteger));
            }
            value
                .parse::<i64>()
                .map(Value::Integer)
                .map_err(|_| make_error(CellErrorKind::IntegerOverflow))
        }
        ColumnType::Decimal => {
            if !decimal_syntax(value) {
                return Err(make_error(CellErrorKind::InvalidDecimal));
            }
            decimal_value(value)
                .map(Value::Decimal)
                .ok_or_else(|| make_error(CellErrorKind::DecimalOverflow))
        }
        ColumnType::Boolean => match value.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(make_error(CellErrorKind::InvalidBoolean)),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        let mut failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            match evaluate_cell(table, row_index, row, spec_index, spec) {
                Ok(value) => output[spec_index] = Some(value),
                Err(error) => {
                    result.errors.push(error);
                    failed = true;
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                }
            }
        }

        result.rows.push(output);
        if policy == ErrorPolicy::Stop && failed {
            break;
        }
    }

    result
}
