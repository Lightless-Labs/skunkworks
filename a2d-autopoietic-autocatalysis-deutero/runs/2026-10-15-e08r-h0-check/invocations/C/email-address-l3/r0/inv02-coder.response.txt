#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if i + 2 >= bytes.len()
        || bytes[i] != b'\r'
        || bytes[i + 1] != b'\n'
        || !is_hws(bytes[i + 2])
    {
        return None;
    }
    let mut j = i + 2;
    while j < bytes.len() && is_hws(bytes[j]) {
        j += 1;
    }
    Some(j)
}

fn add_run(runs: &mut Vec<Run>, start: usize, end: usize) {
    if let Some(last) = runs.last_mut() {
        if last.end == start {
            last.end = end;
            return;
        }
    }
    runs.push(Run { start, end });
}

fn cleanup(input: &str) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let bytes = input.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut map = Vec::with_capacity(bytes.len());
    let mut runs = Vec::new();
    let mut i = 0usize;
    let mut protected = false;

    while i < bytes.len() {
        let b = bytes[i];

        if protected {
            out.push(b);
            map.push(i);
            if b == b'\\' {
                if i + 1 < bytes.len() {
                    i += 1;
                    out.push(bytes[i]);
                    map.push(i);
                }
            } else if b == b'"' {
                protected = false;
            }
            i += 1;
            continue;
        }

        if b == b'"' && out.is_empty() {
            protected = true;
            out.push(b);
            map.push(i);
            i += 1;
            continue;
        }

        if is_hws(b) {
            let start = i;
            while i < bytes.len() && is_hws(bytes[i]) {
                i += 1;
            }
            add_run(&mut runs, start, i);
            continue;
        }

        if b == b'\r' || b == b'\n' {
            if let Some(end) = fold_end(bytes, i) {
                let start = i;
                i = end;
                add_run(&mut runs, start, end);
            } else {
                return Err(EmailError::BadFold { at: i });
            }
            continue;
        }

        if b == b'(' {
            let start = i;
            let outer = i;
            let mut depth = 0usize;
            while i < bytes.len() {
                let c = bytes[i];

                if c == b'\\' {
                    if i + 1 >= bytes.len() || !(0x20..=0x7e).contains(&bytes[i + 1]) {
                        return Err(EmailError::CommentEscape { at: i });
                    }
                    i += 2;
                    continue;
                }

                if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(bytes, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }

                if c == b'(' {
                    depth += 1;
                    if depth > 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    i += 1;
                    continue;
                }

                if c == b')' {
                    depth -= 1;
                    i += 1;
                    if depth == 0 {
                        break;
                    }
                    continue;
                }

                if !(0x20..=0x7e).contains(&c) && c != b'\t' {
                    return Err(EmailError::CommentChar { at: i });
                }
                i += 1;
            }

            if depth != 0 {
                return Err(EmailError::UnclosedComment { at: outer });
            }
            add_run(&mut runs, start, i);
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        out.push(b);
        map.push(i);
        i += 1;
    }

    Ok((out, map, runs))
}

fn pos(map: &[usize], idx: usize, input_len: usize) -> usize {
    map.get(idx).copied().unwrap_or(input_len)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let raw = input.as_bytes();
    if let Some(i) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let (bytes, map, runs) = cleanup(input)?;

    for run in &runs {
        let at_start = run.start > 0 && raw[run.start - 1] == b'@';
        let at_end = run.end < raw.len() && raw[run.end] == b'@';
        if !(run.start == 0 || run.end == raw.len() || at_start || at_end) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut closing_quote = None;
    if bytes[0] == b'"' {
        let mut i = 1usize;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                closing_quote = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if closing_quote.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: pos(&map, 0, raw.len()),
            });
        }
    }

    let mut ats = Vec::new();
    let mut i = closing_quote.map_or(0, |q| q + 1);
    while i < bytes.len() {
        if bytes[i] == b'@' {
            ats.push(i);
        }
        i += 1;
    }

    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: pos(&map, ats[1], raw.len()),
        });
    }

    let at_idx = ats[0];
    if at_idx == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(&map, at_idx, raw.len()),
        });
    }

    let local_bytes = &bytes[..at_idx];
    let (local, quoted) = if local_bytes[0] == b'"' {
        let q = closing_quote.unwrap();
        if q + 1 != at_idx {
            return Err(EmailError::TrailingLocal {
                at: pos(&map, q + 1, raw.len()),
            });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1usize;
        while j < q {
            let b = bytes[j];
            if b == b'\\' {
                if j + 1 >= q || !matches!(bytes[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: pos(&map, j, raw.len()),
                    });
                }
                decoded.push(bytes[j + 1]);
                origins.push(pos(&map, j, raw.len()));
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: pos(&map, j, raw.len()),
                    });
                }
                decoded.push(b);
                origins.push(pos(&map, j, raw.len()));
                j += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: pos(&map, q, raw.len()),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: pos(&map, 64, raw.len()),
            });
        }

        for (j, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || local_bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: pos(&map, j, raw.len()),
                    });
                }
            } else if !(b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)) {
                return Err(EmailError::LocalChar {
                    at: pos(&map, j, raw.len()),
                });
            }
        }

        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start = at_idx + 1;
    let mut domain_end = bytes.len();
    let root_dot = bytes[domain_start..]
        .last()
        .copied()
        .is_some_and(|b| b == b'.');
    if root_dot {
        domain_end -= 1;
    }

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: pos(&map, domain_start, raw.len()),
        });
    }

    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(&map, domain_start + 253, raw.len()),
        });
    }

    let mut labels = Vec::new();
    let mut start = domain_start;
    for j in domain_start..domain_end {
        if bytes[j] == b'.' {
            if j == start {
                return Err(EmailError::EmptyLabel {
                    at: pos(&map, j, raw.len()),
                });
            }
            labels.push((start, j));
            start = j + 1;
        }
    }

    if start == domain_end {
        let empty_at = if root_dot {
            pos(&map, domain_end, raw.len())
        } else {
            pos(&map, domain_end, raw.len())
        };
        return Err(EmailError::EmptyLabel { at: empty_at });
    }
    labels.push((start, domain_end));

    for &(s, e) in &labels {
        if e - s > 63 {
            return Err(EmailError::LabelTooLong {
                at: pos(&map, s + 63, raw.len()),
            });
        }
        if bytes[s] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(&map, s, raw.len()),
            });
        }
        if bytes[e - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(&map, e - 1, raw.len()),
            });
        }
        for j in s..e {
            if !(bytes[j].is_ascii_alphanumeric() || bytes[j] == b'-') {
                return Err(EmailError::DomainChar {
                    at: pos(&map, j, raw.len()),
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: pos(&map, domain_start, raw.len()),
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: pos(&map, last_start, raw.len()),
        });
    }

    let domain = bytes[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}

fn main() {}
