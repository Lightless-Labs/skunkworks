#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.split('/')
        .filter(|part| !part.is_empty())
        .last()
        .is_some_and(|part| part == "." || part == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|part| !part.is_empty()).collect()
}

fn render(parts: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", parts.join("/"))
        }
    } else if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn reduce_plain(path: &str, absolute: bool) -> Result<Vec<String>, PathError> {
    let mut stack = Vec::new();
    let mut offset = 0;

    for part in path.split('/') {
        if !part.is_empty() {
            match part {
                "." => {}
                ".." => {
                    if stack.last().is_some_and(|last: &String| last != "..") {
                        stack.pop();
                    } else if absolute {
                        return Err(PathError::Escape { at: offset });
                    } else {
                        stack.push("..".to_owned());
                    }
                }
                _ => stack.push(part.to_owned()),
            }
        }
        offset += part.len() + 1;
    }
    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let parts = reduce_plain(path, absolute)?;
    Ok(render(&parts, absolute, intent))
}

fn normalized_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized).into_iter().map(str::to_owned).collect()),
        Err(_) => Err(PathError::InvalidCwd),
    }
}

fn normalized_home(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidHome);
    }
    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized).into_iter().map(str::to_owned).collect()),
        Err(_) => Err(PathError::InvalidHome),
    }
}

fn prepare_request<'a>(path: &'a str, home: Option<&[String]>) -> Result<(bool, &'a str, usize), PathError> {
    validate_raw(path)?;

    if path.starts_with('~') {
        if path == "~" {
            let home = home.ok_or(PathError::MissingHome)?;
            let _ = home;
            return Ok((true, "", 1));
        }
        if path.starts_with("~/") {
            home.ok_or(PathError::MissingHome)?;
            return Ok((true, &path[2..], 2));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    let absolute = path.starts_with('/');
    Ok((absolute, path, 0))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = normalized_context(cwd)?;
    let home_parts = match home {
        Some(h) => Some(normalized_home(h)?),
        None => None,
    };

    let (absolute, remainder, _) = prepare_request(path, home_parts.as_deref())?;
    let intent = directory_intent(path);
    let mut stack = if path.starts_with('~') {
        home_parts.unwrap()
    } else if absolute {
        Vec::new()
    } else {
        cwd_parts
    };

    let mut offset = if path.starts_with("~/") { 2 } else if path == "~" { 1 } else { 0 };
    for part in remainder.split('/') {
        if !part.is_empty() {
            match part {
                "." => {}
                ".." => {
                    if stack.pop().is_none() {
                        return Err(PathError::Escape { at: offset });
                    }
                }
                _ => stack.push(part.to_owned()),
            }
        }
        offset += part.len() + 1;
    }

    Ok(render(&stack, true, intent))
}

struct Frame<'a> {
    raw: &'a str,
    offset_base: usize,
    parts: Vec<&'a str>,
    next: usize,
    active_link: Option<usize>,
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = normalized_context(cwd)?;
    let home_parts = match home {
        Some(h) => Some(normalized_home(h)?),
        None => None,
    };

    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path).is_ok_and(|n| n == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((_, first)) = seen.iter().find(|(p, _)| *p == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }

    let (absolute, remainder, base_offset) = prepare_request(path, home_parts.as_deref())?;
    let intent = directory_intent(path);
    let tilde = path.starts_with('~');
    let mut stack = if tilde {
        home_parts.unwrap()
    } else if absolute {
        Vec::new()
    } else {
        cwd_parts
    };

    let mut frames = Vec::new();
    if tilde && path == "~" {
        frames.push(Frame {
            raw: path,
            offset_base: 1,
            parts: Vec::new(),
            next: 0,
            active_link: None,
        });
    } else {
        frames.push(Frame {
            raw: if base_offset == 0 { path } else { remainder },
            offset_base: base_offset,
            parts: components(remainder),
            next: 0,
            active_link: None,
        });
    }

    let mut active = Vec::new();
    let mut expansions = 0usize;

    while let Some(frame) = frames.last_mut() {
        if frame.next >= frame.parts.len() {
            let done = frames.pop().unwrap();
            if let Some(index) = done.active_link {
                if active.last() == Some(&index) {
                    active.pop();
                }
            }
            continue;
        }

        let part = frame.parts[frame.next];
        frame.next += 1;

        let part_at = if frame.raw.is_empty() {
            frame.offset_base
        } else {
            let start = frame
                .raw
                .match_indices(part)
                .map(|(i, _)| i)
                .nth(frame.next - 1)
                .unwrap_or(0);
            frame.offset_base + start
        };

        match part {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: part_at });
                }
            }
            _ => {
                stack.push(part.to_owned());
                let key = format!("/{}", stack.join("/"));
                if let Some((link_index, link)) = links
                    .iter()
                    .enumerate()
                    .find(|(_, link)| link.path == key)
                {
                    if active.contains(&link_index) {
                        return Err(PathError::Loop { index: link_index });
                    }
                    if expansions >= max_expansions {
                        return Err(PathError::Limit { index: link_index });
                    }
                    expansions += 1;
                    active.push(link_index);

                    stack.pop();
                    let target = link.target.as_str();
                    let target_absolute = target.starts_with('/');
                    if target_absolute {
                        stack.clear();
                    }
                    frames.push(Frame {
                        raw: target,
                        offset_base: 0,
                        parts: components(target),
                        next: 0,
                        active_link: Some(link_index),
                    });
                }
            }
        }
    }

    Ok(render(&stack, true, intent))
}
