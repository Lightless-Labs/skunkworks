    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .last()
            .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str, base_offset: usize) -> Vec<(String, usize)> {
    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((path[start..i].to_owned(), base_offset + start));
    }

    result
}

fn render(parts: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", parts.join("/"))
    } else if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };

    if absolute && parts.is_empty() {
        result = "/".to_owned();
    }

    if intent && result != "/" && result != "." {
        result.push('/');
    }

    result
}

fn reduce(
    parts: &mut Vec<String>,
    component: &str,
    absolute: bool,
    at: usize,
) -> Result<(), PathError> {
    match component {
        "." => {}
        ".." => {
            if parts.last().is_some_and(|last| last != "..") {
                parts.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                parts.push("..".to_owned());
            }
        }
        _ => parts.push(component.to_owned()),
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut parts = Vec::new();

    for (component, at) in components(path, 0) {
        reduce(&mut parts, &component, absolute, at)?;
    }

    Ok(render(&parts, absolute, intent))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(components(&normalized, 0)
        .into_iter()
        .map(|(component, _)| component)
        .collect())
}

fn prepare_request<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(bool, &'a str, usize, Vec<String>), PathError> {
    validate_raw(path)?;

    if path.starts_with('~') {
        if path == "~" || path.starts_with("~/") {
            let home = home.ok_or(PathError::MissingHome)?;
            let remainder_start = if path == "~" { path.len() } else { 2 };
            let remainder = &path[remainder_start..];
            let comps = components(remainder, remainder_start);
            return Ok((true, remainder, remainder_start, comps.into_iter().map(|x| x.0).collect()));
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    }

    let absolute = path.starts_with('/');
    let comps = components(path, 0);
    Ok((
        absolute,
        path,
        0,
        comps.into_iter().map(|(component, _)| component).collect(),
    ))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = match home {
        Some(value) => Some(validate_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let intent = directory_intent(path);

    let (absolute, request_parts) = if path.starts_with('~') {
        if path == "~" || path.starts_with("~/") {
            let start = home_parts.ok_or(PathError::MissingHome)?;
            let remainder_start = if path == "~" { path.len() } else { 2 };
            let mut request = Vec::new();
            for (component, at) in components(&path[remainder_start..], remainder_start) {
                request.push((component, at));
            }
            let mut parts = start;
            for (component, at) in request {
                reduce(&mut parts, &component, true, at)?;
            }
            return Ok(render(&parts, true, intent));
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {

    } else {
        (true, components(path, 0))
    };

    let mut parts = if absolute { Vec::new() } else { cwd_parts };
    for (component, at) in request_parts {
        reduce(&mut parts, &component, true, at)?;
    }

    Ok(render(&parts, true, intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .map(|normalized| normalized == link.path)
                .unwrap_or(false);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });

            .iter()
            .position(|earlier| earlier.path == link.path)
        {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
    }
    Ok(())
}

struct WorkComponent {
    value: String,
    at: usize,
    source: Source,
}

#[derive(Clone, Copy)]
enum Source {
    Request,
    Target,
}

fn resolve_links_inner(
    parts: &mut Vec<String>,
    queue: &mut Vec<WorkComponent>,
    links: &[Link],
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
) -> Result<(), PathError> {
    while !queue.is_empty() {
        let item = queue.remove(0);

        match item.value.as_str() {
            "." => continue,
            ".." => {
                if parts.pop().is_none() {
                    return Err(PathError::Escape { at: item.at });
                }
                continue;
            }
            _ => {}
        }

        parts.push(item.value.clone());
        let current = format!("/{}", parts.join("/"));
        if let Some(index) = links.iter().position(|link| link.path == current) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if *expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            *expansions += 1;
            active.push(index);

            let link = &links[index];
            let target_absolute = link.target.starts_with('/');
            let mut target_queue: Vec<WorkComponent> = components(&link.target, 0)
                .into_iter()
                .map(|(value, at)| WorkComponent {
                    value,
                    at,
                    source: Source::Target,
                })
                .collect();

            let saved_parts = if target_absolute {
                let old = std::mem::take(parts);
                old
            } else {
                parts.pop();
                Vec::new()
            };

            resolve_links_inner(
                parts,
                &mut target_queue,
                links,
                active,
                expansions,
                max_expansions,
            )?;

            active.pop();

            if !queue.is_empty() {
                let _ = saved_parts;
            }
        }
    }

    Ok(())
}

    links: &[Link],
    max_expansions: usize,

            )
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });

    } else {
        (cwd_parts, path, 0)
    };

        .map(|(value, at)| WorkComponent {
            value,
            at,
            source: Source::Request,

    resolve_links_inner(
        &mut parts,
        &mut queue,
        links,
        &mut Vec::new(),
        &mut expansions,
        max_expansions,
    )?;

    Ok(render(&parts, true, intent))
}

pub struct Link {

#[derive(Clone, Copy)]
struct Component<'a> {
    text: &'a str,
    at: usize,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str, base: usize) -> Vec<Component<'_>> {
    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push(Component {
            text: &path[start..i],
            at: base + start,
        });
    }

    result
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path, 0)
            .last()
            .is_some_and(|component| component.text == "." || component.text == "..")
}

fn render_absolute(stack: &[String], intent: bool) -> String {
    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    if intent && !stack.is_empty() {
        result.push('/');
    }
    result
}

fn render_relative(stack: &[String], intent: bool) -> String {
    if stack.is_empty() {
        return ".".to_string();
    }

    let mut result = stack.join("/");
    if intent {
        result.push('/');
    }
    result
}

fn reduce(
    stack: &mut Vec<String>,
    component: Component<'_>,
    absolute: bool,
) -> Result<(), PathError> {
    match component.text {
        "." => {}
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at: component.at });
            } else {
                stack.push("..".to_string());
            }
        }
        ordinary => stack.push(ordinary.to_string()),
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let mut stack = Vec::new();

    for component in components(path, 0) {
        reduce(&mut stack, component, absolute)?;
    }

    let intent = directory_intent(path);
    if absolute {
        Ok(render_absolute(&stack, intent))
    } else {
        Ok(render_relative(&stack, intent))
    }
}

fn validated_context(path: &str, invalid: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(invalid);
    }
    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized, 0)
            .into_iter()
            .map(|component| component.text.to_string())
            .collect()),
        Err(_) => Err(invalid),
    }
}

fn tilde_remainder<'a>(path: &'a str, home: Option<&str>) -> Result<(&'a str, bool), PathError> {
    if !path.starts_with('~') {
        return Ok((path, false));
    }

    if path == "~" || path.starts_with("~/") {
        if home.is_none() {
            return Err(PathError::MissingHome);
        }
        let remainder = if path == "~" { "" } else { &path[2..] };
        return Ok((remainder, true));
    }

    Err(PathError::UnsupportedTilde { at: 1 })
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validated_context(cwd, PathError::InvalidCwd)?;
    let home_components = match home {
        Some(value) => Some(validated_context(value, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let (request, expanded_tilde) = tilde_remainder(path, home)?;
    let absolute = path.starts_with('/');
    let mut stack = if absolute {
        Vec::new()
    } else if expanded_tilde {
        home_components.unwrap_or_default()
    } else {
        cwd_components
    };

    for component in components(request, if expanded_tilde { 2 } else { 0 }) {
        reduce(&mut stack, component, true)?;
    }

    Ok(render_absolute(&stack, directory_intent(path)))
}

            .position(|previous| previous.path == link.path)
        {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
    }
    Ok(())
}

struct LinkTraversal<'a> {
    links: &'a [Link],
    max_expansions: usize,
    expansions: usize,
    active: Vec<usize>,
    stack: Vec<String>,
}

impl LinkTraversal<'_> {
    fn process_component(
        &mut self,
        component: Component<'_>,
        absolute: bool,
    ) -> Result<(), PathError> {
        match component.text {
            "." => Ok(()),
            ".." => {
                if self.stack.pop().is_none() {
                    if absolute {
                        Err(PathError::Escape { at: component.at })
                    } else {
                        self.stack.push("..".to_string());
                        Ok(())
                    }
                } else {
                    Ok(())
                }
            }
            ordinary => {
                self.stack.push(ordinary.to_string());
                self.expand_match()
            }
        }
    }

    fn expand_match(&mut self) -> Result<(), PathError> {
        let matched = {
            let current = render_absolute(&self.stack, false);
            self.links.iter().position(|link| link.path == current)
        };

        let Some(index) = matched else {
            return Ok(());
        };

        if self.active.contains(&index) {
            return Err(PathError::Loop { index });
        }
        if self.expansions >= self.max_expansions {
            return Err(PathError::Limit { index });
        }
        self.expansions += 1;

        let target = &self.links[index].target;
        let target_absolute = target.starts_with('/');
        if target_absolute {
            self.stack.clear();
        } else {
            self.stack.pop();
        }

        self.active.push(index);
        let result = self.process_path(target, 0, true);
        self.active.pop();
        result
    }

    fn process_path(
        &mut self,
        path: &str,
        base: usize,
        absolute: bool,
    ) -> Result<(), PathError> {
        for component in components(path, base) {
            self.process_component(component, absolute)?;
        }
        Ok(())
    }
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = validated_context(cwd, PathError::InvalidCwd)?;
    let home_components = match home {
        Some(value) => Some(validated_context(value, PathError::InvalidHome)?),
        None => None,
    };

    validate_links(links)?;
    validate_raw(path)?;
    let (request, expanded_tilde) = tilde_remainder(path, home)?;

    let absolute = path.starts_with('/');
    let mut traversal = LinkTraversal {
        links,
        max_expansions,
        expansions: 0,
        active: Vec::new(),
        stack: if absolute {
            Vec::new()
        } else if expanded_tilde {
            home_components.unwrap_or_default()
        } else {
            cwd_components
        },
    };

    if !absolute {
        let starting = traversal.stack.clone();
        traversal.stack.clear();
        for component in starting {
            traversal.stack.push(component);
            traversal.expand_match()?;
        }
    }

    traversal.process_path(
        request,
        if expanded_tilde { 2 } else { 0 },
        true,
    )?;

    Ok(render_absolute(
        &traversal.stack,
        directory_intent(path),
    ))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.split('/')
        .filter(|part| !part.is_empty())
        .last()
        .is_some_and(|part| part == "." || part == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|part| !part.is_empty()).collect()
}

fn render(parts: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", parts.join("/"))
        }
    } else if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn reduce_plain(path: &str, absolute: bool) -> Result<Vec<String>, PathError> {
    let mut stack = Vec::new();
    let mut offset = 0;

    for part in path.split('/') {
        if !part.is_empty() {
            match part {
                "." => {}
                ".." => {
                    if stack.last().is_some_and(|last: &String| last != "..") {
                        stack.pop();
                    } else if absolute {
                        return Err(PathError::Escape { at: offset });
                    } else {
                        stack.push("..".to_owned());
                    }
                }
                _ => stack.push(part.to_owned()),
            }
        }
        offset += part.len() + 1;
    }
    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let parts = reduce_plain(path, absolute)?;
    Ok(render(&parts, absolute, intent))
}

fn normalized_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized).into_iter().map(str::to_owned).collect()),
        Err(_) => Err(PathError::InvalidCwd),
    }
}

fn normalized_home(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidHome);
    }
    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized).into_iter().map(str::to_owned).collect()),
        Err(_) => Err(PathError::InvalidHome),
    }
}

fn prepare_request<'a>(path: &'a str, home: Option<&[String]>) -> Result<(bool, &'a str, usize), PathError> {
    validate_raw(path)?;

    if path.starts_with('~') {
        if path == "~" {
            let home = home.ok_or(PathError::MissingHome)?;
            let _ = home;
            return Ok((true, "", 1));
        }
        if path.starts_with("~/") {
            home.ok_or(PathError::MissingHome)?;
            return Ok((true, &path[2..], 2));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    let absolute = path.starts_with('/');
    Ok((absolute, path, 0))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = normalized_context(cwd)?;
    let home_parts = match home {
        Some(h) => Some(normalized_home(h)?),
        None => None,
    };

    let (absolute, remainder, _) = prepare_request(path, home_parts.as_deref())?;
    let intent = directory_intent(path);
    let mut stack = if path.starts_with('~') {
        home_parts.unwrap()
    } else if absolute {
        Vec::new()
    } else {
        cwd_parts
    };

    let mut offset = if path.starts_with("~/") { 2 } else if path == "~" { 1 } else { 0 };
    for part in remainder.split('/') {
        if !part.is_empty() {
            match part {
                "." => {}
                ".." => {
                    if stack.pop().is_none() {
                        return Err(PathError::Escape { at: offset });
                    }
                }
                _ => stack.push(part.to_owned()),
            }
        }
        offset += part.len() + 1;
    }

    Ok(render(&stack, true, intent))
}

struct Frame<'a> {
    raw: &'a str,
    offset_base: usize,
    parts: Vec<&'a str>,
    next: usize,
    active_link: Option<usize>,
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = normalized_context(cwd)?;
    let home_parts = match home {
        Some(h) => Some(normalized_home(h)?),
        None => None,
    };

    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path).is_ok_and(|n| n == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((_, first)) = seen.iter().find(|(p, _)| *p == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }

    let (absolute, remainder, base_offset) = prepare_request(path, home_parts.as_deref())?;
    let intent = directory_intent(path);
    let tilde = path.starts_with('~');
    let mut stack = if tilde {
        home_parts.unwrap()
    } else if absolute {
        Vec::new()
    } else {
        cwd_parts
    };

    let mut frames = Vec::new();
    if tilde && path == "~" {
        frames.push(Frame {
            raw: path,
            offset_base: 1,
            parts: Vec::new(),
            next: 0,
            active_link: None,
        });
    } else {
        frames.push(Frame {
            raw: if base_offset == 0 { path } else { remainder },
            offset_base: base_offset,
            parts: components(remainder),
            next: 0,
            active_link: None,
        });
    }

    let mut active = Vec::new();
    let mut expansions = 0usize;

    while let Some(frame) = frames.last_mut() {
        if frame.next >= frame.parts.len() {
            let done = frames.pop().unwrap();
            if let Some(index) = done.active_link {
                if active.last() == Some(&index) {
                    active.pop();
                }
            }
            continue;
        }

        let part = frame.parts[frame.next];
        frame.next += 1;

        let part_at = if frame.raw.is_empty() {
            frame.offset_base
        } else {
            let start = frame
                .raw
                .match_indices(part)
                .map(|(i, _)| i)
                .nth(frame.next - 1)
                .unwrap_or(0);
            frame.offset_base + start
        };

        match part {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: part_at });
                }
            }
            _ => {
                stack.push(part.to_owned());
                let key = format!("/{}", stack.join("/"));
                if let Some((link_index, link)) = links
                    .iter()
                    .enumerate()
                    .find(|(_, link)| link.path == key)
                {
                    if active.contains(&link_index) {
                        return Err(PathError::Loop { index: link_index });
                    }
                    if expansions >= max_expansions {
                        return Err(PathError::Limit { index: link_index });
                    }
                    expansions += 1;
                    active.push(link_index);

                    stack.pop();
                    let target = link.target.as_str();
                    let target_absolute = target.starts_with('/');
                    if target_absolute {
                        stack.clear();
                    }
                    frames.push(Frame {
                        raw: target,
                        offset_base: 0,
                        parts: components(target),
                        next: 0,
                        active_link: Some(link_index),
                    });
                }
            }
        }
    }

    Ok(render(&stack, true, intent))
}
