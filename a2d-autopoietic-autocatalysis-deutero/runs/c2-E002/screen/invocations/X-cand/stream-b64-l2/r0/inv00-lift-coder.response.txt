#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut i = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && i < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[i];
                }
                self.tail_len += 1;
                i += 1;
            }

            if self.tail_len == 3 {
                encode_triplet(
                    self.tail[0],
                    self.tail[1],
                    input[i - 1],
                    &mut output,
                );
                self.tail_len = 0;
            } else {
                return Ok(output);
            }
        }

        while i + 3 <= input.len() {
            encode_triplet(input[i], input[i + 1], input[i + 2], &mut output);
            i += 3;
        }

        let remaining = input.len() - i;
        if remaining > 0 {
            self.tail[0] = input[i];
            self.tail_len = 1;
        }
        if remaining > 1 {
            self.tail[1] = input[i + 1];
            self.tail_len = 2;
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            1 => encode_one(self.tail[0], &mut output),
            2 => encode_two(self.tail[0], self.tail[1], &mut output),
            _ => {}
        }
        self.tail_len = 0;
        Ok(output)
    }
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    group_len: usize,
    pending_second_padding: Option<usize>,
    terminal_padding: bool,
    offset: usize,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            group_len: 0,
            pending_second_padding: None,
            terminal_padding: false,
            offset: 0,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let current_offset = self.offset;
            self.offset += 1;

            let result = self.consume(byte, current_offset, &mut output);
            if let Err(error) = result {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            let error = Error::BadPadding { offset };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        if self.group_len != 0 {
            let error = Error::Truncated {
                offset: self.offsets[0],
            };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        self.finished = true;
        Ok(Vec::new())
    }

    fn consume(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.pending_second_padding.is_some() {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }

            let first = self.group[0];
            let second = self.group[1];
            output.push((decode_digit(first) << 2) | (decode_digit(second) >> 4));
            self.group_len = 0;
            self.pending_second_padding = None;
            self.terminal_padding = true;
            return Ok(());
        }

        if byte == b'=' {
            match self.group_len {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    self.group[3] = b'=';
                    self.offsets[3] = offset;
                    self.pending_second_padding = Some(self.offsets[2]);
                    return Ok(());
                }
                3 => {
                    let first = decode_digit(self.group[0]);
                    let second = decode_digit(self.group[1]);
                    let third = decode_digit(self.group[2]);
                    output.push((first << 2) | (second >> 4));
                    output.push((second << 4) | (third >> 2));
                    self.group_len = 0;
                    self.terminal_padding = true;
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        let Some(_) = alphabet_index(byte) else {
            return Err(Error::InvalidByte { offset, byte });
        };

        self.group[self.group_len] = byte;
        self.offsets[self.group_len] = offset;
        self.group_len += 1;

        if self.group_len == 4 {
            let a = decode_digit(self.group[0]);
            let b = decode_digit(self.group[1]);
            let c = decode_digit(self.group[2]);
            let d = decode_digit(self.group[3]);
            output.push((a << 2) | (b >> 4));
            output.push((b << 4) | (c >> 2));
            output.push((c << 6) | d);
            self.group_len = 0;
        }

        Ok(())
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("new encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

fn encode_triplet(a: u8, b: u8, c: u8, output: &mut String) {
    output.push(alphabet((a >> 2) as usize));
    output.push(alphabet((((a & 0x03) << 4) | (b >> 4)) as usize));
    output.push(alphabet((((b & 0x0f) << 2) | (c >> 6)) as usize));
    output.push(alphabet((c & 0x3f) as usize));
}

fn encode_one(a: u8, output: &mut String) {
    output.push(alphabet((a >> 2) as usize));
    output.push(alphabet(((a & 0x03) << 4) as usize));
    output.push('=');
    output.push('=');
}

fn encode_two(a: u8, b: u8, output: &mut String) {
    output.push(alphabet((a >> 2) as usize));
    output.push(alphabet((((a & 0x03) << 4) | (b >> 4)) as usize));
    output.push(alphabet(((b & 0x0f) << 2) as usize));
    output.push('=');
}

fn alphabet(index: usize) -> char {
    const ALPHABET: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    ALPHABET[index] as char
}

fn alphabet_index(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn decode_digit(byte: u8) -> u8 {
    alphabet_index(byte).expect("only alphabet bytes are decoded")
}
