            tail_len: 0,
            finished: false,
        }
    }

        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut i = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && i < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[i];
                }
                self.tail_len += 1;
                i += 1;
            }

                encode_triplet(
                    self.tail[0],
                    self.tail[1],
                    input[i - 1],
                    &mut output,
                );
                self.tail_len = 0;
            } else {
                return Ok(output);
            }
        }

        while i + 3 <= input.len() {
            encode_triplet(input[i], input[i + 1], input[i + 2], &mut output);
            i += 3;
        }

        let remaining = input.len() - i;
        if remaining > 0 {
            self.tail[0] = input[i];
            self.tail_len = 1;
        }
        if remaining > 1 {
            self.tail[1] = input[i + 1];
            self.tail_len = 2;
        }

        Ok(output)
    }

            return Err(Error::AlreadyFinished);

            return Err(Error::AlreadyFinished);

                offset: self.offsets[0],

            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        self.finished = true;
        Ok(Vec::new())
    }

        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,

        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

                return Err(Error::BadPadding { offset });

            let second = self.group[1];
            output.push((decode_digit(first) << 2) | (decode_digit(second) >> 4));
            self.group_len = 0;
            self.pending_second_padding = None;
            self.terminal_padding = true;
            return Ok(());

                    self.terminal_padding = true;
                    return Ok(());
                }
                _ => unreachable!(),
            }

        let Some(_) = alphabet_index(byte) else {
            return Err(Error::InvalidByte { offset, byte });

        self.offsets[self.group_len] = offset;

            output.push((b << 4) | (c >> 2));
            output.push((c << 6) | d);
            self.group_len = 0;
        }

        Ok(())
    }

pub struct Decoder {

    terminal_padding: bool,
    offset: usize,

    sticky_error: Option<Error>,
}

    pub fn new() -> Self {
        Self {
            group: [Symbol { value: 0, offset: 0 }; 4],
            group_len: 0,
            pending_second_padding: None,
            terminal_padding: false,
            offset: 0,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let current_offset = self.offset;
            self.offset += 1;

            if self.terminal_padding {
                return self.fail(Error::BadPadding {
                    offset: current_offset,
                });
            }

            if self.pending_second_padding.is_some() {
                if byte != b'=' {
                    return self.fail(Error::BadPadding {
                        offset: current_offset,
                    });

                    value: 0,
                    offset: current_offset,
                };
                self.group_len = 4;
                self.pending_second_padding = None;
                decode_group(&self.group, &mut output);
                self.terminal_padding = true;
                self.group_len = 0;
                continue;
            }

            if byte == b'=' {
                match self.group_len {
                    0 | 1 => {
                        return self.fail(Error::BadPadding {
                            offset: current_offset,
                        });
                    }
                    2 => {
                        self.group[2] = Symbol {
                            value: 0,
                            offset: current_offset,
                        };
                        self.group_len = 3;
                        self.pending_second_padding = Some(current_offset);
                    }
                    3 => {
                        self.group[3] = Symbol {
                            value: 0,
                            offset: current_offset,
                        };
                        self.group_len = 4;
                        decode_group(&self.group, &mut output);
                        self.terminal_padding = true;
                        self.group_len = 0;
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            let value = match digit(byte) {
                Some(value) => value,
                None => {
                    return self.fail(Error::InvalidByte {
                        offset: current_offset,
                        byte,
                    });
                }
            };

            self.group[self.group_len] = Symbol {
                value,
                offset: current_offset,
            };
            self.group_len += 1;
            if self.group_len == 4 {
                decode_group(&self.group, &mut output);
                self.group_len = 0;
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {

        }
        if self.group_len != 0 {
            return self.fail(Error::Truncated {
                offset: self.group[0].offset,
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }

    fn fail(&mut self, error: Error) -> Result<Vec<u8>, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }

fn decode_group(group: &[Symbol; 4], output: &mut Vec<u8>) {
    let a = group[0].value;
    let b = group[1].value;
    output.push((a << 2) | (b >> 4));

    if group[2].value != 0 || group[3].value != 0 {
        output.push((b << 4) | (group[2].value >> 2));
        if group[3].value != 0 {
            output.push((group[2].value << 6) | group[3].value);
        }
    } else if group[2].offset != 0 || group[3].offset != 0 {
        // Padding is represented by zero values; offsets distinguish it from digits.
        if group[2].offset != 0 && group[3].offset != 0 {
            // xx== emits only the first byte.
        } else {
            // xxx= emits the second byte, including when the third digit is A.
            output.push(b << 4);
        }
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut out = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        out.push_str(std::str::from_utf8(&encode_group(chunk[0], chunk[1], chunk[2])).unwrap());
    }

    match chunks.remainder() {
        [] => {}
        [a] => {
            out.push(ALPHABET[(a >> 2) as usize] as char);
            out.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
            out.push_str("==");
        }
        [a, b] => {
            out.push(ALPHABET[(a >> 2) as usize] as char);
            out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            out.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
            out.push('=');
        }
        _ => unreachable!(),
    }

    out
}

pub struct Encoder {
    pending: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            pending: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut combined = std::mem::take(&mut self.pending);
        combined.extend_from_slice(input);
        let complete_len = combined.len() / 3 * 3;
        let output = encode(&combined[..complete_len]);
        self.pending.extend_from_slice(&combined[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        Ok(encode(&self.pending))
    }
}

pub struct Decoder {
    group: Vec<Symbol>,
    offset: usize,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: Vec::with_capacity(4),
            offset: 0,
            terminal_padding: false,
            pending_second_padding: None,
            finished: false,
            sticky_error: None,
        }
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    fn emit_group(&mut self, output: &mut Vec<u8>) {
        let a = digit(self.group[0].byte).unwrap();
        let b = digit(self.group[1].byte).unwrap();
        let c = if self.group[2].byte == b'=' {
            0
        } else {
            digit(self.group[2].byte).unwrap()
        };
        let d = if self.group[3].byte == b'=' {
            0
        } else {
            digit(self.group[3].byte).unwrap()
        };

        output.push((a << 2) | (b >> 4));
        if self.group[2].byte != b'=' {
            output.push((b << 4) | (c >> 2));
        }
        if self.group[3].byte != b'=' {
            output.push((c << 6) | d);
        }

        if self.group[2].byte == b'=' || self.group[3].byte == b'=' {
            self.terminal_padding = true;
        }
        self.group.clear();
        self.pending_second_padding = None;
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let offset = self.offset;
            self.offset += 1;

            if self.terminal_padding {
                return Err(self.fail(Error::BadPadding { offset }));
            }

            if let Some(padding_offset) = self.pending_second_padding {
                if byte != b'=' {
                    return Err(self.fail(Error::BadPadding { offset }));
                }
                self.group.push(Symbol { byte, offset });
                let _ = padding_offset;
                self.emit_group(&mut output);
                continue;
            }

            let position = self.group.len();
            if byte == b'=' {
                match position {
                    0 | 1 => return Err(self.fail(Error::BadPadding { offset })),
                    2 => {
                        self.group.push(Symbol { byte, offset });
                        self.pending_second_padding = Some(offset);
                    }
                    3 => {
                        self.group.push(Symbol { byte, offset });
                        self.emit_group(&mut output);
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            if digit(byte).is_none() {
                return Err(self.fail(Error::InvalidByte { offset, byte }));
            }

            self.group.push(Symbol { byte, offset });
            if self.group.len() == 4 {
                self.emit_group(&mut output);
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            let error = self.fail(Error::BadPadding { offset });
            return Err(error);
        }
        if let Some(first) = self.group.first() {
            let error = self.fail(Error::Truncated {
                offset: first.offset,
            });
            return Err(error);
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn encode_group(bytes: &[u8]) -> String {
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    let mut out = String::with_capacity(4);
    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);

    if bytes.len() > 1 {
        out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    } else {
        out.push('=');
    }

    if bytes.len() > 2 {
        out.push(ALPHABET[(c & 0x3f) as usize] as char);
    } else {
        out.push('=');
    }

    out
}

pub fn encode(input: &[u8]) -> String {
    let mut out = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);
    for chunk in &mut chunks {
        out.push_str(&encode_group(chunk));
    }
    if !chunks.remainder().is_empty() {
        out.push_str(&encode_group(chunks.remainder()));
    }
    out
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

pub struct Encoder {
    tail: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut data = Vec::with_capacity(self.tail.len() + input.len());
        data.extend_from_slice(&self.tail);
        data.extend_from_slice(input);

        let complete_len = data.len() / 3 * 3;
        let mut out = String::with_capacity(complete_len / 3 * 4);
        for chunk in data[..complete_len].chunks_exact(3) {
            out.push_str(&encode_group(chunk));
        }
        self.tail.clear();
        self.tail.extend_from_slice(&data[complete_len..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let out = if self.tail.is_empty() {
            String::new()
        } else {
            encode_group(&self.tail)
        };
        self.tail.clear();
        Ok(out)
    }
}

pub struct Decoder {

    terminal_padding: bool,
    offset: usize,
    finished: bool,
    error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: Vec::with_capacity(4),
            pending_second_padding: None,
            terminal_padding: false,
            offset: 0,
            finished: false,
            error: None,
        }
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.error = Some(error.clone());
        Err(error)
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();

        for &byte in input {
            let offset = self.offset;
            self.offset += 1;

            if self.terminal_padding {
                return self.fail(Error::BadPadding { offset });
            }

            if let Some(padding_offset) = self.pending_second_padding {
                if byte != b'=' {
                    return self.fail(Error::BadPadding { offset });
                }
                let first = digit(self.group[0].byte).unwrap();
                let second = digit(self.group[1].byte).unwrap();
                out.push((first << 2) | (second >> 4));
                self.group.clear();
                self.pending_second_padding = None;
                self.terminal_padding = true;
                let _ = padding_offset;
                continue;
            }

            let position = self.group.len();
            if byte == b'=' {
                match position {
                    0 | 1 => return self.fail(Error::BadPadding { offset }),
                    2 => {
                        self.group.push(Symbol { byte, offset });
                        self.pending_second_padding = Some(offset);
                    }
                    3 => {
                        let a = digit(self.group[0].byte).unwrap();
                        let b = digit(self.group[1].byte).unwrap();
                        let c = digit(self.group[2].byte).unwrap();
                        out.push((a << 2) | (b >> 4));
                        out.push((b << 4) | (c >> 2));
                        self.group.clear();
                        self.terminal_padding = true;
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            let Some(value) = digit(byte) else {
                return self.fail(Error::InvalidByte { offset, byte });
            };

            self.group.push(Symbol { byte, offset });
            if self.group.len() == 4 {
                let a = digit(self.group[0].byte).unwrap();
                let b = digit(self.group[1].byte).unwrap();
                let c = digit(self.group[2].byte).unwrap();
                let d = digit(self.group[3].byte).unwrap();
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
                out.push((c << 6) | d);
                self.group.clear();
            }
            let _ = value;
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            return self.fail(Error::BadPadding { offset });
        }
        if let Some(first) = self.group.first() {
            return self.fail(Error::Truncated {
                offset: first.offset,
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }
