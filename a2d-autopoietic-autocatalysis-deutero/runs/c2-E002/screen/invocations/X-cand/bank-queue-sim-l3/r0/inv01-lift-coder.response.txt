```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId {
        index: usize,
    },
    DuplicateId {
        index: usize,
        id: u32,
    },
    BadPriority {
        index: usize,
        value: u8,
    },
    TimeOverflow {
        customer: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct Waiting {
    index: usize,
    paused_at: Option<u64>,
    remaining: u64,
}

struct Running {
    index: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn wait_cmp(a: &Waiting, b: &Waiting, customers: &[Customer]) -> std::cmp::Ordering {
    let ca = &customers[a.index];
    let cb = &customers[b.index];

    ca.priority
        .cmp(&cb.priority)
        .then_with(|| match (a.paused_at, b.paused_at) {
            (Some(ta), Some(tb)) => ta.cmp(&tb).then_with(|| ca.id.cmp(&cb.id)),
            (Some(_), None) => std::cmp::Ordering::Less,
            (None, Some(_)) => std::cmp::Ordering::Greater,
            (None, None) => ca
                .arrival
                .cmp(&cb.arrival)
                .then_with(|| cb.id.cmp(&ca.id)),
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::cmp::Ordering;

    let mut seen = std::collections::HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrival_order: Vec<usize> = (0..customers.len()).collect();
    arrival_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut completed = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut started = vec![false; customers.len()];

    while completed < customers.len() {
        let next_arrival_time = arrival_order
            .get(next_arrival)
            .map(|&i| customers[i].arrival);
        let next_finish = running.as_ref().map(|r| r.finish);
        let next_work = running.as_ref().map(|r| r.work_at);

        let mut next_time: Option<u64> = None;
        for t in [next_arrival_time, next_finish, next_work].into_iter().flatten() {
            if next_time.map_or(true, |cur| t < cur) {
                next_time = Some(t);
            }
        }
        let time = next_time.expect("unfinished simulation must have a pending event");

        // A scheduled finish is processed before arrivals at the same tick.
        if running.as_ref().is_some_and(|r| r.finish == time) {
            let r = running.take().unwrap();
            trace.push(Event {
                time,
                customer: customers[r.index].id,
                kind: Kind::Finish,
            });
            completed += 1;
        }

        // Process the entire arrival cohort before considering dispatch or preemption.
        while next_arrival < arrival_order.len()
            && customers[arrival_order[next_arrival]].arrival == time
        {
            let index = arrival_order[next_arrival];
            trace.push(Event {
                time,
                customer: customers[index].id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                index,
                paused_at: None,
                remaining: customers[index].service,
            });
            next_arrival += 1;
        }

        if let Some(r) = running.as_ref() {
            if time >= r.work_at {
                if let Some(best) = waiting
                    .iter()
                    .min_by(|a, b| wait_cmp(a, b, customers))
                {
                    if customers[best.index].priority < customers[r.index].priority {
                        let best_id = customers[best.index].id;
                        let r = running.take().unwrap();
                        let remaining = r.finish - time;
                        trace.push(Event {
                            time,
                            customer: customers[r.index].id,
                            kind: Kind::Preempt {
                                remaining,
                                by: best_id,
                            },
                        });
                        waiting.push(Waiting {
                            index: r.index,
                            paused_at: Some(time),
                            remaining,
                        });
                    }
                }
            }
        }

        while running.is_none() && !waiting.is_empty() {
            let best_pos = (0..waiting.len())
                .min_by(|&a, &b| wait_cmp(&waiting[a], &waiting[b], customers))
                .unwrap();
            let selected = waiting.remove(best_pos);
            let customer = &customers[selected.index];
            let dispatch_id = customer.id;

            let cost = if last_dispatched.is_some_and(|id| id != dispatch_id) {
                switch_cost
            } else {
                0
            };
            let work_at = time.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: dispatch_id,
                time,
            })?;
            let finish = work_at
                .checked_add(selected.remaining)
                .ok_or(Error::TimeOverflow {
                    customer: dispatch_id,
                    time,
                })?;

            let kind = if started[selected.index] {
                Kind::Resume {
                    remaining: selected.remaining,
                    work_at,
                    finish,
                }
            } else {
                started[selected.index] = true;
                Kind::Start {
                    remaining: selected.remaining,
                    work_at,
                    finish,
                }
            };

            trace.push(Event {
                time,
                customer: dispatch_id,
                kind,
            });
            last_dispatched = Some(dispatch_id);

            if finish == time {
                trace.push(Event {
                    time,
                    customer: dispatch_id,
                    kind: Kind::Finish,
                });
                completed += 1;
            } else {
                running = Some(Running {
                    index: selected.index,
                    remaining: selected.remaining,
                    work_at,
                    finish,
                });
            }
        }
    }

    Ok(trace)
}
```
