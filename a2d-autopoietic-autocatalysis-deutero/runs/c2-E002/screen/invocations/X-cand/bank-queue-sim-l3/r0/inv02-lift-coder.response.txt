#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    id: u32,
    arrival: u64,
    service: u64,
    priority: u8,
    started: bool,
    pause_time: Option<u64>,
}

#[derive(Clone)]
struct Running {
    job_index: usize,
    work_at: u64,
    finish: u64,
}

fn waiter_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    a.priority
        .cmp(&b.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(at), Some(bt)) => at.cmp(&bt).then_with(|| a.id.cmp(&b.id)),
            (Some(_), None) => std::cmp::Ordering::Less,
            (None, Some(_)) => std::cmp::Ordering::Greater,
            (None, None) => a
                .arrival
                .cmp(&b.arrival)
                .then_with(|| b.id.cmp(&a.id)),
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut jobs: Vec<Job> = customers
        .iter()
        .map(|c| Job {
            id: c.id,
            arrival: c.arrival,
            service: c.service,
            priority: c.priority,
            started: false,
            pause_time: None,
        })
        .collect();

    let mut arrival_order: Vec<usize> = (0..jobs.len()).collect();
    arrival_order.sort_by(|&a, &b| {
        jobs[a]
            .arrival
            .cmp(&jobs[b].arrival)
            .then_with(|| jobs[a].priority.cmp(&jobs[b].priority))
            .then_with(|| jobs[b].id.cmp(&jobs[a].id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut running: Option<Running> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut next_arrival = 0usize;
    let mut finished_count = 0usize;
    let mut now = 0u64;

    while finished_count < jobs.len() {
        if let Some(active) = running.as_ref() {
            if active.finish == now {
                let idx = active.job_index;
                events.push(Event {
                    time: now,
                    customer: jobs[idx].id,
                    kind: Kind::Finish,
                });
                running = None;
                finished_count += 1;
            }
        }

        while next_arrival < arrival_order.len()
            && jobs[arrival_order[next_arrival]].arrival == now
        {
            let idx = arrival_order[next_arrival];
            events.push(Event {
                time: now,
                customer: jobs[idx].id,
                kind: Kind::Arrive,
            });
            waiting.push(idx);
            next_arrival += 1;
        }

        if let Some(active) = running.as_ref() {
            if now >= active.work_at {
                if let Some((best_pos, &best_idx)) = waiting
                    .iter()
                    .enumerate()
                    .min_by(|(_, &a), (_, &b)| waiter_cmp(&jobs[a], &jobs[b]))
                {
                    let active_idx = active.job_index;
                    if jobs[best_idx].priority < jobs[active_idx].priority {
                        let by = jobs[best_idx].id;
                        let remaining = active.finish - now;
                        events.push(Event {
                            time: now,
                            customer: jobs[active_idx].id,
                            kind: Kind::Preempt { remaining, by },
                        });
                        jobs[active_idx].pause_time = Some(now);
                        waiting.push(active_idx);
                        let _ = best_pos;
                        running = None;
                    }
                }
            }
        }

        if running.is_none() && !waiting.is_empty() {
            let best_pos = (0..waiting.len())
                .min_by(|&a, &b| waiter_cmp(&jobs[waiting[a]], &jobs[waiting[b]]))
                .unwrap();
            let idx = waiting.remove(best_pos);
            let remaining = if jobs[idx].started {
                match events.iter().rev().find_map(|event| {
                    if event.customer == jobs[idx].id {
                        if let Kind::Preempt { remaining, .. } = event.kind {
                            return Some(remaining);
                        }
                    }
                    None
                }) {
                    Some(value) => value,
                    None => jobs[idx].service,
                }
            } else {
                jobs[idx].service
            };

            let cost = if last_dispatched == Some(jobs[idx].id) || last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: jobs[idx].id,
                time: now,
            })?;
            let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                customer: jobs[idx].id,
                time: now,
            })?;

            let kind = if jobs[idx].started {
                Kind::Resume {
                    remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining,
                    work_at,
                    finish,
                }
            };
            events.push(Event {
                time: now,
                customer: jobs[idx].id,
                kind,
            });
            jobs[idx].started = true;
            jobs[idx].pause_time = None;
            last_dispatched = Some(jobs[idx].id);
            running = Some(Running {
                job_index: idx,
                work_at,
                finish,
            });

            if finish == now {
                events.push(Event {
                    time: now,
                    customer: jobs[idx].id,
                    kind: Kind::Finish,
                });
                running = None;
                finished_count += 1;
                continue;
            }
        }

        if finished_count == jobs.len() {
            break;
        }

        let mut next_tick: Option<u64> = None;
        if next_arrival < arrival_order.len() {
            next_tick = Some(jobs[arrival_order[next_arrival]].arrival);
        }
        if let Some(active) = running.as_ref() {
            for t in [active.finish, active.work_at] {
                if t > now && next_tick.map_or(true, |n| t < n) {
                    next_tick = Some(t);
                }
            }
        }

        match next_tick {
            Some(t) => now = t,
            None => break,
        }
    }

    Ok(events)
}
