```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobRecord {
    job: Job,
    dependencies: Vec<usize>,
    state: State,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<JobRecord>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        use std::collections::{HashMap, HashSet};

        let mut ids = HashMap::new();
        for (index, job) in jobs.iter().enumerate() {
            if job.id.is_empty()
                || job.id.len() > 16
                || !job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
            {
                return Err(Error::BadId(index));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(index));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(index));
            }
            if ids.insert(job.id.as_str(), index).is_some() {
                return Err(Error::Duplicate(index));
            }
        }

        let mut resolved_dependencies = Vec::with_capacity(jobs.len());
        for (job_index, job) in jobs.iter().enumerate() {
            let mut seen = HashSet::new();
            let mut dependencies = Vec::with_capacity(job.dependencies.len());
            for (dependency_index, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency.as_str()) {
                    return Err(Error::DuplicateDependency(job_index, dependency_index));
                }
                let Some(&target) = ids.get(dependency.as_str()) else {
                    return Err(Error::MissingDependency(job_index, dependency_index));
                };
                dependencies.push(target);
            }
            resolved_dependencies.push(dependencies);
        }

        fn has_cycle(
            node: usize,
            dependencies: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[node] == 1 {
                return true;
            }
            if marks[node] == 2 {
                return false;
            }
            marks[node] = 1;
            for &dependency in &dependencies[node] {
                if has_cycle(dependency, dependencies, marks) {
                    return true;
                }
            }
            marks[node] = 2;
            false
        }

        let mut marks = vec![0u8; jobs.len()];
        for node in 0..jobs.len() {
            if has_cycle(node, &resolved_dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let records = jobs
            .into_iter()
            .zip(resolved_dependencies)
            .map(|(job, dependencies)| JobRecord {
                job,
                dependencies,
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs: records,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.cycle_started
            && self.jobs.iter().any(|record| {
                matches!(record.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for record in &mut self.jobs {
            record.state = State::Waiting { ready_at: now };
            record.attempts = 0;
        }
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for (index, record) in self.jobs.iter().enumerate() {
            let State::Waiting { ready_at } = record.state else {
                continue;
            };
            if ready_at > now
                || !record
                    .dependencies
                    .iter()
                    .all(|&dependency| matches!(self.jobs[dependency].state, State::Succeeded))
            {
                continue;
            }

            let age_levels = if self.aging_ms == 0 {
                0
            } else {
                (now - ready_at) / self.aging_ms
            };
            let effective = if self.aging_ms == 0 {
                record.job.priority
            } else {
                (u64::from(record.job.priority) + age_levels).min(9) as u8
            };

            if best.is_none_or(|(_, best_priority)| effective > best_priority) {
                best = Some((index, effective));
            }
        }

        self.last_time = now;
        let Some((index, _)) = best else {
            return Ok(None);
        };

        let record = &mut self.jobs[index];
        record.attempts += 1;
        record.state = State::Running {
            attempt: record.attempts,
        };
        Ok(Some(Ticket {
            id: record.job.id.clone(),
            attempt: record.attempts,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(index) = self.jobs.iter().position(|record| record.job.id == id) else {
            return Err(Error::Unknown);
        };
        if !matches!(self.jobs[index].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        self.last_time = now;
        if success {
            self.jobs[index].state = State::Succeeded;
        } else if self.jobs[index].attempts == self.jobs[index].job.max_attempts {
            self.jobs[index].state = State::Failed;
        } else {
            let attempt = self.jobs[index].attempts;
            let mut delay = self.retry_base;
            for _ in 1..attempt {
                delay = delay.saturating_mul(2).min(self.retry_cap);
                if delay == self.retry_cap {
                    break;
                }
            }
            self.jobs[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }

        if matches!(self.jobs[index].state, State::Failed) {
            loop {
                let mut changed = false;
                for record in &mut self.jobs {
                    if !matches!(record.state, State::Waiting { .. }) {
                        continue;
                    }
                    if record.dependencies.iter().any(|&dependency| {
                        matches!(
                            self.jobs[dependency].state,
                            State::Failed | State::Blocked
                        )
                    }) {
                        record.state = State::Blocked;
                        changed = true;
                    }
                }
                if !changed {
                    break;
                }
            }
        }

        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        self.jobs
            .iter()
            .find(|record| record.job.id == id)
            .map(|record| record.state.clone())
            .ok_or(Error::Unknown)
    }
}
```
