```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobData {
    id: String,
    priority: u8,
    dependencies: Vec<usize>,
    max_attempts: u8,
}

struct Runtime {
    state: State,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<JobData>,
    states: Vec<Runtime>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    has_cycle: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut seen = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if job.id.is_empty()
                || job.id.len() > 16
                || !job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
            {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if job.max_attempts == 0 || job.max_attempts > 32 {
                return Err(Error::BadAttempts(i));
            }
            if seen.insert(job.id.as_str(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut resolved = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut deps = Vec::with_capacity(job.dependencies.len());
            let mut dep_seen = std::collections::HashSet::new();
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !dep_seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match seen.get(dep.as_str()) {
                    Some(&idx) => deps.push(idx),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            resolved.push(deps);
        }

        fn visit(
            node: usize,
            deps: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[node] == 1 {
                return false;
            }
            if marks[node] == 2 {
                return true;
            }
            marks[node] = 1;
            for &dep in &deps[node] {
                if !visit(dep, deps, marks) {
                    return false;
                }
            }
            marks[node] = 2;
            true
        }

        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if !visit(i, &resolved, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let data = jobs
            .into_iter()
            .zip(resolved)
            .map(|(job, dependencies)| JobData {
                id: job.id,
                priority: job.priority,
                dependencies,
                max_attempts: job.max_attempts,
            })
            .collect::<Vec<_>>();
        let states = data
            .iter()
            .map(|_| Runtime {
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs: data,
            states,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            has_cycle: false,
        })
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn index_of(&self, id: &str) -> Option<usize> {
        self.jobs.iter().position(|job| job.id == id)
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;
        if self.has_cycle
            && self.states.iter().any(|runtime| {
                matches!(runtime.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for runtime in &mut self.states {
            runtime.state = State::Waiting { ready_at: now };
            runtime.attempts = 0;
        }
        self.has_cycle = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for i in 0..self.jobs.len() {
            let ready_at = match self.states[i].state {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            if !self.jobs[i]
                .dependencies
                .iter()
                .all(|&dep| matches!(self.states[dep].state, State::Succeeded))
            {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                self.jobs[i].priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                (self.jobs[i].priority as u64 + age).min(9) as u8
            };
            if best.map_or(true, |(_, p)| effective > p) {
                best = Some((i, effective));
            }
        }

        self.last_time = now;
        let Some((i, _)) = best else {
            return Ok(None);
        };

        self.states[i].attempts += 1;
        let attempt = self.states[i].attempts;
        self.states[i].state = State::Running { attempt };
        Ok(Some(Ticket {
            id: self.jobs[i].id.clone(),
            attempt,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }
        let i = self.index_of(id).ok_or(Error::Unknown)?;
        let attempt = match self.states[i].state {
            State::Running { attempt } => attempt,
            _ => return Err(Error::NotRunning),
        };

        if success {
            self.states[i].state = State::Succeeded;
        } else if attempt == self.jobs[i].max_attempts {
            self.states[i].state = State::Failed;
            loop {
                let mut changed = false;
                for j in 0..self.jobs.len() {
                    if matches!(self.states[j].state, State::Waiting { .. })
                        && self.jobs[j].dependencies.iter().any(|&dep| {
                            matches!(
                                self.states[dep].state,
                                State::Failed | State::Blocked
                            )
                        })
                    {
                        self.states[j].state = State::Blocked;
                        changed = true;
                    }
                }
                if !changed {
                    break;
                }
            }
        } else {
            let shift = attempt.saturating_sub(1) as u32;
            let delay = self
                .retry_base
                .checked_shl(shift)
                .unwrap_or(u64::MAX)
                .min(self.retry_cap);
            self.states[i].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }

        self.last_time = now;
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }
        let i = self.index_of(id).ok_or(Error::Unknown)?;
        Ok(self.states[i].state.clone())
    }
}
```
