#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobData {
    id: String,
    priority: u8,
    dependencies: Vec<usize>,
    max_attempts: u8,
    attempts: u8,
    state: State,
}

pub struct Runner {
    jobs: Vec<JobData>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut seen = std::collections::HashMap::new();

        for (i, job) in jobs.iter().enumerate() {
            let id = job.id.as_bytes();
            if id.is_empty()
                || id.len() > 16
                || !id.iter().all(|b| {
                    b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-'
                })
            {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if job.max_attempts == 0 || job.max_attempts > 32 {
                return Err(Error::BadAttempts(i));
            }
            if seen.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            seen.insert(job.id.clone(), i);
        }

        let mut resolved = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut dependencies = Vec::with_capacity(job.dependencies.len());
            let mut dep_seen = std::collections::HashSet::new();
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !dep_seen.insert(dep) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match seen.get(dep) {
                    Some(&index) => dependencies.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            resolved.push(dependencies);
        }

        let mut color = vec![0u8; jobs.len()];
        fn visit(i: usize, graph: &[Vec<usize>], color: &mut [u8]) -> bool {
            if color[i] == 1 {
                return true;
            }
            if color[i] == 2 {
                return false;
            }
            color[i] = 1;
            for &dep in &graph[i] {
                if visit(dep, graph, color) {
                    return true;
                }
            }
            color[i] = 2;
            false
        }

        for i in 0..jobs.len() {
            if visit(i, &resolved, &mut color) {
                return Err(Error::Cycle);
            }
        }

        let data = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| JobData {
                id: job.id,
                priority: job.priority,
                dependencies: resolved[i].clone(),
                max_attempts: job.max_attempts,
                attempts: 0,
                state: State::Waiting { ready_at: 0 },
            })
            .collect();

        Ok(Self {
            jobs: data,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;
        if self.cycle_started
            && self.jobs.iter().any(|job| {
                matches!(job.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for job in &mut self.jobs {
            job.attempts = 0;
            job.state = State::Waiting { ready_at: now };
        }
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for (i, job) in self.jobs.iter().enumerate() {
            let ready_at = match job.state {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            if !job
                .dependencies
                .iter()
                .all(|&dep| matches!(self.jobs[dep].state, State::Succeeded))
            {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                job.priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                (job.priority as u64 + age).min(9) as u8
            };

            if best.is_none_or(|(best_i, best_priority)| {
                effective > best_priority || (effective == best_priority && i < best_i)
            }) {
                best = Some((i, effective));
            }
        }

        self.last_time = now;
        let Some((index, _)) = best else {
            return Ok(None);
        };

        let job = &mut self.jobs[index];
        job.attempts += 1;
        job.state = State::Running {
            attempt: job.attempts,
        };
        Ok(Some(Ticket {
            id: job.id.clone(),
            attempt: job.attempts,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let index = self
            .jobs
            .iter()
            .position(|job| job.id == id)
            .ok_or(Error::Unknown)?;

        if !matches!(self.jobs[index].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        self.last_time = now;
        if success {
            self.jobs[index].state = State::Succeeded;
            return Ok(());
        }

        if self.jobs[index].attempts == self.jobs[index].max_attempts {
            self.jobs[index].state = State::Failed;
            self.propagate_blocked();
        } else {
            let attempt = self.jobs[index].attempts;
            let delay = self.retry_delay(attempt);
            self.jobs[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }
        Ok(())
    }

    fn retry_delay(&self, attempt: u8) -> u64 {
        let mut delay = self.retry_base;
        for _ in 1..attempt {
            if delay >= self.retry_cap || delay > self.retry_cap / 2 {
                return self.retry_cap;
            }
            delay *= 2;
        }
        delay.min(self.retry_cap)
    }

    fn propagate_blocked(&mut self) {
        loop {
            let to_block: Vec<usize> = self
                .jobs
                .iter()
                .enumerate()
                .filter_map(|(i, job)| {
                    if !matches!(job.state, State::Waiting { .. }) {
                        return None;
                    }
                    if job.dependencies.iter().any(|&dep| {
                        matches!(self.jobs[dep].state, State::Failed | State::Blocked)
                    }) {
                        Some(i)
                    } else {
                        None
                    }
                })
                .collect();

            if to_block.is_empty() {
                break;
            }
            for index in to_block {
                self.jobs[index].state = State::Blocked;
            }
        }
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        self.jobs
            .iter()
            .find(|job| job.id == id)
            .map(|job| job.state.clone())
            .ok_or(Error::Unknown)
    }
}
