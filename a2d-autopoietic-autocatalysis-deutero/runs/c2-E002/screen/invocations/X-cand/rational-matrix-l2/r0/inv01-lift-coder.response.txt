use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let mut n = num;
        let mut d = den;
        if d < 0 {
            n = n.checked_neg().ok_or(RationalError::Overflow)?;
            d = d.checked_neg().ok_or(RationalError::Overflow)?;
        }

        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;

        let n = i64::try_from(n).map_err(|_| RationalError::Overflow)?;
        let d = i64::try_from(d).map_err(|_| RationalError::Overflow)?;
        if d <= 0 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n, den: d })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let d1 = self.den as i128;
        let d2 = rhs.den as i128;
        let g = gcd_u128(d1 as u128, d2 as u128) as i128;
        let left_scale = d2 / g;
        let right_scale = d1 / g;
        let n = a
            .checked_mul(left_scale)
            .and_then(|x| b.checked_mul(right_scale).and_then(|y| x.checked_add(y)))
            .ok_or(RationalError::Overflow)?;
        let d = d1.checked_mul(left_scale).ok_or(RationalError::Overflow)?;
        Self::from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let d1 = self.den as i128;
        let d2 = rhs.den as i128;
        let g = gcd_u128(d1 as u128, d2 as u128) as i128;
        let left_scale = d2 / g;
        let right_scale = d1 / g;
        let n = a
            .checked_mul(left_scale)
            .and_then(|x| b.checked_mul(right_scale).and_then(|y| x.checked_sub(y)))
            .ok_or(RationalError::Overflow)?;
        let d = d1.checked_mul(left_scale).ok_or(RationalError::Overflow)?;
        Self::from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let mut a = self.num as i128;
        let mut b = rhs.num as i128;
        let mut c = self.den as i128;
        let mut d = rhs.den as i128;

        let g1 = gcd_u128(a.unsigned_abs(), d as u128) as i128;
        let g2 = gcd_u128(b.unsigned_abs(), c as u128) as i128;
        a /= g1;
        d /= g1;
        b /= g2;
        c /= g2;

        let n = a.checked_mul(b).ok_or(RationalError::Overflow)?;
        let den = c.checked_mul(d).ok_or(RationalError::Overflow)?;
        Self::from_i128(n, den)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }

        let mut a = self.num as i128;
        let mut b = rhs.den as i128;
        let mut c = self.den as i128;
        let mut d = rhs.num as i128;

        let g1 = gcd_u128(a.unsigned_abs(), d.unsigned_abs()) as i128;
        let g2 = gcd_u128(b as u128, c as u128) as i128;
        a /= g1;
        d /= g1;
        b /= g2;
        c /= g2;

        let n = a.checked_mul(b).ok_or(RationalError::Overflow)?;
        let den = c.checked_mul(d).ok_or(RationalError::Overflow)?;
        Self::from_i128(n, den)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = self.num.checked_neg().ok_or(RationalError::Overflow)?;
        Ok(Self { num: n, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0usize;
        if bytes[i] == b'-' {
            i += 1;
        }

        let num_start = 0usize;
        let num_digits_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_digits_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let slash = if i < bytes.len() && bytes[i] == b'/' {
            let at = i;
            i += 1;
            let den_start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            Some((at, den_start, i))
        } else {
            None
        };

        if i < bytes.len() {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let num_end = slash.map_or(bytes.len(), |(at, _, _)| at);
        let num_token = &bytes[num_start..num_end];
        let neg = num_token.first() == Some(&b'-');
        let digits = if neg { &num_token[1..] } else { num_token };

        if digits.len() >= 2 && digits[0] == b'0' {
            return Err(parse_err(if neg { 2 } else { 1 }, ParseKind::NonCanonical));
        }
        if neg && digits == b"0" {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        let num_mag = parse_magnitude(digits);
        let num_limit = if neg { 1u128 << 63 } else { i64::MAX as u128 };
        let num_mag = match num_mag {
            Some(v) if v <= num_limit => v,
            _ => return Err(parse_err(0, ParseKind::OutOfRange)),
        };

        let num = if neg {
            if num_mag == 1u128 << 63 {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };

        let Some((slash_at, den_start, den_end)) = slash else {
            return Ok(Self { num, den: 1 });
        };

        let den_digits = &bytes[den_start..den_end];
        if den_digits.len() >= 2 && den_digits[0] == b'0' {
            return Err(parse_err(den_start + 1, ParseKind::NonCanonical));
        }

        let den_mag = match parse_magnitude(den_digits) {
            Some(v) if v <= i64::MAX as u128 => v,
            _ => return Err(parse_err(den_start, ParseKind::OutOfRange)),
        };

        if den_mag == 0 {
            return Err(parse_err(den_start, ParseKind::ZeroDenominator));
        }
        if den_mag == 1 {
            return Err(parse_err(slash_at, ParseKind::NonCanonical));
        }
        if num == 0 {
            return Err(parse_err(slash_at, ParseKind::NonCanonical));
        }
        if gcd_u128(num.unsigned_abs() as u128, den_mag) > 1 {
            return Err(parse_err(slash_at, ParseKind::NonCanonical));
        }

        Ok(Self { num, den: den_mag as i64 })
    }
}

fn parse_magnitude(digits: &[u8]) -> Option<u128> {
    let mut value = 0u128;
    for &b in digits {
        value = value.checked_mul(10)?.checked_add((b - b'0') as u128)?;
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|base| base.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data.get(index).copied().ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }

        let out_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(out_len);
        let zero = Rational { num: 0, den: 1 };

        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }
}
