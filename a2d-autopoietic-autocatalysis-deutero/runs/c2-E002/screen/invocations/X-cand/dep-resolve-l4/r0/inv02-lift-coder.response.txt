fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort_by_key(|c| constraint_key(*c));
    out.dedup();
    out
}

fn satisfies(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(e: &[Evidence]) -> Vec<Evidence> {
    let mut out = e.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn append_evidence(mut first: String, evidence: &[Evidence]) -> String {
    for e in evidence_sorted(evidence) {
        first.push_str("\n  C:");
        first.push_str(&constraint_text(e.constraint));
        first.push_str(" via ");
        first.push_str(&e.path.join(" -> "));
    }
    first
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => append_evidence(
            format!("missing|package={}", encode(package)),
            evidence,
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            append_evidence(
                format!("no-match|package={}|constraints={cs}", encode(package)),
                evidence,
            )
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            append_evidence(
                format!(
                    "conflict|package={}|selected={}|constraints={cs}",
                    encode(package),
                    version_text(*selected)
                ),
                evidence,
            )
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

struct Search<'a> {
    catalog: &'a [Package],
    by_name: BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    attempts: usize,
    budget: usize,
    best: Option<(usize, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

fn platform_name(p: Platform) -> &'static str {
    match p {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn platform_matches(p: Platform, t: Target) -> bool {
    matches!(p, Platform::Any)
        || matches!((p, t), (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows))
}

impl<'a> Search<'a> {
    fn active(&self, assigned: &BTreeMap<String, Version>) -> Active {
        let mut a = Active::default();
        for (i, r) in self.roots.iter().enumerate() {
            let mut fs = r.features.clone();
            fs.sort();
            fs.dedup();
            let tok = format!(
                "root[{}]:{}@{}+{{{}}}",
                i,
                r.package,
                constraint_text(r.constraint),
                fs.join(",")
            );
            a.origins
                .entry(r.package.clone())
                .or_default()
                .push(Origin {
                    constraint: r.constraint,
                    path: vec![tok.clone()],
                });
            a.requests
                .entry(r.package.clone())
                .or_default()
                .extend(r.features.iter().cloned());
            a.paths
                .entry(r.package.clone())
                .and_modify(|p| {
                    if (1, vec![tok.clone()]) < (p.len(), p.clone()) {
                        *p = vec![tok.clone()];
                    }
                })
                .or_insert(vec![tok]);
        }

        loop {
            let mut changed = false;
            let names: Vec<String> = assigned.keys().cloned().collect();
            for source in names {
                let Some(&version) = assigned.get(&source) else {
                    continue;
                };
                let Some(pkg) = self.by_name.get(&source) else {
                    continue;
                };
                let Some(rel) = pkg.releases.iter().find(|r| r.version == version) else {
                    continue;
                };

                let mut activated = a.requests.get(&source).cloned().unwrap_or_default();
                let mut queue: Vec<String> = activated.iter().cloned().collect();
                let mut idx = 0;
                while idx < queue.len() {
                    let f = queue[idx].clone();
                    idx += 1;
                    if let Some(decl) = rel.features.iter().find(|d| d.name == f) {
                        for en in &decl.enables {
                            if activated.insert(en.clone()) {
                                queue.push(en.clone());
                                changed = true;
                            }
                        }
                    }
                }
                if activated != a.requests.get(&source).cloned().unwrap_or_default() {
                    a.requests.insert(source.clone(), activated.clone());
                    changed = true;
                }

                let source_path = a.paths.get(&source).cloned().unwrap_or_default();
                for (j, dep) in rel.deps.iter().enumerate() {
                    if !platform_matches(dep.platform, self.target)
                        || dep
                            .gate
                            .as_ref()
                            .is_some_and(|g| !activated.contains(g))
                    {
                        continue;
                    }
                    let mut fs = dep.features.clone();
                    fs.sort();
                    fs.dedup();
                    let gate = dep.gate.as_deref().unwrap_or("*");
                    let tok = format!(
                        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        source,
                        version_text(version),
                        j,
                        dep.package,
                        constraint_text(dep.constraint),
                        fs.join(","),
                        gate,
                        platform_name(dep.platform)
                    );
                    let path = {
                        let mut p = source_path.clone();
                        p.push(tok);
                        p
                    };
                    let origins = a.origins.entry(dep.package.clone()).or_default();
                    if !origins.iter().any(|o| o.constraint == dep.constraint && o.path == path) {
                        origins.push(Origin {
                            constraint: dep.constraint,
                            path: path.clone(),
                        });
                        changed = true;
                    }
                    let entry = a.paths.entry(dep.package.clone()).or_insert_with(|| path.clone());
                    if (path.len(), &path) < (entry.len(), entry) {
                        *entry = path;
                        changed = true;
                    }
                    let req = a.requests.entry(dep.package.clone()).or_default();
                    for f in &dep.features {
                        if req.insert(f.clone()) {
                            changed = true;
                        }
                    }
                }
            }
            if !changed {
                break;
            }
        }
        a
    }

    fn evidence(active: &Active, package: &str) -> Vec<Evidence> {
        active
            .origins
            .get(package)
            .into_iter()
            .flatten()
            .map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            })
            .collect()
    }

    fn requirements(active: &Active, package: &str) -> Vec<Constraint> {
        canonical_constraints(
            &active
                .origins
                .get(package)
                .into_iter()
                .flatten()
                .map(|o| o.constraint)
                .collect::<Vec<_>>(),
        )
    }

    fn inspect(&mut self, assigned: &BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let active = self.active(assigned);
        let active_names: Vec<String> = active.origins.keys().cloned().collect();

        if let Some(p) = active_names.iter().find(|p| !self.by_name.contains_key(*p)) {
            return Err(ResolveError::Missing {
                package: p.clone(),
                evidence: Self::evidence(&active, p),
            });
        }

        for (p, version) in assigned {
            let Some(origins) = active.origins.get(p) else {
                continue;
            };
            if origins.iter().any(|o| !satisfies(*version, o.constraint)) {
                return Err(ResolveError::Conflict {
                    package: p.clone(),
                    selected: *version,
                    requirements: Self::requirements(&active, p),
                    evidence: Self::evidence(&active, p),
                });
            }
        }

        for p in &active_names {
            if assigned.contains_key(p) {
                continue;
            }
            let reqs = Self::requirements(&active, p);
            let candidates = self.by_name[*p]
                .releases
                .iter()
                .filter(|r| reqs.iter().all(|c| satisfies(r.version, *c)))
                .count();
            if candidates == 0 {
                return Err(ResolveError::NoMatch {
                    package: p.clone(),
                    requirements: reqs,
                    evidence: Self::evidence(&active, p),
                });
            }
        }

        for (p, version) in assigned {
            let Some(reqs) = active.requests.get(p) else {
                continue;
            };
            let rel = self.by_name[p]
                .releases
                .iter()
                .find(|r| r.version == *version)
                .expect("assigned release must exist");
            for f in reqs {
                if !rel.features.iter().any(|decl| decl.name == *f) {
                    return Err(ResolveError::MissingFeature {
                        package: p.clone(),
                        selected: *version,
                        feature: f.clone(),
                        path: active.paths.get(p).cloned().unwrap_or_default(),
                    });
                }
            }
        }

        let next = active_names
            .iter()
            .find(|p| !assigned.contains_key(*p))
            .cloned();
        if let Some(package) = next {
            let reqs = Self::requirements(&active, &package);
            let mut candidates: Vec<Version> = self.by_name[&package]
                .releases
                .iter()
                .map(|r| r.version)
                .filter(|v| reqs.iter().all(|c| satisfies(*v, *c)))
                .collect();
            candidates.sort_by(|a, b| b.cmp(a));
            if let Some(lock) = self.locks.get(&package) {
                if let Some(pos) = candidates.iter().position(|v| v == lock) {
                    candidates.remove(pos);
                    candidates.insert(0, *lock);
                }
            }

            let mut first = None;
            for version in candidates {
                if self.attempts == self.budget {
                    return Err(ResolveError::Limit {
                        attempts: self.budget,
                    });
                }
                self.attempts += 1;
                let mut child = assigned.clone();
                child.insert(package.clone(), version);
                match self.inspect(&child) {
                    Ok(()) => {}
                    Err(ResolveError::Limit { attempts }) => {
                        return Err(ResolveError::Limit { attempts })
                    }
                    Err(e) => {
                        if first.is_none() {
                            first = Some(e);
                        }
                        continue;
                    }
                }
            }
            if let Some(e) = first {
                return Err(e);
            }
            return Ok(());
        }

        let cost = self
            .locks
            .iter()
            .filter(|(p, v)| assigned.get(*p) != Some(*v))
            .count();
        let replace = match &self.best {
            None => true,
            Some((best_cost, best)) if cost < *best_cost => true,
            Some((best_cost, best)) if cost == *best_cost => {
                let names: BTreeSet<String> = self
                    .by_name
                    .keys()
                    .cloned()
                    .chain(assigned.keys().cloned())
                    .collect();
                let mut cmp = std::cmp::Ordering::Equal;
                for n in names {
                    cmp = match (assigned.get(&n), best.get(&n)) {
                        (None, None) => std::cmp::Ordering::Equal,
                        (None, Some(_)) => std::cmp::Ordering::Less,
                        (Some(_), None) => std::cmp::Ordering::Greater,
                        (Some(a), Some(b)) => a.cmp(b),
                    };
                    if cmp != std::cmp::Ordering::Equal {
                        break;
                    }
                }
                cmp == std::cmp::Ordering::Greater
            }
            _ => false,
        };
        if replace {
            self.best = Some((cost, assigned.clone()));
        }
        Ok(())
    }

    fn enumerate(&mut self, assigned: &BTreeMap<String, Version>) -> Result<(), ResolveError> {
        match self.inspect(assigned) {
            Ok(()) => Ok(()),
            Err(ResolveError::Limit { attempts }) => Err(ResolveError::Limit { attempts }),
            Err(e) => {
                if self.first_error.is_none() {
                    self.first_error = Some(e);
                }
                Ok(())
            }
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: r.package.clone(),
            });
        }
        for (f, name) in r.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{i}].feature[{f}]"),
                    name: name.clone(),
                });
            }
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: p.name.clone(),
            });
        }
        for (j, r) in p.releases.iter().enumerate() {
            for (k, d) in r.deps.iter().enumerate() {
                let site = format!("catalog[{i}].release[{j}].dep[{k}]");
                if !valid_name(&d.package) {
                    return Err(ResolveError::InvalidName {
                        site,
                        name: d.package.clone(),
                    });
                }
                if let Some(g) = &d.gate {
                    if !valid_name(g) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].dep[{k}].gate"),
                            name: g.clone(),
                        });
                    }
                }
                for (f, name) in d.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].dep[{k}].feature[{f}]"),
                            name: name.clone(),
                        });
                    }
                }
            }
            for (f, feat) in r.features.iter().enumerate() {
                if !valid_name(&feat.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].feature[{f}]"),
                        name: feat.name.clone(),
                    });
                }
                for (e, name) in feat.enables.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{i}].release[{j}].feature[{f}].enable[{e}]"
                            ),
                            name: name.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for p in catalog {
        if !seen_packages.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }
    for p in catalog {
        let mut versions = BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }
    for p in catalog {
        for r in &p.releases {
            let mut features = BTreeSet::new();
            for f in &r.features {
                if !features.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: r.version,
                        feature: f.name.clone(),
                    });
                }
            }
        }
    }

    for (i, l) in locks.iter().enumerate() {
        if !valid_name(&l.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: l.package.clone(),
            });
        }
    }
    let mut lock_map = BTreeMap::new();
    for l in locks {
        if lock_map.insert(l.package.clone(), l.version).is_some() {
            return Err(ResolveError::DuplicateLock {
                package: l.package.clone(),
            });
        }
    }

    let by_name: BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let mut search = Search {
        catalog,
        by_name,
        roots,
        locks: lock_map.clone(),
        target,
        attempts: 0,
        budget: max_attempts,
        best: None,
        first_error: None,
    };

    search.enumerate(&BTreeMap::new())?;
    let Some((_, chosen)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::Missing {
            package: String::new(),
            evidence: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let final_active = search.active(&chosen);
    let mut change_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    change_names.extend(chosen.keys().cloned());
    let mut changes = Vec::new();
    for package in change_names {
        let before = lock_map.get(&package).copied();
        let after = chosen.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .by_name
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active
                    .origins
                    .get(&package)
                    .into_iter()
                    .flatten()
                    .any(|o| !satisfies(old, o.constraint))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
