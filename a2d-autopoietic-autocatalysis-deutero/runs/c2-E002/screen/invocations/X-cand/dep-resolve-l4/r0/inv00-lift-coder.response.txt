fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut result = cs.to_vec();
    result.sort_by_key(|c| constraint_key(*c));
    result.dedup();
    result
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() || b.len() > 24 || !b[0].is_ascii_lowercase() || b[b.len() - 1] == b'-' {
        return false;
    }
    b.iter()
        .all(|x| x.is_ascii_lowercase() || x.is_ascii_digit() || *x == b'-')
}

fn invalid(site: impl Into<String>, name: &str) -> ResolveError {
    ResolveError::InvalidName {
        site: site.into(),
        name: name.to_string(),
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn append_evidence(line: &mut String, evidence: &[Evidence]) {
    for e in evidence_sorted(evidence) {
        line.push_str("\n  C:");
        line.push_str(&constraint_text(e.constraint));
        line.push_str(" via ");
        line.push_str(&e.path.join(" -> "));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut line = format!("missing|package={}", encode(package));
            append_evidence(&mut line, evidence);
            line
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut line = format!("no-match|package={}|constraints={}", encode(package), cs);
            append_evidence(&mut line, evidence);
            line
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut line = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                cs
            );
            append_evidence(&mut line, evidence);
            line
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone, Default)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    evidence: BTreeMap<String, Vec<Evidence>>,
    features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
    active_deps: BTreeMap<String, BTreeSet<usize>>,
}

fn target_name(target: Target) -> &'static str {
    match target {
        Target::Linux => "linux",
        Target::Mac => "mac",
        Target::Windows => "windows",
    }
}

fn platform_matches(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

fn feature_index(release: &Release) -> HashMap<&str, &Feature> {
    release
        .features
        .iter()
        .map(|f| (f.name.as_str(), f))
        .collect()
}

fn dep_token(
    source: &str,
    release: &Release,
    dep_index: usize,
    dep: &Dependency,
) -> String {
    let fs = {
        let mut v = dep.features.clone();
        v.sort();
        v.dedup();
        v.join(",")
    };
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(release.version),
        dep_index,
        dep.package,
        constraint_text(dep.constraint),
        fs,
        gate,
        platform
    )
}

fn root_token(root: &Requirement, index: usize) -> String {
    let mut fs = root.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        root.package,
        constraint_text(root.constraint),
        fs.join(",")
    )
}

fn compute_closure(
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    catalog: &BTreeMap<String, &Package>,
    target: Target,
) -> Closure {
    let mut closure = Closure::default();
    let mut origins: BTreeMap<String, Vec<Origin>> = BTreeMap::new();
    let mut requests: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();

    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(root, i)];
        origins.entry(root.package.clone()).or_default().push(Origin {
            constraint: root.constraint,
            path: path.clone(),
        });
        requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        closure.paths.entry(root.package.clone()).or_insert(path);
    }

    loop {
        let old_origins = origins.clone();
        let old_requests = requests.clone();
        let old_deps = closure.active_deps.clone();
        let old_paths = closure.paths.clone();

        let packages: Vec<String> = selected.keys().cloned().collect();
        for package_name in packages {
            let Some(version) = selected.get(&package_name).copied() else {
                continue;
            };
            let Some(package) = catalog.get(&package_name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };
            let declared = feature_index(release);
            let mut activated = requests.get(&package_name).cloned().unwrap_or_default();

            loop {
                let before = activated.len();
                let current: Vec<String> = activated.iter().cloned().collect();
                for f in current {
                    if let Some(def) = declared.get(f.as_str()) {
                        activated.extend(def.enables.iter().cloned());
                    }
                }
                if activated.len() == before {
                    break;
                }
            }
            requests.insert(package_name.clone(), activated.clone());

            for (di, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target) {
                    continue;
                }
                if dep
                    .gate
                    .as_ref()
                    .is_some_and(|gate| !activated.contains(gate))
                {
                    continue;
                }
                closure
                    .active_deps
                    .entry(package_name.clone())
                    .or_default()
                    .insert(di);
                let source_path = closure
                    .paths
                    .get(&package_name)
                    .cloned()
                    .unwrap_or_default();
                let mut path = source_path;
                path.push(dep_token(&package_name, release, di, dep));
                let origin = Origin {
                    constraint: dep.constraint,
                    path: path.clone(),
                };
                let list = origins.entry(dep.package.clone()).or_default();
                if !list.iter().any(|o| {
                    o.constraint == origin.constraint && o.path == origin.path
                }) {
                    list.push(origin);
                }
                requests
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());
                let current_path = closure.paths.get(&dep.package);
                if current_path.is_none_or(|old| {
                    path.len() < old.len() || (path.len() == old.len() && path < *old)
                }) {
                    closure.paths.insert(dep.package.clone(), path);
                }
            }
        }

        if origins == old_origins
            && requests == old_requests
            && closure.active_deps == old_deps
            && closure.paths == old_paths
        {
            break;
        }
    }

    for (package, os) in origins {
        let mut constraints = Vec::new();
        let mut evidence = Vec::new();
        for o in os {
            constraints.push(o.constraint);
            evidence.push(Evidence {
                constraint: o.constraint,
                path: o.path,
            });
        }
        closure
            .constraints
            .insert(package.clone(), canonical_constraints(&constraints));
        closure.evidence.insert(package, evidence);
    }
    closure.features = requests;
    closure
}

fn error_at_node(
    closure: &Closure,
    selected: &BTreeMap<String, Version>,
    catalog: &BTreeMap<String, &Package>,
) -> Option<ResolveError> {
    for package in closure.constraints.keys() {
        if !catalog.contains_key(package) {
            return Some(ResolveError::Missing {
                package: package.clone(),
                evidence: closure.evidence.get(package).cloned().unwrap_or_default(),
            });
        }
    }

    for (package, version) in selected {
        let Some(cs) = closure.constraints.get(package) else {
            continue;
        };
        if cs.iter().any(|c| !matches(*version, *c)) {
            return Some(ResolveError::Conflict {
                package: package.clone(),
                selected: *version,
                requirements: cs.clone(),
                evidence: closure.evidence.get(package).cloned().unwrap_or_default(),
            });
        }
    }

    for (package, version) in selected {
        let Some(pkg) = catalog.get(package) else {
            continue;
        };
        let Some(release) = pkg.releases.iter().find(|r| r.version == *version) else {
            continue;
        };
        let activated = closure.features.get(package).cloned().unwrap_or_default();
        for feature in activated {
            if !release.features.iter().any(|f| f.name == feature) {
                return Some(ResolveError::MissingFeature {
                    package: package.clone(),
                    selected: *version,
                    feature,
                    path: closure.paths.get(package).cloned().unwrap_or_default(),
                });
            }
        }
    }

    for package in closure.constraints.keys() {
        if selected.contains_key(package) {
            continue;
        }
        let cs = closure.constraints.get(package).unwrap();
        let has_candidate = catalog
            .get(package)
            .map(|p| {
                p.releases
                    .iter()
                    .any(|r| cs.iter().all(|c| matches(r.version, *c)))
            })
            .unwrap_or(false);
        if !has_candidate {
            return Some(ResolveError::NoMatch {
                package: package.clone(),
                requirements: cs.clone(),
                evidence: closure.evidence.get(package).cloned().unwrap_or_default(),
            });
        }
    }

    None
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: &'a BTreeMap<String, &'a Package>,
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>, Closure)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let closure = compute_closure(self.roots, selected, self.catalog, self.target);

        if let Some(err) = error_at_node(&closure, selected, self.catalog) {
            if self.first_error.is_none() {
                self.first_error = Some(err);
            }
            return Ok(());
        }

        let next = closure
            .constraints
            .keys()
            .find(|p| !selected.contains_key(*p))
            .cloned();

        let Some(package_name) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(p, v)| selected.get(*p) != Some(*v))
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_sel, _)) => {
                    if cost != *best_cost {
                        cost < *best_cost
                    } else {
                        self.better_vector(selected, best_sel)
                    }
                }
            };
            if replace {
                self.best = Some((cost, selected.clone(), closure));
            }
            return Ok(());
        };

        let cs = closure.constraints.get(&package_name).cloned().unwrap_or_default();
        let Some(pkg) = self.catalog.get(&package_name) else {
            return Ok(());
        };
        let mut candidates: Vec<Version> = pkg
            .releases
            .iter()
            .filter(|r| cs.iter().all(|c| matches(r.version, *c)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();
        if let Some(lock_version) = self.locks.get(&package_name) {
            if let Some(pos) = candidates.iter().position(|v| v == lock_version) {
                let locked = candidates.remove(pos);
                candidates.insert(0, locked);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package_name.clone(), candidate);
            self.visit(selected)?;
            selected.remove(&package_name);
        }
        Ok(())
    }

    fn better_vector(
        &self,
        left: &BTreeMap<String, Version>,
        right: &BTreeMap<String, Version>,
    ) -> bool {
        let mut names: BTreeSet<String> = self.catalog.keys().cloned().collect();
        names.extend(left.keys().cloned());
        names.extend(right.keys().cloned());
        for name in names {
            match (left.get(&name), right.get(&name)) {
                (Some(a), Some(b)) if a != b => return a > b,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                _ => {}
            }
        }
        false
    }
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(invalid(format!("root[{}]", i), &root.package));
        }
        for (f, name) in root.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(invalid(format!("root[{}].feature[{}]", i, f), name));
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(invalid(format!("catalog[{}]", i), &package.name));
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(invalid(base.clone(), &dep.package));
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(invalid(format!("{}.gate", base), gate));
                    }
                }
                for (f, name) in dep.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(invalid(format!("{}.feature[{}]", base, f), name));
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].feature[{}]", i, j, f);
                if !valid_name(&feature.name) {
                    return Err(invalid(base.clone(), &feature.name));
                }
                for (e, name) in feature.enables.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(invalid(format!("{}.enable[{}]", base, e), name));
                    }
                }
            }
        }
    }

    let mut package_names = HashSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = HashSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = HashSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(invalid(format!("lock[{}]", i), &lock.package));
        }
    }
    let mut lock_names = HashSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut search = Search {
        roots,
        catalog: &catalog_map,
        locks: &lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut selected = BTreeMap::new();
    search.visit(&mut selected)?;

    let Some((_, chosen, final_closure)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_vec: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut all_names: BTreeSet<String> = locks.iter().map(|l| l.package.clone()).collect();
    all_names.extend(chosen.keys().cloned());
    let mut changes = Vec::new();

    for name in all_names {
        let before = lock_map.get(&name).copied();
        let after = chosen.get(&name).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&name)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_closure
                    .constraints
                    .get(&name)
                    .is_some_and(|cs| cs.iter().any(|c| !matches(old, *c)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
