#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty()
        || bytes.len() > 24
        || !bytes[0].is_ascii_lowercase()
        || bytes[bytes.len() - 1] == b'-'
    {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .iter()
        .copied()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(value: &str) -> String {
    let mut result = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            result.push(*byte as char);
        } else {
            result.push_str(&format!("%{:02X}", byte));
        }
    }
    result
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(wanted) => version == wanted,
        Constraint::AtLeast(minimum) => version >= minimum,
        Constraint::Before(bound) => version < bound,
        Constraint::Caret(base) => {
            version >= base
                && if base.major > 0 {
                    version.major == base.major
                } else if base.minor > 0 {
                    version.major == 0 && version.minor == base.minor
                } else {
                    version == base
                }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| matches_constraint(version, *constraint))
}

#[derive(Clone)]
struct SearchResult {
    assignments: Vec<Option<Version>>,
    constraints: Vec<Vec<Constraint>>,
}

struct Search<'a> {
    catalog: &'a [Package],
    package_index: &'a std::collections::BTreeMap<String, usize>,
    roots: &'a [Requirement],
    locks: &'a [Lock],
    attempts: usize,
    max_attempts: usize,
    best: Option<SearchResult>,
    best_cost: usize,
    first_error: Option<ResolveError>,
    saw_solution: bool,
}

impl<'a> Search<'a> {
    fn active_constraints(
        &self,
        assignments: &[Option<Version>],
    ) -> Vec<Vec<Constraint>> {
        let mut active = vec![Vec::new(); self.catalog.len()];
        for root in self.roots {
            if let Some(&index) = self.package_index.get(&root.package) {
                active[index].push(root.constraint);
            }
        }
        for (index, assigned) in assignments.iter().enumerate() {
            if let Some(version) = assigned {
                if let Some(release) = self.catalog[index]
                    .releases
                    .iter()
                    .find(|release| release.version == *version)
                {
                    for dep in &release.deps {
                        if let Some(&dep_index) = self.package_index.get(&dep.package) {
                            active[dep_index].push(dep.constraint);
                        }
                    }
                }
            }
        }
        for constraints in &mut active {
            canonicalize(constraints);
        }
        active
    }

    fn current_error(
        &self,
        assignments: &[Option<Version>],
        active: &[Vec<Constraint>],
    ) -> Option<ResolveError> {
        let mut active_names: Vec<&str> = Vec::new();
        for (index, constraints) in active.iter().enumerate() {
            if !constraints.is_empty() {
                active_names.push(&self.catalog[index].name);
            }
        }
        // A package is active even when its constraints happen to be empty only
        // in no valid input path; every activation contributes one constraint.
        active_names.sort_unstable();

        for name in &active_names {
            if !self.package_index.contains_key(*name) {
                return Some(ResolveError::Missing {
                    package: (*name).to_string(),
                });
            }
        }

        let mut indices: Vec<usize> = (0..self.catalog.len()).collect();
        indices.sort_by(|a, b| self.catalog[*a].name.cmp(&self.catalog[*b].name));

        for index in &indices {
            if let Some(version) = assignments[*index] {
                if !matches_all(version, &active[*index]) {
                    return Some(ResolveError::Conflict {
                        package: self.catalog[*index].name.clone(),
                        selected: version,
                        requirements: active[*index].clone(),
                    });
                }
            }
        }

        for index in &indices {
            if !active[*index].is_empty() && assignments[*index].is_none() {
                let has_match = self.catalog[*index]
                    .releases
                    .iter()
                    .any(|release| matches_all(release.version, &active[*index]));
                if !has_match {
                    return Some(ResolveError::NoMatch {
                        package: self.catalog[*index].name.clone(),
                        requirements: active[*index].clone(),
                    });
                }
            }
        }
        None
    }

    fn lock_version(&self, package: &str) -> Option<Version> {
        self.locks
            .iter()
            .find(|lock| lock.package == package)
            .map(|lock| lock.version)
    }

    fn solution_cost(&self, assignments: &[Option<Version>]) -> usize {
        self.locks
            .iter()
            .filter(|lock| {
                self.package_index
                    .get(&lock.package)
                    .and_then(|index| assignments[*index])
                    != Some(lock.version)
            })
            .count()
    }

    fn better_vector(&self, candidate: &[Option<Version>], current: &[Option<Version>]) -> bool {
        let mut indices: Vec<usize> = (0..self.catalog.len()).collect();
        indices.sort_by(|a, b| self.catalog[*a].name.cmp(&self.catalog[*b].name));
        for index in indices {
            match (candidate[index], current[index]) {
                (Some(a), Some(b)) if a != b => return a > b,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                _ => {}
            }
        }
        false
    }

    fn record_solution(
        &mut self,
        assignments: &[Option<Version>],
        active: &[Vec<Constraint>],
    ) {
        self.saw_solution = true;
        let cost = self.solution_cost(assignments);
        let replace = match &self.best {
            None => true,
            Some(best) if cost < self.best_cost => true,
            Some(best) if cost == self.best_cost => {
                self.better_vector(assignments, &best.assignments)
            }
            _ => false,
        };
        if replace {
            self.best_cost = cost;
            self.best = Some(SearchResult {
                assignments: assignments.to_vec(),
                constraints: active.to_vec(),
            });
        }
    }

    fn visit(&mut self, assignments: &mut Vec<Option<Version>>) -> Result<(), ResolveError> {
        let active = self.active_constraints(assignments);
        if let Some(error) = self.current_error(assignments, &active) {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        let mut indices: Vec<usize> = (0..self.catalog.len()).collect();
        indices.sort_by(|a, b| self.catalog[*a].name.cmp(&self.catalog[*b].name));
        let next = indices
            .into_iter()
            .find(|index| !active[*index].is_empty() && assignments[*index].is_none());

        let Some(index) = next else {
            self.record_solution(assignments, &active);
            return Ok(());
        };

        let mut candidates: Vec<Version> = self.catalog[index]
            .releases
            .iter()
            .map(|release| release.version)
            .filter(|version| matches_all(*version, &active[index]))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(locked) = self.lock_version(&self.catalog[index].name) {
            if let Some(position) = candidates.iter().position(|version| *version == locked) {
                candidates.remove(position);
                candidates.insert(0, locked);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assignments[index] = Some(candidate);
            let result = self.visit(assignments);
            assignments[index] = None;
            result?;
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut package_index = std::collections::BTreeMap::new();
    for (index, package) in catalog.iter().enumerate() {
        package_index.insert(package.name.clone(), index);
    }

    let mut search = Search {
        catalog,
        package_index: &package_index,
        roots,
        locks,
        attempts: 0,
        max_attempts,
        best: None,
        best_cost: usize::MAX,
        first_error: None,
        saw_solution: false,
    };
    let mut assignments = vec![None; catalog.len()];
    search.visit(&mut assignments)?;

    let Some(best) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let mut selected = Vec::new();
    for (index, version) in best.assignments.iter().enumerate() {
        if let Some(version) = version {
            selected.push(Selection {
                package: catalog[index].name.clone(),
                version: *version,
            });
        }
    }
    selected.sort_by(|a, b| a.package.cmp(&b.package));

    let before: std::collections::BTreeMap<&str, Version> = locks
        .iter()
        .map(|lock| (lock.package.as_str(), lock.version))
        .collect();
    let after: std::collections::BTreeMap<&str, Version> = selected
        .iter()
        .map(|selection| (selection.package.as_str(), selection.version))
        .collect();
    let mut names = std::collections::BTreeSet::new();
    names.extend(before.keys().copied());
    names.extend(after.keys().copied());

    let mut changes = Vec::new();
    for name in names {
        let old = before.get(name).copied();
        let new = after.get(name).copied();
        if old == new {
            continue;
        }
        let reason = match (old, new) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old_version), Some(_)) => {
                let exists = package_index
                    .get(name)
                    .map(|index| {
                        catalog[*index]
                            .releases
                            .iter()
                            .any(|release| release.version == old_version)
                    })
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let constraints = package_index
                        .get(name)
                        .map(|index| &best.constraints[*index][..])
                        .unwrap_or(&[]);
                    if !matches_all(old_version, constraints) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name.to_string(),
            before: old,
            after: new,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
