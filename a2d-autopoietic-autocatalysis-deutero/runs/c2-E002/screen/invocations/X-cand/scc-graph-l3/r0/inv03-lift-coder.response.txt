pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }
    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adj = vec![Vec::<usize>::new(); n];
    let mut rev = vec![Vec::<usize>::new(); n];
    for &(u, v) in edges {
        adj[u].push(v);
        rev[v].push(u);
    }

    // Iterative Kosaraju avoids recursion on long paths.
    let mut seen = vec![false; n];
    let mut finish = Vec::with_capacity(n);
    for start in 0..n {
        if seen[start] {
            continue;
        }
        seen[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((v, next)) = stack.last_mut() {
            if *next < adj[*v].len() {
                let w = adj[*v][*next];
                *next += 1;
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                let (done, _) = stack.pop().unwrap();
                finish.push(done);
            }
        }
    }

    let mut raw_id = vec![usize::MAX; n];
    let mut raw_components: Vec<Vec<usize>> = Vec::new();
    for &start in finish.iter().rev() {
        if raw_id[start] != usize::MAX {
            continue;
        }
        let id = raw_components.len();
        let mut members = Vec::new();
        let mut stack = vec![start];
        raw_id[start] = id;
        while let Some(v) = stack.pop() {
            members.push(v);
            for &w in &rev[v] {
                if raw_id[w] == usize::MAX {
                    raw_id[w] = id;
                    stack.push(w);
                }
            }
        }
        members.sort_unstable();
        raw_components.push(members);
    }

    let mut order: Vec<usize> = (0..raw_components.len()).collect();
    order.sort_by(|&a, &b| {
        raw_components[b]
            .len()
            .cmp(&raw_components[a].len())
            .then_with(|| raw_components[a][0].cmp(&raw_components[b][0]))
    });

    let mut remap = vec![0usize; raw_components.len()];
    let mut components = Vec::with_capacity(raw_components.len());
    for (new_id, old_id) in order.into_iter().enumerate() {
        remap[old_id] = new_id;
        components.push(raw_components[old_id].clone());
    }
    let component_of: Vec<usize> = raw_id.iter().map(|&id| remap[id]).collect();
    let c = components.len();

    let mut cyclic: Vec<bool> = components.iter().map(|m| m.len() > 1).collect();
    let mut internal_edges = vec![0usize; c];
    let mut arc_map = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_map.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            entry.1 = entry.1.min(i);
        }
    }

    let mut arcs: Vec<Arc> = arc_map
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::<usize>::new(); c];
    let mut indegree = vec![0usize; c];
    let mut predecessors = vec![Vec::<usize>::new(); c];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        indegree[arc.to] += 1;
        predecessors[arc.to].push(arc.from);
    }

    // Binary heap key: greater internal edge count, then greater minimum vertex.
    let mut ready = std::collections::BinaryHeap::<(usize, usize, usize)>::new();
    for id in 0..c {
        if indegree[id] == 0 {
            ready.push((internal_edges[id], components[id][0], id));
        }
    }
    let mut topo = Vec::with_capacity(c);
    while let Some((_, _, v)) = ready.pop() {
        topo.push(v);
        for &w in &outgoing[v] {
            indegree[w] -= 1;
            if indegree[w] == 0 {
                ready.push((internal_edges[w], components[w][0], w));
            }
        }
    }

    let mut depth = vec![0usize; c];
    for &v in &topo {
        for &w in &outgoing[v] {
            depth[w] = depth[w].max(depth[v] + 1);
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        let n = self.component_of.len();
        if from >= n {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        let mut outgoing = vec![Vec::<usize>::new(); self.components.len()];
        for arc in &self.arcs {
            outgoing[arc.from].push(arc.to);
        }
        for neighbors in &mut outgoing {
            neighbors.sort_unstable();
        }

        fn shortest(
            start: usize,
            goal: usize,
            outgoing: &[Vec<usize>],
        ) -> Option<Vec<usize>> {
            let mut parent = vec![usize::MAX; outgoing.len()];
            let mut queue = std::collections::VecDeque::new();
            parent[start] = start;
            queue.push_back(start);
            while let Some(v) = queue.pop_front() {
                if v == goal {
                    break;
                }
                for &w in &outgoing[v] {
                    if parent[w] == usize::MAX {
                        parent[w] = v;
                        queue.push_back(w);
                    }
                }
            }
            if parent[goal] == usize::MAX {
                return None;
            }
            let mut route = vec![goal];
            let mut v = goal;
            while v != start {
                v = parent[v];
                route.push(v);
            }
            route.reverse();
            Some(route)
        }

        if let Some(route) = shortest(a, b, &outgoing) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest(b, a, &outgoing) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}
