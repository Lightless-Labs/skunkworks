#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|field| field == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|field| field == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_records(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut start = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = Vec::new();
    let mut index = start;
    let mut record = 0usize;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;
    let mut record_started = false;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == 0 {
            return Err(CsvError::Nul { record, byte: index });
        }

        if !in_quotes && byte == b'\r' {
            if bytes.get(index + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr { record, byte: index });
            }
            fields.push(String::from_utf8(field).expect("valid UTF-8 field"));
            records.push(fields);
            fields = Vec::new();
            field = Vec::new();
            record += 1;
            record_started = false;
            after_quote = false;
            index += 2;
            continue;
        }

        if in_quotes {
            if byte == quote {
                if bytes.get(index + 1) == Some(&quote) {
                    field.push(quote);
                    index += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    index += 1;
                }
            } else {
                field.push(byte);
                index += 1;
            }
            continue;
        }

        if after_quote {
            if byte == delimiter {
                fields.push(String::from_utf8(field).expect("valid UTF-8 field"));
                field = Vec::new();
                after_quote = false;
                record_started = true;
                index += 1;
            } else if byte == b'\n' {
                fields.push(String::from_utf8(field).expect("valid UTF-8 field"));
                records.push(fields);
                fields = Vec::new();
                field = Vec::new();
                record += 1;
                record_started = false;
                after_quote = false;
                index += 1;
            } else {
                return Err(CsvError::TrailingByte { record, byte: index });
            }
            continue;
        }

        if byte == b'\n' {
            fields.push(String::from_utf8(field).expect("valid UTF-8 field"));
            records.push(fields);
            fields = Vec::new();
            field = Vec::new();
            record += 1;
            record_started = false;
            index += 1;
        } else if byte == delimiter {
            fields.push(String::from_utf8(field).expect("valid UTF-8 field"));
            field = Vec::new();
            record_started = true;
            index += 1;
        } else if byte == quote {
            if field.is_empty() {
                in_quotes = true;
                quote_open = index;
                record_started = true;
                index += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record, byte: index });
            }
        } else {
            field.push(byte);
            record_started = true;
            index += 1;
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_open,
        });
    }

    if record_started || after_quote || !fields.is_empty() || !field.is_empty() {
        fields.push(String::from_utf8(field).expect("valid UTF-8 field"));
        records.push(fields);
    }

    Ok(records)
}

fn records_to_table(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let mut iter = records.into_iter();
    if !config.headers {
        return Ok(Table {
            header: None,
            rows: iter.collect(),
            duplicates: config.duplicates,
        });
    }

    let Some(header) = iter.next() else {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    };

    if config.duplicates == DuplicatePolicy::Reject {
        let mut seen: Vec<(&str, usize)> = Vec::new();
        for (second, name) in header.iter().enumerate() {
            if let Some((_, first)) = seen.iter().find(|(prior, _)| *prior == name) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first: *first,
                    second,
                });
            }
            seen.push((name, second));
        }
    }

    let expected = header.len();
    let mut rows = Vec::new();
    for (record, row) in iter.enumerate() {
        if row.len() != expected {
            return Err(CsvError::HeaderWidth {
                record: record + 1,
                expected,
                actual: row.len(),
            });
        }
        rows.push(row);
    }

    Ok(Table {
        header: Some(header),
        rows,
        duplicates: config.duplicates,
    })
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_records(input, b',', b'"')
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }

        let result = parse_records(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| records_to_table(records, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}
