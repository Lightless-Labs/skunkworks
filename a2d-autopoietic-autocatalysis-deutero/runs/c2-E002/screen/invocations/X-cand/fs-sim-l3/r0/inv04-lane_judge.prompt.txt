use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn validate_and_normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            self.cwd
                .split('/')
                .filter(|part| !part.is_empty())
                .map(str::to_owned)
                .collect()
        };

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                _ => components.push(part.to_owned()),
            }
        }

        Ok(Self::canonical(&components))
    }

    fn components(path: &str) -> Vec<&str> {
        path.split('/').filter(|part| !part.is_empty()).collect()
    }

    fn prefix(components: &[&str], end: usize) -> String {
        if end == 0 {
            "/".to_owned()
        } else {
            format!("/{}", components[..end].join("/"))
        }
    }

    fn resolve_directory(&self, path: &str) -> Result<(), Error> {
        let components = Self::components(path);
        let mut current = String::new();

        for (i, component) in components.iter().enumerate() {
            current.push('/');
            current.push_str(component);

            match self.entries.get(&current) {
                None => return Err(Error::Missing(current)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) if i + 1 < components.len() => {
                    return Err(Error::NotDir(current));
                }
                Some(Entry::File(_)) => {}
            }
        }
        Ok(())
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let components = Self::components(path);
        let basename = match components.last() {
            Some(name) => (*name).to_owned(),
            None => return Ok(("/".to_owned(), String::new())),
        };
        let parent = Self::prefix(&components, components.len() - 1);
        self.resolve_directory(&parent)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => Ok((parent, basename)),
            Some(Entry::File(_)) => Err(Error::NotDir(parent)),
            None => Err(Error::Missing(parent)),
        }
    }

    fn is_at_or_below(path: &str, ancestor: &str) -> bool {
        if ancestor == "/" {
            return path.starts_with('/');
        }
        path == ancestor
            || path
                .strip_prefix(ancestor)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn direct_children(&self, path: &str) -> Vec<(String, Kind)> {
        let depth = Self::components(path).len();
        let mut children = self
            .entries
            .iter()
            .filter_map(|(candidate, entry)| {
                let parts = Self::components(candidate);
                if parts.len() == depth + 1
                    && Self::prefix(&parts, depth) == path
                {
                    Some((parts[depth].to_owned(), entry.kind()))
                } else {
                    None
                }
            })
            .collect::<Vec<_>>();
        children.sort_by(|a, b| a.0.cmp(&b.0));
        children
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.validate_and_normalize(path)?;
        let (parent, _) = self.resolve_parent(&path)?;

        if self.entries.contains_key(&path) {
            return Err(Error::Exists(path));
        }
        self.entries.insert(path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.validate_and_normalize(path)?;
        let (parent, _) = self.resolve_parent(&path)?;

        match self.entries.get(&path) {
            Some(Entry::Dir) => return Err(Error::IsDir(path)),
            Some(Entry::File(_)) | None => {}
        }

        let _ = parent;
        self.entries
            .insert(path, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.validate_and_normalize(path)?;
        self.resolve_directory(&path)?;
        match self.entries.get(&path) {
            None => Err(Error::Missing(path)),
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            Some(Entry::File(data)) => Ok(data.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.validate_and_normalize(path)?;
        self.resolve_directory(&path)?;
        match self.entries.get(&path) {
            None => Err(Error::Missing(path)),
            Some(Entry::File(_)) => Err(Error::NotDir(path)),
            Some(Entry::Dir) => Ok(self.direct_children(&path)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.validate_and_normalize(path)?;
        self.resolve_directory(&path)?;
        match self.entries.get(&path) {
            None => Err(Error::Missing(path)),
            Some(Entry::File(_)) => Err(Error::NotDir(path)),
            Some(Entry::Dir) => {
                self.cwd = path;
                Ok(())
            }
        }
    }

    pub fn rename(
        &mut self,
        from: &str,
        to: &str,
        overwrite: bool,
    ) -> Result<(), Error> {
        let source = self.validate_and_normalize(from)?;
        let destination = self.validate_and_normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, _) = self.resolve_parent(&source)?;
        let source_entry = match self.entries.get(&source) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(source)),
        };

        if matches!(source_entry, Entry::Dir)
            && Self::is_at_or_below(&self.cwd, &source)
        {
            return Err(Error::Busy(source));
        }

        let (_destination_parent, _) = self.resolve_parent(&destination)?;
        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::is_at_or_below(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir)
            && Self::is_at_or_below(&destination, &source)
        {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(ref existing) = destination_entry {
            match (&source_entry, existing) {
                (Entry::Dir, Entry::File(_)) => {
                    return Err(Error::NotDir(destination));
                }
                (Entry::File(_), Entry::Dir) => {
                    return Err(Error::IsDir(destination));
                }
                (Entry::Dir, Entry::Dir) => {
                    if !self.direct_children(&destination).is_empty() {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                _ => {}
            }
        }

        let source_prefix = format!("{}/", source.trim_end_matches('/'));
        let mut moved = Vec::new();
        for (key, entry) in &self.entries {
            if key == &source || key.starts_with(&source_prefix) {
                let suffix = &key[source.len()..];
                moved.push((format!("{destination}{suffix}"), entry.clone()));
            }
        }

        if let Some(existing) = destination_entry {
            if matches!(existing, Entry::Dir) {
                self.entries.remove(&destination);
            } else {
                self.entries.remove(&destination);
            }
        }

        let remove_keys = self
            .entries
            .keys()
            .filter(|key| *key == &source || key.starts_with(&source_prefix))
            .cloned()
            .collect::<Vec<_>>();
        for key in remove_keys {
            self.entries.remove(&key);
        }

        let _ = source_parent;
        for (key, entry) in moved {
            self.entries.insert(key, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.validate_and_normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&path)?;
        let entry = match self.entries.get(&path) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(path)),
        };

        if matches!(entry, Entry::Dir) && Self::is_at_or_below(&self.cwd, &path) {
            return Err(Error::Busy(path));
        }

        let prefix = format!("{}/", path.trim_end_matches('/'));
        let has_children = self
            .entries
            .keys()
            .any(|key| key.starts_with(&prefix));

        if matches!(entry, Entry::Dir) && has_children && !recursive {
            return Err(Error::NotEmpty(path));
        }

        if recursive && matches!(entry, Entry::Dir) {
            let keys = self
                .entries
                .keys()
                .filter(|key| *key == &path || key.starts_with(&prefix))
                .cloned()
                .collect::<Vec<_>>();
            for key in keys {
                self.entries.remove(&key);
            }
        } else {
            self.entries.remove(&path);
        }
        Ok(())
    }
}

use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let absolute = path.starts_with('/');
        let mut components: Vec<String> = if absolute || self.cwd == "/" {
            Vec::new()
        } else {
            self.cwd[1..].split('/').map(str::to_string).collect()
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                other => components.push(other.to_string()),
            }
        }

        if components.is_empty() {
            Ok("/".to_string())
        } else {
            Ok(format!("/{}", components.join("/")))
        }
    }

    fn parent_and_name(path: &str) -> Option<(String, String)> {
        if path == "/" {
            return None;
        }
        let split = path.rfind('/').expect("canonical path contains slash");
        let parent = if split == 0 {
            "/".to_string()
        } else {
            path[..split].to_string()
        };
        Some((parent, path[split + 1..].to_string()))
    }

    fn resolve_parent(&self, path: &str) -> Result<String, Error> {
        let (parent, _) = Self::parent_and_name(path).expect("non-root path");
        self.resolve_dir(&parent)?;
        Ok(parent)
    }

    fn resolve_dir(&self, path: &str) -> Result<(), Error> {
        let mut prefix = String::new();
        if path == "/" {
            return Ok(());
        }
        for component in path[1..].split('/') {
            prefix.push('/');
            prefix.push_str(component);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn resolve_entry(&self, path: &str) -> Result<&Entry, Error> {
        if path == "/" {
            return self.entries.get("/").ok_or_else(|| Error::Missing("/".to_string()));
        }
        self.resolve_parent(path)?;
        self.entries
            .get(path)
            .ok_or_else(|| Error::Missing(path.to_string()))
    }

    fn is_at_or_below(path: &str, dir: &str) -> bool {
        if dir == "/" {
            return path.starts_with('/');
        }
        path == dir
            || path
                .strip_prefix(dir)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn direct_children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                if candidate.as_str() == path || !Self::is_at_or_below(candidate, path) {
                    return false;
                }
                let remainder = if path == "/" {
                    &candidate[1..]
                } else {
                    &candidate[path.len() + 1..]
                };
                !remainder.is_empty() && !remainder.contains('/')
            })
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Exists(path));
        }
        self.resolve_parent(&path)?;
        if self.entries.contains_key(&path) {
            return Err(Error::Exists(path));
        }
        self.entries.insert(path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::IsDir(path));
        }
        self.resolve_parent(&path)?;
        match self.entries.get(&path) {
            Some(Entry::Dir) => return Err(Error::IsDir(path)),
            Some(Entry::File(_)) | None => {}
        }
        self.entries
            .insert(path, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        match self.resolve_entry(&path)? {
            Entry::Dir => Err(Error::IsDir(path)),
            Entry::File(data) => Ok(data.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        match self.resolve_entry(&path)? {
            Entry::File(_) => return Err(Error::NotDir(path)),
            Entry::Dir => {}
        }

        let mut children: Vec<(String, Kind)> = self
            .direct_children(&path)
            .into_iter()
            .map(|child| {
                let name = child.rsplit('/').next().unwrap_or("").to_string();
                let kind = self.entries.get(&child).expect("listed entry exists").kind();
                (name, kind)
            })
            .collect();
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        match self.resolve_entry(&path)? {
            Entry::Dir => {
                self.cwd = path;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(path)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&source)?;
        if !self.entries.contains_key(&source) {
            return Err(Error::Missing(source));
        }

        let source_is_dir = matches!(self.entries.get(&source), Some(Entry::Dir));
        if source_is_dir && Self::is_at_or_below(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        self.resolve_parent(&destination)?;
        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::is_at_or_below(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if source_is_dir && Self::is_at_or_below(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination);
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(existing) = destination_entry {
            match (source_is_dir, existing) {
                (true, Entry::File(_)) => return Err(Error::NotDir(destination)),
                (false, Entry::Dir) => return Err(Error::IsDir(destination)),
                (true, Entry::Dir) => {
                    if !self.direct_children(&destination).is_empty() {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (false, Entry::File(_)) => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| {
                **path == source || (source_is_dir && Self::is_at_or_below(path, &source))
            })
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        let replaced: Vec<String> = if destination_entry.is_some() {
            self.entries
                .keys()
                .filter(|path| {
                    **path == destination
                        || (matches!(destination_entry, Some(Entry::Dir))
                            && Self::is_at_or_below(path, &destination))
                })
                .cloned()
                .collect()
        } else {
            Vec::new()
        };

        for path in replaced {
            self.entries.remove(&path);
        }
        for (old_path, _) in &moving {
            self.entries.remove(old_path);
        }

        for (old_path, entry) in moving {
            let suffix = &old_path[source.len()..];
            let new_path = format!("{destination}{suffix}");
            self.entries.insert(new_path, entry);
        }

        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&path)?;
        let entry = self
            .entries
            .get(&path)
            .ok_or_else(|| Error::Missing(path.clone()))?;

        let is_dir = matches!(entry, Entry::Dir);
        if is_dir && Self::is_at_or_below(&self.cwd, &path) {
            return Err(Error::Busy(path));
        }

        let children = if is_dir {
            self.direct_children(&path)
        } else {
            Vec::new()
        };
        if is_dir && !recursive && !children.is_empty() {
            return Err(Error::NotEmpty(path));
        }

        if recursive && is_dir {
            let to_remove: Vec<String> = self
                .entries
                .keys()
                .filter(|candidate| {
                    **candidate == path || Self::is_at_or_below(candidate, &path)
                })
                .cloned()
                .collect();
            for candidate in to_remove {
                self.entries.remove(&candidate);
            }
        } else {
            self.entries.remove(&path);
        }

        Ok(())
    }
}

use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let absolute = path.starts_with('/');
        let mut components: Vec<&str> = Vec::new();

        if !absolute && self.cwd != "/" {
            components.extend(self.cwd[1..].split('/'));
        }

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                other => components.push(other),
            }
        }

        if components.is_empty() {
            Ok("/".to_string())
        } else {
            let mut result = String::new();
            for component in components {
                result.push('/');
                result.push_str(component);
            }
            Ok(result)
        }
    }

    fn parent(path: &str) -> String {
        if path == "/" {
            return "/".to_string();
        }
        match path.rfind('/') {
            Some(0) => "/".to_string(),
            Some(i) => path[..i].to_string(),
            None => "/".to_string(),
        }
    }

    fn basename(path: &str) -> &str {
        path.rsplit('/').next().unwrap_or("")
    }

    fn is_within(path: &str, directory: &str) -> bool {
        if directory == "/" {
            return path.starts_with('/');
        }
        path == directory
            || (path.starts_with(directory)
                && path.as_bytes().get(directory.len()) == Some(&b'/'))
    }

    fn resolve_directory(&self, path: &str) -> Result<(), Error> {
        let mut prefix = String::new();
        for component in path.split('/').skip(1) {
            prefix.push('/');
            prefix.push_str(component);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn check_parent(&self, path: &str) -> Result<(), Error> {
        let parent = Self::parent(path);
        self.resolve_directory(&parent)
    }

    fn children(&self, path: &str) -> Vec<(String, Kind)> {
        let prefix = if path == "/" {
            "/".to_string()
        } else {
            format!("{}/", path)
        };
        let mut children = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == path || !entry_path.starts_with(&prefix) {
                continue;
            }
            let remainder = &entry_path[prefix.len()..];
            if !remainder.is_empty() && !remainder.contains('/') {
                children.push((remainder.to_string(), entry.kind()));
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        children
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        self.check_parent(&path)?;
        if self.entries.contains_key(&path) {
            return Err(Error::Exists(path));
        }
        self.entries.insert(path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        self.check_parent(&path)?;
        match self.entries.get(&path) {
            Some(Entry::Dir) => return Err(Error::IsDir(path)),
            Some(Entry::File(_)) => {}
            None => {}
        }
        self.entries.insert(path, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        self.check_parent(&path)?;
        match self.entries.get(&path) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            None => Err(Error::Missing(path)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        self.resolve_directory(&path)?;
        match self.entries.get(&path) {
            Some(Entry::Dir) => Ok(self.children(&path)),
            Some(Entry::File(_)) => Err(Error::NotDir(path)),
            None => Err(Error::Missing(path)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        self.resolve_directory(&path)?;
        match self.entries.get(&path) {
            Some(Entry::Dir) => {
                self.cwd = path;
                Ok(())
            }
            Some(Entry::File(_)) => Err(Error::NotDir(path)),
            None => Err(Error::Missing(path)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = self.normalize(from)?;
        let to = self.normalize(to)?;

        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        self.check_parent(&from)?;
        let source = self
            .entries
            .get(&from)
            .cloned()
            .ok_or_else(|| Error::Missing(from.clone()))?;

        if matches!(source, Entry::Dir) && Self::is_within(&self.cwd, &from) {
            return Err(Error::Busy(from));
        }

        self.check_parent(&to)?;

        if matches!(self.entries.get(&to), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &to)
        {
            return Err(Error::Busy(to));
        }

        if from == to {
            return Ok(());
        }

        if matches!(source, Entry::Dir) && Self::is_within(&to, &from) {
            return Err(Error::Cycle(to));
        }

        let destination = self.entries.get(&to).cloned();
        if destination.is_some() && !overwrite {
            return Err(Error::Exists(to));
        }

        if let Some(destination_entry) = destination.as_ref() {
            match (&source, destination_entry) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(to)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(to)),
                (Entry::Dir, Entry::Dir) if !self.children(&to).is_empty() => {
                    return Err(Error::NotEmpty(to));
                }
                _ => {}
            }
        }

        let old_prefix = format!("{}/", from);
        let new_prefix = format!("{}/", to);
        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| *path == &from || path.starts_with(&old_prefix))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if let Some(destination_entry) = destination {
            if matches!(destination_entry, Entry::Dir) {
                self.entries.remove(&to);
            } else {
                self.entries.remove(&to);
            }
        }

        for (path, _) in &moving {
            self.entries.remove(path);
        }

        for (path, entry) in moving {
            let new_path = if path == from {
                to.clone()
            } else {
                format!("{}{}", new_prefix, &path[old_prefix.len()..])
            };
            self.entries.insert(new_path, entry);
        }

        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }

        self.check_parent(&path)?;
        let entry = self
            .entries
            .get(&path)
            .ok_or_else(|| Error::Missing(path.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &path) {
            return Err(Error::Busy(path));
        }

        let children = self.children(&path);
        if matches!(entry, Entry::Dir) && !recursive && !children.is_empty() {
            return Err(Error::NotEmpty(path));
        }

        let prefix = format!("{}/", path);
        let to_remove: Vec<String> = self
            .entries
            .keys()
            .filter(|entry_path| *entry_path == &path || entry_path.starts_with(&prefix))
            .cloned()
            .collect();
        for entry_path in to_remove {
            self.entries.remove(&entry_path);
        }
        Ok(())
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate_and_normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(index) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(index));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components: Vec<&str> = Vec::new();
        if !path.starts_with('/') {
            components.extend(self.cwd.split('/').filter(|part| !part.is_empty()));
        }

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                _ => components.push(part),
            }
        }

        if components.is_empty() {
            Ok("/".to_string())
        } else {
            Ok(format!("/{}", components.join("/")))
        }
    }

    fn parts(path: &str) -> impl Iterator<Item = &str> {
        path.split('/').filter(|part| !part.is_empty())
    }

    fn parent_path(path: &str) -> String {
        match path.rfind('/') {
            Some(0) | None => "/".to_string(),
            Some(index) => path[..index].to_string(),
        }
    }

    fn resolve_dir(&self, path: &str) -> Result<(), Error> {
        let mut prefix = String::new();
        for part in Self::parts(path) {
            prefix.push('/');
            prefix.push_str(part);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn resolve_parent(&self, path: &str) -> Result<String, Error> {
        let parent = Self::parent_path(path);
        self.resolve_dir(&parent)?;
        Ok(parent)
    }

    fn contains_path(directory: &str, path: &str) -> bool {
        if directory == "/" {
            return path.starts_with('/');
        }
        path == directory
            || path
                .strip_prefix(directory)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn children_of(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                if path == "/" {
                    candidate
                        .strip_prefix('/')
                        .is_some_and(|rest| !rest.is_empty() && !rest.contains('/'))
                } else {
                    candidate
                        .strip_prefix(path)
                        .and_then(|rest| rest.strip_prefix('/'))
                        .is_some_and(|rest| !rest.is_empty() && !rest.contains('/'))
                }
            })
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.validate_and_normalize(path)?;
        let parent = Self::parent_path(&path);
        self.resolve_dir(&parent)?;

        if self.entries.contains_key(&path) {
            return Err(Error::Exists(path));
        }

        self.entries.insert(path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.validate_and_normalize(path)?;
        if path == "/" {
            return Err(Error::IsDir(path));
        }

        let parent = Self::parent_path(&path);
        self.resolve_dir(&parent)?;

        match self.entries.get(&path) {
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            Some(Entry::File(_)) | None => {
                self.entries.insert(path, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.validate_and_normalize(path)?;
        if path == "/" {
            return Err(Error::IsDir(path));
        }

        self.resolve_parent(&path)?;
        match self.entries.get(&path) {
            None => Err(Error::Missing(path)),
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            Some(Entry::File(data)) => Ok(data.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.validate_and_normalize(path)?;
        self.resolve_dir(&path)?;
        match self.entries.get(&path) {
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => return Err(Error::NotDir(path)),
            None => return Err(Error::Missing(path)),
        }

        let mut result = Vec::new();
        for child in self.children_of(&path) {
            let basename = child.rsplit('/').next().unwrap_or("").to_string();
            let kind = self.entries.get(&child).expect("child entry exists").kind();
            result.push((basename, kind));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.validate_and_normalize(path)?;
        self.resolve_dir(&path)?;
        match self.entries.get(&path) {
            Some(Entry::Dir) => {
                self.cwd = path;
                Ok(())
            }
            Some(Entry::File(_)) | None => Err(Error::NotDir(path)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = self.validate_and_normalize(from)?;
        let to = self.validate_and_normalize(to)?;

        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        let source_parent = Self::parent_path(&from);
        self.resolve_dir(&source_parent)?;
        let source = match self.entries.get(&from) {
            None => return Err(Error::Missing(from)),
            Some(entry) => entry.clone(),
        };

        if matches!(source, Entry::Dir) && Self::contains_path(&from, &self.cwd) {
            return Err(Error::Busy(from));
        }

        let destination_parent = Self::parent_path(&to);
        self.resolve_dir(&destination_parent)?;

        if matches!(self.entries.get(&to), Some(Entry::Dir))
            && Self::contains_path(&to, &self.cwd)
        {
            return Err(Error::Busy(to));
        }

        if from == to {
            return Ok(());
        }

        if matches!(source, Entry::Dir) && Self::contains_path(&from, &to) {
            return Err(Error::Cycle(to));
        }

        let destination = self.entries.get(&to).cloned();
        if destination.is_some() && !overwrite {
            return Err(Error::Exists(to));
        }

        if let Some(destination_entry) = destination.as_ref() {
            match (&source, destination_entry) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(to)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(to)),
                (Entry::Dir, Entry::Dir) => {
                    if !self.children_of(&to).is_empty() {
                        return Err(Error::NotEmpty(to));
                    }
                }
                (Entry::File(_), Entry::File(_)) => {}
            }
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter_map(|(old_path, entry)| {
                if Self::contains_path(&from, old_path) {
                    let suffix = &old_path[from.len()..];
                    Some((format!("{to}{suffix}"), entry.clone()))
                } else {
                    None
                }
            })
            .collect();

        self.entries
            .retain(|old_path, _| !Self::contains_path(&from, old_path));
        if destination.is_some() {
            self.entries
                .retain(|old_path, _| !Self::contains_path(&to, old_path));
        }
        for (new_path, entry) in moved {
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.validate_and_normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }

        let parent = Self::parent_path(&path);
        self.resolve_dir(&parent)?;
        let entry = match self.entries.get(&path) {
            None => return Err(Error::Missing(path)),
            Some(entry) => entry,
        };

        if matches!(entry, Entry::Dir) && Self::contains_path(&path, &self.cwd) {
            return Err(Error::Busy(path));
        }

        if matches!(entry, Entry::Dir) && !recursive && !self.children_of(&path).is_empty() {
            return Err(Error::NotEmpty(path));
        }

        self.entries
            .retain(|candidate, _| !Self::contains_path(&path, candidate));
        Ok(())
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
