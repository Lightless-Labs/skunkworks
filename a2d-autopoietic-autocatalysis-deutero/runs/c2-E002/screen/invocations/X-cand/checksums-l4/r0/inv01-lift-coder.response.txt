#[derive(Debug, Clone)]
struct State {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
}

impl State {
    fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    fn from_seed(crc32: u32, adler32: u32) -> Self {
        Self {
            crc: crc32 ^ 0xffff_ffff,
            adler_a: adler32 & 0xffff,
            adler_b: adler32 >> 16,
        }
    }

    fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc ^= u32::from(byte);
            for _ in 0..8 {
                if self.crc & 1 != 0 {
                    self.crc = (self.crc >> 1) ^ 0xedb8_8320;
                } else {
                    self.crc >>= 1;
                }
            }

            self.adler_a = (self.adler_a + u32::from(byte)) % 65_521;
            self.adler_b = (self.adler_b + self.adler_a) % 65_521;
        }
    }

    fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index != 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }

    Ok(())
}

#[derive(Clone, Copy)]
struct Fletcher32State {
    a: u32,
    b: u32,
    pending: Option<u8>,
}

impl Fletcher32State {
    fn new() -> Self {
        Self {
            a: 0,
            b: 0,
            pending: None,
        }
    }

    fn word(&mut self, word: u16) {
        self.a = (self.a + u32::from(word)) % 65_535;
        self.b = (self.b + self.a) % 65_535;
    }

    fn update(&mut self, data: &[u8]) {
        let mut index = 0;

        if let Some(high) = self.pending.take() {
            if let Some(&low) = data.first() {
                self.word(u16::from_be_bytes([high, low]));
                index = 1;
            } else {
                self.pending = Some(high);
                return;
            }
        }

        while index + 1 < data.len() {
            self.word(u16::from_be_bytes([data[index], data[index + 1]]));
            index += 2;
        }

        if index < data.len() {
            self.pending = Some(data[index]);
        }
    }

    fn finish(mut self) -> u32 {
        if let Some(high) = self.pending.take() {
            self.word(u16::from(high) << 8);
        }
        (self.b << 16) | self.a
    }
}

#[derive(Clone, Copy)]
struct MultiState {
    base: State,
    fletcher16_a: u32,
    fletcher16_b: u32,
    fletcher32: Fletcher32State,
    crc16: u16,
    variant: Crc16,
}

impl MultiState {
    fn new(variant: Crc16) -> Self {
        let crc16 = match variant {
            Crc16::Xmodem => 0,
            Crc16::False | Crc16::X25 => 0xffff,
        };
        Self {
            base: State::new(),
            fletcher16_a: 0,
            fletcher16_b: 0,
            fletcher32: Fletcher32State::new(),
            crc16,
            variant,
        }
    }

    fn update(&mut self, data: &[u8]) {
        self.base.update(data);

        for &byte in data {
            self.fletcher16_a = (self.fletcher16_a + u32::from(byte)) % 255;
            self.fletcher16_b = (self.fletcher16_b + self.fletcher16_a) % 255;

            match self.variant {
                Crc16::Xmodem | Crc16::False => {
                    self.crc16 ^= u16::from(byte) << 8;
                    for _ in 0..8 {
                        self.crc16 = if self.crc16 & 0x8000 != 0 {
                            (self.crc16 << 1) ^ 0x1021
                        } else {
                            self.crc16 << 1
                        };
                    }
                }
                Crc16::X25 => {
                    self.crc16 ^= u16::from(byte);
                    for _ in 0..8 {
                        self.crc16 = if self.crc16 & 1 != 0 {
                            (self.crc16 >> 1) ^ 0x8408
                        } else {
                            self.crc16 >> 1
                        };
                    }
                }
            }
        }

        self.fletcher32.update(data);
    }

    fn finish(self) -> Multi {
        let crc16 = match self.variant {
            Crc16::X25 => self.crc16 ^ 0xffff,
            Crc16::Xmodem | Crc16::False => self.crc16,
        };
        Multi {
            base: self.base.checksum(),
            fletcher16: ((self.fletcher16_b as u16) << 8) | self.fletcher16_a as u16,
            fletcher32: self.fletcher32.finish(),
            crc16,
        }
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut state = State::new();
    state.update(data);
    state.checksum().crc32
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut state = State::new();

pub fn checksums(data: &[u8]) -> Hashes {

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {

pub struct Stream {
    state: State,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            state: State::new(),
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            state: State::from_seed(crc32, adler32),
        })
    }

    pub fn checksum(&self) -> Hashes {
        self.state.checksum()
    }

    pub fn update(&mut self, data: &[u8]) {
        self.state.update(data);
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.state.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        self.state = State::new();
    }

    pub fn snapshot(&self) -> String {
        let checksum = self.checksum();
        format!("c{:08x}:a{:08x}", checksum.crc32, checksum.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let crc = parse_hex_u32(bytes, 1, 9).map_err(|offset| SnapshotError::Hex { offset })?;
        let adler = parse_hex_u32(bytes, 11, 19).map_err(|offset| SnapshotError::Hex { offset })?;
        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn parse_hex_digit(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some(u32::from(byte - b'0')),
        b'a'..=b'f' => Some(u32::from(byte - b'a' + 10)),
        _ => None,
    }
}

fn parse_hex_u32(bytes: &[u8], start: usize, end: usize) -> Result<u32, usize> {
    let mut value = 0u32;
    for (offset, &byte) in bytes.iter().enumerate().take(end).skip(start) {
        let digit = parse_hex_digit(byte).ok_or(offset)?;
        value = (value << 4) | digit;
    }
    Ok(value)
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {

    state.update(data);
    state.finish()
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }

    let mut even = [0u32; 32];
    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut length = right_len;
    let mut crc = left;

    loop {
        gf2_matrix_square(&mut even, &odd);
        if length & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }

        gf2_matrix_square(&mut odd, &even);
        if length & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }
    }

    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }

    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = State::new();
    let mut len = 0u64;
    for span in spans {
        state.update(&data[span.start..span.end]);
        len += (span.end - span.start) as u64;
    }
    let hashes = state.checksum();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, hashes.crc32, hashes.adler32
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for range in [4..20, 25..33, 40..48] {
        for offset in range {
            if parse_hex_digit(bytes[offset]).is_none() {
                return Err(VerifyError::Hex { offset });
            }
        }
    }

    let length = parse_hex_u64(bytes, 4, 20);
    let expected_crc = parse_hex_u32(bytes, 25, 33).expect("digits validated");
    let expected_adler = parse_hex_u32(bytes, 40, 48).expect("digits validated");

    let mut state = State::new();
    let mut selected_len = 0u64;
    for span in spans {
        state.update(&data[span.start..span.end]);
        selected_len += (span.end - span.start) as u64;
    }

    if length != selected_len {
        return Err(VerifyError::Length);
    }

    let actual = state.checksum();
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

fn parse_hex_u64(bytes: &[u8], start: usize, end: usize) -> u64 {
    let mut value = 0u64;
    for &byte in &bytes[start..end] {
        value = (value << 4) | u64::from(parse_hex_digit(byte).unwrap());
    }
    value
}
