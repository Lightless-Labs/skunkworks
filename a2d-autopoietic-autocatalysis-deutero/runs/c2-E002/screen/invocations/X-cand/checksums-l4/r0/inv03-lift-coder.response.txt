#[derive(Clone, Copy)]
struct State {
    crc: u32,
    a: u32,
    b: u32,
    f16: u32,
    f32_a: u32,
    f32_b: u32,
    f32_pending: Option<u8>,
    c16: u16,
}

impl State {
    fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
            f16: 0,
            f32_a: 0,
            f32_b: 0,
            f32_pending: None,
            c16: 0,
        }
    }

    fn update(&mut self, data: &[u8], variant: Crc16) {
        for &x in data {
            self.crc ^= u32::from(x);
            for _ in 0..8 {
                let low = self.crc & 1;
                self.crc >>= 1;
                if low != 0 {
                    self.crc ^= 0xedb8_8320;
                }
            }

            self.a = (self.a + u32::from(x)) % 65_521;
            self.b = (self.b + self.a) % 65_521;

            self.f16 = (self.f16 + u32::from(x)) % 255;
            self.f16_b_update();

            if let Some(high) = self.f32_pending.take() {
                self.f32_word(u16::from_be_bytes([high, x]));
            } else {
                self.f32_pending = Some(x);
            }

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    self.c16 ^= u16::from(x) << 8;
                    for _ in 0..8 {
                        let top = self.c16 & 0x8000;
                        self.c16 <<= 1;
                        if top != 0 {
                            self.c16 ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    self.c16 ^= u16::from(x);
                    for _ in 0..8 {
                        let low = self.c16 & 1;
                        self.c16 >>= 1;
                        if low != 0 {
                            self.c16 ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    fn f16_b_update(&mut self) {
        // The Fletcher-16 second accumulator is updated alongside the first.
        // Its value is kept in the high half of f16.
        let a = self.f16;
        let b = (self.f16 >> 16).wrapping_add(a) % 255;
        self.f16 = (b << 16) | a;
    }

    fn f32_word(&mut self, word: u16) {
        self.f32_a = (self.f32_a + u32::from(word)) % 65_535;
        self.f32_b = (self.f32_b + self.f32_a) % 65_535;
    }

    fn hashes(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    fn finish_f32(&self) -> u32 {
        let mut a = self.f32_a;
        let mut b = self.f32_b;
        if let Some(high) = self.f32_pending {
            a = (a + (u32::from(high) << 8)) % 65_535;
            b = (b + a) % 65_535;
        }
        (b << 16) | a
    }

    fn finish_c16(&self, variant: Crc16) -> u16 {
        match variant {
            Crc16::False => self.c16 ^ 0,
            Crc16::X25 => self.c16 ^ 0xffff,
            Crc16::Xmodem => self.c16,
        }
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (i, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: i });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.end,
            });
        }
        if i > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: i });
        }
        previous_end = span.end;
    }
    Ok(())
}

fn scan_spans(data: &[u8], spans: &[Span], state: &mut State, variant: Crc16) {
    for span in spans {
        state.update(&data[span.start..span.end], variant);
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut state = State::new();
    state.update(data, Crc16::Xmodem);
    state.hashes().crc32
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &x in data {
        a = (a + u32::from(x)) % 65_521;
        b = (b + a) % 65_521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = State::new();
    scan_spans(data, spans, &mut state, Crc16::Xmodem);
    Ok(state.hashes())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &x in data {
        a = (a + u32::from(x)) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut state = State::new();
    state.update(data, Crc16::Xmodem);
    state.finish_f32()
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = State::new();
    scan_spans(data, spans, &mut state, Crc16::Xmodem);
    Ok(state.finish_f32())
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut state = State::new();
    state.c16 = if variant == Crc16::False || variant == Crc16::X25 {
        0xffff
    } else {
        0
    };
    state.update(data, variant);
    state.finish_c16(variant)
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = State::new();
    state.c16 = if variant == Crc16::False || variant == Crc16::X25 {
        0xffff
    } else {
        0
    };
    scan_spans(data, spans, &mut state, variant);
    Ok(Multi {
        base: state.hashes(),
        fletcher16: ((state.f16 >> 16) << 8 | (state.f16 & 0xff)) as u16,
        fletcher32: state.finish_f32(),
        crc16: state.finish_c16(variant),
    })
}

impl Stream {
    pub fn new() -> Self {
        let state = State::new();
        Self {
            crc: state.crc,
            a: state.a,
            b: state.b,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        for &x in data {
            self.crc ^= u32::from(x);
            for _ in 0..8 {
                let low = self.crc & 1;
                self.crc >>= 1;
                if low != 0 {
                    self.crc ^= 0xedb8_8320;
                }
            }
            self.a = (self.a + u32::from(x)) % 65_521;
            self.b = (self.b + self.a) % 65_521;
        }
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        let fresh = Self::new();
        self.crc = fresh.crc;
        self.a = fresh.a;
        self.b = fresh.b;
    }

    pub fn snapshot(&self) -> String {
        let h = self.checksum();
        format!("c{:08x}:a{:08x}", h.crc32, h.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        let mut crc = 0u32;
        let mut adler = 0u32;
        for i in 1..9 {
            crc = (crc << 4) | u32::from(hex_value(bytes[i]).ok_or(SnapshotError::Hex { offset: i })?);
        }
        for i in 11..19 {
            adler = (adler << 4)
                | u32::from(hex_value(bytes[i]).ok_or(SnapshotError::Hex { offset: i })?);
        }
        Self::from_seed(crc, adler)
    }
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        _ => None,
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }
    let mut matrix = [0u32; 32];
    for (i, item) in matrix.iter_mut().enumerate() {
        let mut r = 1u32 << i;
        for _ in 0..8 {
            let low = r & 1;
            r >>= 1;
            if low != 0 {
                r ^= 0xedb8_8320;
            }
        }
        *item = r;
    }
    let mut n = right_len;
    let mut value = left;
    while n != 0 {
        if n & 1 != 0 {
            value = matrix_times(&matrix, value);
        }
        n >>= 1;
        if n != 0 {
            matrix = matrix_square(&matrix);
        }
    }
    value ^ right
}

fn matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut i = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[i];
        }
        vector >>= 1;
        i += 1;
    }
    sum
}

fn matrix_square(matrix: &[u32; 32]) -> [u32; 32] {
    let mut square = [0u32; 32];
    for i in 0..32 {
        square[i] = matrix_times(matrix, matrix[i]);
    }
    square
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (i, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: i })?;
    }
    for (i, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: i });
        }
    }
    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = State::new();
    let mut len = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        len += selected.len() as u64;
        state.update(selected, Crc16::Xmodem);
    }
    let hashes = state.hashes();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, hashes.crc32, hashes.adler32
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for range in [4..20, 25..33, 40..48] {
        for offset in range {
            if hex_value(bytes[offset]).is_none() {
                return Err(VerifyError::Hex { offset });
            }
        }
    }

    let len = parse_hex_u64(&bytes[4..20]);
    let expected_crc = parse_hex_u32(&bytes[25..33]);
    let expected_adler = parse_hex_u32(&bytes[40..48]);

    let mut state = State::new();
    let mut selected_len = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        selected_len += selected.len() as u64;
        state.update(selected, Crc16::Xmodem);
    }
    if len != selected_len {
        return Err(VerifyError::Length);
    }
    let actual = state.hashes();
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

fn parse_hex_u64(bytes: &[u8]) -> u64 {
    bytes.iter().fold(0u64, |value, &b| {
        (value << 4) | u64::from(hex_value(b).expect("validated hexadecimal digit"))
    })
}

fn parse_hex_u32(bytes: &[u8]) -> u32 {
    bytes.iter().fold(0u32, |value, &b| {
        (value << 4) | u32::from(hex_value(b).expect("validated hexadecimal digit"))
    })
}
