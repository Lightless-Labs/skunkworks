```rust
use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = n.checked_neg().ok_or(RationalError::Overflow)?;
        d = d.checked_neg().ok_or(RationalError::Overflow)?;
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128);
    let rn = n / g as i128;
    let rd = d / g as i128;
    if rn < i64::MIN as i128 || rn > i64::MAX as i128 || rd < 1 || rd > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational { num: rn as i64, den: rd as i64 })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        rational_from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        rational_from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = (self.num as i128).checked_neg().ok_or(RationalError::Overflow)?;
        rational_from_i128(n, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::Empty });
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
                }
                slash = Some(i);
            } else if b.is_ascii_digit() {
                continue;
            } else if b == b'-' && i == 0 {
                continue;
            } else {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
        }

        let num_end = slash.unwrap_or(bytes.len());
        let num_start = usize::from(bytes[0] == b'-');
        if num_start == num_end {
            return Err(RationalError::Parse { at: bytes.len(), kind: ParseKind::InvalidByte });
        }
        for i in num_start..num_end {
            if !bytes[i].is_ascii_digit() {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
        }

        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(RationalError::Parse { at: bytes.len(), kind: ParseKind::InvalidByte });
            }
            for i in s + 1..bytes.len() {
                if !bytes[i].is_ascii_digit() {
                    return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
                }
            }
        }

        if num_end - num_start >= 2 && bytes[num_start] == b'0' {
            return Err(RationalError::Parse { at: num_start + 1, kind: ParseKind::NonCanonical });
        }
        if bytes[0] == b'-' && num_end == 2 && bytes[1] == b'0' {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::NonCanonical });
        }
        if let Some(s) = slash {
            if bytes.len() - s - 1 >= 2 && bytes[s + 1] == b'0' {
                return Err(RationalError::Parse { at: s + 2, kind: ParseKind::NonCanonical });
            }
        }

        let negative = bytes[0] == b'-';
        let mut mag: u128 = 0;
        let limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        for &b in &bytes[num_start..num_end] {
            mag = mag.saturating_mul(10).saturating_add((b - b'0') as u128);
            if mag > limit {
                return Err(RationalError::Parse { at: 0, kind: ParseKind::OutOfRange });
            }
        }

        let mut den_mag = 1u128;
        if let Some(s) = slash {
            for &b in &bytes[s + 1..] {
                den_mag = den_mag.saturating_mul(10).saturating_add((b - b'0') as u128);
                if den_mag > i64::MAX as u128 {
                    return Err(RationalError::Parse { at: s + 1, kind: ParseKind::OutOfRange });
                }
            }
            if den_mag == 0 {
                return Err(RationalError::Parse { at: s + 1, kind: ParseKind::ZeroDenominator });
            }
            if den_mag == 1 {
                return Err(RationalError::Parse { at: s, kind: ParseKind::NonCanonical });
            }
            if mag == 0 {
                return Err(RationalError::Parse { at: s, kind: ParseKind::NonCanonical });
            }
            if gcd_u128(mag, den_mag) > 1 {
                return Err(RationalError::Parse { at: s, kind: ParseKind::NonCanonical });
            }
        }

        let n = if negative {
            if mag == 1u128 << 63 { i64::MIN } else { -(mag as i64) }
        } else {
            mag as i64
        };
        Ok(Rational { num: n, den: den_mag as i64 })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut out = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                out.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data: out })
    }

    fn eliminate(&self, pivot_cols: usize) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..pivot_cols {
            if p == matrix.rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for row in p..matrix.rows {
                let x = matrix.data[row * matrix.cols + col];
                if x.num == 0 {
                    continue;
                }
                let key = (x.den, x.num.unsigned_abs(), row);
                if chosen.map_or(true, |q| {
                    let y = matrix.data[q * matrix.cols + col];
                    key < (y.den, y.num.unsigned_abs(), q)
                }) {
                    chosen = Some(row);
                }
            }
            let Some(q) = chosen else { continue };

            if q != p {
                for c in 0..matrix.cols {
                    matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = matrix.data[p * matrix.cols + col];
            let mut normalized = Vec::with_capacity(matrix.cols);
            for c in 0..matrix.cols {
                let value = matrix.data[p * matrix.cols + c]
                    .div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic { row: p, col: c, step: p, source })?;
                normalized.push(value);
            }
            for c in 0..matrix.cols {
                matrix.data[p * matrix.cols + c] = normalized[c];
            }

            for row in 0..matrix.rows {
                if row == p {
                    continue;
                }
                let factor = matrix.data[row * matrix.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..matrix.cols {
                    let product = factor.mul(&matrix.data[p * matrix.cols + c])
                        .map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                    let value = matrix.data[row * matrix.cols + c].sub(&product)
                        .map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                    matrix.data[row * matrix.cols + c] = value;
                }
            }

            pivots.push(col);
            p += 1;
        }

        Ok(Reduction { matrix, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(if r == c { Rational { num: 1, den: 1 } } else { Rational { num: 0, den: 1 } });
            }
        }
        let augmented = Matrix { rows: n, cols: width, data };
        let reduced = augmented.eliminate(n)?;
        if reduced.pivots.len() < n {
            return Err(MatrixError::Singular { rank: reduced.pivots.len() });
        }
        let mut inverse = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                inverse.push(reduced.matrix.data[r * width + n + c]);
            }
        }
        Ok(Matrix { rows: n, cols: n, data: inverse })
    }
}
```
