use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_parts(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_parts(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_parts(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_parts(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_parts(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::from_parts(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_parts(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        if b.is_empty() {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::Empty });
        }

        let mut i = 0;
        if b[0] == b'-' {
            i = 1;
        }
        let num_start = i;
        while i < b.len() && b[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(RationalError::Parse { at: i, kind: if i == b.len() { ParseKind::InvalidByte } else { ParseKind::InvalidByte } });
        }
        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < b.len() {
            if b[i] != b'/' {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
            i += 1;
            den_start = Some(i);
            while i < b.len() && b[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
            den_end = Some(i);
        }
        if i < b.len() {
            return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
        }

        let negative = b[0] == b'-';
        let digits_start = num_start;
        let num_digits = num_end - digits_start;
        if num_digits >= 2 && b[digits_start] == b'0' {
            return Err(RationalError::Parse { at: digits_start + 1, kind: ParseKind::NonCanonical });
        }
        if negative && num_digits == 1 && b[digits_start] == b'0' {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::NonCanonical });
        }
        if let (Some(ds), Some(de)) = (den_start, den_end) {
            if de - ds >= 2 && b[ds] == b'0' {
                return Err(RationalError::Parse { at: ds + 1, kind: ParseKind::NonCanonical });
            }
        }

        let limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let mut mag = 0u128;
        for &ch in &b[digits_start..num_end] {
            mag = mag.saturating_mul(10).saturating_add((ch - b'0') as u128);
            if mag > limit {
                return Err(RationalError::Parse { at: 0, kind: ParseKind::OutOfRange });
            }
        }
        let n = if negative {
            if mag == (1u128 << 63) { i64::MIN } else { -(mag as i64) }
        } else {
            mag as i64
        };

        let (d, slash_at) = if let (Some(ds), Some(de)) = (den_start, den_end) {
            let mut dm = 0u128;
            for &ch in &b[ds..de] {
                dm = dm.saturating_mul(10).saturating_add((ch - b'0') as u128);
                if dm > i64::MAX as u128 {
                    return Err(RationalError::Parse { at: ds, kind: ParseKind::OutOfRange });
                }
            }
            (Some(dm as i64), Some(ds - 1))
        } else {
            (None, None)
        };

        if let (Some(d), Some(ds)) = (d, den_start) {
            if d == 0 {
                return Err(RationalError::Parse { at: ds, kind: ParseKind::ZeroDenominator });
            }
        }
        if let (Some(d), Some(slash)) = (d, slash_at) {
            if d == 1 {
                return Err(RationalError::Parse { at: slash, kind: ParseKind::NonCanonical });
            }
            if n == 0 {
                return Err(RationalError::Parse { at: slash, kind: ParseKind::NonCanonical });
            }
            if gcd_u128(n.unsigned_abs() as u128, d as u128) > 1 {
                return Err(RationalError::Parse { at: slash, kind: ParseKind::NonCanonical });
            }
        }

        Self::new(n, d.unwrap_or(1)).map_err(|_| RationalError::Overflow)
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let len = self.data.len();
        let mut data = Vec::with_capacity(len);
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    fn eliminate(&self, pivot_cols: usize) -> Result<(Self, Vec<usize>, Vec<(usize, usize)>), MatrixError> {
        let mut out = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..pivot_cols {
            if p == out.rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for row in p..out.rows {
                let v = out.data[row * out.cols + col];
                if v.num == 0 {
                    continue;
                }
                match chosen {
                    None => chosen = Some(row),
                    Some(old) => {
                        let a = (v.den, v.num.unsigned_abs(), row);
                        let ov = out.data[old * out.cols + col];
                        let b = (ov.den, ov.num.unsigned_abs(), old);
                        if a.cmp(&b) == Ordering::Less {
                            chosen = Some(row);
                        }
                    }
                }
            }
            let Some(q) = chosen else { continue };
            if q != p {
                for c in 0..out.cols {
                    out.data.swap(p * out.cols + c, q * out.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = out.data[p * out.cols + col];
            for c in 0..out.cols {
                let current = out.data[p * out.cols + c];
                let value = current.div(&pivot).map_err(|source| MatrixError::Arithmetic {
                    row: p, col: c, step: p, source,
                })?;
                out.data[p * out.cols + c] = value;
            }

            for row in 0..out.rows {
                if row == p {
                    continue;
                }
                let factor = out.data[row * out.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..out.cols {
                    let product = factor.mul(&out.data[p * out.cols + c]).map_err(|source| MatrixError::Arithmetic {
                        row, col: c, step: p, source,
                    })?;
                    let value = out.data[row * out.cols + c].sub(&product).map_err(|source| MatrixError::Arithmetic {
                        row, col: c, step: p, source,
                    })?;
                    out.data[row * out.cols + c] = value;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok((out, pivots, swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let (matrix, pivots, swaps) = self.eliminate(self.cols)?;
        Ok(Reduction { matrix, pivots, swaps })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        let zero = Rational { num: 0, den: 1 };
        let one = Rational { num: 1, den: 1 };
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(if r == c { one } else { zero });
            }
        }
        let augmented = Self { rows: n, cols: width, data };
        let (reduced, pivots, _) = augmented.eliminate(n)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }
        let mut result = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                result.push(reduced.data[r * width + n + c]);
            }
        }
        Ok(Self { rows: n, cols: n, data: result })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}
