#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|n| n == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|n| n == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut records = Vec::new();

    if i == bytes.len() {
        return Ok(records);
    }

    let mut record_index = 0usize;
    let mut row = Vec::new();
    let mut field = String::new();
    let mut field_started = false;
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: after quote
    let mut quote_open = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: i });
        }

        if state != 2 && b == b'\r' {
            if i + 1 < bytes.len() && bytes[i + 1] == b'\n' {
                row.push(field);
                records.push(row);
                row = Vec::new();
                field = String::new();
                field_started = false;
                state = 0;
                i += 2;
                record_index += 1;
                continue;
            }
            return Err(CsvError::BareCr { record: record_index, byte: i });
        }

        match state {
            0 => {
                if b == delimiter {
                    row.push(String::new());
                    field_started = false;
                    i += 1;
                } else if b == quote {
                    state = 2;
                    field_started = true;
                    quote_open = i;
                    i += 1;
                } else if b == b'\n' {
                    row.push(String::new());
                    records.push(row);
                    row = Vec::new();
                    field = String::new();
                    field_started = false;
                    state = 0;
                    i += 1;
                    record_index += 1;
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    field_started = true;
                    state = 1;
                    i += ch.len_utf8();
                }
            }
            1 => {
                if b == delimiter {
                    row.push(std::mem::take(&mut field));
                    field_started = false;
                    state = 0;
                    i += 1;
                } else if b == b'\n' {
                    row.push(std::mem::take(&mut field));
                    records.push(row);
                    row = Vec::new();
                    field = String::new();
                    field_started = false;
                    state = 0;
                    i += 1;
                    record_index += 1;
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote { record: record_index, byte: i });
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                }
            }
            2 => {
                if b == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = 3;
                        i += 1;
                    }
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                }
            }
            _ => {
                if b == delimiter {
                    row.push(std::mem::take(&mut field));
                    field_started = false;
                    state = 0;
                    i += 1;
                } else if b == b'\n' {
                    row.push(std::mem::take(&mut field));
                    records.push(row);
                    row = Vec::new();
                    field = String::new();
                    field_started = false;
                    state = 0;
                    i += 1;
                    record_index += 1;
                } else {
                    return Err(CsvError::TrailingByte { record: record_index, byte: i });
                }
            }
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: quote_open });
    }

    if !row.is_empty() || field_started || !field.is_empty() || state == 3 {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    for second in 0..header.len() {
        for first in 0..second {
            if header[first] == header[second] {
                return Some(CsvError::DuplicateHeader {
                    name: header[second].clone(),
                    first,
                    second,
                });
            }
        }
    }
    None
}

fn table_from_records(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let mut records = records.into_iter();
    let header = if config.headers { records.next() } else { None };
    let rows: Vec<Vec<String>> = records.collect();

    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(e) = duplicate_header(h) {
                return Err(e);
            }
        }
        for (n, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: n + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table { header, rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), cached: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = if b[0] == b'-' { 1 } else { 0 };
    if start == b.len() { return false; }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    (b'1'..=b'9').contains(&b[start]) && b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = if b[0] == b'-' { 1 } else { 0 };
    let dot = match b[start..].iter().position(|&c| c == b'.') {
        Some(p) => start + p,
        None => return false,
    };
    if dot + 3 != b.len() || !b[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    if dot == start { return false; }
    let int = &b[start..dot];
    if int.len() > 1 && int[0] == b'0' { return false; }
    if int == b"0" {
        return !(start == 1 && b[0] == b'-' && b[dot + 1..] == *b"00");
    }
    (b'1'..=b'9').contains(&int[0]) && int[1..].iter().all(u8::is_ascii_digit)
}

fn eval_cell(s: &str, kind: ColumnType, nullable: bool) -> Result<Value, CellErrorKind> {
    if nullable && s.is_empty() {
        return Ok(Value::Null);
    }
    if s.is_empty() && kind != ColumnType::Text {
        return Err(CellErrorKind::Empty);
    }
    match kind {
        ColumnType::Text => Ok(Value::Text(s.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(s) { return Err(CellErrorKind::InvalidInteger); }
            s.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(s) { return Err(CellErrorKind::InvalidDecimal); }
            let neg = s.starts_with('-');
            let body = if neg { &s[1..] } else { s };
            let mut parts = body.split('.');
            let whole = parts.next().unwrap().parse::<i128>().unwrap();
            let frac = parts.next().unwrap().parse::<i128>().unwrap();
            let magnitude = whole * 100 + frac;
            let signed = if neg { -magnitude } else { magnitude };
            i64::try_from(signed).map(Value::Decimal).map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut stop = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            if stop {
                output.push(None);
                continue;
            }
            let col = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let evaluated = match col {
                None => Err((None, CellErrorKind::UnknownColumn)),
                Some(index) => match row.get(index) {
                    None => Err((Some(index), CellErrorKind::MissingCell)),
                    Some(cell) => eval_cell(cell, spec.kind, spec.nullable)
                        .map_err(|kind| (Some(index), kind)),
                },
            };
            match evaluated {
                Ok(value) => output.push(Some(value)),
                Err((column, kind)) => {
                    output.push(None);
                    result.errors.push(CellError { row: row_index, spec: spec_index, column, kind });
                    if policy != ErrorPolicy::Collect {
                        stop = true;
                    }
                    if policy == ErrorPolicy::Stop {
                        result.rows.push(output);
                        return result;
                    }
                }
            }
        }
        result.rows.push(output);
    }
    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut all_records: Vec<(usize, &Vec<String>)> = Vec::new();
    if let Some(header) = &table.header {
        all_records.push((0, header));
    }
    let row_start = if table.header.is_some() { 1 } else { 0 };
    for (i, row) in table.rows.iter().enumerate() {
        all_records.push((row_start + i, row));
    }

    for (record, row) in &all_records {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul { record: *record, column, byte });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in &all_records {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record: *record });
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();

    if config.mode == WriteMode::Preserve {
        if let Some(header) = &table.header {
            output_records.push(header.clone());
        }
        output_records.extend(table.rows.iter().cloned());
    } else if let Some(header) = &table.header {
        let mut selected: Vec<(String, usize)> = Vec::new();
        for (i, name) in header.iter().enumerate() {
            let chosen = match table.duplicates {
                DuplicatePolicy::KeepLast => !header[i + 1..].iter().any(|n| n == name),
                DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                    !header[..i].iter().any(|n| n == name)
                }
            };
            if chosen {
                selected.push((name.clone(), i));
            }
        }
        if table.duplicates == DuplicatePolicy::Reject {
            if let Some(CsvError::DuplicateHeader { name, first, second }) = duplicate_header(header) {
                return Err(WriteError::DuplicateHeader { name, first, second });
            }
        }
        selected.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
        let indices: Vec<usize> = selected.iter().map(|x| x.1).collect();
        output_records.push(selected.into_iter().map(|x| x.0).collect());
        let mut rows: Vec<Vec<String>> = table.rows.iter()
            .map(|row| indices.iter().map(|&i| row[i].clone()).collect())
            .collect();
        rows.sort_by(|a, b| cmp_rows(a, b));
        output_records.extend(rows);
    } else {
        let mut rows = table.rows.clone();
        rows.sort_by(|a, b| cmp_rows(a, b));
        output_records.extend(rows);
    }

    if output_records.is_empty() {
        return Ok(String::new());
    }

    let canonical = config.mode == WriteMode::Canonical;
    let terminator: &[u8] = if canonical || config.line_ending == LineEnding::Lf {
        b"\n"
    } else {
        b"\r\n"
    };
    let mut out = String::new();

    for (record_index, row) in output_records.iter().enumerate() {
        if record_index > 0 {
            out.push_str(std::str::from_utf8(terminator).unwrap());
        }
        for (column_index, field) in row.iter().enumerate() {
            if column_index > 0 {
                out.push(config.delimiter as char);
            }
            let bytes = field.as_bytes();
            let sole_empty = row.len() == 1 && field.is_empty();
            let first_bom = record_index == 0 && column_index == 0 && field.starts_with('\u{feff}');
            let quote_field = bytes.contains(&config.delimiter)
                || bytes.contains(&config.quote)
                || bytes.contains(&b'\r')
                || bytes.contains(&b'\n')
                || field.starts_with(' ') || field.starts_with('\t')
                || field.ends_with(' ') || field.ends_with('\t')
                || first_bom
                || sole_empty;
            if quote_field {
                out.push(config.quote as char);
                for &b in bytes {
                    if b == config.quote {
                        out.push(config.quote as char);
                    }
                    out.push(b as char);
                }
                out.push(config.quote as char);
            } else {
                out.push_str(field);
            }
        }
    }
    if canonical || config.final_terminator {
        out.push_str(std::str::from_utf8(terminator).unwrap());
    }
    Ok(out)
}

fn cmp_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let ordering = x.as_bytes().cmp(y.as_bytes());
        if ordering != std::cmp::Ordering::Equal {
            return ordering;
        }
    }
    a.len().cmp(&b.len())
}
