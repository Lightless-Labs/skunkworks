#[derive(Debug, Clone, Copy, PartialEq, Eq)]

    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|s| s == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut records = Vec::new();
    if i == bytes.len() {
        return Ok(records);
    }

    let mut record = Vec::new();
    let mut field = String::new();
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut field_started = false;
    let mut quote_start = 0usize;
    let mut record_started = false;
    let mut record_index = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: i });
        }

        if !in_quotes && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record: record_index, byte: i });
        }

        if in_quotes {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let start = i;
                if b.is_ascii() {
                    field.push(b as char);
                    i += 1;
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                }
                let _ = start;
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                after_quote = false;
                field_started = false;
                record_started = true;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                after_quote = false;
                field_started = false;
                record_started = false;
                record_index += 1;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte { record: record_index, byte: i });
        }

        if b == delimiter {
            record.push(std::mem::take(&mut field));
            field_started = false;
            record_started = true;
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            field_started = false;
            record_started = false;
            record_index += 1;
            i += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote { record: record_index, byte: i });
            }
            in_quotes = true;
            quote_start = i;
            field_started = true;
            record_started = true;
            i += 1;
        } else {
            field_started = true;
            record_started = true;
            if b.is_ascii() {
                field.push(b as char);
                i += 1;
            } else {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                i += ch.len_utf8();
            }
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: quote_start });
    }
    if record_started || field_started || after_quote || !record.is_empty() {
        record.push(field);
        records.push(record);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(prior, _)| *prior == name) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first: *first,
                second,
            });
        }
        seen.push((name, second));
    }
    None
}

fn table_from_records(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut iter = records.into_iter();
            (iter.next(), iter.collect::<Vec<_>>())
        }
    } else {
        (None, records)
    };

    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header(h) {
                return Err(error);
            }
        }
        for (index, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table { header, rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    buffer: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, buffer: String::new(), cached: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.buffer.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.buffer, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.buffer.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    (b'1'..=b'9').contains(&b[start]) && b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if b.len() < start + 4 { return false; }
    let dot = b.len() - 3;
    if b[dot] != b'.' || !b[dot + 1..].iter().all(u8::is_ascii_digit) { return false; }
    let integral = &b[start..dot];
    if integral.is_empty() { return false; }
    let valid = if integral == b"0" {
        true
    } else {
        (b'1'..=b'9').contains(&integral[0]) && integral[1..].iter().all(u8::is_ascii_digit)
    };
    if !valid { return false; }
    !(b.first() == Some(&b'-') && integral == b"0" && &b[dot + 1..] == b"00")
}

fn evaluate_cell(s: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if s.is_empty() {
        if spec.nullable { return Ok(Value::Null); }
        if spec.kind != ColumnType::Text { return Err(CellErrorKind::Empty); }
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(s.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(s) { return Err(CellErrorKind::InvalidInteger); }
            s.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(s) { return Err(CellErrorKind::InvalidDecimal); }
            let negative = s.starts_with('-');
            let unsigned = if negative { &s[1..] } else { s };
            let dot = unsigned.len() - 3;
            let whole: i128 = unsigned[..dot].parse().unwrap();
            let frac: i128 = unsigned[dot + 1..].parse().unwrap();
            let magnitude = whole * 100 + frac;
            let signed = if negative { -magnitude } else { magnitude };
            i64::try_from(signed).map(Value::Decimal).map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut output = TypedResult { rows: Vec::new(), errors: Vec::new() };
    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut out_row = vec![None; columns.len()];
        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (column, result) = match resolved {
                None => (None, Err(CellErrorKind::UnknownColumn)),
                Some(index) => match row.get(index) {
                    None => (Some(index), Err(CellErrorKind::MissingCell)),
                    Some(cell) => (Some(index), evaluate_cell(cell, spec)),
                },
            };
            match result {
                Ok(value) => out_row[spec_index] = Some(value),
                Err(kind) => {
                    output.errors.push(CellError { row: row_index, spec: spec_index, column, kind });
                    if policy == ErrorPolicy::Stop {
                        output.rows.push(out_row);
                        break 'rows;
                    }
                    if policy == ErrorPolicy::FirstPerRow {
                        output.rows.push(out_row);
                        continue 'rows;
                    }
                }
            }
        }
        output.rows.push(out_row);
    }
    output
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn write_duplicate(header: &[String]) -> Option<WriteError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(prior, _)| *prior == name) {
            return Some(WriteError::DuplicateHeader {
                name: name.clone(), first: *first, second,
            });
        }
        seen.push((name, second));
    }
    None
}

fn row_cmp(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let cmp = x.as_bytes().cmp(y.as_bytes());
        if cmp != std::cmp::Ordering::Equal { return cmp; }
    }
    a.len().cmp(&b.len())
}

fn encode_field(field: &str, delimiter: u8, quote: u8, first: bool, sole_empty: bool) -> String {
    let bytes = field.as_bytes();
    let edge_space = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
    let bom = first && field.starts_with('\u{feff}');
    let must_quote = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || edge_space
        || bom
        || sole_empty;
    if !must_quote { return field.to_string(); }

    let mut result = String::new();
    result.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            result.push(quote as char);
            result.push(quote as char);
        } else {
            result.push(ch);
        }
    }
    result.push(quote as char);
    result
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original_records: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header { original_records.push(header); }
    original_records.extend(table.rows.iter());

    let mut record_base = 0usize;
    for row in &original_records {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|b| *b == 0) {
                return Err(WriteError::Nul { record: record_base, column, byte });
            }
        }
        record_base += 1;
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1, expected: header.len(), actual: row.len(),
                });
            }
        }
    }

    for (record, row) in original_records.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    if config.mode == WriteMode::Canonical
        && table.header.is_some()
        && table.duplicates == DuplicatePolicy::Reject
    {
        if let Some(header) = &table.header {
            if let Some(error) = write_duplicate(header) { return Err(error); }
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header { output_records.push(header.clone()); }
            output_records.extend(table.rows.clone());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut retained: Vec<usize> = Vec::new();
                for (i, name) in header.iter().enumerate() {
                    let chosen = match table.duplicates {
                        DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name) == Some(i),
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                            header.iter().position(|s| s == name) == Some(i)
                        }
                    };
                    if chosen { retained.push(i); }
                }
                retained.sort_by(|a, b| header[*a].as_bytes().cmp(header[*b].as_bytes()));
                output_records.push(retained.iter().map(|i| header[*i].clone()).collect());
                let mut projected: Vec<Vec<String>> = table.rows.iter().map(|row| {
                    retained.iter().map(|i| row[*i].clone()).collect()
                }).collect();
                projected.sort_by(|a, b| row_cmp(a, b));
                output_records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| row_cmp(a, b));
                output_records.extend(rows);
            }
        }
    }

    if output_records.is_empty() { return Ok(String::new()); }

    let mut result = String::new();
    let terminator = match (config.mode, config.line_ending) {
        (WriteMode::Canonical, _) | (_, LineEnding::Lf) => "\n",
        (_, LineEnding::CrLf) => "\r\n",
    };
    for (record_index, row) in output_records.iter().enumerate() {
        if record_index > 0 { result.push_str(terminator); }
        let sole_empty = row.len() == 1 && row[0].is_empty();
        for (column_index, field) in row.iter().enumerate() {
            if column_index > 0 { result.push(config.delimiter as char); }
            result.push_str(&encode_field(
                field,
                config.delimiter,
                config.quote,
                record_index == 0 && column_index == 0,
                sole_empty,
            ));
        }
    }
    let final_terminator = match config.mode {
        WriteMode::Canonical => true,
        WriteMode::Preserve => config.final_terminator,
    };
    if final_terminator { result.push_str(terminator); }
    Ok(result)
}
