use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let high = hex_value(bytes[i + 1]);
            let low = hex_value(bytes[i + 2]);
            match (high, low) {
                (Some(high), Some(low)) => {
                    decoded.push((high << 4) | low);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let decoded_at = error.valid_up_to();
            let original_at = origins.get(decoded_at).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut encoded = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        let safe = matches!(
            byte,
            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~'
        ) || match component {
            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => matches!(byte, b'/' | b'?'),
            Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
        };

        if safe {
            encoded.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            encoded.push('%');
            encoded.push(HEX[(byte >> 4) as usize] as char);
            encoded.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    encoded
}

fn validate_component(
    component: Component,
    value: &str,
    offset: usize,
) -> Result<(), ParseError> {
    match decode_component(component, value) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: offset + at }),
        Err(CodecError::InvalidUtf8 { at }) => {
            Err(ParseError::InvalidUtf8 { at: offset + at })
        }
    }
}

fn parse_port(port: &str, start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }

    for (index, byte) in port.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + index });
        }
    }

    let bytes = port.as_bytes();
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }

    let mut value: u32 = 0;
    for byte in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
        {
            Some(value) => value,
            None => return Err(ParseError::InvalidPort { at: start }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }

    if value == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }

    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(colon) => colon,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|relative| authority_start + relative)
        .unwrap_or(input.len());

    let authority = &input[authority_start..authority_end];
    let authority_bytes = authority.as_bytes();
    let mut at_signs = authority_bytes
        .iter()
        .enumerate()
        .filter(|(_, byte)| **byte == b'@')
        .map(|(index, _)| index);

    let first_at = at_signs.next();
    let second_at = at_signs.next();
    if let Some(second) = second_at {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + second,
        });
    }

    let (userinfo, hostport_start) = if let Some(user_at) = first_at {
        if user_at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + user_at,
            });
        }
        (
            Some(&authority[..user_at]),
            user_at + 1,
        )
    } else {
        (None, 0)
    };

    let hostport = &authority[hostport_start..];
    let hostport_offset = authority_start + hostport_start;
    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let closing = hostport.as_bytes()[1..]
            .iter()
            .position(|&byte| byte == b']')
            .map(|relative| relative + 1);
        let closing = match closing {
            Some(closing) => closing,
            None => {
                return Err(ParseError::InvalidIpv6 {
                    at: hostport_offset,
                })
            }
        };
        let interior = &hostport[1..closing];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 {
                at: hostport_offset,
            });
        }
        ipv6 = true;
        let suffix = &hostport[closing + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(
                &suffix[1..],
                hostport_offset + closing + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_offset + closing + 1,
            });
        };
        (interior.to_ascii_lowercase(), port)
    } else {
        let host_colon = hostport.as_bytes().iter().position(|&byte| byte == b':');
        let (host, port) = if let Some(host_colon) = host_colon {
            let port = parse_port(
                &hostport[host_colon + 1..],
                hostport_offset + host_colon + 1,
            )?;
            (&hostport[..host_colon], Some(port))
        } else {
            (hostport, None)
        };

        if host.is_empty() {
            return Err(ParseError::InvalidHost {
                at: hostport_offset,
            });
        }
        for (index, byte) in host.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_offset + index,
                });
            }
        }

        (host.to_ascii_lowercase(), port)
    };

    let suffix = &input[authority_end..];
    let fragment_position = suffix.find('#');
    let before_fragment = fragment_position
        .map(|position| &suffix[..position])
        .unwrap_or(suffix);
    let fragment = fragment_position.map(|position| &suffix[position + 1..]);

    let query_position = before_fragment.find('?');
    let path = query_position
        .map(|position| &before_fragment[..position])
        .unwrap_or(before_fragment);
    let query = query_position.map(|position| &before_fragment[position + 1..]);

    let path_offset = authority_end;
    let query_offset = query_position.map(|position| authority_end + position);
    let fragment_offset = fragment_position.map(|position| authority_end + position + 1);

    if let Some(userinfo) = userinfo {
        validate_component(
            Component::UserInfo,
            userinfo,
            authority_start,
        )?;
    }
    if !path.is_empty() {
        validate_component(Component::Path, path, path_offset)?;
    }
    if let (Some(query), Some(query_offset)) = (query, query_offset) {
        validate_component(Component::Query, query, query_offset + 1)?;
    }
    if let (Some(fragment), Some(fragment_offset)) = (fragment, fragment_offset) {
        validate_component(Component::Fragment, fragment, fragment_offset)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo: userinfo.map(str::to_owned),
        host,
        port,
        path: if path.is_empty() {
            "/".to_owned()
        } else {
            path.to_owned()
        },
        query: query.map(str::to_owned),
        fragment: fragment.map(str::to_owned),
        ipv6,
    })
}
