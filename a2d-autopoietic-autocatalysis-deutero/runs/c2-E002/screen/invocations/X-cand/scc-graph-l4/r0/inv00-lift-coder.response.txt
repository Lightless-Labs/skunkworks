pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EdgeStatus {
    Duplicate,
    Internal,
    Link,
    Merge,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Step {
    pub edge_index: usize,
    pub status: EdgeStatus,
    pub before: (usize, usize),
    pub after: (usize, usize),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub edge_index: usize,
    pub before: Vec<usize>,
    pub after: usize,
    pub members: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Batch {
    pub steps: Vec<Step>,
    pub merges: Vec<Merge>,
    pub graph: Graph,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

fn validate(n: usize, edges: &[(usize, usize)]) -> Result<(), GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }
    Ok(())
}

fn finish_graph(
    n: usize,
    edges: &[(usize, usize)],
    components: Vec<Vec<usize>>,
) -> Graph {
    let count = components.len();
    let mut component_of = vec![0; n];
    for (id, members) in components.iter().enumerate() {
        for &v in members {
            component_of[v] = id;
        }
    }

    let mut cyclic: Vec<bool> = components.iter().map(|c| c.len() > 1).collect();
    let mut internal_edges = vec![0; count];
    let mut arc_data = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (i, &(from, to)) in edges.iter().enumerate() {
        let a = component_of[from];
        let b = component_of[to];
        if a == b {
            internal_edges[a] += 1;
            if from == to {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_data.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            entry.1 = entry.1.min(i);
        }
    }

    let mut arcs: Vec<Arc> = arc_data
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::<usize>::new(); count];
    let mut indegree = vec![0usize; count];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    let mut ready = std::collections::BTreeSet::<(usize, usize, usize)>::new();
    for id in 0..count {
        if indegree[id] == 0 {
            let smallest = components[id][0];
            ready.insert((internal_edges[id], smallest, id));
        }
    }

    let mut topo = Vec::with_capacity(count);
    let mut depth = vec![0usize; count];
    while let Some(&(internal, smallest, id)) = ready.iter().next_back() {
        ready.remove(&(internal, smallest, id));
        topo.push(id);
        for &to in &outgoing[id] {
            depth[to] = depth[to].max(depth[id] + 1);
            indegree[to] -= 1;
            if indegree[to] == 0 {
                ready.insert((internal_edges[to], components[to][0], to));
            }
        }
    }

    Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    }
}

fn analyze_valid(n: usize, edges: &[(usize, usize)]) -> Graph {
    let mut adjacency = vec![Vec::<usize>::new(); n];
    let mut reverse = vec![Vec::<usize>::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse[to].push(from);
    }

    let mut seen = vec![false; n];
    let mut order = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((v, next)) = stack.pop() {
            if next < adjacency[v].len() {
                stack.push((v, next + 1));
                let to = adjacency[v][next];
                if !seen[to] {
                    seen[to] = true;
                    stack.push((to, 0));
                }
            } else {
                order.push(v);
            }
        }
    }

    seen.fill(false);
    let mut components = Vec::<Vec<usize>>::new();
    for &root in order.iter().rev() {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut members = Vec::new();
        let mut stack = vec![root];
        while let Some(v) = stack.pop() {
            members.push(v);
            for &to in &reverse[v] {
                if !seen[to] {
                    seen[to] = true;
                    stack.push(to);
                }
            }
        }
        members.sort_unstable();
        components.push(members);
    }

    components.sort_by(|a, b| b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0])));
    finish_graph(n, edges, components)
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    validate(n, edges)?;
    Ok(analyze_valid(n, edges))
}

fn shortest_route(graph: &Graph, start: usize, end: usize) -> Option<Vec<usize>> {
    let n = graph.components.len();
    let mut outgoing = vec![Vec::<usize>::new(); n];
    for arc in &graph.arcs {
        outgoing[arc.from].push(arc.to);
    }
    for neighbors in &mut outgoing {
        neighbors.sort_unstable();
        neighbors.dedup();
    }

    let mut seen = vec![false; n];
    let mut parent = vec![usize::MAX; n];
    let mut queue = std::collections::VecDeque::new();
    seen[start] = true;
    queue.push_back(start);

    while let Some(v) = queue.pop_front() {
        if v == end {
            break;
        }
        for &to in &outgoing[v] {
            if !seen[to] {
                seen[to] = true;
                parent[to] = v;
                queue.push_back(to);
            }
        }
    }

    if !seen[end] {
        return None;
    }
    let mut route = Vec::new();
    let mut current = end;
    loop {
        route.push(current);
        if current == start {
            break;
        }
        current = parent[current];
    }
    route.reverse();
    Some(route)
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        if let Some(route) = shortest_route(self, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_route(self, b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

#[derive(Clone, Debug)]
pub struct Incremental {
    n: usize,
    edges: Vec<(usize, usize)>,
    graph: Graph,
    history: Vec<Merge>,
}

impl Incremental {
    pub fn new(n: usize, edges: &[(usize, usize)]) -> Result<Self, GraphError> {
        let graph = analyze(n, edges)?;
        Ok(Self {
            n,
            edges: edges.to_vec(),
            graph,
            history: Vec::new(),
        })
    }

    pub fn snapshot(&self) -> Graph {
        self.graph.clone()
    }

    pub fn history(&self) -> Vec<Merge> {
        self.history.clone()
    }

    pub fn insert(&mut self, batch_edges: &[(usize, usize)]) -> Result<Batch, GraphError> {
        if self.edges.len() + batch_edges.len() > MAX_EDGES {
            return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
        }
        for (i, &(from, to)) in batch_edges.iter().enumerate() {
            if from >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 0,
                    vertex: from,
                });
            }
            if to >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 1,
                    vertex: to,
                });
            }
        }

        let mut steps = Vec::with_capacity(batch_edges.len());
        let mut merges = Vec::new();

        for &(from, to) in batch_edges {
            let edge_index = self.edges.len();
            let before = (
                self.graph.component_of[from],
                self.graph.component_of[to],
            );
            let duplicate = self.edges.iter().any(|&(x, y)| x == from && y == to);
            self.edges.push((from, to));
            self.graph = analyze_valid(self.n, &self.edges);
            let after = (
                self.graph.component_of[from],
                self.graph.component_of[to],
            );

            let status = if duplicate {
                EdgeStatus::Duplicate
            } else if before.0 == before.1 {
                EdgeStatus::Internal
            } else if after.0 == after.1 {
                EdgeStatus::Merge
            } else {
                EdgeStatus::Link
            };

            if status == EdgeStatus::Merge {
                let mut members = self.graph.components[after.0].clone();
                members.sort_unstable();

                let mut before_ids = Vec::new();
                for (id, component) in self.graph.components.iter().enumerate() {
                    if component.iter().any(|&v| {
                        self.graph.component_of[v] == after.0
                            && (before.0 == id || before.1 == id)
                    }) {
                        before_ids.push(id);
                    }
                }

                // Recover the pre-insertion component IDs from the pre-insertion graph.
                // IDs other than the merged component are unchanged except for canonical
                // reordering, so identify all old components by their vertices.
                let mut old_graph = analyze_valid(self.n, &self.edges[..edge_index]);
                let mut old_ids = vec![false; old_graph.components.len()];
                old_ids[before.0] = true;
                old_ids[before.1] = true;

                let mut united = Vec::new();
                for (old_id, old_component) in old_graph.components.iter().enumerate() {
                    if old_ids[old_id] {
                        united.extend_from_slice(old_component);
                    }
                }
                united.sort_unstable();
                united.dedup();

                let mut all_before = Vec::new();
                for old_id in 0..old_graph.components.len() {
                    if old_graph.components[old_id]
                        .iter()
                        .any(|v| united.binary_search(v).is_ok())
                    {
                        all_before.push(old_id);
                    }
                }

                // An inserted edge can merge components beyond its two endpoint SCCs.
                // Determine the full set by checking each old component's membership
                // in the newly formed component.
                all_before.clear();
                for (old_id, old_component) in old_graph.components.iter().enumerate() {
                    if old_component
                        .iter()
                        .any(|&v| self.graph.component_of[v] == after.0)
                    {
                        all_before.push(old_id);
                    }
                }

                before_ids.clear();
                let event = Merge {
                    edge_index,
                    before: all_before,
                    after: after.0,
                    members,
                };
                merges.push(event.clone());
                self.history.push(event);
            }

            steps.push(Step {
                edge_index,
                status,
                before,
                after,
            });
        }

        Ok(Batch {
            steps,
            merges,
            graph: self.snapshot(),
        })
    }
}
