use std::collections::{BTreeMap, BTreeSet, BinaryHeap, VecDeque};

pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EdgeStatus {
    Duplicate,
    Internal,
    Link,
    Merge,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Step {
    pub edge_index: usize,
    pub status: EdgeStatus,
    pub before: (usize, usize),
    pub after: (usize, usize),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub edge_index: usize,
    pub before: Vec<usize>,
    pub after: usize,
    pub members: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Batch {
    pub steps: Vec<Step>,
    pub merges: Vec<Merge>,
    pub graph: Graph,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

fn validate(n: usize, edges: &[(usize, usize)]) -> Result<(), GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }
    Ok(())
}

fn reachable(adj: &[Vec<usize>], start: usize, goal: usize) -> bool {
    if start == goal {
        return true;
    }
    let mut seen = vec![false; adj.len()];
    let mut stack = vec![start];
    seen[start] = true;
    while let Some(v) = stack.pop() {
        for &w in &adj[v] {
            if w == goal {
                return true;
            }
            if !seen[w] {
                seen[w] = true;
                stack.push(w);
            }
        }
    }
    false
}

fn shortest_lex_route(adj: &[Vec<usize>], start: usize, goal: usize) -> Vec<usize> {
    let n = adj.len();
    let mut dist = vec![usize::MAX; n];
    let mut queue = VecDeque::new();
    dist[goal] = 0;
    queue.push_back(goal);

    // Compute distances to goal by traversing the reversed graph.
    let mut rev = vec![Vec::new(); n];
    for (v, outs) in adj.iter().enumerate() {
        for &w in outs {
            rev[w].push(v);
        }
    }
    while let Some(v) = queue.pop_front() {
        for &p in &rev[v] {
            if dist[p] == usize::MAX {
                dist[p] = dist[v] + 1;
                queue.push_back(p);
            }
        }
    }

    let mut route = vec![start];
    let mut cur = start;
    while cur != goal {
        let next = adj[cur]
            .iter()
            .copied()
            .filter(|&v| dist[v] != usize::MAX && dist[v] + 1 == dist[cur])
            .min()
            .expect("reachable route must have a next component");
        route.push(next);
        cur = next;
    }
    route
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    validate(n, edges)?;

    let mut adj = vec![Vec::new(); n];
    let mut rev = vec![Vec::new(); n];
    for &(u, v) in edges {
        adj[u].push(v);
        rev[v].push(u);
    }

    // Iterative Kosaraju first pass.
    let mut seen = vec![false; n];
    let mut order = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((v, next_i)) = stack.pop() {
            if next_i < adj[v].len() {
                stack.push((v, next_i + 1));
                let w = adj[v][next_i];
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                order.push(v);
            }
        }
    }

    let mut raw_components = Vec::<Vec<usize>>::new();
    let mut assigned = vec![false; n];
    for &root in order.iter().rev() {
        if assigned[root] {
            continue;
        }
        assigned[root] = true;
        let mut members = Vec::new();
        let mut stack = vec![root];
        while let Some(v) = stack.pop() {
            members.push(v);
            for &w in &rev[v] {
                if !assigned[w] {
                    assigned[w] = true;
                    stack.push(w);
                }
            }
        }
        members.sort_unstable();
        raw_components.push(members);
    }

    raw_components.sort_by(|a, b| {
        b.len().cmp(&a.len()).then_with(|| a[0].cmp(&b[0]))
    });

    let components = raw_components;
    let mut component_of = vec![0usize; n];
    for (cid, members) in components.iter().enumerate() {
        for &v in members {
            component_of[v] = cid;
        }
    }

    let c = components.len();
    let mut cyclic: Vec<bool> = components.iter().map(|m| m.len() > 1).collect();
    let mut internal_edges = vec![0usize; c];
    let mut arc_data = BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (i, &(u, v)) in edges.iter().enumerate() {
        let a = component_of[u];
        let b = component_of[v];
        if a == b {
            internal_edges[a] += 1;
            if u == v {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_data.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            if i < entry.1 {
                entry.1 = i;
            }
        }
    }

    let mut arcs: Vec<Arc> = arc_data
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut dag = vec![Vec::new(); c];
    let mut indegree = vec![0usize; c];
    for arc in &arcs {
        dag[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    // BinaryHeap is a max heap; tie-break by greatest smallest member.
    let mut ready = BinaryHeap::<(usize, usize, usize)>::new();
    for cid in 0..c {
        if indegree[cid] == 0 {
            ready.push((internal_edges[cid], components[cid][0], cid));
        }
    }

    let mut topo = Vec::with_capacity(c);
    let mut depth = vec![0usize; c];
    while let Some((_, _, cid)) = ready.pop() {
        topo.push(cid);
        for &to in &dag[cid] {
            depth[to] = depth[to].max(depth[cid] + 1);
            indegree[to] -= 1;
            if indegree[to] == 0 {
                ready.push((internal_edges[to], components[to][0], to));
            }
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        let mut adj = vec![Vec::new(); self.components.len()];
        for arc in &self.arcs {
            adj[arc.from].push(arc.to);
        }
        for outs in &mut adj {
            outs.sort_unstable();
            outs.dedup();
        }

        if reachable(&adj, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route: shortest_lex_route(&adj, a, b),
            });
        }
        if reachable(&adj, b, a) {
            let mut route = shortest_lex_route(&adj, b, a);
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

#[derive(Clone, Debug)]
pub struct Incremental {
    n: usize,
    edges: Vec<(usize, usize)>,
    history: Vec<Merge>,
}

impl Incremental {
    pub fn new(n: usize, edges: &[(usize, usize)]) -> Result<Self, GraphError> {
        validate(n, edges)?;
        Ok(Self {
            n,
            edges: edges.to_vec(),
            history: Vec::new(),
        })
    }

    pub fn snapshot(&self) -> Graph {
        analyze(self.n, &self.edges).expect("retained edges are always valid")
    }

    pub fn history(&self) -> Vec<Merge> {
        self.history.clone()
    }

    pub fn insert(&mut self, edges: &[(usize, usize)]) -> Result<Batch, GraphError> {
        if self.edges.len() + edges.len() > MAX_EDGES {
            return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
        }
        for (i, &(from, to)) in edges.iter().enumerate() {
            if from >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 0,
                    vertex: from,
                });
            }
            if to >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: i,
                    side: 1,
                    vertex: to,
                });
            }
        }

        let mut steps = Vec::with_capacity(edges.len());
        let mut merges = Vec::new();
        let mut seen_pairs: BTreeSet<(usize, usize)> = self.edges.iter().copied().collect();

        for &(u, v) in edges {
            let edge_index = self.edges.len();
            let before_graph = self.snapshot();
            let before = (
                before_graph.component_of[u],
                before_graph.component_of[v],
            );
            let duplicate = seen_pairs.contains(&(u, v));

            self.edges.push((u, v));
            seen_pairs.insert((u, v));

            let after_graph = self.snapshot();
            let after = (after_graph.component_of[u], after_graph.component_of[v]);

            let status = if duplicate {
                EdgeStatus::Duplicate
            } else if before.0 == before.1 {
                EdgeStatus::Internal
            } else if after.0 == after.1 {
                EdgeStatus::Merge
            } else {
                EdgeStatus::Link
            };

            if status == EdgeStatus::Merge {
                let after_id = after.0;
                let before_ids: BTreeSet<usize> = before_graph
                    .component_of
                    .iter()
                    .enumerate()
                    .filter_map(|(vertex, &cid)| {
                        if after_graph.component_of[vertex] == after_id {
                            Some(cid)
                        } else {
                            None
                        }
                    })
                    .collect();
                let members = after_graph.components[after_id].clone();
                let event = Merge {
                    edge_index,
                    before: before_ids.into_iter().collect(),
                    after: after_id,
                    members,
                };
                merges.push(event.clone());
                self.history.push(event);
            }

            steps.push(Step {
                edge_index,
                status,
                before,
                after,
            });
        }

        Ok(Batch {
            steps,
            merges,
            graph: self.snapshot(),
        })
    }
}
