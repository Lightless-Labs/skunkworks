fn is_separator(ch: char) -> bool {

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !chars.is_empty() {
                words.push(Word { chars, markers });
                chars = Vec::new();
                markers = Vec::new();
            }
        } else if ch == '~' {
            markers.push(chars.len());
        } else {
            chars.push(ch);
        }
    }

    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }

    words
}

fn append_piece(line: &mut Line, text: String) {
    if !line.pieces.is_empty() {
        line.content_len += 1;
    }
    line.content_len += text.chars().count();
    line.pieces.push(Piece { text });
}

fn emit_line(line: Line, width: usize, justify: bool, line_index: usize, final_line: bool) -> String {
    let mut out = " ".repeat(line.indent);

    if justify && !final_line && line.pieces.len() >= 2 {
        let gaps = line.pieces.len() - 1;
        let extra = width.saturating_sub(line.indent + line.content_len);
        let each = extra / gaps;
        let remainder = extra % gaps;
        let start_gap = line_index % gaps;

        for (i, piece) in line.pieces.iter().enumerate() {
            if i > 0 {
                let gap = i - 1;
                let additional = each + usize::from((gap + gaps - start_gap) % gaps < remainder);
                out.push_str(&" ".repeat(1 + additional));
            }
            out.push_str(&piece.text);
        }
    } else {
        for (i, piece) in line.pieces.iter().enumerate() {
            if i > 0 {
                out.push(' ');
            }
            out.push_str(&piece.text);
        }
    }

    out
}

                    pieces: Vec::new(),

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let scalars: Vec<Scalar> = input
        .char_indices()
        .map(|(byte_index, ch)| Scalar { ch, byte_index })
        .collect();

    for scalar in &scalars {
        if scalar.ch.is_control() && !matches!(scalar.ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl {
                index: scalar.byte_index,
            });
        }
    }

    for i in 0..scalars.len() {
        if scalars[i].ch != '~' {
            continue;
        }
        let valid_side = |index: usize| {
            index < scalars.len()
                && scalars[index].ch != '~'
                && !is_separator(scalars[index].ch)
        };
        if i == 0 || i + 1 >= scalars.len() || !valid_side(i - 1) || !valid_side(i + 1) {
            return Err(WrapError::InvalidMarker {
                index: scalars[i].byte_index,
            });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0usize;
    while i < scalars.len() {
        let ch = scalars[i].ch;
        if ch == '\r' {
            if i + 1 < scalars.len() && scalars[i + 1].ch == '\n' {
                i += 1;
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut pos = 0usize;

    while pos < normalized.len() {
        if normalized[pos] == '\n' {
            let run_start = pos;
            while pos < normalized.len() && normalized[pos] == '\n' {
                pos += 1;
            }
            let run_len = pos - run_start;
            if run_len >= 2 {
                let words = parse_words(&normalized[start..run_start]);
                output.extend(layout_paragraph(&words, options));
                output.extend(std::iter::repeat_n(String::new(), run_len - 1));
                start = pos;
            }
        } else {
            pos += 1;
        }
    }

    let words = parse_words(&normalized[start..]);
    output.extend(layout_paragraph(&words, options));

    Ok(output)
}

    InvalidMarker { index: usize },
}

    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let valid_neighbor = |j: usize| {
                j < chars.len() && chars[j].1 != '~' && !is_separator(chars[j].1)
            };
            if i == 0 || i + 1 == chars.len() || !valid_neighbor(i - 1) || !valid_neighbor(i + 1) {
                return Err(WrapError::InvalidMarker {
                    index: chars[i].0,
                });
            }
        }
    }

    let mut normalized = String::with_capacity(input.len());
    let mut it = input.chars().peekable();
    while let Some(ch) = it.next() {
        if ch == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let normalized_chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut start = 0;
    let mut i = 0;
    while i < normalized_chars.len() {
        if normalized_chars[i] == '\n' {
            let run_start = i;
            while i < normalized_chars.len() && normalized_chars[i] == '\n' {
                i += 1;
            }
            if i - run_start >= 2 {
                layout_segment(&normalized_chars[start..run_start], options, &mut output);
                for _ in 1..(i - run_start) {
                    output.push(String::new());
                }
                start = i;
            }
        } else {
            i += 1;
        }
    }
    layout_segment(&normalized_chars[start..], options, &mut output);

    Ok(output)
}

fn layout_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words = Vec::<Piece>::new();
    let mut word = Vec::new();
    let mut breaks = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !word.is_empty() {
                words.push(Piece {
                    text: std::mem::take(&mut word),
                    breaks: std::mem::take(&mut breaks),
                });
            }
        } else if ch == '~' {
            breaks.push(word.len());
        } else {
            word.push(ch);
        }
    }
    if !word.is_empty() {
        words.push(Piece { text: word, breaks });
    }
    if words.is_empty() {
        return;
    }

    let mut lines = Vec::<Line>::new();
    let mut line_index = 0usize;
    let mut indent = options.first_indent;
    let mut capacity = options.width - indent;
    let mut pieces = Vec::<Vec<char>>::new();
    let mut content_len = 0usize;

    for word in &words {
        let mut offset = 0usize;
        while offset < word.text.len() {
            let remaining = word.text.len() - offset;
            let gap = usize::from(!pieces.is_empty());

            if remaining + gap <= capacity.saturating_sub(content_len) {
                if gap != 0 {
                    content_len += 1;
                }
                pieces.push(word.text[offset..].to_vec());
                content_len += remaining;
                offset = word.text.len();
                continue;
            }

            let room = capacity.saturating_sub(content_len);
            let usable = word
                .breaks
                .iter()
                .copied()
                .filter(|&b| b > offset && b < word.text.len() && b - offset + 1 <= room)
                .collect::<Vec<_>>();

            let chosen = usable.last().copied().filter(|&b| {
                if pieces.is_empty() {
                    true
                } else {
                    let used = b - offset + 1;
                    let unused = capacity - content_len - used;
                    let baseline = capacity - content_len;
                    unused * unused + options.hyphen_cost < baseline * baseline
                }
            });

            if let Some(break_at) = chosen {
                if !pieces.is_empty() {
                    content_len += 1;
                }
                let mut fragment = word.text[offset..break_at].to_vec();
                fragment.push('-');
                content_len += fragment.len();
                pieces.push(fragment);
                lines.push(Line {
                    indent,
                    pieces: std::mem::take(&mut pieces),
                });
                line_index += 1;
                indent = options.rest_indent;
                capacity = options.width - indent;
                content_len = 0;
                offset = break_at;
                continue;
            }

            if !pieces.is_empty() {
                lines.push(Line {
                    indent,
                    pieces: std::mem::take(&mut pieces),
                });
                line_index += 1;
                indent = options.rest_indent;
                capacity = options.width - indent;
                content_len = 0;
                continue;
            }

            if remaining > capacity {
                let end = offset + capacity;
                pieces.push(word.text[offset..end].to_vec());
                content_len = capacity;
                lines.push(Line {
                    indent,
                    pieces: std::mem::take(&mut pieces),
                });
                line_index += 1;
                indent = options.rest_indent;
                capacity = options.width - indent;
                content_len = 0;
                offset = end;
            } else {
                pieces.push(word.text[offset..].to_vec());
                content_len += remaining;
                offset = word.text.len();
            }
        }
    }

    if !pieces.is_empty() {
        lines.push(Line { indent, pieces });
    }

    let final_line = lines.len().saturating_sub(1);
    for (idx, line) in lines.into_iter().enumerate() {
        let piece_count = line.pieces.len();
        let gaps = piece_count.saturating_sub(1);
        let base_width = line.indent
            + line.pieces.iter().map(Vec::len).sum::<usize>()
            + gaps;
        let extra = options.width.saturating_sub(base_width);
        let justify = options.justify && idx != final_line && piece_count >= 2;

        let mut rendered = String::new();
        rendered.extend(std::iter::repeat(' ').take(line.indent));
        for (piece_idx, piece) in line.pieces.iter().enumerate() {
            if piece_idx > 0 {
                let mut spaces = 1;
                if justify && gaps > 0 {
                    spaces += extra / gaps;
                    let first_extra = idx % gaps;
                    let offset = piece_idx - 1;
                    if (offset + gaps - first_extra) % gaps < extra % gaps {
                        spaces += 1;
                    }
                }
                rendered.extend(std::iter::repeat(' ').take(spaces));
            }
            rendered.extend(piece.iter());
        }
        output.push(rendered);
    }

    let _ = line_index;
}

struct Piece {

struct Line {
    pieces: Vec<Piece>,
    indent: usize,

impl Line {
    fn content_len(&self) -> usize {
        self.pieces.iter().map(|p| p.text.chars().count()).sum::<usize>()
            + self.pieces.len().saturating_sub(1)
    }

    fn is_empty(&self) -> bool {
        self.pieces.is_empty()
    }

    fn push_piece(&mut self, text: String, word_id: usize) {
        self.pieces.push(Piece { text, word_id });
    }

    fn render(&self, options: &WrapOptions, line_index: usize, final_line: bool) -> String {
        let mut output = " ".repeat(self.indent);
        let gaps = self.pieces.len().saturating_sub(1);
        let content_width = self.content_len();
        let extra = if options.justify && !final_line && self.pieces.len() >= 2 {
            options.width.saturating_sub(self.indent + content_width)
        } else {
            0
        };

        let base = if gaps > 0 { extra / gaps } else { 0 };
        let remainder = if gaps > 0 { extra % gaps } else { 0 };
        let start_gap = if gaps > 0 { line_index % gaps } else { 0 };

        for (i, piece) in self.pieces.iter().enumerate() {
            if i > 0 {
                output.push(' ');
                if options.justify && !final_line && self.pieces.len() >= 2 {
                    output.push_str(&" ".repeat(base));
                    let cyclic_index = (i - 1 + gaps - start_gap) % gaps;
                    if cyclic_index < remainder {
                        output.push(' ');
                    }
                }
            }
            output.push_str(&piece.text);
        }
        output
    }
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if ch == '~' {
            markers.push(chars.len());
        } else {
            chars.push(ch);
        }
    }

    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }
    words
}

fn wrap_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let words = parse_words(segment);
    if words.is_empty() {
        return;
    }

    let mut completed = Vec::<Line>::new();
    let mut line = Line {
        pieces: Vec::new(),
        indent: options.first_indent,
    };
    let mut paragraph_line_index = 0;

    for (word_id, word) in words.iter().enumerate() {
        let mut offset = 0usize;
        while offset < word.chars.len() {
            let capacity = options.width - line.indent;
            let existing = line.content_len();
            let gap = usize::from(!line.is_empty());
            let room_for_word = capacity.saturating_sub(existing + gap);
            let suffix_len = word.chars.len() - offset;

            if suffix_len <= room_for_word {
                let text: String = word.chars[offset..].iter().collect();
                line.push_piece(text, word_id);
                offset = word.chars.len();
                continue;
            }

            let available = capacity.saturating_sub(existing);
            let mut rightmost: Option<(usize, usize)> = None;
            for &marker in &word.markers {
                if marker <= offset || marker >= word.chars.len() {
                    continue;
                }
                let prefix_len = marker - offset;
                if prefix_len + 1 <= available {
                    rightmost = Some((marker, prefix_len));
                }
            }

            let accepted = if let Some((marker, prefix_len)) = rightmost {
                if line.is_empty() {
                    Some((marker, prefix_len))
                } else {
                    let unused = available - prefix_len - 1;
                    let baseline = capacity.saturating_sub(existing);
                    let lhs = (unused as u128) * (unused as u128)
                        + options.hyphen_cost as u128;
                    let rhs = (baseline as u128) * (baseline as u128);
                    (lhs < rhs).then_some((marker, prefix_len))
                }
            } else {
                None
            };

            if let Some((marker, prefix_len)) = accepted {
                let mut text: String = word.chars[offset..offset + prefix_len].iter().collect();
                text.push('-');
                line.push_piece(text, word_id);
                completed.push(line);
                paragraph_line_index += 1;
                line = Line {
                    pieces: Vec::new(),
                    indent: options.rest_indent,
                };
                offset = marker;
                continue;
            }

            if !line.is_empty() {
                completed.push(line);
                paragraph_line_index += 1;
                line = Line {
                    pieces: Vec::new(),
                    indent: options.rest_indent,
                };
                continue;
            }

            let capacity = options.width - line.indent;
            if suffix_len > capacity {
                let text: String = word.chars[offset..offset + capacity].iter().collect();
                line.push_piece(text, word_id);
                completed.push(line);
                paragraph_line_index += 1;
                line = Line {
                    pieces: Vec::new(),
                    indent: options.rest_indent,
                };
                offset += capacity;
            } else {
                let text: String = word.chars[offset..].iter().collect();
                line.push_piece(text, word_id);
                offset = word.chars.len();
            }
        }
    }

    if !line.is_empty() {
        completed.push(line);
    }

    let total_lines = completed.len();
    for (i, line) in completed.iter().enumerate() {
        output.push(line.render(options, i, i + 1 == total_lines));
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    for (i, &(byte_index, ch)) in original.iter().enumerate() {
        if ch != '~' {
            continue;
        }
        let valid_neighbor = |neighbor: Option<char>| {
            neighbor.map_or(false, |c| c != '~' && !is_separator(c))
        };
        if !valid_neighbor(i.checked_sub(1).and_then(|j| original.get(j).map(|x| x.1)))
            || !valid_neighbor(original.get(i + 1).map(|x| x.1))
        {
            return Err(WrapError::InvalidMarker { index: byte_index });
        }
    }

    let mut normalized = Vec::<char>::new();
    let mut iter = input.chars().peekable();
    while let Some(ch) = iter.next() {
        if ch == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut i = 0usize;

    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut end = i + 1;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            let run_len = end - i;
            if run_len >= 2 {
                wrap_segment(&normalized[start..i], options, &mut output);
                for _ in 1..run_len {
                    output.push(String::new());
                }
                start = end;
            }
            i = end;
        } else {
            i += 1;
        }
    }

    wrap_segment(&normalized[start..], options, &mut output);
    Ok(output)
}

    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect::<Vec<_>>();
    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let valid_neighbor = |j: usize| {
            j < chars.len() && chars[j].1 != '~' && !is_separator(chars[j].1)
        };
        if i == 0 || i + 1 >= chars.len() || !valid_neighbor(i - 1) || !valid_neighbor(i + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }

    let mut normalized = String::new();
    let mut input_chars = input.chars().peekable();
    while let Some(c) = input_chars.next() {
        if c == '\r' {
            if input_chars.peek() == Some(&'\n') {
                input_chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let normalized_chars: Vec<char> = normalized.chars().collect();
    let mut paragraphs: Vec<Vec<Word>> = Vec::new();
    let mut current_paragraph = Vec::new();
    let mut current_word = Vec::new();

    let flush_word = |word_chars: &mut Vec<char>, paragraph: &mut Vec<Word>| {
        if !word_chars.is_empty() {
            let mut visible = Vec::new();
            let mut markers = Vec::new();
            for c in word_chars.drain(..) {
                if c == '~' {
                    markers.push(visible.len());
                } else {
                    visible.push(c);
                }
            }
            paragraph.push(Word {
                chars: visible,
                markers,
            });
        }
    };

    let mut i = 0;
    while i < normalized_chars.len() {
        let c = normalized_chars[i];
        if c == '\n' {
            let mut end = i;
            while end < normalized_chars.len() && normalized_chars[end] == '\n' {
                end += 1;
            }
            if end - i >= 2 {
                flush_word(&mut current_word, &mut current_paragraph);
                paragraphs.push(std::mem::take(&mut current_paragraph));
                for _ in 1..(end - i) {
                    paragraphs.push(Vec::new());
                }
                current_paragraph = Vec::new();
            } else {
                flush_word(&mut current_word, &mut current_paragraph);
            }
            i = end;
        } else if is_separator(c) {
            flush_word(&mut current_word, &mut current_paragraph);
            i += 1;
        } else {
            current_word.push(c);
            i += 1;
        }
    }
    flush_word(&mut current_word, &mut current_paragraph);
    paragraphs.push(current_paragraph);

    let mut output = Vec::new();
    for paragraph in paragraphs {
        if paragraph.is_empty() {
            output.push(String::new());
            continue;
        }

        let mut lines: Vec<Line> = Vec::new();
        let mut line_index = 0usize;
        let mut line = Line {
            pieces: Vec::new(),
            indent: options.first_indent,
        };

        for (word_id, word) in paragraph.iter().enumerate() {
            let mut offset = 0usize;
            while offset < word.chars.len() {
                let capacity = options.width - line.indent;
                let used = line.content_len();
                let room = capacity.saturating_sub(used + usize::from(!line.pieces.is_empty()));
                let suffix_len = word.chars.len() - offset;

                if suffix_len <= room {
                    let text: String = word.chars[offset..].iter().collect();
                    line.pieces.push(Piece { text, word_id });
                    offset = word.chars.len();
                    continue;
                }

                    .filter(|&m| m > offset && m < word.chars.len())
                    .filter(|&m| m - offset + 1 <= room)
                    .collect::<Vec<_>>();
                let chosen = usable.last().copied();

                if let Some(marker) = chosen {
                    let prefix_len = marker - offset;
                    let unused = room - prefix_len - 1;
                    let accept = if line.pieces.is_empty() {
                        true
                    } else {
                        let baseline = capacity.saturating_sub(used);
                        unused.saturating_mul(unused).saturating_add(options.hyphen_cost)
                            < baseline.saturating_mul(baseline)
                    };

                    if accept {
                        let mut text: String = word.chars[offset..marker].iter().collect();
                        text.push('-');
                        line.pieces.push(Piece { text, word_id });
                        lines.push(line);
                        line_index += 1;
                        line = Line {
                            pieces: Vec::new(),
                            indent: options.rest_indent,
                        };
                        offset = marker;
                        continue;
                    }
                }

                if !line.pieces.is_empty() {
                    lines.push(line);
                    line_index += 1;
                    line = Line {
                        pieces: Vec::new(),
                        indent: options.rest_indent,
                    };
                    continue;
                }

                if suffix_len > capacity {
                    let end = offset + capacity;
                    let text: String = word.chars[offset..end].iter().collect();
                    line.pieces.push(Piece { text, word_id });
                    lines.push(line);
                    line_index += 1;
                    line = Line {
                        pieces: Vec::new(),
                        indent: options.rest_indent,
                    };
                    offset = end;
                } else {
                    let text: String = word.chars[offset..].iter().collect();
                    line.pieces.push(Piece { text, word_id });
                    offset = word.chars.len();
                }
            }
        }

        if !line.pieces.is_empty() {
            lines.push(line);
        }

        for (idx, line) in lines.iter().enumerate() {
            let content_width = line.content_len();
            let gaps = line.pieces.len().saturating_sub(1);
            let justify = options.justify && idx + 1 < lines.len() && gaps >= 1;
            let extra = if justify {
                options.width.saturating_sub(line.indent + content_width)
            } else {
                0
            };
            let each = if gaps > 0 { extra / gaps } else { 0 };
            let remainder = if gaps > 0 { extra % gaps } else { 0 };
            let start_gap = if gaps > 0 { line_index.wrapping_sub(lines.len() - 1 - idx) % gaps } else { 0 };

            let mut rendered = " ".repeat(line.indent);
            for (piece_idx, piece) in line.pieces.iter().enumerate() {
                if piece_idx > 0 {
                    rendered.push(' ');
                    if justify {
                        let gap = piece_idx - 1;
                        let bonus = usize::from((gap + gaps - start_gap) % gaps < remainder);
                        rendered.push_str(&" ".repeat(each + bonus));
                    }
                }
                rendered.push_str(&piece.text);
            }
            output.push(rendered);
        }
    }

    Ok(output)
}
