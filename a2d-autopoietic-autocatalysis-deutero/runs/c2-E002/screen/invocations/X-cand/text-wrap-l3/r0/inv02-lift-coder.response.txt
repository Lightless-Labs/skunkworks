#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Piece {
    text: String,
    word_id: usize,
}

struct Line {
    pieces: Vec<Piece>,
    indent: usize,
}

impl Line {
    fn content_len(&self) -> usize {
        self.pieces.iter().map(|p| p.text.chars().count()).sum::<usize>()
            + self.pieces.len().saturating_sub(1)
    }
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect::<Vec<_>>();
    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let valid_neighbor = |j: usize| {
            j < chars.len() && chars[j].1 != '~' && !is_separator(chars[j].1)
        };
        if i == 0 || i + 1 >= chars.len() || !valid_neighbor(i - 1) || !valid_neighbor(i + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }

    let mut normalized = String::new();
    let mut input_chars = input.chars().peekable();
    while let Some(c) = input_chars.next() {
        if c == '\r' {
            if input_chars.peek() == Some(&'\n') {
                input_chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let normalized_chars: Vec<char> = normalized.chars().collect();
    let mut paragraphs: Vec<Vec<Word>> = Vec::new();
    let mut current_paragraph = Vec::new();
    let mut current_word = Vec::new();

    let flush_word = |word_chars: &mut Vec<char>, paragraph: &mut Vec<Word>| {
        if !word_chars.is_empty() {
            let mut visible = Vec::new();
            let mut markers = Vec::new();
            for c in word_chars.drain(..) {
                if c == '~' {
                    markers.push(visible.len());
                } else {
                    visible.push(c);
                }
            }
            paragraph.push(Word {
                chars: visible,
                markers,
            });
        }
    };

    let mut i = 0;
    while i < normalized_chars.len() {
        let c = normalized_chars[i];
        if c == '\n' {
            let mut end = i;
            while end < normalized_chars.len() && normalized_chars[end] == '\n' {
                end += 1;
            }
            if end - i >= 2 {
                flush_word(&mut current_word, &mut current_paragraph);
                paragraphs.push(std::mem::take(&mut current_paragraph));
                for _ in 1..(end - i) {
                    paragraphs.push(Vec::new());
                }
                current_paragraph = Vec::new();
            } else {
                flush_word(&mut current_word, &mut current_paragraph);
            }
            i = end;
        } else if is_separator(c) {
            flush_word(&mut current_word, &mut current_paragraph);
            i += 1;
        } else {
            current_word.push(c);
            i += 1;
        }
    }
    flush_word(&mut current_word, &mut current_paragraph);
    paragraphs.push(current_paragraph);

    let mut output = Vec::new();
    for paragraph in paragraphs {
        if paragraph.is_empty() {
            output.push(String::new());
            continue;
        }

        let mut lines: Vec<Line> = Vec::new();
        let mut line_index = 0usize;
        let mut line = Line {
            pieces: Vec::new(),
            indent: options.first_indent,
        };

        for (word_id, word) in paragraph.iter().enumerate() {
            let mut offset = 0usize;
            while offset < word.chars.len() {
                let capacity = options.width - line.indent;
                let used = line.content_len();
                let room = capacity.saturating_sub(used + usize::from(!line.pieces.is_empty()));
                let suffix_len = word.chars.len() - offset;

                if suffix_len <= room {
                    let text: String = word.chars[offset..].iter().collect();
                    line.pieces.push(Piece { text, word_id });
                    offset = word.chars.len();
                    continue;
                }

                let usable = word.markers.iter().copied()
                    .filter(|&m| m > offset && m < word.chars.len())
                    .filter(|&m| m - offset + 1 <= room)
                    .collect::<Vec<_>>();
                let chosen = usable.last().copied();

                if let Some(marker) = chosen {
                    let prefix_len = marker - offset;
                    let unused = room - prefix_len - 1;
                    let accept = if line.pieces.is_empty() {
                        true
                    } else {
                        let baseline = capacity.saturating_sub(used);
                        unused.saturating_mul(unused).saturating_add(options.hyphen_cost)
                            < baseline.saturating_mul(baseline)
                    };

                    if accept {
                        let mut text: String = word.chars[offset..marker].iter().collect();
                        text.push('-');
                        line.pieces.push(Piece { text, word_id });
                        lines.push(line);
                        line_index += 1;
                        line = Line {
                            pieces: Vec::new(),
                            indent: options.rest_indent,
                        };
                        offset = marker;
                        continue;
                    }
                }

                if !line.pieces.is_empty() {
                    lines.push(line);
                    line_index += 1;
                    line = Line {
                        pieces: Vec::new(),
                        indent: options.rest_indent,
                    };
                    continue;
                }

                let capacity = options.width - line.indent;
                if suffix_len > capacity {
                    let end = offset + capacity;
                    let text: String = word.chars[offset..end].iter().collect();
                    line.pieces.push(Piece { text, word_id });
                    lines.push(line);
                    line_index += 1;
                    line = Line {
                        pieces: Vec::new(),
                        indent: options.rest_indent,
                    };
                    offset = end;
                } else {
                    let text: String = word.chars[offset..].iter().collect();
                    line.pieces.push(Piece { text, word_id });
                    offset = word.chars.len();
                }
            }
        }

        if !line.pieces.is_empty() {
            lines.push(line);
        }

        for (idx, line) in lines.iter().enumerate() {
            let content_width = line.content_len();
            let gaps = line.pieces.len().saturating_sub(1);
            let justify = options.justify && idx + 1 < lines.len() && gaps >= 1;
            let extra = if justify {
                options.width.saturating_sub(line.indent + content_width)
            } else {
                0
            };
            let each = if gaps > 0 { extra / gaps } else { 0 };
            let remainder = if gaps > 0 { extra % gaps } else { 0 };
            let start_gap = if gaps > 0 { line_index.wrapping_sub(lines.len() - 1 - idx) % gaps } else { 0 };

            let mut rendered = " ".repeat(line.indent);
            for (piece_idx, piece) in line.pieces.iter().enumerate() {
                if piece_idx > 0 {
                    rendered.push(' ');
                    if justify {
                        let gap = piece_idx - 1;
                        let bonus = usize::from((gap + gaps - start_gap) % gaps < remainder);
                        rendered.push_str(&" ".repeat(each + bonus));
                    }
                }
                rendered.push_str(&piece.text);
            }
            output.push(rendered);
        }
    }

    Ok(output)
}
