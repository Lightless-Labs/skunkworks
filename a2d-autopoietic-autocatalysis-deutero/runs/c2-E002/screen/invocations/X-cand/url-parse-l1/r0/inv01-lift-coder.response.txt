#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();

    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }

    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map_or(input.len(), |offset| authority_start + offset);
    let authority = &bytes[authority_start..authority_end];

    let mut at_positions = authority
        .iter()
        .enumerate()
        .filter_map(|(offset, &byte)| (byte == b'@').then_some(offset));

    let first_at = at_positions.next();
    if let Some(second_at) = at_positions.next() {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + second_at,
        });
    }

    let (userinfo, host_start_offset) = match first_at {
        Some(at) if at == 0 => {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        Some(at) => (
            Some(input[authority_start..authority_start + at].to_owned()),
            at + 1,
        ),
        None => (None, 0),
    };

    let host_port = &authority[host_start_offset..];
    let host_port_start = authority_start + host_start_offset;
    let host_colon = host_port.iter().position(|&byte| byte == b':');

    let (host_bytes, port_bytes, port_start) = match host_colon {
        Some(offset) => (
            &host_port[..offset],
            Some(&host_port[offset + 1..]),
            host_port_start + offset + 1,
        ),
        None => (host_port, None, 0),
    };

    if host_bytes.is_empty() {
        return Err(ParseError::InvalidHost { at: host_port_start });
    }

    for (offset, &byte) in host_bytes.iter().enumerate() {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost {
                at: host_port_start + offset,
            });
        }
    }

    let port = match port_bytes {
        None => None,
        Some([]) => return Err(ParseError::InvalidPort { at: port_start }),
        Some(port_bytes) => {
            for (offset, &byte) in port_bytes.iter().enumerate() {
                if !byte.is_ascii_digit() {
                    return Err(ParseError::InvalidPort {
                        at: port_start + offset,
                    });
                }
            }

            if port_bytes.len() >= 2 && port_bytes[0] == b'0' {
                return Err(ParseError::InvalidPort { at: port_start });
            }

            let mut value: u32 = 0;
            for &byte in port_bytes {
                value = match value
                    .checked_mul(10)
                    .and_then(|value| value.checked_add((byte - b'0') as u32))
                {
                    Some(value) => value,
                    None => return Err(ParseError::InvalidPort { at: port_start }),
                };
            }

            if !(1..=65535).contains(&value) {
                return Err(ParseError::InvalidPort { at: port_start });
            }

            Some(value as u16)
        }
    };

    let fragment_delimiter = bytes[authority_end..]
        .iter()
        .position(|&byte| byte == b'#')
        .map(|offset| authority_end + offset);

    let before_fragment_end = fragment_delimiter.unwrap_or(input.len());
    let query_delimiter = bytes[authority_end..before_fragment_end]
        .iter()
        .position(|&byte| byte == b'?')
        .map(|offset| authority_end + offset);

    let path_end = query_delimiter.unwrap_or(before_fragment_end);
    let path = if path_end == authority_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };

    let query = query_delimiter.map(|start| {
        let end = fragment_delimiter.unwrap_or(input.len());
        input[start + 1..end].to_owned()
    });

    let fragment = fragment_delimiter.map(|start| input[start + 1..].to_owned());

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host: std::str::from_utf8(host_bytes)
            .expect("validated host bytes are ASCII")
            .to_ascii_lowercase(),
        port,
        path,
        query,
        fragment,
    })
}
