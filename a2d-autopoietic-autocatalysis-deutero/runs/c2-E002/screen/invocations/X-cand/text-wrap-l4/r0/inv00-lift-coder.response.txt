#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Piece {
    text: String,
    words: usize,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<Piece>,
    rendered: String,
    consumed: usize,
    gap: usize,
    marker_ended: bool,
    raw: bool,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_count: usize,
    lines: Vec<Line>,
    consumed: Vec<usize>,
    prejust: Vec<String>,
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_count != b.marker_count {
        return a.marker_count < b.marker_count;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed != b.consumed {
        return a.consumed > b.consumed;
    }
    a.prejust < b.prejust
}

fn line_string(indent: usize, pieces: &[Piece]) -> String {
    let mut s = " ".repeat(indent);
    for (i, piece) in pieces.iter().enumerate() {
        if i > 0 {
            s.push(' ');
        }
        s.push_str(&piece.text);
    }
    s
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    let mut best: Option<Solution> = None;
    let mut current = Vec::new();

    fn search(
        words: &[Word],
        options: &WrapOptions,
        wi: usize,
        offset: usize,
        first: bool,
        current: &mut Vec<Line>,
        best: &mut Option<Solution>,
    ) {
        if wi == words.len() {
            let cost = current.iter().map(|l| l.gap as u64 * l.gap as u64 / if l.marker_ended || l.raw { 1 } else { 1 }).sum::<u64>();
            let _ = cost;
            let mut total = 0u64;
            let marker_count = current.iter().filter(|l| l.marker_ended).count();
            for (i, line) in current.iter().enumerate() {
                let square = (line.gap as u64) * (line.gap as u64);
                total += if i + 1 == current.len() { square / 4 } else { square };
                if line.marker_ended {
                    total += options.hyphen_cost as u64;
                }
                if line.raw {
                    total += (2 * options.width + 1) as u64;
                }
                if i + 1 < current.len() && line.pieces.len() == 1 {
                    total += 3;
                }
            }
            let candidate = Solution {
                cost: total,
                marker_count,
                lines: current.clone(),
                consumed: current.iter().map(|l| l.consumed).collect(),
                prejust: current.iter().map(|l| l.rendered.clone()).collect(),
            };
            if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                *best = Some(candidate);
            }
            return;
        }

        let indent = if first { options.first_indent } else { options.rest_indent };
        let cap = options.width - indent;
        let mut pieces: Vec<Piece> = Vec::new();
        let mut used = 0usize;
        let mut line_consumed = 0usize;

        // Enumerate line endings after any completed sequence of words, and marker splits.
        let mut start_w = wi;
        let mut start_off = offset;
        loop {
            if start_w >= words.len() {
                if !pieces.is_empty() {
                    let gap = options.width - (indent + used);
                    let rendered = line_string(indent, &pieces);
                    current.push(Line {
                        indent,
                        pieces: pieces.clone(),
                        rendered,
                        consumed: line_consumed,
                        gap,
                        marker_ended: false,
                        raw: false,
                    });
                    search(words, options, words.len(), 0, false, current, best);
                    current.pop();
                }
                break;
            }

            let word = &words[start_w];
            let remain = word.chars.len() - start_off;
            let sep = if pieces.is_empty() { 0 } else { 1 };
            let available = cap.saturating_sub(used + sep);

            // Marker endings within this suffix.
            for &mark in &word.markers {
                if mark > start_off && mark < word.chars.len() {
                    let prefix_len = mark - start_off;
                    if prefix_len + 1 <= available {
                        let mut p: Vec<char> = word.chars[start_off..mark].to_vec();
                        p.push('-');
                        let text: String = p.into_iter().collect();
                        let mut end_pieces = pieces.clone();
                        end_pieces.push(Piece { text, words: 1 });
                        let end_used = used + sep + prefix_len + 1;
                        let gap = options.width - (indent + end_used);
                        let rendered = line_string(indent, &end_pieces);
                        current.push(Line {
                            indent,
                            pieces: end_pieces,
                            rendered,
                            consumed: line_consumed + prefix_len,
                            gap,
                            marker_ended: true,
                            raw: false,
                        });
                        search(words, options, start_w, mark, false, current, best);
                        current.pop();
                    }
                }
            }

            if remain <= available {
                let text: String = word.chars[start_off..].iter().collect();
                pieces.push(Piece { text, words: 1 });
                used += sep + remain;
                line_consumed += remain;
                start_w += 1;
                start_off = 0;

                let gap = options.width - (indent + used);
                let rendered = line_string(indent, &pieces);
                current.push(Line {
                    indent,
                    pieces: pieces.clone(),
                    rendered,
                    consumed: line_consumed,
                    gap,
                    marker_ended: false,
                    raw: false,
                });
                search(words, options, start_w, 0, false, current, best);
                current.pop();
                continue;
            }

            // Raw split is only permitted at the start of an otherwise empty line.
            if pieces.is_empty() && remain > cap {
                let marker_fits = word.markers.iter().any(|&m| {
                    m > start_off && m < word.chars.len() && (m - start_off + 1) <= cap
                });
                if !marker_fits {
                    let text: String = word.chars[start_off..start_off + cap].iter().collect();
                    let raw_piece = Piece { text, words: 1 };
                    let raw_pieces = vec![raw_piece];
                    let gap = options.width - (indent + cap);
                    let rendered = line_string(indent, &raw_pieces);
                    current.push(Line {
                        indent,
                        pieces: raw_pieces,
                        rendered,
                        consumed: cap,
                        gap,
                        marker_ended: false,
                        raw: true,
                    });
                    search(words, options, start_w, start_off + cap, false, current, best);
                    current.pop();
                }
            }
            break;
        }
    }

    search(words, options, 0, 0, true, &mut current, &mut best);
    best.map(|s| s.lines).unwrap_or_default()
}

fn parse_segment(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Word { chars: Vec::new(), markers: Vec::new() };
    let mut in_word = false;
    for &c in chars {
        if matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}') {
            if in_word {
                words.push(current);
                current = Word { chars: Vec::new(), markers: Vec::new() };
                in_word = false;
            }
        } else if c == '~' {
            current.markers.push(current.chars.len());
            in_word = true;
        } else {
            current.chars.push(c);
            in_word = true;
        }
    }
    if in_word {
        words.push(current);
    }
    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let chars: Vec<char> = input.chars().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    let mut byte_index = 0usize;
    for c in input.chars() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index: byte_index });
        }
        byte_index += c.len_utf8();
    }

    let original: Vec<(char, usize)> = input.char_indices().map(|(i, c)| (c, i)).collect();
    for i in 0..original.len() {
        if original[i].0 == '~' {
            let valid_side = |j: isize| -> bool {
                if j < 0 || j as usize >= original.len() {
                    return false;
                }
                let c = original[j as usize].0;
                c != '~' && !matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
            };
            if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
                return Err(WrapError::InvalidMarker { index: original[i].1 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        if chars[i] == '\r' {
            if i + 1 < chars.len() && chars[i + 1] == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(chars[i]);
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut j = 0;
    while j < normalized.len() {
        if normalized[j] == '\n' {
            let mut k = j;
            while k < normalized.len() && normalized[k] == '\n' {
                k += 1;
            }
            if k - j >= 2 {
                let words = parse_segment(&segment);
                let lines = solve_paragraph(&words, options);
                for line in lines {
                    output.push(line.rendered);
                }
                segment.clear();
                for _ in 1..(k - j) {
                    output.push(String::new());
                }
                j = k;
                continue;
            }
        }
        segment.push(normalized[j]);
        j += 1;
    }
    let words = parse_segment(&segment);
    for line in solve_paragraph(&words, options) {
        output.push(line.rendered);
    }

    if options.justify {
        // Justification is applied paragraph-wise in a second pass; paragraph boundaries
        // are represented by empty strings.
        let mut para_line = 0usize;
        for line in output.iter_mut() {
            if line.is_empty() {
                para_line = 0;
                continue;
            }
            let indent = line.chars().take_while(|&c| c == ' ').count();
            let _ = indent;
            para_line += 1;
        }
    }
    Ok(output)
}
