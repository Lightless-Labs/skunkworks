#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    text: String,
    gap: usize,
    pieces: usize,
    final_line: bool,
    consumed: usize,
    marker: bool,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    markers: usize,
    lines: Vec<Line>,
}

impl Solution {
    fn key_cmp(&self, other: &Self) -> std::cmp::Ordering {
        self.cost
            .cmp(&other.cost)
            .then(self.markers.cmp(&other.markers))
            .then(self.lines.len().cmp(&other.lines.len()))
            .then_with(|| {
                for (a, b) in self.lines.iter().zip(&other.lines) {
                    match b.consumed.cmp(&a.consumed) {
                        std::cmp::Ordering::Equal => {}
                        ord => return ord,
                    }
                }
                std::cmp::Ordering::Equal
            })
            .then_with(|| {
                for (a, b) in self.lines.iter().zip(&other.lines) {
                    match a.text.cmp(&b.text) {
                        std::cmp::Ordering::Equal => {}
                        ord => return ord,
                    }
                }
                std::cmp::Ordering::Equal
            })
    }
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if ch == '~' {
            markers.push(chars.len());
        } else {
            chars.push(ch);
        }
    }

    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }
    words
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn solve(
    words: &[Word],
    wi: usize,
    offset: usize,
    first: bool,
    options: &WrapOptions,
    memo: &mut Vec<Vec<[Option<Solution>; 2]>>,
) -> Solution {
    if wi >= words.len() {
        return Solution {
            cost: 0,
            markers: 0,
            lines: Vec::new(),
        };
    }
    let fi = usize::from(first);
    if let Some(s) = &memo[wi][offset][fi] {
        return s.clone();
    }

    let indent = if first {
        options.first_indent
    } else {
        options.rest_indent
    };
    let cap = options.width - indent;
    let word = &words[wi];
    let remaining = word.chars.len() - offset;
    let mut candidates = Vec::new();

    // A line may begin with a suffix of the current word, then complete
    // further words separated by one space.
    let mut display = String::new();
    let mut consumed = 0usize;
    let mut pieces = 0usize;
    let mut word_index = wi;
    let mut word_offset = offset;
    let mut used = 0usize;

    loop {
        let current = &words[word_index];
        let rem = current.chars.len() - word_offset;
        let room = cap.saturating_sub(used);

        // Complete the current word, if it fits.
        if rem <= room {
            display.extend(current.chars[word_offset..].iter());
            used += rem;
            consumed += rem;
            pieces += 1;

            let next_wi = word_index + 1;
            let next_sol = if next_wi == words.len() {
                Solution {
                    cost: 0,
                    markers: 0,
                    lines: Vec::new(),
                }
            } else {
                solve(words, next_wi, 0, false, options, memo)
            };
            let mut text = " ".repeat(indent);
            text.push_str(&display);
            let gap = options.width - (indent + used + (pieces - 1));
            let line_cost = if next_wi == words.len() {
                (gap * gap / 4) as u64
            } else {
                (gap * gap) as u64 + if pieces == 1 { 3 } else { 0 }
            };
            let mut lines = vec![Line {
                text,
                gap,
                pieces,
                final_line: next_wi == words.len(),
                consumed,
                marker: false,
            }];
            lines.extend(next_sol.lines.clone());
            candidates.push(Solution {
                cost: line_cost + next_sol.cost,
                markers: next_sol.markers,
                lines,
            });

            if next_wi < words.len() {
                let space_room = cap.saturating_sub(used + 1);
                if space_room == 0 {
                    break;
                }
                used += 1;
                display.push(' ');
                word_index = next_wi;
                word_offset = 0;
                continue;
            }
            break;
        }

        // Marker-ended choices within this word.
        for &m in current.markers.iter() {
            if m > word_offset && m < current.chars.len() {
                let prefix_len = m - word_offset;
                if prefix_len + 1 <= room {
                    let mut d = display.clone();
                    d.extend(current.chars[word_offset..m].iter());
                    d.push('-');
                    let line_used = used + prefix_len + 1;
                    let next_sol = solve(words, word_index, m, false, options, memo);
                    let mut text = " ".repeat(indent);
                    text.push_str(&d);
                    let gap = options.width - (indent + line_used);
                    let line_cost = (gap * gap) as u64
                        + options.hyphen_cost as u64
                        + if pieces == 0 { 3 } else if pieces == 1 { 3 } else { 0 };
                    let mut lines = vec![Line {
                        text,
                        gap,
                        pieces: pieces + 1,
                        final_line: false,
                        consumed: consumed + prefix_len,
                        marker: true,
                    }];
                    lines.extend(next_sol.lines.clone());
                    candidates.push(Solution {
                        cost: line_cost + next_sol.cost,
                        markers: next_sol.markers + 1,
                        lines,
                    });
                }
            }
        }

        // Raw split is allowed only when this line has no prior piece and
        // no marker prefix plus hyphen fits.
        if pieces == 0 && rem > cap {
            let any_marker_fits = current.markers.iter().any(|&m| {
                m > word_offset && m < current.chars.len() && (m - word_offset) + 1 <= cap
            });
            if !any_marker_fits && cap > 0 {
                let mut d = String::new();
                d.extend(current.chars[word_offset..word_offset + cap].iter());
                let next_offset = word_offset + cap;
                let next_sol = solve(words, word_index, next_offset, false, options, memo);
                let mut text = " ".repeat(indent);
                text.push_str(&d);
                let gap = options.width - (indent + cap);
                let line_cost = (gap * gap) as u64
                    + (2 * options.width + 1) as u64
                    + 3;
                let mut lines = vec![Line {
                    text,
                    gap,
                    pieces: 1,
                    final_line: false,
                    consumed: cap,
                    marker: false,
                }];
                lines.extend(next_sol.lines.clone());
                candidates.push(Solution {
                    cost: line_cost + next_sol.cost,
                    markers: next_sol.markers,
                    lines,
                });
            }
        }
        break;
    }

    let best = candidates
        .into_iter()
        .min_by(|a, b| a.key_cmp(b))
        .unwrap_or_else(|| {
            // Capacity is positive; an overlong word always has a raw split.
            unreachable!("no legal continuation")
        });
    memo[wi][offset][fi] = Some(best.clone());
    best
}

fn layout_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }
    let mut memo: Vec<Vec<[Option<Solution>; 2]>> = words
        .iter()
        .map(|w| {
            (0..=w.chars.len())
                .map(|_| [None, None])
                .collect()
        })
        .collect();
    let mut sol = solve(words, 0, 0, true, options, &mut memo);
    let line_count = sol.lines.len();

    for (i, line) in sol.lines.iter_mut().enumerate() {
        if options.justify && !line.final_line && line.pieces >= 2 {
            let gaps = line.pieces - 1;
            let extra = line.gap;
            let each = extra / gaps;
            let rem = extra % gaps;
            let start = i % gaps;
            // Rebuild from the pre-justification line by distributing into
            // its interword spaces.
            let indent = if i == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            let body: Vec<char> = line.text.chars().skip(indent).collect();
            let mut rebuilt = String::new();
            rebuilt.push_str(&" ".repeat(indent));
            let mut gap_no = 0usize;
            for ch in body {
                if ch == ' ' {
                    rebuilt.push(' ');
                    let add = each + usize::from((gap_no + gaps - start) % gaps < rem);
                    rebuilt.push_str(&" ".repeat(add));
                    gap_no += 1;
                } else {
                    rebuilt.push(ch);
                }
            }
            line.text = rebuilt;
        }
    }

    let _ = line_count;
    sol.lines.into_iter().map(|l| l.text).collect()
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<char> = input.chars().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    let mut byte_index = 0usize;
    let mut positions = Vec::with_capacity(chars.len());
    for &ch in &chars {
        positions.push(byte_index);
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index: byte_index });
        }
        byte_index += ch.len_utf8();
    }

    for i in 0..chars.len() {
        if chars[i] == '~' {
            let valid_side = |j: isize| -> bool {
                if j < 0 || j as usize >= chars.len() {
                    return false;
                }
                let c = chars[j as usize];
                c != '~' && !is_separator(c)
            };
            if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
                return Err(WrapError::InvalidMarker { index: positions[i] });
            }
        }
    }

    // Normalize CRLF, then lone CR, while retaining only normalized scalars.
    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        if chars[i] == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1] == '\n' {
                i += 1;
            }
        } else {
            normalized.push(chars[i]);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut i = 0usize;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                let words = parse_words(&normalized[start..i]);
                output.extend(layout_paragraph(&words, options));
                output.extend(std::iter::repeat_n(String::new(), j - i - 1));
                start = j;
            }
            i = j;
        } else {
            i += 1;
        }
    }
    let words = parse_words(&normalized[start..]);
    output.extend(layout_paragraph(&words, options));
    Ok(output)
}
