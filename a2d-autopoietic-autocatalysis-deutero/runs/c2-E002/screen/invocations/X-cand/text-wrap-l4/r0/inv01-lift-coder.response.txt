#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct CandidateLine {
    next_word: usize,
    next_offset: usize,
    ended_at_marker: bool,
    raw_split: bool,
    pieces: usize,
    displayed: String,
    consumed: usize,
    gap: usize,
}

#[derive(Clone)]
struct Layout {
    cost: u64,
    marker_lines: usize,
    lines: Vec<String>,
    consumed: Vec<usize>,
}

fn better(a: &Layout, b: &Layout) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    for (x, y) in a.consumed.iter().zip(&b.consumed) {
        if x != y {
            return x > y;
        }
    }
    a.lines < b.lines
}

fn add_line_candidates(
    words: &[Word],
    start_word: usize,
    start_offset: usize,
    first: bool,
    options: &WrapOptions,
) -> Vec<CandidateLine> {
    let indent = if first {
        options.first_indent
    } else {
        options.rest_indent
    };
    let capacity = options.width - indent;
    let mut out = Vec::new();
    let mut used = 0usize;
    let mut pieces = 0usize;
    let mut display = String::new();
    let mut consumed = 0usize;
    let mut wi = start_word;
    let mut off = start_offset;

    while wi < words.len() {
        let word = &words[wi];
        let remaining = word.chars.len() - off;
        let separator = if pieces == 0 { 0 } else { 1 };
        let room = capacity.saturating_sub(used + separator);

        if remaining <= room {
            if separator != 0 {
                display.push(' ');
                used += 1;
            }
            for &c in &word.chars[off..] {
                display.push(c);
            }
            used += remaining;
            consumed += remaining;
            pieces += 1;
            out.push(CandidateLine {
                next_word: wi + 1,
                next_offset: 0,
                ended_at_marker: false,
                raw_split: false,
                pieces,
                displayed: display.clone(),
                consumed,
                gap: options.width - indent - used,
            });
            wi += 1;
            off = 0;
            if wi >= words.len() {
                break;
            }
            continue;
        }

        let marker_fits: Vec<usize> = word.markers.iter().copied()
            .filter(|&m| m > off && m < word.chars.len() && m - off + 1 <= room)
            .collect();

        for m in marker_fits {
            let mut d = display.clone();
            if separator != 0 {
                d.push(' ');
            }
            for &c in &word.chars[off..m] {
                d.push(c);
            }
            d.push('-');
            let line_used = used + separator + (m - off) + 1;
            out.push(CandidateLine {
                next_word: wi,
                next_offset: m,
                ended_at_marker: true,
                raw_split: false,
                pieces: pieces + 1,
                displayed: d,
                consumed: consumed + (m - off),
                gap: options.width - indent - line_used,
            });
        }

        if pieces == 0 && remaining > capacity && marker_fits.is_empty() {
            let take = capacity;
            let mut d = String::new();
            for &c in &word.chars[off..off + take] {
                d.push(c);
            }
            out.push(CandidateLine {
                next_word: wi,
                next_offset: off + take,
                ended_at_marker: false,
                raw_split: true,
                pieces: 1,
                displayed: d,
                consumed: take,
                gap: options.width - indent - take,
            });
        }

        break;
    }
    out
}

fn optimize_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }

    let mut memo: Vec<Vec<Option<Layout>>> = words
        .iter()
        .map(|w| vec![None; w.chars.len() + 1])
        .collect();
    let mut done: Vec<Vec<bool>> = words
        .iter()
        .map(|w| vec![false; w.chars.len() + 1])
        .collect();

    fn solve(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut [Vec<Option<Layout>>],
        done: &mut [Vec<bool>],
    ) -> Layout {
        if wi == words.len() {
            return Layout {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed: Vec::new(),
            };
        }

        if !first && done[wi][off] {
            return memo[wi][off].as_ref().unwrap().clone();
        }

        let candidates = add_line_candidates(words, wi, off, first, options);
        let mut best: Option<Layout> = None;

        for c in candidates {
            let final_line = c.next_word == words.len();
            let tail = solve(
                c.next_word,
                c.next_offset,
                false,
                words,
                options,
                memo,
                done,
            );
            let mut cost = if final_line {
                (c.gap * c.gap / 4) as u64
            } else {
                (c.gap * c.gap) as u64
            };
            if c.ended_at_marker {
                cost += options.hyphen_cost as u64;
            }
            if c.raw_split {
                cost += (2 * options.width + 1) as u64;
            }
            if !final_line && c.pieces == 1 {
                cost += 3;
            }

            let indent = if first {
                options.first_indent
            } else {
                options.rest_indent
            };
            let mut line = " ".repeat(indent);
            line.push_str(&c.displayed);

            let mut layout = Layout {
                cost: cost + tail.cost,
                marker_lines: usize::from(c.ended_at_marker) + tail.marker_lines,
                lines: vec![line],
                consumed: vec![c.consumed],
            };
            layout.lines.extend(tail.lines);
            layout.consumed.extend(tail.consumed);

            if best.as_ref().map_or(true, |b| better(&layout, b)) {
                best = Some(layout);
            }
        }

        let result = best.unwrap_or(Layout {
            cost: u64::MAX,
            marker_lines: usize::MAX,
            lines: Vec::new(),
            consumed: Vec::new(),
        });
        if !first {
            done[wi][off] = true;
            memo[wi][off] = Some(result.clone());
        }
        result
    }

    let mut layout = solve(0, 0, true, words, options, &mut memo, &mut done);
    if options.justify {
        for (i, line) in layout.lines.iter_mut().enumerate() {
            let indent = if i == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            let body = line[indent..].to_string();
            let pieces: Vec<&str> = body.split(' ').collect();
            if i + 1 == layout.lines.len() || pieces.len() < 2 {
                continue;
            }
            let extra = options.width - line.chars().count();
            let gaps = pieces.len() - 1;
            let each = extra / gaps;
            let rem = extra % gaps;
            let mut rendered = " ".repeat(indent);
            for (j, piece) in pieces.iter().enumerate() {
                if j > 0 {
                    let gap_index = j - 1;
                    let bonus = usize::from((gap_index + gaps - (i % gaps)) % gaps < rem);
                    rendered.push_str(&" ".repeat(1 + each + bonus));
                }
                rendered.push_str(piece);
            }
            *line = rendered;
        }
    }
    layout.lines
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars = input.chars().count();
    if chars > 512 {
        return Err(WrapError::InputTooLong { chars });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && c != '\t' && c != '\r' && c != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let source: Vec<(usize, char)> = input.char_indices().collect();
    let is_sep = |c: char| {
        c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\u{00A0}'
    };
    for i in 0..source.len() {
        if source[i].1 == '~' {
            let valid_left = i > 0 && source[i - 1].1 != '~' && !is_sep(source[i - 1].1);
            let valid_right =
                i + 1 < source.len() && source[i + 1].1 != '~' && !is_sep(source[i + 1].1);
            if !valid_left || !valid_right {
                return Err(WrapError::InvalidMarker {
                    index: source[i].0,
                });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut it = input.chars().peekable();
    while let Some(c) = it.next() {
        if c == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0usize;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                append_segment(&segment, options, &mut output);
                segment.clear();
                for _ in 1..(j - i) {
                    output.push(String::new());
                }
                i = j;
                continue;
            }
        }
        segment.push(normalized[i]);
        i += 1;
    }
    append_segment(&segment, options, &mut output);
    Ok(output)
}

fn append_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let is_sep = |c: char| {
        c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\u{00A0}'
    };
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &c in segment {
        if is_sep(c) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(chars.len());
        } else {
            chars.push(c);
        }
    }
    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }
    output.extend(optimize_paragraph(&words, options));
}
