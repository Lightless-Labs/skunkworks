#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let normalized = input.replace("\r\n", "\n").replace('\r', "\n");
    let chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut index = 0;

    while index < chars.len() {
        if chars[index] != '\n' {
            index += 1;
            continue;
        }

        let run_start = index;
        while index < chars.len() && chars[index] == '\n' {
            index += 1;
        }
        let run_len = index - run_start;

        if run_len >= 2 {
            wrap_segment(&chars[segment_start..run_start], options, &mut output);
            output.extend((1..run_len).map(|_| String::new()));
            segment_start = index;
        }
    }

    wrap_segment(&chars[segment_start..], options, &mut output);
    Ok(output)
}

fn is_separator(ch: char) -> bool {
    ch == ' ' || ch == '\t' || ch == '\n' || ch == '\r' || ch == '\u{00A0}'
}

fn wrap_segment(chars: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words: Vec<Vec<char>> = Vec::new();
    let mut word = Vec::new();

    for &ch in chars {
        if is_separator(ch) {
            if !word.is_empty() {
                words.push(std::mem::take(&mut word));
            }
        } else {
            word.push(ch);
        }
    }
    if !word.is_empty() {
        words.push(word);
    }

    let mut first_line = true;
    let mut line = String::new();
    let mut content_len = 0usize;

    for word in words {
        let mut suffix = word;

        while !suffix.is_empty() {
            let indent = if first_line {
                options.first_indent
            } else {
                options.rest_indent
            };
            let capacity = options.width - indent;

            if suffix.len() <= capacity {
                let needed = suffix.len() + usize::from(content_len > 0);
                if content_len + needed <= capacity {
                    if content_len > 0 {
                        line.push(' ');
                        content_len += 1;
                    }
                    for ch in suffix.drain(..) {
                        line.push(ch);
                        content_len += 1;
                    }
                } else {
                    emit_line(&mut line, content_len, indent, output);
                    first_line = false;
                }
                continue;
            }

            if content_len == 0 {
                let remainder = suffix.split_off(capacity);
                for ch in suffix.drain(..) {
                    line.push(ch);
                    content_len += 1;
                }
                emit_line(&mut line, content_len, indent, output);
                first_line = false;
                suffix = remainder;
            } else {
                let available = capacity - content_len - 1;
                if available >= 2 {
                    line.push(' ');
                    content_len += 1;
                    for ch in suffix.drain(..available) {
                        line.push(ch);
                        content_len += 1;
                    }
                    emit_line(&mut line, content_len, indent, output);
                    first_line = false;
                } else {
                    emit_line(&mut line, content_len, indent, output);
                    first_line = false;
                }
            }
        }
    }

    if content_len > 0 {
        let indent = if first_line {
            options.first_indent
        } else {
            options.rest_indent
        };
        emit_line(&mut line, content_len, indent, output);
    }
}

fn emit_line(line: &mut String, content_len: usize, indent: usize, output: &mut Vec<String>) {
    if content_len == 0 {
        return;
    }
    let mut result = String::with_capacity(indent + line.len());
    result.extend(std::iter::repeat(' ').take(indent));
    result.push_str(line);
    output.push(result);
    line.clear();
}
