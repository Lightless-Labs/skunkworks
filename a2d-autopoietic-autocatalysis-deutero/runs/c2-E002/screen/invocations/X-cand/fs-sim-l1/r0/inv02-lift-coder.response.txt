```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn validate(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut components = Vec::new();
        let mut start = 1;

        for i in 1..=bytes.len() {
            if i == bytes.len() || bytes[i] == b'/' {
                if i == start {
                    return Err(Error::InvalidPath(i.saturating_sub(1)));
                }
                let component = &path[start..i];
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                components.push(component);
                start = i + 1;
            }
        }

        Ok(components)
    }

    fn path_for(components: &[&str], end: usize) -> String {
        if end == 0 {
            "/".to_string()
        } else {
            format!("/{}", components[..end].join("/"))
        }
    }

    fn lookup<'a>(&'a self, components: &[&str]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        for (i, component) in components.iter().enumerate() {
            let prefix = Self::path_for(components, i);
            let children = match current {
                Entry::Dir(children) => children,
                Entry::File(_) => return Err(Error::NotDir(prefix)),
            };
            current = children
                .get(*component)
                .ok_or_else(|| Error::Missing(Self::path_for(components, i + 1)))?;
        }
        Ok(current)
    }

    fn parent_mut<'a>(
        &'a mut self,
        components: &[&str],
    ) -> Result<&'a mut BTreeMap<String, Entry>, Error> {
        let mut current = &mut self.root;
        for (i, component) in components[..components.len() - 1].iter().enumerate() {
            let prefix = Self::path_for(components, i + 1);
            current = match current {
                Entry::Dir(children) => match children.get_mut(*component) {
                    Some(Entry::Dir(children)) => children,
                    Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
                    None => return Err(Error::Missing(prefix)),
                },
                Entry::File(_) => return Err(Error::NotDir(Self::path_for(components, i))),
            };
        }
        Ok(current)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }

        let parent = self.parent_mut(&components)?;
        let name = components[components.len() - 1];
        if parent.contains_key(name) {
            return Err(Error::Exists(Self::path_for(&components, components.len())));
        }
        parent.insert(name.to_string(), Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }

        let parent = self.parent_mut(&components)?;
        let name = components[components.len() - 1];
        let full_path = Self::path_for(&components, components.len());
        match parent.get_mut(name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(full_path)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                parent.insert(name.to_string(), Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::validate(path)?;
        match self.lookup(&components)? {
            Entry::Dir(_) => Err(Error::IsDir(Self::path_for(&components, components.len()))),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::validate(path)?;
        let entry = self.lookup(&components)?;
        match entry {
            Entry::File(_) => Err(Error::NotDir(Self::path_for(&components, components.len()))),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }
}
```
