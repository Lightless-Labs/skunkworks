```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn components(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut components = Vec::new();
        let mut start = 1;
        for (i, component) in path[1..].split('/').enumerate() {
            let offset = if i == 0 {
                1
            } else {
                1 + path[1..].match_indices('/').nth(i - 1).unwrap().0 + 1
            };
            if component.is_empty() || component == "." || component == ".." {
                let invalid_offset = if component.is_empty() {
                    if offset >= path.len() {
                        path.len() - 1
                    } else {
                        offset
                    }
                } else {
                    offset
                };
                return Err(Error::InvalidPath(invalid_offset));
            }
            components.push(component);
            start = offset + component.len() + 1;
        }
        let _ = start;
        Ok(components)
    }

    fn prefix(components: &[&str], count: usize) -> String {
        if count == 0 {
            return "/".to_owned();
        }
        let mut result = String::new();
        for component in &components[..count] {
            result.push('/');
            result.push_str(component);
        }
        result
    }

    fn lookup<'a>(&'a self, components: &[&str]) -> Result<&'a Entry, Error> {
        let mut entry = &self.root;
        for (i, component) in components.iter().enumerate() {
            match entry {
                Entry::Dir(children) => {
                    entry = children
                        .get(*component)
                        .ok_or_else(|| Error::Missing(Self::prefix(components, i + 1)))?;
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::prefix(components, i)));
                }
            }
        }
        Ok(entry)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::components(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let parent_components = &components[..components.len() - 1];
        let parent_path = Self::prefix(&components, components.len() - 1);
        let parent = self.lookup(parent_components)?;
        let children = match parent {
            Entry::Dir(children) => children,
            Entry::File(_) => return Err(Error::NotDir(parent_path)),
        };

        let name = components[components.len() - 1];
        if children.contains_key(name) {
            return Err(Error::Exists(Self::prefix(&components, components.len())));
        }

        let parent = self.lookup_mut(parent_components)?;
        if let Entry::Dir(children) = parent {
            children.insert(name.to_owned(), Entry::Dir(BTreeMap::new()));
        }
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::components(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let parent_components = &components[..components.len() - 1];
        let parent_path = Self::prefix(&components, components.len() - 1);
        match self.lookup(parent_components)? {
            Entry::Dir(_) => {}
            Entry::File(_) => return Err(Error::NotDir(parent_path)),
        }

        let name = components[components.len() - 1];
        let final_path = Self::prefix(&components, components.len());
        let parent = self.lookup_mut(parent_components)?;
        if let Entry::Dir(children) = parent {
            match children.get_mut(name) {
                Some(Entry::Dir(_)) => return Err(Error::IsDir(final_path)),
                Some(Entry::File(contents)) => contents.clear(),
                None => {
                    children.insert(name.to_owned(), Entry::File(Vec::new()));
                }
            }
            if let Some(Entry::File(contents)) = children.get_mut(name) {
                contents.extend_from_slice(data);
            }
        }
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::components(path)?;
        match self.lookup(&components)? {
            Entry::Dir(_) => Err(Error::IsDir(Self::prefix(&components, components.len()))),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::components(path)?;
        match self.lookup(&components)? {
            Entry::File(_) => Err(Error::NotDir(Self::prefix(&components, components.len()))),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }

    fn lookup_mut<'a>(&'a mut self, components: &[&str]) -> Result<&'a mut Entry, Error> {
        let mut entry = &mut self.root;
        for (i, component) in components.iter().enumerate() {
            match entry {
                Entry::Dir(children) => {
                    entry = children
                        .get_mut(*component)
                        .ok_or_else(|| Error::Missing(Self::prefix(components, i + 1)))?;
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::prefix(components, i)));
                }
            }
        }
        Ok(entry)
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
