pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let Some(high) = hex_value(bytes[i + 1]) else {
                return Err(CodecError::BadEscape { at: i });
            };
            let Some(low) = hex_value(bytes[i + 2]) else {
                return Err(CodecError::BadEscape { at: i });
            };
            decoded.push((high << 4) | low);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let at = origins
                .get(error.valid_up_to())
                .copied()
                .unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        if is_unreserved(byte) || component_safe_extra(component, byte) {
            out.push(byte as char);
        } else {
            push_percent(&mut out, byte);
        }
    }
    out
}

    }
    if let Some(query) = query.as_deref() {
        let query_at = authority_end + query_relative.unwrap() + 1;
        validation_components.push((query, query_at));
    }
    if let Some(fragment) = fragment.as_deref() {
        let fragment_at = authority_end + fragment_relative.unwrap() + 1;
        validation_components.push((fragment, fragment_at));
    }

    for (component, offset) in validation_components {
        let kind = if userinfo.as_deref() == Some(component)
            && offset == authority_start
            && at_positions.first().is_some()
        {
            Component::UserInfo
        } else if query.as_deref() == Some(component) {
            Component::Query
        } else if fragment.as_deref() == Some(component) {
            Component::Fragment
        } else {
            Component::Path
        };
        if let Err(error) = decode_component(kind, component) {
            return Err(rebase_parse_error(error, offset));
        }
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|error| match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let mut stack: Vec<&str> = Vec::new();
    let mut last_navigation = false;
    for segment in path[1..].split('/') {
        let decoded = decode_component(Component::Path, segment).map_err(|error| match error {
            CodecError::BadEscape { at } => PathError::BadEscape { at },
            CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
        })?;
        if decoded == "." {
            last_navigation = true;
        } else if decoded == ".." {
            if !stack.is_empty() {
                stack.pop();
            }
            last_navigation = true;
        } else {
            stack.push(segment);
            last_navigation = false;
        }
    }

    if last_navigation {
        stack.push("");
    }

    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let mut base_url = parse_url(base).map_err(ResolveError::Base)?;
    base_url.path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    for (at, &byte) in reference.as_bytes().iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
        }
    }

    let delimiter = reference
        .as_bytes()
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference.as_bytes()[..delimiter].contains(&b':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let (ref_before_fragment, fragment) = match reference.find('#') {
        Some(i) => (&reference[..i], Some(reference[i + 1..].to_owned())),
        None => (reference, None),
    };
    let (reference_path, reference_query) = match ref_before_fragment.find('?') {
        Some(i) => (
            &ref_before_fragment[..i],
            Some(ref_before_fragment[i + 1..].to_owned()),
        ),
        None => (ref_before_fragment, None),
    };

    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_url.path.clone(),
            if reference_query.is_some() {
                reference_query
            } else {
                base_url.query.clone()
            },
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = base_url.path.rfind('/').map(|i| &base_url.path[..=i]).unwrap_or("/");
            format!("{slash}{reference_path}")
        };
        (merged, reference_query)
    };

    let constructed = build_raw_url(
        &base_url,
        &selected_path,
        selected_query.as_deref(),
        fragment.as_deref(),
    );
    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();

    for (field_start, field) in split_with_offsets(input, b'&') {
        if field.is_empty() {
            continue;
        }

        let (raw_key, raw_value, has_equals) = match field.find('=') {
            Some(i) => (&field[..i], &field[i + 1..], true),
            None => (field, "", false),
        };
        let _ = has_equals;

        let key_input = raw_key.replace('+', " ");
        let key = decode_component(Component::Query, &key_input).map_err(|error| {
            rebase_query_error(error, field_start)
        })?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }

        let value_start = field_start + raw_key.len() + 1;
        let value_input = raw_value.replace('+', " ");
        let value = decode_component(Component::Query, &value_input)
            .map_err(|error| rebase_query_error(error, value_start))?;

        if let Some((_, values)) = entries.iter_mut().find(|(existing, _)| *existing == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing_values)) = merged.iter_mut().find(|(existing, _)| existing == key) {
            existing_values.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = form_encode(&key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, form_encode(&value)));
        }
    }
    fields.join("&")
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let normalized_path =
        normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match url.query.as_deref() {
        Some(query) => Some(parse_query(query).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(userinfo) = url.userinfo.as_deref() {
        let decoded =
            decode_component(Component::UserInfo, userinfo).map_err(|error| {
                CanonicalError::Parse(match error {
                    CodecError::BadEscape { at } => ParseError::BadEscape { at },
                    CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at },
                })
            })?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }

    if url.ipv6 {
        let address = url
            .host
            .parse::<std::net::Ipv6Addr>()
            .map_err(|_| CanonicalError::Parse(ParseError::InvalidIpv6 { at: 0 }))?;
        out.push('[');
        out.push_str(&address.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }

    let omit_port = matches!(
        (url.scheme.as_str(), url.port),
        ("http", Some(80)) | ("https", Some(443))
    );
    if let Some(port) = url.port.filter(|_| !omit_port) {
        out.push(':');
        out.push_str(&port.to_string());
    }

    let mut canonical_segments = Vec::new();
    for segment in normalized_path.split('/') {
        let decoded = decode_component(Component::Path, segment).map_err(|error| {
            CanonicalError::Path(match error {
                CodecError::BadEscape { at } => PathError::BadEscape { at },
                CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
            })
        })?;
        canonical_segments.push(encode_unreserved_only(&decoded));
    }
    out.push_str(&canonical_segments.join("/"));

    if let Some(map) = query_map {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }
    if let Some(fragment) = url.fragment.as_deref() {
        let decoded = decode_component(Component::Fragment, fragment).map_err(|error| {
            CanonicalError::Parse(match error {
                CodecError::BadEscape { at } => ParseError::BadEscape { at },
                CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at },
            })
        })?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(out)
}

fn parse_port(bytes: &[u8], start: usize) -> Result<u16, ParseError> {
    if bytes.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }

    if let Some((offset, _)) = bytes.iter().enumerate().find(|(_, byte)| !byte.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: start + offset });
    }
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }

    let mut value: u32 = 0;
    for &byte in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
        {
            Some(value) => value,
            None => return Err(ParseError::InvalidPort { at: start }),
        };
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn component_safe_extra(component: Component, byte: u8) -> bool {
    match component {
        Component::UserInfo => byte == b':',
        Component::Path => byte == b'/',
        Component::Query => matches!(byte, b'/' | b'?'),
        Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
    }
}

fn push_percent(out: &mut String, byte: u8) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    out.push('%');
    out.push(HEX[(byte >> 4) as usize] as char);
    out.push(HEX[(byte & 0x0f) as usize] as char);
}

fn rebase_parse_error(error: CodecError, offset: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

fn rebase_query_error(error: CodecError, offset: usize) -> QueryError {
    match error {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    }
}

fn split_with_offsets(input: &str, delimiter: u8) -> Vec<(usize, &str)> {
    let bytes = input.as_bytes();
    let mut result = Vec::new();
    let mut start = 0;
    for (i, &byte) in bytes.iter().enumerate() {
        if byte == delimiter {
            result.push((start, &input[start..i]));
            start = i + 1;
        }
    }
    result.push((start, &input[start..]));
    result
}

fn form_encode(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        if is_unreserved(byte) {
            out.push(byte as char);
        } else if byte == b' ' {
            out.push('+');
        } else {
            push_percent(&mut out, byte);
        }
    }
    out
}

fn encode_unreserved_only(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        if is_unreserved(byte) {
            out.push(byte as char);
        } else {
            push_percent(&mut out, byte);
        }
    }
    out
}

fn build_raw_url(
    base: &Url,
    path: &str,
    query: Option<&str>,
    fragment: Option<&str>,
) -> String {
    let mut out = format!("{}://", base.scheme);
    if let Some(userinfo) = base.userinfo.as_deref() {
        out.push_str(userinfo);
        out.push('@');
    }
    if base.ipv6 {
        out.push('[');
        out.push_str(&base.host);
        out.push(']');
    } else {
        out.push_str(&base.host);
    }
    if let Some(port) = base.port {
        out.push(':');
        out.push_str(&port.to_string());
    }
    out.push_str(path);
    if let Some(query) = query {
        out.push('?');
        out.push_str(query);
    }
    if let Some(fragment) = fragment {
        out.push('#');
        out.push_str(fragment);
    }
    out
}

    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(h), Some(l)) => {
                    decoded.push((h << 4) | l);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let decoded_at = error.valid_up_to();
            Err(CodecError::InvalidUtf8 {
                at: origins.get(decoded_at).copied().unwrap_or(input.len()),
            })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for byte in text.bytes() {
        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let component_safe = match component {
            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => matches!(byte, b'/' | b'?'),
            Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
        };

        if unreserved || component_safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    validate_global_bytes(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(index) => index,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if colon + 2 >= bytes.len() || bytes[colon + 1] != b'/' || bytes[colon + 2] != b'/' {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(offset, b)| (b == b'@').then_some(offset))
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, userinfo_end) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let host_start = authority_start + userinfo_end;
    let host_port = &authority[userinfo_end..];
    let mut ipv6 = false;
    let (host, port_text, port_start) = if host_port.starts_with('[') {
        let open_at = host_start;
        let close = match host_port.as_bytes().iter().position(|&b| b == b']') {
            Some(index) => index,
            None => return Err(ParseError::InvalidIpv6 { at: open_at }),
        };
        let interior = &host_port[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open_at });
        }
        ipv6 = true;
        let suffix_start = close + 1;
        let suffix = &host_port[suffix_start..];
        let (port, port_offset) = if suffix.is_empty() {
            (None, None)
        } else if suffix.starts_with(':') {
            (Some(&suffix[1..]), Some(host_start + suffix_start + 1))
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + suffix_start,
            });
        };
        (interior.to_ascii_lowercase(), port, port_offset)
    } else {
        let split = host_port.as_bytes().iter().position(|&b| b == b':');
        let (raw_host, port, port_offset) = match split {
            Some(index) => (
                &host_port[..index],
                Some(&host_port[index + 1..]),
                Some(host_start + index + 1),
            ),
            None => (host_port, None, None),
        };
        if raw_host.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (offset, byte) in raw_host.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + offset,
                });
            }
        }
        (raw_host.to_ascii_lowercase(), port, port_offset)
    };

    let port = if let Some(text) = port_text {
        let start = port_start.unwrap_or(authority_end);
        if text.is_empty() {
            return Err(ParseError::InvalidPort { at: start });
        }
        if let Some(offset) = text.bytes().position(|b| !b.is_ascii_digit()) {
            return Err(ParseError::InvalidPort { at: start + offset });
        }
        if text.len() >= 2 && text.as_bytes()[0] == b'0' {
            return Err(ParseError::InvalidPort { at: start });
        }
        let mut value: u32 = 0;
        for byte in text.bytes() {
            value = match value.checked_mul(10).and_then(|v| v.checked_add((byte - b'0') as u32)) {
                Some(v) => v,
                None => return Err(ParseError::InvalidPort { at: start }),
            };
        }
        if !(1..=65535).contains(&value) {
            return Err(ParseError::InvalidPort { at: start });
        }
        Some(value as u16)
    } else {
        None
    };

    let fragment_mark = bytes[authority_end..]
        .iter()
        .position(|&b| b == b'#')
        .map(|offset| authority_end + offset);
    let before_fragment_end = fragment_mark.unwrap_or(bytes.len());
    let query_mark = bytes[authority_end..before_fragment_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|offset| authority_end + offset);

    let path_end = query_mark.unwrap_or(before_fragment_end);
    let raw_path = &input[authority_end..path_end];
    let path = if raw_path.is_empty() {
        "/".to_owned()
    } else {
        raw_path.to_owned()
    };

    let query = query_mark.map(|mark| input[mark + 1..before_fragment_end].to_owned());
    let fragment = fragment_mark.map(|mark| input[mark + 1..].to_owned());

    if let Some(ref raw_userinfo) = userinfo {
        let offset = authority_start;
        decode_component(Component::UserInfo, raw_userinfo)
            .map_err(|error| codec_to_parse(error, offset))?;
    }
    if !raw_path.is_empty() {
        decode_component(Component::Path, raw_path)
            .map_err(|error| codec_to_parse(error, authority_end))?;
    }
    if let (Some(mark), Some(raw_query)) = (query_mark, query.as_deref()) {
        decode_component(Component::Query, raw_query)
            .map_err(|error| codec_to_parse(error, mark + 1))?;
    }
    if let (Some(mark), Some(raw_fragment)) = (fragment_mark, fragment.as_deref()) {
        decode_component(Component::Fragment, raw_fragment)
            .map_err(|error| codec_to_parse(error, mark + 1))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }

    let decoded_path = decode_component(Component::Path, path).map_err(|error| match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let decoded_segments: Vec<&str> = decoded_path[1..].split('/').collect();
    let terminal_navigation = decoded_segments
        .last()
        .map(|segment| *segment == "." || *segment == "..")
        .unwrap_or(false);

    let mut stack: Vec<&str> = Vec::new();
    for (raw, decoded) in raw_segments.iter().zip(decoded_segments.iter()) {
        if *decoded == "." {
            continue;
        } else if *decoded == ".." {
            stack.pop();
        } else {
            stack.push(raw);
        }
    }

    if terminal_navigation {
        stack.push("");
    }

    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn split_reference_components(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let bytes = reference.as_bytes();
    let fragment_mark = bytes.iter().position(|&b| b == b'#');
    let before_fragment_end = fragment_mark.unwrap_or(bytes.len());
    let query_mark = bytes[..before_fragment_end]
        .iter()
        .position(|&b| b == b'?');

    let path_end = query_mark.unwrap_or(before_fragment_end);
    let path = &reference[..path_end];
    let query = query_mark.map(|mark| &reference[mark + 1..before_fragment_end]);
    let fragment = fragment_mark.map(|mark| &reference[mark + 1..]);
    (path, query, fragment)
}

fn normalized_url(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_url = normalized_url(base_url)?;

    validate_global_bytes(reference).map_err(ResolveError::Reference)?;

    let first_delimiter = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delimiter].contains(':') {
        let url = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalized_url(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        return normalized_url(url);
    }

    let (reference_path, reference_query, reference_fragment) =
        split_reference_components(reference);

    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_url.path.clone(),
            if reference_query.is_some() {
                reference_query.map(str::to_owned)
            } else {
                base_url.query.clone()
            },
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = base_url.path.rfind('/').unwrap_or(0);
            format!("{}{}", &base_url.path[..=slash], reference_path)
        };
        (merged, reference_query.map(str::to_owned))
    };

    let userinfo = base_url
        .userinfo
        .as_ref()
        .map(|value| format!("{}@", value))
        .unwrap_or_default();
    let host = if base_url.ipv6 {
        format!("[{}]", base_url.host)
    } else {
        base_url.host.clone()
    };
    let port = base_url
        .port
        .map(|value| format!(":{}", value))
        .unwrap_or_default();
    let query = selected_query
        .as_ref()
        .map(|value| format!("?{}", value))
        .unwrap_or_default();
    let fragment = reference_fragment
        .map(|value| format!("#{}", value))
        .unwrap_or_default();
    let constructed = format!(
        "{}://{}{}{}{}{}{}",
        base_url.scheme, userinfo, host, port, selected_path, query, fragment
    );

    let url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    normalized_url(url)
}

fn form_decode_plus(input: &str) -> String {
    input.replace('+', " ")
}

fn map_codec_query(error: CodecError, base: usize) -> QueryError {
    match error {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut field_start = 0usize;

    for field in input.split('&') {
        let current_start = field_start;
        field_start += field.len() + 1;

        if field.is_empty() {
            continue;
        }

        let equals = field.find('=');
        let (raw_key, raw_value, value_offset) = match equals {
            Some(index) => (&field[..index], &field[index + 1..], index + 1),
            None => (field, "", field.len()),
        };

        let key_input = form_decode_plus(raw_key);
        let key = decode_component(Component::Query, &key_input)
            .map_err(|error| map_codec_query(error, current_start))?;

        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: current_start });
        }

        let value_input = form_decode_plus(raw_value);
        let value = decode_component(Component::Query, &value_input)
            .map_err(|error| map_codec_query(error, current_start + value_offset))?;

        if let Some((_, values)) = entries.iter_mut().find(|(existing, _)| *existing == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

fn form_encode(text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for byte in text.bytes() {
        if byte == b' ' {
            output.push('+');
        } else if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(&str, Vec<&str>)> = Vec::new();

    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, collected)) = merged.iter_mut().find(|(existing, _)| *existing == key) {
            collected.extend(values.iter().map(String::as_str));
        } else {
            merged.push((key.as_str(), values.iter().map(String::as_str).collect()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = form_encode(key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, form_encode(value)));
        }
    }
    fields.join("&")
}

fn encode_unreserved(text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for byte in text.bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match url.query.as_deref() {
        Some(query) => Some(parse_query(query).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut output = format!("{}://", url.scheme);
    if let Some(userinfo) = url.userinfo.as_deref() {
        let decoded = decode_component(Component::UserInfo, userinfo)
            .map_err(|error| CanonicalError::Parse(codec_to_parse(error, 0)))?;
        output.push_str(&encode_component(Component::UserInfo, &decoded));
        output.push('@');
    }

    if url.ipv6 {
        let parsed = Ipv6Addr::from_str(&url.host)
            .map_err(|_| CanonicalError::Parse(ParseError::InvalidIpv6 { at: 0 }))?;
        output.push('[');
        output.push_str(&parsed.to_string());
        output.push(']');
    } else {
        output.push_str(&url.host);
    }

    if let Some(port) = url.port {
        let default = (url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443);
        if !default {
            output.push(':');
            output.push_str(&port.to_string());
        }
    }

    let canonical_segments = path
        .split('/')
        .map(|segment| {
            decode_component(Component::Path, segment)
                .map(|decoded| encode_unreserved(&decoded))
                .map_err(|error| CanonicalError::Path(match error {
                    CodecError::BadEscape { at } => PathError::BadEscape { at },
                    CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
                }))
        })
        .collect::<Result<Vec<_>, _>>()?;
    output.push_str(&canonical_segments.join("/"));

    if let Some(map) = query_map {
        output.push('?');
        output.push_str(&serialise_query(&map));
    }

    if let Some(fragment) = url.fragment.as_deref() {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|error| CanonicalError::Parse(codec_to_parse(error, 0)))?;
        output.push('#');
        output.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(output)
}
