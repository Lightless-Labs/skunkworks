use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let source = input.as_bytes();
    let mut decoded = Vec::with_capacity(source.len());
    let mut offsets = Vec::with_capacity(source.len());
    let mut i = 0;

    while i < source.len() {
        if source[i] == b'%' {
            if i + 2 >= source.len()
                || hex_value(source[i + 1]).is_none()
                || hex_value(source[i + 2]).is_none()
            {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push(
                hex_value(source[i + 1]).unwrap() * 16 + hex_value(source[i + 2]).unwrap(),
            );
            offsets.push(i);
            i += 3;
        } else {
            decoded.push(source[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let decoded_at = error.valid_up_to();
            Err(CodecError::InvalidUtf8 {
                at: offsets.get(decoded_at).copied().unwrap_or(input.len()),
            })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::new();
    for byte in text.bytes() {
        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let extra_safe = match component {
            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => matches!(byte, b'/' | b'?'),
            Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
        };

        if unreserved || extra_safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn validate_global_bytes(input: &str) -> Result<(), ParseError> {
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

fn parse_port(input: &str, start: usize) -> Result<u16, ParseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }

    if let Some(index) = bytes.iter().position(|byte| !byte.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: start + index });
    }

    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }

    let mut value: u32 = 0;
    for byte in bytes {
        value = value
            .saturating_mul(10)
            .saturating_add((byte - b'0') as u32);
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

fn map_codec_parse(error: CodecError, base: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    validate_global_bytes(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|byte| *byte == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for at in 1..colon {
        let byte = bytes[at];
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let after_marker = colon + 3;
    let authority_end = bytes[after_marker..]
        .iter()
        .position(|byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|relative| after_marker + relative)
        .unwrap_or(input.len());
    let authority = &input[after_marker..authority_end];

    let mut at_signs = authority
        .bytes()
        .enumerate()
        .filter(|(_, byte)| *byte == b'@')
        .map(|(index, _)| index);
    let first_at = at_signs.next();
    if let Some(second_at) = at_signs.next() {
        return Err(ParseError::InvalidUserInfo {
            at: after_marker + second_at,
        });
    }

    let (userinfo, host_port, host_port_start) = match first_at {
        Some(index) if index == 0 => {
            return Err(ParseError::InvalidUserInfo {
                at: after_marker + index,
            });
        }
        Some(index) => (
            Some(authority[..index].to_owned()),
            &authority[index + 1..],
            after_marker + index + 1,
        ),
        None => (None, authority, after_marker),
    };

    let mut ipv6 = false;
    let (host_raw, port, host_start) = if host_port.starts_with('[') {
        let close = match host_port.as_bytes().iter().position(|byte| *byte == b']') {
            Some(index) => index,
            None => return Err(ParseError::InvalidIpv6 { at: host_port_start }),
        };
        let interior = &host_port[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_port_start });
        }

        let suffix = &host_port[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(
                &suffix[1..],
                host_port_start + close + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_port_start + close + 1,
            });
        };
        ipv6 = true;
        (interior.to_ascii_lowercase(), port, host_port_start)
    } else {
        let host_colon = host_port.as_bytes().iter().position(|byte| *byte == b':');
        let (host, port) = match host_colon {
            Some(index) => {
                let port = parse_port(&host_port[index + 1..], host_port_start + index + 1)?;
                (&host_port[..index], Some(port))
            }
            None => (host_port, None),
        };

        if host.is_empty() {
            return Err(ParseError::InvalidHost { at: host_port_start });
        }
        for (index, byte) in host.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_port_start + index,
                });
            }
        }
        (host.to_ascii_lowercase(), port, host_port_start)
    };
    let _ = host_start;

    let rest = &input[authority_end..];
    let fragment_relative = rest.find('#');
    let before_fragment = fragment_relative.map(|index| &rest[..index]).unwrap_or(rest);
    let fragment = fragment_relative.map(|index| rest[index + 1..].to_owned());

    let query_relative = before_fragment.find('?');
    let path_raw = query_relative
        .map(|index| &before_fragment[..index])
        .unwrap_or(before_fragment);
    let query = query_relative.map(|index| before_fragment[index + 1..].to_owned());

    let path = if path_raw.is_empty() {
        "/".to_owned()
    } else {
        path_raw.to_owned()
    };

    if let Some(userinfo) = userinfo.as_deref() {
        decode_component(Component::UserInfo, userinfo)
            .map_err(|error| map_codec_parse(error, after_marker))?;
    }

    let path_offset = authority_end;
    if !path_raw.is_empty() {
        decode_component(Component::Path, path_raw)
            .map_err(|error| map_codec_parse(error, path_offset))?;
    }

    if let Some(query) = query.as_deref() {
        let query_offset = authority_end + query_relative.unwrap() + 1;
        decode_component(Component::Query, query)
            .map_err(|error| map_codec_parse(error, query_offset))?;
    }

    if let Some(fragment) = fragment.as_deref() {
        let fragment_offset = authority_end + fragment_relative.unwrap() + 1;
        decode_component(Component::Fragment, fragment)
            .map_err(|error| map_codec_parse(error, fragment_offset))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host: host_raw,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn map_codec_path(error: CodecError) -> PathError {
    match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(map_codec_path)?;

    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_navigation = false;

    for (index, segment) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(map_codec_path)?;
        let is_dot = decoded == ".";
        let is_dotdot = decoded == "..";
        terminal_navigation = index + 1 == segments.len() && (is_dot || is_dotdot);

        if is_dot {
            continue;
        } else if is_dotdot {
            stack.pop();
        } else {
            stack.push(segment);
        }
    }

    if terminal_navigation {
        stack.push("");
    }

    Ok(format!("/{}", stack.join("/")))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    validate_global_bytes(reference).map_err(ResolveError::Reference)?;

    let bytes = reference.as_bytes();
    let first_delimiter = bytes
        .iter()
        .position(|byte| matches!(byte, b'/' | b'?' | b'#'))
        .unwrap_or(bytes.len());

    if bytes[..first_delimiter].contains(&b':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let fragment_start = reference.find('#');
    let before_fragment = fragment_start
        .map(|index| &reference[..index])
        .unwrap_or(reference);
    let fragment = fragment_start.map(|index| reference[index + 1..].to_owned());

    let query_start = before_fragment.find('?');
    let reference_path = query_start
        .map(|index| &before_fragment[..index])
        .unwrap_or(before_fragment);
    let reference_query =
        query_start.map(|index| before_fragment[index + 1..].to_owned());

    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_path,
            if reference_query.is_some() {
                reference_query
            } else {
                base_url.query.clone()
            },
        )
    } else {
        let merged_path = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = base_path.rfind('/').unwrap_or(0);
            format!("{}{}", &base_path[..=slash], reference_path)
        };
        (merged_path, reference_query)
    };

    let userinfo = base_url
        .userinfo
        .as_deref()
        .map(|value| format!("{value}@"))
        .unwrap_or_default();
    let host = if base_url.ipv6 {
        format!("[{}]", base_url.host)
    } else {
        base_url.host.clone()
    };
    let port = base_url
        .port
        .map(|value| format!(":{value}"))
        .unwrap_or_default();
    let query = selected_query
        .as_deref()
        .map(|value| format!("?{value}"))
        .unwrap_or_default();
    let fragment_text = fragment
        .as_deref()
        .map(|value| format!("#{value}"))
        .unwrap_or_default();

    let constructed = format!(
        "{}://{}{}{}{}{}{}",
        base_url.scheme, userinfo, host, port, selected_path, query, fragment_text
    );
    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct QueryMap {
    pub entries: Vec<(String, Vec<String>)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CanonicalError {
    Parse(ParseError),
    Path(PathError),
    Query(QueryError),
}

fn map_codec_query(error: CodecError, base: usize) -> QueryError {
    match error {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: base + at },
    }
}

fn form_decode_part(input: &str, base: usize) -> Result<String, QueryError> {
    let replaced = input.replace('+', " ");
    decode_component(Component::Query, &replaced).map_err(|error| map_codec_query(error, base))
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();

    for (field_start, field) in input.split('&').scan(0usize, |offset, field| {
        let start = *offset;
        *offset += field.len() + 1;
        Some((start, field))
    }) {
        if field.is_empty() {
            continue;
        }

        let (raw_key, raw_value) = match field.find('=') {
            Some(index) => (&field[..index], &field[index + 1..]),
            None => (field, ""),
        };

        let key = form_decode_part(raw_key, field_start)?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }
        let value = form_decode_part(raw_value, field_start + raw_key.len() + usize::from(field.contains('='))?;

        if let Some((_, values)) = entries.iter_mut().find(|(existing, _)| *existing == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

fn form_encode(text: &str) -> String {
    let mut output = String::new();
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for byte in text.bytes() {
        if byte == b' ' {
            output.push('+');
        } else if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(&str, Vec<&str>)> = Vec::new();

    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, collected)) = merged.iter_mut().find(|(existing, _)| *existing == key) {
            collected.extend(values.iter().map(String::as_str));
        } else {
            merged.push((
                key.as_str(),
                values.iter().map(String::as_str).collect(),
            ));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        for value in values {
            fields.push(format!("{}={}", form_encode(key), form_encode(value)));
        }
    }
    fields.join("&")
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let normalized_path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let parsed_query = match url.query.as_deref() {
        Some(query) => Some(parse_query(query).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut output = String::new();
    output.push_str(&url.scheme);
    output.push_str("://");

    if let Some(userinfo) = url.userinfo.as_deref() {
        let decoded = decode_component(Component::UserInfo, userinfo)
            .map_err(|error| CanonicalError::Parse(map_codec_parse(error, 0)))?;
        output.push_str(&encode_component(Component::UserInfo, &decoded));
        output.push('@');
    }

    if url.ipv6 {
        let address = Ipv6Addr::from_str(&url.host).unwrap().to_string();
        output.push('[');
        output.push_str(&address);
        output.push(']');
    } else {
        output.push_str(&url.host);
    }

    if let Some(port) = url.port {
        let omit = (url.scheme == "http" && port == 80)
            || (url.scheme == "https" && port == 443);
        if !omit {
            output.push(':');
            output.push_str(&port.to_string());
        }
    }

    let mut encoded_segments = Vec::new();
    for segment in normalized_path.split('/') {
        let decoded = decode_component(Component::Path, segment)
            .map_err(|error| CanonicalError::Path(map_codec_path(error)))?;
        encoded_segments.push(encode_unreserved(&decoded));
    }
    output.push_str(&encoded_segments.join("/"));

    if let Some(query) = parsed_query {
        output.push('?');
        output.push_str(&serialise_query(&query));
    }
    if let Some(fragment) = url.fragment.as_deref() {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|error| CanonicalError::Parse(map_codec_parse(error, 0)))?;
        output.push('#');
        output.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(output)
}

fn encode_unreserved(text: &str) -> String {
    let mut output = String::new();
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for byte in text.bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}
