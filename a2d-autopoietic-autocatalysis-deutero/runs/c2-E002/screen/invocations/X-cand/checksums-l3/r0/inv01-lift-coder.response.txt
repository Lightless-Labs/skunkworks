#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        update_base(&mut self.crc, &mut self.adler_a, &mut self.adler_b, data);
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let sums = self.checksum();
        format!("c{:08x}:a{:08x}", sums.crc32, sums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut crc = 0u32;
        for (offset, &byte) in bytes[1..9].iter().enumerate() {
            crc = (crc << 4) | u32::from(hex_digit(byte).ok_or(SnapshotError::Hex { offset: offset + 1 })?);
        }
        let mut adler = 0u32;
        for (i, &byte) in bytes[11..19].iter().enumerate() {
            let offset = i + 11;
            adler = (adler << 4) | u32::from(hex_digit(byte).ok_or(SnapshotError::Hex { offset })?);
        }
        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn hex_digit(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn update_base(crc: &mut u32, a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *crc ^= u32::from(byte);
        for _ in 0..8 {
            let low = *crc & 1;
            *crc >>= 1;
            if low != 0 {
                *crc ^= 0xedb8_8320;
            }
        }
        *a = (*a + u32::from(byte)) % 65521;
        *b = (*b + *a) % 65521;
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    update_base(&mut crc, &mut a, &mut b, data);
    crc ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 65521;
        b = (b + a) % 65521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    stream.update_spans(data, spans)?;
    Ok(stream.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u64;
    let mut b = 0u64;
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        let word = (u16::from(pair[0]) << 8) | u16::from(pair[1]);
        a = (a + u64::from(word)) % 65535;
        b = (b + a) % 65535;
    }
    if let Some(&last) = chunks.remainder().first() {
        let word = u16::from(last) << 8;
        a = (a + u64::from(word)) % 65535;
        b = (b + a) % 65535;
    }
    ((b as u32) << 16) | a as u32
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u64;
    let mut b = 0u64;
    let mut pending_high: Option<u8> = None;

    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending_high.take() {
                let word = (u16::from(high) << 8) | u16::from(byte);
                a = (a + u64::from(word)) % 65535;
                b = (b + a) % 65535;
            } else {
                pending_high = Some(byte);
            }
        }
    }

    if let Some(high) = pending_high {
        let word = u16::from(high) << 8;
        a = (a + u64::from(word)) % 65535;
        b = (b + a) % 65535;
    }

    Ok(((b as u32) << 16) | a as u32)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            let mut crc = if variant == Crc16::Xmodem { 0u16 } else { 0xffff };
            for &byte in data {
                crc ^= u16::from(byte) << 8;
                for _ in 0..8 {
                    let high = crc & 0x8000;
                    crc = crc.wrapping_shl(1);
                    if high != 0 {
                        crc ^= 0x1021;
                    }
                }
            }
            crc
        }
        Crc16::X25 => {
            let mut crc = 0xffffu16;
            for &byte in data {
                crc ^= u16::from(byte);
                for _ in 0..8 {
                    let low = crc & 1;
                    crc >>= 1;
                    if low != 0 {
                        crc ^= 0x8408;
                    }
                }
            }
            crc ^ 0xffff
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut crc = 0xffff_ffffu32;
    let mut adler_a = 1u32;
    let mut adler_b = 0u32;
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32_a = 0u64;
    let mut f32_b = 0u64;
    let mut f32_pending: Option<u8> = None;
    let mut crc16_state = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        for &byte in &data[span.start..span.end] {
            update_base(&mut crc, &mut adler_a, &mut adler_b, &[byte]);

            f16_a = (f16_a + u32::from(byte)) % 255;
            f16_b = (f16_b + f16_a) % 255;

            if let Some(high) = f32_pending.take() {
                let word = (u16::from(high) << 8) | u16::from(byte);
                f32_a = (f32_a + u64::from(word)) % 65535;
                f32_b = (f32_b + f32_a) % 65535;
            } else {
                f32_pending = Some(byte);
            }

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    crc16_state ^= u16::from(byte) << 8;
                    for _ in 0..8 {
                        let high = crc16_state & 0x8000;
                        crc16_state = crc16_state.wrapping_shl(1);
                        if high != 0 {
                            crc16_state ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    crc16_state ^= u16::from(byte);
                    for _ in 0..8 {
                        let low = crc16_state & 1;
                        crc16_state >>= 1;
                        if low != 0 {
                            crc16_state ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    if let Some(high) = f32_pending {
        let word = u16::from(high) << 8;
        f32_a = (f32_a + u64::from(word)) % 65535;
        f32_b = (f32_b + f32_a) % 65535;
    }

    if variant == Crc16::X25 {
        crc16_state ^= 0xffff;
    }

    Ok(Multi {
        base: Hashes {
            crc32: crc ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: ((f16_b << 8) | f16_a) as u16,
        fletcher32: ((f32_b as u32) << 16) | f32_a as u32,
        crc16: crc16_state,
    })
}
