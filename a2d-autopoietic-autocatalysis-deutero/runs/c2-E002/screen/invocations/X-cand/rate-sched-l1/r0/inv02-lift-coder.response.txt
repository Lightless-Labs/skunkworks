```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum State {
    Queued,
    Cancelled,
    Completed,
}

#[derive(Debug, Clone)]
struct Entry {
    ready_at: u64,
    priority: u8,
    sequence: u64,
    state: State,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u64,
    jobs: std::collections::HashMap<String, Entry>,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            jobs: std::collections::HashMap::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.contains_key(&job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        self.next_sequence = self.next_sequence.checked_add(1).expect("sequence overflow");
        self.jobs.insert(
            job.id,
            Entry {
                ready_at: job.ready_at,
                priority: job.priority,
                sequence,
                state: State::Queued,
            },
        );
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }

        let entry = self.jobs.get_mut(id).ok_or(Error::Unknown)?;
        match entry.state {
            State::Queued => entry.state = State::Cancelled,
            State::Cancelled => return Err(Error::Cancelled),
            State::Completed => return Err(Error::Completed),
        }

        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }

        let selected = self
            .jobs
            .iter()
            .filter(|(_, entry)| entry.state == State::Queued && entry.ready_at <= now)
            .map(|(id, entry)| {
                (
                    effective_priority(entry.priority, entry.ready_at, now, self.aging_ms),
                    entry.sequence,
                    id.clone(),
                )
            })
            .max_by(|a, b| a.0.cmp(&b.0).then_with(|| b.1.cmp(&a.1)));

        self.last_time = now;

        let Some((effective_priority, _, id)) = selected else {
            return Ok(None);
        };

        if let Some(entry) = self.jobs.get_mut(&id) {
            entry.state = State::Completed;
        }

        Ok(Some(Dispatch {
            id,
            effective_priority,
        }))
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn effective_priority(priority: u8, ready_at: u64, now: u64, aging_ms: u64) -> u8 {
    if aging_ms == 0 {
        return priority;
    }
    let age_steps = (now - ready_at) / aging_ms;
    (priority as u64 + age_steps).min(9) as u8
}
```
