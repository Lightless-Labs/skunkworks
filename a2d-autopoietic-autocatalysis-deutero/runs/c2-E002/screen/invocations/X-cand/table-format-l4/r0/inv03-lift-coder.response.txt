#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let n = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != n {
            return Err(TableError::Ragged {
                row: row_index,
                expected: n,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }

    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let min_col_width = if matches!(options.mode, Mode::Markdown) {
        3
    } else {
        1
    };
    let overhead = match options.mode {
        Mode::Compact => 2 * (n - 1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let minimum = overhead + n * min_col_width;

    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            let chars: Vec<(usize, char)> = cell.char_indices().collect();
            let mut index = 0;
            while index < chars.len() {
                let (byte, ch) = chars[index];
                if ch == '\r' {
                    if index + 1 >= chars.len() || chars[index + 1].1 != '\n' {
                        return Err(TableError::InvalidChar {
                            row: row_index,
                            column: column_index,
                            byte,
                        });
                    }
                    index += 2;
                    continue;
                }
                if (ch.is_control() && ch != '\n') || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
                index += 1;
            }
        }
    }

    let mut logical: Vec<Vec<Vec<Vec<Token>>>> = Vec::with_capacity(rows.len());
    let mut widths = vec![min_col_width; n];

    for row in rows {
        let mut processed_row = Vec::with_capacity(n);
        for (column, cell) in row.iter().enumerate() {
            let normalized = cell.replace("\r\n", "\n");
            let mut lines = Vec::new();
            for line in normalized.split('\n') {
                let trimmed = line.trim_end_matches(' ');
                let mut tokens = Vec::new();
                for ch in trimmed.chars() {
                    if matches!(options.mode, Mode::Markdown) && ch == '|' {
                        tokens.push(Token {
                            text: "\\|".to_owned(),
                            width: 2,
                        });
                    } else if matches!(options.mode, Mode::Markdown) && ch == '\\' {
                        tokens.push(Token {
                            text: "\\\\".to_owned(),
                            width: 2,
                        });
                    } else {
                        tokens.push(Token {
                            text: ch.to_string(),
                            width: 1,
                        });
                    }
                }
                let line_width: usize = tokens.iter().map(|token| token.width).sum();
                widths[column] = widths[column].max(line_width);
                lines.push(tokens);
            }
            processed_row.push(lines);
        }
        logical.push(processed_row);
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let candidate = (0..n)
                .filter(|&column| widths[column] > min_col_width)
                .max_by_key(|&column| (widths[column], column));
            if let Some(column) = candidate {
                widths[column] -= 1;
            } else {
                break;
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let mut output = String::new();

    if matches!(options.mode, Mode::Grid) {
        append_grid_border(&mut output, &widths);
    }

    for (row_index, row) in logical.iter().enumerate() {
        let mut cell_fragments: Vec<Vec<Vec<Token>>> = Vec::with_capacity(n);
        let mut row_height = 1;

        for column in 0..n {
            let mut fragments = Vec::new();
            for line in &row[column] {
                let line_fragments = split_line(line, widths[column], &overflows[column]);
                row_height = row_height.max(line_fragments.len());
                fragments.extend(line_fragments);
            }
            cell_fragments.push(fragments);
        }

        for fragment_index in 0..row_height {
            match options.mode {
                Mode::Grid | Mode::Markdown => {
                    output.push('|');
                    for column in 0..n {
                        output.push(' ');
                        let fragment = cell_fragments[column]
                            .get(fragment_index)
                            .cloned()
                            .unwrap_or_default();
                        append_aligned(
                            &mut output,
                            &fragment,
                            widths[column],
                            &alignments[column],
                            true,
                        );
                        output.push(' ');
                        output.push('|');
                    }
                    output.push('\n');
                }
                Mode::Compact => {
                    for column in 0..n {
                        if column > 0 {
                            output.push_str("  ");
                        }
                        let fragment = cell_fragments[column]
                            .get(fragment_index)
                            .cloned()
                            .unwrap_or_default();
                        append_aligned(
                            &mut output,
                            &fragment,
                            widths[column],
                            &alignments[column],
                            column + 1 != n,
                        );
                    }
                    output.push('\n');
                }
            }
        }

        if matches!(options.mode, Mode::Grid) && options.header && row_index == 0 {
            if rows.len() > 1 {
                append_grid_border(&mut output, &widths);
            }
        }

        if matches!(options.mode, Mode::Markdown) && options.header && row_index == 0 {
            append_markdown_marker(&mut output, &widths, &alignments);
        }
    }

    if matches!(options.mode, Mode::Grid) {
        append_grid_border(&mut output, &widths);
    }

    Ok(output)
}

fn split_line(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<Vec<Token>> {
    if tokens.is_empty() {
        return vec![Vec::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total: usize = tokens.iter().map(|token| token.width).sum();
            if total <= width {
                return vec![tokens.to_vec()];
            }

            let mut fragment = Vec::new();
            let mut used = 0;
            let limit = width.saturating_sub(1);
            for token in tokens {
                if used + token.width > limit {
                    break;
                }
                used += token.width;
                fragment.push(token.clone());
            }
            fragment.push(Token {
                text: "…".to_owned(),
                width: 1,
            });
            vec![fragment]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                result.push(tokens[start..end].to_vec());
                start = end;
            }
            result
        }
    }
}

fn append_aligned(
    output: &mut String,
    tokens: &[Token],
    width: usize,
    alignment: &Alignment,
    include_right_padding: bool,
) {
    let text_width: usize = tokens.iter().map(|token| token.width).sum();
    let extra = width.saturating_sub(text_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };

    for _ in 0..left {
        output.push(' ');
    }
    for token in tokens {
        output.push_str(&token.text);
    }
    if include_right_padding {
        for _ in 0..right {
            output.push(' ');
        }
    }
}

fn append_grid_border(output: &mut String, widths: &[usize]) {
    output.push('+');
    for &width in widths {
        for _ in 0..(width + 2) {
            output.push('-');
        }
        output.push('+');
    }
    output.push('\n');
}

fn append_markdown_marker(
    output: &mut String,
    widths: &[usize],
    alignments: &[Alignment],
) {
    output.push('|');
    for (column, &width) in widths.iter().enumerate() {
        output.push(' ');
        match alignments[column] {
            Alignment::Left => {
                output.push(':');
                for _ in 1..width {
                    output.push('-');
                }
            }
            Alignment::Right => {
                for _ in 1..width {
                    output.push('-');
                }
                output.push(':');
            }
            Alignment::Center => {
                output.push(':');
                for _ in 2..width {
                    output.push('-');
                }
                output.push(':');
            }
        }
        output.push(' ');
        output.push('|');
    }
    output.push('\n');
}
