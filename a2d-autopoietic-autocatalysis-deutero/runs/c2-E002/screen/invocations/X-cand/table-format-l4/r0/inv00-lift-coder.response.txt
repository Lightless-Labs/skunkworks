#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

type LogicalLine = Vec<Token>;
type CellLines = Vec<LogicalLine>;

fn invalid_char(c: char, next: Option<char>) -> bool {
    match c {
        '\r' => next != Some('\n'),
        '\n' => false,
        '\u{2028}' | '\u{2029}' => true,
        _ => c.is_control(),
    }
}

fn validate_cell(
    s: &str,
    row: usize,
    column: usize,
) -> Result<(), TableError> {
    let mut iter = s.char_indices().peekable();
    while let Some((byte, c)) = iter.next() {
        let next = iter.peek().map(|(_, ch)| *ch);
        if invalid_char(c, next) {
            return Err(TableError::InvalidChar { row, column, byte });
        }
        if c == '\r' {
            iter.next();
        }
    }
    Ok(())
}

fn tokenize_line(line: &str, markdown: bool) -> LogicalLine {
    let mut tokens = Vec::new();
    for c in line.chars() {
        if markdown && c == '|' {
            tokens.push(Token { text: "\\|".to_owned(), width: 2 });
        } else if markdown && c == '\\' {
            tokens.push(Token { text: "\\\\".to_owned(), width: 2 });
        } else {
            tokens.push(Token { text: c.to_string(), width: 1 });
        }
    }
    tokens
}

fn prepare_cell(s: &str, markdown: bool) -> CellLines {
    let normalized = s.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            let trimmed = line.trim_end_matches(' ');
            tokenize_line(trimmed, markdown)
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn split_fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<Vec<Token>> {
    if tokens.is_empty() {
        return vec![Vec::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.to_vec()];
            }
            let available = width.saturating_sub(1);
            let mut used = 0;
            let mut prefix = Vec::new();
            for token in tokens {
                if used + token.width > available {
                    break;
                }
                used += token.width;
                prefix.push(token.clone());
            }
            prefix.push(Token { text: "…".to_owned(), width: 1 });
            vec![prefix]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                result.push(tokens[start..end].to_vec());
                start = end;
            }
            result
        }
    }
}

fn fragment_text(tokens: &[Token]) -> String {
    let mut text = String::new();
    for token in tokens {
        text.push_str(&token.text);
    }
    text
}

fn aligned(text: &str, content_width: usize, width: usize, alignment: &Alignment) -> String {
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    format!("{}{}{}", " ".repeat(left), text, " ".repeat(right))
}

fn border(widths: &[usize]) -> String {
    let mut line = String::from("+");
    for width in widths {
        line.push_str(&"-".repeat(width + 2));
        line.push('+');
    }
    line.push('\n');
    line
}

fn data_line(
    fragments: &[Vec<Token>],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    match mode {
        Mode::Compact => {
            let mut line = String::new();
            for col in 0..widths.len() {
                if col > 0 {
                    line.push_str("  ");
                }
                let tokens = fragments.get(col).map(Vec::as_slice).unwrap_or(&[]);
                let text = fragment_text(tokens);
                let content_width = token_width(tokens);
                let mut padded = aligned(&text, content_width, widths[col], &alignments[col]);
                if col + 1 == widths.len() {
                    let right_padding = match alignments[col] {
                        Alignment::Left => widths[col].saturating_sub(content_width),
                        Alignment::Center => (widths[col].saturating_sub(content_width) + 1) / 2,
                        Alignment::Right => 0,
                    };
                    for _ in 0..right_padding {
                        padded.pop();
                    }
                }
                line.push_str(&padded);
            }
            line.push('\n');
            line
        }
        _ => {
            let mut line = String::from("|");
            for col in 0..widths.len() {
                let tokens = fragments.get(col).map(Vec::as_slice).unwrap_or(&[]);
                let text = fragment_text(tokens);
                let content_width = token_width(tokens);
                line.push(' ');
                line.push_str(&aligned(&text, content_width, widths[col], &alignments[col]));
                line.push_str(" |");
            }
            line.push('\n');
            line
        }
    }
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged { row, expected: n, actual: cells.len() });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount { expected: n, actual: options.alignments.len() });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount { expected: n, actual: options.overflow.len() });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let min_col = if matches!(options.mode, Mode::Markdown) { 3 } else { 1 };
    let overhead = if matches!(options.mode, Mode::Compact) {
        2 * (n - 1)
    } else {
        3 * n + 1
    };
    let minimum = overhead + n * min_col;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow { minimum, provided: options.max_width });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            validate_cell(cell, row, column)?;
        }
    }

    let markdown = matches!(options.mode, Mode::Markdown);
    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let prepared: Vec<Vec<CellLines>> = rows
        .iter()
        .map(|row| row.iter().map(|cell| prepare_cell(cell, markdown)).collect())
        .collect();

    let mut widths = vec![min_col; n];
    for row in &prepared {
        for (col, cell) in row.iter().enumerate() {
            for line in cell {
                widths[col] = widths[col].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let candidate = (0..n)
                .rev()
                .find(|&col| widths[col] > min_col)
                .expect("minimum budget was validated");
            widths[candidate] -= 1;
        }
    }

    let mut output = String::new();
    if matches!(options.mode, Mode::Grid) {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in prepared.iter().enumerate() {
        let mut cell_fragments: Vec<Vec<Vec<Token>>> = Vec::with_capacity(n);
        let mut height = 1;
        for (col, cell) in row.iter().enumerate() {
            let mut fragments = Vec::new();
            for line in cell {
                let parts = split_fragments(line, widths[col], &overflows[col]);
                height = height.max(parts.len());
                fragments.extend(parts);
            }
            cell_fragments.push(fragments);
        }

        for fragment_index in 0..height {
            let fragments: Vec<Vec<Token>> = (0..n)
                .map(|col| cell_fragments[col].get(fragment_index).cloned().unwrap_or_default())
                .collect();
            output.push_str(&data_line(&fragments, &widths, &alignments, &options.mode));
        }

        if markdown && row_index == 0 {
            let mut marker = String::from("|");
            for col in 0..n {
                let width = widths[col];
                let segment = match alignments[col] {
                    Alignment::Left => format!(":{}", "-".repeat(width - 1)),
                    Alignment::Right => format!("{}:", "-".repeat(width - 1)),
                    Alignment::Center => format!(":{}:", "-".repeat(width - 2)),
                };
                marker.push(' ');
                marker.push_str(&segment);
                marker.push_str(" |");
            }
            marker.push('\n');
            output.push_str(&marker);
        }

        if matches!(options.mode, Mode::Grid) && options.header && row_index == 0 && rows.len() > 1 {
            output.push_str(&border(&widths));
        }
    }

    if matches!(options.mode, Mode::Grid) {
        output.push_str(&border(&widths));
    }
    Ok(output)
}
