#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
struct Entry {
    job: Job,
    state: State,
    dependencies: Vec<usize>,
    input_order: usize,
    remaining: Option<u64>,
    segment_start: u64,
}

pub struct Pool {
    entries: Vec<Entry>,
    by_id: std::collections::HashMap<String, usize>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    has_run: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base < 1
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut by_id = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if let Some(tag) = &job.tag {
                if !valid_token(tag, 1, 8) {
                    return Err(Error::BadTag(i));
                }
            }
            if by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            by_id.insert(job.id.clone(), i);
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut resolved = Vec::with_capacity(job.dependencies.len());
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                let Some(&index) = by_id.get(dep) else {
                    return Err(Error::MissingDependency(i, j));
                };
                resolved.push(index);
            }
            dependencies.push(resolved);
        }

        fn visit(
            node: usize,
            deps: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[node] == 1 {
                return false;
            }
            if marks[node] == 2 {
                return true;
            }
            marks[node] = 1;
            for &dep in &deps[node] {
                if !visit(dep, deps, marks) {
                    return false;
                }
            }
            marks[node] = 2;
            true
        }

        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if !visit(i, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let entries = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| Entry {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                job,
                dependencies: dependencies[i].clone(),
                input_order: i,
                remaining: None,
                segment_start: 0,
            })
            .collect();

        Ok(Self {
            entries,
            by_id,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let Some(&index) = self.by_id.get(id) else {
            return Err(Error::Unknown);
        };
        Ok(self.entries[index].state.clone())
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if self.has_run && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut lines = Vec::new();
        let first_time = if self.has_run { self.horizon } else { 0 };
        let mut current = first_time;
        let mut first_phase = true;

        loop {
            if !first_phase && current == first_time {
                // The previous successful run already processed this timestamp.
            } else {
                self.process_timestamp(current, &mut lines);
            }
            first_phase = false;

            let Some(next) = self.next_event_after(current, until) else {
                break;
            };
            current = next;
        }

        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }

    fn process_timestamp(&mut self, time: u64, lines: &mut Vec<String>) {
        let mut completions: Vec<usize> = self
            .entries
            .iter()
            .enumerate()
            .filter_map(|(i, entry)| match entry.state {
                State::Running { finish_at, .. } if finish_at == time => Some(i),
                _ => None,
            })
            .collect();
        completions.sort_by(|&a, &b| self.entries[a].job.id.cmp(&self.entries[b].job.id));

        for i in completions {
            let attempt = match self.entries[i].state {
                State::Running { attempt, .. } => attempt,
                _ => continue,
            };
            let id = self.entries[i].job.id.clone();
            if attempt > self.entries[i].job.failures {
                lines.push(format!("{time} finish {id} {attempt} ok"));
                self.entries[i].state = State::Succeeded;
                self.entries[i].remaining = None;
            } else {
                lines.push(format!("{time} finish {id} {attempt} fail"));
                self.entries[i].remaining = None;
                if attempt >= self.entries[i].job.max_attempts {
                    self.entries[i].state = State::Failed;
                } else {
                    let shift = u32::from(attempt - 1);
                    let delay = self
                        .retry_base
                        .checked_mul(1u64.checked_shl(shift).unwrap_or(u64::MAX))
                        .unwrap_or(u64::MAX)
                        .min(self.retry_cap);
                    let ready_at = time.saturating_add(delay);
                    lines.push(format!("{time} retry {id} {attempt} {ready_at}"));
                    self.entries[i].state = State::Waiting { ready_at, attempt };
                }
            }
        }

        loop {
            let newly_blocked: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, entry)| {
                    if matches!(entry.state, State::Waiting { .. })
                        && entry.dependencies.iter().any(|&d| {
                            matches!(
                                self.entries[d].state,
                                State::Failed | State::Blocked
                            )
                        })
                    {
                        Some(i)
                    } else {
                        None
                    }
                })
                .collect();
            if newly_blocked.is_empty() {
                break;
            }
            for &i in &newly_blocked {
                self.entries[i].state = State::Blocked;
                self.entries[i].remaining = None;
            }
            let mut sorted = newly_blocked;
            sorted.sort_by(|&a, &b| self.entries[a].job.id.cmp(&self.entries[b].job.id));
            for i in sorted {
                let dep = self.entries[i]
                    .dependencies
                    .iter()
                    .filter(|&&d| {
                        matches!(
                            self.entries[d].state,
                            State::Failed | State::Blocked
                        )
                    })
                    .min_by_key(|&&d| self.entries[d].job.id.as_str())
                    .map(|&d| self.entries[d].job.id.clone())
                    .unwrap_or_default();
                lines.push(format!("{time} skip {} {dep}", self.entries[i].job.id));
            }
        }

        loop {
            let mut candidates: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, entry)| match entry.state {
                    State::Waiting { ready_at, .. }
                        if ready_at <= time
                            && entry
                                .dependencies
                                .iter()
                                .all(|&d| matches!(self.entries[d].state, State::Succeeded)) =>
                    {
                        Some(i)
                    }
                    _ => None,
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, time);
                let pb = self.effective_priority(b, time);
                pb.cmp(&pa)
                    .then_with(|| self.entries[a].input_order.cmp(&self.entries[b].input_order))
            });

            let mut changed = false;
            for candidate in candidates {
                if let Some(victim) = self.select_victim(candidate) {
                    let victim_id = self.entries[victim].job.id.clone();
                    let candidate_id = self.entries[candidate].job.id.clone();
                    let remaining = match self.entries[victim].state {
                        State::Running { finish_at, .. } => finish_at - time,
                        _ => continue,
                    };
                    let attempt = match self.entries[victim].state {
                        State::Running { attempt, .. } => attempt,
                        _ => continue,
                    };
                    lines.push(format!("{time} preempt {victim_id} by {candidate_id}"));
                    self.entries[victim].state = State::Waiting {
                        ready_at: time,
                        attempt,
                    };
                    self.entries[victim].remaining = Some(remaining);
                    self.start_job(candidate, time, lines);
                    changed = true;
                    break;
                }

                if self.can_start(candidate) {
                    self.start_job(candidate, time, lines);
                    changed = true;
                    break;
                }
            }
            if !changed {
                break;
            }
        }
    }

    fn effective_priority(&self, index: usize, time: u64) -> u8 {
        let entry = &self.entries[index];
        if self.aging_ms == 0 {
            return entry.job.priority;
        }
        let ready_at = match entry.state {
            State::Waiting { ready_at, .. } => ready_at,
            _ => time,
        };
        let age = time.saturating_sub(ready_at) / self.aging_ms;
        (u64::from(entry.job.priority) + age).min(9) as u8
    }

    fn occupied_slots(&self) -> usize {
        self.entries
            .iter()
            .filter(|entry| matches!(entry.state, State::Running { .. }))
            .count()
    }

    fn tag_in_use(&self, tag: &str, excluding: Option<usize>) -> bool {
        self.entries.iter().enumerate().any(|(i, entry)| {
            Some(i) != excluding
                && matches!(entry.state, State::Running { .. })
                && entry.job.tag.as_deref() == Some(tag)
        })
    }

    fn can_start(&self, candidate: usize) -> bool {
        if self.occupied_slots() >= self.capacity {
            return false;
        }
        self.entries[candidate]
            .job
            .tag
            .as_deref()
            .is_none_or(|tag| !self.tag_in_use(tag, None))
    }

    fn select_victim(&self, candidate: usize) -> Option<usize> {
        let cand = &self.entries[candidate].job;
        let mut victims: Vec<usize> = self
            .entries
            .iter()
            .enumerate()
            .filter_map(|(i, entry)| {
                if i == candidate || !matches!(entry.state, State::Running { .. }) {
                    return None;
                }
                let victim = &entry.job;
                if cand.priority < victim.priority.saturating_add(2) {
                    return None;
                }
                let slot_frees = self.occupied_slots() >= self.capacity;
                let tag_frees = cand
                    .tag
                    .as_deref()
                    .is_none_or(|tag| !self.tag_in_use(tag, Some(i)));
                if slot_frees && tag_frees {
                    Some(i)
                } else {
                    None
                }
            })
            .collect();

        victims.sort_by(|&a, &b| {
            self.entries[a]
                .job
                .priority
                .cmp(&self.entries[b].job.priority)
                .then_with(|| {
                    let sa = self.entries[a].segment_start;
                    let sb = self.entries[b].segment_start;
                    sb.cmp(&sa)
                })
                .then_with(|| self.entries[b].job.id.cmp(&self.entries[a].job.id))
        });
        victims.into_iter().next()
    }

    fn start_job(&mut self, index: usize, time: u64, lines: &mut Vec<String>) {
        let (attempt, remaining) = match self.entries[index].state {
            State::Waiting { attempt, .. } => {
                if let Some(remaining) = self.entries[index].remaining.take() {
                    (attempt, remaining)
                } else {
                    let next_attempt = attempt.saturating_add(1);
                    (next_attempt, self.entries[index].job.duration)
                }
            }
            _ => return,
        };
        let finish_at = time.saturating_add(remaining);
        let id = self.entries[index].job.id.clone();
        lines.push(format!("{time} start {id} {attempt} {remaining}"));
        self.entries[index].state = State::Running { attempt, finish_at };
        self.entries[index].segment_start = time;
    }

    fn next_event_after(&self, current: u64, until: u64) -> Option<u64> {
        let mut next: Option<u64> = None;
        for entry in &self.entries {
            let event = match entry.state {
                State::Waiting { ready_at, .. } => Some(ready_at),
                State::Running { finish_at, .. } => Some(finish_at),
                _ => None,
            };
            if let Some(event) = event {
                if event > current && event <= until {
                    next = Some(next.map_or(event, |n| n.min(event)));
                }
            }
        }
        next
    }
}

fn valid_token(s: &str, min_len: usize, max_len: usize) -> bool {
    let bytes = s.as_bytes();
    (min_len..=max_len).contains(&bytes.len())
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}
