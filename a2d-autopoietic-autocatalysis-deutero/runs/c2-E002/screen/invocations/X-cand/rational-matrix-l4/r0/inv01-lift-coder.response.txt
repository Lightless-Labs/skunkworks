#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn normalize_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let negative = (num < 0) ^ (den < 0);
    let n = num.unsigned_abs();
    let d = den.unsigned_abs();
    let g = gcd_u128(n, d);
    let n = n / g;
    let d = d / g;

    if d > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }

    let max_negative = (i64::MAX as u128) + 1;
    let normalized_num = if negative {
        if n > max_negative {
            return Err(RationalError::Overflow);
        }
        if n == max_negative {
            i64::MIN
        } else {
            -(n as i64)
        }
    } else {
        if n > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        n as i64
    };

    Ok(Rational {
        num: normalized_num,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        normalize_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            + (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        normalize_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            - (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        normalize_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        normalize_i128(
            (self.num as i128) * (rhs.num as i128),
            (self.den as i128) * (rhs.den as i128),
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        normalize_i128(
            (self.num as i128) * (rhs.den as i128),
            (self.den as i128) * (rhs.num as i128),
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Rational {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let fail = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(fail(0, ParseKind::Empty));
        }

        let mut i = 0usize;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }
        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        let numerator_end = i;

        if numerator_start == numerator_end {
            if i == bytes.len() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            return Err(fail(i, ParseKind::InvalidByte));
        }

        let mut slash = None;
        let mut denominator_start = 0usize;
        let mut denominator_end = 0usize;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            slash = Some(i);
            i += 1;
            denominator_start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            denominator_end = i;
            if denominator_start == denominator_end {
                if i == bytes.len() {
                    return Err(fail(i, ParseKind::InvalidByte));
                }
                return Err(fail(i, ParseKind::InvalidByte));
            }
            if i < bytes.len() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
        }

        let numerator_digits = &bytes[numerator_start..numerator_end];
        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(fail(
                if negative { numerator_start + 1 } else { numerator_start + 1 },
                ParseKind::NonCanonical,
            ));
        }
        if negative && numerator_digits == b"0" {
            return Err(fail(0, ParseKind::NonCanonical));
        }

        if let Some(slash_at) = slash {
            let denominator_digits = &bytes[denominator_start..denominator_end];
            if denominator_digits.len() >= 2 && denominator_digits[0] == b'0' {
                return Err(fail(denominator_start + 1, ParseKind::NonCanonical));
            }
            let _ = slash_at;
        }

        let max_n = if negative {
            (i64::MAX as u128) + 1
        } else {
            i64::MAX as u128
        };
        let mut nmag = 0u128;
        for &b in numerator_digits {
            nmag = nmag.saturating_mul(10).saturating_add((b - b'0') as u128);
            if nmag > max_n {
                return Err(fail(0, ParseKind::OutOfRange));
            }
        }

        let mut dmag = 1u128;
        if slash.is_some() {
            dmag = 0;
            for &b in &bytes[denominator_start..denominator_end] {
                dmag = dmag
                    .saturating_mul(10)
                    .saturating_add((b - b'0') as u128);
                if dmag > i64::MAX as u128 {
                    return Err(fail(denominator_start, ParseKind::OutOfRange));
                }
            }
            if dmag == 0 {
                return Err(fail(denominator_start, ParseKind::ZeroDenominator));
            }
        }

        if slash.is_some() && dmag == 1 {
            return Err(fail(slash.unwrap(), ParseKind::NonCanonical));
        }
        if slash.is_some() && nmag == 0 {
            return Err(fail(slash.unwrap(), ParseKind::NonCanonical));
        }
        if slash.is_some() && gcd_u128(nmag, dmag) > 1 {
            return Err(fail(slash.unwrap(), ParseKind::NonCanonical));
        }

        let n = if negative {
            if nmag == (i64::MAX as u128) + 1 {
                i64::MIN
            } else {
                -(nmag as i64)
            }
        } else {
            nmag as i64
        };
        Ok(Rational {
            num: n,
            den: dmag as i64,
        })
    }
}

impl std::fmt::Display for Rational {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch {
        coefficients: usize,
        rhs: usize,
    },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

fn eliminate(
    rows: usize,
    cols: usize,
    mut data: Vec<Rational>,
    candidate_cols: usize,
) -> Result<Reduction, MatrixError> {
    let mut p = 0usize;
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();

    for col in 0..candidate_cols {
        if p == rows {
            break;
        }

        let mut chosen: Option<usize> = None;
        for row in p..rows {
            let idx = row * cols + col;
            if data[idx].num == 0 {
                continue;
            }
            let key = (
                data[idx].den,
                data[idx].num.unsigned_abs(),
                row,
            );
            let replace = match chosen {
                None => true,
                Some(old) => {
                    let oldv = data[old * cols + col];
                    key < (oldv.den, oldv.num.unsigned_abs(), old)
                }
            };
            if replace {
                chosen = Some(row);
            }
        }

        let Some(q) = chosen else {
            continue;
        };

        if q != p {
            for c in 0..cols {
                data.swap(p * cols + c, q * cols + c);
            }
            swaps.push((p, q));
        }

        let pivot = data[p * cols + col];
        for c in 0..cols {
            let idx = p * cols + c;
            data[idx] = data[idx].div(&pivot).map_err(|source| {
                MatrixError::Arithmetic {
                    row: p,
                    col: c,
                    step: p,
                    source,
                }
            })?;
        }

        for row in 0..rows {
            if row == p {
                continue;
            }
            let factor = data[row * cols + col];
            if factor.num == 0 {
                continue;
            }
            for c in 0..cols {
                let product = factor
                    .mul(&data[p * cols + c])
                    .map_err(|source| MatrixError::Arithmetic {
                        row,
                        col: c,
                        step: p,
                        source,
                    })?;
                data[row * cols + c] = data[row * cols + c]
                    .sub(&product)
                    .map_err(|source| MatrixError::Arithmetic {
                        row,
                        col: c,
                        step: p,
                        source,
                    })?;
            }
        }

        pivots.push(col);
        p += 1;
    }

    Ok(Reduction {
        matrix: Matrix { rows, cols, data },
        pivots,
        swaps,
    })
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for row in 0..self.cols {
            for col in 0..self.rows {
                data.push(self.data[col * self.cols + row]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let out_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let zero = Rational { num: 0, den: 1 };
        let mut data = Vec::with_capacity(out_len);

        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let left = self.data[row * self.cols + k];
                    let right = rhs.data[k * rhs.cols + col];
                    let term = left.mul(&right).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col,
                            step: k,
                            source,
                        }
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self.rows, self.cols, self.data.clone(), self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let length = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;

        let zero = Rational { num: 0, den: 1 };
        let one = Rational { num: 1, den: 1 };
        let mut augmented = Vec::with_capacity(length);
        for row in 0..n {
            for col in 0..n {
                augmented.push(self.data[row * n + col]);
            }
            for col in 0..n {
                augmented.push(if row == col { one } else { zero });
            }
        }

        let reduction = eliminate(n, width, augmented, n)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: reduction.pivots.len(),
            });
        }

        let mut result = Vec::with_capacity(n * n);
        for row in 0..n {
            for col in 0..n {
                result.push(reduction.matrix.data[row * width + n + col]);
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: result,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }

        let n = self.cols;
        let k = rhs.cols;
        let width = n
            .checked_add(k)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let length = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut augmented = Vec::with_capacity(length);
        for row in 0..self.rows {
            for col in 0..n {
                augmented.push(self.data[row * n + col]);
            }
            for col in 0..k {
                augmented.push(rhs.data[row * k + col]);
            }
        }

        let reduction = eliminate(self.rows, width, augmented, n)
            .map_err(SolveError::Matrix)?;
        let rank = reduction.pivots.len();

        for row in 0..self.rows {
            let coefficient_zero = (0..n)
                .all(|col| reduction.matrix.data[row * width + col].num == 0);
            if coefficient_zero {
                for rhs_col in 0..k {
                    if reduction.matrix.data[row * width + n + rhs_col].num != 0 {
                        let start = row * width;
                        let witness =
                            reduction.matrix.data[start..start + width].to_vec();
                        return Ok(Solution::None {
                            row,
                            rhs_col,
                            witness,
                        });
                    }
                }
            }
        }

        let mut free = Vec::new();
        let mut pivot_mask = vec![false; n];
        for &col in &reduction.pivots {
            pivot_mask[col] = true;
        }
        for col in 0..n {
            if !pivot_mask[col] {
                free.push(col);
            }
        }

        let particular_len = n.checked_mul(k).ok_or_else(|| {
            SolveError::Matrix(MatrixError::ShapeOverflow)
        })?;
        let zero = Rational { num: 0, den: 1 };
        let mut particular = vec![zero; particular_len];
        for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
            for rhs_col in 0..k {
                particular[pivot_col * k + rhs_col] =
                    reduction.matrix.data[pivot_row * width + n + rhs_col];
            }
        }
        let particular_matrix = Matrix {
            rows: n,
            cols: k,
            data: particular,
        };

        if rank == n {
            return Ok(Solution::Unique {
                values: particular_matrix,
                pivots: reduction.pivots,
            });
        }

        let basis_len = n.checked_mul(free.len()).ok_or_else(|| {
            SolveError::Matrix(MatrixError::ShapeOverflow)
        })?;
        let one = Rational { num: 1, den: 1 };
        let mut basis = vec![zero; basis_len];

        for (basis_col, &free_col) in free.iter().enumerate() {
            basis[free_col * free.len() + basis_col] = one;
            for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
                let coefficient =
                    reduction.matrix.data[pivot_row * width + free_col];
                let negative = coefficient.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col,
                        col: basis_col,
                        step: rank,
                        source,
                    })
                })?;
                basis[pivot_col * free.len() + basis_col] = negative;
            }
        }

        Ok(Solution::Infinite {
            particular: particular_matrix,
            free,
            basis: Matrix {
                rows: n,
                cols: reduction.pivots.capacity().min(usize::MAX),
                data: Vec::new(),
            },
            pivots: reduction.pivots,
        })
    }
}
