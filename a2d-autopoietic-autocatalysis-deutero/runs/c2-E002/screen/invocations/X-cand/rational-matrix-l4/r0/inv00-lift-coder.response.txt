use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

const I64_MAX_U: u128 = i64::MAX as u128;
const I64_MIN_MAG_U: u128 = 1u128 << 63;

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_wide(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let neg = (num < 0) ^ (den < 0);
    let nmag = num.unsigned_abs();
    let dmag = den.unsigned_abs();
    let g = gcd_u128(nmag, dmag);
    let n = nmag / g;
    let d = dmag / g;

    if d == 0 || d > I64_MAX_U {
        return Err(RationalError::Overflow);
    }
    if (!neg && n > I64_MAX_U) || (neg && n > I64_MIN_MAG_U) {
        return Err(RationalError::Overflow);
    }

    let out_num = if neg {
        if n == I64_MIN_MAG_U {
            i64::MIN
        } else {
            -(n as i64)
        }
    } else {
        n as i64
    };
    Ok(Rational {
        num: out_num,
        den: d as i64,
    })
}

fn abs_i64_u(n: i64) -> u128 {
    n.unsigned_abs() as u128
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        from_wide(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            + (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        from_wide(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            - (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        from_wide(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.num as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        from_wide(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let n = (self.num as i128) * (rhs.den as i128);
        let d = (self.den as i128) * (rhs.num as i128);
        from_wide(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(RationalError::Parse {
                        at: i,
                        kind: ParseKind::InvalidByte,
                    });
                }
                slash = Some(i);
                continue;
            }
            if b.is_ascii_digit() {
                continue;
            }
            if b == b'-' && i == 0 {
                continue;
            }
            return Err(RationalError::Parse {
                at: i,
                kind: ParseKind::InvalidByte,
            });
        }

        let num_end = slash.unwrap_or(bytes.len());
        let num_start = usize::from(bytes[0] == b'-');
        if num_start == num_end {
            return Err(RationalError::Parse {
                at: text.len(),
                kind: ParseKind::InvalidByte,
            });
        }

        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(RationalError::Parse {
                    at: text.len(),
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let den_start = slash.map(|s| s + 1);
        if let Some(ds) = den_start {
            if ds == bytes.len() {
                return Err(RationalError::Parse {
                    at: text.len(),
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let num_digits = num_end - num_start;
        if num_digits >= 2 && bytes[num_start] == b'0' {
            return Err(RationalError::Parse {
                at: num_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }
        if bytes[0] == b'-' && num_digits == 1 && bytes[1] == b'0' {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::NonCanonical,
            });
        }

        if let Some(ds) = den_start {
            let den_digits = bytes.len() - ds;
            if den_digits >= 2 && bytes[ds] == b'0' {
                return Err(RationalError::Parse {
                    at: ds + 1,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let neg = bytes[0] == b'-';
        let mut nmag: u128 = 0;
        for &b in &bytes[num_start..num_end] {
            nmag = nmag * 10 + (b - b'0') as u128;
            let limit = if neg { I64_MIN_MAG_U } else { I64_MAX_U };
            if nmag > limit {
                return Err(RationalError::Parse {
                    at: 0,
                    kind: ParseKind::OutOfRange,
                });
            }
        }

        let mut dmag: u128 = 1;
        if let Some(ds) = den_start {
            for &b in &bytes[ds..] {
                dmag = dmag * 10 + (b - b'0') as u128;
                if dmag > I64_MAX_U {
                    return Err(RationalError::Parse {
                        at: ds,
                        kind: ParseKind::OutOfRange,
                    });
                }
            }

            if dmag == 0 {
                return Err(RationalError::Parse {
                    at: ds,
                    kind: ParseKind::ZeroDenominator,
                });
            }
            if dmag == 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
            if nmag == 0 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
            if gcd_u128(nmag, dmag) > 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let num = if neg {
            if nmag == I64_MIN_MAG_U {
                i64::MIN
            } else {
                -(nmag as i64)
            }
        } else {
            nmag as i64
        };
        Ok(Self {
            num,
            den: dmag as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(
        rows: usize,
        cols: usize,
        data: Vec<Rational>,
    ) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|v| v.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data
            .get(index)
            .copied()
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let len = self.data.len();
        let mut data = Vec::with_capacity(len);
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    fn eliminate(
        &self,
        candidate_cols: usize,
    ) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..candidate_cols {
            if p == matrix.rows {
                break;
            }

            let mut best: Option<(usize, (i64, u128, usize))> = None;
            for row in p..matrix.rows {
                let value = matrix.data[row * matrix.cols + col];
                if value.num == 0 {
                    continue;
                }
                let key = (value.den, abs_i64_u(value.num), row);
                if best.as_ref().map_or(true, |(_, current)| key < *current) {
                    best = Some((row, key));
                }
            }
            let Some((q, _)) = best else {
                continue;
            };

            if q != p {
                for c in 0..matrix.cols {
                    matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = matrix.data[p * matrix.cols + col];
            let mut normalized = Vec::with_capacity(matrix.cols);
            for c in 0..matrix.cols {
                normalized.push(
                    matrix.data[p * matrix.cols + c]
                        .div(&pivot)
                        .map_err(|source| MatrixError::Arithmetic {
                            row: p,
                            col: c,
                            step: p,
                            source,
                        })?,
                );
            }
            for c in 0..matrix.cols {
                matrix.data[p * matrix.cols + c] = normalized[c];
            }

            for row in 0..matrix.rows {
                if row == p {
                    continue;
                }
                let factor = matrix.data[row * matrix.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..matrix.cols {
                    let product = factor
                        .mul(&matrix.data[p * matrix.cols + c])
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                    matrix.data[row * matrix.cols + c] = matrix.data[row * matrix.cols + c]
                        .sub(&product)
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                }
            }

            pivots.push(col);
            p += 1;
        }

        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }

        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut augmented = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n {
                augmented.push(self.data[r * n + c]);
            }
            for c in 0..n {
                augmented.push(Rational {
                    num: i64::from(r == c),
                    den: 1,
                });
            }
        }

        let reduction = Matrix {
            rows: n,
            cols: width,
            data: augmented,
        }
        .eliminate(n)?;

        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: reduction.pivots.len(),
            });
        }

        let mut inverse = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n {
                inverse.push(reduction.matrix.data[r * width + n + c]);
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: inverse,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }

        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut augmented = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..n {
                augmented.push(self.data[r * n + c]);
            }
            for c in 0..k {
                augmented.push(rhs.data[r * k + c]);
            }
        }

        let reduction = Matrix {
            rows: self.rows,
            cols: width,
            data: augmented,
        }
        .eliminate(n)
        .map_err(SolveError::Matrix)?;

        let rank = reduction.pivots.len();
        for row in 0..self.rows {
            let coefficients_zero = (0..n)
                .all(|c| reduction.matrix.data[row * width + c].num == 0);
            if !coefficients_zero {
                continue;
            }
            for rhs_col in 0..k {
                if reduction.matrix.data[row * width + n + rhs_col].num != 0 {
                    let witness = reduction.matrix.data[row * width..(row + 1) * width].to_vec();
                    return Ok(Solution::None {
                        row,
                        rhs_col,
                        witness,
                    });
                }
            }
        }

        let particular_len = n
            .checked_mul(k)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![Rational { num: 0, den: 1 }; particular_len];
        for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
            for c in 0..k {
                particular_data[pivot_col * k + c] =
                    reduction.matrix.data[pivot_row * width + n + c];
            }
        }
        let particular = Matrix {
            rows: n,
            cols: k,
            data: particular_data,
        };

        let mut free = Vec::new();
        let mut pivot_index = 0usize;
        for col in 0..n {
            if pivot_index < rank && reduction.pivots[pivot_index] == col {
                pivot_index += 1;
            } else {
                free.push(col);
            }
        }

        if rank == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots: reduction.pivots,
            });
        }

        let basis_len = n
            .checked_mul(free.len())
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![Rational { num: 0, den: 1 }; basis_len];
        for (j, &free_col) in free.iter().enumerate() {
            basis_data[free_col * free.len() + j] = Rational { num: 1, den: 1 };
            for (i, &pivot_col) in reduction.pivots.iter().enumerate() {
                let value = reduction.matrix.data[i * width + free_col];
                basis_data[pivot_col * free.len() + j] =
                    value.neg().map_err(|source| SolveError::Matrix(
                        MatrixError::Arithmetic {
                            row: pivot_col,
                            col: j,
                            step: rank,
                            source,
                        },
                    ))?;
            }
        }

        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix {
                rows: n,
                cols: basis_len.checked_div(n.max(1)).unwrap_or(0),
                data: basis_data,
            },
            pivots: reduction.pivots,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}
