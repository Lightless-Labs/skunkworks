use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = normalize_path(path, &self.cwd)?;
        if path == "/" {
            return Err(Error::Exists(path));
        }
        let (parent, name) = split_parent(&path);
        let parent = self.resolve(&parent)?;
        self.require_dir(&parent)?;
        let dest = join_path(&parent, &name);
        if self.entries.contains_key(&dest) {
            return Err(Error::Exists(dest));
        }
        self.entries.insert(dest, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = normalize_path(path, &self.cwd)?;
        let resolved = self.resolve_for_write(&path)?;
        if resolved == "/" {
            return Err(Error::IsDir(resolved));
        }
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let (parent, _) = split_parent(&resolved);
                self.require_dir(&parent)?;
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = normalize_path(path, &self.cwd)?;
        let resolved = self.resolve(&path)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = normalize_path(path, &self.cwd)?;
        let resolved = self.resolve(&path)?;
        self.require_dir(&resolved)?;
        let mut result = Vec::new();
        for (child, entry) in &self.entries {
            if child == "/" {
                continue;
            }
            let (parent, name) = split_parent(child);
            if parent == resolved {
                result.push((name, entry.kind()));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = normalize_path(path, &self.cwd)?;
        let resolved = self.resolve(&path)?;
        self.require_dir(&resolved)?;
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = normalize_path(from, &self.cwd)?;
        let to = normalize_path(to, &self.cwd)?;
        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        let (from_parent, from_name) = split_parent(&from);
        let from_parent = self.resolve(&from_parent)?;
        self.require_dir(&from_parent)?;
        let source = join_path(&from_parent, &from_name);
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && contains_path(&source, &self.cwd) {
            return Err(Error::Busy(source));
        }

        let (to_parent, to_name) = split_parent(&to);
        let to_parent = self.resolve(&to_parent)?;
        self.require_dir(&to_parent)?;
        let destination = join_path(&to_parent, &to_name);

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && contains_path(&destination, &self.cwd)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && is_proper_descendant(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(ref dest_entry) = destination_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if self.has_children(&destination) {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                _ => {}
            }
        }

        if destination_entry.is_some() {
            self.entries.remove(&destination);
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| contains_path(&source, path))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        for (old_path, _) in &moved {
            self.entries.remove(old_path);

                destination.clone()
            } else {
                format!("{destination}{suffix}")
            };
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = normalize_path(path, &self.cwd)?;
        if path == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = split_parent(&path);
        let parent = self.resolve(&parent)?;
        self.require_dir(&parent)?;
        let resolved = join_path(&parent, &name);
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && contains_path(&resolved, &self.cwd) {
            return Err(Error::Busy(resolved));
        }
        if matches!(entry, Entry::Dir) && self.has_children(&resolved) && !recursive {
            return Err(Error::NotEmpty(resolved));
        }

        let removed: Vec<String> = self
            .entries
            .keys()
            .filter(|p| contains_path(&resolved, p))
            .cloned()
            .collect();
        for p in removed {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let link = normalize_path(link, &self.cwd)?;
        validate_target(target)?;
        if link == "/" {
            return Err(Error::Exists(link));
        }
        let (parent, name) = split_parent(&link);
        let parent = self.resolve(&parent)?;
        self.require_dir(&parent)?;
        let resolved = join_path(&parent, &name);
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries
            .insert(resolved, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let path = normalize_path(path, &self.cwd)?;
        if path == "/" {
            return Err(Error::InvalidTarget(0));
        }
        let (parent, name) = split_parent(&path);
        let parent = self.resolve(&parent)?;
        self.require_dir(&parent)?;
        let final_path = join_path(&parent, &name);
        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let path = normalize_path(path, &self.cwd)?;
        let resolved = self.resolve(&path)?;
        let mut rows = Vec::new();
        for (candidate, entry) in &self.entries {
            if contains_path(&resolved, candidate) {
                let size = match entry {
                    Entry::Dir => self.subtree_size(candidate),
                    Entry::File(data) => data.len() as u64,
                    Entry::Link(target) => target.len() as u64,
                };
                rows.push((candidate.clone(), size));
            }
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }

    fn resolve(&self, path: &str) -> Result<String, Error> {
        self.resolve_with_missing_final(path, false)
    }

    fn resolve_for_write(&self, path: &str) -> Result<String, Error> {
        self.resolve_with_missing_final(path, true)
    }

    fn resolve_with_missing_final(
        &self,
        path: &str,
        allow_missing_final: bool,
    ) -> Result<String, Error> {
        let mut current = String::from("/");
        let components = components(path);
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        for (index, component) in components.iter().enumerate() {
            let candidate = join_path(&current, component);
            let entry = match self.entries.get(&candidate) {
                Some(entry) => entry,
                None => {
                    if allow_missing_final && index + 1 == components.len() {
                        return Ok(candidate);
                    }
                    return Err(Error::Missing(candidate));
                }
            };

            if matches!(entry, Entry::Link(_)) {
                let target = match entry {
                    Entry::Link(target) => target,
                    _ => unreachable!(),
                };
                if !visited.insert(candidate.clone()) {
                    return Err(Error::Loop(candidate));
                }
                if expansions >= 8 {
                    return Err(Error::Depth(candidate));
                }
                expansions += 1;

                let base = if target.starts_with('/') {
                    String::from("/")
                } else {
                    current.clone()
                };
                let mut expanded = normalize_target(target, &base)?;
                let remaining = &components[index + 1..];
                for item in remaining {
                    expanded = join_path(&expanded, item);
                }
                let expanded = normalize_absolute(&expanded);
                return self.resolve_from_expansion(
                    &expanded,
                    allow_missing_final,
                    visited,
                    expansions,
                );
            }

            if index + 1 < components.len() && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(candidate));
            }
            current = candidate;
        }
        Ok(current)
    }

    fn resolve_from_expansion(
        &self,
        path: &str,
        allow_missing_final: bool,
        mut visited: BTreeSet<String>,
        mut expansions: usize,
    ) -> Result<String, Error> {
        let mut current = String::from("/");
        let components = components(path);

        for (index, component) in components.iter().enumerate() {
            let candidate = join_path(&current, component);

                None => {
                    if allow_missing_final && index + 1 == components.len() {
                        return Ok(candidate);
                    }
                    return Err(Error::Missing(candidate));
                }
            };

            if let Entry::Link(target) = entry {
                if !visited.insert(candidate.clone()) {
                    return Err(Error::Loop(candidate));
                }
                if expansions >= 8 {
                    return Err(Error::Depth(candidate));
                }

                    String::from("/")
                } else {
                    current.clone()
                };

                }

                    visited,

            Some(Entry::Dir) => Ok(()),
            Some(_) => Err(Error::NotDir(path.to_owned())),
            None => Err(Error::Missing(path.to_owned())),
        }
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries
            .keys()
            .any(|candidate| candidate != path && contains_path(path, candidate))
    }

    fn subtree_size(&self, path: &str) -> u64 {
        self.entries
            .iter()
            .filter(|(candidate, _)| candidate.as_str() != path && contains_path(path, candidate))
            .map(|(_, entry)| match entry {
                Entry::Dir => 0,
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
            })
            .sum()
    }
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(index) = path.as_bytes().iter().position(|b| *b == 0) {
        return Err(Error::InvalidPath(index));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn validate_target(target: &str) -> Result<(), Error> {
    if target.is_empty() {
        return Err(Error::InvalidTarget(0));
    }
    if let Some(index) = target.as_bytes().iter().position(|b| *b == 0) {
        return Err(Error::InvalidTarget(index));
    }
    Ok(())
}

fn normalize_path(path: &str, cwd: &str) -> Result<String, Error> {
    validate_path(path)?;
    let base = if path.starts_with('/') {
        String::from("/")
    } else {
        cwd.to_owned()
    };
    Ok(normalize_with_base(path, &base))
}

fn normalize_target(target: &str, base: &str) -> Result<String, Error> {
    validate_target(target)?;
    let root = if target.starts_with('/') {
        String::from("/")
    } else {
        base.to_owned()
    };
    Ok(normalize_with_base(target, &root))
}

fn normalize_with_base(path: &str, base: &str) -> String {
    let mut parts = components(base);
    for component in path.split('/') {
        match component {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            value => parts.push(value.to_owned()),
        }
    }
    render_components(&parts)
}

fn normalize_absolute(path: &str) -> String {
    let mut parts = Vec::new();
    for component in path.split('/') {
        match component {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            value => parts.push(value.to_owned()),
        }
    }
    render_components(&parts)
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect()
}

fn render_components(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn split_parent(path: &str) -> (String, String) {
    if path == "/" {
        return ("/".to_owned(), String::new());
    }
    match path.rfind('/') {
        Some(0) => ("/".to_owned(), path[1..].to_owned()),
        Some(index) => (path[..index].to_owned(), path[index + 1..].to_owned()),
        None => ("/".to_owned(), path.to_owned()),
    }
}

fn join_path(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{name}")
    } else {
        format!("{parent}/{name}")
    }
}

fn contains_path(parent: &str, candidate: &str) -> bool {
    if parent == "/" {
        candidate != "/" && candidate.starts_with('/')
    } else {
        candidate == parent
            || candidate
                .strip_prefix(parent)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }
}

fn is_proper_descendant(candidate: &str, parent: &str) -> bool {
    candidate != parent && contains_path(parent, candidate)
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));

    if path.is_empty() {
        return Err(Error::InvalidPath(0));

fn normalize_from(base: &str, path: &str) -> String {
    let mut parts: Vec<&str> = Vec::new();
    if !path.starts_with('/') {
        parts.extend(base.split('/').filter(|s| !s.is_empty()));
    }
    for component in path.split('/') {
        match component {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other),
        }
    }
    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn parent_of(path: &str) -> String {
    if path == "/" {
        return "/".to_string();
    }
    match path.rfind('/') {
        Some(0) | None => "/".to_string(),
        Some(i) => path[..i].to_string(),
    }
}

fn basename(path: &str) -> &str {
    path.rsplit('/').next().unwrap_or("")
}

fn is_within(path: &str, directory: &str) -> bool {
    if directory == "/" {
        return path.starts_with('/');
    }
    path == directory
        || (path.starts_with(directory)
            && path.as_bytes().get(directory.len()) == Some(&b'/'))
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn entry_kind(entry: &Entry) -> Kind {
        match entry {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }

    fn resolve(&self, path: &str, follow_final: bool, create_final: bool) -> Result<String, Error> {
        let mut pending: Vec<String> = path
            .split('/')
            .filter(|s| !s.is_empty())
            .map(str::to_string)
            .collect();
        let mut physical: Vec<String> = Vec::new();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        while !pending.is_empty() {
            let component = pending.remove(0);
            physical.push(component);
            let candidate = format!("/{}", physical.join("/"));
            let final_component = pending.is_empty();
            match self.entries.get(&candidate) {
                None if create_final && final_component => return Ok(candidate),
                None => return Err(Error::Missing(candidate)),
                Some(Entry::Dir) => {}
                Some(_) if !final_component => return Err(Error::NotDir(candidate)),
                Some(Entry::Link(target)) if follow_final || !final_component => {
                    if !visited.insert(candidate.clone()) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    expansions += 1;

                    let parent = parent_of(&candidate);
                    let mut expanded = normalize_from(&parent, target);
                    if !pending.is_empty() {
                        if expanded != "/" {
                            expanded.push('/');
                        }
                        expanded.push_str(&pending.join("/"));
                    }
                    let normalized = normalize_from("/", &expanded);
                    pending = normalized
                        .split('/')
                        .filter(|s| !s.is_empty())
                        .map(str::to_string)
                        .collect();
                    physical.clear();
                }
                Some(_) => {}
            }
        }

        let resolved = if physical.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", physical.join("/"))
        };
        if !self.entries.contains_key(&resolved) {
            if create_final {
                return Ok(resolved);
            }
            return Err(Error::Missing(resolved));
        }
        Ok(resolved)
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        if path == "/" {
            return Ok(("/".to_string(), String::new()));
        }
        let parent = parent_of(path);
        let name = basename(path).to_string();
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, name)),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn children(&self, path: &str) -> Vec<String> {
        let prefix = if path == "/" {
            "/".to_string()
        } else {
            format!("{}/", path)
        };
        self.entries
            .keys()
            .filter(|p| {
                p.starts_with(&prefix)
                    && *p != path
                    && !p[prefix.len()..].contains('/')
            })
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        validate_path(path)?;
        let normalized = normalize_from(&self.cwd, path);
        let (parent, name) = self.resolve_parent(&normalized)?;
        let destination = if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        validate_path(path)?;
        let normalized = normalize_from(&self.cwd, path);
        let resolved = self.resolve(&normalized, true, true)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) | None => {
                let (parent, _) = self.resolve_parent(&resolved)?;
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                if !self.entries.contains_key(&parent) {
                    return Err(Error::Missing(parent));
                }
                Ok(())
            }
            Some(Entry::Link(_)) => Err(Error::Loop(resolved)),
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        validate_path(path)?;
        let normalized = normalize_from(&self.cwd, path);
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => Err(Error::Loop(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        validate_path(path)?;
        let normalized = normalize_from(&self.cwd, path);

                    let kind = Self::entry_kind(&self.entries[&p]);
                    (basename(&p).to_string(), kind)
                })
                .collect()),
            _ => Err(Error::NotDir(resolved)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        validate_path(path)?;
        let normalized = normalize_from(&self.cwd, path);
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            _ => Err(Error::NotDir(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        validate_path(from)?;
        validate_path(to)?;
        let from_normalized = normalize_from(&self.cwd, from);
        let to_normalized = normalize_from(&self.cwd, to);
        if from_normalized == "/" || to_normalized == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&from_normalized)?;
        let source = if source_parent == "/" {
            format!("/{}", source_name)
        } else {
            format!("{}/{}", source_parent, source_name)
        };
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let (dest_parent, dest_name) = self.resolve_parent(&to_normalized)?;
        let destination = if dest_parent == "/" {
            format!("/{}", dest_name)
        } else {
            format!("{}/{}", dest_parent, dest_name)
        };
        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && is_within(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && is_within(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let existing = self.entries.get(&destination).cloned();
        if existing.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }
        if let Some(existing_entry) = existing.as_ref() {
            match (&source_entry, existing_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&destination).is_empty() {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| is_within(p, &source))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();

        if existing.is_some() {
            let remove_dest: Vec<String> = self
                .entries
                .keys()
                .filter(|p| is_within(p, &destination))
                .cloned()
                .collect();
            for p in remove_dest {
                self.entries.remove(&p);
            }
        }
        for (old, _) in &moving {
            self.entries.remove(old);
        }
        for (old, entry) in moving {
            let suffix = &old[source.len()..];
            let new_path = format!("{}{}", destination, suffix);
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        validate_path(path)?;
        let normalized = normalize_from(&self.cwd, path);
        if normalized == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&normalized)?;
        let target = if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        let entry = self
            .entries
            .get(&target)
            .ok_or_else(|| Error::Missing(target.clone()))?;
        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        let children = self.children(&target);
        if matches!(entry, Entry::Dir) && !recursive && !children.is_empty() {
            return Err(Error::NotEmpty(target));
        }
        let remove: Vec<String> = self
            .entries
            .keys()
            .filter(|p| is_within(p, &target))
            .cloned()
            .collect();
        for p in remove {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        validate_path(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let normalized = normalize_from(&self.cwd, link);
        let (parent, name) = self.resolve_parent(&normalized)?;
        let destination = if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries
            .insert(destination, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        validate_path(path)?;
        let normalized = normalize_from(&self.cwd, path);
        if normalized == "/" {
            return Err(Error::InvalidTarget(0));
        }
        let (parent, name) = self.resolve_parent(&normalized)?;
        let final_path = if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        validate_path(path)?;
        let normalized = normalize_from(&self.cwd, path);
        let resolved = self.resolve(&normalized, true, false)?;
        let entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;
        if !matches!(entry, Entry::Dir | Entry::File(_) | Entry::Link(_)) {
            return Err(Error::NotDir(resolved));
        }

        let mut paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| is_within(p, &resolved))
            .cloned()
            .collect();
        paths.sort();

        let mut sizes = BTreeMap::new();
        for p in &paths {
            let size = match &self.entries[p] {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            sizes.insert(p.clone(), size);
        }

        for p in paths.iter().rev() {
            let size = sizes[p];
            if size == 0 {
                continue;
            }
            let parent = parent_of(p);
            if is_within(&parent, &resolved) && parent != *p {
                if let Some(parent_size) = sizes.get_mut(&parent) {
                    *parent_size += size;
                }
            }
        }

        Ok(paths
            .into_iter()
            .map(|p| {
                let size = sizes.get(&p).copied().unwrap_or(0);
                (p, size)
            })
            .collect())
    }
}

            Some(Entry::File(_)) => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {

                Ok(())

        let resolved_destination_parent = self.resolve_full(&destination_parent)?;
        self.require_dir(&resolved_destination_parent)?;
        let resolved_destination = join_path(&resolved_destination_parent, &destination_name);

        if matches!(self.entries.get(&resolved_destination), Some(Entry::Dir))
            && is_within(&self.cwd, &resolved_destination)
        {
            return Err(Error::Busy(resolved_destination));
        }

        if resolved_source == resolved_destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir)
            && is_proper_descendant(&resolved_destination, &resolved_source)
        {
            return Err(Error::Cycle(resolved_destination));
        }

        let destination_entry = self.entries.get(&resolved_destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(resolved_destination));
        }

        if let Some(ref target_entry) = destination_entry {
            match (&source_entry, target_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if self.has_children(&resolved_destination) {
                        return Err(Error::NotEmpty(resolved_destination));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(resolved_destination)),
                (_, Entry::Dir) => return Err(Error::IsDir(resolved_destination)),
                _ => {}
            }
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| is_within(path, &resolved_source))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if destination_entry.is_some() {
            self.entries
                .retain(|path, _| !is_within(path, &resolved_destination));
        }
        self.entries
            .retain(|path, _| !is_within(path, &resolved_source));

        for (old_path, entry) in moved {
            let suffix = old_path
                .strip_prefix(&resolved_source)
                .expect("source subtree prefix");
            let new_path = format!("{resolved_destination}{suffix}");
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = split_path(&canonical);
        let resolved_parent = self.resolve_full(&parent)?;
        self.require_dir(&resolved_parent)?;
        let resolved = join_path(&resolved_parent, &name);
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &resolved) {
            return Err(Error::Busy(resolved));
        }
        if matches!(entry, Entry::Dir) && self.has_children(&resolved) && !recursive {
            return Err(Error::NotEmpty(resolved));
        }

        self.entries.retain(|candidate, _| !is_within(candidate, &resolved));
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical_link = self.normalize(link)?;
        validate_target(target)?;

        if canonical_link == "/" {
            return Err(Error::Exists(canonical_link));
        }
        let (parent, name) = split_path(&canonical_link);
        let resolved_parent = self.resolve_full(&parent)?;
        self.require_dir(&resolved_parent)?;
        let resolved_link = join_path(&resolved_parent, &name);
        if self.entries.contains_key(&resolved_link) {
            return Err(Error::Exists(resolved_link));
        }
        self.entries
            .insert(resolved_link, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::InvalidTarget(0));
        }
        let (parent, name) = split_path(&canonical);
        let resolved_parent = self.resolve_full(&parent)?;
        self.require_dir(&resolved_parent)?;
        let resolved = join_path(&resolved_parent, &name);
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve_full(&canonical)?;
        let entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;
        if !matches!(entry, Entry::Dir | Entry::File(_) | Entry::Link(_)) {
            return Err(Error::Missing(resolved));
        }

        let mut rows = Vec::new();
        let mut descendants: Vec<(&String, &Entry)> = self
            .entries
            .iter()
            .filter(|(entry_path, _)| is_within(entry_path, &resolved))
            .collect();
        descendants.sort_by(|a, b| a.0.cmp(b.0));

        for (entry_path, entry) in descendants {
            let size = match entry {
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => self
                    .entries
                    .iter()
                    .filter(|(candidate, child)| {
                        matches!(child, Entry::File(_) | Entry::Link(_))
                            && is_within(candidate, entry_path)
                    })
                    .map(|(_, child)| match child {
                        Entry::File(bytes) => bytes.len() as u64,
                        Entry::Link(target) => target.len() as u64,
                        Entry::Dir => 0,
                    })
                    .sum(),
            };
            rows.push((entry_path.clone(), size));
        }
        Ok(rows)
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(index) = path.as_bytes().iter().position(|byte| *byte == 0) {
            return Err(Error::InvalidPath(index));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let absolute = path.starts_with('/');
        let mut components: Vec<String> = if absolute {
            Vec::new()
        } else {
            components(&self.cwd)
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                other => components.push(other.to_string()),
            }
        }
        Ok(make_path(&components))
    }

    fn require_dir(&self, path: &str) -> Result<(), Error> {
        match self.entries.get(path) {
            Some(Entry::Dir) => Ok(()),
            Some(_) => Err(Error::NotDir(path.to_string())),
            None => Err(Error::Missing(path.to_string())),
        }
    }

    fn resolve_full(&self, path: &str) -> Result<String, Error> {
        self.resolve(path, true)
    }

    fn resolve(&self, path: &str, follow_final: bool) -> Result<String, Error> {
        let mut current = path.to_string();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        loop {
            let parts = components(&current);
            let mut prefix_parts: Vec<String> = Vec::new();
            let mut restarted = false;

            for (index, component) in parts.iter().enumerate() {
                prefix_parts.push(component.clone());
                let prefix = make_path(&prefix_parts);
                let is_final = index + 1 == parts.len();

                match self.entries.get(&prefix) {
                    None => return Err(Error::Missing(prefix)),
                    Some(Entry::Dir) => {}
                    Some(Entry::File(_)) => {
                        if !is_final {
                            return Err(Error::NotDir(prefix));
                        }
                    }
                    Some(Entry::Link(target)) if !is_final || follow_final => {
                        if !visited.insert(prefix.clone()) {
                            return Err(Error::Loop(prefix));
                        }
                        if expansions >= 8 {
                            return Err(Error::Depth(prefix));
                        }
                        expansions += 1;

                        let target_base = if target.starts_with('/') {
                            Vec::new()
                        } else {
                            prefix_parts[..prefix_parts.len() - 1].to_vec()
                        };
                        let mut expanded = target_base;
                        for target_part in target.split('/') {
                            match target_part {
                                "" | "." => {}
                                ".." => {
                                    expanded.pop();
                                }
                                other => expanded.push(other.to_string()),
                            }
                        }
                        expanded.extend(parts[index + 1..].iter().cloned());
                        current = make_path(&expanded);
                        restarted = true;
                        break;
                    }
                    Some(Entry::Link(_)) => {}
                }
            }

            if !restarted {
                return Ok(current);
            }
        }
    }

    fn resolve_write(&self, path: &str) -> Result<String, Error> {
        if path == "/" {
            return Ok("/".to_string());
        }

        let mut current = path.to_string();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        loop {
            let parts = components(&current);
            let mut prefix_parts: Vec<String> = Vec::new();
            let mut restarted = false;

            for (index, component) in parts.iter().enumerate() {
                prefix_parts.push(component.clone());
                let prefix = make_path(&prefix_parts);
                let is_final = index + 1 == parts.len();

                match self.entries.get(&prefix) {
                    None if is_final => return Ok(prefix),
                    None => return Err(Error::Missing(prefix)),
                    Some(Entry::Dir) => {}
                    Some(Entry::File(_)) => {
                        if !is_final {
                            return Err(Error::NotDir(prefix));
                        }
                    }
                    Some(Entry::Link(target)) => {
                        if !visited.insert(prefix.clone()) {
                            return Err(Error::Loop(prefix));
                        }
                        if expansions >= 8 {
                            return Err(Error::Depth(prefix));
                        }
                        expansions += 1;

                        let target_base = if target.starts_with('/') {
                            Vec::new()
                        } else {
                            prefix_parts[..prefix_parts.len() - 1].to_vec()
                        };
                        let mut expanded = target_base;
                        for target_part in target.split('/') {
                            match target_part {
                                "" | "." => {}
                                ".." => {
                                    expanded.pop();
                                }
                                other => expanded.push(other.to_string()),
                            }
                        }
                        expanded.extend(parts[index + 1..].iter().cloned());
                        current = make_path(&expanded);
                        restarted = true;
                        break;
                    }
                }
            }

            if !restarted {
                return Ok(current);
            }
        }
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries.keys().any(|candidate| {
            candidate != path && is_proper_descendant(candidate, path)
        })
    }
}

fn validate_target(target: &str) -> Result<(), Error> {
    if target.is_empty() {
        return Err(Error::InvalidTarget(0));
    }
    if let Some(index) = target.as_bytes().iter().position(|byte| *byte == 0) {
        return Err(Error::InvalidTarget(index));
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    if path == "/" {
        Vec::new()
    } else {
        path.split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_string)
            .collect()
    }
}

fn make_path(components: &[String]) -> String {
    if components.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", components.join("/"))
    }
}

fn split_path(path: &str) -> (String, String) {
    let index = path.rfind('/').expect("canonical absolute path");
    let name = path[index + 1..].to_string();
    let parent = if index == 0 {
        "/".to_string()
    } else {
        path[..index].to_string()
    };
    (parent, name)
}

fn join_path(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{name}")
    } else {
        format!("{parent}/{name}")
    }
}

fn is_within(path: &str, root: &str) -> bool {
    if root == "/" {
        return path.starts_with('/');
    }
    path == root
        || path
            .strip_prefix(root)
            .map(|suffix| suffix.starts_with('/'))
            .unwrap_or(false)
}

fn is_proper_descendant(path: &str, root: &str) -> bool {
    path != root && is_within(path, root)
}

use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {

        Self::new()
    }

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        let (parent, name) = split_parent(&canonical);
        let physical_parent = self.resolve_full(&parent)?;
        self.require_dir(&physical_parent)?;

        let target = join_child(&physical_parent, &name);
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        let (parent, name) = split_parent(&canonical);
        let physical_parent = self.resolve_full(&parent)?;
        self.require_dir(&physical_parent)?;

        let target = join_child(&physical_parent, &name);
        let resolved = self.resolve_from(target.clone(), true, true)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve_full(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) | None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve_full(&canonical)?;
        self.require_dir(&resolved)?;

        let mut result = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == &resolved {
                continue;
            }
            if let Some(name) = immediate_child_name(&resolved, entry_path) {
                result.push((name.to_owned(), entry.kind()));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve_full(&canonical)?;
        self.require_dir(&resolved)?;
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = normalize(from, &self.cwd)?;
        let to = normalize(to, &self.cwd)?;
        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        let (from_parent, from_name) = split_parent(&from);
        let source_parent = self.resolve_full(&from_parent)?;
        self.require_dir(&source_parent)?;
        let source = join_child(&source_parent, &from_name);
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let (to_parent, to_name) = split_parent(&to);
        let dest_parent = self.resolve_full(&to_parent)?;
        self.require_dir(&dest_parent)?;
        let dest = join_child(&dest_parent, &to_name);

        if matches!(self.entries.get(&dest), Some(Entry::Dir)) && is_within(&self.cwd, &dest) {
            return Err(Error::Busy(dest));
        }

        if source == dest {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && is_within(&dest, &source) && dest != source {
            return Err(Error::Cycle(dest));
        }

        let dest_entry = self.entries.get(&dest).cloned();
        if dest_entry.is_some() && !overwrite {
            return Err(Error::Exists(dest));
        }

        if let Some(existing) = &dest_entry {
            match (&source_entry, existing) {
                (Entry::Dir, Entry::Dir) => {
                    if self.has_children(&dest) {
                        return Err(Error::NotEmpty(dest));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dest)),
                (_, Entry::Dir) => return Err(Error::IsDir(dest)),
                _ => {}
            }
        }

        let moved = self.collect_subtree(&source);
        if dest_entry.is_some() {
            self.remove_subtree_entries(&dest);
        }
        self.remove_subtree_entries(&source);
        for (old_path, entry) in moved {
            let suffix = old_path.strip_prefix(&source).unwrap_or("");
            let new_path = format!("{dest}{suffix}");
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        if canonical == "/" {
            return Err(Error::Root);
        }

        let (parent, name) = split_parent(&canonical);
        let physical_parent = self.resolve_full(&parent)?;
        self.require_dir(&physical_parent)?;
        let target = join_child(&physical_parent, &name);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && !recursive && self.has_children(&target) {
            return Err(Error::NotEmpty(target));
        }
        self.remove_subtree_entries(&target);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical_link = normalize(link, &self.cwd)?;
        validate_target(target)?;

        let (parent, name) = split_parent(&canonical_link);
        let physical_parent = self.resolve_full(&parent)?;
        self.require_dir(&physical_parent)?;
        let destination = join_child(&physical_parent, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries
            .insert(destination, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let (parent, name) = split_parent(&canonical);
        let physical_parent = self.resolve_full(&parent)?;
        self.require_dir(&physical_parent)?;
        let target_path = join_child(&physical_parent, &name);
        match self.entries.get(&target_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(target_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve_full(&canonical)?;
        let entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        let paths: Vec<(String, &Entry)> = self
            .entries
            .iter()
            .filter(|(candidate, _)| **candidate == resolved || is_within(candidate, &resolved))
            .map(|(p, e)| (p.clone(), e))
            .collect();

        let mut rows = Vec::with_capacity(paths.len());
        for (candidate, entry) in paths {
            let size = match entry {
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => self
                    .entries
                    .iter()
                    .filter(|(p, e)| {
                        matches!(e, Entry::File(_) | Entry::Link(_))
                            && is_within(p, &candidate)
                    })
                    .map(|(_, e)| match e {
                        Entry::File(bytes) => bytes.len() as u64,
                        Entry::Link(target) => target.len() as u64,
                        Entry::Dir => 0,
                    })
                    .sum(),
            };
            rows.push((candidate, size));
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        let _ = entry;
        Ok(rows)
    }

    fn require_dir(&self, path: &str) -> Result<(), Error> {
        match self.entries.get(path) {
            Some(Entry::Dir) => Ok(()),
            Some(_) => Err(Error::NotDir(path.to_owned())),
            None => Err(Error::Missing(path.to_owned())),
        }
    }

    fn resolve_full(&self, path: &str) -> Result<String, Error> {
        self.resolve_from(path.to_owned(), true, false)
    }

    fn resolve_from(
        &self,
        initial: String,
        follow_final: bool,
        allow_missing_final: bool,
    ) -> Result<String, Error> {
        let mut pending = components(&initial);
        let mut current: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                return Ok(path_from_components(&current));
            }

            let component = pending.remove(0);
            current.push(component);
            let candidate = path_from_components(&current);
            let final_component = pending.is_empty();

            match self.entries.get(&candidate) {
                None => {
                    if allow_missing_final && final_component {
                        return Ok(candidate);
                    }
                    return Err(Error::Missing(candidate));
                }
                Some(Entry::Dir) => {
                    if !final_component {
                        continue;
                    }
                    return Ok(candidate);
                }
                Some(Entry::File(_)) => {
                    if !final_component {
                        return Err(Error::NotDir(candidate));
                    }
                    return Ok(candidate);
                }
                Some(Entry::Link(target)) => {
                    if final_component && !follow_final {
                        return Ok(candidate);
                    }
                    if !visited.insert(candidate.clone()) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    expansions += 1;

                    let mut expanded = if target.starts_with('/') {
                        normalize(target, "/")?
                    } else {
                        let parent = path_from_components(
                            &current[..current.len().saturating_sub(1)],
                        );
                        normalize(target, &parent)?
                    };
                    let mut combined = components(&expanded);
                    combined.extend(pending.into_iter());
                    expanded = path_from_components(&combined);
                    pending = components(&expanded);
                    current.clear();
                }
            }
        }
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries
            .keys()
            .any(|candidate| candidate != path && is_within(candidate, path))
    }

    fn collect_subtree(&self, path: &str) -> Vec<(String, Entry)> {
        self.entries
            .iter()
            .filter(|(candidate, _)| **candidate == path || is_within(candidate, path))
            .map(|(candidate, entry)| (candidate.clone(), entry.clone()))
            .collect()
    }

    fn remove_subtree_entries(&mut self, path: &str) {
        let doomed: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| **candidate == path || is_within(candidate, path))
            .cloned()
            .collect();
        for candidate in doomed {
            self.entries.remove(&candidate);
        }
    }
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(index) = path.as_bytes().iter().position(|byte| *byte == 0) {
        return Err(Error::InvalidPath(index));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn normalize(path: &str, cwd: &str) -> Result<String, Error> {
    validate_path(path)?;
    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        components(cwd)
    };

    for component in path.split('/') {
        match component {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other.to_owned()),
        }
    }
    Ok(path_from_components(&parts))
}

fn validate_target(target: &str) -> Result<(), Error> {
    if target.is_empty() {
        return Err(Error::InvalidTarget(0));
    }
    if let Some(index) = target.as_bytes().iter().position(|byte| *byte == 0) {
        return Err(Error::InvalidTarget(index));
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect()
}

fn path_from_components(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn split_parent(path: &str) -> (String, String) {
    if path == "/" {
        return ("/".to_owned(), String::new());
    }
    match path.rfind('/') {
        Some(0) => ("/".to_owned(), path[1..].to_owned()),
        Some(index) => (path[..index].to_owned(), path[index + 1..].to_owned()),
        None => ("/".to_owned(), path.to_owned()),
    }
}

fn join_child(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{name}")
    } else {
        format!("{parent}/{name}")
    }
}

fn is_within(candidate: &str, parent: &str) -> bool {
    if parent == "/" {
        return candidate.starts_with('/') && candidate != "/";
    }
    candidate
        .strip_prefix(parent)
        .is_some_and(|suffix| suffix.starts_with('/'))
}

fn immediate_child_name<'a>(parent: &str, candidate: &'a str) -> Option<&'a str> {
    let suffix = if parent == "/" {
        candidate.strip_prefix('/')?
    } else {
        candidate.strip_prefix(parent)?.strip_prefix('/')?
    };
    if suffix.is_empty() || suffix.contains('/') {
        None
    } else {
        Some(suffix)
    }
}
