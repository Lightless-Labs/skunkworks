#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        for (offset, byte) in text.bytes().enumerate() {
            if byte.is_ascii_whitespace() && byte != b' ' {
                return Err(CronError::Layout(offset));
            }
        }

        let spaces: Vec<usize> = text
            .bytes()
            .enumerate()
            .filter_map(|(i, byte)| (byte == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let separator = spaces[0];
        if separator == 0 || separator + 1 == text.len() {
            return Err(CronError::Layout(separator));
        }

        let minute_text = &text[..separator];
        let hour_text = &text[separator + 1..];
        let mut minutes = [false; 60];
        let mut hours = [false; 24];

        parse_field(minute_text, 0, 59, &mut minutes)?;
        parse_field(hour_text, separator + 1, 23, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    field: &str,
    offset: usize,
    maximum: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut start = 0;

    loop {
        let end = field[start..]
            .find(',')
            .map(|relative| start + relative)
            .unwrap_or(field.len());

        if start == end {
            return Err(CronError::Syntax(offset + start));
        }

        parse_item(&field[start..end], offset + start, maximum, selected)?;

        if end == field.len() {
            break;
        }
        start = end + 1;
    }

    Ok(())
}

fn parse_item(
    item: &str,
    offset: usize,
    maximum: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = item.as_bytes();
    let slash = bytes.iter().position(|&byte| byte == b'/');

    if let Some(first_slash) = slash {
        if let Some(second_slash) = bytes[first_slash + 1..]
            .iter()
            .position(|&byte| byte == b'/')
        {
            return Err(CronError::Syntax(offset + first_slash + 1 + second_slash));
        }
    }

    let base_end = slash.unwrap_or(bytes.len());
    let base = &item[..base_end];

    if let Some(slash_pos) = slash {
        if base != "*" && !base.contains('-') {
            return Err(CronError::Syntax(offset + slash_pos));
        }
    }

    let (range_start, range_end, dash_pos) = if base == "*" {
        (0, maximum, None)
    } else if let Some(dash) = base.as_bytes().iter().position(|&byte| byte == b'-') {
        let left = parse_number(&base[..dash], offset, maximum)?;
        let right = parse_number(&base[dash + 1..], offset + dash + 1, maximum)?;
        if left > right {
            return Err(CronError::Descending(offset + dash));
        }
        (left, right, Some(dash))
    } else {
        let value = parse_number(base, offset, maximum)?;
        (value, value, None)
    };

    let step = if let Some(slash_pos) = slash {
        parse_step(
            &item[slash_pos + 1..],
            offset + slash_pos + 1,
            maximum,
        )?
    } else {
        1
    };

    let _ = dash_pos;
    let mut value = range_start;
    loop {
        selected[value] = true;
        if range_end - value < step {
            break;
        }
        value += step;
    }

    Ok(())
}

fn parse_number(fragment: &str, offset: usize, maximum: usize) -> Result<usize, CronError> {
    let bytes = fragment.as_bytes();

    if bytes.is_empty() {
        return Err(CronError::Syntax(offset));
    }

    if let Some(index) = bytes.iter().position(|byte| !byte.is_ascii_digit()) {
        return Err(CronError::Syntax(offset + index));
    }

    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let mut value = 0usize;
    for &byte in bytes {
        value = value
            .saturating_mul(10)
            .saturating_add((byte - b'0') as usize);
        if value > maximum {
            return Err(CronError::Bounds(offset));
        }
    }

    Ok(value)
}

fn parse_step(fragment: &str, offset: usize, maximum: usize) -> Result<usize, CronError> {
    let bytes = fragment.as_bytes();

    if bytes.is_empty() {
        return Err(CronError::Syntax(offset));
    }

    if let Some(index) = bytes.iter().position(|byte| !byte.is_ascii_digit()) {
        return Err(CronError::Syntax(offset + index));
    }

    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let mut value = 0usize;
    for &byte in bytes {
        value = value
            .saturating_mul(10)
            .saturating_add((byte - b'0') as usize);
        if value > maximum {
            return Err(CronError::Step(offset));
        }
    }

    if value == 0 || value > maximum {
        return Err(CronError::Step(offset));
    }

    Ok(value)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|job| job.id == id) {
            return Err(Error::Duplicate);
        }

        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let order = self.jobs.len();
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if self.last_poll.is_some_and(|last| now < last) {
            return Err(Error::TimeReversed);
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower_exclusive = self.last_poll;
        let first_minute = match lower_exclusive {
            None => 0,
            Some(last) => last / 60_000 + 1,
        };
        let last_minute = now / 60_000;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        if first_minute <= last_minute {
            for job in &self.jobs {
                let matching_first = first_minute
                    + ((0..60).find(|offset| {
                        let candidate = first_minute + offset;
                        job.cron.matches(candidate)
                    }).unwrap_or(60));

                match job.policy {
                    Policy::All => {
                        let mut minute = matching_first;
                        while minute <= last_minute {
                            if job.cron.matches(minute) {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: minute * 60_000,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                            minute += 1;
                        }
                    }
                    Policy::Once => {
                        let mut latest = None;
                        let mut minute = matching_first;
                        while minute <= last_minute {
                            if job.cron.matches(minute) {
                                latest = Some(minute);
                            }
                            minute += 1;
                        }
                        if let Some(minute) = latest {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                    Policy::Skip => {
                        if now % 60_000 == 0 {
                            let minute = now / 60_000;
                            if minute >= first_minute && job.cron.matches(minute) {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: now,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|entry| entry.0).collect())
    }
}
