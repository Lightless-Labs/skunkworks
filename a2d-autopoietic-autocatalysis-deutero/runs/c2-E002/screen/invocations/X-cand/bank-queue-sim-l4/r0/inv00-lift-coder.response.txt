use std::cmp::Ordering;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege {
        waited: u64,
    },
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        time: u64,
        stage: OverflowStage,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;

    // Compare remainder with denominator - remainder to avoid doubling it.
    let complement = denominator - remainder;
    match remainder.cmp(&complement) {
        Ordering::Less => Some(quotient),
        Ordering::Greater => Some(quotient + 1),
        Ordering::Equal => {
            if quotient & 1 == 0 {
                Some(quotient)
            } else {
                Some(quotient + 1)
            }
        }
    }
}

#[derive(Clone)]
struct Person {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    waited: u64,
    entry: u64,
    ever_started: bool,
    paused_at: Option<u64>,
    accepted: bool,
    terminal: bool,
}

#[derive(Clone, Copy)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    dispatch_at: u64,
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn waiting_cmp(a: &Person, b: &Person) -> Ordering {
    a.customer.priority.cmp(&b.customer.priority)
        .then_with(|| match (a.paused_at, b.paused_at) {
            (Some(pa), Some(pb)) => pa.cmp(&pb).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a.customer.arrival.cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn best_waiter(waiting: &[usize], people: &[Person]) -> Option<usize> {
    waiting.iter().copied().min_by(|&a, &b| waiting_cmp(&people[a], &people[b]))
}

fn expiration_order(waiting: &[usize], people: &[Person], now: u64) -> Vec<usize> {
    let mut expired: Vec<usize> = waiting.iter().copied()
        .filter(|&i| people[i].deadline.is_some_and(|d| d <= now))
        .collect();
    expired.sort_by(|&a, &b| {
        people[a].deadline.cmp(&people[b].deadline)
            .then_with(|| people[a].customer.id.cmp(&people[b].customer.id))
    });
    expired
}

fn account_wait(p: &mut Person, now: u64, stats: &mut Stats) {
    let segment = now - p.entry;
    p.waited = p.waited.saturating_add(segment);
    stats.wait_ticks += segment as u128;
    if let Some(deadline) = p.deadline {
        let remaining = deadline.saturating_sub(now);
        p.deadline = Some(now.saturating_add(remaining));
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index].iter().any(|previous| previous.id == c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        if let Some(patience) = c.patience {
            if c.arrival.checked_add(patience).is_none() {
                return Err(Error::DeadlineOverflow { index, id: c.id });
            }
        }
    }

    let mut people: Vec<Person> = customers.iter().cloned().map(|c| Person {
        deadline: c.patience.and_then(|p| c.arrival.checked_add(p)),
        remaining: c.service,
        waited: 0,
        entry: c.arrival,
        ever_started: false,
        paused_at: None,
        accepted: false,
        terminal: false,
        customer: c,
    }).collect();

    let mut arrivals: Vec<usize> = (0..people.len()).collect();
    arrivals.sort_by(|&a, &b| {
        people[a].customer.arrival.cmp(&people[b].customer.arrival)
            .then_with(|| people[a].customer.priority.cmp(&people[b].customer.priority))
            .then_with(|| people[b].customer.id.cmp(&people[a].customer.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut ever_dispatched = false;
    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    while stats.completed + stats.reneged + stats.balked < people.len() {
        // Jump to the next event time; at time zero, process arrivals at zero.
        if active.is_none() && waiting.is_empty() {
            let next = arrivals.get(next_arrival).map(|&i| people[i].customer.arrival);
            if let Some(t) = next {
                if t > now {
                    now = t;
                }
            }
        }

        // A: Finish at this tick first.
        if let Some(a) = active {
            if a.finish == now {
                let p = &mut people[a.index];
                emit(&mut events, now, p.customer.id, Kind::Finish);
                p.terminal = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - p.customer.arrival) as u128;
                stats.completion_order.push(p.customer.id);
                active = None;
            }
        }

        // B: Expire those already waiting.
        for i in expiration_order(&waiting, &people, now) {
            waiting.retain(|&x| x != i);
            let p = &mut people[i];
            let waited = p.waited + (now - p.entry);
            stats.wait_ticks += (now - p.entry) as u128;
            emit(&mut events, now, p.customer.id, Kind::Renege { waited });
            p.terminal = true;
            stats.reneged += 1;
        }

        // C: Process the complete arrival cohort in its specified order.
        while next_arrival < arrivals.len()
            && people[arrivals[next_arrival]].customer.arrival == now
        {
            let i = arrivals[next_arrival];
            next_arrival += 1;
            let p = &people[i];
            emit(&mut events, now, p.customer.id, Kind::Arrive);

            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= gate {
                emit(&mut events, now, p.customer.id, Kind::Balk);
                stats.balked += 1;
            } else {
                let p = &mut people[i];
                p.accepted = true;
                p.entry = now;
                waiting.push(i);
                stats.accepted += 1;
            }
        }

        // D: Reconsider preemption after the full cohort.
        if let Some(a) = active {
            if now >= a.work_at {
                if let Some(best) = best_waiter(&waiting, &people) {
                    if people[best].customer.priority < people[a.index].customer.priority {
                        let remaining = a.finish - now;
                        let paused_id = people[a.index].customer.id;
                        let by = people[best].customer.id;
                        emit(&mut events, now, paused_id, Kind::Preempt { remaining, by });
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - a.dispatch_at) as u128;

                        let paused = &mut people[a.index];
                        paused.remaining = remaining;
                        paused.entry = now;
                        paused.paused_at = Some(now);
                        if let Some(budget) = paused.customer.patience {
                            let consumed = paused.waited;
                            let full = budget;
                            let remaining_budget = full.saturating_sub(consumed);
                            paused.deadline = Some(now.checked_add(remaining_budget).ok_or(
                                Error::TimeOverflow {
                                    customer: paused_id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                },
                            )?);
                        }
                        waiting.push(a.index);
                        active = None;
                    }
                }
            }
        }

        // E: Dispatch while possible, accounting for inline completions.
        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&waiting, &people) else {
                break;
            };
            waiting.retain(|&x| x != i);
            let p = &mut people[i];
            account_wait(p, now, &mut stats);

            let cost = if ever_dispatched && last_dispatched != Some(p.customer.id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: p.customer.id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(p.remaining).ok_or(Error::TimeOverflow {
                customer: p.customer.id,
                time: now,
                stage: OverflowStage::Finish,
            })?;

            let kind = if p.ever_started {
                Kind::Resume { remaining: p.remaining, work_at, finish }
            } else {
                p.ever_started = true;
                stats.started += 1;
                Kind::Start { remaining: p.remaining, work_at, finish }
            };
            emit(&mut events, now, p.customer.id, kind);
            let id = p.customer.id;
            last_dispatched = Some(id);
            ever_dispatched = true;
            active = Some(Active { index: i, work_at, finish, dispatch_at: now });

            if finish == now {
                emit(&mut events, now, id, Kind::Finish);
                p.terminal = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - p.customer.arrival) as u128;
                stats.completion_order.push(id);
                active = None;
                continue;
            }
            break;
        }

        // F: Expire newly queued customers after dispatch has been exhausted.
        for i in expiration_order(&waiting, &people, now) {
            waiting.retain(|&x| x != i);
            let p = &mut people[i];
            let segment = now - p.entry;
            stats.wait_ticks += segment as u128;
            let waited = p.waited + segment;
            emit(&mut events, now, p.customer.id, Kind::Renege { waited });
            p.terminal = true;
            stats.reneged += 1;
        }

        if stats.completed + stats.reneged + stats.balked >= people.len() {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |t: u64| {
            if t > now {
                next_time = Some(next_time.map_or(t, |old| old.min(t)));
            }
        };
        if next_arrival < arrivals.len() {
            consider(people[arrivals[next_arrival]].customer.arrival);
        }
        if let Some(a) = active {
            consider(a.finish);
            if a.work_at > now {
                consider(a.work_at);
            }
        }
        for &i in &waiting {
            if let Some(deadline) = people[i].deadline {
                consider(deadline);
            }
        }
        if let Some(t) = next_time {
            now = t;
        } else {
            break;
        }
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap_or(0)
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap_or(0)
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128)
            .unwrap_or(0)
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}
