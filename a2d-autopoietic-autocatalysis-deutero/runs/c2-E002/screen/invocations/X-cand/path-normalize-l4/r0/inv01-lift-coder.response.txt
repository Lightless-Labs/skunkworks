use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

#[derive(Clone, Copy)]
struct Component<'a> {
    value: &'a str,
    at: usize,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<Component<'_>> {
    path.split('/')
        .enumerate()
        .filter_map(|(index, value)| {
            if value.is_empty() {
                None
            } else {
                let at = path[..].match_indices(value).find(|(at, _)| {
                    *at == index || path.as_bytes().get(*at + value.len()) == Some(&b'/')
                });
                let offset = at.map_or(index, |(at, _)| at);
                Some(Component { value, at: offset })
            }
        })
        .collect()
}

fn raw_components(path: &str, base_offset: usize) -> Vec<Component<'_>> {
    let mut out = Vec::new();
    let mut start = 0;
    for (i, b) in path.bytes().enumerate() {
        if b == b'/' {
            if start < i {
                out.push(Component {
                    value: &path[start..i],
                    at: base_offset + start,
                });
            }
            start = i + 1;
        }
    }
    if start < path.len() {
        out.push(Component {
            value: &path[start..],
            at: base_offset + start,
        });
    }
    out
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || raw_components(path, 0)
            .last()
            .is_some_and(|c| c.value == "." || c.value == "..")
}

fn render(stack: &[String], absolute: bool, directory: bool) -> String {
    let mut out = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if absolute && stack.is_empty() {
        out = "/".to_owned();
    }

    if directory && out != "/" && out != "." && !out.is_empty() {
        out.push('/');
    }
    out
}

fn reduce_plain(
    path: &str,
    initial: Vec<String>,
    absolute: bool,
    base_offset: usize,
) -> Result<Vec<String>, PathError> {
    let mut stack = initial;
    for component in raw_components(path, base_offset) {
        match component.value {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|part| part != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: component.at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            value => stack.push(value.to_owned()),
        }
    }
    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let stack = reduce_plain(path, Vec::new(), absolute, 0)?;
    Ok(render(&stack, absolute, intent))
}

fn validate_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }
    match normalize(path) {
        Ok(normalized) => Ok(raw_components(&normalized, 0)
            .into_iter()
            .map(|c| c.value.to_owned())
            .collect()),
        Err(_) => Err(error),
    }
}

fn prepare_request<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(bool, &'a str, usize, Vec<String>), PathError> {
    validate_raw(path)?;

    if path.starts_with('~') {
        if path.len() > 1 && !path.as_bytes()[1..].starts_with(b"/") {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home = home.ok_or(PathError::MissingHome)?;
        let remainder = if path == "~" {
            ""
        } else {
            &path[2..]
        };
        let offset = if path == "~" { path.len() } else { 2 };
        let start = raw_components(home, 0)
            .into_iter()
            .map(|c| c.value.to_owned())
            .collect();
        Ok((false, remainder, offset, start))
    } else {
        Ok((path.starts_with('/'), path, 0, Vec::new()))
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd, PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(validate_context(home, PathError::InvalidHome)?),
        None => None,
    };

    let intent = directory_intent(path);
    let (absolute, request, offset, tilde_start) =
        prepare_request(path, home)?;

    let start = if path.starts_with('~') {
        tilde_start
    } else if absolute {
        Vec::new()
    } else {
        cwd_components
    };

    let stack = reduce_plain(request, start, true, offset)?;
    Ok(render(&stack, true, intent))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut keys = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize(&link.path).is_ok_and(|p| p == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = keys.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        keys.insert(link.path.clone(), index);
    }
    Ok(keys)
}

fn stack_path(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

struct Walker<'a> {
    links: &'a [Link],
    keys: &'a HashMap<String, usize>,
    active: Vec<usize>,
    expansions: usize,
    max_expansions: usize,
}

impl Walker<'_> {
    fn walk<'a>(
        &mut self,
        input: &'a str,
        offset: usize,
        stack: &mut Vec<String>,
    ) -> Result<(), PathError> {
        for component in raw_components(input, offset) {
            match component.value {
                "." => continue,
                ".." => {
                    if stack.pop().is_none() {
                        return Err(PathError::Escape { at: component.at });
                    }
                    continue;
                }
                value => stack.push(value.to_owned()),
            }

            let key = stack_path(stack);
            if let Some(&index) = self.keys.get(&key) {
                if self.active.contains(&index) {
                    return Err(PathError::Loop { index });
                }
                if self.expansions >= self.max_expansions {
                    return Err(PathError::Limit { index });
                }
                self.expansions += 1;
                self.active.push(index);

                let target = &self.links[index].target;
                let result = if target.starts_with('/') {
                    stack.clear();
                    self.walk(target, 0, stack)
                } else {
                    stack.pop();
                    self.walk(target, 0, stack)
                };

                self.active.pop();
                result?;
            }
        }
        Ok(())
    }
}

fn resolve_links_internal(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd, PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(validate_context(home, PathError::InvalidHome)?),
        None => None,
    };
    let keys = validate_links(links)?;
    let intent = directory_intent(path);
    let (absolute, request, offset, tilde_start) = prepare_request(path, home)?;

    let mut walker = Walker {
        links,
        keys: &keys,
        active: Vec::new(),
        expansions: 0,
        max_expansions,
    };

    let mut stack = if path.starts_with('~') {
        tilde_start
    } else if absolute {
        Vec::new()
    } else {
        cwd_components
    };

    if path.starts_with('~') {
        if let Some(home) = home_components {
            stack = home;
        }
    }

    walker.walk(request, offset, &mut stack)?;
    Ok(render(&stack, true, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_links_internal(path, cwd, home, links, max_expansions)
}

fn absolute_components(path: &str) -> Vec<String> {
    raw_components(path, 0)
        .into_iter()
        .map(|c| c.value.to_owned())
        .collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target)
        .take_while(|(a, b)| a == b)
        .count();

    let mut parts = Vec::new();
    parts.extend((common..base.len()).map(|_| "..".to_owned()));
    parts.extend(target[common..].iter().cloned());

    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base =
        resolve_links_internal(base, cwd, home, links, max_expansions)?;
    let resolved_target =
        resolve_links_internal(target, cwd, home, links, max_expansions)?;

    let base_components = absolute_components(&resolved_base);
    let target_components = absolute_components(&resolved_target);
    let target_directory = resolved_target.ends_with('/');
    let mut candidates = vec![route(&base_components, &target_components)];

    for link in links {
        let probe = resolve_links_internal(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else {
            continue;
        };
        let probed_components = absolute_components(&probe);

        if target_components.starts_with(&probed_components) {
            let suffix = &target_components[probed_components.len()..];
            let mut candidate = route(&base_components, &absolute_components(&link.path));
            if !suffix.is_empty() {
                if candidate == "." {
                    candidate = suffix.join("/");
                } else {
                    candidate.push('/');
                    candidate.push_str(&suffix.join("/"));
                }
            }
            candidates.push(candidate);
        }
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_directory && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        let Ok(resolved) =
            resolve_links_internal(&candidate, &resolved_base, home, links, max_expansions)
        else {
            continue;
        };

        if absolute_components(&resolved) == target_components {
            valid.push(candidate);
        }
    }

    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Empty)
}
