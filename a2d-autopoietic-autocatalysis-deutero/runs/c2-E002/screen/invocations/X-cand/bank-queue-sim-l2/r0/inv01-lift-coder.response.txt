```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    busy_until: Option<u64>,
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|prior| prior.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|prior| prior.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut customers_sorted: Vec<Customer> = customers.to_vec();
    customers_sorted.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            busy_until: None,
            idle_since: server.opens,
        })
        .collect();

    let mut events = Vec::new();
    let mut next_customer = 0usize;
    let mut waiting: Vec<usize> = Vec::new();
    let mut time = 0u64;

    loop {
        let mut had_event = false;

        let mut finishing: Vec<(u32, usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                state
                    .busy_until
                    .filter(|&finish| finish == time)
                    .map(|finish| {
                        let customer = customers
                            .iter()
                            .find(|c| c.id == state_busy_customer_placeholder())
                            .map_or(0, |c| c.id);
                        (state.id, index, finish, customer)
                    })
                    .map(|(id, index, finish, customer)| (id, index, finish, customer))
            })
            .map(|(id, index, _, customer)| (id, index, customer))
            .collect();

        finishing.sort_by_key(|&(id, _, _)| id);
        for (_, index, _) in &finishing {
            let state = &mut states[*index];
            state.busy_until = None;
            state.idle_since = time;
        }
        for (_, _, customer) in finishing {
            events.push(Event::Finish {
                time,
                customer,
                server: states
                    .iter()
                    .find(|s| s.busy_until.is_none() && s.id != 0)
                    .map_or(0, |s| s.id),
            });
            had_event = true;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.opens == time).then_some(index)
            })
            .collect();
        opening.sort_by_key(|&index| std::cmp::Reverse(states[index].id));
        for index in opening {
            states[index].opened = true;
            states[index].idle_since = time;
            events.push(Event::Open {
                time,
                server: states[index].id,
            });
            had_event = true;
        }

        let start = next_customer;
        while next_customer < customers_sorted.len()
            && customers_sorted[next_customer].arrival == time
        {
            next_customer += 1;
        }
        for customer in &customers_sorted[start..next_customer] {
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            waiting.push(
                customers
                    .iter()
                    .position(|c| c.id == customer.id)
                    .expect("customer index exists"),
            );
            had_event = true;
        }

        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &customers[customer_index];
            let server_id = states[server_index].id;
            let finish = customer
                .service
                .checked_add(time)
                .and_then(|value| value.checked_add(states[server_index].overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time,
                })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server_id,
                finish,
            });
            had_event = true;

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = time;
                states[server_index].busy_until = None;
            } else {
                states[server_index].busy_until = Some(finish);
            }
        }

        let next_arrival = customers_sorted
            .get(next_customer)
            .map(|customer| customer.arrival);
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.opens)
            .min();
        let next_finish = states.iter().filter_map(|state| state.busy_until).min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&candidate| candidate > time)
            .min();

        if let Some(next_time) = next_time {
            time = next_time;
        } else if !had_event && waiting.is_empty() && next_customer == customers_sorted.len() {
            break;
        } else if waiting.is_empty() && next_customer == customers_sorted.len() {
            break;
        } else {
            break;
        }
    }

    Ok(events)
}
```
