```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u128,
    jobs: Vec<Entry>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Status {
    Queued,
    Cancelled,
    Completed,
}

struct Entry {
    id: String,
    ready_at: u64,
    priority: u8,
    sequence: u128,
    status: Status,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            jobs: Vec::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|entry| entry.id == job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        self.jobs.push(Entry {
            id: job.id,
            ready_at: job.ready_at,
            priority: job.priority,
            sequence,
            status: Status::Queued,
        });
        self.next_sequence += 1;
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }

        let Some(entry) = self.jobs.iter_mut().find(|entry| entry.id == id) else {
            return Err(Error::Unknown);
        };

        match entry.status {
            Status::Queued => entry.status = Status::Cancelled,
            Status::Cancelled => return Err(Error::Cancelled),
            Status::Completed => return Err(Error::Completed),
        }

        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }

        let selected = self
            .jobs
            .iter()
            .enumerate()
            .filter(|(_, entry)| entry.status == Status::Queued && entry.ready_at <= now)
            .map(|(index, entry)| {
                let effective_priority = if self.aging_ms == 0 {
                    entry.priority
                } else {
                    let gained = (now - entry.ready_at) / self.aging_ms;
                    (entry.priority as u64 + gained).min(9) as u8
                };
                (index, entry.sequence, effective_priority)
            })
            .max_by(|a, b| a.2.cmp(&b.2).then_with(|| b.1.cmp(&a.1)));

        self.last_time = now;

        let Some((index, _, effective_priority)) = selected else {
            return Ok(None);
        };

        let entry = &mut self.jobs[index];
        entry.status = Status::Completed;
        Ok(Some(Dispatch {
            id: entry.id.clone(),
            effective_priority,
        }))
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes
            .iter()
            .all(|&byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-')
}
```
