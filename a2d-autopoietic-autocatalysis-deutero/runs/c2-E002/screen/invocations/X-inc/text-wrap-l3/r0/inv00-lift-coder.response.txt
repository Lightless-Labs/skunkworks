#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    breaks: Vec<usize>,
}

#[derive(Clone)]
struct Piece {
    text: String,
    has_word: bool,
}

struct Line {
    indent: usize,
    pieces: Vec<Piece>,
    gaps: Vec<usize>,
    content_len: usize,
}

impl Line {
    fn new(indent: usize) -> Self {
        Self {
            indent,
            pieces: Vec::new(),
            gaps: Vec::new(),
            content_len: 0,
        }
    }

    fn is_empty(&self) -> bool {
        self.content_len == 0
    }

    fn add_piece(&mut self, text: String, has_word: bool, gap: usize) {
        if !self.is_empty() {
            self.content_len += gap;
            self.gaps.push(gap);
        }
        self.content_len += text.chars().count();
        self.pieces.push(Piece { text, has_word });
    }

    fn render(&self, width: usize, justify: bool, final_line: bool, line_index: usize) -> String {
        let mut out = " ".repeat(self.indent);
        let word_pieces = self.pieces.iter().filter(|p| p.has_word).count();
        let gap_count = self.gaps.len();
        let extra = if justify && !final_line && word_pieces >= 2 && gap_count > 0 {
            width.saturating_sub(self.indent + self.content_len)
        } else {
            0
        };

        let base_extra = if gap_count > 0 { extra / gap_count } else { 0 };
        let remainder = if gap_count > 0 { extra % gap_count } else { 0 };
        let start_gap = if gap_count > 0 { line_index % gap_count } else { 0 };

        let mut gap_index = 0;
        for (piece_index, piece) in self.pieces.iter().enumerate() {
            if piece_index > 0 {
                let bonus = if gap_count > 0
                    && (gap_index + gap_count - start_gap) % gap_count < remainder
                {
                    1
                } else {
                    0
                };
                out.push_str(&" ".repeat(self.gaps[gap_index] + base_extra + bonus));
                gap_index += 1;
            }
            out.push_str(&piece.text);
        }
        out
    }
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn validate_markers(input: &str) -> Result<(), WrapError> {
    let chars: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let valid_left = i > 0 && chars[i - 1].1 != '~' && !is_separator(chars[i - 1].1);
        let valid_right =
            i + 1 < chars.len() && chars[i + 1].1 != '~' && !is_separator(chars[i + 1].1);
        if !valid_left || !valid_right {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }
    Ok(())
}

fn parse_segments(normalized: &str) -> Vec<(Vec<Word>, usize)> {
    let chars: Vec<char> = normalized.chars().collect();
    let mut segments = Vec::new();
    let mut start = 0;
    let mut i = 0;

    while i < chars.len() {
        if chars[i] == '\n' {
            let run_start = i;
            while i < chars.len() && chars[i] == '\n' {
                i += 1;
            }
            if i - run_start >= 2 {
                segments.push((parse_words(&chars[start..run_start]), i - run_start - 1));
                start = i;
            }
        } else {
            i += 1;
        }
    }
    segments.push((parse_words(&chars[start..]), 0));
    segments
}

fn parse_words(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        while i < chars.len() && is_separator(chars[i]) {
            i += 1;
        }
        if i == chars.len() {
            break;
        }
        let mut word_chars = Vec::new();
        let mut breaks = Vec::new();
        while i < chars.len() && !is_separator(chars[i]) {
            if chars[i] == '~' {
                breaks.push(word_chars.len());
            } else {
                word_chars.push(chars[i]);
            }
            i += 1;
        }
        words.push(Word {
            chars: word_chars,
            breaks,
        });
    }
    words
}

fn chars_to_string(chars: &[char]) -> String {
    chars.iter().collect()
}

fn layout_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }

    let mut lines = Vec::new();
    let mut line = Line::new(options.first_indent);
    let mut first_line = true;

    for word in words {
        let mut offset = 0usize;
        while offset < word.chars.len() {
            let capacity = options.width - line.indent;
            let available = capacity.saturating_sub(line.content_len);
            let gap = if line.is_empty() { 0 } else { 1 };
            let suffix_len = word.chars.len() - offset;

            if suffix_len <= available.saturating_sub(gap) {
                let text = chars_to_string(&word.chars[offset..]);
                line.add_piece(text, true, gap);
                offset = word.chars.len();
                continue;
            }

            let room = available.saturating_sub(gap);
            let mut usable = None;
            for &break_at in &word.breaks {
                if break_at > offset && break_at < word.chars.len() {
                    let prefix_len = break_at - offset;
                    if prefix_len + 1 <= room {
                        usable = Some(break_at);
                    }
                }
            }

            if let Some(break_at) = usable {
                let prefix_len = break_at - offset;
                let unused = room - prefix_len - 1;
                let accepted = line.is_empty()
                    || unused.saturating_mul(unused).saturating_add(options.hyphen_cost)
                        < available.saturating_mul(available);

                if accepted {
                    let mut text = chars_to_string(&word.chars[offset..break_at]);
                    text.push('-');
                    line.add_piece(text, true, gap);
                    lines.push(line.render(options.width, options.justify, false, lines.len()));
                    line = Line::new(options.rest_indent);
                    first_line = false;
                    offset = break_at;
                    while offset < word.chars.len()
                        && word.breaks.binary_search(&offset).is_ok()
                    {
                        offset += 1;
                    }
                    continue;
                }
            }

            if !line.is_empty() {
                lines.push(line.render(options.width, options.justify, false, lines.len()));
                line = Line::new(options.rest_indent);
                first_line = false;
                continue;
            }

            let capacity = options.width - line.indent;
            if suffix_len > capacity {
                let end = offset + capacity;
                line.add_piece(chars_to_string(&word.chars[offset..end]), true, 0);
                lines.push(line.render(options.width, options.justify, false, lines.len()));
                line = Line::new(options.rest_indent);
                first_line = false;
                offset = end;
                while offset < word.chars.len()
                    && word.breaks.binary_search(&offset).is_ok()
                {
                    offset += 1;
                }
            } else {
                line.add_piece(chars_to_string(&word.chars[offset..]), true, 0);
                offset = word.chars.len();
            }
        }
    }

    if !line.is_empty() {
        lines.push(line.render(options.width, options.justify, true, lines.len()));
    }

    let _ = first_line;
    lines
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    validate_markers(input)?;

    let mut normalized = String::new();
    let mut chars = input.chars().peekable();
    while let Some(ch) = chars.next() {
        if ch == '\r' {
            if chars.peek() == Some(&'\n') {
                chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    if normalized.is_empty() {
        return Ok(Vec::new());
    }

    let segments = parse_segments(&normalized);
    let mut output = Vec::new();
    for (words, boundary_empty_count) in segments {
        output.extend(layout_paragraph(&words, options));
        output.extend(std::iter::repeat(String::new()).take(boundary_empty_count));
    }
    Ok(output)
}
