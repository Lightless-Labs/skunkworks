impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|v| v == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|v| v == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = Vec::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut opening_quote = 0usize;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: i });
        }

        if !in_quotes && b == b'\r' {
            if bytes.get(i + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr { record: record_index, byte: i });
            }
            record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
            records.push(std::mem::take(&mut record));
            record_index += 1;
            i += 2;
            after_quote = false;
            continue;
        }

        if in_quotes {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                field.push(b);
                i += 1;
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                after_quote = false;
                i += 1;
                continue;
            }
            if b == b'\n' {
                record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                records.push(std::mem::take(&mut record));
                record_index += 1;
                after_quote = false;
                i += 1;
                continue;
            }
            return Err(CsvError::TrailingByte { record: record_index, byte: i });
        }

        if b == delimiter {
            record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
            i += 1;
        } else if b == b'\n' {
            record.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
            records.push(std::mem::take(&mut record));
            record_index += 1;
            i += 1;
        } else if b == quote {
            if field.is_empty() {
                in_quotes = true;
                opening_quote = i;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record: record_index, byte: i });
            }
        } else {
            field.push(b);
            i += 1;
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: opening_quote });
    }

    if !record.is_empty() || !field.is_empty() || after_quote || bytes.last() != Some(&b'\n') {
        record.push(String::from_utf8(field).unwrap());
        records.push(record);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn read_parsed(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| CsvError::InvalidConfig { component })?;
    let mut records = parse_dialect(input, config.delimiter, config.quote)?;
    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(ref names) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: Vec<(&str, usize)> = Vec::new();
            for (second, name) in names.iter().enumerate() {
                if let Some((_, first)) = seen.iter().find(|(n, _)| *n == name) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: *first,
                        second,
                    });
                }
                seen.push((name, second));
            }
        }

        for (idx, row) in records.iter().enumerate() {
            if row.len() != names.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1,
                    expected: names.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table { header, rows: records, duplicates: config.duplicates })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), result: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = read_parsed(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    if !(b'1'..=b'9').contains(&b[start]) { return false; }
    b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if b.len() < start + 4 { return false; }
    let dot = b.len() - 3;
    if b[dot] != b'.' || !b[dot + 1..].iter().all(u8::is_ascii_digit) { return false; }
    let integer = &b[start..dot];
    if integer.is_empty() { return false; }
    if integer == b"0" { return true; }
    integer[0].is_ascii_digit()
        && integer[0] != b'0'
        && integer.iter().all(u8::is_ascii_digit)
}

fn parse_decimal_value(s: &str) -> Option<i64> {
    let negative = s.starts_with('-');
    let body = if negative { &s[1..] } else { s };
    let (whole, frac) = body.split_once('.')?;
    let magnitude = whole.parse::<i128>().ok()?.checked_mul(100)?
        .checked_add(frac.parse::<i128>().ok()?)?;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).ok()
}

fn convert_cell(s: &str, kind: ColumnType) -> Result<Value, CellErrorKind> {
    match kind {
        ColumnType::Text => Ok(Value::Text(s.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(s) { return Err(CellErrorKind::InvalidInteger); }
            s.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(s) { return Err(CellErrorKind::InvalidDecimal); }
            parse_decimal_value(s).map(Value::Decimal).ok_or(CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut out = TypedResult { rows: Vec::new(), errors: Vec::new() };
    for (row_index, row) in table.rows.iter().enumerate() {
        let mut values = vec![None; columns.len()];
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(n) => table.column_index(n),
            };
            let Some(column) = resolved else {
                out.errors.push(CellError {
                    row: row_index, spec: spec_index, column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                row_failed = true;
                if policy != ErrorPolicy::Collect { break; }
                continue;
            };
            let Some(cell) = row.get(column) else {
                out.errors.push(CellError {
                    row: row_index, spec: spec_index, column: Some(column),
                    kind: CellErrorKind::MissingCell,
                });
                row_failed = true;
                if policy != ErrorPolicy::Collect { break; }
                continue;
            };

            let result = if cell.is_empty() {
                if spec.nullable {
                    Ok(Value::Null)
                } else if spec.kind == ColumnType::Text {
                    Ok(Value::Text(String::new()))
                } else {
                    Err(CellErrorKind::Empty)
                }
            } else {
                convert_cell(cell, spec.kind)
            };

            match result {
                Ok(value) => values[spec_index] = Some(value),
                Err(kind) => {
                    out.errors.push(CellError { row: row_index, spec: spec_index, column: Some(column), kind });
                    row_failed = true;
                    if policy != ErrorPolicy::Collect { break; }
                }
            }
            if policy == ErrorPolicy::Stop && row_failed {
                break;
            }
        }
        out.rows.push(values);
        if policy == ErrorPolicy::Stop && row_failed {
            break;
        }
    }
    out
}

fn duplicate_header(names: &[String]) -> Option<(String, usize, usize)> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in names.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(n, _)| *n == name) {
            return Some((name.clone(), *first, second));
        }
        seen.push((name, second));
    }
    None
}

fn field_has_nul(field: &str) -> Option<usize> {
    field.as_bytes().iter().position(|&b| b == 0)
}

fn quote_field(field: &str, delimiter: u8, quote: u8, first_record: bool, first_field: bool, sole: bool) -> String {
    let bytes = field.as_bytes();
    let edge_ws = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
    let starts_bom = field.starts_with('\u{feff}');
    let needs = bytes.contains(&delimiter) || bytes.contains(&quote)
        || bytes.contains(&b'\r') || bytes.contains(&b'\n') || edge_ws
        || (first_record && first_field && starts_bom)
        || (sole && field.is_empty());
    if !needs {
        return field.to_string();
    }
    let mut out = String::new();
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
            out.push(quote as char);
        } else {
            out.push(ch);
        }
    }
    out.push(quote as char);
    out
}

fn compare_rows(a: &Vec<String>, b: &Vec<String>) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let ord = x.as_bytes().cmp(y.as_bytes());
        if ord != std::cmp::Ordering::Equal { return ord; }
    }
    a.len().cmp(&b.len())
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original: Vec<(usize, &Vec<String>)> = Vec::new();
    if let Some(header) = &table.header {
        original.push((0, header));
    }
    let data_start = usize::from(table.header.is_some());
    for (i, row) in table.rows.iter().enumerate() {
        original.push((data_start + i, row));
    }

    for (record, row) in &original {
        for (column, cell) in row.iter().enumerate() {
            if let Some(byte) = field_has_nul(cell) {
                return Err(WriteError::Nul { record: *record, column, byte });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1, expected: header.len(), actual: row.len(),
                });
            }
        }
    }

    for (record, row) in &original {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record: *record });
        }
    }

    if config.mode == WriteMode::Canonical
        && table.header.is_some()
        && table.duplicates == DuplicatePolicy::Reject
    {
        if let Some((name, first, second)) = duplicate_header(table.header.as_ref().unwrap()) {
            return Err(WriteError::DuplicateHeader { name, first, second });
        }
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header { records.push(header.clone()); }
            records.extend(table.rows.clone());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut chosen: Vec<(String, usize)> = Vec::new();
                for (idx, name) in header.iter().enumerate() {
                    if let Some(pos) = chosen.iter().position(|(n, _)| n == name) {
                        if table.duplicates == DuplicatePolicy::KeepLast {
                            chosen[pos].1 = idx;
                        }
                    } else {
                        chosen.push((name.clone(), idx));
                    }
                }
                chosen.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
                let indexes: Vec<usize> = chosen.iter().map(|(_, i)| *i).collect();
                records.push(chosen.iter().map(|(n, _)| n.clone()).collect());
                let mut projected: Vec<Vec<String>> = table.rows.iter()
                    .map(|row| indexes.iter().map(|&i| row[i].clone()).collect())
                    .collect();
                projected.sort_by(compare_rows);
                records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(compare_rows);
                records = rows;
            }
        }
    }

    if records.is_empty() { return Ok(String::new()); }
    let canonical = config.mode == WriteMode::Canonical;
    let mut output = String::new();
    for (ri, row) in records.iter().enumerate() {
        if ri > 0 {
            output.push_str(if canonical || config.line_ending == LineEnding::Lf { "\n" } else { "\r\n" });
        }
        for (ci, field) in row.iter().enumerate() {
            if ci > 0 { output.push(config.delimiter as char); }
            output.push_str(&quote_field(field, config.delimiter, config.quote, ri == 0, ci == 0, row.len() == 1));
        }
    }
    if canonical || config.final_terminator {
        output.push('\n');
        if !canonical && config.line_ending == LineEnding::CrLf {
            let len = output.len();
            output.insert(len - 1, '\r');
        }
    }
    Ok(output)
}
