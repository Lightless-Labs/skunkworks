#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    text: String,
    consumed: usize,
    marker_ended: bool,
    pieces: usize,
    gap: usize,
}

#[derive(Clone)]
struct Candidate {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed_vector: Vec<usize>,
    text_vector: Vec<String>,
}

fn better(a: &Candidate, b: &Candidate) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    for (x, y) in a.consumed_vector.iter().zip(&b.consumed_vector) {
        if x != y {
            return x > y;
        }
    }
    for (x, y) in a.text_vector.iter().zip(&b.text_vector) {
        if x != y {
            return x < y;
        }
    }
    false
}

fn render_line(
    words: &[Word],
    start_w: usize,
    start_off: usize,
    end_w: usize,
    end_off: usize,
    marker: bool,
    indent: usize,
) -> (String, usize, usize) {
    let mut text = " ".repeat(indent);
    let mut pieces = 0;
    let mut consumed = 0;

    for wi in start_w..=end_w {
        let lo = if wi == start_w { start_off } else { 0 };
        let hi = if wi == end_w {
            end_off
        } else {
            words[wi].chars.len()
        };
        if wi != start_w {
            text.push(' ');
        }
        pieces += 1;
        for &c in &words[wi].chars[lo..hi] {
            text.push(c);
            consumed += 1;
        }
    }

    if marker {
        text.push('-');
    }
    (text, pieces, consumed)
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    fn rec(
        words: &[Word],
        options: &WrapOptions,
        wi: usize,
        off: usize,
        first: bool,
        memo: &mut [Vec<[Option<Candidate>; 2]>],
    ) -> Candidate {
        if wi == words.len() {
            return Candidate {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed_vector: Vec::new(),
                text_vector: Vec::new(),
            };
        }

        let first_slot = usize::from(first);
        if let Some(cached) = &memo[wi][off][first_slot] {
            return cached.clone();
        }

        let width = options.width;
        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let cap = width - indent;
        let mut best: Option<Candidate> = None;

        for end_w in wi..words.len() {
            let start_off = if end_w == wi { off } else { 0 };
            let mut chars_used = 0usize;

            for k in start_off..words[end_w].chars.len() {
                chars_used += 1;
                let pieces_before = end_w - wi;
                let displayed = chars_used + pieces_before;
                if displayed > cap {
                    break;
                }

                let complete = k + 1 == words[end_w].chars.len();
                if complete {
                    let (text, pieces, consumed) =
                        render_line(words, wi, off, end_w, k + 1, false, indent);
                    let gap = width - indent - displayed;
                    let tail = rec(words, options, end_w + 1, 0, false, memo);
                    let mut candidate = Candidate {
                        cost: tail.cost
                            + if tail.lines.is_empty() {
                                (gap * gap / 4) as u64
                            } else {
                                (gap * gap) as u64
                            },
                        marker_lines: tail.marker_lines,
                        lines: Vec::new(),
                        consumed_vector: Vec::new(),
                        text_vector: Vec::new(),
                    };
                    if !tail.lines.is_empty() && pieces == 1 {
                        candidate.cost += 3;
                    }
                    candidate.lines.push(Line {
                        text: text.clone(),
                        consumed,
                        marker_ended: false,
                        pieces,
                        gap,
                    });
                    candidate.lines.extend(tail.lines);
                    candidate.consumed_vector.push(consumed);
                    candidate.consumed_vector.extend(tail.consumed_vector);
                    candidate.text_vector.push(text);
                    candidate.text_vector.extend(tail.text_vector);
                    if best.as_ref().is_none_or(|b| better(&candidate, b)) {
                        best = Some(candidate);
                    }
                }

                for &mark in &words[end_w].markers {
                    if mark != k + 1 {
                        continue;
                    }
                    let displayed_with_hyphen = displayed + 1;
                    if displayed_with_hyphen > cap {
                        continue;
                    }

                    let (text, pieces, consumed) =
                        render_line(words, wi, off, end_w, mark, true, indent);
                    let gap = width - indent - displayed_with_hyphen;
                    let tail = rec(words, options, end_w, mark, false, memo);
                    let mut candidate = Candidate {
                        cost: tail.cost
                            + (gap * gap) as u64
                            + options.hyphen_cost as u64,
                        marker_lines: tail.marker_lines + 1,
                        lines: Vec::new(),
                        consumed_vector: Vec::new(),
                        text_vector: Vec::new(),
                    };
                    if pieces == 1 {
                        candidate.cost += 3;
                    }
                    candidate.lines.push(Line {
                        text: text.clone(),
                        consumed,
                        marker_ended: true,
                        pieces,
                        gap,
                    });
                    candidate.lines.extend(tail.lines);
                    candidate.consumed_vector.push(consumed);
                    candidate.consumed_vector.extend(tail.consumed_vector);
                    candidate.text_vector.push(text);
                    candidate.text_vector.extend(tail.text_vector);
                    if best.as_ref().is_none_or(|b| better(&candidate, b)) {
                        best = Some(candidate);
                    }
                }
            }

            if end_w > wi && chars_used == 0 {
                break;
            }
            if end_w + 1 < words.len() {
                let projected = chars_used + 1 + (end_w - wi + 1);
                if projected > cap {
                    break;
                }
            }
        }

        let remaining = words[wi].chars.len() - off;
        let markers_fit = words[wi]
            .markers
            .iter()
            .any(|&m| m > off && m - off + 1 <= cap);

        if remaining > cap && !markers_fit {
            let end = off + cap;
            let (text, pieces, consumed) =
                render_line(words, wi, off, wi, end, false, indent);
            let gap = width - indent - cap;
            let tail = rec(words, options, wi, end, false, memo);
            let mut candidate = Candidate {
                cost: tail.cost + (gap * gap) as u64 + (2 * width + 1) as u64,
                marker_lines: tail.marker_lines,
                lines: Vec::new(),
                consumed_vector: Vec::new(),
                text_vector: Vec::new(),
            };
            if !tail.lines.is_empty() && pieces == 1 {
                candidate.cost += 3;
            }
            candidate.lines.push(Line {
                text: text.clone(),
                consumed,
                marker_ended: false,
                pieces,
                gap,
            });
            candidate.lines.extend(tail.lines);
            candidate.consumed_vector.push(consumed);
            candidate.consumed_vector.extend(tail.consumed_vector);
            candidate.text_vector.push(text);
            candidate.text_vector.extend(tail.text_vector);
            if best.as_ref().is_none_or(|b| better(&candidate, b)) {
                best = Some(candidate);
            }
        }

        let result = best.unwrap_or(Candidate {
            cost: u64::MAX / 4,
            marker_lines: usize::MAX / 4,
            lines: Vec::new(),
            consumed_vector: Vec::new(),
            text_vector: Vec::new(),
        });
        memo[wi][off][first_slot] = Some(result.clone());
        result
    }

    let mut memo: Vec<Vec<[Option<Candidate>; 2]>> = words
        .iter()
        .map(|word| {
            (0..=word.chars.len())
                .map(|_| [None, None])
                .collect()
        })
        .collect();
    rec(words, options, 0, 0, true, &mut memo).lines
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn normalize_and_segments(input: &str) -> Vec<Vec<char>> {
    let mut normalized = Vec::new();
    let mut it = input.chars().peekable();
    while let Some(c) = it.next() {
        if c == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut segments = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                segments.push(current);
                current = Vec::new();
                for _ in 1..(j - i) {
                    segments.push(Vec::new());
                }
                i = j;
                continue;
            }
        }
        current.push(normalized[i]);
        i += 1;
    }
    segments.push(current);
    segments
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut i = 0;

    while i < segment.len() {
        while i < segment.len() && is_separator(segment[i]) {
            i += 1;
        }
        if i == segment.len() {
            break;
        }

        let mut chars = Vec::new();
        let mut markers = Vec::new();
        while i < segment.len() && !is_separator(segment[i]) {
            if segment[i] == '~' {
                markers.push(chars.len());
            } else {
                chars.push(segment[i]);
            }
            i += 1;
        }
        words.push(Word { chars, markers });
    }

    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    if original.len() > 512 {
        return Err(WrapError::InputTooLong {
            chars: original.len(),
        });
    }

    for &(index, c) in &original {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..original.len() {
        if original[i].1 == '~' {
            let left_ok = i > 0 && original[i - 1].1 != '~' && !is_separator(original[i - 1].1);
            let right_ok = i + 1 < original.len()
                && original[i + 1].1 != '~'
                && !is_separator(original[i + 1].1);
            if !left_ok || !right_ok {
                return Err(WrapError::InvalidMarker {
                    index: original[i].0,
                });
            }
        }
    }

    let mut output = Vec::new();
    for segment in normalize_and_segments(input) {
        let words = parse_words(&segment);
        if words.is_empty() {
            continue;
        }

        let mut lines = solve_paragraph(&words, options);
        let line_count = lines.len();
        for (idx, line) in lines.iter_mut().enumerate() {
            if options.justify && idx + 1 < line_count && line.pieces >= 2 {
                let gaps = line.pieces - 1;
                let extra = line.gap;
                let base = extra / gaps;
                let remainder = extra % gaps;

                let indent = if idx == 0 {
                    options.first_indent
                } else {
                    options.rest_indent
                };
                let content: Vec<char> = line.text.chars().skip(indent).collect();
                let mut parts: Vec<String> = Vec::new();
                let mut part = String::new();
                for c in content {
                    if c == ' ' {
                        parts.push(part);
                        part = String::new();
                    } else {
                        part.push(c);
                    }
                }
                parts.push(part);

                let mut rebuilt = " ".repeat(indent);
                for (i, p) in parts.iter().enumerate() {
                    if i > 0 {
                        let gap_index = i - 1;
                        let cyclic = (gap_index + gaps - (idx % gaps)) % gaps;
                        let add = base + usize::from(cyclic < remainder);
                        rebuilt.push_str(&" ".repeat(1 + add));
                    }
                    rebuilt.push_str(p);
                }
                line.text = rebuilt;
            }
            output.push(line.text.clone());
        }
    }

    let normalized_segments = normalize_and_segments(input);
    if normalized_segments.len() <= 1 {
        return Ok(output);
    }

    let mut result = Vec::new();
    let mut line_iter = output.into_iter();
    for (i, segment) in normalized_segments.iter().enumerate() {
        let words = parse_words(segment);
        if !words.is_empty() {
            let count = solve_paragraph(&words, options).len();
            for _ in 0..count {
                if let Some(line) = line_iter.next() {
                    result.push(line);
                }
            }
        }
        if i + 1 < normalized_segments.len() {
            result.push(String::new());
        }
    }
    Ok(result)
}
