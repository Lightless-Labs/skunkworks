#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    text: String,
    consumed: usize,
    marker_ended: bool,
    raw_split: bool,
    pieces: usize,
    gap: usize,
}

#[derive(Clone)]
struct Candidate {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed_vector: Vec<usize>,
    text_vector: Vec<String>,
}

fn better(a: &Candidate, b: &Candidate) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    for (x, y) in a.consumed_vector.iter().zip(&b.consumed_vector) {
        if x != y {
            return x > y;
        }
    }
    for (x, y) in a.text_vector.iter().zip(&b.text_vector) {
        if x != y {
            return x < y;
        }
    }
    false
}

fn word_spelling_len(words: &[Word], wi: usize, off: usize) -> usize {
    words[wi].chars.len() - off
}

fn render_line(
    words: &[Word],
    start_w: usize,
    start_off: usize,
    end_w: usize,
    end_off: usize,
    marker: bool,
    raw: bool,
    indent: usize,
) -> (String, usize, usize) {
    let mut s = " ".repeat(indent);
    let mut pieces = 0usize;
    let mut consumed = 0usize;

    for wi in start_w..=end_w {
        let lo = if wi == start_w { start_off } else { 0 };
        let hi = if wi == end_w { end_off } else { words[wi].chars.len() };
        if wi != start_w {
            s.push(' ');
        }
        pieces += 1;
        for &c in &words[wi].chars[lo..hi] {
            s.push(c);
            consumed += 1;
        }
    }
    if marker {
        s.push('-');
    }
    let _ = raw;
    (s, pieces, consumed)
}

fn normalize_and_segments(input: &str) -> Vec<Vec<char>> {
    let mut normalized = Vec::new();
    let mut it = input.chars().peekable();
    while let Some(c) = it.next() {
        if c == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut segments = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                segments.push(current);
                current = Vec::new();
                for _ in 1..(j - i) {
                    segments.push(Vec::new());
                }
                i = j;
                continue;
            }
        }
        current.push(normalized[i]);
        i += 1;
    }
    segments.push(current);
    segments
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    let mut best: Vec<Vec<Option<Candidate>>> = words
        .iter()
        .map(|w| vec![None; w.chars.len() + 1])
        .collect();
    let width = options.width;

    fn rec(
        words: &[Word],
        options: &WrapOptions,
        wi: usize,
        off: usize,
        first: bool,
        memo: &mut [Vec<Option<Candidate>>],
    ) -> Candidate {
        if wi >= words.len() {
            return Candidate {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed_vector: Vec::new(),
                text_vector: Vec::new(),
            };
        }

        if off == 0 {
            if let Some(c) = &memo[wi][off] {
                return c.clone();
            }
        }

        let width = options.width;
        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let cap = width - indent;
        let mut best_candidate: Option<Candidate> = None;

        for end_w in wi..words.len() {
            let start_off = if end_w == wi { off } else { 0 };
            let mut used = 0usize;
            let mut displayed = 0usize;

            for k in start_off..words[end_w].chars.len() {
                let next_used = used + 1;
                let next_displayed = displayed + 1;
                let prior_words = end_w - wi;
                let display_with_gaps = next_displayed + prior_words;
                if display_with_gaps > cap {
                    break;
                }
                used = next_used;
                displayed = next_displayed;

                let end_complete = k + 1 == words[end_w].chars.len();
                if end_complete {
                    let (text, pieces, consumed) =
                        render_line(words, wi, off, end_w, k + 1, false, false, indent);
                    let full_width = indent + displayed + (end_w - wi);
                    let gap = width - full_width;
                    let tail = if k + 1 < words[end_w].chars.len() {
                        rec(words, options, end_w, k + 1, false, memo)
                    } else {
                        rec(words, options, end_w + 1, 0, false, memo)
                    };
                    let mut cand = Candidate {
                        cost: tail.cost + if tail.lines.is_empty() {
                            (gap * gap / 4) as u64
                        } else {
                            (gap * gap) as u64
                        },
                        marker_lines: tail.marker_lines,
                        lines: Vec::new(),
                        consumed_vector: Vec::new(),
                        text_vector: Vec::new(),
                    };
                    if !tail.lines.is_empty() && pieces == 1 {
                        cand.cost += 3;
                    }
                    let mut line = Line {
                        text,
                        consumed,
                        marker_ended: false,
                        raw_split: false,
                        pieces,
                        gap,
                    };
                    let _ = &mut line;
                    cand.lines.push(line);
                    cand.lines.extend(tail.lines);
                    cand.consumed_vector.push(consumed);
                    cand.consumed_vector.extend(tail.consumed_vector);
                    cand.text_vector.push(cand.lines[0].text.clone());
                    cand.text_vector.extend(tail.text_vector);
                    if best_candidate.as_ref().map_or(true, |b| better(&cand, b)) {
                        best_candidate = Some(cand);
                    }
                }

                for &mark in &words[end_w].markers {
                    if mark == k + 1 {
                        let display_hyphen = displayed + 1 + (end_w - wi);
                        if display_hyphen <= cap {
                            let (text, pieces, consumed) =
                                render_line(words, wi, off, end_w, mark, true, false, indent);
                            let gap = width - indent - display_hyphen;
                            let tail = rec(words, options, end_w, mark, false, memo);
                            let mut cand = Candidate {
                                cost: tail.cost
                                    + (gap * gap) as u64
                                    + options.hyphen_cost as u64,
                                marker_lines: tail.marker_lines + 1,
                                lines: Vec::new(),
                                consumed_vector: Vec::new(),
                                text_vector: Vec::new(),
                            };
                            if pieces == 1 {
                                cand.cost += 3;
                            }
                            cand.lines.push(Line {
                                text,
                                consumed,
                                marker_ended: true,
                                raw_split: false,
                                pieces,
                                gap,
                            });
                            cand.lines.extend(tail.lines);
                            cand.consumed_vector.push(consumed);
                            cand.consumed_vector.extend(tail.consumed_vector);
                            cand.text_vector.push(cand.lines[0].text.clone());
                            cand.text_vector.extend(tail.text_vector);
                            if best_candidate.as_ref().map_or(true, |b| better(&cand, b)) {
                                best_candidate = Some(cand);
                            }
                        }
                    }
                }
            }

            if end_w > wi && displayed == 0 {
                break;
            }
            if end_w + 1 < words.len() {
                let projected = displayed + 1 + (end_w - wi + 1);
                if projected > cap {
                    break;
                }
            }
        }

        let remaining = word_spelling_len(words, wi, off);
        let markers_fit = words[wi]
            .markers
            .iter()
            .any(|&m| m > off && m - off + 1 <= cap);
        if remaining > cap && !markers_fit {
            let end = off + cap;
            let (text, pieces, consumed) =
                render_line(words, wi, off, wi, end, false, true, indent);
            let gap = width - indent - cap;
            let tail = rec(words, options, wi, end, false, memo);
            let mut cand = Candidate {
                cost: tail.cost + (gap * gap) as u64 + (2 * width + 1) as u64,
                marker_lines: tail.marker_lines,
                lines: Vec::new(),
                consumed_vector: Vec::new(),
                text_vector: Vec::new(),
            };
            if !tail.lines.is_empty() && pieces == 1 {
                cand.cost += 3;
            }
            cand.lines.push(Line {
                text,
                consumed,
                marker_ended: false,
                raw_split: true,
                pieces,
                gap,
            });
            cand.lines.extend(tail.lines);
            cand.consumed_vector.push(consumed);
            cand.consumed_vector.extend(tail.consumed_vector);
            cand.text_vector.push(cand.lines[0].text.clone());
            cand.text_vector.extend(tail.text_vector);
            if best_candidate.as_ref().map_or(true, |b| better(&cand, b)) {
                best_candidate = Some(cand);
            }
        }

        let result = best_candidate.unwrap_or_else(|| Candidate {
            cost: u64::MAX / 4,
            marker_lines: usize::MAX / 4,
            lines: Vec::new(),
            consumed_vector: Vec::new(),
            text_vector: Vec::new(),
        });
        if off == 0 {
            memo[wi][off] = Some(result.clone());
        }
        result
    }

    let candidate = rec(words, options, 0, 0, true, &mut best);
    candidate.lines
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }
    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let left_ok = i > 0 && chars[i - 1].1 != '~' && !is_separator(chars[i - 1].1);
            let right_ok =
                i + 1 < chars.len() && chars[i + 1].1 != '~' && !is_separator(chars[i + 1].1);
            if !left_ok || !right_ok {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut output = Vec::new();
    for segment in normalize_and_segments(input) {
        let mut words = Vec::new();
        let mut i = 0;
        while i < segment.len() {
            while i < segment.len() && is_separator(segment[i]) {
                i += 1;
            }
            if i == segment.len() {
                break;
            }
            let mut chars = Vec::new();
            let mut markers = Vec::new();
            while i < segment.len() && !is_separator(segment[i]) {
                if segment[i] == '~' {
                    markers.push(chars.len());
                } else {
                    chars.push(segment[i]);
                }
                i += 1;
            }
            words.push(Word { chars, markers });
        }

        if words.is_empty() {
            continue;
        }

        let mut lines = solve_paragraph(&words, options);
        let n = lines.len();
        for idx in 0..n {
            if options.justify && idx + 1 < n && lines[idx].pieces >= 2 {
                let gaps = lines[idx].pieces - 1;
                let extra = lines[idx].gap;
                let base = extra / gaps;
                let rem = extra % gaps;
                let mut rebuilt = String::new();
                let indent = if idx == 0 {
                    options.first_indent
                } else {
                    options.rest_indent
                };
                rebuilt.push_str(&" ".repeat(indent));
                let content = lines[idx].text.chars().skip(indent).collect::<Vec<_>>();
                let mut word_parts: Vec<String> = Vec::new();
                let mut part = String::new();
                for c in content {
                    if c == ' ' {
                        word_parts.push(part);
                        part = String::new();
                    } else {
                        part.push(c);
                    }
                }
                word_parts.push(part);
                for (g, p) in word_parts.iter().enumerate() {
                    if g > 0 {
                        let cycle_start = idx % gaps;
                        let cyclic = (g - 1 + gaps - cycle_start) % gaps;
                        let add = base + usize::from(cyclic < rem);
                        rebuilt.push_str(&" ".repeat(1 + add));
                    }
                    rebuilt.push_str(p);
                }
                lines[idx].text = rebuilt;
            }
            output.push(lines[idx].text.clone());
        }
        // Paragraph boundary empties are supplied by the segment splitter.
        // Empty segments between boundaries are retained below through boundary markers.
    }

    // Reconstruct boundary empty lines from normalized input.
    let normalized = normalize_and_segments(input);
    let boundary_count = normalized.len().saturating_sub(1);
    if boundary_count > 0 {
        let mut result = Vec::new();
        let mut line_iter = output.into_iter();
        for (si, seg) in normalized.iter().enumerate() {
            if seg.iter().any(|&c| !is_separator(c)) {
                let mut count = 0usize;
                // Paragraph output groups are tracked by rerunning the deterministic layout.
                let mut words = Vec::new();
                let mut i = 0;
                while i < seg.len() {
                    while i < seg.len() && is_separator(seg[i]) {
                        i += 1;
                    }
                    if i == seg.len() { break; }
                    let mut wc = Vec::new();
                    let mut ms = Vec::new();
                    while i < seg.len() && !is_separator(seg[i]) {
                        if seg[i] == '~' { ms.push(wc.len()); } else { wc.push(seg[i]); }
                        i += 1;
                    }
                    words.push(Word { chars: wc, markers: ms });
                }
                count = solve_paragraph(&words, options).len();
                for _ in 0..count {
                    if let Some(line) = line_iter.next() { result.push(line); }
                }
            }
            if si + 1 < normalized.len() {
                result.push(String::new());
            }
        }
        return Ok(result);
    }

    Ok(output)
}
