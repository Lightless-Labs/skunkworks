#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        for (i, b) in text.bytes().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = text
            .bytes()
            .enumerate()
            .filter_map(|(i, b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 || split + 1 == text.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(&text[..split], 0, 59, &mut minutes)?;
        parse_field(&text[split + 1..], split + 1, 23, &mut hours)?;
        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    field: &str,
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut start = 0;
    loop {
        let end = field[start..]
            .find(',')
            .map(|n| start + n)
            .unwrap_or(field.len());
        if start == end {
            return Err(CronError::Syntax(offset + start));
        }
        parse_item(&field[start..end], offset + start, max, selected)?;
        if end == field.len() {
            break;
        }
        start = end + 1;
    }
    Ok(())
}

fn parse_item(
    item: &str,
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let slash = item.find('/');
    if let Some(s) = slash {
        if let Some(second) = item[s + 1..].find('/') {
            return Err(CronError::Syntax(offset + s + 1 + second));
        }
    }

    let base_end = slash.unwrap_or(item.len());
    let base = &item[..base_end];
    let step_offset = slash.map(|s| offset + s + 1);

    let is_star = base == "*";
    let dash = base.find('-');
    if step_offset.is_some() && !is_star && dash.is_none() {
        return Err(CronError::Syntax(offset + slash.unwrap()));
    }

    let (lo, hi, dash_at) = if is_star {
        (0, max, None)
    } else if let Some(d) = dash {
        let left = parse_number(&base[..d], offset, max)?;
        let right = parse_number(&base[d + 1..], offset + d + 1, max)?;
        if left > right {
            return Err(CronError::Descending(offset + d));
        }
        (left, right, Some(d))
    } else {
        let n = parse_number(base, offset, max)?;
        (n, n, None)
    };

    let step = if let Some(step_start) = step_offset {
        let value = parse_step(&item[slash.unwrap() + 1..], step_start)?;
        let step_max = if max == 59 { 60 } else { 24 };
        if value == 0 || value > step_max {
            return Err(CronError::Step(step_start));
        }
        value
    } else {
        1
    };

    let _ = dash_at;
    let mut n = lo;
    while n <= hi {
        selected[n] = true;
        match n.checked_add(step) {
            Some(next) if next > n => n = next,
            _ => break,
        }
    }
    Ok(())
}

fn parse_number(fragment: &str, start: usize, max: usize) -> Result<usize, CronError> {
    validate_digits(fragment, start)?;
    if fragment.len() > 1 && fragment.as_bytes()[0] == b'0' {
        return Err(CronError::Number(start));
    }
    let mut value = 0usize;
    for b in fragment.bytes() {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as usize))
            .ok_or(CronError::Bounds(start))?;
        if value > max {
            return Err(CronError::Bounds(start));
        }
    }
    Ok(value)
}

fn parse_step(fragment: &str, start: usize) -> Result<usize, CronError> {
    validate_digits(fragment, start)?;
    if fragment.len() > 1 && fragment.as_bytes()[0] == b'0' {
        return Err(CronError::Number(start));
    }
    let mut value = 0usize;
    for b in fragment.bytes() {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as usize))
            .ok_or(CronError::Step(start))?;
    }
    Ok(value)
}

fn validate_digits(fragment: &str, start: usize) -> Result<(), CronError> {
    if fragment.is_empty() {
        return Err(CronError::Syntax(start));
    }
    for (i, b) in fragment.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(CronError::Syntax(start + i));
        }
    }
    Ok(())
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|job| job.id == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order: self.jobs.len(),
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower_exclusive = self.last_poll;
        let first_minute = match lower_exclusive {
            None => 0,
            Some(last) => last / 60_000 + 1,
        };
        let last_minute = now / 60_000;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let matching: Vec<u64> = if first_minute > last_minute {
                Vec::new()
            } else {
                (first_minute..=last_minute)
                    .filter(|&minute| job.cron.matches(minute))
                    .collect()
            };

            match job.policy {
                Policy::All => {
                    for minute in matching {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Once => {
                    if let Some(minute) = matching.last() {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: *minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0 && matching.last().is_some_and(|m| *m * 60_000 == now) {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: now,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|entry| entry.0).collect())
    }
}
