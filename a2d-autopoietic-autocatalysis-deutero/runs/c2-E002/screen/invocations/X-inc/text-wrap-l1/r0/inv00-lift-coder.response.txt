#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(32..=126).contains(&byte) && !matches!(byte, b'\t' | b'\n' | b'\r') {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();
    let mut start = 0;

    while start < bytes.len() {
        while start < bytes.len() && matches!(bytes[start], b' ' | b'\t' | b'\n' | b'\r') {
            start += 1;
        }
        if start == bytes.len() {
            break;
        }

        let mut end = start;
        while end < bytes.len() && !matches!(bytes[end], b' ' | b'\t' | b'\n' | b'\r') {
            end += 1;
        }
        let mut suffix = &bytes[start..end];

        while !suffix.is_empty() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            if suffix.len() <= width {
                let needed = suffix.len() + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(std::str::from_utf8(suffix).unwrap());
                    suffix = &[];
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            } else if line.is_empty() {
                line.push_str(std::str::from_utf8(&suffix[..width]).unwrap());
                lines.push(std::mem::take(&mut line));
                suffix = &suffix[width..];
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(std::str::from_utf8(&suffix[..available]).unwrap());
                    lines.push(std::mem::take(&mut line));
                    suffix = &suffix[available..];
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }

        start = end;
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}
