```rust
use std::cmp::Ordering;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let complement = denominator - remainder;
    Some(match remainder.cmp(&complement) {
        Ordering::Less => quotient,
        Ordering::Greater => quotient + 1,
        Ordering::Equal if quotient % 2 == 0 => quotient,
        Ordering::Equal => quotient + 1,
    })
}

#[derive(Clone)]
struct Waiting {
    index: usize,
    entry: u64,
    deadline: Option<u64>,
    waited: u64,
    paused: bool,
    pause_time: u64,
    remaining: u64,
}

#[derive(Clone)]
struct Active {
    index: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiting_cmp(a: &Waiting, b: &Waiting, customers: &[Customer]) -> Ordering {
    let ca = &customers[a.index];
    let cb = &customers[b.index];
    ca.priority
        .cmp(&cb.priority)
        .then_with(|| match (a.paused, b.paused) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (true, true) => a.pause_time.cmp(&b.pause_time).then(ca.id.cmp(&cb.id)),
            (false, false) => ca.arrival.cmp(&cb.arrival).then(cb.id.cmp(&ca.id)),
        })
}

fn add_time(customer: u32, time: u64, amount: u64, stage: OverflowStage) -> Result<u64, Error> {
    time.checked_add(amount).ok_or(Error::TimeOverflow {
        customer,
        time,
        stage,
    })
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow {
                index,
                id: c.id,
            })?),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let n = customers.len();
    let mut arrivals: Vec<usize> = (0..n).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then(customers[a].priority.cmp(&customers[b].priority))
            .then(customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0;
    let mut now = arrivals
        .first()
        .map(|&i| customers[i].arrival)
        .unwrap_or(0);
    let mut last_dispatched: Option<u32> = None;
    let mut ever_started = vec![false; n];
    let mut completed = vec![false; n];
    let mut total_waited = vec![0u64; n];
    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };
    let mut last_event_time = 0;

    let emit = |events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind| {
        events.push(Event { time, customer, kind });
    };

    while stats.completed + stats.reneged + stats.balked < n {
        if let Some(a) = active.as_ref() {
            if a.finish == now {
                let a = active.take().unwrap();
                let c = &customers[a.index];
                emit(&mut events, now, c.id, Kind::Finish);
                stats.completed += 1;
                completed[a.index] = true;
                stats.turnaround_ticks += (now - c.arrival) as u128;
                stats.completion_order.push(c.id);
                last_event_time = now;
            }
        }

        let mut expired: Vec<usize> = waiting
            .iter()
            .enumerate()
            .filter_map(|(i, w)| w.deadline.filter(|&d| d <= now).map(|d| (i, d)))
            .map(|(i, _)| i)
            .collect();
        expired.sort_by(|&a, &b| {
            waiting[a]
                .deadline
                .cmp(&waiting[b].deadline)
                .then(customers[waiting[a].index].id.cmp(&customers[waiting[b].index].id))
        });
        for wi in expired.into_iter().rev() {
            let w = waiting.remove(wi);
            let c = &customers[w.index];
            let segment = now - w.entry;
            total_waited[w.index] += segment;
            stats.wait_ticks += segment as u128;
            let waited = total_waited[w.index];
            emit(&mut events, now, c.id, Kind::Renege { waited });
            stats.reneged += 1;
            last_event_time = now;
        }

        while next_arrival < n && customers[arrivals[next_arrival]].arrival == now {
            let idx = arrivals[next_arrival];
            next_arrival += 1;
            let c = &customers[idx];
            emit(&mut events, now, c.id, Kind::Arrive);
            last_event_time = now;
            let server_idle = active.is_none();
            let limit = config.waiting_capacity + usize::from(server_idle);
            if waiting.len() >= limit {
                emit(&mut events, now, c.id, Kind::Balk);
                stats.balked += 1;
                last_event_time = now;
            } else {
                stats.accepted += 1;
                waiting.push(Waiting {
                    index: idx,
                    entry: now,
                    deadline: initial_deadlines[idx],
                    waited: 0,
                    paused: false,
                    pause_time: 0,
                    remaining: c.service,
                });
            }
        }

        if let Some(a) = active.as_ref() {
            if now >= a.work_at && !waiting.is_empty() {
                let best = (0..waiting.len())
                    .min_by(|&i, &j| waiting_cmp(&waiting[i], &waiting[j], customers))
                    .unwrap();
                if customers[waiting[best].index].priority
                    < customers[a.index].priority
                {
                    let a = active.take().unwrap();
                    let paused_id = customers[a.index].id;
                    let by = customers[waiting[best].index].id;
                    let remaining = a.finish - now;
                    emit(
                        &mut events,
                        now,
                        paused_id,
                        Kind::Preempt { remaining, by },
                    );
                    stats.preemptions += 1;
                    stats.busy_ticks += (now - a.work_at.min(now)) as u128;
                    last_event_time = now;

                    let deadline = match customers[a.index].patience {
                        Some(_) => Some(add_time(
                            paused_id,
                            now,
                            customers[a.index].patience.unwrap(),
                            OverflowStage::Patience,
                        )?),
                        None => None,
                    };
                    waiting.push(Waiting {
                        index: a.index,
                        entry: now,
                        deadline,
                        waited: total_waited[a.index],
                        paused: true,
                        pause_time: now,
                        remaining,
                    });
                }
            }
        }

        loop {
            if active.is_some() || waiting.is_empty() {
                break;
            }
            let best = (0..waiting.len())
                .min_by(|&i, &j| waiting_cmp(&waiting[i], &waiting[j], customers))
                .unwrap();
            let w = waiting.remove(best);
            let c = &customers[w.index];
            let segment = now - w.entry;
            total_waited[w.index] += segment;
            stats.wait_ticks += segment as u128;

            let switch = if last_dispatched.is_some() && last_dispatched != Some(c.id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = add_time(c.id, now, switch, OverflowStage::WorkAt)?;
            let finish = add_time(c.id, work_at, w.remaining, OverflowStage::Finish)?;
            stats.busy_ticks += (finish - now) as u128;
            let kind = if ever_started[w.index] {
                Kind::Resume {
                    remaining: w.remaining,
                    work_at,
                    finish,
                }
            } else {
                ever_started[w.index] = true;
                stats.started += 1;
                Kind::Start {
                    remaining: w.remaining,
                    work_at,
                    finish,
                }
            };
            last_dispatched = Some(c.id);
            emit(&mut events, now, c.id, kind);
            last_event_time = now;
            if finish == now {
                emit(&mut events, now, c.id, Kind::Finish);
                stats.completed += 1;
                completed[w.index] = true;
                stats.turnaround_ticks += (now - c.arrival) as u128;
                stats.completion_order.push(c.id);
                last_event_time = now;
            } else {
                active = Some(Active {
                    index: w.index,
                    remaining: w.remaining,
                    work_at,
                    finish,
                });
            }
        }

        let mut post_expired: Vec<usize> = waiting
            .iter()
            .enumerate()
            .filter_map(|(i, w)| w.deadline.filter(|&d| d <= now).map(|_| i))
            .collect();
        post_expired.sort_by(|&a, &b| {
            waiting[a]
                .deadline
                .cmp(&waiting[b].deadline)
                .then(customers[waiting[a].index].id.cmp(&customers[waiting[b].index].id))
        });
        for wi in post_expired.into_iter().rev() {
            let w = waiting.remove(wi);
            let c = &customers[w.index];
            let segment = now - w.entry;
            total_waited[w.index] += segment;
            stats.wait_ticks += segment as u128;
            emit(
                &mut events,
                now,
                c.id,
                Kind::Renege {
                    waited: total_waited[w.index],
                },
            );
            stats.reneged += 1;
            last_event_time = now;
        }

        if stats.completed + stats.reneged + stats.balked >= n {
            break;
        }

        let mut next_time: Option<u64> = None;
        if next_arrival < n {
            next_time = Some(customers[arrivals[next_arrival]].arrival);
        }
        if let Some(a) = active.as_ref() {
            next_time = Some(next_time.map_or(a.finish, |t| t.min(a.finish)));
            if a.work_at > now {
                next_time = Some(next_time.map_or(a.work_at, |t| t.min(a.work_at)));
            }
        }
        for w in &waiting {
            if let Some(d) = w.deadline {
                if d > now {
                    next_time = Some(next_time.map_or(d, |t| t.min(d)));
                }
            }
        }
        match next_time {
            Some(t) if t > now => now = t,
            _ => break,
        }
    }

    stats.elapsed = if events.is_empty() { 0 } else { last_event_time };
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10_000, stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}
```
