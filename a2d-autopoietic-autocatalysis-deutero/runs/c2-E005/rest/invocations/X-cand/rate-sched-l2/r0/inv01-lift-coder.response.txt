#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    reserved_ids: Vec<String>,
    last_poll: Option<u64>,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b' ' { Some(i) } else { None })
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];

        parse_field(&text[..split], 0, 59, &mut minutes)?;
        parse_field(&text[split + 1..], split + 1, 23, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    field: &str,
    offset: usize,
    max: u32,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = field.as_bytes();
    let mut start = 0;

    loop {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| start + p)
            .unwrap_or(bytes.len());

        if start == end {
            return Err(CronError::Syntax(offset + start));
        }

        parse_item(&field[start..end], offset + start, max, selected)?;

        if end == bytes.len() {
            break;
        }
        start = end + 1;
    }

    Ok(())
}

fn parse_item(
    item: &str,
    offset: usize,
    max: u32,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = item.as_bytes();

    let slashes: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
        .collect();

    if slashes.len() > 1 {
        return Err(CronError::Syntax(offset + slashes[1]));
    }

    let slash = slashes.first().copied();
    let base_end = slash.unwrap_or(bytes.len());

    if let Some(slash_pos) = slash {
        let base = &bytes[..base_end];
        if base != b"*" && !base.contains(&b'-') {
            return Err(CronError::Syntax(offset + slash_pos));
        }
    }

    let (first, last, dash_pos) = if &bytes[..base_end] == b"*" {
        (0, max, None)
    } else if let Some(dash) = bytes[..base_end].iter().position(|&b| b == b'-') {
        let left = parse_number(&bytes[..dash], offset, max)?;
        let right = parse_number(&bytes[dash + 1..base_end], offset + dash + 1, max)?;
        if left > right {
            return Err(CronError::Descending(offset + dash));
        }
        (left, right, Some(dash))
    } else {
        let value = parse_number(&bytes[..base_end], offset, max)?;
        (value, value, None)
    };

    let step = if let Some(slash_pos) = slash {
        let step_start = slash_pos + 1;
        let step_bytes = &bytes[step_start..];
        let value = parse_number(step_bytes, offset + step_start, max)?;
        if value == 0 || value > max + 1 {
            return Err(CronError::Step(offset + step_start));
        }
        value
    } else {
        1
    };

    let _ = dash_pos;
    let mut value = first;
    while value <= last {
        selected[value as usize] = true;
        match value.checked_add(step) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }

    Ok(())
}

fn parse_number(fragment: &[u8], offset: usize, max: u32) -> Result<u32, CronError> {
    if fragment.is_empty() {
        return Err(CronError::Syntax(offset));
    }

    for (i, &b) in fragment.iter().enumerate() {
        if !b.is_ascii_digit() {
            return Err(CronError::Syntax(offset + i));
        }
    }

    if fragment.len() > 1 && fragment[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let mut value: u32 = 0;
    for &b in fragment {
        value = match value
            .checked_mul(10)
            .and_then(|n| n.checked_add((b - b'0') as u32))
        {
            Some(n) => n,
            None => return Err(CronError::Bounds(offset)),
        };
    }

    if value > max {
        return Err(CronError::Bounds(offset));
    }

    Ok(value)
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            reserved_ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        let id_bytes = id.as_bytes();
        if id_bytes.is_empty()
            || id_bytes.len() > 16
            || !id_bytes
                .iter()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
        {
            return Err(Error::BadId);
        }

        if priority > 9 {
            return Err(Error::BadPriority);
        }

        if self.reserved_ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }

        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let owned_id = id.to_owned();
        self.reserved_ids.push(owned_id.clone());
        self.jobs.push(Job {
            id: owned_id,
            cron,
            priority,
            policy,
            order: self.jobs.len(),
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let previous = self.last_poll;
        let lower_exclusive = previous;
        let upper = now / 60_000;
        let lower_inclusive = match previous {
            Some(last) => last / 60_000 + 1,
            None => 0,
        };

        let mut occurrences = Vec::new();

        for job in &self.jobs {
            let first_minute = lower_inclusive;
            if first_minute > upper {
                continue;
            }

            match job.policy {
                Policy::All => {
                    for minute in first_minute..=upper {
                        if job.cron.matches(minute) {
                            occurrences.push(Occurrence {
                                id: job.id.clone(),
                                at: minute * 60_000,
                            });
                        }
                    }
                }
                Policy::Once => {
                    let mut latest = None;
                    for minute in first_minute..=upper {
                        if job.cron.matches(minute) {
                            latest = Some(minute);
                        }
                    }
                    if let Some(minute) = latest {
                        occurrences.push(Occurrence {
                            id: job.id.clone(),
                            at: minute * 60_000,
                        });
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0 {
                        let minute = now / 60_000;
                        let in_interval = match lower_exclusive {
                            Some(last) => minute * 60_000 > last,
                            None => true,
                        };
                        if in_interval && job.cron.matches(minute) {
                            occurrences.push(Occurrence {
                                id: job.id.clone(),
                                at: now,
                            });
                        }
                    }
                }
            }
        }

        occurrences.sort_by(|a, b| {
            a.at.cmp(&b.at).then_with(|| {
                let ja = self.jobs.iter().find(|j| j.id == a.id).unwrap();
                let jb = self.jobs.iter().find(|j| j.id == b.id).unwrap();
                jb.priority
                    .cmp(&ja.priority)
                    .then_with(|| ja.order.cmp(&jb.order))
            })
        });

        self.last_poll = Some(now);
        Ok(occurrences)
    }
}
