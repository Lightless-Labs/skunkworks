#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    insertion_order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let minute_field = &text[..split];
        let hour_field = &text[split + 1..];

        let mut minutes = [false; 60];
        let mut hours = [false; 24];

        parse_field(minute_field, 0, 60, &mut minutes)?;
        parse_field(hour_field, split + 1, 24, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    field: &str,
    offset: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut start = 0usize;

    loop {
        let end = field[start..]
            .find(',')
            .map(|relative| start + relative)
            .unwrap_or(field.len());

        if start == end {
            return Err(CronError::Syntax(offset + start));
        }

        parse_item(&field[start..end], offset + start, limit, selected)?;

        if end == field.len() {
            break;
        }
        start = end + 1;
    }

    Ok(())
}

fn parse_item(
    item: &str,
    offset: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = item.as_bytes();

    let slash = bytes.iter().position(|&b| b == b'/');
    if let Some(first_slash) = slash {
        if let Some(second_relative) = bytes[first_slash + 1..]
            .iter()
            .position(|&b| b == b'/')
        {
            return Err(CronError::Syntax(offset + first_slash + 1 + second_relative));
        }
    }

    let (base_end, step_text, step_offset) = match slash {
        Some(i) => (&item[..i], Some(&item[i + 1..]), offset + i + 1),
        None => (item, None, 0),
    };

    if slash.is_some() && base_end != "*" && !base_end.contains('-') {
        return Err(CronError::Syntax(offset + slash.unwrap()));
    }

    let (range_start, range_end) = if base_end == "*" {
        (0, limit - 1)
    } else if let Some(dash) = base_end.as_bytes().iter().position(|&b| b == b'-') {
        let left = parse_number(&base_end[..dash], offset, limit)?;
        let right = parse_number(
            &base_end[dash + 1..],
            offset + dash + 1,
            limit,
        )?;
        if left > right {
            return Err(CronError::Descending(offset + dash));
        }
        (left, right)
    } else {
        let number = parse_number(base_end, offset, limit)?;
        (number, number)
    };

    let step = if let Some(step_text) = step_text {
        parse_step(step_text, step_offset, limit)?
    } else {
        1
    };

    let mut value = range_start;
    while value <= range_end {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_number(fragment: &str, offset: usize, limit: usize) -> Result<usize, CronError> {
    let bytes = fragment.as_bytes();

    if bytes.is_empty() {
        return Err(CronError::Syntax(offset));
    }
    if let Some(i) = bytes.iter().position(|b| !b.is_ascii_digit()) {
        return Err(CronError::Syntax(offset + i));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let mut value = 0usize;
    for &b in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|n| n.checked_add((b - b'0') as usize))
        {
            Some(n) => n,
            None => return Err(CronError::Bounds(offset)),
        };
        if value >= limit {
            return Err(CronError::Bounds(offset));
        }
    }

    Ok(value)
}

fn parse_step(fragment: &str, offset: usize, limit: usize) -> Result<usize, CronError> {
    let bytes = fragment.as_bytes();

    if bytes.is_empty() {
        return Err(CronError::Syntax(offset));
    }
    if let Some(i) = bytes.iter().position(|b| !b.is_ascii_digit()) {
        return Err(CronError::Syntax(offset + i));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let max_step = limit;
    let mut value = 0usize;
    for &b in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|n| n.checked_add((b - b'0') as usize))
        {
            Some(n) => n,
            None => return Err(CronError::Step(offset)),
        };
        if value > max_step {
            return Err(CronError::Step(offset));
        }
    }

    if value == 0 || value > max_step {
        return Err(CronError::Step(offset));
    }

    Ok(value)
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|job| job.id == id) {
            return Err(Error::Duplicate);
        }

        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let insertion_order = self.jobs.len();

        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            insertion_order,
        });

        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        const MINUTE_MS: u64 = 60_000;
        let first_minute = match self.last_poll {
            Some(last) => last / MINUTE_MS + 1,
            None => 0,
        };
        let last_minute = now / MINUTE_MS;

        let mut output = Vec::new();

        if first_minute <= last_minute {
            for job in &self.jobs {
                match job.policy {
                    Policy::All => {
                        for minute in first_minute..=last_minute {
                            if job.cron.matches(minute) {
                                output.push(Occurrence {
                                    id: job.id.clone(),
                                    at: minute * MINUTE_MS,
                                });
                            }
                        }
                    }
                    Policy::Once => {
                        let mut latest = None;
                        for minute in first_minute..=last_minute {
                            if job.cron.matches(minute) {
                                latest = Some(minute);
                            }
                        }
                        if let Some(minute) = latest {
                            output.push(Occurrence {
                                id: job.id.clone(),
                                at: minute * MINUTE_MS,
                            });
                        }
                    }
                    Policy::Skip => {
                        if now % MINUTE_MS == 0 {
                            let minute = now / MINUTE_MS;
                            if minute >= first_minute
                                && minute <= last_minute
                                && job.cron.matches(minute)
                            {
                                output.push(Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                });
                            }
                        }
                    }
                }
            }
        }

        output.sort_by(|a, b| {
            let a_job = self.jobs.iter().find(|job| job.id == a.id).unwrap();
            let b_job = self.jobs.iter().find(|job| job.id == b.id).unwrap();

            a.at.cmp(&b.at)
                .then_with(|| b_job.priority.cmp(&a_job.priority))
                .then_with(|| a_job.insertion_order.cmp(&b_job.insertion_order))
        });

        self.last_poll = Some(now);
        Ok(output)
    }
}
