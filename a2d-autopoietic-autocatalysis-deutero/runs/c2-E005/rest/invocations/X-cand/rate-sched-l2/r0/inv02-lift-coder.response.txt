#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 {
            return Err(CronError::Layout(0));
        }
        if split + 1 == text.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(&text[..split], 0, 59, &mut minutes)?;
        parse_field(&text[split + 1..], split + 1, 23, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    field: &str,
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut start = 0;
    loop {
        let end = field[start..]
            .find(',')
            .map(|i| start + i)
            .unwrap_or(field.len());
        parse_item(&field[start..end], offset + start, max, selected)?;
        if end == field.len() {
            break;
        }
        start = end + 1;
    }
    Ok(())
}

fn parse_number(fragment: &str, start: usize, max: usize) -> Result<usize, CronError> {
    let bytes = fragment.as_bytes();
    if bytes.is_empty() {
        return Err(CronError::Syntax(start));
    }
    for (i, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() {
            return Err(CronError::Syntax(start + i));
        }
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for &b in bytes {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as usize)) {
            Some(v) => v,
            None => return Err(CronError::Bounds(start)),
        };
    }
    if value > max {
        return Err(CronError::Bounds(start));
    }
    Ok(value)
}

fn parse_item(
    item: &str,
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = item.as_bytes();

    if let Some(slash) = bytes.iter().position(|&b| b == b'/') {
        if let Some(second) = bytes[slash + 1..].iter().position(|&b| b == b'/') {
            return Err(CronError::Syntax(offset + slash + 1 + second));
        }
    }

    let slash = bytes.iter().position(|&b| b == b'/');
    let (base, step_fragment, step_offset) = match slash {
        Some(i) => {
            let base = &item[..i];
            if base != "*" && !base.contains('-') {
                return Err(CronError::Syntax(offset + i));
            }
            (base, Some(&item[i + 1..]), offset + i + 1)
        }
        None => (item, None, 0),
    };

    let (range_start, range_end) = if base == "*" {
        (0, max)
    } else if base.is_empty() {
        return Err(CronError::Syntax(offset));
    } else if let Some(dash) = base.as_bytes().iter().position(|&b| b == b'-') {
        let left = parse_number(&base[..dash], offset, max)?;
        let right = parse_number(&base[dash + 1..], offset + dash + 1, max)?;
        if left > right {
            return Err(CronError::Descending(offset + dash));
        }
        (left, right)
    } else {
        let value = parse_number(base, offset, max)?;
        (value, value)
    };

    let step = match step_fragment {
        Some(fragment) => {
            let value = parse_number(fragment, step_offset, max)?;
            if value == 0 {
                return Err(CronError::Step(step_offset));
            }
            value
        }
        None => 1,
    };

    let mut value = range_start;
    loop {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) if next <= range_end => value = next,
            _ => break,
        }
    }
    Ok(())
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|job| job.id == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let order = self.jobs.len();
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower = self.last_poll;
        let mut output: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            match job.policy {
                Policy::All => {
                    let mut boundary = match lower {
                        Some(last) => (last / 60_000)
                            .checked_add(1)
                            .and_then(|m| m.checked_mul(60_000)),
                        None => Some(0),
                    };
                    while let Some(at) = boundary {
                        if at > now {
                            break;
                        }
                        if job.cron.matches(at / 60_000) {
                            output.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                        boundary = at.checked_add(60_000);
                    }
                }
                Policy::Once => {
                    let mut boundary = match lower {
                        Some(last) => (last / 60_000)
                            .checked_add(1)
                            .and_then(|m| m.checked_mul(60_000)),
                        None => Some(0),
                    };
                    let mut latest = None;
                    while let Some(at) = boundary {
                        if at > now {
                            break;
                        }
                        if job.cron.matches(at / 60_000) {
                            latest = Some(at);
                        }
                        boundary = at.checked_add(60_000);
                    }
                    if let Some(at) = latest {
                        output.push((
                            Occurrence {
                                id: job.id.clone(),
                                at,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0
                        && lower.map_or(true, |last| now > last)
                        && job.cron.matches(now / 60_000)
                    {
                        output.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: now,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
            }
        }

        output.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(output.into_iter().map(|(occurrence, _, _)| occurrence).collect())
    }
}
