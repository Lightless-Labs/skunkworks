#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let Some(high) = hex_value(bytes[i + 1]) else {
                return Err(CodecError::BadEscape { at: i });
            };
            let Some(low) = hex_value(bytes[i + 2]) else {
                return Err(CodecError::BadEscape { at: i });
            };
            decoded.push((high << 4) | low);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let offset = error.valid_up_to();
            Err(CodecError::InvalidUtf8 {
                at: origins.get(offset).copied().unwrap_or(input.len()),
            })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };
        if safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn scan_global_bytes(input: &str) -> Result<(), ParseError> {
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

fn parse_port(port: &str, at: usize) -> Result<Option<u16>, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }

    for (i, byte) in port.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: at + i });
        }
    }

    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at });
    }

    let mut value: u32 = 0;
    for byte in port.bytes() {
        value = match value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
        {
            Some(value) => value,
            None => return Err(ParseError::InvalidPort { at }),
        };
        if value > u16::MAX as u32 {
            return Err(ParseError::InvalidPort { at });
        }
    }

    if value == 0 {
        return Err(ParseError::InvalidPort { at });
    }
    Ok(Some(value as u16))
}

fn validate_component(component: Component, value: &str, base: usize) -> Result<(), ParseError> {
    match decode_component(component, value) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: base + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: base + at }),
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    scan_global_bytes(input)?;

    let bytes = input.as_bytes();
    let Some(colon) = bytes.iter().position(|&byte| byte == b':') else {
        return Err(ParseError::InvalidScheme { at: input.len() });
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }
    let scheme = input[..colon].to_ascii_lowercase();

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(input.len());
    let authority = &input[authority_start..authority_end];

    let mut at_signs = authority.match_indices('@');
    let first_at = at_signs.next();
    let second_at = at_signs.next();
    if let Some((second, _)) = second_at {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + second,
        });
    }

    let (userinfo, host_port_start) = if let Some((position, _)) = first_at {
        if position == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + position,
            });
        }
        (
            Some(authority[..position].to_owned()),
            authority_start + position + 1,
        )
    } else {
        (None, authority_start)
    };

    let host_port = &input[host_port_start..authority_end];
    let mut ipv6 = false;
    let (host, port) = if host_port.starts_with('[') {
        let Some(close_relative) = host_port.as_bytes()[1..]
            .iter()
            .position(|&byte| byte == b']')
        else {
            return Err(ParseError::InvalidIpv6 { at: host_port_start });
        };
        let close = close_relative + 1;
        let interior = &host_port[1..close];
        if interior.is_empty()
            || interior.contains('%')
            || interior.parse::<std::net::Ipv6Addr>().is_err()
        {
            return Err(ParseError::InvalidIpv6 { at: host_port_start });
        }

        let suffix = &host_port[close + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if let Some(port_text) = suffix.strip_prefix(':') {
            parse_port(port_text, host_port_start + close + 2)?
        } else {
            return Err(ParseError::InvalidHost {
                at: host_port_start + close + 1,
            });
        };
        ipv6 = true;
        (interior.to_ascii_lowercase(), parsed_port)
    } else {
        let split = host_port.find(':');
        let (host_text, port_text) = match split {
            Some(position) => (
                &host_port[..position],
                Some((&host_port[position + 1..], host_port_start + position + 1)),
            ),
            None => (host_port, None),
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_port_start });
        }
        for (offset, byte) in host_text.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_port_start + offset,
                });
            }
        }
        let parsed_port = match port_text {
            Some((text, at)) => parse_port(text, at)?,
            None => None,
        };
        (host_text.to_ascii_lowercase(), parsed_port)
    };

    let fragment_start = bytes[authority_end..]
        .iter()
        .position(|&byte| byte == b'#')
        .map(|offset| authority_end + offset);
    let before_fragment_end = fragment_start.unwrap_or(input.len());
    let fragment = fragment_start.map(|start| input[start + 1..].to_owned());

    let query_start = bytes[authority_end..before_fragment_end]
        .iter()
        .position(|&byte| byte == b'?')
        .map(|offset| authority_end + offset);
    let before_query_end = query_start.unwrap_or(before_fragment_end);
    let query = query_start.map(|start| input[start + 1..before_fragment_end].to_owned());

    let path = if authority_end == before_query_end {
        "/".to_owned()
    } else {
        input[authority_end..before_query_end].to_owned()
    };

    if let Some(userinfo_value) = userinfo.as_deref() {
        validate_component(Component::UserInfo, userinfo_value, authority_start)?;
    }
    if path != "/" || authority_end != before_query_end {
        validate_component(Component::Path, &path, authority_end)?;
    }
    if let (Some(query_value), Some(start)) = (query.as_deref(), query_start) {
        validate_component(Component::Query, query_value, start + 1)?;
    }
    if let (Some(fragment_value), Some(start)) = (fragment.as_deref(), fragment_start) {
        validate_component(Component::Fragment, fragment_value, start + 1)?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|error| match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_navigation = false;

    for (index, segment) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(|error| match error {
            CodecError::BadEscape { at } => PathError::BadEscape { at: at + 1 },
            CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at: at + 1 },
        })?;
        let is_last = index + 1 == segments.len();
        if decoded == "." {
            terminal_navigation = is_last;
        } else if decoded == ".." {
            if !stack.is_empty() {
                stack.pop();
            }
            terminal_navigation = is_last;
        } else {
            stack.push(segment);
            terminal_navigation = false;
        }
    }

    if terminal_navigation {
        stack.push("");
    }
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn map_path_error(error: PathError) -> ResolveError {
    ResolveError::Path(error)
}

fn normalise_url_path(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(map_path_error)?;
    Ok(url)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_url = normalise_url_path(base_url)?;

    scan_global_bytes(reference).map_err(ResolveError::Reference)?;

    let bytes = reference.as_bytes();
    let first_delimiter = bytes
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if bytes[..first_delimiter].contains(&b':') {
        let parsed = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalise_url_path(parsed);
    }

    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let parsed = parse_url(&constructed).map_err(ResolveError::Reference)?;
        return normalise_url_path(parsed);
    }

    let fragment_start = reference.find('#');
    let before_fragment_end = fragment_start.unwrap_or(reference.len());
    let fragment = fragment_start.map(|start| reference[start + 1..].to_owned());
    let query_start = reference[..before_fragment_end].find('?');
    let before_query_end = query_start.unwrap_or(before_fragment_end);
    let reference_query =
        query_start.map(|start| reference[start + 1..before_fragment_end].to_owned());
    let reference_path = &reference[..before_query_end];

    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_url.path.clone(),
            if query_start.is_none() {
                base_url.query.clone()
            } else {
                reference_query
            },
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = base_url.path.rfind('/').map(|index| index + 1).unwrap_or(0);
            format!("{}{}", &base_url.path[..slash], reference_path)
        };
        (merged, reference_query)
    };

    let userinfo = base_url
        .userinfo
        .as_ref()
        .map(|value| format!("{}@", value))
        .unwrap_or_default();
    let host = if base_url.ipv6 {
        format!("[{}]", base_url.host)
    } else {
        base_url.host.clone()
    };
    let port = base_url
        .port
        .map(|value| format!(":{}", value))
        .unwrap_or_default();
    let path = if selected_path.is_empty() {
        "/".to_owned()
    } else {
        selected_path
    };
    let query = selected_query
        .map(|value| format!("?{}", value))
        .unwrap_or_default();
    let fragment = fragment
        .map(|value| format!("#{}", value))
        .unwrap_or_default();

    let constructed = format!(
        "{}://{}{}{}{}{}{}",
        base_url.scheme, userinfo, host, port, path, query, fragment
    );
    let parsed = parse_url(&constructed).map_err(ResolveError::Reference)?;
    normalise_url_path(parsed)
}
