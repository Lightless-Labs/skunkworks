#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Variable<'a> {
    name: &'a str,
    start: usize,
}

#[derive(Debug)]
struct Expression<'a> {
    operator: Option<char>,
    variables: Vec<Variable<'a>>,
}

#[derive(Debug)]
enum Part<'a> {
    Literal(&'a str),
    Expression(Expression<'a>),
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let parts = parse(template)?;
    let mut output = String::new();

    for part in parts {
        match part {
            Part::Literal(text) => output.push_str(text),
            Part::Expression(expression) => {
                let mut fields = Vec::new();
                for variable in expression.variables {
                    if let Some(value) = lookup(vars, variable.name) {
                        fields.push(encode(value, matches!(expression.operator, Some('+') | Some('#'))));
                    }
                }

                if fields.is_empty() {
                    continue;
                }

                if expression.operator == Some('#') {
                    output.push('#');
                }

                let separator = ",";
                output.push_str(&fields.join(separator));
            }
        }
    }

    Ok(output)
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| *value)
}

fn parse(template: &str) -> Result<Vec<Part<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut parts = Vec::new();
    let mut cursor = 0;
    let mut literal_start = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                if literal_start < cursor {
                    parts.push(Part::Literal(&template[literal_start..cursor]));
                }

                let open = cursor;
                let mut close = cursor + 1;
                while close < bytes.len() && bytes[close] != b'}' {
                    close += 1;
                }

                if close == bytes.len() {
                    return Err(Error::Unclosed { pos: open });
                }

                if let Some(inner) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + inner,
                    });
                }

                parts.push(Part::Expression(parse_expression(
                    template,
                    open + 1,
                    close,
                )?));

                cursor = close + 1;
                literal_start = cursor;
            }
            _ => cursor += 1,
        }
    }

    if literal_start < bytes.len() {
        parts.push(Part::Literal(&template[literal_start..]));
    }

    Ok(parts)
}

fn parse_expression(
    template: &str,
    start: usize,
    close: usize,
) -> Result<Expression<'_>, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start;
    let mut operator = None;

    if content_start < close && matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        let byte = bytes[content_start];
        if byte != b'+' && byte != b'#' {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        operator = Some(byte as char);
        content_start += 1;
    }

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..spec_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let name = &template[spec_start..spec_end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        variables.push(Variable {
            name,
            start: spec_start,
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        let byte = bytes[cursor];

        if byte == b'%' && cursor + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[cursor + 1]), hex_value(bytes[cursor + 2])) {
                result.push('%');
                result.push(hex_digit(high));
                result.push(hex_digit(low));
                cursor += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(hex_digit(byte >> 4));
            result.push(hex_digit(byte & 0x0f));
        }
        cursor += 1;
    }

    result
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    b"0123456789ABCDEF"[value as usize] as char
}
