#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("new encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len != 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                }
                self.tail_len += 1;
                index += 1;

                if self.tail_len == 3 {
                    encode_triplet(
                        self.tail[0],
                        self.tail[1],
                        input[index - 1],
                        &mut output,
                    );
                    self.tail_len = 0;
                }
            }
        }

        while index + 3 <= input.len() {
            encode_triplet(
                input[index],
                input[index + 1],
                input[index + 2],
                &mut output,
            );
            index += 3;
        }

        let remaining = input.len() - index;
        self.tail_len = remaining;
        self.tail[..remaining].copy_from_slice(&input[index..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        self.finished = true;
        let mut output = String::new();
        match self.tail_len {
            1 => {
                let a = self.tail[0];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                output.push_str("==");
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => {}
        }
        self.tail_len = 0;
        Ok(output)
    }
}

fn encode_triplet(a: u8, b: u8, c: u8, output: &mut String) {
    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    output.push(ALPHABET[(c & 0x3f) as usize] as char);
}

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    total: usize,
    terminal: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            len: 0,
            total: 0,
            terminal: false,
            pending_second_padding: None,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.total;
            self.total = self.total.saturating_add(1);

            let result = self.consume(byte, offset, &mut output);
            if let Err(error) = result {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            let error = Error::BadPadding { offset };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }
        if self.len != 0 {
            let error = Error::Truncated {
                offset: self.offsets[0],
            };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        self.finished = true;
        Ok(Vec::new())
    }

    fn consume(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if self.terminal {
            return Err(Error::BadPadding { offset });
        }

        if self.pending_second_padding.is_some() && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        match self.len {
            0 | 1 => {
                if byte == b'=' {
                    return Err(Error::BadPadding { offset });
                }
                let value = digit(byte).ok_or(Error::InvalidByte { offset, byte })?;
                self.group[self.len] = value;
                self.offsets[self.len] = offset;
                self.len += 1;
            }
            2 => {
                if byte == b'=' {
                    self.group[2] = 0;
                    self.offsets[2] = offset;
                    self.len = 3;
                    self.pending_second_padding = Some(offset);
                } else {
                    let value = digit(byte).ok_or(Error::InvalidByte { offset, byte })?;
                    self.group[2] = value;
                    self.offsets[2] = offset;
                    self.len = 3;
                }
            }
            3 => {
                if byte == b'=' {
                    self.offsets[3] = offset;
                    if let Some(first_padding) = self.pending_second_padding.take() {
                        let a = self.group[0];
                        let b = self.group[1];
                        output.push((a << 2) | (b >> 4));
                        self.terminal = true;
                        self.len = 0;
                        let _ = first_padding;
                    } else {
                        let a = self.group[0];
                        let b = self.group[1];
                        let c = self.group[2];
                        output.push((a << 2) | (b >> 4));
                        output.push((b << 4) | (c >> 2));
                        self.terminal = true;
                        self.len = 0;
                    }
                } else {
                    let value = digit(byte).ok_or(Error::InvalidByte { offset, byte })?;
                    if self.pending_second_padding.is_some() {
                        return Err(Error::BadPadding { offset });
                    }
                    let a = self.group[0];
                    let b = self.group[1];
                    let c = self.group[2];
                    output.push((a << 2) | (b >> 4));
                    output.push((b << 4) | (c >> 2));
                    output.push((c << 6) | value);
                    self.len = 0;
                }
            }
            _ => unreachable!(),
        }
        Ok(())
    }
}
