#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = String::new();
        let mut i = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && i < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[i];
                }
                self.tail_len += 1;
                i += 1;

                if self.tail_len == 3 {
                    encode_triplet(
                        self.tail[0],
                        self.tail[1],
                        input[i - 1],
                        &mut out,
                    );
                    self.tail_len = 0;
                    break;
                }
            }
        }

        while i + 3 <= input.len() {
            encode_triplet(input[i], input[i + 1], input[i + 2], &mut out);
            i += 3;
        }

        while i < input.len() {
            self.tail[self.tail_len] = input[i];
            self.tail_len += 1;
            i += 1;
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut out = String::new();
        match self.tail_len {
            1 => {
                let a = self.tail[0];
                out.push(ALPHABET[(a >> 2) as usize] as char);
                out.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                out.push_str("==");
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                out.push(ALPHABET[(a >> 2) as usize] as char);
                out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                out.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                out.push('=');
            }
            _ => {}
        }
        self.tail_len = 0;
        Ok(out)
    }
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    group_len: usize,
    pending_second_padding: bool,
    terminal_padding: bool,
    offset: usize,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            group_len: 0,
            pending_second_padding: false,
            terminal_padding: false,
            offset: 0,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &byte in input {
            let current_offset = self.offset;
            self.offset = self.offset.saturating_add(1);

            if self.terminal_padding {
                return self.fail(Error::BadPadding { offset: current_offset });
            }

            if self.pending_second_padding {
                if byte != b'=' {
                    return self.fail(Error::BadPadding { offset: current_offset });
                }
                self.group[3] = b'=';
                self.offsets[3] = current_offset;
                self.group_len = 4;
                self.pending_second_padding = false;
                self.terminal_padding = true;
                emit_group(&self.group, &mut out);
                self.group_len = 0;
                continue;
            }

            let pos = self.group_len;
            if byte == b'=' {
                match pos {
                    0 | 1 => return self.fail(Error::BadPadding { offset: current_offset }),
                    2 => {
                        self.group[2] = b'=';
                        self.offsets[2] = current_offset;
                        self.group_len = 3;
                        self.pending_second_padding = true;
                    }
                    3 => {
                        self.group[3] = b'=';
                        self.offsets[3] = current_offset;
                        self.group_len = 4;
                        self.terminal_padding = true;
                        emit_group(&self.group, &mut out);
                        self.group_len = 0;
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            let Some(value) = decode_digit(byte) else {
                return self.fail(Error::InvalidByte {
                    offset: current_offset,
                    byte,
                });
            };

            self.group[pos] = value;
            self.offsets[pos] = current_offset;
            self.group_len += 1;
            if self.group_len == 4 {
                emit_group(&self.group, &mut out);
                self.group_len = 0;
            }
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if self.pending_second_padding {
            return self.fail(Error::BadPadding {
                offset: self.offsets[2],
            });
        }
        if self.group_len != 0 {
            return self.fail(Error::Truncated {
                offset: self.offsets[0],
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn encode_triplet(a: u8, b: u8, c: u8, out: &mut String) {
    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    out.push(ALPHABET[(c & 0x3f) as usize] as char);
}

fn decode_digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn emit_group(group: &[u8; 4], out: &mut Vec<u8>) {
    let a = group[0];
    let b = group[1];
    out.push((a << 2) | (b >> 4));

    if group[2] != b'=' {
        let c = group[2];
        out.push((b << 4) | (c >> 2));
        if group[3] != b'=' {
            let d = group[3];
            out.push((c << 6) | d);
        }
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).unwrap_or_default();
    output.push_str(&encoder.finish().unwrap_or_default());
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}
