pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous {
        hunk: usize,
        first: usize,
        second: usize,
    },
}

#[derive(Clone)]
enum RowKind {
    Context,
    Remove,
    Add,
}

#[derive(Clone)]
struct Row {
    kind: RowKind,
    payload: String,
    line: usize,
}

struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn parse_number(s: &str) -> Option<usize> {
    if s.is_empty() || (s.len() > 1 && s.starts_with('0')) {
        return None;
    }
    if !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let rest = line.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(" +")?;
    let (new, suffix) = rest.split_once(" @@")?;
    if suffix != "" {
        return None;
    }
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    let (a, b, c, d) = (
        parse_number(a)?,
        parse_number(b)?,
        parse_number(c)?,
        parse_number(d)?,
    );
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    Some((a, b, c, d))
}

fn full_start(start: usize, count: usize) -> usize {
    if count > 0 { start - 1 } else { start }
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }

    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines: Vec<&str> = patch[..patch.len() - 1].split('\n').collect();
    for (i, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(i).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }

    if lines.len() <= 2 || !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_old_end = 0usize;
    let mut preceding_net: i128 = 0;

    while i < lines.len() {
        let header_line = i + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[i]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        let old_start = full_start(a, b);
        let new_start = full_start(c, d);

        i += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;

        while i < lines.len() && !lines[i].starts_with("@@") {
            let row_line = i + 1;
            let row = lines[i];
            if row.is_empty() {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let (kind, payload) = match row.as_bytes()[0] {
                b' ' => (RowKind::Context, &row[1..]),
                b'-' => (RowKind::Remove, &row[1..]),
                b'+' => (RowKind::Add, &row[1..]),
                _ => return Err(ApplyError::BadBody { line: row_line }),
            };
            if payload.contains('\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            match kind {
                RowKind::Context => {
                    old_seen += 1;
                    new_seen += 1;
                }
                RowKind::Remove => old_seen += 1,
                RowKind::Add => new_seen += 1,
            }
            rows.push(Row {
                kind,
                payload: payload.to_string(),
                line: row_line,
            });
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| !matches!(r.kind, RowKind::Context)) {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if old_start < previous_old_end {
            return Err(ApplyError::Misordered { line: header_line });
        }
        let required_new = old_start as i128 + preceding_net;
        if required_new < 0 || required_new != new_start as i128 {
            return Err(ApplyError::Misordered { line: header_line });
        }
        previous_old_end = old_start + b;
        preceding_net += d as i128 - b as i128;

        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }
        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });
    }

    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }
    let lines: Vec<String> = source[..source.len() - 1]
        .split('\n')
        .map(str::to_string)
        .collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

fn abs_diff_i128(a: i128, b: i128) -> u128 {
    (a - b).unsigned_abs()
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;
    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (index, hunk) in hunks.iter().enumerate() {
        let mut lead = 0usize;
        while lead < hunk.rows.len() && matches!(hunk.rows[lead].kind, RowKind::Context) {
            lead += 1;
        }
        let mut trail = 0usize;
        while trail < hunk.rows.len()
            && matches!(hunk.rows[hunk.rows.len() - 1 - trail].kind, RowKind::Context)
        {
            trail += 1;
        }
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, u128)> = None;
        let mut best_starts = Vec::<usize>::new();
        let mut chosen: Option<(usize, usize, usize)> = None;

        for left in 0..=options.fuzz.min(lead) {
            for right in 0..=options.fuzz.min(trail) {
                if left + right > hunk.old_count {
                    continue;
                }
                let mut old_payloads = Vec::new();
                for row in &hunk.rows {
                    if matches!(row.kind, RowKind::Context | RowKind::Remove) {
                        old_payloads.push(row.payload.as_str());
                    }
                }
                let old_payloads = &old_payloads[left..old_payloads.len() - right];

                if hunk.old_count > working.len() {
                    continue;
                }
                for start in 0..=working.len() - hunk.old_count {
                    let distance = abs_diff_i128(start as i128, expected);
                    if distance > options.max_offset as u128 {
                        continue;
                    }
                    let rank = (left + right, distance);
                    let begin = start + left;
                    let end = start + hunk.old_count - right;
                    if old_payloads.len() != end - begin
                        || !old_payloads
                            .iter()
                            .zip(&working[begin..end])
                            .all(|(expected, actual)| *expected == actual)
                    {
                        continue;
                    }

                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(start);
                            chosen = Some((start, left, right));
                        }
                        Some(current) if rank < current => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(start);
                            chosen = Some((start, left, right));
                        }
                        Some(current) if rank == current => {
                            if !best_starts.contains(&start) {
                                best_starts.push(start);
                            }
                            if let Some((chosen_start, chosen_left, _)) = chosen {
                                if start == chosen_start && left < chosen_left {
                                    chosen = Some((start, left, right));
                                }
                            }
                        }
                        _ => {}
                    }
                }
            }
        }

        let Some((start, left, right)) = chosen else {
            return Err(ApplyError::NoMatch { hunk: index + 1 });
        };
        if best_starts.len() > 1 {
            best_starts.sort_unstable();
            return Err(ApplyError::Ambiguous {
                hunk: index + 1,
                first: best_starts[0],
                second: best_starts[1],
            });
        }

        let replace_start = start + left;
        let replace_end = start + hunk.old_count - right;
        let mut replacement = Vec::new();
        let mut context_index = 0usize;
        for row in &hunk.rows {
            match row.kind {
                RowKind::Context => {
                    if context_index >= left && context_index < hunk.rows.len() - right {
                        replacement.push(row.payload.clone());
                    }
                    context_index += 1;
                }
                RowKind::Add => replacement.push(row.payload.clone()),
                RowKind::Remove => {}
            }
        }

        working.splice(replace_start..replace_end, replacement);
        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = start as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}

fn parse_number(bytes: &[u8]) -> Option<usize> {
    if bytes.is_empty() || (bytes.len() > 1 && bytes[0] == b'0') {
        return None;
    }
    let mut value = 0usize;
    for &b in bytes {
        if !b.is_ascii_digit() {
            return None;
        }
        value = value.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if value > 128 {
            return None;
        }
    }
    Some(value)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let b = line.as_bytes();
    if b.len() < 11 || !b.starts_with(b"@@ -") || !b.ends_with(b" @@") {
        return None;
    }
    let core = &b[4..b.len() - 3];
    let comma1 = core.iter().position(|&x| x == b',')?;
    let plus = core.iter().position(|&x| x == b' ')?;
    if plus <= comma1 || core.get(plus + 1) != Some(&b'+') {
        return None;
    }
    let comma2 = core[plus + 2..].iter().position(|&x| x == b',')? + plus + 2;
    if comma2 <= plus + 2 {
        return None;
    }
    let a = parse_number(&core[..comma1])?;
    let count_b = parse_number(&core[comma1 + 1..plus])?;
    let c = parse_number(&core[plus + 2..comma2])?;
    let count_d = parse_number(&core[comma2 + 1..])?;
    if (count_b > 0 && a == 0) || (count_d > 0 && c == 0) {
        return None;
    }
    Some((a, count_b, c, count_d))
}

fn old_full_start(a: usize, b: usize) -> usize {
    if b > 0 { a - 1 } else { a }
}

fn new_full_start(c: usize, d: usize) -> usize {
    if d > 0 { c - 1 } else { c }
}

fn patch_lines(patch: &str) -> Vec<&str> {

}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines = patch_lines(patch);
    for (i, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(i).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() < 3 || !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_end = 0usize;
    let mut net: i64 = 0;
    while i < lines.len() {
        let header_line = i + 1;
        let (a, b, c, d) = parse_hunk_header(lines[i])
            .ok_or(ApplyError::BadHunkHeader { line: header_line })?;
        let old_start = old_full_start(a, b);
        let new_start = new_full_start(c, d);

        i += 1;
        let body_start = i;
        let mut rows = Vec::new();
        while i < lines.len() && !lines[i].starts_with("@@") {
            let raw = lines[i].as_bytes();
            if raw.is_empty() || !matches!(raw[0], b' ' | b'-' | b'+') {
                return Err(ApplyError::BadBody { line: i + 1 });
            }
            if raw[1..].contains(&b'\r') {
                return Err(ApplyError::BadBody { line: i + 1 });
            }
            rows.push(Row {
                prefix: raw[0],
                payload: lines[i][1..].to_owned(),
                line: i + 1,
            });
            i += 1;
        }

        let old_actual = rows.iter().filter(|r| r.prefix != b'+').count();
        let new_actual = rows.iter().filter(|r| r.prefix != b'-').count();
        if old_actual != b || new_actual != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.prefix == b'+' || r.prefix == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if old_start < previous_end {
            return Err(ApplyError::Misordered { line: header_line });
        }
        let required_new = old_start as i64 + net;
        if required_new < 0 || new_start as i64 != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }
        previous_end = old_start + b;
        net += d as i64 - b as i64;

        let lead = rows.iter().take_while(|r| r.prefix == b' ').count();
        let trail = rows.iter().rev().take_while(|r| r.prefix == b' ').count();
        hunks.push(Hunk {
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
            line: header_line,
            lead,
            trail,
        });
        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }
        if i == body_start {
            // Empty bodies are valid structurally only when their declared counts
            // are zero; the required change-row check above rejects those.
        }
    }
    Ok(hunks)
}

fn abs_distance(a: i64, b: i64) -> u64 {
    a.abs_diff(b)
}

    }

    let hunks = parse_patch(patch)?;

    if let Some(byte) = source.bytes().position(|b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let mut working: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source[..source.len() - 1]
            .split('\n')
            .map(str::to_owned)
            .collect()
    };
    if working.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    let mut net: i64 = 0;
    let mut drift: i64 = 0;
    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let expected = hunk.old_start as i64 + net + drift;
        let max_l = options.fuzz.min(hunk.lead);
        let max_r = options.fuzz.min(hunk.trail);
        let mut best_rank: Option<(usize, u64)> = None;
        let mut best_starts: Vec<(usize, usize, usize)> = Vec::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > hunk.old_count {
                    continue;
                }
                let mut match_payloads = Vec::new();
                for row in &hunk.rows {
                    if row.prefix == b' ' || row.prefix == b'-' {
                        match_payloads.push(row.payload.as_str());
                    }
                }
                let trimmed: Vec<&str> = hunk.rows
                    .iter()
                    .enumerate()
                    .filter(|(idx, row)| {
                        !((*idx < l && row.prefix == b' ')
                            || (*idx >= hunk.rows.len().saturating_sub(r)
                                && row.prefix == b' '))
                    })
                    .filter_map(|(_, row)| {
                        if row.prefix == b' ' || row.prefix == b'-' {
                            Some(row.payload.as_str())
                        } else {
                            None
                        }
                    })
                    .collect();

                for s in 0..=working.len().saturating_sub(hunk.old_count) {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = abs_distance(s as i64, expected);
                    if distance > options.max_offset as u64 {
                        continue;
                    }
                    let start = s + l;
                    let end = s + hunk.old_count - r;
                    if end < start || end - start != trimmed.len() {
                        continue;
                    }
                    if working[start..end]
                        .iter()
                        .map(String::as_str)
                        .ne(trimmed.iter().copied())
                    {
                        continue;
                    }
                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push((s, l, r));
                        }
                        Some(old) if rank < old => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push((s, l, r));
                        }
                        Some(old) if rank == old => best_starts.push((s, l, r)),
                        _ => {}
                    }
                }
                let _ = match_payloads;
            }
        }

        if best_starts.is_empty() {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        }
        let mut distinct: Vec<usize> = best_starts.iter().map(|x| x.0).collect();
        distinct.sort_unstable();
        distinct.dedup();
        if distinct.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: distinct[0],
                second: distinct[1],
            });
        }

        let s = distinct[0];
        let (_, l, r) = best_starts
            .iter()
            .filter(|x| x.0 == s)
            .min_by_key(|x| x.1)
            .copied()
            .unwrap();

        let trim_start = l;
        let trim_end = hunk.rows.len().saturating_sub(r);
        let mut replacement = Vec::new();
        for row in &hunk.rows[trim_start..trim_end] {
            if row.prefix == b' ' || row.prefix == b'+' {
                replacement.push(row.payload.clone());
            }
        }
        let replace_start = s + l;
        let replace_end = s + hunk.old_count - r;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }
        drift = s as i64 - (hunk.old_start as i64 + net);
        net += hunk.new_count as i64 - hunk.old_count as i64;
    }

    let mut result = String::new();
    for line in working {
        result.push_str(&line);
        result.push('\n');
    }
    Ok(result)
}

fn parse_number(s: &str) -> Option<i64> {

        return None;
    }
    if s.len() > 1 && s.starts_with('0') {

fn parse_hunk_header(line: &str) -> Option<(i64, i64, i64, i64)> {
    let rest = line.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(" +")?;
    let (new, ending) = rest.split_once(" @@")?;

        return None;

    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;

        parse_number(a)?,
        parse_number(b)?,
        parse_number(c)?,
        parse_number(d)?,

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines = physical_lines(patch);
    for i in 0..2 {
        let expected = if i == 0 { "--- before" } else { "+++ after" };
        if lines.get(i).copied() != Some(expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() <= 2 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_old_end = 0i64;
    let mut net = 0i64;

    while i < lines.len() {
        let header_line = i + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[i]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        if (b > 0 && a == 0) || (d > 0 && c == 0) {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        }
        let old_start = if b > 0 { a - 1 } else { a };
        let new_start = if d > 0 { c - 1 } else { c };
        let old_count = b as usize;
        let new_count = d as usize;

        i += 1;
        let mut rows = Vec::new();
        let mut seen_old = 0usize;
        let mut seen_new = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let row = lines[i];
            if row.is_empty() {
                return Err(ApplyError::BadBody { line: i + 1 });
            }
            let (kind, payload) = row.split_at(1);
            let kind = kind.as_bytes()[0];
            if !matches!(kind, b' ' | b'-' | b'+') || payload.contains('\r') {
                return Err(ApplyError::BadBody { line: i + 1 });
            }
            if kind == b' ' || kind == b'-' {
                seen_old += 1;
            }
            if kind == b' ' || kind == b'+' {
                seen_new += 1;
            }
            rows.push(BodyRow {
                kind,
                payload: payload.to_owned(),
            });
            i += 1;
        }

        if seen_old != old_count || seen_new != new_count {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.kind == b'+' || r.kind == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if old_start < previous_old_end
            || new_start != old_start + net
            || new_start < 0
        {
            return Err(ApplyError::Misordered { line: header_line });
        }
        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }

        previous_old_end = old_start + b;
        net += d - b;
        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count,
            new_start,
            new_count,
            rows,
        });
    }

    Ok(hunks)
}

fn parse_source(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.bytes().position(|b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let lines: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source
            .split_terminator('\n')
            .map(str::to_owned)
            .collect()
    };
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = parse_source(source)?;
    let mut net = 0i64;
    let mut drift = 0i64;

    for (index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| r.kind == b' ').count();
        let trail = hunk.rows.iter().rev().take_while(|r| r.kind == b' ').count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let expected = hunk.old_start + net + drift;

        let mut best_rank: Option<(usize, i64)> = None;
        let mut best_starts = Vec::<(usize, usize, usize)>::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > hunk.old_count {
                    continue;
                }
                let mut old_payloads = Vec::new();
                for row in &hunk.rows[l..hunk.rows.len() - r] {
                    if row.kind == b' ' || row.kind == b'-' {
                        old_payloads.push(row.payload.as_str());
                    }
                }

                let count = hunk.old_count;
                if count > working.len() {
                    continue;
                }
                for s in 0..=working.len() - count {
                    let distance = (s as i64 - expected).abs();
                    if distance > options.max_offset as i64 {
                        continue;
                    }
                    let slice = &working[s + l..s + count - r];
                    if slice.len() != old_payloads.len()
                        || !slice
                            .iter()
                            .zip(&old_payloads)
                            .all(|(a, b)| a == b)
                    {
                        continue;
                    }
                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push((s, l, r));
                        }
                        Some(prev) if rank < prev => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push((s, l, r));
                        }
                        Some(prev) if rank == prev => best_starts.push((s, l, r)),
                        _ => {}
                    }
                }
            }
        }

        if best_starts.is_empty() {
            return Err(ApplyError::NoMatch { hunk: index + 1 });
        }

        best_starts.sort_by_key(|&(s, l, _)| (s, l));
        let distinct: Vec<usize> = best_starts
            .iter()
            .map(|&(s, _, _)| s)
            .fold(Vec::new(), |mut acc, s| {
                if acc.last().copied() != Some(s) {
                    acc.push(s);
                }
                acc
            });
        if distinct.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: index + 1,
                first: distinct[0],
                second: distinct[1],
            });
        }

        let (s, l, r) = best_starts[0];
        let mut replacement = Vec::new();
        for row in &hunk.rows[l..hunk.rows.len() - r] {
            if row.kind == b' ' || row.kind == b'+' {
                replacement.push(row.payload.clone());
            }
        }
        let replace_start = s + l;
        let replace_end = s + hunk.old_count - r;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }
        let previous_net = net;
        net += hunk.new_count as i64 - hunk.old_count as i64;
        drift = s as i64 - (hunk.old_start + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}

    Context,
    Remove,
    Add,

#[derive(Clone)]
struct Row {
    kind: RowKind,
    payload: String,
    line: usize,
}

struct Hunk {

    old_start: usize,
    old_count: usize,

        || !s.bytes().all(|b| b.is_ascii_digit())
        || (s.len() > 1 && s.starts_with('0'))

    (n <= 128).then_some(n)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let rest = line.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(" +")?;
    let (new, ending) = rest.split_once(" @@")?;
    if ending != "" {
        return None;
    }
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    if a.contains(',') || b.contains(',') || c.contains(',') || d.contains(',') {
        return None;
    }
    let a = parse_number(a)?;
    let b = parse_number(b)?;
    let c = parse_number(c)?;
    let d = parse_number(d)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    Some((a, b, c, d))
}

fn full_start(start: usize, count: usize) -> i128 {
    if count > 0 {
        start as i128 - 1
    } else {
        start as i128
    }
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    if patch.is_empty() {
        return validate_source_and_serialize(source);
    }

    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }

    let lines: Vec<&str> = patch[..patch.len() - 1].split('\n').collect();
    for (i, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(i) != Some(expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() < 3 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut idx = 2;
    let mut previous_old_end: Option<i128> = None;
    let mut accumulated_delta: i128 = 0;

    while idx < lines.len() {
        let header_line = idx + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[idx]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        let old_nominal = full_start(a, b);
        let new_nominal = full_start(c, d);

        idx += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while idx < lines.len() && !lines[idx].starts_with("@@") {
            let physical_line = idx + 1;
            let row = lines[idx];
            if row.is_empty() {
                return Err(ApplyError::BadBody { line: physical_line });
            }
            let (kind, payload) = match row.as_bytes()[0] {
                b' ' => (RowKind::Context, &row[1..]),
                b'-' => (RowKind::Remove, &row[1..]),
                b'+' => (RowKind::Add, &row[1..]),
                _ => return Err(ApplyError::BadBody { line: physical_line }),
            };
            if payload.as_bytes().contains(&b'\r') {
                return Err(ApplyError::BadBody { line: physical_line });
            }
            match kind {
                RowKind::Context => {
                    old_seen += 1;
                    new_seen += 1;
                }
                RowKind::Remove => old_seen += 1,
                RowKind::Add => new_seen += 1,
            }
            rows.push(Row {
                kind,
                payload: payload.to_string(),
                line: physical_line,
            });
            idx += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.kind != RowKind::Context) {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if let Some(end) = previous_old_end {
            if old_nominal < end || new_nominal != old_nominal + accumulated_delta {
                return Err(ApplyError::Misordered { line: header_line });
            }
        } else if new_nominal != old_nominal {
            return Err(ApplyError::Misordered { line: header_line });
        }

        previous_old_end = Some(old_nominal + b as i128);
        accumulated_delta += d as i128 - b as i128;

        hunks.push(Hunk {
            line: header_line,
            old_start: old_nominal as usize,
            old_count: b,
            new_start: new_nominal as usize,
            new_count: d,
            rows,
        });

        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut working = source_lines(source)?;
    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| r.kind == RowKind::Context).count();
        let trail = hunk.rows.iter().rev().take_while(|r| r.kind == RowKind::Context).count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, i128)> = None;
        let mut best_starts = Vec::new();
        let mut chosen: Option<(usize, usize, usize)> = None;

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > hunk.old_count {
                    continue;
                }
                let wanted: Vec<&str> = hunk
                    .rows
                    .iter()
                    .filter(|row| row.kind != RowKind::Add)
                    .skip(l)
                    .take(hunk.old_count - l - r)
                    .map(|row| row.payload.as_str())
                    .collect();

                for s in 0..=working.len() {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = (s as i128 - expected).abs();
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    if working[s + l..s + hunk.old_count - r]
                        .iter()
                        .map(String::as_str)
                        .ne(wanted.iter().copied())
                    {
                        continue;
                    }

                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(s);
                            chosen = Some((s, l, r));
                        }
                        Some(old_rank) if rank < old_rank => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(s);
                            chosen = Some((s, l, r));
                        }
                        Some(old_rank) if rank == old_rank => {
                            if !best_starts.contains(&s) {
                                best_starts.push(s);
                            }
                            if let Some((old_s, old_l, _)) = chosen {
                                if s == old_s && l < old_l {
                                    chosen = Some((s, l, r));
                                }
                            }
                        }
                        _ => {}
                    }
                }
            }
        }

        let Some((s, l, r)) = chosen else {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        };
        if best_starts.len() > 1 {
            best_starts.sort_unstable();
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: best_starts[0],
                second: best_starts[1],
            });
        }

        let mut replacement = Vec::new();
        for row in hunk.rows.iter().skip(l) {
            if row.kind != RowKind::Remove {
                replacement.push(row.payload.clone());
            }
        }
        let begin = s + l;
        let end = s + hunk.old_count - r;
        working.splice(begin..end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (hunk.old_start as i128 + previous_net);
    }

    Ok(working
        .into_iter()
        .map(|line| format!("{line}\n"))
        .collect())
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }
    let lines: Vec<String> = source[..source.len() - 1]
        .split('\n')
        .map(str::to_string)
        .collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

fn validate_source_and_serialize(source: &str) -> Result<String, ApplyError> {
    let lines = source_lines(source)?;
    Ok(lines.into_iter().map(|line| format!("{line}\n")).collect())
}
