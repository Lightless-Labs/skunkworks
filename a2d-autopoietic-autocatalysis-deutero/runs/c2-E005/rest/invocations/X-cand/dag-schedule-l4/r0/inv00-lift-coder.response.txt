#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

struct Running {
    task: usize,
    worker: usize,
    finish: u64,
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependency entries in task order and list order.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &limit)) in task.needs.iter().zip(capacity).enumerate() {
            if need > limit {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: limit,
                });
            }
        }
    }

    // Build successor lists and indegrees. Each dependency list is unique
    // after validation.
    let mut indegree: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut children = vec![Vec::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal, always removing the smallest currently available ID.
    let mut available: Vec<usize> = (0..n).filter(|&id| indegree[id] == 0).collect();
    let mut kahn_order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest = available
            .iter()
            .enumerate()
            .min_by_key(|&(_, &id)| id)
            .map(|(index, _)| index)
            .unwrap();
        let id = available.swap_remove(smallest);
        kahn_order.push(id);

        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if kahn_order.len() != n {
        let removed: Vec<bool> = {
            let mut flags = vec![false; n];
            for &id in &kahn_order {
                flags[id] = true;
            }
            flags
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &id in kahn_order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_available = vec![0u64; worker_count];
    let mut used: Vec<u128> = vec![0; capacity.len()];
    let mut running: Vec<Running> = Vec::new();
    let mut trace = String::new();
    trace.push_str(&format!(
        "workers={};resources={}\n",
        worker_count,
        capacity.len()
    ));

    let mut time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete every task finishing now, in task ID order.
        let mut ending: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(index, run)| (run.finish == time).then_some(index))
            .collect();
        ending.sort_by_key(|&index| running[index].task);

        for index in ending.into_iter().rev() {
            let run = running.remove(index);
            let task_id = run.task;
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(time);

            for resource in 0..capacity.len() {
                used[resource] -= u128::from(tasks[task_id].needs[resource]);
            }

            trace.push_str(&format!("{time}|end|{task_id}|{}|", run.worker));
            append_used(&mut trace, &used);
            trace.push('\n');
        }

        // Start one task at a time, reevaluating eligibility and ranking.
        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= time
                        && tasks[id].deps.iter().all(|&dep| completed[dep])
                        && tasks[id]
                            .needs
                            .iter()
                            .enumerate()
                            .all(|(r, &need)| {
                                u128::from(need)
                                    <= u128::from(capacity[r]) - used[r]
                            })
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tail[b]
                    .cmp(&tail[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= time)
                .expect("available worker required for admission");

            let finish = time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let turnaround = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = turnaround;
            for resource in 0..capacity.len() {
                used[resource] += u128::from(tasks[task_id].needs[resource]);
            }

            runs[task_id] = Some(Run {
                start: time,
                finish,
                worker,
            });

            trace.push_str(&format!("{time}|start|{task_id}|{worker}|"));
            append_used(&mut trace, &used);
            trace.push('\n');

            if tasks[task_id].duration == 0 {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(time);
                for resource in 0..capacity.len() {
                    used[resource] -= u128::from(tasks[task_id].needs[resource]);
                }
                trace.push_str(&format!("{time}|end|{task_id}|{worker}|"));
                append_used(&mut trace, &used);
                trace.push('\n');
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the next relevant strictly later event.
        let mut next = u64::MAX;
        let mut has_next = false;

        for run in &running {
            if run.finish > time {
                next = next.min(run.finish);
                has_next = true;
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > time {
                next = next.min(tasks[id].release);
                has_next = true;
            }
        }
        for worker in 0..worker_count {
            if worker_available[worker] > time {
                next = next.min(worker_available[worker]);
                has_next = true;
            }
        }

        if has_next {
            time = next;
        } else {
            // With an acyclic graph, any unstarted task must be waiting for a
            // release or a predecessor completion, both of which are events.
            break;
        }
    }

    trace.push_str(&format!("makespan={makespan}\n"));

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

fn append_used(trace: &mut String, used: &[u128]) {
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, amount) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            trace.push_str(&amount.to_string());
        }
    }
}
