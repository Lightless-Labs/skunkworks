#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn encode_digit(value: u8, alphabet: Alphabet) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + (value - 26),
        52..=61 => b'0' + (value - 52),
        62 => match alphabet {
            Alphabet::Standard => b'+',
            Alphabet::UrlSafe => b'-',
        },
        _ => match alphabet {
            Alphabet::Standard => b'/',
            Alphabet::UrlSafe => b'_',
        },
    }
}

fn raw_encode(input: &[u8], options: Options) -> Vec<u8> {
    let mut out = Vec::with_capacity((input.len() + 2) / 3 * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        let a = chunk[0];
        let b = chunk[1];
        let c = chunk[2];
        out.push(encode_digit(a >> 2, options.alphabet));
        out.push(encode_digit(((a & 0x03) << 4) | (b >> 4), options.alphabet));
        out.push(encode_digit(((b & 0x0f) << 2) | (c >> 6), options.alphabet));
        out.push(encode_digit(c & 0x3f, options.alphabet));
    }

    let tail = chunks.remainder();
    if !tail.is_empty() {
        let a = tail[0];
        out.push(encode_digit(a >> 2, options.alphabet));
        if tail.len() == 1 {
            out.push(encode_digit((a & 0x03) << 4, options.alphabet));
            if options.padding != Padding::Forbidden {
                out.extend_from_slice(b"==");
            }
        } else {
            let b = tail[1];
            out.push(encode_digit(((a & 0x03) << 4) | (b >> 4), options.alphabet));
            out.push(encode_digit((b & 0x0f) << 2, options.alphabet));
            if options.padding != Padding::Forbidden {
                out.push(b'=');
            }
        }
    }

    out
}

fn wrap_symbols(raw: &[u8], width: usize, ending: LineEnding) -> Vec<u8> {
    let separator: &[u8] = match ending {
        LineEnding::Lf => b"\n",
        LineEnding::CrLf => b"\r\n",
    };
    let mut out = Vec::with_capacity(raw.len() + raw.len() / width * separator.len());
    for (i, byte) in raw.iter().enumerate() {
        if i != 0 && i % width == 0 {
            out.extend_from_slice(separator);
        }
        out.push(*byte);
    }
    out
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let raw = raw_encode(input, options);
    let output = match options.whitespace {
        Whitespace::Wrapped { width, ending } => wrap_symbols(&raw, width, ending),
        Whitespace::None | Whitespace::IgnoreAscii => raw,
    };
    Ok(String::from_utf8(output).expect("base64 output is ASCII"))
}

pub struct Encoder {
    options: Options,
    buffered: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            buffered: Vec::new(),
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut combined = Vec::with_capacity(self.buffered.len() + input.len());
        combined.extend_from_slice(&self.buffered);
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 3 * 3;
        let encoded = raw_encode(&combined[..complete_len], self.options);
        self.buffered.clear();
        self.buffered.extend_from_slice(&combined[complete_len..]);

        let output = match self.options.whitespace {
            Whitespace::Wrapped { width, ending } => wrap_symbols(
                &encoded,
                width,
                ending,
            ),
            Whitespace::None | Whitespace::IgnoreAscii => encoded,
        };
        Ok(String::from_utf8(output).expect("base64 output is ASCII"))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        let encoded = raw_encode(&self.buffered, self.options);
        self.buffered.clear();
        let output = match self.options.whitespace {
            Whitespace::Wrapped { width, ending } => wrap_symbols(
                &encoded,
                width,
                ending,
            ),
            Whitespace::None | Whitespace::IgnoreAscii => encoded,
        };
        Ok(String::from_utf8(output).expect("base64 output is ASCII"))
    }
}

#[derive(Clone, Copy)]
enum WrapState {
    Row,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    offset: usize,
    group: [u8; 4],
    group_offsets: [usize; 4],
    group_len: usize,
    pending_second_padding: Option<usize>,
    terminal_padding: bool,
    raw_symbols: usize,
    wrap_state: WrapState,
    sticky_error: Option<Error>,
    finished: bool,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            offset: 0,
            group: [0; 4],
            group_offsets: [0; 4],
            group_len: 0,
            pending_second_padding: None,
            terminal_padding: false,
            raw_symbols: 0,
            wrap_state: WrapState::Row,
            sticky_error: None,
            finished: false,
        })
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    fn emit_group(&mut self, out: &mut Vec<u8>) {
        let a = self.group[0];
        let b = self.group[1];
        let c = self.group[2];
        let d = self.group[3];

        out.push((a << 2) | (b >> 4));
        if c != 64 {
            out.push((b << 4) | (c >> 2));
        }
        if d != 64 {
            out.push((c << 6) | d);
        }

        if c == 64 || d == 64 {
            self.terminal_padding = true;
        }
        self.group_len = 0;
        self.pending_second_padding = None;
    }

    fn process_symbol(&mut self, byte: u8, physical_offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset: physical_offset });
        }

        if let Some(first_padding_offset) = self.pending_second_padding {
            if byte != b'=' {
                return Err(Error::BadPadding { offset: physical_offset });
            }
            self.group[3] = 64;
            self.group_offsets[3] = physical_offset;
            self.group_len = 4;
            self.emit_group(out);
            let _ = first_padding_offset;
            return Ok(());
        }

        match self.group_len {
            0 | 1 => {
                if byte == b'=' {
                    return Err(Error::BadPadding { offset: physical_offset });
                }
                let digit = alphabet_digit(byte, self.options.alphabet)
                    .ok_or(Error::InvalidByte { offset: physical_offset, byte })?;
                self.group[self.group_len] = digit;
                self.group_offsets[self.group_len] = physical_offset;
                self.group_len += 1;
            }
            2 => {
                self.group_offsets[2] = physical_offset;
                if byte == b'=' {
                    self.pending_second_padding = Some(physical_offset);
                    self.group_len = 3;
                } else {
                    let digit = alphabet_digit(byte, self.options.alphabet)
                        .ok_or(Error::InvalidByte { offset: physical_offset, byte })?;
                    self.group[2] = digit;
                    self.group_len = 3;
                }
            }
            3 => {
                self.group_offsets[3] = physical_offset;
                if byte == b'=' {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset: physical_offset });
                    }
                    self.group[3] = 64;
                    self.group_len = 4;
                    self.emit_group(out);
                } else {
                    let digit = alphabet_digit(byte, self.options.alphabet)
                        .ok_or(Error::InvalidByte { offset: physical_offset, byte })?;
                    self.group[3] = digit;
                    self.group_len = 4;
                    self.emit_group(out);
                }
            }
            _ => unreachable!(),
        }

        Ok(())
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &byte in input {
            let current_offset = self.offset;
            self.offset = self.offset.saturating_add(1);

            match self.options.whitespace {
                Whitespace::None => {
                    self.process_symbol(byte, current_offset, &mut out)
                        .map_err(|error| self.fail(error))?;
                }
                Whitespace::IgnoreAscii => {
                    if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                        continue;
                    }
                    self.process_symbol(byte, current_offset, &mut out)
                        .map_err(|error| self.fail(error))?;
                }
                Whitespace::Wrapped { width, ending } => {
                    match self.wrap_state {
                        WrapState::NeedLf { cr_offset } => {
                            if byte != b'\n' {
                                return Err(self.fail(Error::BadWrap { offset: current_offset }));
                            }
                            self.wrap_state = WrapState::NeedSymbol {
                                separator_offset: cr_offset,
                            };
                            continue;
                        }
                        WrapState::NeedSymbol { .. } => {
                            if matches!(byte, b'\n' | b'\r' | 9 | 10 | 11 | 12 | 32) {
                                return Err(self.fail(Error::BadWrap { offset: current_offset }));
                            }
                            self.wrap_state = WrapState::Row;
                        }
                        WrapState::Row => {}
                    }

                    if self.raw_symbols != 0 && self.raw_symbols % width == 0 {
                        match ending {
                            LineEnding::Lf if byte == b'\n' => {
                                self.wrap_state = WrapState::NeedSymbol {
                                    separator_offset: current_offset,
                                };
                                continue;
                            }
                            LineEnding::CrLf if byte == b'\r' => {
                                self.wrap_state = WrapState::NeedLf {
                                    cr_offset: current_offset,
                                };
                                continue;
                            }
                            _ => {
                                return Err(self.fail(Error::BadWrap { offset: current_offset }));
                            }
                        }
                    }

                    if matches!(byte, b'\n' | b'\r' | 9 | 10 | 11 | 12 | 32) {
                        return Err(self.fail(Error::BadWrap { offset: current_offset }));
                    }

                    self.raw_symbols = self.raw_symbols.saturating_add(1);
                    self.process_symbol(byte, current_offset, &mut out)
                        .map_err(|error| self.fail(error))?;
                }
            }
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Whitespace::Wrapped { ending, .. } = self.options.whitespace {
            match self.wrap_state {
                WrapState::NeedLf { cr_offset } => {
                    return Err(self.fail(Error::BadWrap { offset: cr_offset }));
                }
                WrapState::NeedSymbol { separator_offset } => {
                    return Err(self.fail(Error::BadWrap { offset: separator_offset }));
                }
                WrapState::Row => {
                    let _ = ending;
                }
            }
        }

        if let Some(offset) = self.pending_second_padding {
            return Err(self.fail(Error::BadPadding { offset }));
        }

        let mut out = Vec::new();
        if self.group_len == 0 {
            self.finished = true;
            return Ok(out);
        }

        if self.options.padding == Padding::Required {
            return Err(self.fail(Error::Truncated {
                offset: self.group_offsets[0],
            }));
        }

        match self.group_len {
            1 => {
                return Err(self.fail(Error::Truncated {
                    offset: self.group_offsets[0],
                }));
            }
            2 => {
                out.push((self.group[0] << 2) | (self.group[1] >> 4));
            }
            3 => {
                out.push((self.group[0] << 2) | (self.group[1] >> 4));
                out.push((self.group[1] << 4) | (self.group[2] >> 2));
            }
            _ => unreachable!(),
        }

        self.group_len = 0;
        self.finished = true;
        Ok(out)
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}
