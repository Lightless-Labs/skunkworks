fn validate_options(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())

fn alphabet_digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn encoded_digit(value: u8, alphabet: Alphabet) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + value - 26,
        52..=61 => b'0' + value - 52,
        62 => match alphabet {
            Alphabet::Standard => b'+',
            Alphabet::UrlSafe => b'-',
        },
        _ => match alphabet {
            Alphabet::Standard => b'/',
            Alphabet::UrlSafe => b'_',
        },
    }
}

fn encode_raw(input: &[u8], options: Options, allow_padding: bool) -> String {
    let mut output = Vec::with_capacity((input.len() + 2) / 3 * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        let a = chunk[0];
        let b = chunk[1];
        let c = chunk[2];
        output.push(encoded_digit(a >> 2, options.alphabet));
        output.push(encoded_digit(((a & 0x03) << 4) | (b >> 4), options.alphabet));
        output.push(encoded_digit(((b & 0x0f) << 2) | (c >> 6), options.alphabet));
        output.push(encoded_digit(c & 0x3f, options.alphabet));
    }

    let tail = chunks.remainder();
    if !tail.is_empty() {
        let a = tail[0];
        output.push(encoded_digit(a >> 2, options.alphabet));
        if tail.len() == 1 {
            output.push(encoded_digit((a & 0x03) << 4, options.alphabet));
            if allow_padding {
                output.extend_from_slice(b"==");
            }
        } else {
            let b = tail[1];
            output.push(encoded_digit(((a & 0x03) << 4) | (b >> 4), options.alphabet));
            output.push(encoded_digit((b & 0x0f) << 2, options.alphabet));
            if allow_padding {
                output.push(b'=');
            }
        }
    }

    String::from_utf8(output).expect("base64 alphabet is ASCII")
}

fn wrap_encoded(raw: String, whitespace: Whitespace) -> String {
    let Whitespace::Wrapped { width, ending } = whitespace else {
        return raw;
    };
    if raw.len() <= width {
        return raw;
    }

    let separator = match ending {
        LineEnding::Lf => "\n",
        LineEnding::CrLf => "\r\n",
    };

    let mut result = String::with_capacity(raw.len() + raw.len() / width * separator.len());
    for (index, chunk) in raw.as_bytes().chunks(width).enumerate() {
        if index != 0 {
            result.push_str(separator);
        }
        result.push_str(std::str::from_utf8(chunk).expect("ASCII output"));
    }
    result
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate_options(options)?;
    let allow_padding = options.padding != Padding::Forbidden;
    let raw = encode_raw(input, options, allow_padding);
    Ok(wrap_encoded(raw, options.whitespace))
}

pub struct Encoder {
    options: Options,
    buffered: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            buffered: Vec::with_capacity(2),
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        if input.is_empty() {
            return Ok(String::new());
        }

        let mut combined = Vec::with_capacity(self.buffered.len() + input.len());
        combined.extend_from_slice(&self.buffered);
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 3 * 3;
        self.buffered.clear();
        self.buffered.extend_from_slice(&combined[complete_len..]);

        let raw = encode_raw(&combined[..complete_len], self.options, true);
        Ok(wrap_encoded(raw, self.options.whitespace))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        let raw = encode_raw(&self.buffered, self.options, true);
        self.buffered.clear();
        Ok(wrap_encoded(raw, self.options.whitespace))
    }
}

pub struct Decoder {
    options: Options,
    offset: usize,
    symbols: Vec<Symbol>,
    pending_second_padding: Option<usize>,
    terminal_padding: bool,
    wrapped_symbols: usize,
    wrap_state: WrapState,
    finished: bool,
    sticky_error: Option<Error>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum WrapState {
    Row,
    NeedSeparator,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            offset: 0,
            symbols: Vec::with_capacity(4),
            pending_second_padding: None,
            terminal_padding: false,
            wrapped_symbols: 0,
            wrap_state: WrapState::Row,
            finished: false,
            sticky_error: None,
        })
    }

    fn fail(&mut self, error: Error) -> Result<Vec<u8>, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }

    fn scan_symbol(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        let position = self.symbols.len();

        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if let Some(pending_offset) = self.pending_second_padding {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            self.pending_second_padding = None;
            self.symbols.push(Symbol { byte, offset });
            self.emit_group(output);
            self.terminal_padding = true;
            return Ok(());
        }

        if byte == b'=' {
            match position {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset });
                    }
                    self.pending_second_padding = Some(offset);
                    self.symbols.push(Symbol { byte, offset });
                    return Ok(());
                }
                3 => {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset });
                    }
                    if self.symbols[2].byte == b'=' {
                        return Err(Error::BadPadding { offset });
                    }
                    self.symbols.push(Symbol { byte, offset });
                    self.emit_group(output);
                    self.terminal_padding = true;
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        if alphabet_digit(byte, self.options.alphabet).is_none() {
            return Err(Error::InvalidByte { offset, byte });
        }

        self.symbols.push(Symbol { byte, offset });
        if self.symbols.len() == 4 {
            self.emit_group(output);
        }
        Ok(())
    }

    fn emit_group(&mut self, output: &mut Vec<u8>) {
        let digits: Vec<Option<u8>> = self
            .symbols
            .iter()
            .map(|s| {
                if s.byte == b'=' {
                    None
                } else {
                    alphabet_digit(s.byte, self.options.alphabet)
                }
            })
            .collect();

        let a = digits[0].unwrap();
        let b = digits[1].unwrap();
        output.push((a << 2) | (b >> 4));

        if digits[2].is_some() {
            let c = digits[2].unwrap();
            output.push((b << 4) | (c >> 2));
            if digits[3].is_some() {
                output.push((c << 6) | digits[3].unwrap());
            }
        }

        self.symbols.clear();
    }

    fn frame_byte(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => self.scan_symbol(byte, offset, output),
            Whitespace::IgnoreAscii => {
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    Ok(())
                } else {
                    self.scan_symbol(byte, offset, output)
                }
            }
            Whitespace::Wrapped { width, ending } => {
                match self.wrap_state {
                    WrapState::Row => {
                        if byte == b'\n' || byte == b'\r' || matches!(byte, 9 | 11 | 12 | 32) {
                            return Err(Error::BadWrap { offset });
                        }
                        self.scan_symbol(byte, offset, output)?;
                        self.wrapped_symbols += 1;
                        if self.wrapped_symbols == width {
                            self.wrap_state = WrapState::NeedSeparator;
                        }
                        Ok(())
                    }
                    WrapState::NeedSeparator => match ending {
                        LineEnding::Lf if byte == b'\n' => {
                            self.wrapped_symbols = 0;
                            self.wrap_state = WrapState::NeedSymbol {
                                separator_offset: offset,
                            };
                            Ok(())
                        }
                        LineEnding::CrLf if byte == b'\r' => {
                            self.wrap_state = WrapState::NeedLf { cr_offset: offset };
                            Ok(())
                        }
                        _ => Err(Error::BadWrap { offset }),
                    },
                    WrapState::NeedLf { .. } if byte == b'\n' => {
                        self.wrapped_symbols = 0;
                        let cr_offset = match self.wrap_state {
                            WrapState::NeedLf { cr_offset } => cr_offset,
                            _ => unreachable!(),
                        };
                        self.wrap_state = WrapState::NeedSymbol {
                            separator_offset: cr_offset,
                        };
                        Ok(())
                    }
                    WrapState::NeedLf { .. } => Err(Error::BadWrap { offset }),
                    WrapState::NeedSymbol { .. } => {
                        if byte == b'\n' || byte == b'\r' || matches!(byte, 9 | 11 | 12 | 32) {
                            return Err(Error::BadWrap { offset });
                        }
                        self.wrap_state = WrapState::Row;
                        self.scan_symbol(byte, offset, output)?;
                        self.wrapped_symbols = 1;
                        if self.wrapped_symbols == width {
                            self.wrap_state = WrapState::NeedSeparator;
                        }
                        Ok(())
                    }
                }
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;
            if let Err(error) = self.frame_byte(byte, offset, &mut output) {
                return self.fail(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Whitespace::Wrapped { ending, .. } = self.options.whitespace {
            match self.wrap_state {
                WrapState::NeedLf { cr_offset } => {
                    return self.fail(Error::BadWrap { offset: cr_offset });
                }
                WrapState::NeedSymbol { separator_offset } => {
                    return self.fail(Error::BadWrap {
                        offset: separator_offset,
                    });
                }
                _ => {}
            }
            let _ = ending;
        }

        if let Some(offset) = self.pending_second_padding {
            return self.fail(Error::BadPadding { offset });
        }

        let mut output = Vec::new();
        let residual = self.symbols.len();
        if residual != 0 {
            if self.options.padding == Padding::Required {
                return self.fail(Error::Truncated {
                    offset: self.symbols[0].offset,
                });
            }
            match residual {
                1 => {
                    return self.fail(Error::Truncated {
                        offset: self.symbols[0].offset,
                    });
                }
                2 => {
                    let a = alphabet_digit(self.symbols[0].byte, self.options.alphabet).unwrap();
                    let b = alphabet_digit(self.symbols[1].byte, self.options.alphabet).unwrap();
                    output.push((a << 2) | (b >> 4));
                }
                3 => {
                    let a = alphabet_digit(self.symbols[0].byte, self.options.alphabet).unwrap();
                    let b = alphabet_digit(self.symbols[1].byte, self.options.alphabet).unwrap();
                    let c = alphabet_digit(self.symbols[2].byte, self.options.alphabet).unwrap();
                    output.push((a << 2) | (b >> 4));
                    output.push((b << 4) | (c >> 2));
                }
                _ => unreachable!(),
            }
            self.symbols.clear();
        }

        self.finished = true;
        Ok(output)
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {

fn digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn encode_digit(value: u8, alphabet: Alphabet) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + value - 26,
        52..=61 => b'0' + value - 52,
        62 => match alphabet {
            Alphabet::Standard => b'+',
            Alphabet::UrlSafe => b'-',
        },
        _ => match alphabet {
            Alphabet::Standard => b'/',
            Alphabet::UrlSafe => b'_',
        },
    }
}

fn encode_group(bytes: &[u8], alphabet: Alphabet, padding: Padding) -> Vec<u8> {
    let mut out = Vec::with_capacity(4);
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };
    out.push(encode_digit(a >> 2, alphabet));
    out.push(encode_digit(((a & 0x03) << 4) | (b >> 4), alphabet));
    if bytes.len() > 1 {
        out.push(encode_digit(((b & 0x0f) << 2) | (c >> 6), alphabet));
    } else if padding != Padding::Forbidden {
        out.push(b'=');
    }
    if bytes.len() > 2 {
        out.push(encode_digit(c & 0x3f, alphabet));
    } else if padding != Padding::Forbidden {
        out.push(b'=');
    }
    out
}

fn apply_wrapping(raw: Vec<u8>, whitespace: Whitespace) -> String {
    let Whitespace::Wrapped { width, ending } = whitespace else {
        return String::from_utf8(raw).expect("base64 output is ASCII");
    };
    let sep: &[u8] = match ending {
        LineEnding::Lf => b"\n",
        LineEnding::CrLf => b"\r\n",
    };
    let mut out = Vec::with_capacity(raw.len() + raw.len() / width * sep.len());
    for (i, chunk) in raw.chunks(width).enumerate() {
        if i != 0 {
            out.extend_from_slice(sep);
        }
        out.extend_from_slice(chunk);
    }
    String::from_utf8(out).expect("base64 output is ASCII")
}

pub struct Encoder {
    options: Options,
    buffered: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            buffered: Vec::new(),
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.buffered.extend_from_slice(input);
        let complete = self.buffered.len() / 3 * 3;
        if complete == 0 {
            return Ok(String::new());
        }
        let mut raw = Vec::with_capacity(complete / 3 * 4);
        for chunk in self.buffered[..complete].chunks_exact(3) {
            raw.extend_from_slice(&encode_group(
                chunk,
                self.options.alphabet,
                self.options.padding,
            ));
        }
        self.buffered.drain(..complete);
        Ok(apply_wrapping(raw, self.options.whitespace))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        let raw = if self.buffered.is_empty() {
            Vec::new()
        } else {
            encode_group(
                &self.buffered,
                self.options.alphabet,
                self.options.padding,
            )
        };
        self.buffered.clear();
        Ok(apply_wrapping(raw, self.options.whitespace))
    }
}

#[derive(Clone, Copy)]
enum Symbol {
    Digit(u8),
    Pad,
}

struct WrappedState {
    width: usize,
    ending: LineEnding,
    row_count: usize,
    awaiting_separator: bool,
    separator_start: Option<usize>,
    awaiting_lf: Option<usize>,
    awaiting_symbol: Option<usize>,
}

pub struct Decoder {
    options: Options,
    group: Vec<(Symbol, usize)>,
    offset: usize,
    terminal_padding: bool,
    wrapped: Option<WrappedState>,
    finished: bool,
    sticky: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        let wrapped = match options.whitespace {
            Whitespace::Wrapped { width, ending } => Some(WrappedState {
                width,
                ending,
                row_count: 0,
                awaiting_separator: false,
                separator_start: None,
                awaiting_lf: None,
                awaiting_symbol: None,
            }),
            _ => None,
        };
        Ok(Self {
            options,
            group: Vec::with_capacity(4),
            offset: 0,
            terminal_padding: false,
            wrapped,
            finished: false,
            sticky: None,
        })
    }

    fn remember_error(&mut self, error: Error) -> Error {
        self.sticky = Some(error.clone());
        error
    }

    fn scan_symbol(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }
        let pos = self.group.len();
        if pos == 0 || pos == 1 {
            if byte == b'=' {
                return Err(Error::BadPadding { offset });
            }
        } else if pos == 2 && matches!(self.group[1].0, Symbol::Pad) {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
        } else if pos == 2 && byte == b'=' {
            self.group.push((Symbol::Pad, offset));
            return Ok(());
        } else if pos == 3 && matches!(self.group[2].0, Symbol::Pad) && byte != b'=' {
            return Err(Error::BadPadding { offset });
        } else if pos == 3 && byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            self.group.push((Symbol::Pad, offset));
            self.emit_group(out)?;
            self.terminal_padding = true;
            return Ok(());
        }

        let value = digit(byte, self.options.alphabet)
            .ok_or(Error::InvalidByte { offset, byte })?;
        self.group.push((Symbol::Digit(value), offset));

        if self.group.len() == 4 {
            if self.group.iter().any(|(s, _)| matches!(s, Symbol::Pad)) {
                return Err(Error::BadPadding { offset });
            }
            self.emit_group(out)?;
        }
        Ok(())
    }

    fn emit_group(&mut self, out: &mut Vec<u8>) -> Result<(), Error> {
        let vals: Vec<u8> = self
            .group
            .iter()
            .filter_map(|(s, _)| match s {
                Symbol::Digit(v) => Some(*v),
                Symbol::Pad => None,
            })
            .collect();
        match self.group.len() {
            4 if vals.len() == 4 => {
                out.push((vals[0] << 2) | (vals[1] >> 4));
                out.push((vals[1] << 4) | (vals[2] >> 2));
                out.push((vals[2] << 6) | vals[3]);
                self.group.clear();
            }
            4 if vals.len() == 2 => {
                out.push((vals[0] << 2) | (vals[1] >> 4));
                self.group.clear();
                self.terminal_padding = true;
            }
            4 if vals.len() == 3 => {
                out.push((vals[0] << 2) | (vals[1] >> 4));
                out.push((vals[1] << 4) | (vals[2] >> 2));
                self.group.clear();
                self.terminal_padding = true;
            }
            _ => return Err(Error::BadPadding {
                offset: self.group.last().map_or(self.offset, |(_, o)| *o),
            }),
        }
        Ok(())
    }

    fn frame_byte(&mut self, byte: u8, offset: usize) -> Result<bool, Error> {
        if let Some(w) = self.wrapped.as_mut() {
            if let Some(cr_offset) = w.awaiting_lf {
                if byte != b'\n' {
                    return Err(Error::BadWrap { offset });
                }
                w.awaiting_lf = None;
                w.awaiting_symbol = Some(w.separator_start.unwrap_or(cr_offset));
                return Ok(false);
            }
            if let Some(start) = w.awaiting_symbol {
                w.awaiting_symbol = None;
                w.row_count = 0;
                w.awaiting_separator = false;
                w.separator_start = None;
                if byte == b'\r' || byte == b'\n' || byte == b' ' || byte == b'\t' || byte == 0x0b || byte == 0x0c {
                    return Err(Error::BadWrap { offset });
                }
                return Ok(true);
            }
            if w.awaiting_separator {
                match w.ending {
                    LineEnding::Lf => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        w.awaiting_symbol = Some(offset);
                        w.separator_start = Some(offset);
                        return Ok(false);
                    }
                    LineEnding::CrLf => {
                        if byte != b'\r' {
                            return Err(Error::BadWrap { offset });
                        }
                        w.separator_start = Some(offset);
                        w.awaiting_lf = Some(offset);
                        return Ok(false);
                    }
                }
            }
            if byte == b'\r' || byte == b'\n' || byte == b' ' || byte == b'\t' || byte == 0x0b || byte == 0x0c {
                return Err(Error::BadWrap { offset });
            }
            if w.row_count >= w.width {
                return Err(Error::BadWrap { offset });
            }
            w.row_count += 1;
            if w.row_count == w.width {
                w.awaiting_separator = true;
            }
            Ok(true)
        } else if self.options.whitespace == Whitespace::IgnoreAscii
            && matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
        {
            Ok(false)
        } else {
            Ok(true)
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            let error = Error::AlreadyFinished;
            self.sticky = Some(error.clone());
            return Err(error);
        }
        let mut out = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset = self.offset.checked_add(1).unwrap_or(usize::MAX);
            let framed = match self.frame_byte(byte, offset) {
                Ok(v) => v,
                Err(e) => return Err(self.remember_error(e)),
            };
            if !framed {
                continue;
            }
            if let Err(e) = self.scan_symbol(byte, offset, &mut out) {
                return Err(self.remember_error(e));
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            let error = Error::AlreadyFinished;
            self.sticky = Some(error.clone());
            return Err(error);
        }
        if let Some(w) = &self.wrapped {
            if let Some(cr) = w.awaiting_lf {
                let e = Error::BadWrap { offset: cr };
                return Err(self.remember_error(e));
            }
            if let Some(start) = w.awaiting_symbol {
                let e = Error::BadWrap { offset: start };
                return Err(self.remember_error(e));
            }
        }
        if self.group.len() == 4 {
            unreachable!();
        }
        if self.group.len() == 3 && matches!(self.group[2].0, Symbol::Pad) {
            let e = Error::BadPadding {
                offset: self.group[2].1,
            };
            return Err(self.remember_error(e));
        }
        if !self.group.is_empty() {
            if self.options.padding == Padding::Required {
                let e = Error::Truncated {
                    offset: self.group[0].1,
                };
                return Err(self.remember_error(e));
            }
            match self.group.len() {
                1 => {
                    let e = Error::Truncated {
                        offset: self.group[0].1,
                    };
                    return Err(self.remember_error(e));
                }
                2 => {
                    let (Symbol::Digit(a), _) = self.group[0] else {
                        unreachable!()
                    };
                    let (Symbol::Digit(b), _) = self.group[1] else {
                        unreachable!()
                    };
                    self.finished = true;
                    return Ok(vec![(a << 2) | (b >> 4)]);
                }
                3 => {
                    let (Symbol::Digit(a), _) = self.group[0] else {
                        unreachable!()
                    };
                    let (Symbol::Digit(b), _) = self.group[1] else {
                        unreachable!()
                    };
                    let (Symbol::Digit(c), _) = self.group[2] else {
                        unreachable!()
                    };
                    self.finished = true;
                    return Ok(vec![(a << 2) | (b >> 4), (b << 4) | (c >> 2)]);
                }
                _ => unreachable!(),
            }
        }
        self.finished = true;
        Ok(Vec::new())
    }
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut raw = String::new();
    raw.push_str(&encoder.push(input)?);
    raw.push_str(&encoder.finish()?);
    if matches!(options.whitespace, Whitespace::Wrapped { .. }) {
        let mut symbols = Vec::new();
        let mut buffered = Vec::new();
        for chunk in input.chunks(3) {
            buffered.extend_from_slice(&encode_group(chunk, options.alphabet, options.padding));
        }
        symbols = buffered;
        Ok(apply_wrapping(symbols, options.whitespace))
    } else {
        Ok(raw)
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}
