```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

enum State {
    Queued,
    Cancelled,
    Completed,
}

struct Entry {
    job: Job,
    sequence: u64,
    state: State,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u64,
    entries: HashMap<String, Entry>,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            entries: HashMap::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        self.check_time(now)?;

        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.entries.contains_key(&job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        self.entries.insert(
            job.id.clone(),
            Entry {
                job,
                sequence,
                state: State::Queued,
            },
        );
        self.next_sequence += 1;
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        let entry = self.entries.get(id).ok_or(Error::Unknown)?;
        match entry.state {
            State::Queued => {}
            State::Cancelled => return Err(Error::Cancelled),
            State::Completed => return Err(Error::Completed),
        }

        self.entries.get_mut(id).unwrap().state = State::Cancelled;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        self.check_time(now)?;

        let selected = self
            .entries
            .iter()
            .filter_map(|(id, entry)| {
                if !matches!(entry.state, State::Queued) || entry.job.ready_at > now {
                    return None;
                }

                let effective_priority = if self.aging_ms == 0 {
                    entry.job.priority
                } else {
                    let gained = (now - entry.job.ready_at) / self.aging_ms;
                    (u64::from(entry.job.priority) + gained).min(9) as u8
                };

                Some((
                    id.clone(),
                    effective_priority,
                    entry.sequence,
                ))
            })
            .max_by(|a, b| {
                a.1.cmp(&b.1).then_with(|| b.2.cmp(&a.2))
            });

        self.last_time = now;
        match selected {
            Some((id, effective_priority, _)) => {
                self.entries.get_mut(&id).unwrap().state = State::Completed;
                Ok(Some(Dispatch {
                    id,
                    effective_priority,
                }))
            }
            None => Ok(None),
        }
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes
            .iter()
            .all(|&b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
}
```
