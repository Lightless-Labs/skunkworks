```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    jobs: Vec<Entry>,
    reserved: Vec<String>,
    next_sequence: u128,
}

struct Entry {
    id: String,
    ready_at: u64,
    priority: u8,
    sequence: u128,
    status: Status,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Status {
    Queued,
    Cancelled,
    Completed,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            jobs: Vec::new(),
            reserved: Vec::new(),
            next_sequence: 0,
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        self.check_time(now)?;

        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved.iter().any(|id| id == &job.id) {
            return Err(Error::Duplicate);
        }

        self.reserved.push(job.id.clone());
        self.jobs.push(Entry {
            id: job.id,
            ready_at: job.ready_at,
            priority: job.priority,
            sequence: self.next_sequence,
            status: Status::Queued,
        });
        self.next_sequence = self.next_sequence.saturating_add(1);
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        let index = self.jobs.iter().position(|job| job.id == id);
        let Some(index) = index else {
            return Err(Error::Unknown);
        };

        match self.jobs[index].status {
            Status::Queued => self.jobs[index].status = Status::Cancelled,
            Status::Cancelled => return Err(Error::Cancelled),
            Status::Completed => return Err(Error::Completed),
        }

        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        self.check_time(now)?;

        let mut selected: Option<(usize, u8, u128)> = None;

        for (index, job) in self.jobs.iter().enumerate() {
            if job.status != Status::Queued || job.ready_at > now {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                job.priority
            } else {
                let age = (now - job.ready_at) / self.aging_ms;
                let boosted = u64::from(job.priority).saturating_add(age).min(9);
                boosted as u8
            };

            let replace = match selected {
                None => true,
                Some((_, best_priority, best_sequence)) => {
                    effective > best_priority
                        || (effective == best_priority && job.sequence < best_sequence)
                }
            };

            if replace {
                selected = Some((index, effective, job.sequence));
            }
        }

        self.last_time = now;

        let Some((index, effective_priority, _)) = selected else {
            return Ok(None);
        };

        let job = &mut self.jobs[index];
        job.status = Status::Completed;
        Ok(Some(Dispatch {
            id: job.id.clone(),
            effective_priority,
        }))
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes
            .iter()
            .all(|&b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
}
```
