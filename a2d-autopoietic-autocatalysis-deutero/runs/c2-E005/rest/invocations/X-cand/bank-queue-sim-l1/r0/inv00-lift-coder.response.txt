#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut ids = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut queue: VecDeque<&Customer> = VecDeque::new();
    let mut next_arrival = 0usize;
    let mut current_time = 0u64;
    let mut finish_time: Option<u64> = None;
    let mut previous_finish: Option<u64> = None;

    while next_arrival < arrivals.len() || finish_time.is_some() || !queue.is_empty() {
        let next_arrival_time = arrivals
            .get(next_arrival)
            .map(|customer| customer.arrival);

        let tick = match (finish_time, next_arrival_time) {
            (Some(finish), Some(arrival)) => finish.min(arrival),
            (Some(finish), None) => finish,
            (None, Some(arrival)) => arrival,
            (None, None) => break,
        };

        current_time = tick;

        if finish_time == Some(tick) {
            let finished_id = match events.last().map(|event: &Event| &event.kind) {
                Some(Kind::Start { .. }) => events.last().unwrap().customer,
                _ => {
                    // The occupied customer's ID is retained in this local slot
                    // by finding its matching scheduled finish below.
                    let mut id = 0;
                    for event in events.iter().rev() {
                        if let Kind::Start { finish, .. } = event.kind {
                            if finish == tick {
                                id = event.customer;
                                break;
                            }
                        }
                    }
                    id
                }
            };
            events.push(Event {
                time: tick,
                customer: finished_id,
                kind: Kind::Finish,
            });
            finish_time = None;
            previous_finish = Some(tick);
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == tick {
            let customer = arrivals[next_arrival];
            events.push(Event {
                time: tick,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            queue.push_back(customer);
            next_arrival += 1;
        }

        if finish_time.is_none() {
            if let Some(customer) = queue.pop_front() {
                let wake = previous_finish.map_or(true, |finish| tick > finish);
                let base_finish = tick.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: tick,
                    },
                )?;
                let final_finish = base_finish
                    .checked_add(if wake { wake_cost } else { 0 })
                    .ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time: tick,
                    })?;

                events.push(Event {
                    time: tick,
                    customer: customer.id,
                    kind: Kind::Start {
                        finish: final_finish,
                        wake,
                    },
                });

                if final_finish == tick {
                    events.push(Event {
                        time: tick,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    previous_finish = Some(tick);
                } else {
                    finish_time = Some(final_finish);
                }
            }
        }
    }

    Ok(events)
}
