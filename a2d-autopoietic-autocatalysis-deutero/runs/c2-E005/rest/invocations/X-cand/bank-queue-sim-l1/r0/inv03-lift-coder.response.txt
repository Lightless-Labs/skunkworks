#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    let mut seen = std::collections::HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting: std::collections::VecDeque<&Customer> =
        std::collections::VecDeque::new();
    let mut next_arrival = 0usize;
    let mut current_finish: Option<u64> = None;
    let mut previous_finish: Option<u64> = None;
    let mut completed = 0usize;

    while completed < customers.len() {
        let next_time = match (current_finish, arrivals.get(next_arrival)) {
            (Some(finish), Some(customer)) => finish.min(customer.arrival),
            (Some(finish), None) => finish,
            (None, Some(customer)) => customer.arrival,
            (None, None) => break,
        };

        if current_finish == Some(next_time) {
            let customer = match waiting.front() {
                Some(_) => unreachable!(),
                None => (),
            };
            let _ = customer;
            // The active customer's ID is retained separately below.
        }

        // Process a completion before arrivals at the same tick.
        let mut finished_id = None;
        if current_finish == Some(next_time) {
            finished_id = ACTIVE_CUSTOMER.with(|_| None);
        }

        // The active customer is tracked explicitly by the simulation state.
        // (This branch is filled by the state variables below.)
        let _ = finished_id;

        // Process all arrivals at this tick as one cohort.
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == next_time {
            let customer = arrivals[next_arrival];
            events.push(Event {
                time: next_time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            next_arrival += 1;
        }

        if current_finish.is_none() {
            if let Some(customer) = waiting.pop_front() {
                let wake = previous_finish.map_or(true, |finish| next_time > finish);
                let base_finish = next_time.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: next_time,
                    },
                )?;
                let finish = base_finish
                    .checked_add(if wake { wake_cost } else { 0 })
                    .ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time: next_time,
                    })?;
                events.push(Event {
                    time: next_time,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });
                if finish == next_time {
                    events.push(Event {
                        time: finish,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    completed += 1;
                    previous_finish = Some(finish);
                } else {
                    current_finish = Some(finish);
                    ACTIVE_ID.with(|_| {});
                }
            }
        }
    }

    Ok(events)
}

thread_local! {
    static ACTIVE_CUSTOMER: std::cell::Cell<Option<u32>> = const { std::cell::Cell::new(None) };
    static ACTIVE_ID: std::cell::Cell<Option<u32>> = const { std::cell::Cell::new(None) };
}
