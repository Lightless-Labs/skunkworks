```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut queue: VecDeque<&Customer> = VecDeque::new();
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut busy: Option<(u64, u32)> = None;
    let mut previous_finish: Option<u64> = None;

    loop {
        let next_time = match (busy, arrivals.get(next_arrival)) {
            (Some((finish, _)), Some(customer)) => finish.min(customer.arrival),
            (Some((finish, _)), None) => finish,
            (None, Some(customer)) => customer.arrival,
            (None, None) => break,
        };
        now = next_time;

        if let Some((finish, id)) = busy {
            if finish == now {
                events.push(Event {
                    time: now,
                    customer: id,
                    kind: Kind::Finish,
                });
                busy = None;
                previous_finish = Some(now);
            }
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival];
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            queue.push_back(customer);
            next_arrival += 1;
        }

        if busy.is_none() {
            if let Some(customer) = queue.pop_front() {
                let wake = previous_finish.map_or(true, |finish| now > finish);
                let service_finish = now.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: now,
                    },
                )?;
                let finish = service_finish
                    .checked_add(if wake { wake_cost } else { 0 })
                    .ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time: now,
                    })?;

                events.push(Event {
                    time: now,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });

                if finish == now {
                    events.push(Event {
                        time: now,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    previous_finish = Some(now);
                } else {
                    busy = Some((finish, customer.id));
                }
            }
        }
    }

    Ok(events)
}
```
