#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let normalized: String = input
        .chars()
        .map(|ch| if ch == '\r' { '\n' } else { ch })
        .collect();
    let chars: Vec<char> = normalized.chars().collect();

    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut i = 0;

    while i < chars.len() {
        if chars[i] == '\n' {
            let run_start = i;
            while i < chars.len() && chars[i] == '\n' {
                i += 1;
            }
            if i - run_start >= 2 {
                wrap_segment(&chars[segment_start..run_start], options, &mut output);
                for _ in 1..(i - run_start) {
                    output.push(String::new());
                }
                segment_start = i;
            }
        } else {
            i += 1;
        }
    }
    wrap_segment(&chars[segment_start..], options, &mut output);

    Ok(output)
}

fn is_separator(ch: char) -> bool {
    ch == ' ' || ch == '\t' || ch == '\n' || ch == '\r' || ch == '\u{00A0}'
}

fn wrap_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let words: Vec<Vec<char>> = segment
        .split(|&ch| is_separator(ch))
        .filter(|part| !part.is_empty())
        .map(|part| part.to_vec())
        .collect();

    if words.is_empty() {
        return;
    }

    let mut first_line = true;
    let mut line = String::new();
    let mut content_len = 0usize;

    for word in words {
        let mut offset = 0usize;
        while offset < word.len() {
            let indent = if first_line {
                options.first_indent
            } else {
                options.rest_indent
            };
            let capacity = options.width - indent;
            let suffix_len = word.len() - offset;
            let empty_line = content_len == 0;

            if suffix_len <= capacity {
                let needed = suffix_len + usize::from(!empty_line);
                if content_len + needed <= capacity {
                    if empty_line {
                        line = " ".repeat(indent);
                    } else {
                        line.push(' ');
                        content_len += 1;
                    }
                    for &ch in &word[offset..] {
                        line.push(ch);
                    }
                    content_len += suffix_len;
                    offset = word.len();
                } else {
                    output.push(line);
                    first_line = false;
                    line = String::new();
                    content_len = 0;
                }
            } else if empty_line {
                line = " ".repeat(indent);
                for &ch in &word[offset..offset + capacity] {
                    line.push(ch);
                }
                content_len = capacity;
                offset += capacity;
                output.push(line);
                first_line = false;
                line = String::new();
                content_len = 0;
            } else {
                let available = capacity.saturating_sub(content_len + 1);
                if available >= 2 {
                    line.push(' ');
                    content_len += 1;
                    for &ch in &word[offset..offset + available] {
                        line.push(ch);
                    }
                    content_len += available;
                    offset += available;
                    output.push(line);
                    first_line = false;
                    line = String::new();
                    content_len = 0;
                } else {
                    output.push(line);
                    first_line = false;
                    line = String::new();
                    content_len = 0;
                }
            }
        }
    }

    if content_len > 0 {
        output.push(line);
    }
}
