#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let normalized = input.replace("\r\n", "\n").replace('\r', "\n");
    let chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut i = 0;

    while i < chars.len() {
        if chars[i] == '\n' {
            let mut end = i + 1;
            while end < chars.len() && chars[end] == '\n' {
                end += 1;
            }
            if end - i >= 2 {
                wrap_segment(&chars[segment_start..i], options, &mut output);
                for _ in 1..(end - i) {
                    output.push(String::new());
                }
                segment_start = end;
            }
            i = end;
        } else {
            i += 1;
        }
    }

    wrap_segment(&chars[segment_start..], options, &mut output);
    Ok(output)
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn wrap_segment(chars: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words: Vec<Vec<char>> = Vec::new();
    let mut word = Vec::new();

    for &ch in chars {
        if is_separator(ch) {
            if !word.is_empty() {
                words.push(std::mem::take(&mut word));
            }
        } else {
            word.push(ch);
        }
    }
    if !word.is_empty() {
        words.push(word);
    }

    if words.is_empty() {
        return;
    }

    let mut first_line = true;
    let mut line_content = String::new();
    let mut content_len = 0usize;

    for word in words {
        let mut suffix = word;
        let mut offset = 0usize;

        while offset < suffix.len() {
            let indent = if first_line {
                options.first_indent
            } else {
                options.rest_indent
            };
            let capacity = options.width - indent;
            let remaining = suffix.len() - offset;

            if remaining <= capacity {
                if content_len == 0 {
                    line_content.extend(suffix[offset..].iter());
                    content_len += remaining;
                    offset = suffix.len();
                } else if content_len + 1 + remaining <= capacity {
                    line_content.push(' ');
                    line_content.extend(suffix[offset..].iter());
                    content_len += 1 + remaining;
                    offset = suffix.len();
                } else {
                    emit_line(
                        &mut line_content,
                        indent,
                        output,
                    );
                    content_len = 0;
                    first_line = false;
                }
            } else if content_len == 0 {
                line_content.extend(suffix[offset..offset + capacity].iter());
                offset += capacity;
                emit_line(&mut line_content, indent, output);
                content_len = 0;
                first_line = false;
            } else {
                let available = capacity - content_len - 1;
                if available >= 2 {
                    line_content.push(' ');
                    line_content.extend(suffix[offset..offset + available].iter());
                    content_len += 1 + available;
                    offset += available;
                    emit_line(&mut line_content, indent, output);
                    content_len = 0;
                    first_line = false;
                } else {
                    emit_line(&mut line_content, indent, output);
                    content_len = 0;
                    first_line = false;
                }
            }
        }
    }

    if content_len > 0 {
        let indent = if first_line {
            options.first_indent
        } else {
            options.rest_indent
        };
        emit_line(&mut line_content, indent, output);
    }
}

fn emit_line(line_content: &mut String, indent: usize, output: &mut Vec<String>) {
    let mut line = String::with_capacity(indent + line_content.len());
    line.extend(std::iter::repeat(' ').take(indent));
    line.push_str(line_content);
    output.push(line);
    line_content.clear();
}
