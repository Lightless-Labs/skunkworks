#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged {
        row: usize,
        expected: usize,
        actual: usize,
    },
    InvalidChar {
        row: usize,
        column: usize,
        byte: usize,
    },
    TooWide { column: usize, width: usize },
}

pub fn render(rows: &[Vec<String>]) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let expected = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != expected {
            return Err(TableError::Ragged {
                row,
                expected,
                actual: cells.len(),
            });
        }
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if !(' '..='~').contains(&ch) {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }
        }
    }

    let trimmed: Vec<Vec<&str>> = rows
        .iter()
        .map(|cells| {
            cells
                .iter()
                .map(|cell| cell.trim_end_matches(' '))
                .collect()
        })
        .collect();

    let mut widths = vec![1usize; expected];
    for cells in &trimmed {
        for (column, cell) in cells.iter().enumerate() {
            widths[column] = widths[column].max(cell.len());
        }
    }

    for (column, &width) in widths.iter().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let border_len = widths.iter().try_fold(1usize, |total, &width| {
        total.checked_add(width.checked_add(3)?)
    });

    let mut output = String::new();
    if let Some(capacity) = border_len
        .and_then(|len| len.checked_mul(rows.len().checked_add(1)?))
        .and_then(|borders| {
            rows.len()
                .checked_mul(expected.checked_mul(4)?.checked_add(2)?)?
                .checked_add(borders)
        })
    {
        output.reserve(capacity);
    }

    let append_border = |output: &mut String| {
        output.push('+');
        for &width in &widths {
            for _ in 0..width + 2 {
                output.push('-');
            }
            output.push('+');
        }
        output.push('\n');
    };

    append_border(&mut output);
    for cells in &trimmed {
        output.push('|');
        for (column, cell) in cells.iter().enumerate() {
            output.push(' ');
            output.push_str(cell);
            for _ in cell.len()..widths[column] {
                output.push(' ');
            }
            output.push_str(" |");
        }
        output.push('\n');
    }
    append_border(&mut output);

    Ok(output)
}
