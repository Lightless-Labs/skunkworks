#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
}

pub fn render(rows: &[Vec<String>]) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    if let Some(row) = rows.iter().position(Vec::is_empty) {
        return Err(TableError::EmptyRow { row });
    }

    let expected = rows[0].len();
    if let Some((row, cells)) = rows
        .iter()
        .enumerate()
        .find(|(_, cells)| cells.len() != expected)
    {
        return Err(TableError::Ragged {
            row,
            expected,
            actual: cells.len(),
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            if let Some((byte, _)) = cell
                .bytes()
                .enumerate()
                .find(|(_, byte)| !(b' '..=b'~').contains(byte))
            {
                return Err(TableError::InvalidChar { row, column, byte });
            }
        }
    }

    let mut trimmed = Vec::with_capacity(rows.len());
    let mut widths = vec![1usize; expected];

    for cells in rows {
        let mut row = Vec::with_capacity(expected);
        for (column, cell) in cells.iter().enumerate() {
            let content = cell.trim_end_matches(' ');
            widths[column] = widths[column].max(content.len());
            row.push(content);
        }
        trimmed.push(row);
    }

    if let Some((column, &width)) = widths
        .iter()
        .enumerate()
        .find(|(_, width)| **width > 32)
    {
        return Err(TableError::TooWide { column, width });
    }

    let mut output = String::new();

    for &width in &widths {
        output.push('+');
        output.extend(std::iter::repeat_n('-', width + 2));
    }
    output.push_str("+\n");

    for row in &trimmed {
        output.push('|');
        for (column, content) in row.iter().enumerate() {
            let width = widths[column];
            output.push(' ');
            output.push_str(content);
            output.extend(std::iter::repeat_n(' ', width - content.len()));
            output.push_str(" |");
        }
        output.push('\n');
    }

    for &width in &widths {
        output.push('+');
        output.extend(std::iter::repeat_n('-', width + 2));
    }
    output.push_str("+\n");

    Ok(output)
}
