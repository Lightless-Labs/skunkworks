#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 {
        return false;
    }
    if !bytes[0].is_ascii_lowercase() || bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn constraint_matches(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", byte));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn validate_inputs(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_names = std::collections::BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

type Constraints = std::collections::BTreeMap<String, Vec<Constraint>>;
type Assignments = std::collections::BTreeMap<String, Version>;

fn active_constraints(roots: &[Requirement], assignments: &Assignments, catalog: &[Package]) -> Constraints {
    let by_name: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut active = Constraints::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (name, version) in assignments {
        if let Some(package) = by_name.get(name.as_str()) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        canonicalize(constraints);
    }
    active
}

fn matching_releases<'a>(
    package: &'a Package,
    constraints: &[Constraint],
) -> Vec<&'a Release> {
    let mut releases: Vec<&Release> = package
        .releases
        .iter()
        .filter(|release| {
            constraints
                .iter()
                .all(|constraint| constraint_matches(*constraint, release.version))
        })
        .collect();
    releases.sort_by(|a, b| b.version.cmp(&a.version));
    releases
}

fn solution_vector_better(
    candidate: &Assignments,
    incumbent: &Assignments,
    catalog_names: &[String],
) -> bool {
    for name in catalog_names {
        match (candidate.get(name), incumbent.get(name)) {
            (None, None) => {}
            (None, Some(_)) => return false,
            (Some(_), None) => return true,
            (Some(a), Some(b)) if a != b => return a > b,
            _ => {}
        }
    }
    false
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Assignments, Constraints)>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn lock_cost(&self, assignments: &Assignments) -> usize {
        self.locks
            .iter()
            .filter(|lock| assignments.get(&lock.package) != Some(&lock.version))
            .count()
    }

    fn inspect(&mut self, assignments: &mut Assignments) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, assignments, self.catalog);
        let by_name: std::collections::BTreeMap<&str, &Package> =
            self.catalog.iter().map(|p| (p.name.as_str(), p)).collect();

        if let Some(name) = active.keys().find(|name| !by_name.contains_key(name.as_str())) {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in assignments.iter() {
            if let Some(constraints) = active.get(name) {
                if !constraints
                    .iter()
                    .all(|constraint| constraint_matches(*constraint, *version))
                {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if !assignments.contains_key(name) {
                let package = by_name[name.as_str()];
                if matching_releases(package, constraints).is_empty() {
                    return Err(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        let next = active.keys().find(|name| !assignments.contains_key(*name)).cloned();
        let Some(name) = next else {
            let cost = self.lock_cost(assignments);
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_assignments, _)) => {
                    cost < *best_cost
                        || (cost == *best_cost
                            && solution_vector_better(
                                assignments,
                                best_assignments,
                                &self.catalog_names(),
                            ))
                }
            };
            if replace {
                self.best = Some((cost, assignments.clone(), active));
            }
            return Ok(());
        };

        let package = by_name[name.as_str()];
        let constraints = &active[&name];
        let mut candidates = matching_releases(package, constraints);

        if let Some(lock) = self.locks.iter().find(|lock| lock.package == name) {
            if let Some(index) = candidates
                .iter()
                .position(|release| release.version == lock.version)
            {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        for release in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assignments.insert(name.clone(), release.version);
            let child_result = self.inspect(assignments);
            assignments.remove(&name);

            if let Err(error) = child_result {
                match error {
                    ResolveError::Limit { .. } => return Err(error),
                    other => {
                        if self.first_error.is_none() {
                            self.first_error = Some(other);
                        }
                    }
                }
            }
        }

        Ok(())
    }

    fn catalog_names(&self) -> Vec<String> {
        let mut names: Vec<String> = self.catalog.iter().map(|p| p.name.clone()).collect();
        names.sort();
        names
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let mut search = Search {
        catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut assignments = Assignments::new();

    if let Err(error) = search.inspect(&mut assignments) {
        return Err(error);
    }

    let Some((_, solution, final_constraints)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = solution
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let lock_map: std::collections::BTreeMap<&str, Version> = locks
        .iter()
        .map(|lock| (lock.package.as_str(), lock.version))
        .collect();
    let selected_map = &solution;
    let mut change_names = std::collections::BTreeSet::new();
    change_names.extend(lock_map.keys().map(|name| (*name).to_string()));
    change_names.extend(selected_map.keys().cloned());

    let catalog_versions: std::collections::BTreeMap<&str, std::collections::BTreeSet<Version>> =
        catalog
            .iter()
            .map(|package| {
                (
                    package.name.as_str(),
                    package.releases.iter().map(|r| r.version).collect(),
                )
            })
            .collect();

    let mut changes = Vec::new();
    for name in change_names {
        let before = lock_map.get(name.as_str()).copied();
        let after = selected_map.get(&name).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(locked), Some(_)) => {
                if !catalog_versions
                    .get(name.as_str())
                    .is_some_and(|versions| versions.contains(&locked))
                {
                    ChangeReason::Unavailable
                } else if final_constraints
                    .get(&name)
                    .is_some_and(|constraints| {
                        !constraints
                            .iter()
                            .all(|constraint| constraint_matches(*constraint, locked))
                    })
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
