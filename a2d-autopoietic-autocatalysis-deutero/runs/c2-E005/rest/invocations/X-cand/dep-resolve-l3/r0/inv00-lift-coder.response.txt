#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let canonical = canonical_constraints(constraints.to_vec());
    canonical
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(s: &str) -> String {
    let mut out = String::new();
    for byte in s.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", byte));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            encode_text(site),
            encode_text(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode_text(package))
        }
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode_text(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode_text(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            if version < v {
                false
            } else if v.major > 0 {
                version.major == v.major
            } else if v.minor > 0 {
                version.major == 0 && version.minor == v.minor
            } else {
                version == v
            }
        }
    }
}

type Active = std::collections::BTreeMap<String, Vec<Constraint>>;
type Assigned = std::collections::BTreeMap<String, Version>;

fn collect_active(roots: &[Requirement], assigned: &Assigned, by_name: &std::collections::BTreeMap<String, &Package>) -> Active {
    let mut active = Active::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in assigned {
        if let Some(package) = by_name.get(name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        *constraints = canonical_constraints(std::mem::take(constraints));
    }
    active
}

struct Search<'a> {
    roots: &'a [Requirement],
    by_name: std::collections::BTreeMap<String, &'a Package>,
    locks: std::collections::BTreeMap<String, Version>,
    catalog_names: Vec<String>,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Version>, Assigned)>,
    first_error: Option<ResolveError>,
    limited: bool,
}

impl<'a> Search<'a> {
    fn visit(&mut self, assigned: &mut Assigned) {
        if self.limited {
            return;
        }

        let active = collect_active(self.roots, assigned, &self.by_name);

        if let Some(name) = active
            .keys()
            .filter(|name| !self.by_name.contains_key(*name))
            .min()
            .cloned()
        {
            self.record_error(ResolveError::Missing { package: name });
            return;
        }

        for (name, version) in assigned.iter() {
            if let Some(requirements) = active.get(name) {
                if !requirements
                    .iter()
                    .all(|c| matches_constraint(*version, *c))
                {
                    self.record_error(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: requirements.clone(),
                    });
                    return;
                }
            }
        }

        for (name, requirements) in &active {
            if assigned.contains_key(name) {
                continue;
            }
            let package = self.by_name[name];
            if !package.releases.iter().any(|release| {
                requirements
                    .iter()
                    .all(|c| matches_constraint(release.version, *c))
            }) {
                self.record_error(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: requirements.clone(),
                });
                return;
            }
        }

        let next = active.keys().find(|name| !assigned.contains_key(*name)).cloned();
        let Some(package_name) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(name, version)| assigned.get(*name) != Some(*version))
                .count();
            let vector = self
                .catalog_names
                .iter()
                .map(|name| assigned.get(name).copied())
                .collect::<Vec<_>>();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_vector, _)) => {
                    cost < *best_cost
                        || (cost == *best_cost && vector_cmp(&vector, best_vector).is_gt())
                }
            };
            if replace {
                self.best = Some((cost, vector, assigned.clone()));
            }
            return;
        };

        let requirements = active.get(&package_name).cloned().unwrap_or_default();
        let package = self.by_name[&package_name];
        let mut candidates = package
            .releases
            .iter()
            .filter(|release| {
                requirements
                    .iter()
                    .all(|c| matches_constraint(release.version, *c))
            })
            .map(|release| release.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(locked) = self.locks.get(&package_name) {
            if let Some(index) = candidates.iter().position(|v| v == locked) {
                let version = candidates.remove(index);
                candidates.insert(0, version);
            }
        }

        let mut first_local_error = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                self.limited = true;
                return;
            }
            self.attempts += 1;
            assigned.insert(package_name.clone(), candidate);
            self.visit(assigned);
            assigned.remove(&package_name);

            if self.limited {
                return;
            }
            if first_local_error.is_none() && self.best.is_none() {
                first_local_error = self.first_error.clone();
            }
        }

        if self.best.is_none() {
            if let Some(error) = first_local_error {
                self.record_error(error);
            }
        }
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }
}

fn vector_cmp(a: &[Option<Version>], b: &[Option<Version>]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b) {
        let ordering = left.cmp(right);
        if ordering != std::cmp::Ordering::Equal {
            return ordering;
        }
    }
    a.len().cmp(&b.len())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let by_name = catalog
        .iter()
        .map(|package| (package.name.clone(), package))
        .collect::<std::collections::BTreeMap<_, _>>();
    let mut lock_map = std::collections::BTreeMap::new();
    for lock in locks {
        lock_map.insert(lock.package.clone(), lock.version);
    }
    let mut catalog_names = by_name.keys().cloned().collect::<Vec<_>>();
    catalog_names.sort();

    let mut search = Search {
        roots,
        by_name,
        locks: lock_map,
        catalog_names,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        limited: false,
    };
    search.visit(&mut Assigned::new());

    if search.limited {
        return Err(ResolveError::Limit {
            attempts: search.max_attempts,
        });
    }

    let Some((_, _, selected_map)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::Missing {
            package: String::new(),
        }));
    };

    let selected = selected_map
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let final_active = collect_active(roots, &selected_map, &search.by_name);
    let mut names = std::collections::BTreeSet::new();
    names.extend(lock_map.keys().cloned());
    names.extend(selected_map.keys().cloned());

    let mut changes = Vec::new();
    for package in names {
        let before = lock_map.get(&package).copied();
        let after = selected_map.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .by_name
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active.get(&package).is_some_and(|requirements| {
                    !requirements
                        .iter()
                        .all(|c| matches_constraint(old, *c))
                }) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
