#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn canonical_constraints(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn matches(version: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|constraint| match *constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    })
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonical_constraints(&mut canonical);
    canonical
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", byte));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn active_constraints(
    roots: &[Requirement],
    catalog: &[Package],
    assigned: &[(String, Version)],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in assigned {
        if let Some(package) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        canonical_constraints(constraints);
    }
    active
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<Vec<(String, Version)>>,
    best_cost: usize,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn lock_cost(&self, selected: &[(String, Version)]) -> usize {
        self.locks
            .iter()
            .filter(|lock| {
                !selected
                    .iter()
                    .any(|(name, version)| name == &lock.package && *version == lock.version)
            })
            .count()
    }

    fn better_vector(&self, candidate: &[(String, Version)], current: &[(String, Version)]) -> bool {
        let mut names: Vec<&str> = self.catalog.iter().map(|p| p.name.as_str()).collect();
        names.sort_unstable();
        names.dedup();
        for name in names {
            let a = candidate
                .iter()
                .find(|(n, _)| n == name)
                .map(|(_, v)| *v);
            let b = current.iter().find(|(n, _)| n == name).map(|(_, v)| *v);
            match (a, b) {
                (Some(x), Some(y)) if x != y => return x > y,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                _ => {}
            }
        }
        false
    }

    fn visit(&mut self, assigned: &mut Vec<(String, Version)>) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, self.catalog, assigned);

        if let Some(package) = active
            .keys()
            .filter(|name| !self.catalog.iter().any(|p| p.name == **name))
            .min()
        {
            let error = ResolveError::Missing {
                package: package.clone(),
            };
            self.record_error(error.clone());
            return Err(error);
        }

        let mut selected_sorted: Vec<_> = assigned.iter().collect();
        selected_sorted.sort_by(|a, b| a.0.cmp(&b.0));
        for (name, version) in selected_sorted {
            if let Some(constraints) = active.get(name) {
                if !matches(*version, constraints) {
                    let error = ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    };
                    self.record_error(error.clone());
                    return Err(error);
                }
            }
        }

        let mut unselected: Vec<String> = active
            .keys()
            .filter(|name| !assigned.iter().any(|(n, _)| n == *name))
            .cloned()
            .collect();
        unselected.sort();
        for name in &unselected {
            let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
            let constraints = active.get(name).unwrap();
            if !package
                .releases
                .iter()
                .any(|release| matches(release.version, constraints))
            {
                let error = ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                };
                self.record_error(error.clone());
                return Err(error);
            }
        }

        if unselected.is_empty() {
            let mut solution = assigned.clone();
            solution.sort_by(|a, b| a.0.cmp(&b.0));
            let cost = self.lock_cost(&solution);
            let replace = match &self.best {
                None => true,
                Some(best) => cost < self.best_cost || (cost == self.best_cost && self.better_vector(&solution, best)),
            };
            if replace {
                self.best_cost = cost;
                self.best = Some(solution);
            }
            return Ok(());
        }

        let name = unselected.remove(0);
        let constraints = active.get(&name).unwrap();
        let package = self.catalog.iter().find(|p| p.name == name).unwrap();
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .filter(|release| matches(release.version, constraints))
            .map(|release| release.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.iter().find(|lock| lock.package == name) {
            if let Some(index) = candidates.iter().position(|v| *v == lock.version) {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assigned.push((name.clone(), version));
            let child = self.visit(assigned);
            assigned.pop();
            match child {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    return Err(ResolveError::Limit { attempts });
                }
                Err(_) => {}
            }
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        best_cost: usize::MAX,
        first_error: None,
    };
    let mut assigned = Vec::new();
    match search.visit(&mut assigned) {
        Err(error @ ResolveError::Limit { .. }) => return Err(error),
        _ => {}
    }

    let selected_pairs = match search.best {
        Some(solution) => solution,
        None => return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        })),
    };

    let selected: Vec<Selection> = selected_pairs
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let final_active = active_constraints(roots, catalog, &selected_pairs);
    let mut before = std::collections::BTreeMap::<String, Version>::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }
    let mut after = std::collections::BTreeMap::<String, Version>::new();
    for (package, version) in &selected_pairs {
        after.insert(package.clone(), *version);
    }
    let all_names: std::collections::BTreeSet<String> =
        before.keys().chain(after.keys()).cloned().collect();

    let mut changes = Vec::new();
    for package in all_names {
        let b = before.get(&package).copied();
        let a = after.get(&package).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == package)
                    .map(|p| p.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active
                    .get(&package)
                    .map(|constraints| !matches(old, constraints))
                    .unwrap_or(false)
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
