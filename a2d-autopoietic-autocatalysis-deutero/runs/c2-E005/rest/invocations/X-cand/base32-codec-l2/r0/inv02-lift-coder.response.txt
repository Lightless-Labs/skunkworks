#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn decode_value(byte: u8, alphabet: Alphabet) -> Option<u8> {
    alphabet_bytes(alphabet)
        .iter()
        .position(|&candidate| candidate == byte)
        .map(|value| value as u8)
}

pub fn encode(input: &[u8], alphabet: Alphabet, padding: Padding) -> String {
    let symbols = alphabet_bytes(alphabet);
    let mut output = String::new();
    let mut accumulator = 0u16;
    let mut bits = 0u8;

    for &byte in input {
        accumulator = (accumulator << 8) | u16::from(byte);
        bits += 8;

        while bits >= 5 {
            bits -= 5;
            let value = ((accumulator >> bits) & 0x1f) as usize;
            output.push(symbols[value] as char);
        }
    }

    if bits > 0 {
        let value = ((accumulator << (5 - bits)) & 0x1f) as usize;
        output.push(symbols[value] as char);
    }

    if padding == Padding::Required {
        while output.len() % 8 != 0 {
            output.push('=');
        }
    }

    output
}

pub fn decode(
    input: &str,
    alphabet: Alphabet,
    padding: Padding,
) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let n = bytes.len();

    // Pass 1: validate every byte lexically.
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && decode_value(byte, alphabet).is_none() {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let p = bytes.iter().position(|&byte| byte == b'=').unwrap_or(n);

    // Pass 2: validate padding placement and mode.
    match padding {
        Padding::Omitted if p < n => return Err(DecodeError::Padding { offset: p }),
        Padding::Required => {
            if let Some(offset) = bytes[p..].iter().position(|&byte| byte != b'=') {
                return Err(DecodeError::Padding { offset: p + offset });
            }
        }
        Padding::Omitted => {}
    }

    let data_len = p;
    let residue = data_len % 8;

    // Pass 3: validate data length.
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength { offset: n });
    }

    // Pass 4: validate required padding shape.
    if padding == Padding::Required {
        let required_padding = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => unreachable!(),
        };
        if n % 8 != 0 || n - p != required_padding {
            return Err(DecodeError::Padding { offset: p });
        }
    }

    // Pass 5: validate unused low bits.
    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };
    if unused_bits > 0 {
        let last = decode_value(bytes[data_len - 1], alphabet).unwrap();
        if last & ((1u8 << unused_bits) - 1) != 0 {
            return Err(DecodeError::TrailingBits {
                offset: data_len - 1,
            });
        }
    }

    let capacity = data_len
        .checked_mul(5)
        .map(|bits| bits / 8)
        .unwrap_or(0);
    let mut output = Vec::with_capacity(capacity);
    let mut accumulator = 0u16;
    let mut bits = 0u8;

    for &byte in &bytes[..data_len] {
        let value = decode_value(byte, alphabet).unwrap();
        accumulator = (accumulator << 5) | u16::from(value);
        bits += 5;

        while bits >= 8 {
            bits -= 8;
            output.push((accumulator >> bits) as u8);
        }
    }

    Ok(output)
}
