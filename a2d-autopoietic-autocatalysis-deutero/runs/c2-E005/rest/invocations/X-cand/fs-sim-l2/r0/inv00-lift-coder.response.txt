```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(&self.cwd)
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                other => components.push(other.to_string()),
            }
        }

        if components.is_empty() {
            Ok("/".to_string())
        } else {
            Ok(format!("/{}", components.join("/")))
        }
    }

    fn components(path: &str) -> Vec<String> {
        path.split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_string)
            .collect()
    }

    fn parent_and_name(path: &str) -> Option<(String, String)> {
        if path == "/" {
            return None;
        }
        let idx = path.rfind('/')?;
        let parent = if idx == 0 {
            "/".to_string()
        } else {
            path[..idx].to_string()
        };
        Some((parent, path[idx + 1..].to_string()))
    }

    fn lookup_parent(&self, path: &str) -> Result<String, Error> {
        let components = Self::components(path);
        let mut prefix = String::new();

        for component in components.iter().take(components.len().saturating_sub(1)) {
            prefix.push('/');
            prefix.push_str(component);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }

        if prefix.is_empty() {
            prefix.push('/');
        }
        Ok(prefix)
    }

    fn lookup_final(&self, path: &str) -> Result<Option<&Entry>, Error> {
        let parent = self.lookup_parent(path)?;
        if path == "/" {
            return Ok(self.entries.get("/"));
        }
        let (expected_parent, _) = Self::parent_and_name(path).unwrap();
        debug_assert_eq!(parent, expected_parent);

        match self.entries.get(path) {
            Some(entry) => Ok(Some(entry)),
            None => Ok(None),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Exists(canonical));
        }

        let parent = self.lookup_parent(&canonical)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }

        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }

        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::IsDir(canonical));
        }

        let parent = self.lookup_parent(&canonical)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }

        match self.entries.get_mut(&canonical) {
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize(path)?;
        match self.lookup_final(&canonical)? {
            Some(Entry::File(contents)) => Ok(contents.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            None => Err(Error::Missing(canonical)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize(path)?;
        match self.lookup_final(&canonical)? {
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => return Err(Error::NotDir(canonical)),
            None => return Err(Error::Missing(canonical)),
        }

        let prefix = if canonical == "/" {
            "/".to_string()
        } else {
            format!("{}/", canonical)
        };

        let mut result = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == &canonical || !entry_path.starts_with(&prefix) {
                continue;
            }
            let remainder = &entry_path[prefix.len()..];
            if remainder.contains('/') {
                continue;
            }
            let kind = match entry {
                Entry::Dir => Kind::Dir,
                Entry::File(_) => Kind::File,
            };
            result.push((remainder.to_string(), kind));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        match self.lookup_final(&canonical)? {
            Some(Entry::Dir) => {
                self.cwd = canonical;
                Ok(())
            }
            Some(Entry::File(_)) | None => Err(Error::NotDir(canonical)),
        }
    }
}
```
