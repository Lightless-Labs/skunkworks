fn edit_changes(old: &[String], new: &[String], side: usize) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0usize, 0usize);
    let mut base_pos = 0usize;
    let mut run_start: Option<usize> = None;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();
    let mut changes = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 run_start: &mut Option<usize>,
                 deletes: &mut usize,
                 inserts: &mut Vec<String>| {
        if let Some(start) = run_start.take() {
            changes.push(Change {
                start,
                end: start + *deletes,
                replacement: std::mem::take(inserts),
                side,
            });
            *deletes = 0;
        }
    };

    while i < n || j < m {
        let op = if i == n {
            Op::Insert
        } else if j == m {
            Op::Delete
        } else if old[i] == new[j] {
            Op::Keep
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            if del > ins || (del == ins && (i + j) % 2 == 0) {
                Op::Delete
            } else {
                Op::Insert
            }
        };

        match op {
            Op::Keep => {
                flush(
                    &mut changes,
                    &mut run_start,
                    &mut deletes,
                    &mut inserts,
                );
                i += 1;
                j += 1;
                base_pos += 1;

                i += 1;
                base_pos += 1;
                deletes += 1;

                j += 1;
            }
        }
    }

        &mut changes,
        &mut run_start,
        &mut deletes,
        &mut inserts,
    );

}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_empty = a.start == a.end;
    let b_empty = b.start == b.end;

    match (a_empty, b_empty) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn build_components(changes: Vec<Change>) -> Vec<Component> {
    let n = changes.len();
    let mut seen = vec![false; n];
    let mut components = Vec::new();

    for seed in 0..n {
        if seen[seed] {
            continue;
        }
        seen[seed] = true;
        let mut stack = vec![seed];
        let mut members = Vec::new();

        while let Some(i) = stack.pop() {
            members.push(changes[i].clone());
            for j in 0..n {
                if !seen[j] && interacts(&changes[i], &changes[j]) {
                    seen[j] = true;
                    stack.push(j);
                }
            }
        }

        let start = members.iter().map(|c| c.start).min().unwrap();
        let end = members.iter().map(|c| c.end).max().unwrap();
        components.push(Component {
            start,
            end,
            changes: members,
        });
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then(a.end.cmp(&b.end))
    });
    components
}

fn project(base: &[String], component: &Component, side: usize) -> Vec<String> {
    let mut changes: Vec<&Change> = component
        .changes
        .iter()
        .filter(|c| c.side == side)
        .collect();
    changes.sort_by_key(|c| (c.start, c.end));

    let mut out = Vec::new();
    let mut cursor = component.start;

    for change in changes {
        if change.start > cursor {
            out.extend(base[cursor..change.start].iter().cloned());
        }
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    if cursor < component.end {
        out.extend(base[cursor..component.end].iter().cloned());
    }
    out
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    let inputs = [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ];

    for (input, side) in &inputs {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn {
                side: side.clone(),
                byte,
            });
        }
    }

    for (input, side) in &inputs {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side: side.clone() });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = edit_changes(&base_lines, &left_lines, 0);
    changes.extend(edit_changes(&base_lines, &right_lines, 1));
    let components = build_components(changes);

    let mut output = Vec::<String>::new();
    let mut cursor = 0usize;
    let mut regions = Vec::new();

    for component in &components {
        if cursor < component.start {
            output.extend(base_lines[cursor..component.start].iter().cloned());
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(&base_lines, component, 0);
        let right_projection = project(&base_lines, component, 1);

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = regions.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(left_projection.iter().cloned());

                    let region_len = component.end - component.start;
                    let marker_start = if region_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!("||||||| BASE {marker_start},{region_len}"));
                    output.extend(base_projection.iter().cloned());
                    output.push("=======".to_owned());
                    output.extend(right_projection.iter().cloned());
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));

                    regions.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: region_len,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        cursor = component.end;
    }

    output.extend(base_lines[cursor..].iter().cloned());
    let mut text = String::new();
    for line in output {
        text.push_str(&line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: regions.len(),
        regions,
    })
}

fn projection(base: &[String], start: usize, end: usize, changes: &[Change]) -> Vec<String> {
    let mut relevant: Vec<&Change> = changes
        .iter()
        .filter(|c| c.start >= start && c.end <= end)
        .collect();
    relevant.sort_by_key(|c| (c.start, c.end));

    let mut out = Vec::new();
    let mut cursor = start;
    for change in relevant {
        if change.start < cursor {
            continue;
        }
        out.extend(base[cursor..change.start].iter().cloned());
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    out.extend(base[cursor..end].iter().cloned());
    out
}

    let right_changes = canonical_changes(base_lines, &lines[2], 2);

    let mut all_changes = left_changes.clone();
    all_changes.extend(right_changes.clone());

    let count = all_changes.len();
    let mut parent: Vec<usize> = (0..count).collect();

    fn root(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            let r = root(parent, parent[x]);
            parent[x] = r;
        }
        parent[x]
    }

    for i in 0..count {
        for j in i + 1..count {
            if interacts(&all_changes[i], &all_changes[j]) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut components: Vec<Component> = Vec::new();
    let mut roots = Vec::new();
    for i in 0..count {
        let r = root(&mut parent, i);
        if let Some(index) = roots.iter().position(|&x| x == r) {
            components[index].changes.push(all_changes[i].clone());
        } else {
            roots.push(r);
            components.push(Component {
                start: all_changes[i].start,
                end: all_changes[i].end,
                changes: vec![all_changes[i].clone()],
            });
        }
    }

    for component in &mut components {
        component.start = component.changes.iter().map(|c| c.start).min().unwrap();
        component.end = component.changes.iter().map(|c| c.end).max().unwrap();
    }
    components.sort_by_key(|c| (c.start, usize::from(c.start < c.end), c.end));

    let mut output = Vec::<String>::new();
    let mut conflicts = Vec::<Conflict>::new();
    let mut cursor = 0usize;

    for component in components {
        if component.start > cursor {
            append_lines(&mut output, &base_lines[cursor..component.start]);
        }
        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_component: Vec<Change> = component
            .changes
            .iter()
            .filter(|c| c.side == 1)
            .cloned()
            .collect();
        let right_component: Vec<Change> = component
            .changes
            .iter()
            .filter(|c| c.side == 2)
            .cloned()
            .collect();
        let left_projection = projection(
            base_lines,
            component.start,
            component.end,
            &left_component,
        );
        let right_projection = projection(
            base_lines,
            component.start,
            component.end,
            &right_component,
        );

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);
                    let base_start = if component.start < component.end {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!(
                        "||||||| BASE {base_start},{}",
                        component.end - component.start
                    ));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: output.len() - output_start,
                    });
                }
            }
        }
        cursor = cursor.max(component.end);
    }

    if cursor < base_lines.len() {
        append_lines(&mut output, &base_lines[cursor..]);
    }

    let text = if output.is_empty() {
        String::new()
    } else {
        let mut text = output.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        conflicts: conflicts.len(),
        regions: conflicts,
        text,
    })
}

fn lines(input: &str) -> Vec<String> {

fn edit_script(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut i = 0;
    let mut j = 0;
    let mut base_pos = 0;
    let mut changes = Vec::new();
    let mut run_start = None;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 run_start: &mut Option<usize>,
                 deletes: &mut usize,
                 inserts: &mut Vec<String>| {
        if let Some(start) = run_start.take() {
            changes.push(Change {
                start,
                end: start + *deletes,
                replacement: std::mem::take(inserts),
            });
            *deletes = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(&mut changes, &mut run_start, &mut deletes, &mut inserts);
            i += 1;
            j += 1;
            base_pos += 1;
        } else if i == n {
            run_start.get_or_insert(base_pos);
            inserts.push(new[j].clone());
            j += 1;
        } else if j == m {
            run_start.get_or_insert(base_pos);
            deletes += 1;
            i += 1;
            base_pos += 1;
        } else {
            let delete_lcs = lcs[i + 1][j];
            let insert_lcs = lcs[i][j + 1];
            let delete = if delete_lcs > insert_lcs {
                true
            } else if insert_lcs > delete_lcs {
                false
            } else {
                (i + j) % 2 == 0
            };

            run_start.get_or_insert(base_pos);
            if delete {
                deletes += 1;
                i += 1;
                base_pos += 1;
            } else {
                inserts.push(new[j].clone());
                j += 1;
            }
        }
    }

    flush(&mut changes, &mut run_start, &mut deletes, &mut inserts);
    changes
}

fn interact(a: &Change, b: &Change) -> bool {
    let a_positive = a.start < a.end;
    let b_positive = b.start < b.end;

    match (a_positive, b_positive) {
        (true, true) => a.start.max(b.start) < a.end.min(b.end),
        (false, false) => a.start == b.start,
        (false, true) => b.start < a.start && a.start < b.end,
        (true, false) => a.start < b.start && b.start < a.end,
    }
}

fn projection(
    base: &[String],
    start: usize,
    end: usize,
    changes: &[&Change],
) -> Vec<String> {
    let mut result = Vec::new();
    let mut cursor = start;

    for change in changes {
        for line in &base[cursor..change.start] {
            result.push(line.clone());
        }
        result.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    for line in &base[cursor..end] {
        result.push(line.clone());
    }
    result
}

fn push_line(output: &mut Vec<String>, line: String) {
    output.push(line);
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    validate_cr(base, MergeSide::Base)?;
    validate_cr(left, MergeSide::Left)?;
    validate_cr(right, MergeSide::Right)?;

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = edit_script(&base_lines, &left_lines);
    let right_changes = edit_script(&base_lines, &right_lines);

    let mut nodes: Vec<(bool, usize)> = (0..left_changes.len())
        .map(|i| (true, i))
        .chain((0..right_changes.len()).map(|i| (false, i)))
        .collect();

    let node_change = |node: (bool, usize)| -> &Change {
        if node.0 {
            &left_changes[node.1]
        } else {
            &right_changes[node.1]
        }
    };

    let mut parent: Vec<usize> = (0..nodes.len()).collect();

    fn find(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = find(parent, parent[x]);
        }
        parent[x]
    }

    for i in 0..nodes.len() {
        for j in i + 1..nodes.len() {
            if interact(node_change(nodes[i]), node_change(nodes[j])) {
                let ri = find(&mut parent, i);
                let rj = find(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut components: Vec<Component> = Vec::new();
    let mut roots: Vec<usize> = Vec::new();
    for (idx, node) in nodes.iter().copied().enumerate() {
        let root = find(&mut parent, idx);
        let pos = if let Some(pos) = roots.iter().position(|&r| r == root) {
            pos
        } else {
            roots.push(root);
            components.push(Component {
                start: usize::MAX,
                end: 0,
                changes: Vec::new(),
            });
            components.len() - 1
        };
        let change = node_change(node);
        components[pos].start = components[pos].start.min(change.start);
        components[pos].end = components[pos].end.max(change.end);
        components[pos].changes.push(node);
    }

    components.sort_by(|a, b| {
        a.start.cmp(&b.start).then_with(|| {
            let a_zero = a.start == a.end;
            let b_zero = b.start == b.end;
            b_zero.cmp(&a_zero)
        })
    });

    let mut output = Vec::<String>::new();
    let mut conflicts = Vec::<Conflict>::new();
    let mut base_cursor = 0usize;

    for component in components {
        for line in &base_lines[base_cursor..component.start] {
            push_line(&mut output, line.clone());
        }

        let mut lchanges: Vec<&Change> = component
            .changes
            .iter()
            .filter(|(is_left, _)| *is_left)
            .map(|(_, i)| &left_changes[*i])
            .collect();
        let mut rchanges: Vec<&Change> = component
            .changes
            .iter()
            .filter(|(is_left, _)| !*is_left)
            .map(|(_, i)| &right_changes[*i])
            .collect();
        lchanges.sort_by_key(|c| (c.start, c.end));
        rchanges.sort_by_key(|c| (c.start, c.end));

        let base_projection: Vec<String> = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &lchanges,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &rchanges,
        );

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {}", ordinal));
                    output.extend(left_projection.iter().cloned());
                    let base_start = if component.start < component.end {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!(
                        "||||||| BASE {},{}",
                        base_start,
                        component.end - component.start
                    ));
                    output.extend(base_projection.iter().cloned());
                    output.push("=======".to_owned());
                    output.extend(right_projection.iter().cloned());
                    output.push(format!(">>>>>>> RIGHT {}", ordinal));
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    for line in &base_lines[base_cursor..] {
        output.push(line.clone());
    }

    let mut text = String::new();
    for line in output {
        text.push_str(&line);
        text.push('\n');
    }

    Ok(MergeResult {
        conflicts: conflicts.len(),
        text,
        regions: conflicts,
    })
}

fn split_lines(text: &str) -> Vec<String> {
    if text.is_empty() {
        return Vec::new();
    }
    text[..text.len() - 1]
        .split('\n')
        .map(str::to_owned)

fn validate<'a>(
    base: &'a str,
    left: &'a str,
    right: &'a str,
) -> Result<[Vec<String>; 3], MergeError> {
    let texts = [base, left, right];
    let sides = [MergeSide::Base, MergeSide::Left, MergeSide::Right];

    for (text, side) in texts.iter().zip(sides.iter()) {
        if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn {
                side: side.clone(),
                byte,
            });
        }
    }

    for (text, side) in texts.iter().zip(sides.iter()) {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side: side.clone() });
        }
    }

    let lines = [split_lines(base), split_lines(left), split_lines(right)];
    if lines.iter().any(|v| v.len() > 128) {
        return Err(MergeError::TooLarge {
            base_len: lines[0].len(),
            left_len: lines[1].len(),
            right_len: lines[2].len(),
        });
    }
    Ok(lines)
}

#[derive(Clone, Copy)]
enum Op {
    Keep,
    Delete,
    Insert,
}

fn canonical_ops(old: &[String], new: &[String]) -> Vec<Op> {
    let n = old.len();
    let m = new.len();
    let width = m + 1;
    let mut lcs = vec![0usize; (n + 1) * (m + 1)];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            let idx = i * width + j;
            lcs[idx] = if old[i] == new[j] {
                1 + lcs[(i + 1) * width + j + 1]
            } else {
                lcs[(i + 1) * width + j].max(lcs[i * width + j + 1])
            };
        }
    }

    let (mut i, mut j) = (0, 0);
    let mut ops = Vec::with_capacity(n + m);
    while i < n && j < m {
        if old[i] == new[j] {
            ops.push(Op::Keep);
            i += 1;
            j += 1;
        } else {
            let delete_lcs = lcs[(i + 1) * width + j];
            let insert_lcs = lcs[i * width + j + 1];
            if delete_lcs > insert_lcs
                || (delete_lcs == insert_lcs && (i + j) % 2 == 0)
            {
                ops.push(Op::Delete);
                i += 1;
            } else {
                ops.push(Op::Insert);
                j += 1;
            }
        }
    }
    while i < n {
        ops.push(Op::Delete);
        i += 1;
    }
    while j < m {
        ops.push(Op::Insert);
        j += 1;
    }
    ops
}

fn changes_for(old: &[String], new: &[String], side: usize) -> Vec<Change> {
    let ops = canonical_ops(old, new);
    let (mut i, mut j) = (0, 0);
    let mut changes = Vec::new();
    let mut run_start = None;
    let mut deleted = 0usize;
    let mut replacement = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 run_start: &mut Option<usize>,
                 deleted: &mut usize,
                 replacement: &mut Vec<String>| {
        if let Some(start) = run_start.take() {
            changes.push(Change {
                start,
                end: start + *deleted,
                replacement: std::mem::take(replacement),
                side,
            });
            *deleted = 0;
        }
    };

    for op in ops {
        match op {
            Op::Keep => {
                flush(&mut changes, &mut run_start, &mut deleted, &mut replacement);
                i += 1;
                j += 1;
            }
            Op::Delete => {
                run_start.get_or_insert(i);
                deleted += 1;
                i += 1;
            }
            Op::Insert => {
                run_start.get_or_insert(i);
                replacement.push(new[j].clone());
                j += 1;
            }
        }
    }
    flush(&mut changes, &mut run_start, &mut deleted, &mut replacement);
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn projection(
    base: &[String],
    start: usize,
    end: usize,
    changes: &[&Change],
) -> Vec<String> {
    let mut result = Vec::new();
    let mut cursor = start;

    for change in changes {
        if change.start > cursor {
            result.extend(base[cursor..change.start].iter().cloned());
        }
        result.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    if cursor < end {
        result.extend(base[cursor..end].iter().cloned());
    }
    result
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    let [base_lines, left_lines, right_lines] = validate(base, left, right)?;

    let mut changes = changes_for(&base_lines, &left_lines, 0);
    changes.extend(changes_for(&base_lines, &right_lines, 1));

    let mut parent: Vec<usize> = (0..changes.len()).collect();

    fn find(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = find(parent, parent[x]);
        }
        parent[x]
    }

    for i in 0..changes.len() {
        for j in i + 1..changes.len() {
            if interacts(&changes[i], &changes[j]) {
                let a = find(&mut parent, i);
                let b = find(&mut parent, j);
                if a != b {
                    parent[b] = a;
                }
            }
        }
    }

    let mut components: Vec<Component> = Vec::new();
    for i in 0..changes.len() {
        let root = find(&mut parent, i);
        if let Some(component) = components.iter_mut().find(|c| {
            c.changes
                .first()
                .map(|&idx| find(&mut parent, idx) == root)
                .unwrap_or(false)
        }) {
            component.start = component.start.min(changes[i].start);
            component.end = component.end.max(changes[i].end);
            component.changes.push(i);
        } else {
            components.push(Component {
                start: changes[i].start,
                end: changes[i].end,
                changes: vec![i],
            });
        }
    }

    components.sort_by(|a, b| {
        a.start
            .cmp(&b.start)
            .then_with(|| (a.start != a.end).cmp(&(b.start != b.end)))
            .then(a.end.cmp(&b.end))
    });

    let mut output = Vec::<String>::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for component in components {
        if base_cursor < component.start {
            append_lines(&mut output, &base_lines[base_cursor..component.start]);
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let mut left_changes: Vec<&Change> = component
            .changes
            .iter()
            .map(|&i| &changes[i])
            .filter(|c| c.side == 0)
            .collect();
        let mut right_changes: Vec<&Change> = component
            .changes
            .iter()
            .map(|&i| &changes[i])
            .filter(|c| c.side == 1)
            .collect();
        left_changes.sort_by_key(|c| (c.start, c.end));
        right_changes.sort_by_key(|c| (c.start, c.end));

        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &left_changes,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &right_changes,
        );

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    let base_len = component.end - component.start;
                    let base_line_start = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };

                    output.push(format!("<<<<<<< LEFT {}", ordinal));
                    append_lines(&mut output, &left_projection);
                    output.push(format!(
                        "||||||| BASE {},{}",
                        base_line_start, base_len
                    ));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {}", ordinal));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    if base_cursor < base_lines.len() {
        append_lines(&mut output, &base_lines[base_cursor..]);
    }

    let mut text = String::new();
    for line in output {
        text.push_str(&line);
        text.push('\n');
    }

    Ok(MergeResult {
        conflicts: conflicts.len(),
        regions: conflicts,
        text,
    })
}
