#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Debug, Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
}

#[derive(Debug, Clone)]
struct Component {
    start: usize,
    end: usize,
    changes: Vec<(bool, usize)>, // true = left, false = right
}

fn validate_cr(input: &str, side: MergeSide) -> Result<(), MergeError> {
    if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(MergeError::CarriageReturn { side, byte });
    }
    Ok(())
}

fn lines(input: &str) -> Vec<String> {
    if input.is_empty() {
        return Vec::new();
    }
    input[..input.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn edit_script(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut i = 0;
    let mut j = 0;
    let mut base_pos = 0;
    let mut changes = Vec::new();
    let mut run_start = None;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 run_start: &mut Option<usize>,
                 deletes: &mut usize,
                 inserts: &mut Vec<String>| {
        if let Some(start) = run_start.take() {
            changes.push(Change {
                start,
                end: start + *deletes,
                replacement: std::mem::take(inserts),
            });
            *deletes = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(&mut changes, &mut run_start, &mut deletes, &mut inserts);
            i += 1;
            j += 1;
            base_pos += 1;
        } else if i == n {
            run_start.get_or_insert(base_pos);
            inserts.push(new[j].clone());
            j += 1;
        } else if j == m {
            run_start.get_or_insert(base_pos);
            deletes += 1;
            i += 1;
            base_pos += 1;
        } else {
            let delete_lcs = lcs[i + 1][j];
            let insert_lcs = lcs[i][j + 1];
            let delete = if delete_lcs > insert_lcs {
                true
            } else if insert_lcs > delete_lcs {
                false
            } else {
                (i + j) % 2 == 0
            };

            run_start.get_or_insert(base_pos);
            if delete {
                deletes += 1;
                i += 1;
                base_pos += 1;
            } else {
                inserts.push(new[j].clone());
                j += 1;
            }
        }
    }

    flush(&mut changes, &mut run_start, &mut deletes, &mut inserts);
    changes
}

fn interact(a: &Change, b: &Change) -> bool {
    let a_positive = a.start < a.end;
    let b_positive = b.start < b.end;

    match (a_positive, b_positive) {
        (true, true) => a.start.max(b.start) < a.end.min(b.end),
        (false, false) => a.start == b.start,
        (false, true) => b.start < a.start && a.start < b.end,
        (true, false) => a.start < b.start && b.start < a.end,
    }
}

fn projection(
    base: &[String],
    start: usize,
    end: usize,
    changes: &[&Change],
) -> Vec<String> {
    let mut result = Vec::new();
    let mut cursor = start;

    for change in changes {
        for line in &base[cursor..change.start] {
            result.push(line.clone());
        }
        result.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    for line in &base[cursor..end] {
        result.push(line.clone());
    }
    result
}

fn push_line(output: &mut Vec<String>, line: String) {
    output.push(line);
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    validate_cr(base, MergeSide::Base)?;
    validate_cr(left, MergeSide::Left)?;
    validate_cr(right, MergeSide::Right)?;

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = lines(base);
    let left_lines = lines(left);
    let right_lines = lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = edit_script(&base_lines, &left_lines);
    let right_changes = edit_script(&base_lines, &right_lines);

    let mut nodes: Vec<(bool, usize)> = (0..left_changes.len())
        .map(|i| (true, i))
        .chain((0..right_changes.len()).map(|i| (false, i)))
        .collect();

    let node_change = |node: (bool, usize)| -> &Change {
        if node.0 {
            &left_changes[node.1]
        } else {
            &right_changes[node.1]
        }
    };

    let mut parent: Vec<usize> = (0..nodes.len()).collect();

    fn find(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = find(parent, parent[x]);
        }
        parent[x]
    }

    for i in 0..nodes.len() {
        for j in i + 1..nodes.len() {
            if interact(node_change(nodes[i]), node_change(nodes[j])) {
                let ri = find(&mut parent, i);
                let rj = find(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut components: Vec<Component> = Vec::new();
    let mut roots: Vec<usize> = Vec::new();
    for (idx, node) in nodes.iter().copied().enumerate() {
        let root = find(&mut parent, idx);
        let pos = if let Some(pos) = roots.iter().position(|&r| r == root) {
            pos
        } else {
            roots.push(root);
            components.push(Component {
                start: usize::MAX,
                end: 0,
                changes: Vec::new(),
            });
            components.len() - 1
        };
        let change = node_change(node);
        components[pos].start = components[pos].start.min(change.start);
        components[pos].end = components[pos].end.max(change.end);
        components[pos].changes.push(node);
    }

    components.sort_by(|a, b| {
        a.start.cmp(&b.start).then_with(|| {
            let a_zero = a.start == a.end;
            let b_zero = b.start == b.end;
            b_zero.cmp(&a_zero)
        })
    });

    let mut output = Vec::<String>::new();
    let mut conflicts = Vec::<Conflict>::new();
    let mut base_cursor = 0usize;

    for component in components {
        for line in &base_lines[base_cursor..component.start] {
            push_line(&mut output, line.clone());
        }

        let mut lchanges: Vec<&Change> = component
            .changes
            .iter()
            .filter(|(is_left, _)| *is_left)
            .map(|(_, i)| &left_changes[*i])
            .collect();
        let mut rchanges: Vec<&Change> = component
            .changes
            .iter()
            .filter(|(is_left, _)| !*is_left)
            .map(|(_, i)| &right_changes[*i])
            .collect();
        lchanges.sort_by_key(|c| (c.start, c.end));
        rchanges.sort_by_key(|c| (c.start, c.end));

        let base_projection: Vec<String> = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &lchanges,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &rchanges,
        );

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {}", ordinal));
                    output.extend(left_projection.iter().cloned());
                    let base_start = if component.start < component.end {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!(
                        "||||||| BASE {},{}",
                        base_start,
                        component.end - component.start
                    ));
                    output.extend(base_projection.iter().cloned());
                    output.push("=======".to_owned());
                    output.extend(right_projection.iter().cloned());
                    output.push(format!(">>>>>>> RIGHT {}", ordinal));
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    for line in &base_lines[base_cursor..] {
        output.push(line.clone());
    }

    let mut text = String::new();
    for line in output {
        text.push_str(&line);
        text.push('\n');
    }

    Ok(MergeResult {
        conflicts: conflicts.len(),
        text,
        regions: conflicts,
    })
}
