```rust
pub const MAX_DIM: usize = 4096;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Axis {
    Row,
    Column,
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Entry {
    pub row: usize,
    pub col: usize,
    pub value: f64,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Sparse {
    rows: usize,
    cols: usize,
    entries: Vec<Entry>,
}

#[derive(Clone, Debug, PartialEq)]
pub enum Error {
    Shape {
        axis: Axis,
        size: usize,
    },
    Bounds {
        entry: usize,
        axis: Axis,
        index: usize,
        limit: usize,
    },
    NonFinite {
        entry: usize,
    },
    SumOverflow {
        entry: usize,
    },
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        if rows > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Row,
                size: rows,
            });
        }
        if cols > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Column,
                size: cols,
            });
        }

        for (entry_index, entry) in entries.iter().enumerate() {
            if entry.row >= rows {
                return Err(Error::Bounds {
                    entry: entry_index,
                    axis: Axis::Row,
                    index: entry.row,
                    limit: rows,
                });
            }
            if entry.col >= cols {
                return Err(Error::Bounds {
                    entry: entry_index,
                    axis: Axis::Column,
                    index: entry.col,
                    limit: cols,
                });
            }
            if !entry.value.is_finite() {
                return Err(Error::NonFinite { entry: entry_index });
            }
        }

        let mut sums: Vec<((usize, usize), f64)> = Vec::new();
        for (entry_index, entry) in entries.iter().enumerate() {
            let coordinate = (entry.row, entry.col);
            if let Some((_, sum)) = sums.iter_mut().find(|(key, _)| *key == coordinate) {
                *sum += entry.value;
                if !sum.is_finite() {
                    return Err(Error::SumOverflow { entry: entry_index });
                }
            } else {
                // Every input value has already been checked finite. Starting at
                // positive zero and adding the first occurrence preserves the
                // specified ordinary-addition behavior, including signed zero.
                let sum = 0.0 + entry.value;
                if !sum.is_finite() {
                    return Err(Error::SumOverflow { entry: entry_index });
                }
                sums.push((coordinate, sum));
            }
        }

        let mut canonical: Vec<Entry> = sums
            .into_iter()
            .filter_map(|((row, col), value)| {
                if value == 0.0 {
                    None
                } else {
                    Some(Entry { row, col, value })
                }
            })
            .collect();
        canonical.sort_by_key(|entry| (entry.row, entry.col));

        Ok(Self {
            rows,
            cols,
            entries: canonical,
        })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn entries(&self) -> &[Entry] {
        &self.entries
    }
}
```
