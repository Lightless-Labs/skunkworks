```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = normalize_path(path, &self.cwd)?;
        let (parent, name) = split_parent(&canonical);
        let parent = self.resolve_full(&parent)?;
        let destination = join_component(&parent, &name);
        if !matches!(self.entries.get(&parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(parent));
        }
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = normalize_path(path, &self.cwd)?;
        let resolved = self.resolve_for_write(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let (parent, _) = split_parent(&resolved);
                if !matches!(self.entries.get(&parent), Some(Entry::Dir)) {
                    return Err(Error::NotDir(parent));
                }
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = normalize_path(path, &self.cwd)?;
        let resolved = self.resolve_full(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) | None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = normalize_path(path, &self.cwd)?;
        let resolved = self.resolve_full(&canonical)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }

        let mut result = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == "/" || entry_path == &resolved {
                continue;
            }
            let (parent, name) = split_parent(entry_path);
            if parent == resolved {
                let kind = match entry {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                    Entry::Link(_) => Kind::Link,
                };
                result.push((name, kind));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = normalize_path(path, &self.cwd)?;
        let resolved = self.resolve_full(&canonical)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = normalize_path(from, &self.cwd)?;
        let destination = normalize_path(to, &self.cwd)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent_raw, source_name) = split_parent(&source);
        let source_parent = self.resolve_full(&source_parent_raw)?;
        if !matches!(self.entries.get(&source_parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(source_parent));
        }
        let source_path = join_component(&source_parent, &source_name);
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent_raw, destination_name) = split_parent(&destination);
        let destination_parent = self.resolve_full(&destination_parent_raw)?;
        if !matches!(self.entries.get(&destination_parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(destination_parent));
        }
        let destination_path = join_component(&destination_parent, &destination_name);

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && is_within(&self.cwd, &destination_path)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && is_strictly_within(&destination_path, &source_path)
        {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(ref dest_entry) = destination_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if self.has_children(&destination_path) {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        let subtree: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| **p == source_path || is_strictly_within(p, &source_path))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();

        if let Some(dest_entry) = destination_entry {
            if matches!(dest_entry, Entry::Dir) {
                self.entries.remove(&destination_path);
            } else {
                self.entries.remove(&destination_path);
            }
        }

        for (old_path, _) in &subtree {
            self.entries.remove(old_path);
        }
        for (old_path, entry) in subtree {
            let suffix = &old_path[source_path.len()..];
            let new_path = if suffix.is_empty() {
                destination_path.clone()
            } else if destination_path == "/" {
                format!("/{suffix}")
            } else {
                format!("{destination_path}{suffix}")
            };
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = normalize_path(path, &self.cwd)?;
        if canonical == "/" {
            return Err(Error::Root);
        }
        let (parent_raw, name) = split_parent(&canonical);
        let parent = self.resolve_full(&parent_raw)?;
        if !matches!(self.entries.get(&parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(parent));
        }
        let resolved = join_component(&parent, &name);
        let entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &resolved) {
            return Err(Error::Busy(resolved));
        }
        if matches!(entry, Entry::Dir) && self.has_children(&resolved) && !recursive {
            return Err(Error::NotEmpty(resolved));
        }

        let to_remove: Vec<String> = self
            .entries
            .keys()
            .filter(|p| **p == resolved || is_strictly_within(p, &resolved))
            .cloned()
            .collect();
        for p in to_remove {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical_link = normalize_path(link, &self.cwd)?;
        validate_target(target)?;

        let (parent_raw, name) = split_parent(&canonical_link);
        let parent = self.resolve_full(&parent_raw)?;
        if !matches!(self.entries.get(&parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(parent));
        }
        let resolved_link = join_component(&parent, &name);
        if self.entries.contains_key(&resolved_link) {
            return Err(Error::Exists(resolved_link));
        }
        self.entries
            .insert(resolved_link, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = normalize_path(path, &self.cwd)?;
        let (parent_raw, name) = split_parent(&canonical);
        let parent = self.resolve_full(&parent_raw)?;
        if !matches!(self.entries.get(&parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(parent));
        }
        let resolved = join_component(&parent, &name);
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(Entry::Dir) | Some(Entry::File(_)) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = normalize_path(path, &self.cwd)?;
        let resolved = self.resolve_full(&canonical)?;
        let entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        let mut paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| **p == resolved || is_strictly_within(p, &resolved))
            .cloned()
            .collect();
        paths.sort();

        let mut totals = BTreeMap::new();
        for current in paths.iter().rev() {
            let size = match self.entries.get(current).unwrap() {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => {
                    let prefix = if current == "/" {
                        "/".to_string()
                    } else {
                        format!("{current}/")
                    };
                    totals
                        .iter()
                        .filter_map(|(descendant, size)| {
                            if descendant.starts_with(&prefix) {
                                let relative = &descendant[prefix.len()..];
                                if relative.is_empty() || relative.contains('/') {
                                    None
                                } else {
                                    Some(*size)
                                }
                            } else {
                                None
                            }
                        })
                        .sum()
                }
            };
            totals.insert(current.clone(), size);
        }

        let _ = entry;
        let mut rows: Vec<(String, u64)> = paths
            .into_iter()
            .map(|p| {
                let size = match self.entries.get(&p).unwrap() {
                    Entry::Dir => directory_total(self, &p),
                    Entry::File(data) => data.len() as u64,
                    Entry::Link(target) => target.len() as u64,
                };
                (p, size)
            })
            .collect();
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries.keys().any(|p| is_strictly_within(p, path))
    }

    fn resolve_full(&self, path: &str) -> Result<String, Error> {
        self.resolve(path, false)
    }

    fn resolve_for_write(&self, path: &str) -> Result<String, Error> {
        self.resolve(path, true)
    }

    fn resolve(&self, path: &str, allow_missing_final: bool) -> Result<String, Error> {
        let mut pending: Vec<String> = components(path).collect();
        let mut current = Vec::<String>::new();
        let mut visited = BTreeSet::<String>::new();
        let mut expansions = 0usize;

        while !pending.is_empty() {
            let component = pending.remove(0);
            current.push(component);
            let candidate = path_from_components(&current);
            let is_final = pending.is_empty();

            match self.entries.get(&candidate) {
                Some(Entry::Link(target)) => {
                    if visited.contains(&candidate) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    visited.insert(candidate.clone());
                    expansions += 1;

                    let mut target_components = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        current[..current.len() - 1].to_vec()
                    };
                    target_components.extend(components(target));
                    target_components.extend(pending.into_iter());
                    pending = target_components;
                    current.clear();
                }
                Some(Entry::Dir) => {
                    if is_final {
                        return Ok(candidate);
                    }
                }
                Some(Entry::File(_)) => {
                    if is_final {
                        return Ok(candidate);
                    }
                    return Err(Error::NotDir(candidate));
                }
                None => {
                    if is_final && allow_missing_final {
                        return Ok(candidate);
                    }
                    return Err(Error::Missing(candidate));
                }
            }
        }

        let resolved = path_from_components(&current);
        if !self.entries.contains_key(&resolved) {
            if allow_missing_final {
                return Ok(resolved);
            }
            return Err(Error::Missing(resolved));
        }
        Ok(resolved)
    }
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(index) = path.as_bytes().iter().position(|b| *b == 0) {
        return Err(Error::InvalidPath(index));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn normalize_path(path: &str, cwd: &str) -> Result<String, Error> {
    validate_path(path)?;
    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        components(cwd).collect()
    };

    for component in path.split('/') {
        match component {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other.to_string()),
        }
    }
    Ok(path_from_components(&parts))
}

fn components(path: &str) -> impl Iterator<Item = String> + '_ {
    path.split('/')
        .filter(|part| !part.is_empty() && *part != ".")
        .filter_map(|part| {
            if part == ".." {
                None
            } else {
                Some(part.to_string())
            }
        })
}

fn path_from_components(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn split_parent(path: &str) -> (String, String) {
    if path == "/" {
        return ("/".to_string(), String::new());
    }
    match path.rfind('/') {
        Some(0) => ("/".to_string(), path[1..].to_string()),
        Some(index) => (path[..index].to_string(), path[index + 1..].to_string()),
        None => ("/".to_string(), path.to_string()),
    }
}

fn join_component(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{name}")
    } else if name.is_empty() {
        parent.to_string()
    } else {
        format!("{parent}/{name}")
    }
}

fn is_within(path: &str, parent: &str) -> bool {
    path == parent || is_strictly_within(path, parent)
}

fn is_strictly_within(path: &str, parent: &str) -> bool {
    if parent == "/" {
        path.starts_with('/') && path != "/"
    } else {
        path.strip_prefix(parent)
            .is_some_and(|suffix| suffix.starts_with('/'))
    }
}

fn validate_target(target: &str) -> Result<(), Error> {
    if target.is_empty() {
        return Err(Error::InvalidTarget(0));
    }
    if let Some(index) = target.as_bytes().iter().position(|b| *b == 0) {
        return Err(Error::InvalidTarget(index));
    }
    Ok(())
}

fn directory_total(fs: &Fs, path: &str) -> u64 {
    fs.entries
        .iter()
        .filter_map(|(candidate, entry)| {
            if !is_strictly_within(candidate, path) {
                return None;
            }
            match entry {
                Entry::File(data) => Some(data.len() as u64),
                Entry::Link(target) => Some(target.len() as u64),
                Entry::Dir => None,
            }
        })
        .sum()
}
```
