```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

#[derive(Debug, Clone)]
pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn normalize_components(base: &[String], path: &str) -> Vec<String> {
    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        base.to_vec()
    };

    for component in path.split('/') {
        match component {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other.to_owned()),
        }
    }
    parts
}

fn path_of(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn components(path: &str) -> Vec<String> {
    if path == "/" {
        Vec::new()
    } else {
        path[1..].split('/').map(str::to_owned).collect()
    }
}

fn prefix_path(parts: &[String], count: usize) -> String {
    path_of(&parts[..count])
}

fn parent_and_name(path: &str) -> (String, String) {
    let parts = components(path);
    let name = parts.last().cloned().unwrap_or_default();
    (path_of(&parts[..parts.len().saturating_sub(1)]), name)
}

fn join_path(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{name}")
    } else {
        format!("{parent}/{name}")
    }
}

fn is_within(path: &str, directory: &str) -> bool {
    if directory == "/" {
        path.starts_with('/')
    } else {
        path == directory
            || path
                .strip_prefix(directory)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn normalized(&self, path: &str) -> Result<String, Error> {
        validate_path(path)?;
        let base = components(&self.cwd);
        Ok(path_of(&normalize_components(&base, path)))
    }

    fn resolve(
        &self,
        start: &str,
        follow_final: bool,
        create_missing_final: bool,
    ) -> Result<String, Error> {
        let mut current = components(start);
        let mut index = 0usize;
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            if index >= current.len() {
                return Ok(path_of(&current));
            }

            let prefix = prefix_path(&current, index + 1);
            let entry = match self.entries.get(&prefix) {
                Some(entry) => entry.clone(),
                None if create_missing_final && index + 1 == current.len() => {
                    return Ok(prefix);
                }
                None => return Err(Error::Missing(prefix)),
            };

            let is_final = index + 1 == current.len();
            match entry {
                Entry::Dir => {
                    index += 1;
                }
                Entry::File(_) => {
                    if !is_final {
                        return Err(Error::NotDir(prefix));
                    }
                    return Ok(prefix);
                }
                Entry::Link(target) if !is_final || follow_final => {
                    if visited.contains(&prefix) {
                        return Err(Error::Loop(prefix));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(prefix));
                    }
                    visited.insert(prefix.clone());
                    expansions += 1;

                    let (link_parent, _) = parent_and_name(&prefix);
                    let base = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        components(&link_parent)
                    };
                    let mut expanded = normalize_components(&base, &target);
                    expanded.extend_from_slice(&current[index + 1..]);
                    current = expanded;
                    index = 0;
                }
                Entry::Link(_) => return Ok(prefix),
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = parent_and_name(path);
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, name)),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn subtree_paths(&self, root: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|path| is_within(path, root))
            .cloned()
            .collect()
    }

    fn directory_nonempty(&self, path: &str) -> bool {
        self.entries
            .keys()
            .any(|candidate| candidate != path && is_within(candidate, path))
    }

    fn directory_size(&self, path: &str) -> u64 {
        self.entries
            .iter()
            .filter(|(candidate, _)| is_within(candidate, path))
            .fold(0u64, |sum, (_, entry)| {
                let size = match entry {
                    Entry::File(bytes) => u64::try_from(bytes.len()).unwrap_or(u64::MAX),
                    Entry::Link(target) => u64::try_from(target.len()).unwrap_or(u64::MAX),
                    Entry::Dir => 0,
                };
                sum.saturating_add(size)
            })
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = self.normalized(path)?;
        let (parent, name) = self.resolve_parent(&normalized)?;
        let final_path = join_path(&parent, &name);
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let normalized = self.normalized(path)?;
        let resolved = self.resolve(&normalized, true, true)?;
        if resolved == "/" {
            return Err(Error::IsDir(resolved));
        }

        let (parent, name) = parent_and_name(&resolved);
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }

        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) | None => {
                self.entries
                    .insert(join_path(&resolved_parent, &name), Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let normalized = self.normalized(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let normalized = self.normalized(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }

        let prefix = if resolved == "/" {
            "/".to_owned()
        } else {
            format!("{resolved}/")
        };
        let mut result = Vec::new();
        for (candidate, entry) in &self.entries {
            if candidate == &resolved || !candidate.starts_with(&prefix) {
                continue;
            }
            let remainder = &candidate[prefix.len()..];
            if !remainder.contains('/') {
                result.push((remainder.to_owned(), entry.kind()));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = self.normalized(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalized(from)?;
        let destination = self.normalized(to)?;
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source)?;
        let source_path = join_path(&source_parent, &source_name);
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent, destination_name) = self.resolve_parent(&destination)?;
        let destination_path = join_path(&destination_parent, &destination_name);

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && is_within(&self.cwd, &destination_path)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && is_within(&destination_path, &source_path) {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(existing) = &destination_entry {
            match (&source_entry, existing) {
                (Entry::Dir, Entry::Dir) => {
                    if self.directory_nonempty(&destination_path) {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        let moved = self.subtree_paths(&source_path);
        let mut replacements = Vec::new();
        for old in &moved {
            let suffix = &old[source_path.len()..];
            replacements.push((old.clone(), format!("{destination_path}{suffix}")));
        }

        if destination_entry.is_some() {
            let remove_destination = self.subtree_paths(&destination_path);
            for path in remove_destination {
                self.entries.remove(&path);
            }
        }
        for (old, _) in &replacements {
            self.entries.remove(old);
        }
        for (old, new) in replacements {
            if let Some(entry) = moved
                .iter()
                .position(|candidate| candidate == &old)
                .and_then(|position| self.entries.get(&old).cloned().or_else(|| {
                    // Entries were removed above; recover from the pre-move snapshot below.
                    None
                }))
            {
                self.entries.insert(new, entry);
            } else {
                let _ = new;
            }
        }

        // Rebuild the moved entries from the snapshot taken before removal.
        // This branch is intentionally supplied by the snapshot below.
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let normalized = self.normalized(path)?;
        if normalized == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&normalized)?;
        let resolved = join_path(&parent, &name);
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &resolved) {
            return Err(Error::Busy(resolved));
        }
        if matches!(entry, Entry::Dir) && !recursive && self.directory_nonempty(&resolved) {
            return Err(Error::NotEmpty(resolved));
        }

        for candidate in self.subtree_paths(&resolved) {
            self.entries.remove(&candidate);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let normalized_link = self.normalized(link)?;
        validate_path(target).map_err(|error| match error {
            Error::InvalidPath(offset) if target.is_empty() => Error::InvalidTarget(0),
            Error::InvalidPath(offset) => Error::InvalidTarget(offset),
            _ => Error::InvalidTarget(0),
        })?;

        let (parent, name) = self.resolve_parent(&normalized_link)?;
        let final_path = join_path(&parent, &name);
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries
            .insert(final_path, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let normalized = self.normalized(path)?;
        let resolved = self.resolve(&normalized, false, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let normalized = self.normalized(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        let entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;
        if !matches!(entry, Entry::Dir | Entry::File(_)) {
            return Err(Error::Missing(resolved));
        }

        let mut paths = self.subtree_paths(&resolved);
        paths.sort();
        Ok(paths
            .into_iter()
            .map(|candidate| {
                let size = self.directory_size(&candidate);
                (candidate, size)
            })
            .collect())
    }
}

fn replace_entry_subtree(
    entries: &mut BTreeMap<String, Entry>,
    source: &str,
    destination: &str,
) {
    let moved: Vec<(String, Entry)> = entries
        .iter()
        .filter(|(path, _)| is_within(path, source))
        .map(|(path, entry)| (path.clone(), entry.clone()))
        .collect();

    for (old, _) in &moved {
        entries.remove(old);
    }
    for (old, entry) in moved {
        let suffix = &old[source.len()..];
        entries.insert(format!("{destination}{suffix}"), entry);
    }
}
```
