fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort();
    out.dedup();
    out
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| {
        a.constraint
            .cmp(&b.constraint)
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn append_evidence(text: &mut String, evidence: &[Evidence]) {
    for item in evidence_sorted(evidence) {
        text.push_str("\n  C:");
        text.push_str(&constraint_text(item.constraint));
        text.push_str(" via ");
        text.push_str(&item.path.join(" -> "));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut text = format!("missing|package={}", encode(package));
            append_evidence(&mut text, evidence);
            text
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut text = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            append_evidence(&mut text, evidence);
            text
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut text = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            append_evidence(&mut text, evidence);
            text
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn permits(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn platform_active(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

        .map(|(i, p)| (p.name.clone(), i))
        .collect()
}

use std::collections::{BTreeMap, BTreeSet};

#[derive(Clone, Default)]
struct Active {
    constraints: BTreeMap<String, Vec<Constraint>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    incoming: BTreeMap<String, Vec<Evidence>>,
    paths: BTreeMap<String, Vec<String>>,
    activated: BTreeMap<String, BTreeSet<String>>,
    missing_features: Vec<(String, Version, String, Vec<String>)>,
}

fn root_token(i: usize, r: &Requirement) -> String {
    let mut features = r.features.clone();
    features.sort();
    features.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        features.join(",")
    )
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    let mut features = dep.features.clone();
    features.sort();
    features.dedup();
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        features.join(","),
        gate,
        platform
    )
}

fn compute_active(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
    indices: &BTreeMap<String, usize>,
) -> Active {
    let mut active = Active::default();

    for (i, root) in roots.iter().enumerate() {
        active
            .constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        active
            .requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        active
            .incoming
            .entry(root.package.clone())
            .or_default()
            .push(Evidence {
                constraint: root.constraint,
                path: vec![root_token(i, root)],
            });
    }

    loop {
        let old_constraints = active.constraints.clone();
        let old_requests = active.requests.clone();
        let old_incoming = active.incoming.clone();

        let names: Vec<String> = selected.keys().cloned().collect();
        for source in names {
            let version = selected[&source];
            let Some(&pi) = indices.get(&source) else {
                continue;
            };
            let Some(release) = catalog[pi]
                .releases
                .iter()
                .find(|r| r.version == version)
            else {
                continue;
            };

            let mut activated = BTreeSet::new();
            let mut queue: Vec<String> = active
                .requests
                .get(&source)
                .cloned()
                .unwrap_or_default()
                .into_iter()
                .collect();
            while let Some(feature_name) = queue.pop() {
                if !activated.insert(feature_name.clone()) {
                    continue;
                }
                if let Some(declaration) = release
                    .features
                    .iter()
                    .find(|f| f.name == feature_name)
                {
                    for enabled in &declaration.enables {
                        queue.push(enabled.clone());
                    }
                }
            }
            active.activated.insert(source.clone(), activated);
        }

        let names: Vec<String> = selected.keys().cloned().collect();
        for source in names {
            let version = selected[&source];
            let Some(&pi) = indices.get(&source) else {
                continue;
            };
            let Some(release) = catalog[pi]
                .releases
                .iter()
                .find(|r| r.version == version)
            else {
                continue;
            };
            let active_features = active.activated.get(&source).cloned().unwrap_or_default();
            for (di, dep) in release.deps.iter().enumerate() {
                if !platform_active(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !active_features.contains(g))
                {
                    continue;
                }
                let token = dep_token(&source, version, di, dep);
                let path = active
                    .paths
                    .get(&source)
                    .cloned()
                    .unwrap_or_default();
                let mut dep_path = path;
                dep_path.push(token);

                active
                    .constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                active
                    .requests
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());
                active
                    .incoming
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Evidence {
                        constraint: dep.constraint,
                        path: dep_path.clone(),
                    });
                let replace = match active.paths.get(&dep.package) {
                    None => true,
                    Some(old) => {
                        (dep_path.len(), &dep_path) < (old.len(), old)
                    }
                };
                if replace {
                    active.paths.insert(dep.package.clone(), dep_path);
                }
            }
        }

        let mut best: BTreeMap<String, Vec<String>> = BTreeMap::new();
        for (i, root) in roots.iter().enumerate() {
            let path = vec![root_token(i, root)];
            let replace = match best.get(&root.package) {
                None => true,
                Some(old) => (path.len(), &path) < (old.len(), old),
            };
            if replace {
                best.insert(root.package.clone(), path);
            }
        }
        for (source, version) in selected {
            let Some(&pi) = indices.get(source) else {
                continue;
            };
            let Some(release) = catalog[pi].releases.iter().find(|r| r.version == *version) else {
                continue;
            };
            let Some(source_path) = best.get(source).cloned() else {
                continue;
            };
            let active_features = active.activated.get(source).cloned().unwrap_or_default();
            for (di, dep) in release.deps.iter().enumerate() {
                if !platform_active(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !active_features.contains(g))
                {
                    continue;
                }
                let mut path = source_path.clone();
                path.push(dep_token(source, *version, di, dep));
                let replace = match best.get(&dep.package) {
                    None => true,
                    Some(old) => (path.len(), &path) < (old.len(), old),
                };
                if replace {
                    best.insert(dep.package.clone(), path);
                }
            }
        }

        active.paths = best;
        if active.constraints == old_constraints
            && active.requests == old_requests
            && active.incoming == old_incoming
        {
            break;
        }
    }

    active.missing_features.clear();
    for (package, version) in selected {
        let Some(&pi) = indices.get(package) else {
            continue;
        };
        let Some(release) = catalog[pi].releases.iter().find(|r| r.version == *version) else {
            continue;
        };
        let features = active.activated.get(package).cloned().unwrap_or_default();
        for feature in features {
            if !release.features.iter().any(|f| f.name == feature) {
                active.missing_features.push((
                    package.clone(),
                    *version,
                    feature,
                    active.paths.get(package).cloned().unwrap_or_default(),
                ));
            }
        }
    }
    active
}

fn canonical_active(active: &Active) -> BTreeMap<String, Vec<Constraint>> {
    active
        .constraints
        .iter()
        .map(|(name, cs)| (name.clone(), canonical_constraints(cs)))
        .collect()
}

fn dynamic_error(
    catalog: &[Package],
    selected: &BTreeMap<String, Version>,
    active: &Active,
    indices: &BTreeMap<String, usize>,
) -> Option<ResolveError> {
    let missing = active
        .constraints
        .keys()
        .filter(|name| !indices.contains_key(*name))
        .next()
        .cloned();
    if let Some(package) = missing {
        return Some(ResolveError::Missing {
            evidence: active.incoming.get(&package).cloned().unwrap_or_default(),
            package,
        });
    }

    let mut missing_features = active.missing_features.clone();
    missing_features.sort_by(|a, b| {
        a.0.cmp(&b.0)
            .then_with(|| a.2.cmp(&b.2))
    });
    if let Some((package, selected_version, feature, path)) = missing_features.into_iter().next() {
        return Some(ResolveError::MissingFeature {
            package,
            selected: selected_version,
            feature,
            path,
        });
    }

    for (package, version) in selected {
        let constraints = active.constraints.get(package).cloned().unwrap_or_default();
        if constraints.iter().any(|c| !permits(*c, *version)) {
            return Some(ResolveError::Conflict {
                package: package.clone(),
                selected: *version,
                requirements: canonical_constraints(&constraints),
                evidence: active.incoming.get(package).cloned().unwrap_or_default(),
            });
        }
    }

    let all_constraints = canonical_active(active);
    for (package, constraints) in &all_constraints {
        if selected.contains_key(package) {
            continue;
        }
        let Some(&pi) = indices.get(package) else {
            continue;
        };
        let any = catalog[pi].releases.iter().any(|r| {
            constraints.iter().all(|c| permits(*c, r.version))
        });
        if !any {
            return Some(ResolveError::NoMatch {
                package: package.clone(),
                requirements: constraints.clone(),
                evidence: active.incoming.get(package).cloned().unwrap_or_default(),
            });
        }
    }
    None
}

#[derive(Default)]
struct SearchState {
    attempts: usize,
    first_failure: Option<ResolveError>,
    best: Option<(usize, Vec<Version>, BTreeMap<String, Version>, Active)>,
    limit: bool,
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
    indices: &BTreeMap<String, usize>,
    all_names: &[String],
    selected: &mut BTreeMap<String, Version>,
    state: &mut SearchState,
) {
    if state.limit {
        return;
    }

    let active = compute_active(catalog, roots, selected, target, indices);
    if let Some(error) = dynamic_error(catalog, selected, &active, indices) {
        if state.first_failure.is_none() {
            state.first_failure = Some(error);
        }
        return;
    }

    let next = active
        .constraints
        .keys()
        .find(|name| !selected.contains_key(*name))
        .cloned();
    let Some(package) = next else {
        let lock_map: BTreeMap<&str, Version> = locks
            .iter()
            .map(|l| (l.package.as_str(), l.version))
            .collect();
        let cost = lock_map
            .iter()
            .filter(|(p, v)| selected.get(**p) != Some(**v))
            .count();
        let vector: Vec<Version> = all_names
            .iter()
            .filter_map(|name| selected.get(name).copied())
            .collect();
        let full_vector: Vec<(bool, Version)> = all_names
            .iter()
            .map(|name| selected.get(name).map(|v| (true, *v)).unwrap_or((false, Version { major: 0, minor: 0, patch: 0 })))
            .collect();

        let should_replace = match &state.best {
            None => true,
            Some((old_cost, _, old_selected, _)) => {
                cost < *old_cost
                    || (cost == *old_cost
                        && all_names.iter().find_map(|name| {
                            let a = selected.get(name);
                            let b = old_selected.get(name);
                            if a == b {
                                None
                            } else {
                                Some(match (a, b) {
                                    (Some(a), Some(b)) => a > b,
                                    (Some(_), None) => true,
                                    (None, Some(_)) => false,
                                    (None, None) => false,
                                })
                            }
                        }).unwrap_or(false))
            }
        };
        if should_replace {
            let _ = full_vector;
            state.best = Some((cost, vector, selected.clone(), active));
        }
        return;
    };

    let constraints = canonical_constraints(active.constraints.get(&package).map(Vec::as_slice).unwrap_or(&[]));
    let pi = indices[&package];
    let mut candidates: Vec<Version> = catalog[pi]
        .releases
        .iter()
        .filter(|r| constraints.iter().all(|c| permits(*c, r.version)))
        .map(|r| r.version)
        .collect();
    candidates.sort();
    candidates.dedup();
    candidates.reverse();

    if let Some(lock) = locks.iter().find(|l| l.package == package) {
        if let Some(position) = candidates.iter().position(|v| *v == lock.version) {
            let v = candidates.remove(position);
            candidates.insert(0, v);
        }
    }

    for version in candidates {
        if state.attempts == max_attempts {
            state.limit = true;
            return;
        }
        state.attempts += 1;
        selected.insert(package.clone(), version);
        search(
            catalog,
            roots,
            locks,
            target,
            max_attempts,
            indices,
            all_names,
            selected,
            state,
        );
        selected.remove(&package);
        if state.limit {
            return;
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.gate", base),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", base, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].feature[{}]", i, j, f);
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.enable[{}]", base, e),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen_features = BTreeSet::new();
            for feature in &release.features {
                if !seen_features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let indices = package_index(catalog);
    let mut all_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    all_names.sort();

    let mut state = SearchState::default();
    search(
        catalog,
        roots,
        locks,
        target,
        max_attempts,
        &indices,
        &all_names,
        &mut BTreeMap::new(),
        &mut state,
    );

    if state.limit {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }

    let Some((_, _, selected, final_active)) = state.best else {
        return Err(state.first_failure.unwrap_or_else(|| ResolveError::NoMatch {
            package: roots
                .first()
                .map(|r| r.package.clone())
                .unwrap_or_default(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_vec: Vec<Selection> = selected
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut before: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();
    let mut after = selected.clone();
    let names: BTreeSet<String> = before
        .keys()
        .chain(after.keys())
        .cloned()
        .collect();
    let mut changes = Vec::new();
    for package in names {
        let b = before.remove(&package);
        let a = after.remove(&package);
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = indices
                    .get(&package)
                    .is_some_and(|pi| catalog[*pi].releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let constraints = final_active
                        .constraints
                        .get(&package)
                        .cloned()
                        .unwrap_or_default();
                    if constraints.iter().any(|c| !permits(*c, old)) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
