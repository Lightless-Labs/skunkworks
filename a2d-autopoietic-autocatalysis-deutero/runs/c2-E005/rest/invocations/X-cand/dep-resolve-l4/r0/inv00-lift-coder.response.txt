fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty()
        || bytes.len() > 24
        || !bytes[0].is_ascii_lowercase()
        || bytes[bytes.len() - 1] == b'-'
    {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort_by_key(|c| constraint_rank(*c));
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(mut evidence: Vec<Evidence>) -> Vec<Evidence> {
    evidence.sort_by(|a, b| {
        constraint_rank(a.constraint)
            .cmp(&constraint_rank(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    evidence
}

fn error_text_with_evidence(first: String, evidence: &[Evidence]) -> String {
    let sorted = evidence_sorted(evidence.to_vec());
    let mut out = first;
    for item in sorted {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(item.constraint));
        out.push_str(" via ");
        out.push_str(&item.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => error_text_with_evidence(
            format!("missing|package={}", encode(package)),
            evidence,
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => error_text_with_evidence(
            format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => error_text_with_evidence(
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn satisfies(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn target_matches(platform: Platform, target: Target) -> bool {
    matches!(platform, Platform::Any)
        || matches!(
            (platform, target),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn sorted_features(features: &[String]) -> Vec<String> {
    let mut result = features.to_vec();
    result.sort();
    result.dedup();
    result
}

fn root_token(root: &Requirement, index: usize) -> String {
    let features = sorted_features(&root.features).join(",");
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        root.package,
        constraint_text(root.constraint),
        features
    )
}

fn dependency_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    let features = sorted_features(&dep.features).join(",");
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        features,
        gate,
        platform
    )
}

fn better_path(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn add_feature(info: &mut PackageInfo, feature: &str, path: &[String]) -> bool {
    if !info.features.iter().any(|f| f == feature) {
        info.features.push(feature.to_string());
        info.features.sort();
        info.feature_paths.insert(feature.to_string(), path.to_vec());
        true
    } else {
        if let Some(existing) = info.feature_paths.get_mut(feature) {
            if better_path(path, existing) {
                *existing = path.to_vec();
            }
        }
        false
    }
}

fn add_origin(info: &mut PackageInfo, constraint: Constraint, path: Vec<String>) {
    info.origins.push(Origin { constraint, path });
}

fn build_node(
    catalog: &[Package],
    roots: &[Requirement],
    assigned: &std::collections::BTreeMap<String, Version>,
    target: Target,
) -> NodeInfo {
    let by_name: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut node = NodeInfo {
        packages: std::collections::BTreeMap::new(),
        active_deps: std::collections::BTreeMap::new(),
    };

    for (i, root) in roots.iter().enumerate() {
        let info = node.packages.entry(root.package.clone()).or_default();
        add_origin(info, root.constraint, vec![root_token(root, i)]);
        for feature in &root.features {
            let path = vec![root_token(root, i)];
            add_feature(info, feature, &path);
        }
    }

    loop {
        let mut changed = false;
        let names: Vec<String> = node.packages.keys().cloned().collect();
        for name in names {
            let Some(version) = assigned.get(&name).copied() else {
                continue;
            };
            let Some(package) = by_name.get(name.as_str()).copied() else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let declared: std::collections::BTreeMap<&str, &Feature> =
                release.features.iter().map(|f| (f.name.as_str(), f)).collect();

            let enabled_now = node.packages[&name].features.clone();
            for feature_name in enabled_now {
                if let Some(feature) = declared.get(feature_name.as_str()) {
                    let path = node.packages[&name]
                        .feature_paths
                        .get(&feature_name)
                        .cloned()
                        .unwrap_or_default();
                    for enabled in &feature.enables {
                        let info = node.packages.get_mut(&name).unwrap();
                        if add_feature(info, enabled, &path) {
                            changed = true;
                        }
                    }
                }
            }

            for (dep_index, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target) {
                    continue;
                }
                if dep
                    .gate
                    .as_ref()
                    .is_some_and(|gate| !node.packages[&name].features.contains(gate))
                {
                    continue;
                }
                let source_path = node.packages[&name]
                    .origins
                    .iter()
                    .map(|o| o.path.clone())
                    .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.cmp(b)))
                    .unwrap_or_default();
                let mut path = source_path;
                path.push(dependency_token(&name, version, dep_index, dep));

                let dest = node.packages.entry(dep.package.clone()).or_default();
                if !node
                    .active_deps
                    .entry(name.clone())
                    .or_default()
                    .contains(&dep_index)
                {
                    node.active_deps
                        .entry(name.clone())
                        .or_default()
                        .push(dep_index);
                    changed = true;
                }
                let before = dest.origins.len();
                add_origin(dest, dep.constraint, path.clone());
                if dest.origins.len() != before {
                    changed = true;
                }
                for feature in &dep.features {
                    if add_feature(dest, feature, &path) {
                        changed = true;
                    }
                }
            }
        }
        if !changed {
            break;
        }
    }

    node
}

fn canonical_path(info: &PackageInfo) -> Vec<String> {
    info.origins
        .iter()
        .map(|o| o.path.clone())
        .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.cmp(b)))
        .unwrap_or_default()
}

fn package_evidence(info: &PackageInfo) -> Vec<Evidence> {
    evidence_sorted(
        info.origins
            .iter()
            .map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            })
            .collect(),
    )
}

fn package_constraints(info: &PackageInfo) -> Vec<Constraint> {
    canonical_constraints(info.origins.iter().map(|o| o.constraint).collect())
}

fn release_matches(release: &Release, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| satisfies(release.version, *constraint))
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].gate",
                                i, j, k
                            ),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen_features = std::collections::BTreeSet::new();
            for feature in &release.features {
                if !seen_features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    Ok(())
}

#[derive(Clone)]
struct SearchResult {
    selected: std::collections::BTreeMap<String, Version>,
    constraints: std::collections::BTreeMap<String, Vec<Constraint>>,
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<SearchResult>,
    best_cost: usize,
}

impl<'a> Search<'a> {
    fn visit(&mut self, assigned: &mut std::collections::BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let node = build_node(self.catalog, self.roots, assigned, self.target);
        let packages_by_name: std::collections::BTreeMap<&str, &Package> =
            self.catalog.iter().map(|p| (p.name.as_str(), p)).collect();

        for package_name in node.packages.keys() {
            if !packages_by_name.contains_key(package_name.as_str()) {
                let info = &node.packages[package_name];
                return self.record_error(ResolveError::Missing {
                    package: package_name.clone(),
                    evidence: package_evidence(info),
                });
            }
        }

        for (package_name, version) in assigned.iter() {
            let Some(info) = node.packages.get(package_name) else {
                continue;
            };
            let Some(package) = packages_by_name.get(package_name.as_str()) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
                continue;
            };
            let declared: std::collections::BTreeSet<&str> =
                release.features.iter().map(|f| f.name.as_str()).collect();
            for feature in &info.features {
                if !declared.contains(feature.as_str()) {
                    return self.record_error(ResolveError::MissingFeature {
                        package: package_name.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: canonical_path(info),
                    });
                }
            }
        }

        for (package_name, version) in assigned.iter() {
            if let Some(info) = node.packages.get(package_name) {
                let constraints = package_constraints(info);
                if !constraints.iter().all(|c| satisfies(*version, *c)) {
                    return self.record_error(ResolveError::Conflict {
                        package: package_name.clone(),
                        selected: *version,
                        requirements: constraints,
                        evidence: package_evidence(info),
                    });
                }
            }
        }

        for (package_name, info) in &node.packages {
            if assigned.contains_key(package_name) {
                continue;
            }
            let package = packages_by_name[package_name.as_str()];
            let constraints = package_constraints(info);
            let candidates = package
                .releases
                .iter()
                .filter(|r| release_matches(r, &constraints))
                .count();
            if candidates == 0 {
                return self.record_error(ResolveError::NoMatch {
                    package: package_name.clone(),
                    requirements: constraints,
                    evidence: package_evidence(info),
                });
            }
        }

        let next = node
            .packages
            .keys()
            .find(|name| !assigned.contains_key(*name))
            .cloned();

        let Some(package_name) = next else {
            let constraints = node
                .packages
                .iter()
                .map(|(name, info)| (name.clone(), package_constraints(info)))
                .collect();
            let cost = self
                .locks
                .iter()
                .filter(|lock| assigned.get(&lock.package) != Some(&lock.version))
                .count();
            let result = SearchResult {
                selected: assigned.clone(),
                constraints,
            };
            let replace = match &self.best {
                None => true,
                Some(best) => {
                    cost < self.best_cost
                        || (cost == self.best_cost
                            && self.vector_better(&result.selected, &best.selected))
                }
            };
            if replace {
                self.best = Some(result);
                self.best_cost = cost;
            }
            return Ok(());
        };

        let info = &node.packages[&package_name];
        let constraints = package_constraints(info);
        let package = packages_by_name[package_name.as_str()];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .filter(|r| release_matches(r, &constraints))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.iter().find(|l| l.package == package_name) {
            if let Some(index) = candidates.iter().position(|v| *v == lock.version) {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assigned.insert(package_name.clone(), candidate);
            let result = self.visit(assigned);
            assigned.remove(&package_name);
            if let Err(ResolveError::Limit { attempts }) = result {
                return Err(ResolveError::Limit { attempts });
            }
        }

        Ok(())
    }

    fn record_error<T>(&mut self, error: ResolveError) -> Result<T, ResolveError> {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
        Err(self.first_error.clone().unwrap())
    }

    fn vector_better(
        &self,
        a: &std::collections::BTreeMap<String, Version>,
        b: &std::collections::BTreeMap<String, Version>,
    ) -> bool {
        let mut names: Vec<String> = self.catalog.iter().map(|p| p.name.clone()).collect();
        names.sort();
        for name in names {
            match (a.get(&name), b.get(&name)) {
                (Some(av), Some(bv)) if av != bv => return av > bv,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                _ => {}
            }
        }
        false
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let mut search = Search {
        catalog,
        roots,
        locks,
        target,
        max_attempts,
        attempts: 0,
        first_error: None,
        best: None,
        best_cost: usize::MAX,
    };
    let mut assigned = std::collections::BTreeMap::new();
    search.visit(&mut assigned)?;

    let Some(best) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = best
        .selected
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let before: std::collections::BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();
    let after = &best.selected;
    let mut all_names: std::collections::BTreeSet<String> = before.keys().cloned().collect();
    all_names.extend(after.keys().cloned());

    let mut changes = Vec::new();
    for package in all_names {
        let b = before.get(&package).copied();
        let a = after.get(&package).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if best
                    .constraints
                    .get(&package)
                    .is_some_and(|cs| !cs.iter().all(|c| satisfies(old, *c)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
