fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort_by_key(|c| constraint_rank(*c));
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn all_match(cs: &[Constraint], v: Version) -> bool {
    cs.iter().all(|c| matches(*c, v))
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_cmp(a: &Evidence, b: &Evidence) -> std::cmp::Ordering {
    constraint_rank(a.constraint)
        .cmp(&constraint_rank(b.constraint))
        .then_with(|| a.path.cmp(&b.path))
}

fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            format_evidence_lines(format!("missing|package={}", encode(package)), evidence)
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format_evidence_lines(
            format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format_evidence_lines(
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn format_evidence_lines(mut first: String, evidence: &[Evidence]) -> String {
    let mut sorted = evidence.to_vec();
    sorted.sort_by(evidence_cmp);
    for e in sorted {
        first.push_str("\n  C:");
        first.push_str(&constraint_text(e.constraint));
        first.push_str(" via ");
        first.push_str(&e.path.join(" -> "));
    }
    first
}

use std::collections::{BTreeMap, BTreeSet};

fn feature_decl<'a>(release: &'a Release, name: &str) -> Option<&'a Feature> {
    release.features.iter().find(|f| f.name == name)
}

fn target_matches(platform: Platform, target: Target) -> bool {
    matches!(
        (platform, target),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn root_token(i: usize, r: &Requirement) -> String {
    let fs = sorted_names(&r.features).join(",");
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        fs
    )
}

fn sorted_names(names: &[String]) -> Vec<String> {
    let mut xs = names.to_vec();
    xs.sort();
    xs.dedup();
    xs
}

fn dep_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    let fs = sorted_names(&dep.features).join(",");
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        fs,
        gate,
        platform
    )
}

fn better_path(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn append_token(path: &[String], token: String) -> Vec<String> {
    let mut p = path.to_vec();
    p.push(token);
    p
}

fn compute_active(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> (BTreeMap<String, Active>, BTreeMap<String, Vec<String>>) {
    let mut active = BTreeMap::<String, Active>::new();
    let mut paths = BTreeMap::<String, Vec<String>>::new();

    for (i, root) in roots.iter().enumerate() {
        let entry = active.entry(root.package.clone()).or_default();
        entry.constraints.push(root.constraint);
        entry.features.extend(root.features.iter().cloned());
        let p = vec![root_token(i, root)];
        match paths.get(&root.package) {
            None => {
                paths.insert(root.package.clone(), p);
            }
            Some(old) if better_path(&p, old) => {
                paths.insert(root.package.clone(), p);
            }
            _ => {}
        }
    }

    loop {
        let mut changed = false;
        let active_names: Vec<String> = active.keys().cloned().collect();

        for package_name in active_names {
            let Some(version) = selected.get(&package_name).copied() else {
                continue;
            };
            let Some(package) = catalog.get(&package_name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested: Vec<String> = active
                .get(&package_name)
                .map(|a| a.features.iter().cloned().collect())
                .unwrap_or_default();

            for feature_name in requested {
                if let Some(feature) = feature_decl(release, &feature_name) {
                    for enabled in &feature.enables {
                        let entry = active.entry(package_name.clone()).or_default();
                        if entry.features.insert(enabled.clone()) {
                            changed = true;
                        }
                    }
                }
            }

            let current_features = active
                .get(&package_name)
                .map(|a| a.features.clone())
                .unwrap_or_default();

            for (dep_index, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !current_features.contains(gate))
                {
                    continue;
                }

                let source_path = paths
                    .get(&package_name)
                    .cloned()
                    .unwrap_or_default();
                let token = dep_token(&package_name, version, dep_index, dep);
                let candidate_path = append_token(&source_path, token);
                match paths.get(&dep.package) {
                    None => {
                        paths.insert(dep.package.clone(), candidate_path.clone());
                        changed = true;
                    }
                    Some(old) if better_path(&candidate_path, old) => {
                        paths.insert(dep.package.clone(), candidate_path.clone());
                        changed = true;
                    }
                    _ => {}
                }

                let entry = active.entry(dep.package.clone()).or_default();
                entry.constraints.push(dep.constraint);
                entry.features.extend(dep.features.iter().cloned());
                entry.origins.push(Origin {
                    constraint: dep.constraint,
                    path: candidate_path,
                });
            }
        }

        if !changed {
            break;
        }
    }

    for (i, root) in roots.iter().enumerate() {
        active
            .entry(root.package.clone())
            .or_default()
            .origins
            .push(Origin {
                constraint: root.constraint,
                path: vec![root_token(i, root)],
            });
    }

    for item in active.values_mut() {
        item.constraints = canonical_constraints(std::mem::take(&mut item.constraints));
    }

    (active, paths)
}

struct Search<'a> {
    catalog: BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>, BTreeMap<String, Active>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn search(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let (active, paths) =
            compute_active(&self.catalog, self.roots, selected, self.target);

        let missing: Vec<String> = active
            .keys()
            .filter(|name| !self.catalog.contains_key(*name))
            .cloned()
            .collect();
        if let Some(package) = missing.into_iter().next() {
            let evidence = active
                .get(&package)
                .map(|a| evidence_from(a))
                .unwrap_or_default();
            return self.record_error(ResolveError::Missing { package, evidence });
        }

        for (package, version) in selected.iter() {
            let constraints = active
                .get(package)
                .map(|a| a.constraints.clone())
                .unwrap_or_default();
            if !all_match(&constraints, *version) {
                return self.record_error(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: constraints,
                    evidence: active
                        .get(package)
                        .map(|a| evidence_from(a))
                        .unwrap_or_default(),
                });
            }
        }

        for (package, state) in &active {
            if !selected.contains_key(package) {
                let p = self.catalog[package];
                let mut releases: Vec<&Release> = p
                    .releases
                    .iter()
                    .filter(|r| all_match(&state.constraints, r.version))
                    .collect();
                if releases.is_empty() {
                    return self.record_error(ResolveError::NoMatch {
                        package: package.clone(),
                        requirements: state.constraints.clone(),
                        evidence: evidence_from(state),
                    });
                }
            }
        }

        for (package, version) in selected.iter() {
            let release = self.catalog[package]
                .releases
                .iter()
                .find(|r| r.version == *version)
                .unwrap();
            let feature_names = active
                .get(package)
                .map(|a| a.features.iter().cloned().collect::<Vec<_>>())
                .unwrap_or_default();
            for feature in feature_names {
                if feature_decl(release, &feature).is_none() {
                    return self.record_error(ResolveError::MissingFeature {
                        package: package.clone(),
                        selected: *version,
                        feature,
                        path: paths.get(package).cloned().unwrap_or_default(),
                    });
                }
            }
        }

        let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
        let Some(package) = next else {
            let solution: Vec<Selection> = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            let cost = self
                .locks
                .iter()
                .filter(|(p, v)| {
                    selected.get(*p).copied() != Some(**v)
                })
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_solution, _)) => {
                    cost < *best_cost
                        || (cost == *best_cost
                            && solution_cmp(&solution, best_solution, &self.catalog)
                                == std::cmp::Ordering::Greater)
                }
            };
            if replace {
                self.best = Some((cost, solution, active));
            }
            return Ok(());
        };

        let state = &active[&package];
        let package_data = self.catalog[&package];
        let mut candidates: Vec<Version> = package_data
            .releases
            .iter()
            .filter(|r| all_match(&state.constraints, r.version))
            .map(|r| r.version)
            .collect();
        candidates.sort();
        candidates.dedup();
        candidates.reverse();

        if let Some(lock) = self.locks.get(&package) {
            if let Some(index) = candidates.iter().position(|v| v == lock) {
                let version = candidates.remove(index);
                candidates.insert(0, version);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            match self.search(selected) {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    selected.remove(&package);
                    return Err(ResolveError::Limit { attempts });
                }
                Err(err) => {
                    if self.first_error.is_none() {
                        self.first_error = Some(err);
                    }
                }
            }
            selected.remove(&package);
        }
        Ok(())
    }

    fn record_error<T>(&mut self, err: ResolveError) -> Result<T, ResolveError> {
        if self.first_error.is_none() {
            self.first_error = Some(err.clone());
        }
        Err(err)
    }
}

fn evidence_from(active: &Active) -> Vec<Evidence> {
    let mut out: Vec<Evidence> = active
        .origins
        .iter()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect();
    out.sort_by(evidence_cmp);
    out
}

fn solution_cmp(
    a: &[Selection],
    b: &[Selection],
    catalog: &BTreeMap<String, &Package>,
) -> std::cmp::Ordering {
    let names: Vec<&String> = catalog.keys().collect();
    for name in names {
        let av = a.iter().find(|s| &s.package == name).map(|s| s.version);
        let bv = b.iter().find(|s| &s.package == name).map(|s| s.version);
        let order = match (av, bv) {
            (None, None) => std::cmp::Ordering::Equal,
            (None, Some(_)) => std::cmp::Ordering::Less,
            (Some(_), None) => std::cmp::Ordering::Greater,
            (Some(x), Some(y)) => x.cmp(&y),
        };
        if order != std::cmp::Ordering::Equal {
            return order;
        }
    }
    std::cmp::Ordering::Equal
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.gate", base),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", base, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].feature[{}]", i, j, f);
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.enable[{}]", base, e),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let locks_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut search = Search {
        catalog: catalog_map,
        roots,
        locks: locks_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };

    let mut selected = BTreeMap::new();
    match search.search(&mut selected) {
        Err(ResolveError::Limit { attempts }) => {
            return Err(ResolveError::Limit { attempts });
        }
        _ => {}
    }

    let Some((_, selected, final_active)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_map: BTreeMap<String, Version> = selected
        .iter()
        .map(|s| (s.package.clone(), s.version))
        .collect();
    let mut all_names: BTreeSet<String> = locks_map.keys().cloned().collect();
    all_names.extend(selected_map.keys().cloned());

    let mut changes = Vec::new();
    for package in all_names {
        let before = locks_map.get(&package).copied();
        let after = selected_map.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !final_active
                    .get(&package)
                    .map(|a| all_match(&a.constraints, old))
                    .unwrap_or(true)
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
