#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
struct Record {
    job: Job,
    dependencies: Vec<usize>,
    state: State,
    remaining: Option<u64>,
    started_at: u64,
    order: usize,
}

pub struct Pool {
    records: Vec<Record>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    first_run: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job.tag.as_ref().is_some_and(|tag| !valid_tag(tag)) {
                return Err(Error::BadTag(i));
            }
            if ids.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut resolved = Vec::with_capacity(job.dependencies.len());
            for (j, spelling) in job.dependencies.iter().enumerate() {
                if !seen.insert(spelling.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(spelling) {
                    Some(&idx) => resolved.push(idx),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            dependencies.push(resolved);
        }

        let mut marks = vec![0u8; jobs.len()];
        fn visit(node: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
            if marks[node] == 1 {
                return false;
            }
            if marks[node] == 2 {
                return true;
            }
            marks[node] = 1;
            for &dep in &deps[node] {
                if !visit(dep, deps, marks) {
                    return false;
                }
            }
            marks[node] = 2;
            true
        }
        for i in 0..jobs.len() {
            if !visit(i, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let records = jobs
            .into_iter()
            .enumerate()
            .map(|(order, job)| Record {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                job,
                dependencies: dependencies[order].clone(),
                remaining: None,
                started_at: 0,
                order,
            })
            .collect();

        Ok(Self {
            records,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            first_run: true,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        self.records
            .iter()
            .find(|record| record.job.id == id)
            .map(|record| record.state.clone())
            .ok_or(Error::Unknown)
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut lines = Vec::new();
        let mut time = if self.first_run { 0 } else { self.horizon };
        let mut process_current = self.first_run;

        loop {
            if process_current {
                self.process_time(time, &mut lines);
            }
            process_current = false;

            let next = self.next_event_time(time);
            match next {
                Some(next_time) if next_time <= until => {
                    time = next_time;
                    process_current = true;
                }
                _ => break,
            }
        }

        self.horizon = until;
        self.first_run = false;
        Ok(lines)
    }

    fn process_time(&mut self, time: u64, lines: &mut Vec<String>) {
        let mut completions: Vec<usize> = self
            .records
            .iter()
            .enumerate()
            .filter_map(|(i, r)| match r.state {
                State::Running { finish_at, .. } if finish_at == time => Some(i),
                _ => None,
            })
            .collect();
        completions.sort_by(|&a, &b| self.records[a].job.id.cmp(&self.records[b].job.id));

        for i in completions {
            let attempt = match self.records[i].state {
                State::Running { attempt, .. } => attempt,
                _ => continue,
            };
            let succeeded = attempt > self.records[i].job.failures;
            let id = self.records[i].job.id.clone();
            if succeeded {
                self.records[i].state = State::Succeeded;
                self.records[i].remaining = None;
                lines.push(format!("{time} finish {id} {attempt} ok"));
            } else {
                lines.push(format!("{time} finish {id} {attempt} fail"));
                self.records[i].remaining = None;
                if attempt >= self.records[i].job.max_attempts {
                    self.records[i].state = State::Failed;
                } else {
                    let shift = u32::from(attempt - 1);
                    let delay = self
                        .retry_base
                        .checked_shl(shift)
                        .unwrap_or(u64::MAX)
                        .min(self.retry_cap);
                    let ready_at = time.saturating_add(delay);
                    self.records[i].state = State::Waiting { ready_at, attempt };
                    lines.push(format!("{time} retry {id} {attempt} {ready_at}"));
                }
            }
        }

        let mut newly_blocked = Vec::new();
        loop {
            let to_block: Vec<usize> = self
                .records
                .iter()
                .enumerate()
                .filter_map(|(i, record)| {
                    if !matches!(record.state, State::Waiting { .. }) {
                        return None;
                    }
                    record.dependencies.iter().any(|&dep| {
                        matches!(self.records[dep].state, State::Failed | State::Blocked)
                    }).then_some(i)
                })
                .collect();
            if to_block.is_empty() {
                break;
            }
            for i in to_block {
                self.records[i].state = State::Blocked;
                newly_blocked.push(i);
            }
        }
        newly_blocked.sort_by(|&a, &b| self.records[a].job.id.cmp(&self.records[b].job.id));
        for i in newly_blocked {
            let dependency = self.records[i]
                .dependencies
                .iter()
                .filter(|&&dep| {
                    matches!(self.records[dep].state, State::Failed | State::Blocked)
                })
                .map(|&dep| self.records[dep].job.id.as_str())
                .min()
                .unwrap_or("");
            lines.push(format!(
                "{time} skip {} {dependency}",
                self.records[i].job.id
            ));
        }

        loop {
            let mut candidates: Vec<usize> = self
                .records
                .iter()
                .enumerate()
                .filter_map(|(i, record)| match record.state {
                    State::Waiting { ready_at, .. }
                        if ready_at <= time
                            && record.dependencies.iter().all(|&dep| {
                                matches!(self.records[dep].state, State::Succeeded)
                            }) =>
                    {
                        Some(i)
                    }
                    _ => None,
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, time);
                let pb = self.effective_priority(b, time);
                pb.cmp(&pa).then_with(|| self.records[a].order.cmp(&self.records[b].order))
            });

            let mut action = None;
            for candidate in candidates {
                if self.can_start_without_preemption(candidate) {
                    action = Some((candidate, None));
                    break;
                }
                if let Some(victim) = self.select_victim(candidate) {
                    action = Some((candidate, Some(victim)));
                    break;
                }
            }
            let Some((candidate, victim)) = action else {
                break;
            };

            if let Some(victim) = victim {
                let remaining = match self.records[victim].state {
                    State::Running { finish_at, .. } => finish_at - time,
                    _ => 0,
                };
                let victim_id = self.records[victim].job.id.clone();
                let candidate_id = self.records[candidate].job.id.clone();
                let attempt = match self.records[victim].state {
                    State::Running { attempt, .. } => attempt,
                    _ => 0,
                };
                self.records[victim].state = State::Waiting {
                    ready_at: time,
                    attempt,
                };
                self.records[victim].remaining = Some(remaining);
                lines.push(format!(
                    "{time} preempt {victim_id} by {candidate_id}"
                ));
            }

            self.start_job(candidate, time, lines);
        }
    }

    fn effective_priority(&self, i: usize, time: u64) -> u8 {
        if self.aging_ms == 0 {
            return self.records[i].job.priority;
        }
        let ready_at = match self.records[i].state {
            State::Waiting { ready_at, .. } => ready_at,
            _ => time,
        };
        let age = (time - ready_at) / self.aging_ms;
        (u64::from(self.records[i].job.priority) + age).min(9) as u8
    }

    fn can_start_without_preemption(&self, candidate: usize) -> bool {
        let running = self
            .records
            .iter()
            .filter(|r| matches!(r.state, State::Running { .. }))
            .count();
        if running >= self.capacity {
            return false;
        }
        let tag = self.records[candidate].job.tag.as_deref();
        tag.is_none()
            || !self.records.iter().any(|r| {
                matches!(r.state, State::Running { .. }) && r.job.tag.as_deref() == tag
            })
    }

    fn select_victim(&self, candidate: usize) -> Option<usize> {
        let candidate_priority = self.records[candidate].job.priority;
        let candidate_tag = self.records[candidate].job.tag.as_deref();
        let running_count = self
            .records
            .iter()
            .filter(|r| matches!(r.state, State::Running { .. }))
            .count();

        let mut victims: Vec<usize> = self
            .records
            .iter()
            .enumerate()
            .filter_map(|(i, r)| {
                let State::Running { .. } = r.state else {
                    return None;
                };
                if candidate_priority < r.job.priority.saturating_add(2) {
                    return None;
                }
                let resulting_slots_free = running_count - 1 < self.capacity;
                let candidate_tag_free = candidate_tag.is_none()
                    || r.job.tag.as_deref() == candidate_tag
                    || !self.records.iter().any(|other| {
                        matches!(other.state, State::Running { .. })
                            && other.job.tag.as_deref() == candidate_tag
                            && !std::ptr::eq(other, r)
                    });
                (resulting_slots_free && candidate_tag_free).then_some(i)
            })
            .collect();

        victims.sort_by(|&a, &b| {
            self.records[a]
                .job
                .priority
                .cmp(&self.records[b].job.priority)
                .then_with(|| self.records[b].started_at.cmp(&self.records[a].started_at))
                .then_with(|| self.records[b].job.id.cmp(&self.records[a].job.id))
        });
        victims.first().copied()
    }

    fn start_job(&mut self, i: usize, time: u64, lines: &mut Vec<String>) {
        let (attempt, ready_at) = match self.records[i].state {
            State::Waiting { attempt, ready_at } => (attempt, ready_at),
            _ => return,
        };
        let (new_attempt, remaining) = if let Some(remaining) = self.records[i].remaining.take() {
            (attempt, remaining)
        } else {
            (attempt.saturating_add(1), self.records[i].job.duration)
        };
        self.records[i].state = State::Running {
            attempt: new_attempt,
            finish_at: time.saturating_add(remaining),
        };
        self.records[i].started_at = time;
        let _ = ready_at;
        lines.push(format!(
            "{time} start {} {new_attempt} {remaining}",
            self.records[i].job.id
        ));
    }

    fn next_event_time(&self, after: u64) -> Option<u64> {
        self.records
            .iter()
            .filter_map(|record| match record.state {
                State::Waiting { ready_at, .. } if ready_at > after => Some(ready_at),
                State::Running { finish_at, .. } if finish_at > after => Some(finish_at),
                _ => None,
            })
            .min()
    }
}

        && value.len() <= 16
        && value
            .bytes()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')

struct JobData {

    state: State,

}

pub struct Pool {
    jobs: Vec<JobData>,
    capacity: usize,
    aging_ms: u64,

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if let Some(tag) = &job.tag {
                if !valid_token(tag, 1, 8) {
                    return Err(Error::BadTag(i));
                }
            }
            if ids.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            ids.insert(job.id.clone(), i);
        }

        let mut resolved = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (j, spelling) in job.dependencies.iter().enumerate() {
                if !seen.insert(spelling.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(spelling) {
                    Some(&index) => resolved[i].push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
        }

        let mut indegree = vec![0usize; jobs.len()];
        let mut dependents = vec![Vec::new(); jobs.len()];
        for (i, deps) in resolved.iter().enumerate() {
            indegree[i] = deps.len();
            for &dep in deps {
                dependents[dep].push(i);
            }
        }
        let mut queue = std::collections::VecDeque::new();
        for (i, &d) in indegree.iter().enumerate() {
            if d == 0 {
                queue.push_back(i);
            }
        }
        let mut visited = 0;
        while let Some(i) = queue.pop_front() {
            visited += 1;
            for &next in &dependents[i] {
                indegree[next] -= 1;
                if indegree[next] == 0 {
                    queue.push_back(next);
                }
            }
        }
        if visited != jobs.len() {
            return Err(Error::Cycle);
        }

        let data = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| JobData {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                job,
                dependencies: resolved[i].clone(),
                input_index: i,
                remaining: None,
                segment_order: 0,
            })
            .collect();

        Ok(Self {
            jobs: data,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
            next_segment_order: 0,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        self.jobs
            .iter()
            .find(|j| j.job.id == id)
            .map(|j| j.state.clone())
            .ok_or(Error::Unknown)
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if self.has_run && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }
        if self.has_run && until == self.horizon {
            return Ok(Vec::new());
        }

        let mut lines = Vec::new();
        let mut current = if self.has_run {
            self.horizon.checked_add(1)
        } else {
            Some(0)
        };

        while let Some(t) = current {
            if t > until {
                break;
            }

            let mut completed: Vec<usize> = self
                .jobs
                .iter()
                .enumerate()
                .filter_map(|(i, j)| match j.state {
                    State::Running { finish_at, .. } if finish_at == t => Some(i),
                    _ => None,
                })
                .collect();
            completed.sort_by(|&a, &b| self.jobs[a].job.id.cmp(&self.jobs[b].job.id));

            for i in completed {
                let (attempt, finish_at) = match self.jobs[i].state {
                    State::Running { attempt, finish_at } => (attempt, finish_at),
                    _ => continue,
                };
                let _ = finish_at;
                if attempt > self.jobs[i].job.failures {
                    self.jobs[i].state = State::Succeeded;
                    self.jobs[i].remaining = None;
                    lines.push(format!("{} finish {} {} ok", t, self.jobs[i].job.id, attempt));
                } else {
                    lines.push(format!("{} finish {} {} fail", t, self.jobs[i].job.id, attempt));
                    self.jobs[i].remaining = None;
                    if attempt >= self.jobs[i].job.max_attempts {
                        self.jobs[i].state = State::Failed;
                    } else {
                        let delay = retry_delay(self.retry_base, self.retry_cap, attempt);
                        let ready_at = t.saturating_add(delay);
                        self.jobs[i].state = State::Waiting { ready_at, attempt };
                        lines.push(format!(
                            "{} retry {} {} {}",
                            t, self.jobs[i].job.id, attempt, ready_at
                        ));
                    }
                }
            }

            let mut newly_blocked = Vec::new();
            loop {
                let mut found = Vec::new();
                for (i, job) in self.jobs.iter().enumerate() {
                    if matches!(job.state, State::Waiting { .. })
                        && job.dependencies.iter().any(|&d| {
                            matches!(self.jobs[d].state, State::Failed | State::Blocked)
                        })
                    {
                        found.push(i);
                    }
                }
                if found.is_empty() {
                    break;
                }
                for i in found {
                    let dep = self.jobs[i]
                        .dependencies
                        .iter()
                        .filter(|&&d| {
                            matches!(self.jobs[d].state, State::Failed | State::Blocked)
                        })
                        .min_by_key(|&&d| self.jobs[d].job.id.as_str())
                        .copied()
                        .unwrap();
                    self.jobs[i].state = State::Blocked;
                    self.jobs[i].remaining = None;
                    newly_blocked.push((i, dep));
                }
            }
            newly_blocked.sort_by(|(a, _), (b, _)| self.jobs[*a].job.id.cmp(&self.jobs[*b].job.id));
            for (i, dep) in newly_blocked {
                lines.push(format!(
                    "{} skip {} {}",
                    t, self.jobs[i].job.id, self.jobs[dep].job.id
                ));
            }

            self.schedule(t, &mut lines);

            if t == until {
                break;
            }

            let mut next: Option<u64> = None;
            for job in &self.jobs {
                let candidate = match job.state {
                    State::Running { finish_at, .. } if finish_at > t => Some(finish_at),
                    State::Waiting { ready_at, .. } if ready_at > t => Some(ready_at),
                    _ => None,
                };
                if let Some(value) = candidate {
                    next = Some(next.map_or(value, |n| n.min(value)));
                }
            }
            current = next;
        }

        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }

    fn schedule(&mut self, t: u64, lines: &mut Vec<String>) {
        loop {
            let mut candidates: Vec<usize> = self
                .jobs
                .iter()
                .enumerate()
                .filter_map(|(i, job)| {
                    match job.state {
                        State::Waiting { ready_at, .. }
                            if ready_at <= t
                                && job.dependencies.iter().all(|&d| {
                                    matches!(self.jobs[d].state, State::Succeeded)
                                }) =>
                        {
                            Some(i)
                        }
                        _ => None,
                    }
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, t);
                let pb = self.effective_priority(b, t);
                pb.cmp(&pa)
                    .then_with(|| self.jobs[a].input_index.cmp(&self.jobs[b].input_index))
            });

            let mut started = false;
            for candidate in candidates {
                let candidate_tag = self.jobs[candidate].job.tag.as_deref();
                let running: Vec<usize> = self
                    .jobs
                    .iter()
                    .enumerate()
                    .filter_map(|(i, job)| {
                        if matches!(job.state, State::Running { .. }) {
                            Some(i)
                        } else {
                            None
                        }
                    })
                    .collect();

                let tag_busy = candidate_tag.is_some_and(|tag| {
                    running.iter().any(|&i| self.jobs[i].job.tag.as_deref() == Some(tag))
                });

                if running.len() < self.capacity && !tag_busy {
                    self.start_job(candidate, t, None, lines);
                    started = true;
                    break;
                }

                let mut victims: Vec<usize> = running
                    .iter()
                    .copied()
                    .filter(|&victim| {
                        self.jobs[candidate].job.priority
                            >= self.jobs[victim].job.priority.saturating_add(2)
                            && running.len() - 1 < self.capacity
                            && !candidate_tag.is_some_and(|tag| {
                                running
                                    .iter()
                                    .copied()
                                    .filter(|&i| i != victim)
                                    .any(|i| self.jobs[i].job.tag.as_deref() == Some(tag))
                            })
                    })
                    .collect();

                victims.sort_by(|&a, &b| {
                    self.jobs[a]
                        .job
                        .priority
                        .cmp(&self.jobs[b].job.priority)
                        .then_with(|| self.jobs[b].segment_order.cmp(&self.jobs[a].segment_order))
                        .then_with(|| self.jobs[b].job.id.cmp(&self.jobs[a].job.id))
                });

                if let Some(victim) = victims.first().copied() {
                    let remaining = match self.jobs[victim].state {
                        State::Running { finish_at, .. } => finish_at.saturating_sub(t),
                        _ => 0,
                    };
                    lines.push(format!(
                        "{} preempt {} by {}",
                        t, self.jobs[victim].job.id, self.jobs[candidate].job.id
                    ));
                    let attempt = match self.jobs[victim].state {
                        State::Running { attempt, .. } => attempt,
                        _ => 0,
                    };
                    self.jobs[victim].state = State::Waiting {
                        ready_at: t,
                        attempt,
                    };
                    self.jobs[victim].remaining = Some(remaining);
                    self.start_job(candidate, t, None, lines);
                    started = true;
                    break;
                }
            }

            if !started {
                break;
            }
        }
    }

    fn start_job(
        &mut self,
        index: usize,
        t: u64,
        _unused: Option<()>,
        lines: &mut Vec<String>,
    ) {
        let (attempt, remaining) = match self.jobs[index].state {
            State::Waiting { attempt, .. } => {
                if let Some(saved) = self.jobs[index].remaining.take() {
                    (attempt, saved)
                } else {
                    let next_attempt = attempt.saturating_add(1);
                    (next_attempt, self.jobs[index].job.duration)
                }
            }
            _ => return,
        };
        let finish_at = t.saturating_add(remaining);
        self.next_segment_order = self.next_segment_order.saturating_add(1);
        self.jobs[index].segment_order = self.next_segment_order;
        self.jobs[index].state = State::Running { attempt, finish_at };
        lines.push(format!(
            "{} start {} {} {}",
            t, self.jobs[index].job.id, attempt, remaining
        ));
    }

    fn effective_priority(&self, index: usize, t: u64) -> u8 {
        if self.aging_ms == 0 {
            return self.jobs[index].job.priority;
        }
        let ready_at = match self.jobs[index].state {
            State::Waiting { ready_at, .. } => ready_at,
            _ => t,
        };
        let increase = (t - ready_at) / self.aging_ms;
        (self.jobs[index].job.priority as u64 + increase).min(9) as u8
    }
}

fn valid_token(s: &str, min: usize, max: usize) -> bool {
    let bytes = s.as_bytes();
    (min..=max).contains(&bytes.len())
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn retry_delay(base: u64, cap: u64, attempt: u8) -> u64 {
    let mut delay = base;
    for _ in 1..attempt {
        if delay >= cap {
            return cap;
        }
        delay = delay.saturating_mul(2).min(cap);
    }
    delay.min(cap)
}

struct Entry {

pub struct Pool {
    entries: Vec<Entry>,
    capacity: usize,
    aging_ms: u64,

    retry_cap: u64,

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base < 1
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::<String, usize>::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || !(1..=1_000_000_000).contains(&job.duration)
            {
                return Err(Error::BadTime(i));
            }
            if let Some(tag) = &job.tag {
                if !valid_token(tag, 1, 8) {
                    return Err(Error::BadTag(i));
                }
            }
            if ids.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut deps_by_job = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::<String>::new();
            for (j, name) in job.dependencies.iter().enumerate() {
                if !seen.insert(name.clone()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(name) {
                    Some(&index) => deps_by_job[i].push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
        }

        let mut indegree = deps_by_job.iter().map(Vec::len).collect::<Vec<_>>();
        let mut dependents = vec![Vec::new(); jobs.len()];
        for (i, deps) in deps_by_job.iter().enumerate() {
            for &dep in deps {
                dependents[dep].push(i);
            }
        }
        let mut queue = std::collections::VecDeque::new();
        for (i, &degree) in indegree.iter().enumerate() {
            if degree == 0 {
                queue.push_back(i);
            }
        }
        let mut visited = 0;
        while let Some(i) = queue.pop_front() {
            visited += 1;
            for &next in &dependents[i] {
                indegree[next] -= 1;
                if indegree[next] == 0 {
                    queue.push_back(next);
                }
            }
        }
        if visited != jobs.len() {
            return Err(Error::Cycle);
        }

        let entries = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| Entry {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                age_origin: job.release_at,
                remaining: None,
                segment_started: 0,
                job,
                deps: deps_by_job[i].clone(),
                input_order: i,
            })
            .collect();

        Ok(Self {
            entries,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        self.entries
            .iter()
            .find(|entry| entry.job.id == id)
            .map(|entry| entry.state.clone())
            .ok_or(Error::Unknown)
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if self.has_run && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }
        if self.has_run && until == self.horizon {
            return Ok(Vec::new());
        }

        let mut lines = Vec::new();
        let start_time = if self.has_run { self.horizon } else { 0 };
        let mut now = start_time;

        loop {
            self.process_timestamp(now, &mut lines);

            let next = self.next_event_after(now);
            match next {
                Some(t) if t <= until => now = t,
                _ => break,
            }
        }

        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }

    fn process_timestamp(&mut self, now: u64, lines: &mut Vec<String>) {
        let mut completed = self
            .entries
            .iter()
            .enumerate()
            .filter_map(|(i, e)| match e.state {
                State::Running { finish_at, attempt } if finish_at == now => {
                    Some((i, attempt, e.job.id.clone()))
                }
                _ => None,
            })
            .collect::<Vec<_>>();
        completed.sort_by(|a, b| a.2.as_bytes().cmp(b.2.as_bytes()));

        for (i, attempt, id) in completed {
            let success = attempt > self.entries[i].job.failures;
            if success {
                self.entries[i].state = State::Succeeded;
                self.entries[i].remaining = None;
                lines.push(format!("{now} finish {id} {attempt} ok"));
            } else {
                lines.push(format!("{now} finish {id} {attempt} fail"));
                self.entries[i].remaining = None;
                if attempt >= self.entries[i].job.max_attempts {
                    self.entries[i].state = State::Failed;
                } else {
                    let shift = u32::from(attempt.saturating_sub(1));
                    let delay = self.retry_base.checked_shl(shift).unwrap_or(u64::MAX)
                        .min(self.retry_cap);
                    let ready = now.saturating_add(delay);
                    self.entries[i].state = State::Waiting {
                        ready_at: ready,
                        attempt,
                    };
                    self.entries[i].age_origin = ready;
                    lines.push(format!("{now} retry {id} {attempt} {ready}"));
                }
            }
        }

        let mut newly_blocked = Vec::new();
        loop {
            let mut changed = false;
            for i in 0..self.entries.len() {
                if !matches!(self.entries[i].state, State::Waiting { .. }) {
                    continue;
                }
                if let Some(&bad_dep) = self.entries[i].deps.iter().find(|&&d| {
                    matches!(
                        self.entries[d].state,
                        State::Failed | State::Blocked
                    )
                }) {
                    self.entries[i].state = State::Blocked;
                    newly_blocked.push((i, bad_dep));
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
        newly_blocked.sort_by(|(a, _), (b, _)| {
            self.entries[*a]
                .job
                .id
                .as_bytes()
                .cmp(self.entries[*b].job.id.as_bytes())
        });
        for (i, dep) in newly_blocked {
            lines.push(format!(
                "{now} skip {} {}",
                self.entries[i].job.id, self.entries[dep].job.id
            ));
        }

        loop {
            let mut candidates = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, e)| {
                    let ready_at = match e.state {
                        State::Waiting { ready_at, .. } if ready_at <= now => ready_at,
                        _ => return None,
                    };
                    if !e.deps.iter().all(|&d| matches!(self.entries[d].state, State::Succeeded))
                    {
                        return None;
                    }
                    let effective = if self.aging_ms == 0 {
                        e.job.priority
                    } else {
                        let age = now.saturating_sub(e.age_origin) / self.aging_ms;
                        (u64::from(e.job.priority) + age).min(9) as u8
                    };
                    Some((i, effective, ready_at))
                })
                .collect::<Vec<_>>();
            candidates.sort_by(|a, b| {
                b.1.cmp(&a.1)
                    .then_with(|| self.entries[a.0].input_order.cmp(&self.entries[b.0].input_order))
            });

            let mut started = false;
            for (candidate, _, _) in candidates {
                let candidate_tag = self.entries[candidate].job.tag.as_deref();
                let running = self
                    .entries
                    .iter()
                    .enumerate()
                    .filter_map(|(i, e)| match e.state {
                        State::Running {
                            attempt,
                            finish_at,
                        } => Some((i, attempt, finish_at)),
                        _ => None,
                    })
                    .collect::<Vec<_>>();

                let tag_busy = candidate_tag.map_or(false, |tag| {
                    running.iter().any(|&(i, _, _)| {
                        self.entries[i].job.tag.as_deref() == Some(tag)
                    })
                });
                if running.len() < self.capacity && !tag_busy {
                    self.start_job(candidate, now, lines);
                    started = true;
                    break;
                }

                let mut victims = running
                    .iter()
                    .filter_map(|&(i, _, _)| {
                        let victim_tag = self.entries[i].job.tag.as_deref();
                        let slot_free_after = running.len() - 1 < self.capacity;
                        let tag_free_after = candidate_tag.map_or(true, |tag| {
                            victim_tag == Some(tag)
                                || !running.iter().any(|&(other, _, _)| {
                                    other != i
                                        && self.entries[other].job.tag.as_deref() == Some(tag)
                                })
                        });
                        if slot_free_after
                            && tag_free_after
                            && self.entries[candidate].job.priority
                                >= self.entries[i].job.priority.saturating_add(2)
                        {
                            Some(i)
                        } else {
                            None
                        }
                    })
                    .collect::<Vec<_>>();
                victims.sort_by(|&a, &b| {
                    self.entries[a]
                        .job
                        .priority
                        .cmp(&self.entries[b].job.priority)
                        .then_with(|| {
                            self.entries[b]
                                .segment_started
                                .cmp(&self.entries[a].segment_started)
                        })
                        .then_with(|| {
                            self.entries[b]
                                .job
                                .id
                                .as_bytes()
                                .cmp(self.entries[a].job.id.as_bytes())
                        })
                });

                if let Some(&victim) = victims.first() {
                    let (attempt, finish_at) = match self.entries[victim].state {
                        State::Running { attempt, finish_at } => (attempt, finish_at),
                        _ => unreachable!(),
                    };
                    let remaining = finish_at - now;
                    let victim_id = self.entries[victim].job.id.clone();
                    let candidate_id = self.entries[candidate].job.id.clone();
                    self.entries[victim].state = State::Waiting {
                        ready_at: now,
                        attempt,
                    };
                    self.entries[victim].age_origin = now;
                    self.entries[victim].remaining = Some(remaining);
                    lines.push(format!("{now} preempt {victim_id} by {candidate_id}"));
                    self.start_job(candidate, now, lines);
                    started = true;
                    break;
                }
            }
            if !started {
                break;
            }
        }
    }

    fn start_job(&mut self, index: usize, now: u64, lines: &mut Vec<String>) {
        let (attempt, remaining) = match self.entries[index].state {
            State::Waiting { attempt, .. } => {
                if let Some(remaining) = self.entries[index].remaining.take() {
                    (attempt, remaining)
                } else {
                    (attempt.saturating_add(1), self.entries[index].job.duration)
                }
            }
            _ => unreachable!(),
        };
        let finish_at = now.saturating_add(remaining);
        let id = self.entries[index].job.id.clone();
        self.entries[index].segment_started = now;
        self.entries[index].state = State::Running { attempt, finish_at };
        lines.push(format!("{now} start {id} {attempt} {remaining}"));
    }

    fn next_event_after(&self, now: u64) -> Option<u64> {
        let mut next = None;
        for entry in &self.entries {
            let t = match entry.state {
                State::Waiting { ready_at, .. } if ready_at > now => Some(ready_at),
                State::Running { finish_at, .. } if finish_at > now => Some(finish_at),
                _ => None,
            };
            if let Some(t) = t {
                next = Some(next.map_or(t, |current: u64| current.min(t)));
            }
        }
        next
    }
}

fn valid_token(value: &str, min_len: usize, max_len: usize) -> bool {
    let bytes = value.as_bytes();
    (min_len..=max_len).contains(&bytes.len())
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}
