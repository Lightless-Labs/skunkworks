#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    text: String,
    consumed: usize,
    marker_ended: bool,
    pieces: usize,
    gap: usize,
}

#[derive(Clone)]
struct Candidate {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed_vector: Vec<usize>,
    prejustify_vector: Vec<String>,
}

fn separators(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn add_cost(a: u64, b: u64) -> u64 {
    a + b
}

fn candidate_better(a: &Candidate, b: &Candidate) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed_vector != b.consumed_vector {
        return a.consumed_vector > b.consumed_vector;
    }
    a.prejustify_vector < b.prejustify_vector
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    if words.is_empty() {
        return Vec::new();
    }

    let mut states: Vec<(usize, usize, bool)> = Vec::new();
    for wi in 0..words.len() {
        for off in 0..words[wi].chars.len() {
            states.push((wi, off, wi == 0 && off == 0));
        }
    }
    let mut dp: Vec<Option<Candidate>> = vec![None; states.len()];
    let state_index = |wi: usize, off: usize| -> usize {
        states.iter().position(|&(w, o, _)| w == wi && o == off).unwrap()
    };

    let mut positions: Vec<usize> = (0..states.len()).collect();
    positions.sort_by_key(|&i| {
        let (wi, off, _) = states[i];
        (std::cmp::Reverse(wi), std::cmp::Reverse(off))
    });

    for idx in positions {
        let (wi, off, first) = states[idx];
        let indent = if first { options.first_indent } else { options.rest_indent };
        let cap = options.width - indent;
        let mut best: Option<Candidate> = None;

        let mut transitions: Vec<(usize, usize, bool, String, usize, bool, usize, u64)> = Vec::new();

        let mut body = String::new();
        let mut consumed = 0usize;
        let mut pieces = 0usize;
        let mut gap_count = 0usize;
        let mut cursor_w = wi;
        let mut cursor_off = off;

        loop {
            if cursor_w >= words.len() {
                break;
            }

            let word = &words[cursor_w];
            let remaining = word.chars.len() - cursor_off;
            let room_for_word = cap.saturating_sub(consumed + if pieces > 0 { 1 } else { 0 });

            if remaining <= room_for_word {
                if pieces > 0 {
                    body.push(' ');
                    consumed += 1;
                    gap_count += 1;
                }
                for &ch in &word.chars[cursor_off..] {
                    body.push(ch);
                }
                consumed += remaining;
                pieces += 1;
                cursor_w += 1;
                cursor_off = 0;

                let next_is_final = cursor_w == words.len();
                let display_width = indent + consumed;
                let gap = options.width - display_width;
                let line_cost = if next_is_final {
                    (gap * gap / 4) as u64
                } else {
                    (gap * gap) as u64 + if pieces == 1 { 3 } else { 0 }
                };
                let mut text = " ".repeat(indent);
                text.push_str(&body);
                transitions.push((cursor_w, 0, false, text, consumed, false, pieces, line_cost));

                if next_is_final {
                    // Completion transition is handled below as a terminal candidate.
                    let line = Line {
                        text: transitions.last().unwrap().3.clone(),
                        consumed,
                        marker_ended: false,
                        pieces,
                        gap,
                    };
                    let candidate = Candidate {
                        cost: line_cost,
                        marker_lines: 0,
                        consumed_vector: vec![consumed],
                        prejustify_vector: vec![line.text.clone()],
                        lines: vec![line],
                    };
                    if best.as_ref().map_or(true, |b| candidate_better(&candidate, b)) {
                        best = Some(candidate);
                    }
                }
            } else {
                // Marked splits from this word, when the line has room.
                let prefix_base = consumed + if pieces > 0 { 1 } else { 0 };
                let available = cap.saturating_sub(prefix_base);
                for &m in &word.markers {
                    if m <= cursor_off || m >= word.chars.len() {
                        continue;
                    }
                    let prefix_len = m - cursor_off;
                    if prefix_len + 1 <= available {
                        let mut next_body = body.clone();
                        if pieces > 0 {
                            next_body.push(' ');
                        }
                        for &ch in &word.chars[cursor_off..m] {
                            next_body.push(ch);
                        }
                        next_body.push('-');
                        let line_consumed = prefix_base + prefix_len;
                        let gap = options.width - (indent + line_consumed + 1);
                        let line_cost = (gap * gap) as u64
                            + options.hyphen_cost as u64
                            + if pieces == 0 { 3 } else { 0 };
                        let mut text = " ".repeat(indent);
                        text.push_str(&next_body);
                        transitions.push((
                            wi,
                            m,
                            false,
                            text,
                            line_consumed,
                            true,
                            pieces + 1,
                            line_cost,
                        ));
                    }
                }

                // Raw split is permitted only on an otherwise empty line and only if no marker fits.
                if pieces == 0 && remaining > cap {
                    let has_marker_fit = word.markers.iter().any(|&m| {
                        m > cursor_off && m < word.chars.len() && m - cursor_off + 1 <= cap
                    });
                    if !has_marker_fit {
                        let mut text = " ".repeat(indent);
                        for &ch in &word.chars[cursor_off..cursor_off + cap] {
                            text.push(ch);
                        }
                        let gap = options.width - (indent + cap);
                        let line_cost = (gap * gap) as u64 + (2 * options.width + 1) as u64 + 3;
                        transitions.push((
                            wi,
                            cursor_off + cap,
                            false,
                            text,
                            cap,
                            false,
                            1,
                            line_cost,
                        ));
                    }
                }
                break;
            }

            if cursor_w >= words.len() {
                break;
            }
        }

        for (nwi, noff, _, text, line_consumed, marker_ended, line_pieces, line_cost) in transitions {
            let target = if nwi == words.len() {
                None
            } else {
                Some(state_index(nwi, noff))
            };
            let mut suffix = if let Some(t) = target {
                match dp[t].clone() {
                    Some(c) => c,
                    None => continue,
                }
            } else {
                continue;
            };
            let line_gap = options.width.saturating_sub(
                text.chars().count(),
            );
            let mut lines = vec![Line {
                text: text.clone(),
                consumed: line_consumed,
                marker_ended,
                pieces: line_pieces,
                gap: line_gap,
            }];
            lines.append(&mut suffix.lines);
            let mut consumed_vector = vec![line_consumed];
            consumed_vector.append(&mut suffix.consumed_vector);
            let mut prejustify_vector = vec![text];
            prejustify_vector.append(&mut suffix.prejustify_vector);
            let candidate = Candidate {
                cost: add_cost(line_cost, suffix.cost),
                marker_lines: usize::from(marker_ended) + suffix.marker_lines,
                lines,
                consumed_vector,
                prejustify_vector,
            };
            if best.as_ref().map_or(true, |b| candidate_better(&candidate, b)) {
                best = Some(candidate);
            }
        }

        dp[idx] = best;
    }

    let start = state_index(0, 0);
    dp[start].as_ref().map(|c| c.lines.clone()).unwrap_or_default()
}

fn parse_segment(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Word {
        chars: Vec::new(),
        markers: Vec::new(),
    };
    for &c in chars {
        if separators(c) {
            if !current.chars.is_empty() {
                words.push(current);
                current = Word {
                    chars: Vec::new(),
                    markers: Vec::new(),
                };
            }
        } else if c == '~' {
            current.markers.push(current.chars.len());
        } else {
            current.chars.push(c);
        }
    }
    if !current.chars.is_empty() {
        words.push(current);
    }
    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let valid_neighbor = |j: isize| -> bool {
            if j < 0 || j as usize >= chars.len() {
                return false;
            }
            let c = chars[j as usize].1;
            c != '~' && !separators(c)
        };
        if !valid_neighbor(i as isize - 1) || !valid_neighbor(i as isize + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0usize;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(c);
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut pos = 0usize;
    while pos < normalized.len() {
        if normalized[pos] == '\n' {
            let mut end = pos;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            if end - pos >= 2 {
                let words = parse_segment(&segment);
                output.extend(solve_paragraph(&words, options).into_iter().map(|line| line.text));
                segment.clear();
                for _ in 0..(end - pos - 1) {
                    output.push(String::new());
                }
                pos = end;
                continue;
            }
        }
        segment.push(normalized[pos]);
        pos += 1;
    }

    let words = parse_segment(&segment);
    let mut lines = solve_paragraph(&words, options);
    if options.justify {
        let total_lines = lines.len();
        for (line_index, line) in lines.iter_mut().enumerate() {
            if line_index + 1 == total_lines || line.pieces < 2 {
                continue;
            }
            let current_width = line.text.chars().count();
            let extra = options.width.saturating_sub(current_width);
            let gaps = line.pieces - 1;
            if gaps == 0 {
                continue;
            }
            let each = extra / gaps;
            let rem = extra % gaps;
            let indent_len = if line_index == 0 { options.first_indent } else { options.rest_indent };
            let content: String = line.text.chars().skip(indent_len).collect();
            let parts: Vec<&str> = content.split(' ').collect();
            if parts.len() != line.pieces {
                continue;
            }
            let mut rendered = " ".repeat(indent_len);
            for (j, part) in parts.iter().enumerate() {
                if j > 0 {
                    let gap_idx = j - 1;
                    let rotation = line_index % gaps;
                    let extra_one = if (gap_idx + gaps - rotation) % gaps < rem { 1 } else { 0 };
                    rendered.push_str(&" ".repeat(1 + each + extra_one));
                }
                rendered.push_str(part);
            }
            line.text = rendered;
        }
    }
    output.extend(lines.into_iter().map(|line| line.text));
    Ok(output)
}
