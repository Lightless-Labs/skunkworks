fn add_cost(a: u64, b: u64) -> u64 {

fn candidate_better(a: &Candidate, b: &Candidate) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed_vector != b.consumed_vector {
        return a.consumed_vector > b.consumed_vector;
    }
    a.prejustify_vector < b.prejustify_vector
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {

                    gap_count += 1;

                    body.push(ch);

                    (gap * gap) as u64 + if pieces == 1 { 3 } else { 0 }

fn parse_segment(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Word {
        chars: Vec::new(),
        markers: Vec::new(),
    };
    for &c in chars {
        if separators(c) {
            if !current.chars.is_empty() {
                words.push(current);
                current = Word {
                    chars: Vec::new(),
                    markers: Vec::new(),
                };
            }
        } else if c == '~' {
            current.markers.push(current.chars.len());
        } else {
            current.chars.push(c);
        }
    }
    if !current.chars.is_empty() {
        words.push(current);
    }
    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let valid_neighbor = |j: isize| -> bool {
            if j < 0 || j as usize >= chars.len() {
                return false;
            }
            let c = chars[j as usize].1;
            c != '~' && !separators(c)
        };
        if !valid_neighbor(i as isize - 1) || !valid_neighbor(i as isize + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0usize;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(c);
            i += 1;
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut pos = 0usize;
    while pos < normalized.len() {
        if normalized[pos] == '\n' {
            let mut end = pos;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            if end - pos >= 2 {
                let words = parse_segment(&segment);
                output.extend(solve_paragraph(&words, options).into_iter().map(|line| line.text));
                segment.clear();
                for _ in 0..(end - pos - 1) {
                    output.push(String::new());
                }
                pos = end;
                continue;
            }
        }
        segment.push(normalized[pos]);
        pos += 1;
    }

    let words = parse_segment(&segment);
    let mut lines = solve_paragraph(&words, options);
    if options.justify {
        let total_lines = lines.len();
        for (line_index, line) in lines.iter_mut().enumerate() {
            if line_index + 1 == total_lines || line.pieces < 2 {
                continue;
            }
            let current_width = line.text.chars().count();
            let extra = options.width.saturating_sub(current_width);
            let gaps = line.pieces - 1;
            if gaps == 0 {
                continue;
            }
            let each = extra / gaps;
            let rem = extra % gaps;
            let indent_len = if line_index == 0 { options.first_indent } else { options.rest_indent };
            let content: String = line.text.chars().skip(indent_len).collect();
            let parts: Vec<&str> = content.split(' ').collect();
            if parts.len() != line.pieces {
                continue;
            }
            let mut rendered = " ".repeat(indent_len);
            for (j, part) in parts.iter().enumerate() {
                if j > 0 {
                    let gap_idx = j - 1;
                    let rotation = line_index % gaps;
                    let extra_one = if (gap_idx + gaps - rotation) % gaps < rem { 1 } else { 0 };
                    rendered.push_str(&" ".repeat(1 + each + extra_one));
                }
                rendered.push_str(part);
            }
            line.text = rendered;
        }
    }
    output.extend(lines.into_iter().map(|line| line.text));
    Ok(output)
}
