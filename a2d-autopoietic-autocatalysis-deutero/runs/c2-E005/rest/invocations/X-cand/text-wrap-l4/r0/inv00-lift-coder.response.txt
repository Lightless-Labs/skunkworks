#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    text: String,
    consumed: usize,
    marker_ended: bool,
    pieces: usize,
    gap: usize,
    final_line: bool,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_count: usize,
    lines: Vec<Line>,
}

fn solution_better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_count != b.marker_count {
        return a.marker_count < b.marker_count;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }

    let ac: Vec<usize> = a.lines.iter().map(|l| l.consumed).collect();
    let bc: Vec<usize> = b.lines.iter().map(|l| l.consumed).collect();
    if ac != bc {
        return ac > bc;
    }

    let at: Vec<&str> = a.lines.iter().map(|l| l.text.as_str()).collect();
    let bt: Vec<&str> = b.lines.iter().map(|l| l.text.as_str()).collect();
    at < bt
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    use std::collections::HashMap;

    type State = (usize, usize, bool);
    let mut memo: HashMap<State, Option<Solution>> = HashMap::new();

    fn rec(
        state: State,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut HashMap<State, Option<Solution>>,
    ) -> Option<Solution> {
        let (wi, offset, first) = state;
        if wi >= words.len() {
            return Some(Solution {
                cost: 0,
                marker_count: 0,
                lines: Vec::new(),
            });
        }
        if let Some(cached) = memo.get(&state) {
            return cached.clone();
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let capacity = options.width - indent;
        let mut best: Option<Solution> = None;

        let mut consider = |end_wi: usize,
                            end_offset: usize,
                            text: String,
                            consumed: usize,
                            marker_ended: bool,
                            raw: bool,
                            pieces: usize| {
            let next_state = if end_wi == words.len() {
                (words.len(), 0, false)
            } else {
                (end_wi, end_offset, false)
            };
            let tail = rec(next_state, words, options, memo);
            let Some(mut tail) = tail else {
                return;
            };

            let scalar_width = text.chars().count();
            let gap = options.width.saturating_sub(scalar_width);
            let is_final = end_wi == words.len();
            let mut line_cost = (gap * gap) as u64;
            if is_final {
                line_cost /= 4;
            }
            if marker_ended {
                line_cost += options.hyphen_cost as u64;
                tail.marker_count += 1;
            }
            if raw {
                line_cost += (2 * options.width + 1) as u64;
            }
            if !is_final && pieces == 1 {
                line_cost += 3;
            }

            let mut lines = Vec::with_capacity(tail.lines.len() + 1);
            lines.push(Line {
                text,
                consumed,
                marker_ended,
                pieces,
                gap,
                final_line: is_final,
            });
            lines.append(&mut tail.lines);
            let candidate = Solution {
                cost: line_cost + tail.cost,
                marker_count: tail.marker_count,
                lines,
            };
            if best
                .as_ref()
                .map(|old| solution_better(&candidate, old))
                .unwrap_or(true)
            {
                best = Some(candidate);
            }
        };

        let mut assembled = String::new();
        let mut pieces = 0usize;
        let mut total_consumed = 0usize;
        let mut current_wi = wi;
        let mut current_offset = offset;

        while current_wi < words.len() {
            let word = &words[current_wi];
            let remaining = word.chars.len() - current_offset;
            let separator = if pieces == 0 { 0 } else { 1 };
            let room = capacity.saturating_sub(assembled.chars().count() + separator);
            if room == 0 {
                break;
            }

            let available = remaining.min(room);
            if available == remaining {
                if pieces > 0 {
                    assembled.push(' ');
                }
                assembled.extend(word.chars[current_offset..].iter());
                pieces += 1;
                total_consumed += remaining;
                current_wi += 1;
                current_offset = 0;

                let mut full = " ".repeat(indent);
                full.push_str(&assembled);
                consider(
                    current_wi,
                    current_offset,
                    full,
                    total_consumed,
                    false,
                    false,
                    pieces,
                );
                continue;
            }

            if pieces == 0 {
                let split_end = current_offset + available;
                let marker_fits = word.markers.iter().any(|&m| {
                    m > current_offset
                        && m < word.chars.len()
                        && m - current_offset + 1 <= capacity
                });

                for &marker in &word.markers {
                    if marker > current_offset
                        && marker < word.chars.len()
                        && marker - current_offset + 1 <= capacity
                    {
                        let mut displayed: String =
                            word.chars[current_offset..marker].iter().collect();
                        displayed.push('-');
                        let mut full = " ".repeat(indent);
                        full.push_str(&displayed);
                        consider(
                            current_wi,
                            marker,
                            full,
                            marker - current_offset,
                            true,
                            false,
                            1,
                        );
                    }
                }

                if remaining > capacity && !marker_fits {
                    let mut displayed: String =
                        word.chars[current_offset..split_end].iter().collect();
                    let mut full = " ".repeat(indent);
                    full.push_str(&displayed);
                    consider(
                        current_wi,
                        split_end,
                        full,
                        available,
                        false,
                        true,
                        1,
                    );
                    displayed.clear();
                }
            }
            break;
        }

        let result = best;
        memo.insert(state, result.clone());
        result
    }

    rec((0, 0, true), words, options, &mut memo)
        .map(|s| s.lines)
        .unwrap_or_default()
}

fn parse_segment(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Word {
        chars: Vec::new(),
        markers: Vec::new(),
    };
    let mut in_word = false;

    for &ch in segment {
        if matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00a0}') {
            if in_word {
                words.push(current);
                current = Word {
                    chars: Vec::new(),
                    markers: Vec::new(),
                };
                in_word = false;
            }
        } else if ch == '~' {
            current.markers.push(current.chars.len());
            in_word = true;
        } else {
            current.chars.push(ch);
            in_word = true;
        }
    }
    if in_word {
        words.push(current);
    }

    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, ch) in &chars {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let valid_side = |j: usize| {
                j < chars.len()
                    && chars[j].1 != '~'
                    && !matches!(chars[j].1, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
            };
            if i == 0 || i + 1 >= chars.len() || !valid_side(i - 1) || !valid_side(i + 1) {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let ch = chars[i].1;
        if ch == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 1;
            }
        } else {
            normalized.push(ch);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                let words = parse_segment(&segment);
                let mut lines = solve_paragraph(&words, options);
                output.extend(lines.drain(..).map(|line| render_line(line, options, 0)));
                output.extend(std::iter::repeat(String::new()).take(j - i - 1));
                segment.clear();
                i = j;
                continue;
            }
        }
        segment.push(normalized[i]);
        i += 1;
    }

    let words = parse_segment(&segment);
    let lines = solve_paragraph(&words, options);
    let paragraph_offset = 0;
    output.extend(
        lines
            .into_iter()
            .enumerate()
            .map(|(idx, line)| render_line(line, options, paragraph_offset + idx)),
    );

    Ok(output)
}

fn render_line(mut line: Line, options: &WrapOptions, paragraph_line_index: usize) -> String {
    if options.justify && !line.final_line && line.pieces >= 2 {
        let indent = if paragraph_line_index == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        let mut chars: Vec<char> = line.text.chars().collect();
        let body: String = chars.drain(indent..).collect();
        let parts: Vec<&str> = body.split(' ').collect();
        if parts.len() >= 2 {
            let extra = line.gap;
            let gaps = parts.len() - 1;
            let base = extra / gaps;
            let remainder = extra % gaps;
            let start = paragraph_line_index % gaps;
            let mut rendered = " ".repeat(indent);
            for (idx, part) in parts.iter().enumerate() {
                if idx > 0 {
                    let gap_idx = idx - 1;
                    let cyclic = (gap_idx + gaps - start) % gaps;
                    let n = 1 + base + usize::from(cyclic < remainder);
                    rendered.push_str(&" ".repeat(n));
                }
                rendered.push_str(part);
            }
            line.text = rendered;
        }
    }
    line.text
}
