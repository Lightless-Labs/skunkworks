fn digit(byte: u8, options: Options, offset: usize) -> Result<u8, Error> {
    if let Some(index) = alphabet(options).iter().position(|&b| b == byte) {
        return Ok(index as u8);

            Alphabet::Standard => byte == b'-' || byte == b'_',
            Alphabet::UrlSafe => byte == b'+' || byte == b'/',

            return Err(Error::MixedAlphabet { offset });
        }

fn encode_group(bytes: &[u8], options: Options) -> Vec<u8> {
    let table = alphabet(options);
    let mut out = Vec::with_capacity(4);
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    out.push(table[(a >> 2) as usize]);
    out.push(table[(((a & 0x03) << 4) | (b >> 4)) as usize]);

    } else if options.padding != Padding::Forbidden {

pub struct Encoder {
    options: Options,
    buffer: Vec<u8>,
    closed: bool,
    row_symbols: usize,
    pending_separator: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            buffer: Vec::new(),
            closed: false,
            row_symbols: 0,
            pending_separator: false,
        })
    }

    fn append_symbols(&mut self, symbols: &[u8], out: &mut String) {
        let wrapped = match self.options.whitespace {
            Whitespace::Wrapped { width, ending } => Some((width, ending)),
            _ => None,
        };

        for &symbol in symbols {
            if let Some((width, ending)) = wrapped {
                if self.pending_separator {
                    match ending {
                        LineEnding::Lf => out.push('\n'),
                        LineEnding::CrLf => out.push_str("\r\n"),
                    }
                    self.pending_separator = false;
                }
                out.push(symbol as char);
                self.row_symbols += 1;
                if self.row_symbols == width {
                    self.row_symbols = 0;
                    self.pending_separator = true;
                }
            } else {
                out.push(symbol as char);
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut data = Vec::with_capacity(self.buffer.len() + input.len());
        data.extend_from_slice(&self.buffer);
        data.extend_from_slice(input);

        let complete = data.len() / 3 * 3;
        let mut out = String::new();
        for chunk in data[..complete].chunks_exact(3) {
            let symbols = encode_group(chunk, self.options);
            self.append_symbols(&symbols, &mut out);
        }
        self.buffer.clear();
        self.buffer.extend_from_slice(&data[complete..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut out = String::new();
        if !self.buffer.is_empty() {
            let symbols = encode_group(&self.buffer, self.options);
            self.append_symbols(&symbols, &mut out);
            self.buffer.clear();
        }
        self.closed = true;
        Ok(out)
    }
}

enum FrameState {

    NeedLf { cr_offset: usize },
    NeedSymbol { start_offset: usize },

pub struct Decoder {
    options: Options,
    group: Vec<(u8, usize)>,
    pending_second_pad: Option<usize>,
    terminal_padding: bool,
    total_offset: usize,
    symbol_count: usize,
    frame_symbols: usize,
    frame_state: FrameState,
    closed: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: Vec::with_capacity(4),
            pending_second_pad: None,
            terminal_padding: false,
            total_offset: 0,
            symbol_count: 0,
            frame_symbols: 0,
            frame_state: FrameState::Normal,
            closed: false,
            sticky_error: None,
        })
    }

    fn skip_ascii(&self, byte: u8) -> bool {
        self.options.whitespace == Whitespace::IgnoreAscii
            && matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
    }

    fn process_wrapped(&mut self, byte: u8, offset: usize) -> Result<bool, Error> {
        let (width, ending) = match self.options.whitespace {
            Whitespace::Wrapped { width, ending } => (width, ending),
            _ => return Ok(true),
        };

        match self.frame_state {
            FrameState::NeedLf { cr_offset } => {
                if byte != b'\n' {
                    return Err(Error::BadWrap { offset });
                }
                self.frame_state = FrameState::NeedSymbol {
                    start_offset: cr_offset,
                };
                Ok(false)
            }
            FrameState::NeedSymbol { .. } => {
                if byte == b'\r' || byte == b'\n' {
                    return Err(Error::BadWrap { offset });
                }
                self.frame_symbols = 1;
                self.frame_state = FrameState::Normal;
                Ok(true)
            }
            FrameState::Normal => {
                if self.frame_symbols == width {
                    match ending {
                        LineEnding::Lf => {
                            if byte != b'\n' {
                                return Err(Error::BadWrap { offset });
                            }
                            self.frame_state = FrameState::NeedSymbol {
                                start_offset: offset,
                            };
                            Ok(false)
                        }
                        LineEnding::CrLf => {
                            if byte != b'\r' {
                                return Err(Error::BadWrap { offset });
                            }
                            self.frame_state = FrameState::NeedLf { cr_offset: offset };
                            Ok(false)
                        }
                    }
                } else if byte == b'\r' || byte == b'\n' {
                    Err(Error::BadWrap { offset })
                } else {
                    self.frame_symbols += 1;
                    Ok(true)
                }
            }
        }
    }

    fn emit_group(&mut self, out: &mut Vec<u8>) -> Result<(), Error> {
        let g = &self.group;

        if g.len() == 4 && g[2].0 == b'=' && g[3].0 == b'=' {
            let a = digit(g[0].0, self.options, g[0].1)?;
            let b = digit(g[1].0, self.options, g[1].1)?;
            if self.options.strict && (b & 0x0f) != 0 {
                return Err(Error::NonCanonical { offset: g[1].1 });
            }
            out.push((a << 2) | (b >> 4));
            self.terminal_padding = true;
        } else if g.len() == 4 && g[3].0 == b'=' {
            let a = digit(g[0].0, self.options, g[0].1)?;
            let b = digit(g[1].0, self.options, g[1].1)?;
            let c = digit(g[2].0, self.options, g[2].1)?;
            if self.options.strict && (c & 0x03) != 0 {
                return Err(Error::NonCanonical { offset: g[2].1 });
            }
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
            self.terminal_padding = true;
        } else {
            let a = digit(g[0].0, self.options, g[0].1)?;
            let b = digit(g[1].0, self.options, g[1].1)?;
            let c = digit(g[2].0, self.options, g[2].1)?;
            let d = digit(g[3].0, self.options, g[3].1)?;
            out.push((a << 2) | (b >> 4));
            out.push((b << 4) | (c >> 2));
            out.push((c << 6) | d);
        }

        self.group.clear();
        Ok(())
    }

    fn scan_symbol(
        &mut self,
        byte: u8,
        offset: usize,
        out: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if let Some(limit) = self.options.max_symbols {
            if self.symbol_count >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.symbol_count += 1;

        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if let Some(pad_offset) = self.pending_second_pad.take() {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.group.push((byte, offset));
            let _ = pad_offset;
            self.emit_group(out)?;
            return Ok(());
        }

        let position = self.group.len();
        if byte == b'=' {
            match position {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset });
                    }
                    self.group.push((byte, offset));
                    self.pending_second_pad = Some(offset);
                    return Ok(());
                }
                3 => {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset });
                    }
                    self.group.push((byte, offset));
                    self.emit_group(out)?;
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        let value = digit(byte, self.options, offset)?;
        self.group.push((byte, offset));
        if self.group.len() == 4 {
            let _ = value;
            self.emit_group(out)?;
        }
        Ok(())
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &byte in input {
            let offset = self.total_offset;
            self.total_offset = self.total_offset.saturating_add(1);

            if self.skip_ascii(byte) {
                continue;
            }

            let symbol = match self.process_wrapped(byte, offset) {
                Ok(symbol) => symbol,
                Err(error) => return self.fail(error),
            };
            if !symbol {
                continue;
            }

            if let Err(error) = self.scan_symbol(byte, offset, &mut out) {
                return self.fail(error);
            }
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        if let FrameState::NeedLf { cr_offset } = self.frame_state {
            return self.fail(Error::BadWrap { offset: cr_offset });
        }
        if let FrameState::NeedSymbol { start_offset } = self.frame_state {
            return self.fail(Error::BadWrap {
                offset: start_offset,
            });
        }

        if self.pending_second_pad.is_some() {
            let offset = self.pending_second_pad.unwrap();
            return self.fail(Error::BadPadding { offset });
        }

        let mut out = Vec::new();
        if self.group.is_empty() {
            self.closed = true;
            return Ok(out);
        }

        if self.group.iter().any(|(b, _)| *b == b'=') {
            return self.fail(Error::BadPadding {
                offset: self.group.iter().find(|(b, _)| *b == b'=').unwrap().1,
            });
        }

        if self.options.padding == Padding::Required {
            return self.fail(Error::Truncated {
                offset: self.group[0].1,
            });
        }

        match self.group.len() {
            1 => {
                return self.fail(Error::Truncated {
                    offset: self.group[0].1,
                });
            }
            2 => {
                let a = match digit(self.group[0].0, self.options, self.group[0].1) {
                    Ok(v) => v,
                    Err(e) => return self.fail(e),
                };
                let b = match digit(self.group[1].0, self.options, self.group[1].1) {
                    Ok(v) => v,
                    Err(e) => return self.fail(e),
                };
                if self.options.strict && (b & 0x0f) != 0 {
                    return self.fail(Error::NonCanonical {
                        offset: self.group[1].1,
                    });
                }
                out.push((a << 2) | (b >> 4));
            }
            3 => {
                let a = match digit(self.group[0].0, self.options, self.group[0].1) {
                    Ok(v) => v,
                    Err(e) => return self.fail(e),
                };
                let b = match digit(self.group[1].0, self.options, self.group[1].1) {
                    Ok(v) => v,
                    Err(e) => return self.fail(e),
                };
                let c = match digit(self.group[2].0, self.options, self.group[2].1) {
                    Ok(v) => v,
                    Err(e) => return self.fail(e),
                };
                if self.options.strict && (c & 0x03) != 0 {
                    return self.fail(Error::NonCanonical {
                        offset: self.group[2].1,
                    });
                }
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
            }
            _ => unreachable!(),
        }

        self.group.clear();
        self.closed = true;
        Ok(out)
    }
}

fn alphabet_digit(alphabet: Alphabet, byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }

fn is_foreign_alphabet_byte(alphabet: Alphabet, byte: u8) -> bool {
    match alphabet {
        Alphabet::Standard => byte == b'-' || byte == b'_',
        Alphabet::UrlSafe => byte == b'+' || byte == b'/',
    }

fn encoded_symbol(alphabet: Alphabet, value: u8) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + value - 26,
        52..=61 => b'0' + value - 52,
        62 if alphabet == Alphabet::Standard => b'+',
        62 => b'-',
        63 if alphabet == Alphabet::Standard => b'/',
        _ => b'_',
    }

fn append_symbol(out: &mut String, byte: u8) {
    out.push(byte as char);

fn encode_raw(input: &[u8], options: Options) -> Vec<u8> {
    let mut out = Vec::new();
    let mut chunks = input.chunks_exact(3);
    for chunk in &mut chunks {
        let a = chunk[0];
        let b = chunk[1];
        let c = chunk[2];
        out.push(encoded_symbol(options.alphabet, a >> 2));
        out.push(encoded_symbol(options.alphabet, ((a & 0x03) << 4) | (b >> 4)));
        out.push(encoded_symbol(options.alphabet, ((b & 0x0f) << 2) | (c >> 6)));
        out.push(encoded_symbol(options.alphabet, c & 0x3f));
    }
    let tail = chunks.remainder();
    if !tail.is_empty() {
        let a = tail[0];
        out.push(encoded_symbol(options.alphabet, a >> 2));
        if tail.len() == 1 {
            out.push(encoded_symbol(options.alphabet, (a & 0x03) << 4));
            if options.padding != Padding::Forbidden {
                out.extend_from_slice(b"==");
            }
        } else {
            let b = tail[1];
            out.push(encoded_symbol(options.alphabet, ((a & 0x03) << 4) | (b >> 4)));
            out.push(encoded_symbol(options.alphabet, (b & 0x0f) << 2));
            if options.padding != Padding::Forbidden {
                out.push(b'=');
            }
        }
    }
    out
}

fn line_ending_bytes(ending: LineEnding) -> &'static [u8] {
    match ending {
        LineEnding::Lf => b"\n",
        LineEnding::CrLf => b"\r\n",
    }
}

fn wrap_bytes(raw: &[u8], width: usize, ending: LineEnding) -> Vec<u8> {
    let mut out = Vec::with_capacity(
        raw.len().saturating_add(raw.len() / width * line_ending_bytes(ending).len()),
    );
    for (i, byte) in raw.iter().enumerate() {
        if i != 0 && i % width == 0 {
            out.extend_from_slice(line_ending_bytes(ending));
        }
        out.push(*byte);
    }
    out
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let raw = encode_raw(input, options);
    let bytes = match options.whitespace {
        Whitespace::Wrapped { width, ending } => wrap_bytes(&raw, width, ending),
        Whitespace::None | Whitespace::IgnoreAscii => raw,
    };
    Ok(String::from_utf8(bytes).expect("base64 output is ASCII"))
}

struct EncoderState {
    options: Options,
    pending: Vec<u8>,
    row_symbols: usize,
    closed: bool,
}

pub struct Encoder {
    state: EncoderState,
}

fn emit_encoded_symbols(state: &mut EncoderState, symbols: &[u8], out: &mut String) {
    let wrapped = match state.options.whitespace {
        Whitespace::Wrapped { width, ending } => Some((width, ending)),
        _ => None,
    };
    for symbol in symbols {
        if let Some((width, ending)) = wrapped {
            if state.row_symbols == width {
                for byte in line_ending_bytes(ending) {
                    append_symbol(out, *byte);
                }
                state.row_symbols = 0;
            }
            state.row_symbols += 1;
        }
        append_symbol(out, *symbol);
    }
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            state: EncoderState {
                options,
                pending: Vec::new(),
                row_symbols: 0,
                closed: false,
            },
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.state.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut all = std::mem::take(&mut self.state.pending);
        all.extend_from_slice(input);
        let complete_len = all.len() / 3 * 3;
        let raw = encode_raw(&all[..complete_len], self.state.options);
        self.state.pending.extend_from_slice(&all[complete_len..]);
        let mut out = String::new();
        emit_encoded_symbols(&mut self.state, &raw, &mut out);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.state.closed {
            return Err(Error::AlreadyFinished);
        }
        self.state.closed = true;
        let pending = std::mem::take(&mut self.state.pending);
        let raw = encode_raw(&pending, self.state.options);
        let mut out = String::new();
        emit_encoded_symbols(&mut self.state, &raw, &mut out);
        Ok(out)
    }
}

struct DecoderState {
    options: Options,
    group: Vec<Digit>,
    pending_second_pad: Option<usize>,
    terminal_padding: bool,
    symbol_count: usize,
    physical_offset: usize,
    row_symbols: usize,
    framing: FramingState,
    finished: bool,
    sticky: Option<Error>,
}

pub struct Decoder {
    state: DecoderState,
}

fn emit_group(state: &mut DecoderState, out: &mut Vec<u8>, padded: bool) -> Result<(), Error> {
    let group = &state.group;
    if padded {
        if group.len() == 2 {
            if state.options.strict && group[1].value & 0x0f != 0 {
                return Err(Error::NonCanonical {
                    offset: group[1].offset,
                });
            }
            out.push((group[0].value << 2) | (group[1].value >> 4));
        } else {
            if state.options.strict && group[2].value & 0x03 != 0 {
                return Err(Error::NonCanonical {
                    offset: group[2].offset,
                });
            }
            out.push((group[0].value << 2) | (group[1].value >> 4));
            out.push((group[1].value << 4) | (group[2].value >> 2));
        }
    } else {
        out.push((group[0].value << 2) | (group[1].value >> 4));
        out.push((group[1].value << 4) | (group[2].value >> 2));
        out.push((group[2].value << 6) | group[3].value);
    }
    Ok(())
}

fn is_ignored_whitespace(byte: u8) -> bool {
    matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
}

fn decoder_scan_byte(
    state: &mut DecoderState,
    byte: u8,
    offset: usize,
    out: &mut Vec<u8>,
) -> Result<(), Error> {
    match state.options.whitespace {
        Whitespace::IgnoreAscii if is_ignored_whitespace(byte) => return Ok(()),
        Whitespace::Wrapped { width, ending } => {
            match state.framing {
                FramingState::NeedLf { cr_offset } => {
                    if byte != b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    state.framing = FramingState::NeedSymbol {
                        separator_offset: cr_offset,
                    };
                    return Ok(());
                }
                FramingState::NeedSymbol { .. } => {
                    state.framing = FramingState::Normal;
                    state.row_symbols = 0;
                }
                FramingState::Normal => {}
            }
            if state.row_symbols == width {
                match ending {
                    LineEnding::Lf if byte == b'\n' => {
                        state.framing = FramingState::NeedSymbol {
                            separator_offset: offset,
                        };
                        return Ok(());
                    }
                    LineEnding::CrLf if byte == b'\r' => {
                        state.framing = FramingState::NeedLf { cr_offset: offset };
                        return Ok(());
                    }
                    _ => return Err(Error::BadWrap { offset }),
                }
            } else if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                return Err(Error::BadWrap { offset });
            }
        }
        _ => {}
    }

    if let Some(limit) = state.options.max_symbols {
        if state.symbol_count >= limit {
            return Err(Error::Overlong { offset, limit });
        }
    }
    state.symbol_count = state.symbol_count.saturating_add(1);
    if matches!(state.options.whitespace, Whitespace::Wrapped { .. }) {
        state.row_symbols = state.row_symbols.saturating_add(1);
    }

    if state.terminal_padding {
        return Err(Error::BadPadding { offset });
    }
    if let Some(pad_offset) = state.pending_second_pad {
        if byte != b'=' {
            return Err(Error::BadPadding { offset });
        }
        let mut result = Vec::new();
        emit_group(state, &mut result, true)?;
        out.extend(result);
        state.group.clear();
        state.pending_second_pad = None;
        state.terminal_padding = true;
        return Ok(());
    }

    if byte == b'=' {
        match state.group.len() {
            0 | 1 => return Err(Error::BadPadding { offset }),
            2 => {
                if state.options.padding == Padding::Forbidden {
                    return Err(Error::BadPadding { offset });
                }
                state.pending_second_pad = Some(offset);
                return Ok(());
            }
            3 => {
                if state.options.padding == Padding::Forbidden {
                    return Err(Error::BadPadding { offset });
                }
                let mut result = Vec::new();
                emit_group(state, &mut result, true)?;
                out.extend(result);
                state.group.clear();
                state.terminal_padding = true;
                return Ok(());
            }
            _ => return Err(Error::BadPadding { offset }),
        }
    }

    if state.group.len() == 3 {
        let value = match alphabet_digit(state.options.alphabet, byte) {
            Some(value) => value,
            None if state.options.strict && is_foreign_alphabet_byte(state.options.alphabet, byte) => {
                return Err(Error::MixedAlphabet { offset });
            }
            None => return Err(Error::InvalidByte { offset, byte }),
        };
        state.group.push(Digit { value, offset });
        let mut result = Vec::new();
        emit_group(state, &mut result, false)?;
        out.extend(result);
        state.group.clear();
        return Ok(());
    }

    let value = match alphabet_digit(state.options.alphabet, byte) {
        Some(value) => value,
        None if state.options.strict && is_foreign_alphabet_byte(state.options.alphabet, byte) => {
            return Err(Error::MixedAlphabet { offset });
        }
        None => return Err(Error::InvalidByte { offset, byte }),
    };
    state.group.push(Digit { value, offset });
    Ok(())
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            state: DecoderState {
                options,
                group: Vec::new(),
                pending_second_pad: None,
                terminal_padding: false,
                symbol_count: 0,
                physical_offset: 0,
                row_symbols: 0,
                framing: FramingState::Normal,
                finished: false,
                sticky: None,
            },
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.state.sticky {
            return Err(error.clone());
        }
        if self.state.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut out = Vec::new();
        for byte in input {
            let offset = self.state.physical_offset;
            self.state.physical_offset = self.state.physical_offset.saturating_add(1);
            if let Err(error) = decoder_scan_byte(&mut self.state, *byte, offset, &mut out) {
                self.state.sticky = Some(error.clone());
                return Err(error);
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.state.sticky {
            return Err(error.clone());
        }
        if self.state.finished {
            return Err(Error::AlreadyFinished);
        }

        let result = (|| {
            match self.state.framing {
                FramingState::NeedLf { cr_offset } => {
                    return Err(Error::BadWrap { offset: cr_offset });
                }
                FramingState::NeedSymbol { separator_offset } => {
                    return Err(Error::BadWrap {
                        offset: separator_offset,
                    });
                }
                FramingState::Normal => {}
            }
            if let Some(offset) = self.state.pending_second_pad {
                return Err(Error::BadPadding { offset });
            }
            let len = self.state.group.len();
            if len == 0 {
                return Ok(Vec::new());
            }
            if len == 1 || self.state.options.padding == Padding::Required {
                return Err(Error::Truncated {
                    offset: self.state.group[0].offset,
                });
            }
            if len == 2 {
                if self.state.options.strict && self.state.group[1].value & 0x0f != 0 {
                    return Err(Error::NonCanonical {
                        offset: self.state.group[1].offset,
                    });
                }
                return Ok(vec![
                    (self.state.group[0].value << 2) | (self.state.group[1].value >> 4),
                ]);
            }
            if self.state.options.strict && self.state.group[2].value & 0x03 != 0 {
                return Err(Error::NonCanonical {
                    offset: self.state.group[2].offset,
                });
            }
            Ok(vec![
                (self.state.group[0].value << 2) | (self.state.group[1].value >> 4),
                (self.state.group[1].value << 4) | (self.state.group[2].value >> 2),
            ])
        })();

        match result {
            Ok(bytes) => {
                self.state.finished = true;
                Ok(bytes)
            }
            Err(error) => {
                self.state.sticky = Some(error.clone());
                Err(error)
            }
        }
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

        Ok(Self { options, tail: Vec::new(), row: 0, closed: false })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut output = String::new();
        let mut index = 0;

        if !self.tail.is_empty() {
            while self.tail.len() < 3 && index < input.len() {
                self.tail.push(input[index]);
                index += 1;
            }
            if self.tail.len() == 3 {
                let bytes = [self.tail[0], self.tail[1], self.tail[2]];
                append_triple(self.options, &mut self.row, &mut output, &bytes);
                self.tail.clear();
            }
        }

        while index + 3 <= input.len() {
            append_triple(self.options, &mut self.row, &mut output, &input[index..index + 3]);
            index += 3;
        }
        self.tail.extend_from_slice(&input[index..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;
        let mut output = String::new();

        match self.tail.as_slice() {
            [a] => {
                append_symbol(self.options, &mut self.row, &mut output, a >> 2);
                append_symbol(self.options, &mut self.row, &mut output, (a & 0x03) << 4);
                if self.options.padding != Padding::Forbidden {
                    append_padding(self.options, &mut self.row, &mut output, 2);
                }
            }
            [a, b] => {
                append_symbol(self.options, &mut self.row, &mut output, a >> 2);
                append_symbol(
                    self.options,
                    &mut self.row,
                    &mut output,
                    ((a & 0x03) << 4) | (b >> 4),
                );
                append_symbol(self.options, &mut self.row, &mut output, (b & 0x0f) << 2);
                if self.options.padding != Padding::Forbidden {
                    append_padding(self.options, &mut self.row, &mut output, 1);
                }
            }
            _ => {}
        }
        self.tail.clear();
        Ok(output)
    }
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    let mut encoder = Encoder::new(options)?;
    let mut output = encoder.push(input)?;
    output.push_str(&encoder.finish()?);
    Ok(output)
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
    value: Option<u8>,
}

pub struct Decoder {
    options: Options,
    offset: usize,
    symbol_count: usize,
    group: Vec<Symbol>,
    terminal_padding: bool,
    pending_padding: Option<usize>,
    row: usize,
    frame: u8,
    frame_offset: usize,
    after_separator: bool,
    closed: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            offset: 0,
            symbol_count: 0,
            group: Vec::new(),
            terminal_padding: false,
            pending_padding: None,
            row: 0,
            frame: 0,
            frame_offset: 0,
            after_separator: false,
            closed: false,
            sticky_error: None,
        })
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }

    fn complete_group(&mut self, output: &mut Vec<u8>) -> Result<(), Error> {
        let values: Vec<u8> = self.group.iter().filter_map(|symbol| symbol.value).collect();
        output.push((values[0] << 2) | (values[1] >> 4));
        if self.group[2].byte != b'=' {
            output.push((values[1] << 4) | (values[2] >> 2));
        }
        if self.group[3].byte != b'=' {
            output.push((values[2] << 6) | values[3]);
        }

        if self.options.strict {
            let noncanonical = if self.group[2].byte == b'=' && values[1] & 0x0f != 0 {
                Some(self.group[1].offset)
            } else if self.group[3].byte == b'=' && values[2] & 0x03 != 0 {
                Some(self.group[2].offset)
            } else {
                None
            };
            if let Some(offset) = noncanonical {
                return Err(Error::NonCanonical { offset });
            }
        }

        if self.group[2].byte == b'=' || self.group[3].byte == b'=' {
            self.terminal_padding = true;
        }
        self.group.clear();
        Ok(())
    }

    fn process_symbol(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        if let Some(limit) = self.options.max_symbols {
            if self.symbol_count >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.symbol_count += 1;

        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }
        if self.pending_padding.is_some() {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.pending_padding = None;
            self.group.push(Symbol { byte, offset, value: None });
            return self.complete_group(output);
        }

        let position = self.group.len();
        if byte == b'=' {
            if position < 2 || self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            if position == 2 {
                self.pending_padding = Some(offset);
                self.group.push(Symbol { byte, offset, value: None });
                return Ok(());
            }
            if position == 3 {
                self.group.push(Symbol { byte, offset, value: None });
                return self.complete_group(output);
            }
        }

        let value = decode_digit(self.options.alphabet, byte, self.options.strict, offset)?;
        self.group.push(Symbol { byte, offset, value: Some(value) });
        if self.group.len() == 4 {
            self.complete_group(output)?;
        }
        Ok(())
    }

    fn process_byte(&mut self, byte: u8, offset: usize, output: &mut Vec<u8>) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => self.process_symbol(byte, offset, output),
            Whitespace::IgnoreAscii => {
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    Ok(())
                } else {
                    self.process_symbol(byte, offset, output)
                }
            }
            Whitespace::Wrapped { width, ending } => {
                if self.frame == 1 {
                    if byte != b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    self.frame = 0;
                    self.row = 0;
                    self.after_separator = true;
                    return Ok(());
                }
                if self.frame == 2 {
                    if byte != b'\r' {
                        return Err(Error::BadWrap { offset });
                    }
                    self.frame_offset = offset;
                    if ending == LineEnding::Lf {
                        self.frame = 0;
                        self.row = 0;
                        self.after_separator = true;
                    } else {
                        self.frame = 3;
                    }
                    return Ok(());
                }
                if self.frame == 3 {
                    if byte != b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    self.frame = 0;
                    self.row = 0;
                    self.after_separator = true;
                    return Ok(());
                }
                if self.row == width {
                    self.frame_offset = offset;
                    self.frame = match ending {
                        LineEnding::Lf => 1,
                        LineEnding::CrLf => 2,
                    };
                    return self.process_byte(byte, offset, output);
                }
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    return Err(Error::BadWrap { offset });
                }
                self.after_separator = false;
                self.row += 1;
                self.process_symbol(byte, offset, output)
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;
            if let Err(error) = self.process_byte(byte, offset, &mut output) {
                return self.fail(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        if matches!(self.options.whitespace, Whitespace::Wrapped { .. }) {
            if self.frame == 1 || self.frame == 2 || self.after_separator {
                return self.fail(Error::BadWrap { offset: self.frame_offset });
            }
            if self.frame == 3 {
                return self.fail(Error::BadWrap { offset: self.frame_offset });
            }
        }

        if let Some(offset) = self.pending_padding {
            return self.fail(Error::BadPadding { offset });
        }

        if !self.group.is_empty() {
            let count = self.group.len();
            let first_offset = self.group[0].offset;
            if self.options.padding == Padding::Required || count == 1 {
                return self.fail(Error::Truncated { offset: first_offset });
            }
            if self.options.strict {
                let (value, offset) = if count == 2 {
                    (self.group[1].value.unwrap(), self.group[1].offset)
                } else {
                    (self.group[2].value.unwrap(), self.group[2].offset)
                };
                if (count == 2 && value & 0x0f != 0) || (count == 3 && value & 0x03 != 0) {
                    return self.fail(Error::NonCanonical { offset });
                }
            }

            let values: Vec<u8> = self.group.iter().map(|symbol| symbol.value.unwrap()).collect();
            output.push((values[0] << 2) | (values[1] >> 4));
            if count == 3 {
                output.push((values[1] << 4) | (values[2] >> 2));
            }
            self.group.clear();
        }

        self.closed = true;
        Ok(output)
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new(options)?;
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}
