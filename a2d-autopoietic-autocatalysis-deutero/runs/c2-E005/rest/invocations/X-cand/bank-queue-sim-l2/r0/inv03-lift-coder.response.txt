```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    busy: bool,
    idle_since: u64,
    finish_at: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| a.arrival.cmp(&b.arrival).then_with(|| b.id.cmp(&a.id)));

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy: false,
            finish_at: None,
        })
        .collect();

    let mut waiting: Vec<usize> = Vec::new();
    let mut next_arrival = 0usize;
    let mut events = Vec::new();
    let mut now: Option<u64> = None;

    loop {
        let next_finish = states
            .iter()
            .filter_map(|state| state.finish_at)
            .min();
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.server.opens)
            .min();
        let next_arrival_time = arrivals.get(next_arrival).map(|customer| customer.arrival);

        let next_time = match (next_finish, next_open, next_arrival_time) {
            (None, None, None) => break,
            (f, o, a) => [f, o, a].into_iter().flatten().min().unwrap(),
        };

        if now.is_some_and(|previous| next_time < previous) {
            unreachable!("simulation time must not move backwards");
        }
        now = Some(next_time);

        let mut completing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| (state.finish_at == Some(next_time)).then_some(index))
            .collect();
        completing.sort_by_key(|&index| states[index].server.id);
        for index in completing {
            let state = &mut states[index];
            events.push(Event::Finish {
                time: next_time,
                customer: 0,
                server: state.server.id,
            });
            state.finish_at = None;
            state.busy = false;
            state.idle_since = next_time;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.server.opens == next_time).then_some(index)
            })
            .collect();
        opening.sort_by(|&a, &b| states[b].server.id.cmp(&states[a].server.id));
        for index in opening {
            let state = &mut states[index];
            state.opened = true;
            events.push(Event::Open {
                time: next_time,
                server: state.server.id,
            });
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == next_time {
            let customer = &arrivals[next_arrival];
            events.push(Event::Arrive {
                time: next_time,
                customer: customer.id,
            });
            waiting.push(next_arrival);
            next_arrival += 1;
        }

        while !waiting.is_empty() {
            let selected_server = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && !state.busy)
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected_server else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &arrivals[customer_index];
            let server = &states[server_index].server;

            let finish = customer
                .service
                .checked_add(next_time)
                .and_then(|value| value.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: next_time,
                })?;

            events.push(Event::Start {
                time: next_time,
                customer: customer.id,
                server: server.id,
                finish,
            });

            let state = &mut states[server_index];
            state.busy = true;
            state.finish_at = Some(finish);

            if finish == next_time {
                events.push(Event::Finish {
                    time: next_time,
                    customer: customer.id,
                    server: server.id,
                });
                state.busy = false;
                state.finish_at = None;
                state.idle_since = next_time;
            }
        }
    }

    Ok(events)
}
```
