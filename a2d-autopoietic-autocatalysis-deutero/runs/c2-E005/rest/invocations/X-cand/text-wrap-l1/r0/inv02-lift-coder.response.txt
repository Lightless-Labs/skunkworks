#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let bytes = input.as_bytes();

    for (index, &byte) in bytes.iter().enumerate() {
        let allowed = (32..=126).contains(&byte) || matches!(byte, b'\t' | b'\n' | b'\r');
        if !allowed {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();

    for word in input.split(|c: char| matches!(c, ' ' | '\t' | '\n' | '\r')).filter(|word| !word.is_empty()) {
        let word = word.as_bytes();
        let mut offset = 0;

        while offset < word.len() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            let remaining = word.len() - offset;

            if remaining <= width {
                let separator_len = usize::from(!line.is_empty());
                let needed = separator_len.checked_add(remaining).unwrap_or(usize::MAX);

                if line.len().checked_add(needed).is_some_and(|length| length <= width) {
                    if separator_len == 1 {
                        line.push(' ');
                    }
                    line.push_str(std::str::from_utf8(&word[offset..]).expect("validated ASCII"));
                    offset = word.len();
                } else {
                    if !line.is_empty() {
                        lines.push(std::mem::take(&mut line));
                    } else {
                        // A suffix no longer than width always fits on an empty line.
                        unreachable!("suffix should fit on an empty line");
                    }
                }
            } else if line.is_empty() {
                line.push_str(std::str::from_utf8(&word[offset..offset + width]).expect("validated ASCII"));
                offset += width;
                lines.push(std::mem::take(&mut line));
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(
                        std::str::from_utf8(&word[offset..offset + available])
                            .expect("validated ASCII"),
                    );
                    offset += available;
                    lines.push(std::mem::take(&mut line));
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}
