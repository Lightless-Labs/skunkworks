#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventKind {
    Start,
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub task: usize,
    pub worker: usize,
    pub kind: EventKind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub events: Vec<Event>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry in the required order.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    let mut children = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Kahn traversal, always removing the smallest available task ID.
    let mut available = std::collections::BTreeSet::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree.clone();
    while let Some(id) = available.pop_first() {
        order.push(id);
        for &child in &children[id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut set = vec![false; n];
            for &id in &order {
                set[id] = true;
            }
            set
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Bottom levels, computed in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::new();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut running: Vec<Option<(usize, u64)>> = vec![None; workers];
    let mut worker_available = vec![0u64; workers];
    let mut current_time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all positive-duration tasks finishing now, in task ID order.
        let mut finishing: Vec<(usize, usize)> = running
            .iter()
            .enumerate()
            .filter_map(|(worker, entry)| match entry {
                Some((task, finish)) if *finish == current_time => Some((*task, worker)),
                _ => None,
            })
            .collect();
        finishing.sort_unstable_by_key(|&(task, _)| task);

        for (task, worker) in finishing {
            running[worker] = None;
            completed[task] = true;
            completed_count += 1;
            makespan = makespan.max(current_time);
            events.push(Event {
                time: current_time,
                task,
                worker,
                kind: EventKind::Finish,
            });
        }

        loop {
            let worker = (0..workers).find(|&w| {
                running[w].is_none() && worker_available[w] <= current_time
            });
            let Some(worker) = worker else {
                break;
            };

            let selected = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= current_time
                        && tasks[id].deps.iter().all(|&pred| completed[pred])
                })
                .max_by(|&a, &b| {
                    tail[a]
                        .cmp(&tail[b])
                        .then_with(|| tasks[a].priority.cmp(&tasks[b].priority))
                        .then_with(|| b.cmp(&a))
                });

            let Some(task) = selected else {
                break;
            };

            let finish = current_time
                .checked_add(tasks[task].duration)
                .ok_or(Error::Overflow {
                    task,
                    stage: Stage::Finish,
                })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task,
                stage: Stage::Turnaround,
            })?;

            started[task] = true;
            worker_available[worker] = available_at;
            runs[task] = Some(Run {
                start: current_time,
                finish,
                worker,
            });
            events.push(Event {
                time: current_time,
                task,
                worker,
                kind: EventKind::Start,
            });

            if tasks[task].duration == 0 {
                completed[task] = true;
                completed_count += 1;
                makespan = makespan.max(current_time);
                events.push(Event {
                    time: current_time,
                    task,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                running[worker] = Some((task, finish));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > current_time {
                next_time = Some(next_time.map_or(time, |old| old.min(time)));
            }
        };

        for entry in &running {
            if let Some((_, finish)) = entry {
                consider(*finish);
            }
        }
        for (id, task) in tasks.iter().enumerate() {
            if !started[id] {
                consider(task.release);
            }
        }
        for worker in 0..workers {
            if running[worker].is_none() {
                consider(worker_available[worker]);
            }
        }

        current_time = next_time.expect("unfinished tasks must have a future event");
    }

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        events,
    })
}
