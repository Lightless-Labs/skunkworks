    workers: usize,
    turnaround: u64,
) -> Result<Schedule, Error> {

    // Validate every dependency before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Kahn traversal, so children are sorted before they are processed.
    let mut children = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }
    for list in &mut children {
        list.sort_unstable();
    }

    let mut available = std::collections::BinaryHeap::new();
    for task_id in 0..n {
        if indegree[task_id] == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        order.push(task_id);
        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut set = vec![false; n];
            for &task_id in &order {
                set[task_id] = true;
            }
            set
        };
        let blocked = (0..n).filter(|&task_id| !removed[task_id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_successor_tail = children[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]

    let mut worker_available = vec![0u64; workers];
    let mut remaining_predecessors: Vec<usize> =

                task: task_id,
                worker: worker_id,
                kind: EventKind::Finish,

                remaining_predecessors[child] -= 1;
            }

            let worker_id = (0..workers).find(|&w| {
                running[w].is_none() && worker_available[w] <= now
            });
            let Some(worker_id) = worker_id else {
                break;
            };

                    !started[task_id]
                        && tasks[task_id].release <= now
                        && remaining_predecessors[task_id] == 0
                })
                .max_by(|&a, &b| {

                        .cmp(&tails[b])
                        .then_with(|| tasks[a].priority.cmp(&tasks[b].priority))
                        .then_with(|| b.cmp(&a))
                });
            let Some(task_id) = selected else {
                break;
            };

                    task: task_id,
                    stage: Stage::Finish,
                },
            )?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,

                start: now,
                finish,
                worker: worker_id,
            });

                time: now,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Start,
            });

                completed_count += 1;

                events.push(Event {
                    time: now,
                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,
                });
                for &child in &children[task_id] {
                    remaining_predecessors[child] -= 1;
                }
            } else {
                running[worker_id] = Some((task_id, finish));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now {
                next_time = Some(next_time.map_or(time, |current| current.min(time)));
            }
        };

        for worker_id in 0..workers {
            if let Some((_, finish)) = running[worker_id] {
                consider(finish);
            }
            if running[worker_id].is_none() && worker_available[worker_id] > now {
                consider(worker_available[worker_id]);
            }
        }
        for task_id in 0..n {
            if !started[task_id] {
                consider(tasks[task_id].release);
            }

        // predecessors finishes, already included above. If no future event
        // exists, all tasks must have completed; otherwise this is unreachable
        // after successful cycle validation.
        if let Some(next) = next_time {
            now = next;
        } else {
            break;
        }
    }

        .into_iter()
        .map(|run| run.expect("all tasks complete after successful simulation"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

                kind: EventKind::Start,
            });

            if tasks[task].duration == 0 {
                completed[task] = true;
                completed_count += 1;
                makespan = makespan.max(current_time);
                events.push(Event {
                    time: current_time,
                    task,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                running[worker] = Some((task, finish));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > current_time {
                next_time = Some(next_time.map_or(time, |old| old.min(time)));
            }
        };

        for entry in &running {
            if let Some((_, finish)) = entry {
                consider(*finish);
            }
        }
        for (id, task) in tasks.iter().enumerate() {
            if !started[id] {
                consider(task.release);
            }
        }
        for worker in 0..workers {
            if running[worker].is_none() {
                consider(worker_available[worker]);
            }
        }

        current_time = next_time.expect("unfinished tasks must have a future event");
    }

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        events,
    })
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before checking any other condition.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest currently available task first.
    let mut available = std::collections::BinaryHeap::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree;
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        order.push(task_id);
        for &child in &children[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut removed = vec![false; n];
            for &task_id in &order {
                removed[task_id] = true;
            }
            removed
        };
        let blocked = (0..n).filter(|&task_id| !removed[task_id]).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = children[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_available = vec![0u64; workers];
    let mut running: Vec<Running> = Vec::new();
    let mut events = Vec::with_capacity(n.saturating_mul(2));
    let mut time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now, in ascending task ID order.
        let mut finishing: Vec<Running> = running
            .iter()
            .copied()
            .filter(|entry| entry.finish == time)
            .collect();
        finishing.sort_by_key(|entry| entry.task);

        for entry in finishing {
            running.retain(|other| other.task != entry.task);
            completed[entry.task] = true;
            completed_count += 1;
            makespan = makespan.max(entry.finish);
            events.push(Event {
                time,
                task: entry.task,
                worker: entry.worker,
                kind: EventKind::Finish,
            });
        }

        loop {
            let worker = (0..workers).find(|&id| worker_available[id] <= time);
            let Some(worker) = worker else {
                break;
            };

            let selected = (0..n)
                .filter(|&task_id| {
                    !started[task_id]
                        && tasks[task_id].release <= time
                        && tasks[task_id]
                            .deps
                            .iter()
                            .all(|&predecessor| completed[predecessor])
                })
                .max_by(|&a, &b| {
                    tails[a]
                        .cmp(&tails[b])
                        .then_with(|| tasks[a].priority.cmp(&tasks[b].priority))
                        .then_with(|| b.cmp(&a))
                });

            let Some(task_id) = selected else {
                break;
            };

            let finish = time.checked_add(tasks[task_id].duration).ok_or(
                Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                },
            )?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            runs[task_id] = Some(Run {
                start: time,
                finish,
                worker,
            });
            events.push(Event {
                time,
                task: task_id,
                worker,
                kind: EventKind::Start,
            });

            if finish == time {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                events.push(Event {
                    time,
                    task: task_id,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > time && next_time.map_or(true, |current| candidate < current) {
                next_time = Some(candidate);
            }
        };

        for entry in &running {
            consider(entry.finish);
        }
        for (task_id, task) in tasks.iter().enumerate() {
            if !started[task_id] {
                consider(task.release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        match next_time {
            Some(next) => time = next,
            None => break,
        }
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task is started exactly once"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before checking workers or performing any
    // scheduling computation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::new();
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest currently available ID first.
    let mut available = std::collections::BinaryHeap::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(id));
        }
    }
    let mut order = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(id)) = available.pop() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }
    if order.len() != n {
        let removed: Vec<bool> = {
            let mut flags = vec![false; n];
            for &id in &order {
                flags[id] = true;
            }
            flags
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::new();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_deps: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut running = Vec::<(u64, usize, usize)>::new(); // (finish, task, worker)
    let mut worker_available = vec![0u64; workers];
    let mut time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete every positive-duration task finishing now, in task-ID
        // order. Zero-duration tasks are completed immediately on admission.
        let mut finishing: Vec<(u64, usize, usize)> = Vec::new();
        let mut still_running = Vec::with_capacity(running.len());
        for entry in running.drain(..) {
            if entry.0 == time {
                finishing.push(entry);
            } else {
                still_running.push(entry);
            }
        }
        running = still_running;
        finishing.sort_by_key(|entry| entry.1);
        for &(finish, task_id, worker_id) in &finishing {
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(finish);
            events.push(Event {
                time: finish,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Finish,
            });
            for &child in &children[task_id] {
                remaining_deps[child] -= 1;
            }
        }

        loop {
            let worker_id = (0..workers).find(|&w| worker_available[w] <= time);
            let worker_id = match worker_id {
                Some(w) => w,
                None => break,
            };

            let mut selected: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id]
                    || tasks[task_id].release > time
                    || remaining_deps[task_id] != 0
                {
                    continue;
                }
                let better = match selected {
                    None => true,
                    Some(current) => {
                        tails[task_id] > tails[current]
                            || (tails[task_id] == tails[current]
                                && (tasks[task_id].priority > tasks[current].priority
                                    || (tasks[task_id].priority == tasks[current].priority
                                        && task_id < current)))
                    }
                };
                if better {
                    selected = Some(task_id);
                }
            }
            let task_id = match selected {
                Some(id) => id,
                None => break,
            };

            let finish = time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let worker_ready = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker_id] = worker_ready;
            runs[task_id] = Some(Run {
                start: time,
                finish,
                worker: worker_id,
            });
            events.push(Event {
                time,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Start,
            });

            if finish == time {
                // Complete zero-duration tasks before choosing another task.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                events.push(Event {
                    time: finish,
                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,
                });
                for &child in &children[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.push((finish, task_id, worker_id));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > time {
                next_time = Some(next_time.map_or(finish, |current| current.min(finish)));
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > time {
                let release = tasks[task_id].release;
                next_time = Some(next_time.map_or(release, |current| current.min(release)));
            }
        }
        for &ready in &worker_available {
            if ready > time {
                next_time = Some(next_time.map_or(ready, |current| current.min(ready)));
            }
        }

        match next_time {
            Some(next) => time = next,
            None => {
                // With validated acyclic dependencies, an unfinished task
                // must have a future finish, release, or worker availability.
                // This fallback prevents an unexpected infinite loop.
                return Err(Error::Cycle {
                    blocked: (0..n).filter(|&id| !completed[id]).collect(),
                });
            }
        }
    }

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        events,
    })
}
