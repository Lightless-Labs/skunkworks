#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone)]
struct Specifier {
    name: String,
    start: usize,
    modifier: Modifier,
}

#[derive(Debug, Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continue,
}

impl Operator {
    fn from_char(c: char) -> Option<Self> {
        match c {
            '+' => Some(Self::Reserved),
            '#' => Some(Self::Fragment),
            '.' => Some(Self::Dot),
            '/' => Some(Self::Slash),
            ';' => Some(Self::Semicolon),
            '?' => Some(Self::Query),
            '&' => Some(Self::Continue),
            _ => None,
        }
    }

    fn separator(self) -> char {
        match self {
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon => ';',
            Self::Query | Self::Continue => '&',
            Self::Simple | Self::Reserved | Self::Fragment => ',',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Fragment => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Query => Some('?'),
            Self::Continue => Some('&'),
            Self::Simple | Self::Reserved => None,
        }
    }

    fn uses_reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }

    fn names_scalar_fields(self) -> bool {
        matches!(self, Self::Semicolon | Self::Query | Self::Continue)
    }
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn parse_specifier(
    spec: &str,
    start: usize,
    seen: &mut Vec<String>,
) -> Result<Specifier, Error> {
    if spec.is_empty() {
        return Err(Error::EmptyVariable { pos: start });
    }

    let bytes = spec.as_bytes();
    let split = bytes.iter().position(|&b| b == b'*' || b == b':');
    let name_end = split.unwrap_or(bytes.len());

    if name_end == 0 {
        return Err(Error::InvalidName { pos: start });
    }

    if !is_name_start(bytes[0]) {
        return Err(Error::InvalidName { pos: start });
    }

    for (i, &b) in bytes[..name_end].iter().enumerate().skip(1) {
        if !is_name_continue(b) {
            return Err(Error::InvalidName { pos: start + i });
        }
    }

    let name = spec[..name_end].to_string();

    let modifier = if let Some(index) = split {
        let modifier_pos = start + index;
        let suffix = &spec[index..];
        if suffix == "*" {
            Modifier::Explode
        } else if let Some(digits) = suffix.strip_prefix(':') {
            let valid = !digits.is_empty()
                && digits.len() <= 4
                && digits.as_bytes()[0].is_ascii_digit()
                && digits.as_bytes()[0] != b'0'
                && digits.bytes().all(|b| b.is_ascii_digit());
            if !valid {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            let value = digits.parse::<usize>().unwrap_or(0);
            if !(1..=9999).contains(&value) {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            Modifier::Prefix(value)
        } else {
            return Err(Error::InvalidModifier { pos: modifier_pos });
        }
    } else {
        Modifier::None
    };

    if seen.iter().any(|n| n == &name) {
        return Err(Error::DuplicateVariable { pos: start });
    }
    seen.push(name.clone());

    Ok(Specifier {
        name,
        start,
        modifier,
    })
}

fn parse_expression(content: &str, content_start: usize, close_pos: usize) -> Result<(Operator, Vec<Specifier>), Error> {
    let mut operator = Operator::Simple;
    let mut body_start = 0;

    if let Some(first) = content.chars().next() {
        if let Some(op) = Operator::from_char(first) {
            operator = op;
            body_start = first.len_utf8();
        } else if matches!(first, '>' | '<' | '=' | '!' | '$' | '@' | '|' | '%' | '^' | '`' | '~') {
            return Err(Error::UnsupportedOperator {
                pos: content_start,
            });
        }
    }

    let body = &content[body_start..];
    let body_abs = content_start + body_start;
    if body.is_empty() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::new();
    let mut relative = 0;
    for part in body.split(',') {
        let spec = parse_specifier(part, body_abs + relative, &mut seen)?;
        specs.push(spec);
        relative += part.len() + 1;
    }
    Ok((operator, specs))
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(input: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() && bytes[i + 1].is_ascii_hexdigit() && bytes[i + 2].is_ascii_hexdigit() {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
        } else if is_unreserved(b) || (reserved && is_reserved(b)) {
            out.push(b as char);
            i += 1;
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
            i += 1;
        }
    }

    out
}

fn encode_named(name: &str, value: &str, operator: Operator) -> String {
    let encoded = encode(value, operator.uses_reserved());
    if operator == Operator::Semicolon && encoded.is_empty() {
        name.to_string()
    } else {
        format!("{name}={encoded}")
    }
}

fn selected_binding<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn expand_specifier(spec: &Specifier, operator: Operator, value: &Value) -> Result<Vec<String>, Error> {
    if matches!(spec.modifier, Modifier::Prefix(_)) && !matches!(value, Value::Scalar(_)) {
        return Err(Error::TypeMismatch { pos: spec.start });
    }

    match value {
        Value::Scalar(s) => {
            let sliced;
            let text = match spec.modifier {
                Modifier::Prefix(n) => {
                    sliced = s.chars().take(n).collect::<String>();
                    sliced.as_str()
                }
                _ => s.as_str(),
            };
            let field = if operator.names_scalar_fields() {
                encode_named(&spec.name, text, operator)
            } else {
                encode(text, operator.uses_reserved())
            };
            Ok(vec![field])
        }
        Value::List(items) => {
            let exploded = matches!(spec.modifier, Modifier::Explode);
            if items.is_empty() {
                return Ok(Vec::new());
            }
            if exploded {
                Ok(items
                    .iter()
                    .map(|item| {
                        if operator.names_scalar_fields() {
                            encode_named(&spec.name, item, operator)
                        } else {
                            encode(item, operator.uses_reserved())
                        }
                    })
                    .collect())
            } else {
                let joined = items
                    .iter()
                    .map(|item| encode(item, operator.uses_reserved()))
                    .collect::<Vec<_>>()
                    .join(",");
                Ok(vec![if operator.names_scalar_fields() {
                    if operator == Operator::Semicolon && joined.is_empty() {
                        spec.name.clone()
                    } else {
                        format!("{}={joined}", spec.name)
                    }
                } else {
                    joined
                }])
            }
        }
        Value::Map(entries) => {
            let mut keys = Vec::<&str>::new();
            for (key, _) in entries {
                if keys.iter().any(|seen| *seen == key) {
                    return Err(Error::DuplicateKey { pos: spec.start });
                }
                keys.push(key);
            }
            if entries.is_empty() {
                return Ok(Vec::new());
            }

            let exploded = matches!(spec.modifier, Modifier::Explode);
            if exploded {
                Ok(entries
                    .iter()
                    .map(|(key, val)| {
                        let k = encode(key, operator.uses_reserved());
                        let v = encode(val, operator.uses_reserved());
                        if operator == Operator::Semicolon && v.is_empty() {
                            k
                        } else {
                            format!("{k}={v}")
                        }
                    })
                    .collect())
            } else {
                let mut components = Vec::with_capacity(entries.len() * 2);
                for (key, val) in entries {
                    components.push(encode(key, operator.uses_reserved()));
                    components.push(encode(val, operator.uses_reserved()));
                }
                let joined = components.join(",");
                Ok(vec![if operator.names_scalar_fields() {
                    if operator == Operator::Semicolon && joined.is_empty() {
                        spec.name.clone()
                    } else {
                        format!("{}={joined}", spec.name)
                    }
                } else {
                    joined
                }])
            }
        }
    }
}

fn expand_expression(
    operator: Operator,
    specifiers: &[Specifier],
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let mut fields = Vec::new();
    for spec in specifiers {
        if let Some(value) = selected_binding(&spec.name, vars) {
            fields.extend(expand_specifier(spec, operator, value)?);
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut out = String::new();
    if let Some(prefix) = operator.prefix() {
        out.push(prefix);
    }
    out.push_str(&fields.join(&operator.separator().to_string()));
    Ok(out)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::<(usize, usize, Operator, Vec<Specifier>)>::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = bytes[i + 1..]
                    .iter()
                    .position(|&b| b == b'}')
                    .map(|offset| i + 1 + offset);

                let Some(close) = close else {
                    return Err(Error::Unclosed { pos: open });
                };

                if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                let content = &template[open + 1..close];
                let (operator, specifiers) =
                    parse_expression(content, open + 1, close)?;
                expressions.push((open, close + 1, operator, specifiers));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut copied = 0;
    for (start, end, operator, specifiers) in expressions {
        out.push_str(&template[copied..start]);
        out.push_str(&expand_expression(operator, &specifiers, vars)?);
        copied = end;
    }
    out.push_str(&template[copied..]);
    Ok(out)
}
