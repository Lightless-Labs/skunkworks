#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

impl Operator {
    fn separator(self) -> char {
        match self {
            Operator::Simple | Operator::Plus | Operator::Hash => ',',
            Operator::Dot => '.',
            Operator::Slash => '/',
            Operator::Semicolon => ';',
            Operator::Question | Operator::Ampersand => '&',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Operator::Simple | Operator::Plus => None,
            Operator::Hash => Some('#'),
            Operator::Dot => Some('.'),
            Operator::Slash => Some('/'),
            Operator::Semicolon => Some(';'),
            Operator::Question => Some('?'),
            Operator::Ampersand => Some('&'),
        }
    }

    fn reserved(self) -> bool {
        matches!(self, Operator::Plus | Operator::Hash)
    }

    fn named_fields(self) -> bool {
        matches!(
            self,
            Operator::Semicolon | Operator::Question | Operator::Ampersand
        )
    }
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Specifier<'a> {
    name: &'a str,
    modifier: Modifier,
    pos: usize,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_specifier<'a>(raw: &'a str, pos: usize) -> Result<Specifier<'a>, Error> {
    if raw.is_empty() {
        return Err(Error::EmptyVariable { pos });
    }

    let modifier_at = raw.bytes().position(|b| b == b'*' || b == b':');
    let name_end = modifier_at.unwrap_or(raw.len());
    let name = &raw[..name_end];

    if name.is_empty() {
        return Err(Error::InvalidName { pos });
    }

    let name_bytes = name.as_bytes();
    if !is_name_start(name_bytes[0]) {
        return Err(Error::InvalidName { pos });
    }
    for (i, &byte) in name_bytes.iter().enumerate().skip(1) {
        if !is_name_byte(byte) {
            return Err(Error::InvalidName { pos: pos + i });
        }
    }

    let modifier = if let Some(index) = modifier_at {
        let modifier_pos = pos + index;
        let suffix = &raw[index..];
        if suffix == "*" {
            Modifier::Explode
        } else if let Some(digits) = suffix.strip_prefix(':') {
            let bytes = digits.as_bytes();
            if bytes.is_empty()
                || bytes.len() > 4
                || bytes[0] == b'0'
                || !bytes.iter().all(u8::is_ascii_digit)
            {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            let mut value = 0usize;
            for &byte in bytes {
                value = value * 10 + usize::from(byte - b'0');
            }
            if !(1..=9999).contains(&value) {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            Modifier::Prefix(value)
        } else {
            return Err(Error::InvalidModifier { pos: modifier_pos });
        }
    } else {
        Modifier::None
    };

    Ok(Specifier {
        name,
        modifier,
        pos,
    })
}

fn validate_expression<'a>(
    content: &'a str,
    content_pos: usize,
    close_pos: usize,
) -> Result<(Operator, Vec<Specifier<'a>>), Error> {
    let (operator, body, body_pos) = if let Some(first) = content.as_bytes().first().copied() {
        let operator = match first {
            b'+' => Some(Operator::Plus),
            b'#' => Some(Operator::Hash),
            b'.' => Some(Operator::Dot),
            b'/' => Some(Operator::Slash),
            b';' => Some(Operator::Semicolon),
            b'?' => Some(Operator::Question),
            b'&' => Some(Operator::Ampersand),
            _ => None,
        };
        if let Some(operator) = operator {
            (operator, &content[1..], content_pos + 1)
        } else {
            (Operator::Simple, content, content_pos)
        }
    } else {
        (Operator::Simple, content, content_pos)
    };

    if body.is_empty() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specifiers = Vec::new();
    let mut start = 0;
    for (i, byte) in body.bytes().enumerate() {
        if byte == b',' {
            specifiers.push(parse_specifier(
                &body[start..i],
                body_pos + start,
            )?);
            start = i + 1;
        }
    }
    specifiers.push(parse_specifier(&body[start..], body_pos + start)?);

    let mut seen: Vec<&str> = Vec::new();
    for specifier in &specifiers {
        if seen.iter().any(|name| *name == specifier.name) {
            return Err(Error::DuplicateVariable { pos: specifier.pos });
        }
        seen.push(specifier.name);
    }

    Ok((operator, specifiers))
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn upper_hex(byte: u8) -> u8 {
    byte.to_ascii_uppercase()
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
            output.push('%');
            output.push(char::from(upper_hex(bytes[i + 1])));
            output.push(char::from(upper_hex(bytes[i + 2])));
            i += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                    | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || reserved_byte {
            output.push(char::from(byte));
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(char::from(HEX[(byte >> 4) as usize]));
            output.push(char::from(HEX[(byte & 0x0f) as usize]));
        }
        i += 1;
    }

    output
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    if operator == Operator::Semicolon && value.is_empty() {
        name.to_owned()
    } else {
        let mut field = String::with_capacity(name.len() + value.len() + 1);
        field.push_str(name);
        field.push('=');
        field.push_str(&value);
        field
    }
}

fn fields_for(
    specifier: &Specifier<'_>,
    value: &Value,
    operator: Operator,
) -> Result<Vec<String>, Error> {
    let reserved = operator.reserved();
    let explode = matches!(specifier.modifier, Modifier::Explode);

    match (&specifier.modifier, value) {
        (Modifier::Prefix(_), Value::List(_) | Value::Map(_)) => {
            return Err(Error::TypeMismatch { pos: specifier.pos });
        }
        _ => {}
    }

    match value {
        Value::Scalar(raw) => {
            let sliced;
            let value = match specifier.modifier {
                Modifier::Prefix(count) => {
                    sliced = raw.chars().take(count).collect::<String>();
                    sliced.as_str()
                }
                _ => raw.as_str(),
            };
            let encoded = encode(value, reserved);
            Ok(vec![if operator.named_fields() {
                named_field(specifier.name, encoded, operator)
            } else {
                encoded
            }])
        }
        Value::List(items) => {
            if items.is_empty() {
                return Ok(Vec::new());
            }
            if explode {
                Ok(items
                    .iter()
                    .map(|item| {
                        let encoded = encode(item, reserved);
                        if operator.named_fields() {
                            named_field(specifier.name, encoded, operator)
                        } else {
                            encoded
                        }
                    })
                    .collect())
            } else {
                let joined = items
                    .iter()
                    .map(|item| encode(item, reserved))
                    .collect::<Vec<_>>()
                    .join(",");
                Ok(vec![if operator.named_fields() {
                    named_field(specifier.name, joined, operator)
                } else {
                    joined
                }])
            }
        }
        Value::Map(entries) => {
            let mut seen: Vec<&str> = Vec::new();
            for (key, _) in entries {
                if seen.iter().any(|existing| *existing == key) {
                    return Err(Error::DuplicateKey { pos: specifier.pos });
                }
                seen.push(key);
            }

            if entries.is_empty() {
                return Ok(Vec::new());
            }

            if explode {
                Ok(entries
                    .iter()
                    .map(|(key, value)| {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        let mut field = String::with_capacity(
                            encoded_key.len() + encoded_value.len() + 1,
                        );
                        field.push_str(&encoded_key);
                        if operator != Operator::Semicolon || !encoded_value.is_empty() {
                            field.push('=');
                        }
                        field.push_str(&encoded_value);
                        field
                    })
                    .collect())
            } else {
                let mut components = Vec::with_capacity(entries.len() * 2);
                for (key, value) in entries {
                    components.push(encode(key, reserved));
                    components.push(encode(value, reserved));
                }
                let joined = components.join(",");
                Ok(vec![if operator.named_fields() {
                    named_field(specifier.name, joined, operator)
                } else {
                    joined
                }])
            }
        }
    }
}

fn expand_expression(
    operator: Operator,
    specifiers: &[Specifier<'_>],
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let mut fields = Vec::new();
    for specifier in specifiers {
        if let Some(value) = selected_value(specifier.name, vars) {
            fields.extend(fields_for(specifier, value, operator)?);
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    if let Some(prefix) = operator.prefix() {
        output.push(prefix);
    }
    output.push_str(&fields.join(&operator.separator().to_string()));
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let mut output = String::new();
    let mut cursor = 0;

    while cursor < template.len() {
        let next_open = template[cursor..].find('{').map(|i| cursor + i);
        let next_close = template[cursor..].find('}').map(|i| cursor + i);

        match (next_open, next_close) {
            (None, None) => {
                output.push_str(&template[cursor..]);
                break;
            }
            (Some(open), Some(close)) if close < open => {
                return Err(Error::UnexpectedClose { pos: close });
            }
            (Some(open), _) => {
                output.push_str(&template[cursor..open]);
                let close = match template[open + 1..].find('}') {
                    Some(i) => open + 1 + i,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(inner) = template[open + 1..close].find('{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + inner,
                    });
                }

                let content = &template[open + 1..close];
                let (operator, specifiers) =
                    validate_expression(content, open + 1, close)?;
                output.push_str(&expand_expression(operator, &specifiers, vars)?);
                cursor = close + 1;
            }
            (None, Some(close)) => {
                return Err(Error::UnexpectedClose { pos: close });
            }
        }
    }

    Ok(output)
}
