enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Specifier<'a> {
    name: &'a str,
    modifier: Modifier,
    pos: usize,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_specifier<'a>(raw: &'a str, pos: usize) -> Result<Specifier<'a>, Error> {
    if raw.is_empty() {
        return Err(Error::EmptyVariable { pos });
    }

    let modifier_at = raw.bytes().position(|b| b == b'*' || b == b':');
    let name_end = modifier_at.unwrap_or(raw.len());
    let name = &raw[..name_end];

    if name.is_empty() {
        return Err(Error::InvalidName { pos });
    }

    let name_bytes = name.as_bytes();
    if !is_name_start(name_bytes[0]) {
        return Err(Error::InvalidName { pos });
    }
    for (i, &byte) in name_bytes.iter().enumerate().skip(1) {
        if !is_name_byte(byte) {
            return Err(Error::InvalidName { pos: pos + i });
        }
    }

    let modifier = if let Some(index) = modifier_at {
        let modifier_pos = pos + index;
        let suffix = &raw[index..];
        if suffix == "*" {
            Modifier::Explode
        } else if let Some(digits) = suffix.strip_prefix(':') {
            let bytes = digits.as_bytes();
            if bytes.is_empty()
                || bytes.len() > 4
                || bytes[0] == b'0'
                || !bytes.iter().all(u8::is_ascii_digit)
            {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            let mut value = 0usize;
            for &byte in bytes {
                value = value * 10 + usize::from(byte - b'0');
            }
            if !(1..=9999).contains(&value) {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            Modifier::Prefix(value)
        } else {
            return Err(Error::InvalidModifier { pos: modifier_pos });
        }
    } else {
        Modifier::None
    };

    Ok(Specifier {
        name,
        modifier,
        pos,
    })
}

fn validate_expression<'a>(
    content: &'a str,
    content_pos: usize,
    close_pos: usize,
) -> Result<(Operator, Vec<Specifier<'a>>), Error> {
    let (operator, body, body_pos) = if let Some(first) = content.as_bytes().first().copied() {
        let operator = match first {
            b'+' => Some(Operator::Plus),
            b'#' => Some(Operator::Hash),
            b'.' => Some(Operator::Dot),
            b'/' => Some(Operator::Slash),
            b';' => Some(Operator::Semicolon),
            b'?' => Some(Operator::Question),
            b'&' => Some(Operator::Ampersand),
            _ => None,
        };
        if let Some(operator) = operator {
            (operator, &content[1..], content_pos + 1)
        } else {
            (Operator::Simple, content, content_pos)
        }
    } else {
        (Operator::Simple, content, content_pos)
    };

    if body.is_empty() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specifiers = Vec::new();
    let mut start = 0;
    for (i, byte) in body.bytes().enumerate() {
        if byte == b',' {
            specifiers.push(parse_specifier(
                &body[start..i],
                body_pos + start,
            )?);
            start = i + 1;
        }
    }
    specifiers.push(parse_specifier(&body[start..], body_pos + start)?);

    let mut seen: Vec<&str> = Vec::new();
    for specifier in &specifiers {
        if seen.iter().any(|name| *name == specifier.name) {
            return Err(Error::DuplicateVariable { pos: specifier.pos });
        }
        seen.push(specifier.name);
    }

    Ok((operator, specifiers))
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn upper_hex(byte: u8) -> u8 {
    byte.to_ascii_uppercase()
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
            output.push('%');
            output.push(char::from(upper_hex(bytes[i + 1])));
            output.push(char::from(upper_hex(bytes[i + 2])));
            i += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                    | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || reserved_byte {
            output.push(char::from(byte));
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(char::from(HEX[(byte >> 4) as usize]));
            output.push(char::from(HEX[(byte & 0x0f) as usize]));
        }
        i += 1;
    }

    output
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    if operator == Operator::Semicolon && value.is_empty() {
        name.to_owned()
    } else {
        let mut field = String::with_capacity(name.len() + value.len() + 1);
        field.push_str(name);
        field.push('=');
        field.push_str(&value);
        field
    }
}

fn fields_for(
    specifier: &Specifier<'_>,
    value: &Value,
    operator: Operator,
) -> Result<Vec<String>, Error> {
    let reserved = operator.reserved();
    let explode = matches!(specifier.modifier, Modifier::Explode);

    match (&specifier.modifier, value) {
        (Modifier::Prefix(_), Value::List(_) | Value::Map(_)) => {
            return Err(Error::TypeMismatch { pos: specifier.pos });
        }
        _ => {}
    }

    match value {
        Value::Scalar(raw) => {
            let sliced;
            let value = match specifier.modifier {
                Modifier::Prefix(count) => {
                    sliced = raw.chars().take(count).collect::<String>();
                    sliced.as_str()
                }
                _ => raw.as_str(),
            };
            let encoded = encode(value, reserved);
            Ok(vec![if operator.named_fields() {
                named_field(specifier.name, encoded, operator)
            } else {
                encoded
            }])
        }
        Value::List(items) => {
            if items.is_empty() {
                return Ok(Vec::new());
            }
            if explode {
                Ok(items
                    .iter()
                    .map(|item| {
                        let encoded = encode(item, reserved);
                        if operator.named_fields() {
                            named_field(specifier.name, encoded, operator)
                        } else {
                            encoded
                        }
                    })
                    .collect())
            } else {
                let joined = items
                    .iter()
                    .map(|item| encode(item, reserved))
                    .collect::<Vec<_>>()
                    .join(",");
                Ok(vec![if operator.named_fields() {
                    named_field(specifier.name, joined, operator)
                } else {
                    joined
                }])
            }
        }
        Value::Map(entries) => {
            let mut seen: Vec<&str> = Vec::new();
            for (key, _) in entries {
                if seen.iter().any(|existing| *existing == key) {
                    return Err(Error::DuplicateKey { pos: specifier.pos });
                }
                seen.push(key);
            }

            if entries.is_empty() {
                return Ok(Vec::new());
            }

            if explode {
                Ok(entries
                    .iter()
                    .map(|(key, value)| {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        let mut field = String::with_capacity(
                            encoded_key.len() + encoded_value.len() + 1,
                        );
                        field.push_str(&encoded_key);
                        if operator != Operator::Semicolon || !encoded_value.is_empty() {
                            field.push('=');
                        }
                        field.push_str(&encoded_value);
                        field
                    })
                    .collect())
            } else {
                let mut components = Vec::with_capacity(entries.len() * 2);
                for (key, value) in entries {
                    components.push(encode(key, reserved));
                    components.push(encode(value, reserved));
                }
                let joined = components.join(",");
                Ok(vec![if operator.named_fields() {
                    named_field(specifier.name, joined, operator)
                } else {
                    joined
                }])
            }
        }
    }
}

fn expand_expression(
    operator: Operator,
    specifiers: &[Specifier<'_>],
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let mut fields = Vec::new();
    for specifier in specifiers {
        if let Some(value) = selected_value(specifier.name, vars) {
            fields.extend(fields_for(specifier, value, operator)?);
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    if let Some(prefix) = operator.prefix() {
        output.push(prefix);
    }
    output.push_str(&fields.join(&operator.separator().to_string()));
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let mut output = String::new();
    let mut cursor = 0;

    while cursor < template.len() {
        let next_open = template[cursor..].find('{').map(|i| cursor + i);
        let next_close = template[cursor..].find('}').map(|i| cursor + i);

        match (next_open, next_close) {
            (None, None) => {
                output.push_str(&template[cursor..]);
                break;
            }
            (Some(open), Some(close)) if close < open => {
                return Err(Error::UnexpectedClose { pos: close });
            }
            (Some(open), _) => {
                output.push_str(&template[cursor..open]);
                let close = match template[open + 1..].find('}') {
                    Some(i) => open + 1 + i,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(inner) = template[open + 1..close].find('{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + inner,
                    });
                }

                let content = &template[open + 1..close];
                let (operator, specifiers) =
                    validate_expression(content, open + 1, close)?;
                output.push_str(&expand_expression(operator, &specifiers, vars)?);
                cursor = close + 1;
            }
            (None, Some(close)) => {
                return Err(Error::UnexpectedClose { pos: close });
            }
        }
    }

    Ok(output)
}

    }

    let bytes = spec.as_bytes();
    let split = bytes.iter().position(|&b| b == b'*' || b == b':');
    let name_end = split.unwrap_or(bytes.len());

    if name_end == 0 {
        return Err(Error::InvalidName { pos: start });
    }

    if !is_name_start(bytes[0]) {
        return Err(Error::InvalidName { pos: start });
    }

    for (i, &b) in bytes[..name_end].iter().enumerate().skip(1) {
        if !is_name_continue(b) {
            return Err(Error::InvalidName { pos: start + i });
        }
    }

    let name = spec[..name_end].to_string();

    let modifier = if let Some(index) = split {
        let modifier_pos = start + index;
        let suffix = &spec[index..];
        if suffix == "*" {
            Modifier::Explode
        } else if let Some(digits) = suffix.strip_prefix(':') {
            let valid = !digits.is_empty()
                && digits.len() <= 4
                && digits.as_bytes()[0].is_ascii_digit()
                && digits.as_bytes()[0] != b'0'
                && digits.bytes().all(|b| b.is_ascii_digit());
            if !valid {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            let value = digits.parse::<usize>().unwrap_or(0);
            if !(1..=9999).contains(&value) {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
            Modifier::Prefix(value)
        } else {
            return Err(Error::InvalidModifier { pos: modifier_pos });
        }
    } else {
        Modifier::None
    };

    if seen.iter().any(|n| n == &name) {
        return Err(Error::DuplicateVariable { pos: start });
    }
    seen.push(name.clone());

    Ok(Specifier {
        name,
        start,
        modifier,
    })
}

fn parse_expression(content: &str, content_start: usize, close_pos: usize) -> Result<(Operator, Vec<Specifier>), Error> {
    let mut operator = Operator::Simple;
    let mut body_start = 0;

    if let Some(first) = content.chars().next() {
        if let Some(op) = Operator::from_char(first) {
            operator = op;
            body_start = first.len_utf8();
        } else if matches!(first, '>' | '<' | '=' | '!' | '$' | '@' | '|' | '%' | '^' | '`' | '~') {
            return Err(Error::UnsupportedOperator {
                pos: content_start,
            });
        }
    }

    let body = &content[body_start..];
    let body_abs = content_start + body_start;
    if body.is_empty() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::new();
    let mut relative = 0;
    for part in body.split(',') {
        let spec = parse_specifier(part, body_abs + relative, &mut seen)?;
        specs.push(spec);
        relative += part.len() + 1;
    }
    Ok((operator, specs))
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(input: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() && bytes[i + 1].is_ascii_hexdigit() && bytes[i + 2].is_ascii_hexdigit() {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
        } else if is_unreserved(b) || (reserved && is_reserved(b)) {
            out.push(b as char);
            i += 1;
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
            i += 1;
        }
    }

    out
}

fn encode_named(name: &str, value: &str, operator: Operator) -> String {
    let encoded = encode(value, operator.uses_reserved());
    if operator == Operator::Semicolon && encoded.is_empty() {
        name.to_string()
    } else {
        format!("{name}={encoded}")
    }
}

fn selected_binding<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn expand_specifier(spec: &Specifier, operator: Operator, value: &Value) -> Result<Vec<String>, Error> {
    if matches!(spec.modifier, Modifier::Prefix(_)) && !matches!(value, Value::Scalar(_)) {
        return Err(Error::TypeMismatch { pos: spec.start });
    }

    match value {
        Value::Scalar(s) => {
            let sliced;
            let text = match spec.modifier {
                Modifier::Prefix(n) => {
                    sliced = s.chars().take(n).collect::<String>();
                    sliced.as_str()
                }
                _ => s.as_str(),
            };
            let field = if operator.names_scalar_fields() {
                encode_named(&spec.name, text, operator)
            } else {
                encode(text, operator.uses_reserved())
            };
            Ok(vec![field])
        }
        Value::List(items) => {
            let exploded = matches!(spec.modifier, Modifier::Explode);
            if items.is_empty() {
                return Ok(Vec::new());
            }
            if exploded {
                Ok(items
                    .iter()
                    .map(|item| {
                        if operator.names_scalar_fields() {
                            encode_named(&spec.name, item, operator)
                        } else {
                            encode(item, operator.uses_reserved())
                        }
                    })
                    .collect())
            } else {
                let joined = items
                    .iter()
                    .map(|item| encode(item, operator.uses_reserved()))
                    .collect::<Vec<_>>()
                    .join(",");
                Ok(vec![if operator.names_scalar_fields() {
                    if operator == Operator::Semicolon && joined.is_empty() {
                        spec.name.clone()
                    } else {
                        format!("{}={joined}", spec.name)
                    }
                } else {
                    joined
                }])
            }
        }
        Value::Map(entries) => {
            let mut keys = Vec::<&str>::new();
            for (key, _) in entries {
                if keys.iter().any(|seen| *seen == key) {
                    return Err(Error::DuplicateKey { pos: spec.start });
                }
                keys.push(key);
            }
            if entries.is_empty() {
                return Ok(Vec::new());
            }

            let exploded = matches!(spec.modifier, Modifier::Explode);
            if exploded {
                Ok(entries
                    .iter()
                    .map(|(key, val)| {
                        let k = encode(key, operator.uses_reserved());
                        let v = encode(val, operator.uses_reserved());
                        if operator == Operator::Semicolon && v.is_empty() {
                            k
                        } else {
                            format!("{k}={v}")
                        }
                    })
                    .collect())
            } else {
                let mut components = Vec::with_capacity(entries.len() * 2);
                for (key, val) in entries {
                    components.push(encode(key, operator.uses_reserved()));
                    components.push(encode(val, operator.uses_reserved()));
                }
                let joined = components.join(",");
                Ok(vec![if operator.names_scalar_fields() {
                    if operator == Operator::Semicolon && joined.is_empty() {
                        spec.name.clone()
                    } else {
                        format!("{}={joined}", spec.name)
                    }
                } else {
                    joined
                }])
            }
        }
    }
}

fn expand_expression(
    operator: Operator,
    specifiers: &[Specifier],
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let mut fields = Vec::new();
    for spec in specifiers {
        if let Some(value) = selected_binding(&spec.name, vars) {
            fields.extend(expand_specifier(spec, operator, value)?);
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut out = String::new();
    if let Some(prefix) = operator.prefix() {
        out.push(prefix);
    }
    out.push_str(&fields.join(&operator.separator().to_string()));
    Ok(out)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::<(usize, usize, Operator, Vec<Specifier>)>::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = bytes[i + 1..]
                    .iter()
                    .position(|&b| b == b'}')
                    .map(|offset| i + 1 + offset);

                let Some(close) = close else {
                    return Err(Error::Unclosed { pos: open });
                };

                if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                let content = &template[open + 1..close];
                let (operator, specifiers) =
                    parse_expression(content, open + 1, close)?;
                expressions.push((open, close + 1, operator, specifiers));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut copied = 0;
    for (start, end, operator, specifiers) in expressions {
        out.push_str(&template[copied..start]);
        out.push_str(&expand_expression(operator, &specifiers, vars)?);
        copied = end;
    }
    out.push_str(&template[copied..]);
    Ok(out)
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semicolon,
    Query,
    Ampersand,
}

impl Operator {
    fn from_byte(b: u8) -> Option<Self> {
        match b {
            b'+' => Some(Self::Plus),
            b'#' => Some(Self::Hash),
            b'.' => Some(Self::Dot),
            b'/' => Some(Self::Slash),
            b';' => Some(Self::Semicolon),
            b'?' => Some(Self::Query),
            b'&' => Some(Self::Ampersand),
            _ => None,
        }
    }

    fn separator(self) -> u8 {
        match self {
            Self::Simple | Self::Plus | Self::Hash => b',',
            Self::Dot => b'.',
            Self::Slash => b'/',
            Self::Semicolon => b';',
            Self::Query | Self::Ampersand => b'&',
        }
    }

    fn prefix(self) -> Option<u8> {
        match self {
            Self::Simple | Self::Plus => None,
            Self::Hash => Some(b'#'),
            Self::Dot => Some(b'.'),
            Self::Slash => Some(b'/'),
            Self::Semicolon => Some(b';'),
            Self::Query => Some(b'?'),
            Self::Ampersand => Some(b'&'),
        }
    }

    fn named_fields(self) -> bool {
        matches!(self, Self::Semicolon | Self::Query | Self::Ampersand)
    }

    fn reserved(self) -> bool {
        matches!(self, Self::Plus | Self::Hash)
    }
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Specifier<'a> {
    name: &'a str,
    name_pos: usize,
    start: usize,
    modifier: Modifier,
}

struct Expression<'a> {
    operator: Operator,
    specs: Vec<Specifier<'a>>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn parse_expression<'a>(
    template: &'a str,
    content_start: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();
    let mut start = content_start;
    let mut operator = Operator::Simple;

    if start < close {
        if let Some(op) = Operator::from_byte(bytes[start]) {
            operator = op;
            start += 1;
        } else if matches!(bytes[start], b'=' | b'!' | b'@' | b'|' | b'$' | b'%' | b'^') {
            return Err(Error::UnsupportedOperator { pos: start });
        }
    }

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let spec_bytes = &bytes[spec_start..spec_end];
        let modifier_offset = spec_bytes
            .iter()
            .position(|&b| b == b'*' || b == b':')
            .unwrap_or(spec_bytes.len());
        let name_end = spec_start + modifier_offset;

        if name_end == spec_start || !is_name_start(bytes[spec_start]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        let mut i = spec_start + 1;
        while i < name_end {
            if !is_name_continue(bytes[i]) {
                return Err(Error::InvalidName { pos: i });
            }
            i += 1;
        }

        let name = &template[spec_start..name_end];
        let modifier = if name_end == spec_end {
            Modifier::None
        } else {
            let modifier_pos = name_end;
            let suffix = &bytes[modifier_pos..spec_end];
            if suffix == b"*" {
                Modifier::Explode
            } else if suffix.first() == Some(&b':') {
                let digits = &suffix[1..];
                if digits.is_empty()
                    || digits.len() > 4
                    || digits[0] == b'0'
                    || !digits.iter().all(u8::is_ascii_digit)
                {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                let mut value = 0usize;
                for &digit in digits {
                    value = value
                        .checked_mul(10)
                        .and_then(|n| n.checked_add((digit - b'0') as usize))
                        .ok_or(Error::InvalidModifier { pos: modifier_pos })?;
                }
                if !(1..=9999).contains(&value) {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                Modifier::Prefix(value)
            } else {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
        };

        if seen.iter().any(|&previous| previous == name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        specs.push(Specifier {
            name,
            name_pos: spec_start,
            start: spec_start,
            modifier,
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression { operator, specs })
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(b: u8) -> bool {
    matches!(

        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(input: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = input.as_bytes();
    let mut result = String::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() {
            if hex_value(bytes[i + 1]).is_some() && hex_value(bytes[i + 2]).is_some() {
                result.push('%');
                result.push(HEX[hex_value(bytes[i + 1]).unwrap() as usize] as char);
                result.push(HEX[hex_value(bytes[i + 2]).unwrap() as usize] as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(b) || (reserved && is_reserved(b)) {
            result.push(b as char);
        } else {
            result.push('%');
            result.push(HEX[(b >> 4) as usize] as char);
            result.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }

    result
}

fn truncate_chars(value: &str, count: usize) -> &str {
    match value.char_indices().nth(count) {
        Some((index, _)) => &value[..index],
        None => value,
    }
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn encode_named(name: &str, value: &str, operator: Operator) -> String {
    let encoded = encode(value, operator.reserved());
    if operator == Operator::Semicolon && encoded.is_empty() {
        name.to_owned()
    } else {
        let mut field = String::with_capacity(name.len() + encoded.len() + 1);
        field.push_str(name);
        field.push('=');
        field.push_str(&encoded);
        field
    }
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expression.operator;
    let mut fields: Vec<String> = Vec::new();

    for spec in &expression.specs {
        let Some(value) = selected_value(spec.name, vars) else {
            continue;
        };

        match (spec.modifier, value) {
            (Modifier::Prefix(_), Value::List(_) | Value::Map(_)) => {
                return Err(Error::TypeMismatch { pos: spec.start });
            }
            (_, Value::Scalar(scalar)) => {
                let value = match spec.modifier {
                    Modifier::Prefix(count) => truncate_chars(scalar, count),
                    Modifier::None | Modifier::Explode => scalar.as_str(),
                };
                if operator.named_fields() {
                    fields.push(encode_named(spec.name, value, operator));
                } else {
                    fields.push(encode(value, operator.reserved()));
                }
            }
            (modifier, Value::List(items)) => {
                let explode = matches!(modifier, Modifier::Explode);
                if !explode {
                    if !items.is_empty() {
                        let joined = items
                            .iter()
                            .map(|item| encode(item, operator.reserved()))
                            .collect::<Vec<_>>()
                            .join(",");
                        if operator.named_fields() {
                            fields.push(if operator == Operator::Semicolon && joined.is_empty() {
                                spec.name.to_owned()
                            } else {
                                format!("{}={}", spec.name, joined)
                            });
                        } else {
                            fields.push(joined);
                        }
                    }
                } else {
                    for item in items {
                        if operator.named_fields() {
                            fields.push(encode_named(spec.name, item, operator));
                        } else {
                            fields.push(encode(item, operator.reserved()));
                        }
                    }
                }
            }
            (modifier, Value::Map(entries)) => {
                let mut keys: Vec<&str> = Vec::new();
                for (key, _) in entries {
                    if keys.iter().any(|&previous| previous == key) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                    keys.push(key);
                }

                if entries.is_empty() {
                    continue;
                }

                if matches!(modifier, Modifier::Explode) {
                    for (key, value) in entries {
                        let encoded_key = encode(key, operator.reserved());
                        let encoded_value = encode(value, operator.reserved());
                        if operator == Operator::Semicolon && encoded_value.is_empty() {
                            fields.push(encoded_key);
                        } else {
                            fields.push(format!("{}={}", encoded_key, encoded_value));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, operator.reserved()));
                        components.push(encode(value, operator.reserved()));
                    }
                    let joined = components.join(",");
                    if operator.named_fields() {
                        fields.push(if operator == Operator::Semicolon && joined.is_empty() {
                            spec.name.to_owned()
                        } else {
                            format!("{}={}", spec.name, joined)
                        });
                    } else {
                        fields.push(joined);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    if let Some(prefix) = operator.prefix() {
        output.push(prefix as char);
    }
    output.push_str(&fields.join(&(operator.separator() as char).to_string()));
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions: Vec<(usize, usize, Expression<'_>)> = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let close = match bytes[open + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => open + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                let expression = parse_expression(template, open + 1, close)?;
                expressions.push((open, close + 1, expression));
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;
    for (open, end, expression) in &expressions {
        output.push_str(&template[literal_start..*open]);
        output.push_str(&expand_expression(expression, vars)?);
        literal_start = *end;
    }
    output.push_str(&template[literal_start..]);
    Ok(output)
}
