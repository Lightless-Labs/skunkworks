#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    ContextTooLarge { context: usize },
    CarriageReturn { side: Side, byte: usize },
    MissingFinalLf { side: Side },
    TooLarge { old_len: usize, new_len: usize },
}

fn split_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        Vec::new()
    } else {
        text[..text.len() - 1].split('\n').collect()
    }
}

fn validate(text: &str, side: Side) -> Result<(), DiffError> {
    if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn { side, byte });
    }
    if !text.is_empty() && !text.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side });
    }
    Ok(())
}

fn push_edit(edits: &mut Vec<Edit>, edit: Edit) {
    edits.push(edit);
}

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }

    validate(old, Side::Old)?;
    validate(new, Side::New)?;

    let old_lines = split_lines(old);
    let new_lines = split_lines(new);

    if old_lines.len() > 128 || new_lines.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old_lines.len(),
            new_len: new_lines.len(),
        });
    }

    let n = old_lines.len();
    let m = new_lines.len();
    let width = m.checked_add(1).expect("bounded line count");
    let height = n.checked_add(1).expect("bounded line count");
    let cells = height.checked_mul(width).expect("bounded table size");
    let mut lcs = vec![0usize; cells];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            let index = i * width + j;
            lcs[index] = if old_lines[i] == new_lines[j] {
                lcs[(i + 1) * width + (j + 1)] + 1
            } else {
                lcs[(i + 1) * width + j].max(lcs[i * width + (j + 1)])
            };
        }
    }

    let mut edits = Vec::new();
    let (mut i, mut j) = (0usize, 0usize);
    while i < n && j < m {
        if old_lines[i] == new_lines[j] {
            push_edit(
                &mut edits,
                Edit::Keep {
                    old_index: i,
                    new_index: j,
                    line: old_lines[i].to_owned(),
                },
            );
            i += 1;
            j += 1;
        } else {
            let delete_score = lcs[(i + 1) * width + j];
            let insert_score = lcs[i * width + (j + 1)];
            if delete_score > insert_score
                || (delete_score == insert_score
                    && (i.checked_add(j).expect("bounded indices") % 2 == 0))
            {
                push_edit(
                    &mut edits,
                    Edit::Delete {
                        old_index: i,
                        line: old_lines[i].to_owned(),
                    },
                );
                i += 1;
            } else {
                push_edit(
                    &mut edits,
                    Edit::Insert {
                        new_index: j,
                        line: new_lines[j].to_owned(),
                    },
                );
                j += 1;
            }
        }
    }

    while i < n {
        push_edit(
            &mut edits,
            Edit::Delete {
                old_index: i,
                line: old_lines[i].to_owned(),
            },
        );
        i += 1;
    }
    while j < m {
        push_edit(
            &mut edits,
            Edit::Insert {
                new_index: j,
                line: new_lines[j].to_owned(),
            },
        );
        j += 1;
    }

    let changed = edits.iter().any(|edit| !matches!(edit, Edit::Keep { .. }));
    if !changed {
        return Ok(String::new());
    }

    let mut groups = Vec::<(usize, usize)>::new();
    for (idx, edit) in edits.iter().enumerate() {
        if !matches!(edit, Edit::Keep { .. }) {
            if let Some((_, end)) = groups.last_mut() {
                if *end == idx {
                    *end = idx + 1;
                    continue;
                }
            }
            groups.push((idx, idx + 1));
        }
    }

    let merge_distance = context.checked_mul(2).expect("context is at most 8");
    let mut merged = Vec::<(usize, usize)>::new();
    for (start, end) in groups {
        if let Some((_, prev_end)) = merged.last_mut() {
            let keeps_between = edits[*prev_end..start]
                .iter()
                .filter(|edit| matches!(edit, Edit::Keep { .. }))
                .count();
            if keeps_between <= merge_distance {
                *prev_end = end;
                continue;
            }
        }
        merged.push((start, end));
    }

    let mut output = String::from("--- before\n+++ after\n");
    for (start, end) in merged {
        let mut hunk_start = start;
        let mut preceding = 0usize;
        while hunk_start > 0 && preceding < context {
            if matches!(edits[hunk_start - 1], Edit::Keep { .. }) {
                hunk_start -= 1;
                preceding += 1;
            } else {
                break;
            }
        }

        let mut hunk_end = end;
        let mut following = 0usize;
        while hunk_end < edits.len() && following < context {
            if matches!(edits[hunk_end], Edit::Keep { .. }) {
                hunk_end += 1;
                following += 1;
            } else {
                break;
            }
        }

        let mut old_count = 0usize;
        let mut new_count = 0usize;
        let (mut old_before, mut new_before) = (0usize, 0usize);
        for edit in &edits[..hunk_start] {
            match edit {
                Edit::Keep { .. } => {
                    old_before += 1;
                    new_before += 1;
                }
                Edit::Delete { .. } => old_before += 1,
                Edit::Insert { .. } => new_before += 1,
            }
        }
        for edit in &edits[hunk_start..hunk_end] {
            match edit {
                Edit::Keep { .. } => {
                    old_count += 1;
                    new_count += 1;
                }
                Edit::Delete { .. } => old_count += 1,
                Edit::Insert { .. } => new_count += 1,
            }
        }

        let old_start = if old_count > 0 {
            old_before + 1
        } else {
            old_before
        };
        let new_start = if new_count > 0 {
            new_before + 1
        } else {
            new_before
        };

        use std::fmt::Write;
        writeln!(
            output,
            "@@ -{},{} +{},{} @@",
            old_start, old_count, new_start, new_count
        )
        .expect("writing to String cannot fail");

        for edit in &edits[hunk_start..hunk_end] {
            match edit {
                Edit::Keep { line, .. } => {
                    output.push(' ');
                    output.push_str(line);
                    output.push('\n');
                }
                Edit::Delete { line, .. } => {
                    output.push('-');
                    output.push_str(line);
                    output.push('\n');
                }
                Edit::Insert { line, .. } => {
                    output.push('+');
                    output.push_str(line);
                    output.push('\n');
                }
            }
        }
    }

    Ok(output)
}
