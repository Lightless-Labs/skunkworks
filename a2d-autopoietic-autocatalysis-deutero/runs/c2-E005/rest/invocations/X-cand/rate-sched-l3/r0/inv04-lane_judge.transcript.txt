pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobRuntime {
    state: State,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<Job>,
    dependency_indices: Vec<Vec<usize>>,
    runtime: Vec<JobRuntime>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    has_cycle: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();

        for (index, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(index));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(index));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(index));
            }
            if ids.insert(job.id.as_str(), index).is_some() {
                return Err(Error::Duplicate(index));
            }
        }

        let mut dependency_indices = Vec::with_capacity(jobs.len());

        for (job_index, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut indices = Vec::with_capacity(job.dependencies.len());

            for (dependency_index, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency.as_str()) {
                    return Err(Error::DuplicateDependency(job_index, dependency_index));
                }
                match ids.get(dependency.as_str()) {
                    Some(&index) => indices.push(index),
                    None => return Err(Error::MissingDependency(job_index, dependency_index)),
                }
            }

            dependency_indices.push(indices);
        }

        if graph_has_cycle(&dependency_indices) {
            return Err(Error::Cycle);
        }

        let runtime = (0..jobs.len())
            .map(|_| JobRuntime {
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            dependency_indices,
            runtime,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            has_cycle: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;

        if self.has_cycle
            && self.runtime.iter().any(|job| {
                matches!(job.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for job in &mut self.runtime {
            job.state = State::Waiting { ready_at: now };
            job.attempts = 0;
        }

        self.has_cycle = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;

        for index in 0..self.jobs.len() {
            let ready_at = match self.runtime[index].state {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };

            if !self.dependency_indices[index]
                .iter()
                .all(|&dependency| matches!(self.runtime[dependency].state, State::Succeeded))
            {
                continue;
            }

            let effective_priority = if self.aging_ms == 0 {
                self.jobs[index].priority
            } else {
                let age_steps = (now - ready_at) / self.aging_ms;
                let priority = u64::from(self.jobs[index].priority).saturating_add(age_steps);
                priority.min(9) as u8
            };

            if best.map_or(true, |(_, best_priority)| effective_priority > best_priority) {
                best = Some((index, effective_priority));
            }
        }

        self.last_time = now;

        if let Some((index, _)) = best {
            let attempt = self.runtime[index].attempts.saturating_add(1);
            self.runtime[index].attempts = attempt;
            self.runtime[index].state = State::Running { attempt };
            Ok(Some(Ticket {
                id: self.jobs[index].id.clone(),
                attempt,
            }))
        } else {
            Ok(None)
        }
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let index = self
            .jobs
            .iter()
            .position(|job| job.id == id)
            .ok_or(Error::Unknown)?;

        let attempt = match self.runtime[index].state {
            State::Running { attempt } => attempt,
            _ => return Err(Error::NotRunning),
        };

        if success {
            self.runtime[index].state = State::Succeeded;
        } else if attempt == self.jobs[index].max_attempts {
            self.runtime[index].state = State::Failed;
            self.propagate_blocked();
        } else {
            let delay = self.retry_delay(attempt);
            self.runtime[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }

        self.last_time = now;
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let index = self
            .jobs
            .iter()
            .position(|job| job.id == id)
            .ok_or(Error::Unknown)?;

        Ok(self.runtime[index].state.clone())
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn retry_delay(&self, attempt: u8) -> u64 {
        let mut delay = self.retry_base;
        for _ in 1..attempt {
            if delay >= self.retry_cap {
                return self.retry_cap;
            }
            delay = delay.saturating_mul(2).min(self.retry_cap);
        }
        delay.min(self.retry_cap)
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut newly_blocked = Vec::new();

            for index in 0..self.jobs.len() {
                if !matches!(self.runtime[index].state, State::Waiting { .. }) {
                    continue;
                }

                if self.dependency_indices[index].iter().any(|&dependency| {
                    matches!(
                        self.runtime[dependency].state,
                        State::Failed | State::Blocked
                    )
                }) {
                    newly_blocked.push(index);
                }
            }

            if newly_blocked.is_empty() {
                break;
            }

            for index in newly_blocked {
                self.runtime[index].state = State::Blocked;
            }
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 16
        && bytes
            .iter()
            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-')
}

fn graph_has_cycle(dependencies: &[Vec<usize>]) -> bool {
    fn visit(
        node: usize,
        dependencies: &[Vec<usize>],
        marks: &mut [u8],
    ) -> bool {
        if marks[node] == 1 {
            return true;
        }
        if marks[node] == 2 {
            return false;
        }

        marks[node] = 1;
        for &dependency in &dependencies[node] {
            if visit(dependency, dependencies, marks) {
                return true;
            }
        }
        marks[node] = 2;
        false
    }

    let mut marks = vec![0; dependencies.len()];
    for node in 0..dependencies.len() {
        if visit(node, dependencies, &mut marks) {
            return true;
        }
    }
    false
}

pub struct Runner {
    jobs: Vec<Job>,
    dependency_indices: Vec<Vec<usize>>,
    states: Vec<State>,
    attempts: Vec<u8>,

    retry_base: u64,

}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if job.id.is_empty()
                || job.id.len() > 16
                || !job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
            {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if ids.insert(job.id.as_str(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut dependency_indices = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut deps = Vec::with_capacity(job.dependencies.len());
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(dep.as_str()) {
                    Some(&index) => deps.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            dependency_indices.push(deps);
        }

        fn visit(
            node: usize,
            deps: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[node] == 1 {
                return false;
            }
            if marks[node] == 2 {
                return true;
            }
            marks[node] = 1;
            for &dep in &deps[node] {
                if !visit(dep, deps, marks) {
                    return false;
                }
            }
            marks[node] = 2;
            true
        }

        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if !visit(i, &dependency_indices, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let states = vec![State::Waiting { ready_at: 0 }; jobs.len()];
        let attempts = vec![0; jobs.len()];
        Ok(Self {
            jobs,
            dependency_indices,
            states,
            attempts,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.cycle_started
            && self.states.iter().any(|state| {
                matches!(state, State::Waiting { .. } | State::Running { .. })

            return Err(Error::Busy);
        }

        self.states = vec![State::Waiting { ready_at: now }; self.jobs.len()];
        self.attempts.fill(0);
        self.last_time = now;
        self.cycle_started = true;
        Ok(())

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for i in 0..self.jobs.len() {
            let ready_at = match self.states[i] {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            if !self.dependency_indices[i]
                .iter()
                .all(|&dep| matches!(self.states[dep], State::Succeeded))
            {
                continue;
            }

            let priority = if self.aging_ms == 0 {
                self.jobs[i].priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                let effective = self.jobs[i].priority as u64 + age;
                effective.min(9) as u8
            };

            if best.map_or(true, |(_, p)| priority > p) {
                best = Some((i, priority));
            }
        }

        self.last_time = now;
        if let Some((i, _)) = best {
            self.attempts[i] += 1;
            let attempt = self.attempts[i];
            self.states[i] = State::Running { attempt };
            Ok(Some(Ticket {
                id: self.jobs[i].id.clone(),
                attempt,
            }))
        } else {
            Ok(None)
        }
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let index = match self.jobs.iter().position(|job| job.id == id) {
            Some(i) => i,
            None => return Err(Error::Unknown),
        };

        let attempt = match self.states[index] {
            State::Running { attempt } => attempt,
            _ => return Err(Error::NotRunning),
        };

        self.last_time = now;
        if success {
            self.states[index] = State::Succeeded;
        } else if attempt == self.jobs[index].max_attempts {
            self.states[index] = State::Failed;
            self.propagate_blocked();
        } else {
            let shift = u32::from(attempt.saturating_sub(1));
            let delay = self.retry_base.checked_shl(shift).unwrap_or(u64::MAX);
            let delay = delay.min(self.retry_cap);
            self.states[index] = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }
        Ok(())
    }

            .jobs
            .iter()
            .position(|job| job.id == id)
            .ok_or(Error::Unknown)?;
        Ok(self.states[index].clone())
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut changed = false;
            for i in 0..self.jobs.len() {
                if matches!(self.states[i], State::Waiting { .. })
                    && self.dependency_indices[i].iter().any(|&dep| {
                        matches!(self.states[dep], State::Failed | State::Blocked)
                    })
                {
                    self.states[i] = State::Blocked;
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
    }
}

    retry_base: u64,
    retry_cap: u64,

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        for (index, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(index));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(index));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(index));
            }
            if jobs[..index].iter().any(|prior| prior.id == job.id) {
                return Err(Error::Duplicate(index));
            }
        }

        let mut dependency_indices = Vec::with_capacity(jobs.len());
        for (job_index, job) in jobs.iter().enumerate() {
            let mut indices = Vec::with_capacity(job.dependencies.len());
            for (dependency_index, dependency) in job.dependencies.iter().enumerate() {
                if job.dependencies[..dependency_index]
                    .iter()
                    .any(|prior| prior == dependency)
                {
                    return Err(Error::DuplicateDependency(job_index, dependency_index));
                }
                let Some(found) = jobs.iter().position(|candidate| candidate.id == *dependency)
                else {
                    return Err(Error::MissingDependency(job_index, dependency_index));
                };
                indices.push(found);
            }
            dependency_indices.push(indices);
        }

        if contains_cycle(&dependency_indices) {
            return Err(Error::Cycle);
        }

        let states = jobs
            .iter()
            .map(|_| JobState {
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            dependency_indices,
            states,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            has_cycle: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;
        if self.has_cycle
            && self.states.iter().any(|job_state| {
                matches!(job_state.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for job_state in &mut self.states {
            job_state.state = State::Waiting { ready_at: now };
            job_state.attempts = 0;
        }
        self.has_cycle = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for index in 0..self.jobs.len() {
            let State::Waiting { ready_at } = self.states[index].state else {
                continue;
            };
            if ready_at > now
                || self.dependency_indices[index].iter().any(|&dependency| {
                    !matches!(self.states[dependency].state, State::Succeeded)
                })
            {
                continue;
            }

            let effective_priority = if self.aging_ms == 0 {
                self.jobs[index].priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                let aged = u64::from(self.jobs[index].priority).saturating_add(age);
                aged.min(9) as u8
            };

            if best.is_none_or(|(_, best_priority)| effective_priority > best_priority) {
                best = Some((index, effective_priority));
            }
        }

        let ticket = if let Some((index, _)) = best {
            let attempt = self.states[index].attempts + 1;
            self.states[index].attempts = attempt;
            self.states[index].state = State::Running { attempt };
            Some(Ticket {
                id: self.jobs[index].id.clone(),
                attempt,
            })
        } else {
            None
        };

        self.last_time = now;
        Ok(ticket)
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let Some(index) = self.jobs.iter().position(|job| job.id == id) else {
            return Err(Error::Unknown);
        };
        let State::Running { .. } = self.states[index].state else {
            return Err(Error::NotRunning);
        };

        if success {
            self.states[index].state = State::Succeeded;
        } else if self.states[index].attempts == self.jobs[index].max_attempts {
            self.states[index].state = State::Failed;
        } else {
            let delay = retry_delay(
                self.states[index].attempts,
                self.retry_base,
                self.retry_cap,
            );
            self.states[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }

        if matches!(self.states[index].state, State::Failed) {
            self.propagate_blocked();
        }

        self.last_time = now;
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }
        let Some(index) = self.jobs.iter().position(|job| job.id == id) else {
            return Err(Error::Unknown);
        };
        Ok(self.states[index].state.clone())
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut newly_blocked = Vec::new();
            for index in 0..self.jobs.len() {
                if !matches!(self.states[index].state, State::Waiting { .. }) {
                    continue;
                }
                if self.dependency_indices[index].iter().any(|&dependency| {
                    matches!(
                        self.states[dependency].state,
                        State::Failed | State::Blocked
                    )
                }) {
                    newly_blocked.push(index);
                }
            }

            if newly_blocked.is_empty() {
                break;
            }
            for index in newly_blocked {
                self.states[index].state = State::Blocked;
            }
        }
    }
}

fn valid_id(id: &str) -> bool {
    (1..=16).contains(&id.len())
        && id
            .bytes()
            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-')
}

fn contains_cycle(dependencies: &[Vec<usize>]) -> bool {
    fn visit(node: usize, dependencies: &[Vec<usize>], colors: &mut [u8]) -> bool {
        if colors[node] == 1 {
            return true;
        }
        if colors[node] == 2 {
            return false;
        }
        colors[node] = 1;
        for &dependency in &dependencies[node] {
            if visit(dependency, dependencies, colors) {
                return true;
            }
        }
        colors[node] = 2;
        false
    }

    let mut colors = vec![0; dependencies.len()];
    (0..dependencies.len()).any(|node| visit(node, dependencies, &mut colors))
}

fn retry_delay(attempt: u8, retry_base: u64, retry_cap: u64) -> u64 {
    let mut delay = retry_base;
    for _ in 1..attempt {
        if delay >= retry_cap || delay > retry_cap / 2 {
            return retry_cap;
        }
        delay *= 2;
    }
    delay.min(retry_cap)
}

pub enum State {

    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {

    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct Entry {
    job: Job,
    dependencies: Vec<usize>,
    attempts: u8,
    state: Option<State>,
}

pub struct Runner {
    entries: Vec<Entry>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            let valid_id = !job.id.is_empty()
                && job.id.len() <= 16
                && job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-');
            if !valid_id {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if ids.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            ids.insert(job.id.clone(), i);
        }

        let mut entries = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.into_iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut dependencies = Vec::with_capacity(job.dependencies.len());
            for (j, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency.clone()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                let Some(&index) = ids.get(dependency) else {
                    return Err(Error::MissingDependency(i, j));
                };
                dependencies.push(index);
            }
            entries.push(Entry {
                job,
                dependencies,
                attempts: 0,
                state: None,
            });
        }

        fn visit(index: usize, entries: &[Entry], marks: &mut [u8]) -> bool {
            if marks[index] == 1 {
                return true;
            }
            if marks[index] == 2 {
                return false;
            }
            marks[index] = 1;
            for &dependency in &entries[index].dependencies {
                if visit(dependency, entries, marks) {
                    return true;
                }
            }
            marks[index] = 2;
            false
        }

        let mut marks = vec![0; entries.len()];
        for i in 0..entries.len() {
            if visit(i, &entries, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        Ok(Self {
            entries,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.cycle_started
            && self.entries.iter().any(|entry| {
                matches!(entry.state, Some(State::Waiting { .. } | State::Running { .. }))
            })
        {
            return Err(Error::Busy);
        }

        for entry in &mut self.entries {
            entry.attempts = 0;
            entry.state = Some(State::Waiting { ready_at: now });
        }
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut chosen: Option<(usize, u8)> = None;
        for (i, entry) in self.entries.iter().enumerate() {
            let ready_at = match entry.state {
                Some(State::Waiting { ready_at }) if ready_at <= now => ready_at,
                _ => continue,
            };
            if !entry.dependencies.iter().all(|&dependency| {
                matches!(
                    self.entries[dependency].state,
                    Some(State::Succeeded)
                )
            }) {
                continue;
            }

            let age = if self.aging_ms == 0 {
                0
            } else {
                ((now - ready_at) / self.aging_ms).min(9) as u8
            };
            let effective = entry.job.priority.saturating_add(age).min(9);
            if chosen.map_or(true, |(best_i, best_priority)| {
                effective > best_priority || (effective == best_priority && i < best_i)
            }) {
                chosen = Some((i, effective));
            }
        }

        self.last_time = now;
        let Some((index, _)) = chosen else {
            return Ok(None);
        };
        let entry = &mut self.entries[index];
        entry.attempts += 1;
        entry.state = Some(State::Running {
            attempt: entry.attempts,
        });
        Ok(Some(Ticket {
            id: entry.job.id.clone(),
            attempt: entry.attempts,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(index) = self.entries.iter().position(|entry| entry.job.id == id) else {
            return Err(Error::Unknown);
        };
        if !matches!(self.entries[index].state, Some(State::Running { .. })) {
            return Err(Error::NotRunning);
        }

        self.last_time = now;
        if success {
            self.entries[index].state = Some(State::Succeeded);
        } else if self.entries[index].attempts == self.entries[index].job.max_attempts {
            self.entries[index].state = Some(State::Failed);
            self.propagate_blocked();
        } else {
            let attempt = self.entries[index].attempts;
            let mut delay = self.retry_base;
            for _ in 1..attempt {
                delay = delay.saturating_mul(2).min(self.retry_cap);
            }
            delay = delay.min(self.retry_cap);
            self.entries[index].state = Some(State::Waiting {
                ready_at: now.saturating_add(delay),
            });
        }
        Ok(())
    }

    fn propagate_blocked(&mut self) {
        loop {
            let to_block: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, entry)| {
                    if !matches!(entry.state, Some(State::Waiting { .. })) {
                        return None;
                    }
                    entry
                        .dependencies
                        .iter()
                        .any(|&dependency| {
                            matches!(
                                self.entries[dependency].state,
                                Some(State::Failed | State::Blocked)
                            )
                        })
                        .then_some(i)
                })
                .collect();

            if to_block.is_empty() {
                break;
            }
            for i in to_block {
                self.entries[i].state = Some(State::Blocked);
            }
        }
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(entry) = self.entries.iter().find(|entry| entry.job.id == id) else {
            return Err(Error::Unknown);
        };
        entry.state.clone().ok_or(Error::NoCycle)
    }
}
