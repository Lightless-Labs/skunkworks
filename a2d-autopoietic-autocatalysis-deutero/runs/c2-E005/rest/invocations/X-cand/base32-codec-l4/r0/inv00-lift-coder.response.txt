#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut combined = self.pending.clone();
        combined.extend_from_slice(input);

        let complete_len = (combined.len() / 5) * 5;
        let output = encode_bytes(&combined[..complete_len], self.config.alphabet);

        self.pending = combined[complete_len..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let output = encode_final(&self.pending, self.config);
        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn encode_bytes(input: &[u8], alphabet: Alphabet) -> String {
    let symbols = alphabet_bytes(alphabet);
    let mut output = String::with_capacity(input.len() * 8 / 5);

    for chunk in input.chunks_exact(5) {
        let b0 = chunk[0] as u32;
        let b1 = chunk[1] as u32;
        let b2 = chunk[2] as u32;
        let b3 = chunk[3] as u32;
        let b4 = chunk[4] as u32;

        let values = [
            b0 >> 3,
            ((b0 & 0x07) << 2) | (b1 >> 6),
            (b1 >> 1) & 31,
            ((b1 & 0x01) << 4) | (b2 >> 4),
            ((b2 & 0x0f) << 1) | (b3 >> 7),
            (b3 >> 2) & 31,
            ((b3 & 0x03) << 3) | (b4 >> 5),
            b4 & 31,
        ];

        for value in values {
            output.push(symbols[value as usize] as char);
        }
    }

    output
}

fn encode_final(input: &[u8], config: Config) -> String {
    if input.is_empty() {
        return String::new();
    }

    let symbols = alphabet_bytes(config.alphabet);
    let bit_count = input.len() * 8;
    let symbol_count = (bit_count + 4) / 5;
    let mut output = String::with_capacity(if config.padding == Padding::Required {
        8
    } else {
        symbol_count
    });

    for symbol_index in 0..symbol_count {
        let mut value = 0u8;
        for bit_index in 0..5 {
            let stream_bit = symbol_index * 5 + bit_index;
            value <<= 1;
            if stream_bit < bit_count {
                let byte = input[stream_bit / 8];
                let bit = (byte >> (7 - (stream_bit % 8))) & 1;
                value |= bit;
            }
        }
        output.push(symbols[value as usize] as char);
    }

    if config.padding == Padding::Required {
        while output.len() % 8 != 0 {
            output.push('=');
        }
    }

    output
}

pub struct Decoder {
    config: Config,
    position: usize,
    data_count: usize,
    bit_buffer: u8,
    bit_count: u8,
    padding_count: usize,
    first_padding_offset: Option<usize>,
    expected_padding: Option<usize>,
    last_data_offset: Option<usize>,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            position: 0,
            data_count: 0,
            bit_buffer: 0,
            bit_count: 0,
            padding_count: 0,
            first_padding_offset: None,
            expected_padding: None,
            last_data_offset: None,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            if byte == b'=' {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }

                if next.first_padding_offset.is_none() {
                    let residue = next.data_count % 8;
                    let expected = match residue {
                        2 => 6,
                        4 => 4,
                        5 => 3,
                        7 => 1,
                        _ => return Err(StreamError::Padding { offset }),
                    };
                    next.first_padding_offset = Some(offset);
                    next.expected_padding = Some(expected);
                }

                let expected = next.expected_padding.unwrap();
                if next.padding_count >= expected {
                    return Err(StreamError::Padding { offset });
                }

                next.padding_count += 1;
                next.position += 1;
                continue;
            }

            let (value, canonical_byte) = match decode_symbol(byte, next.config.alphabet) {
                Some(found) => found,
                None => return Err(StreamError::InvalidByte { offset, byte }),
            };

            if next.config.canonical && byte != canonical_byte {
                return Err(StreamError::NonCanonical {
                    offset,
                    expected: canonical_byte,
                });
            }

            if next.first_padding_offset.is_some() {
                return Err(StreamError::Padding { offset });
            }

            next.data_count += 1;
            next.last_data_offset = Some(offset);
            next.bit_buffer = (next.bit_buffer << 5) | value;
            next.bit_count += 5;

            while next.bit_count >= 8 {
                next.bit_count -= 8;
                output.push((next.bit_buffer >> next.bit_count) as u8);
                if next.bit_count == 0 {
                    next.bit_buffer = 0;
                } else {
                    next.bit_buffer &= (1u16.wrapping_shl(next.bit_count as u32) - 1) as u8;
                }
            }

            next.position += 1;
        }

        self.copy_from(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
            return Err(StreamError::InvalidLength {
                offset: self.position,
            });
        }

        if self.config.padding == Padding::Required {
            let expected = match residue {
                0 => 0,
                2 => 6,
                4 => 4,
                5 => 3,
                7 => 1,
                _ => unreachable!(),
            };
            if self.padding_count != expected {
                return Err(StreamError::Padding {
                    offset: self.first_padding_offset.unwrap_or(self.position),
                });
            }
        }

        let unused_bits = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => unreachable!(),
        };
        if unused_bits != 0 && self.bit_buffer != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(self.position),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            position: self.position,
            data_count: self.data_count,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            padding_count: self.padding_count,
            first_padding_offset: self.first_padding_offset,
            expected_padding: self.expected_padding,
            last_data_offset: self.last_data_offset,
            closed: self.closed,
        }
    }

    fn copy_from(&mut self, other: Self) {
        self.position = other.position;
        self.data_count = other.data_count;
        self.bit_buffer = other.bit_buffer;
        self.bit_count = other.bit_count;
        self.padding_count = other.padding_count;
        self.first_padding_offset = other.first_padding_offset;
        self.expected_padding = other.expected_padding;
        self.last_data_offset = other.last_data_offset;
        self.closed = other.closed;
    }
}

fn decode_symbol(byte: u8, alphabet: Alphabet) -> Option<(u8, u8)> {
    let upper = byte.to_ascii_uppercase();
    let symbols = alphabet_bytes(alphabet);
    symbols
        .iter()
        .position(|&symbol| symbol == upper)
        .map(|value| (value as u8, upper))
}
