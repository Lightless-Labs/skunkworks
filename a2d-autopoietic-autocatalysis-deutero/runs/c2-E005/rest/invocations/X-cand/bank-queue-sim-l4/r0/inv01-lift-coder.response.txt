#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(q + u128::from(round_up))
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    ever_started: bool,
    waited: u64,
    queue_since: u64,
    pause_time: Option<u64>,
    active: bool,
}

#[derive(Clone, Copy)]
struct Running {
    index: usize,
    since: u64,
    work_at: u64,
    finish: u64,
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn best_waiter(waiting: &[usize], jobs: &[Job]) -> Option<usize> {
    waiting.iter().copied().min_by(|&a, &b| {
        let x = &jobs[a];
        let y = &jobs[b];
        x.customer.priority.cmp(&y.customer.priority)
            .then_with(|| y.pause_time.is_some().cmp(&x.pause_time.is_some()))
            .then_with(|| match (x.pause_time, y.pause_time) {
                (Some(px), Some(py)) => px.cmp(&py),
                (None, None) => x.customer.arrival.cmp(&y.customer.arrival)
                    .then_with(|| y.customer.id.cmp(&x.customer.id)),
                _ => std::cmp::Ordering::Equal,
            })
            .then_with(|| x.customer.id.cmp(&y.customer.id))
    })
}

fn deadline_order(a: usize, b: usize, jobs: &[Job]) -> std::cmp::Ordering {
    jobs[a].deadline.cmp(&jobs[b].deadline)
        .then_with(|| jobs[a].customer.id.cmp(&jobs[b].customer.id))
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::<u32>::new();
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow { index, id: c.id });
            }
        }
    }

    let mut stats = Stats {
        accepted: 0, balked: 0, reneged: 0, completed: 0, started: 0,
        preemptions: 0, elapsed: 0, busy_ticks: 0, wait_ticks: 0,
        turnaround_ticks: 0, mean_wait_milli: 0, mean_turnaround_milli: 0,
        utilization_bps: 0, completion_order: Vec::new(),
    };
    let mut events = Vec::new();
    if customers.is_empty() {
        return Ok(Report { events, stats });
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| customers[a].arrival.cmp(&customers[b].arrival)
        .then_with(|| customers[a].priority.cmp(&customers[b].priority))
        .then_with(|| customers[b].id.cmp(&customers[a].id)));

    let mut jobs: Vec<Job> = customers.iter().cloned().map(|customer| Job {
        deadline: customer.patience.and_then(|p| customer.arrival.checked_add(p)),
        remaining: customer.service,
        customer,
        ever_started: false,
        waited: 0,
        queue_since: 0,
        pause_time: None,
        active: false,
    }).collect();

    let mut waiting: Vec<usize> = Vec::new();
    let mut running: Option<Running> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut cursor = 0usize;
    let mut now = customers.iter().map(|c| c.arrival).min().unwrap();
    let mut last_event_time = 0u64;

    loop {
        if let Some(r) = running {
            if r.finish == now {
                let id = jobs[r.index].customer.id;
                emit(&mut events, now, id, Kind::Finish);
                last_event_time = now;
                jobs[r.index].active = false;
                stats.completed += 1;
                stats.turnaround_ticks += (now - jobs[r.index].customer.arrival) as u128;
                stats.completion_order.push(id);
                running = None;
            }
        }

        let mut expired: Vec<usize> = waiting.iter().copied()
            .filter(|&i| jobs[i].deadline.is_some_and(|d| d <= now)).collect();
        expired.sort_by(|&a, &b| deadline_order(a, b, &jobs));
        for i in expired {
            if let Some(pos) = waiting.iter().position(|&x| x == i) {
                waiting.remove(pos);
                let segment = now - jobs[i].queue_since;
                jobs[i].waited = jobs[i].waited.saturating_add(segment);
                stats.wait_ticks += segment as u128;
                emit(&mut events, now, jobs[i].customer.id, Kind::Renege { waited: jobs[i].waited });
                last_event_time = now;
                stats.reneged += 1;
            }
        }

        while cursor < order.len() && customers[order[cursor]].arrival == now {
            let i = order[cursor];
            cursor += 1;
            let id = jobs[i].customer.id;
            emit(&mut events, now, id, Kind::Arrive);
            last_event_time = now;
            let gate = config.waiting_capacity + usize::from(running.is_none());
            if waiting.len() >= gate {
                emit(&mut events, now, id, Kind::Balk);
                last_event_time = now;
                stats.balked += 1;
            } else {
                stats.accepted += 1;
                jobs[i].queue_since = now;
                waiting.push(i);
            }
        }

        if let Some(r) = running {
            if now >= r.work_at {
                if let Some(w) = best_waiter(&waiting, &jobs) {
                    if jobs[w].customer.priority < jobs[r.index].customer.priority {
                        let remaining = r.finish - now;
                        let id = jobs[r.index].customer.id;
                        emit(&mut events, now, id, Kind::Preempt { remaining, by: jobs[w].customer.id });
                        last_event_time = now;
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - r.since) as u128;
                        jobs[r.index].remaining = remaining;
                        jobs[r.index].queue_since = now;
                        jobs[r.index].pause_time = Some(now);
                        jobs[r.index].active = false;
                        if let Some(budget) = jobs[r.index].customer.patience {
                            jobs[r.index].deadline = Some(now.checked_add(budget).ok_or(Error::TimeOverflow {
                                customer: id, time: now, stage: OverflowStage::Patience,
                            })?);
                        }
                        waiting.push(r.index);
                        running = None;
                    }
                }
            }
        }

        while running.is_none() {
            let Some(i) = best_waiter(&waiting, &jobs) else { break };
            waiting.retain(|&x| x != i);
            let segment = now - jobs[i].queue_since;
            jobs[i].waited = jobs[i].waited.saturating_add(segment);
            stats.wait_ticks += segment as u128;
            if let Some(budget) = jobs[i].customer.patience {
                jobs[i].deadline = Some(jobs[i].deadline.unwrap().saturating_sub(segment).saturating_add(now));
                let _ = budget;
            }
            let id = jobs[i].customer.id;
            let switching = last_dispatched.is_some_and(|previous| previous != id);
            let cost = if switching { config.switch_cost } else { 0 };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: id, time: now, stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(jobs[i].remaining).ok_or(Error::TimeOverflow {
                customer: id, time: now, stage: OverflowStage::Finish,
            })?;
            let kind = if jobs[i].ever_started {
                Kind::Resume { remaining: jobs[i].remaining, work_at, finish }
            } else {
                jobs[i].ever_started = true;
                stats.started += 1;
                Kind::Start { remaining: jobs[i].remaining, work_at, finish }
            };
            emit(&mut events, now, id, kind);
            last_event_time = now;
            last_dispatched = Some(id);
            jobs[i].active = true;
            running = Some(Running { index: i, since: now, work_at, finish });
            if finish == now {
                let r = running.take().unwrap();
                emit(&mut events, now, id, Kind::Finish);
                last_event_time = now;
                jobs[i].active = false;
                stats.completed += 1;
                stats.turnaround_ticks += (now - jobs[i].customer.arrival) as u128;
                stats.completion_order.push(id);
                running = None;
                let _ = r;
                continue;
            }
            break;
        }

        let mut phase_f: Vec<usize> = waiting.iter().copied()
            .filter(|&i| jobs[i].deadline.is_some_and(|d| d <= now)).collect();
        phase_f.sort_by(|&a, &b| deadline_order(a, b, &jobs));
        for i in phase_f {
            if let Some(pos) = waiting.iter().position(|&x| x == i) {
                waiting.remove(pos);
                let segment = now - jobs[i].queue_since;
                jobs[i].waited = jobs[i].waited.saturating_add(segment);
                stats.wait_ticks += segment as u128;
                emit(&mut events, now, jobs[i].customer.id, Kind::Renege { waited: jobs[i].waited });
                last_event_time = now;
                stats.reneged += 1;
            }
        }

        if stats.completed + stats.reneged + stats.balked == customers.len() {
            break;
        }

        let mut next = None::<u64>;
        let mut consider = |t: u64| {
            if t > now && next.map_or(true, |n| t < n) { next = Some(t); }
        };
        if cursor < order.len() {
            consider(customers[order[cursor]].arrival);
        }
        if let Some(r) = running {
            consider(r.finish);
            if r.work_at > now { consider(r.work_at); }
        }
        for &i in &waiting {
            if let Some(d) = jobs[i].deadline { consider(d); }
        }
        if let Some(t) = next {
            now = t;
        } else {
            break;
        }
    }

    stats.elapsed = last_event_time;
    stats.mean_wait_milli = rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap_or(0);
    stats.mean_turnaround_milli = rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap_or(0);
    stats.utilization_bps = rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128).unwrap_or(0) as u32;
    Ok(Report { events, stats })
}
