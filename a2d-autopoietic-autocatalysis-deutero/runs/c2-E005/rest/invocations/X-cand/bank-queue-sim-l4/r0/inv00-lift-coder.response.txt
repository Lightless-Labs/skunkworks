#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };
    Some(if round_up { quotient + 1 } else { quotient })
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    waited: u64,
    queue_since: u64,
    paused: bool,
    pause_time: u64,
    first_started: bool,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    protection_ended: bool,
    interval_start: u64,
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn waiter_better(a: usize, b: usize, jobs: &[Job]) -> bool {
    let x = &jobs[a];
    let y = &jobs[b];
    if x.customer.priority != y.customer.priority {
        return x.customer.priority < y.customer.priority;
    }
    if x.paused != y.paused {
        return x.paused;
    }
    if x.paused {
        if x.pause_time != y.pause_time {
            return x.pause_time < y.pause_time;
        }
        x.customer.id < y.customer.id
    } else if x.customer.arrival != y.customer.arrival {
        x.customer.arrival < y.customer.arrival
    } else {
        x.customer.id > y.customer.id
    }
}

fn best_waiter(waiting: &[usize], jobs: &[Job]) -> Option<usize> {
    waiting.iter().copied().reduce(|best, candidate| {
        if waiter_better(candidate, best, jobs) {
            candidate
        } else {
            best
        }
    })
}

fn overflow(customer: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow { customer, time, stage }
}

fn expire_waiters(
    now: u64,
    waiting: &mut Vec<usize>,
    jobs: &mut [Job],
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<(u64, u32, usize)> = waiting
        .iter()
        .filter_map(|&i| jobs[i].deadline.filter(|&d| d <= now).map(|d| (d, jobs[i].customer.id, i)))
        .collect();
    expired.sort_by_key(|&(deadline, id, _)| (deadline, id));

    for (_, _, index) in expired {
        if let Some(pos) = waiting.iter().position(|&i| i == index) {
            waiting.swap_remove(pos);
            let segment = now - jobs[index].queue_since;
            jobs[index].waited = jobs[index].waited.saturating_add(segment);
            stats.wait_ticks += segment as u128;
            stats.reneged += 1;
            emit(events, now, jobs[index].customer.id, Kind::Renege { waited: jobs[index].waited });
        }
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut seen = Vec::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen.contains(&customer.id) {
            return Err(Error::DuplicateId { index, id: customer.id });
        }
        seen.push(customer.id);
        if customer.priority > 3 {
            return Err(Error::BadPriority { index, value: customer.priority });
        }
        if let Some(patience) = customer.patience {
            if customer.arrival.checked_add(patience).is_none() {
                return Err(Error::DeadlineOverflow { index, id: customer.id });
            }
        }
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then(customers[a].priority.cmp(&customers[b].priority))
            .then(customers[b].id.cmp(&customers[a].id))
    });

    let mut jobs: Vec<Job> = customers
        .iter()
        .cloned()
        .map(|customer| Job {
            remaining: customer.service,
            deadline: customer.patience.map(|p| customer.arrival + p),
            queue_since: customer.arrival,
            customer,
            waited: 0,
            paused: false,
            pause_time: 0,
            first_started: false,
        })
        .collect();

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };
    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut ever_dispatched = false;

    while stats.completed + stats.reneged + stats.balked < customers.len() {
        let mut next_time: Option<u64> = None;
        let mut consider = |t: u64| {
            if t >= now && next_time.map_or(true, |old| t < old) {
                next_time = Some(t);
            }
        };

        if next_arrival < order.len() {
            consider(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = &active {
            consider(a.finish);
            if !a.protection_ended {
                consider(a.work_at);
            }
        }
        for &i in &waiting {
            if let Some(d) = jobs[i].deadline {
                consider(d);
            }
        }

        if active.is_none() && !waiting.is_empty() {
            consider(now);
        }
        let Some(tick) = next_time else { break };
        now = tick;

        if let Some(a) = active.as_ref() {
            if a.finish == now {
                let a = active.take().unwrap();
                let job = &jobs[a.index];
                stats.busy_ticks += (now - a.interval_start) as u128;
                stats.completed += 1;
                stats.turnaround_ticks += (now - job.customer.arrival) as u128;
                stats.completion_order.push(job.customer.id);
                emit(&mut events, now, job.customer.id, Kind::Finish);
            } else if !a.protection_ended && a.work_at == now {
                active.as_mut().unwrap().protection_ended = true;
            }
        }

        expire_waiters(now, &mut waiting, &mut jobs, &mut events, &mut stats);

        while next_arrival < order.len() && customers[order[next_arrival]].arrival == now {
            let i = order[next_arrival];
            next_arrival += 1;
            let id = jobs[i].customer.id;
            emit(&mut events, now, id, Kind::Arrive);

            let limit = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= limit {
                stats.balked += 1;
                emit(&mut events, now, id, Kind::Balk);
            } else {
                stats.accepted += 1;
                jobs[i].queue_since = now;
                waiting.push(i);
            }
        }

        if let Some(a) = active.as_ref() {
            if a.protection_ended {
                if let Some(best) = best_waiter(&waiting, &jobs) {
                    if jobs[best].customer.priority < jobs[a.index].customer.priority {
                        let a = active.take().unwrap();
                        let job = &mut jobs[a.index];
                        let remaining = a.finish - now;
                        stats.busy_ticks += (now - a.interval_start) as u128;
                        emit(
                            &mut events,
                            now,
                            job.customer.id,
                            Kind::Preempt { remaining, by: jobs[best].customer.id },
                        );
                        stats.preemptions += 1;
                        job.remaining = remaining;
                        job.paused = true;
                        job.pause_time = now;
                        job.queue_since = now;
                        job.deadline = match job.customer.patience {
                            Some(_) => Some(now.checked_add(job.customer.patience.unwrap()).ok_or_else(|| {
                                overflow(job.customer.id, now, OverflowStage::Patience)
                            })?),
                            None => None,
                        };
                        waiting.push(a.index);
                    }
                }
            }
        }

        loop {
            if active.is_some() {
                break;
            }
            let Some(index) = best_waiter(&waiting, &jobs) else {
                break;
            };
            let pos = waiting.iter().position(|&i| i == index).unwrap();
            waiting.swap_remove(pos);

            let segment = now - jobs[index].queue_since;
            jobs[index].waited = jobs[index].waited.saturating_add(segment);
            stats.wait_ticks += segment as u128;

            let id = jobs[index].customer.id;
            let switch = if !ever_dispatched || last_dispatched == Some(id) {
                0
            } else {
                config.switch_cost
            };
            let work_at = now.checked_add(switch).ok_or_else(|| overflow(id, now, OverflowStage::WorkAt))?;
            let finish = work_at
                .checked_add(jobs[index].remaining)
                .ok_or_else(|| overflow(id, now, OverflowStage::Finish))?;

            let remaining = jobs[index].remaining;
            let resume = jobs[index].first_started;
            if resume {
                emit(&mut events, now, id, Kind::Resume { remaining, work_at, finish });
            } else {
                jobs[index].first_started = true;
                stats.started += 1;
                emit(&mut events, now, id, Kind::Start { remaining, work_at, finish });
            }

            ever_dispatched = true;
            last_dispatched = Some(id);
            active = Some(Active {
                index,
                work_at,
                finish,
                protection_ended: work_at == now,
                interval_start: now,
            });

            if finish == now {
                active = None;
                stats.busy_ticks += 0;
                stats.completed += 1;
                stats.turnaround_ticks += (now - jobs[index].customer.arrival) as u128;
                stats.completion_order.push(id);
                emit(&mut events, now, id, Kind::Finish);
                continue;
            }
        }

        expire_waiters(now, &mut waiting, &mut jobs, &mut events, &mut stats);

        if next_arrival >= order.len()
            && active.is_none()
            && waiting.is_empty()
        {
            break;
        }

        // If the only remaining work is an active interval whose finish is at
        // this same tick, the next iteration processes it before anything else.
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(1000 * stats.wait_ticks, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(1000 * stats.turnaround_ticks, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(10000 * stats.busy_ticks, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}
