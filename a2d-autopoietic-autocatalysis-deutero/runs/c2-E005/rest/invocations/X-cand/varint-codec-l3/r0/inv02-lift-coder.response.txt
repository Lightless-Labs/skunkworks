#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Message {
    pub fields: Vec<Field>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Field {
    UInt { tag: u32, value: u64 },
    Bytes { tag: u32, data: Vec<u8> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
    LengthOutOfBounds { offset: usize, length: u64 },
    InvalidTag { offset: usize, tag: u64 },
    InvalidWire { offset: usize, wire: u8 },
    OutOfOrder { offset: usize, previous: u32, tag: u32 },
    Duplicate { offset: usize, tag: u32 },
}

fn read_varint(input: &[u8], mut pos: usize, end: usize) -> Result<(u64, usize), Error> {
    let start = pos;
    let mut value = 0u64;

    for index in 0..10 {
        if pos >= end {
            return Err(Error::Truncated { offset: end, start });
        }

        let byte = input[pos];
        let payload = byte & 0x7f;

        if index == 9 && (payload > 1 || (byte & 0x80) != 0) {
            return Err(Error::Overflow { offset: pos, start });
        }

        value |= (payload as u64) << (index * 7);
        pos += 1;

        if byte & 0x80 == 0 {
            return Ok((value, pos));
        }
    }

    unreachable!()
}

fn write_varint(mut value: u64, output: &mut Vec<u8>) {
    while value >= 0x80 {
        output.push((value as u8) | 0x80);
        value >>= 7;
    }
    output.push(value as u8);
}

pub fn encode_message(message: &Message) -> Result<Vec<u8>, Error> {
    let mut body = Vec::new();
    let mut previous: Option<(u32, bool)> = None;

    for (index, field) in message.fields.iter().enumerate() {
        let (tag, is_bytes) = match field {
            Field::UInt { tag, .. } => (*tag, false),
            Field::Bytes { tag, .. } => (*tag, true),
        };

        if !(1..=15).contains(&tag) {
            return Err(Error::InvalidTag {
                offset: index,
                tag: tag as u64,
            });
        }

        if let Some((prev_tag, prev_bytes)) = previous {
            if tag < prev_tag {
                return Err(Error::OutOfOrder {
                    offset: index,
                    previous: prev_tag,
                    tag,
                });
            }
            if tag == prev_tag && !(prev_bytes && is_bytes) {
                return Err(Error::Duplicate { offset: index, tag });
            }
        }

        previous = Some((tag, is_bytes));

        match field {
            Field::UInt { value, .. } => {
                write_varint((tag as u64) << 3, &mut body);
                write_varint(*value, &mut body);
            }
            Field::Bytes { data, .. } => {
                write_varint(((tag as u64) << 3) | 2, &mut body);
                write_varint(data.len() as u64, &mut body);
                body.extend_from_slice(data);
            }
        }
    }

    let mut output = Vec::new();
    write_varint(body.len() as u64, &mut output);
    output.extend_from_slice(&body);
    Ok(output)
}

pub fn decode_message(input: &[u8], start: usize) -> Result<(Message, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let (body_length, body_start) = read_varint(input, start, input.len())?;
    let available = input.len() - body_start;
    if body_length > available as u64 {
        return Err(Error::LengthOutOfBounds {
            offset: start,
            length: body_length,
        });
    }

    let body_len = body_length as usize;
    let frame_end = body_start + body_len;
    let mut pos = body_start;
    let mut fields = Vec::new();
    let mut previous: Option<(u32, u8)> = None;

    while pos < frame_end {
        let key_start = pos;
        let (key, next) = read_varint(input, pos, frame_end)?;
        pos = next;

        let tag64 = key >> 3;
        let wire = (key & 7) as u8;

        if !(1..=255).contains(&tag64) {
            return Err(Error::InvalidTag {
                offset: key_start,
                tag: tag64,
            });
        }
        let tag = tag64 as u32;

        if !matches!(wire, 0 | 1 | 2 | 5) {
            return Err(Error::InvalidWire {
                offset: key_start,
                wire,
            });
        }

        if let Some((prev_tag, prev_wire)) = previous {
            if tag < prev_tag {
                return Err(Error::OutOfOrder {
                    offset: key_start,
                    previous: prev_tag,
                    tag,
                });
            }
            if tag == prev_tag && !(prev_wire == 2 && wire == 2) {
                return Err(Error::Duplicate {
                    offset: key_start,
                    tag,
                });
            }
        }
        previous = Some((tag, wire));

        match wire {
            0 => {
                let (value, after) = read_varint(input, pos, frame_end)?;
                pos = after;
                if tag <= 15 {
                    fields.push(Field::UInt { tag, value });
                }
            }
            1 | 5 => {
                let needed = if wire == 1 { 8usize } else { 4usize };
                let remaining = frame_end - pos;
                if remaining < needed {
                    return Err(Error::LengthOutOfBounds {
                        offset: key_start,
                        length: needed as u64,
                    });
                }
                pos += needed;
            }
            2 => {
                let length_start = pos;
                let (length, after_length) = read_varint(input, pos, frame_end)?;
                pos = after_length;
                let remaining = frame_end - pos;
                if length > remaining as u64 {
                    return Err(Error::LengthOutOfBounds {
                        offset: length_start,
                        length,
                    });
                }
                let data_len = length as usize;
                if tag <= 15 {
                    if let Some(Field::Bytes {
                        tag: existing_tag,
                        data,
                    }) = fields.last_mut()
                    {
                        if *existing_tag == tag {
                            data.extend_from_slice(&input[pos..pos + data_len]);
                        } else {
                            fields.push(Field::Bytes {
                                tag,
                                data: input[pos..pos + data_len].to_vec(),
                            });
                        }
                    } else {
                        fields.push(Field::Bytes {
                            tag,
                            data: input[pos..pos + data_len].to_vec(),
                        });
                    }
                }
                pos += data_len;
            }
            _ => unreachable!(),
        }
    }

    Ok((Message { fields }, frame_end))
}
