#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Debug, Clone)]
struct Token {
    text: String,
    width: usize,
}

#[derive(Debug, Clone)]
struct LogicalLine {
    tokens: Vec<Token>,
    width: usize,
}

fn invalid_char(ch: char, next: Option<char>) -> bool {
    if ch == '\r' {
        return next != Some('\n');
    }
    ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}'
}

fn prepare_cell(
    cell: &str,
    markdown: bool,
) -> Vec<LogicalLine> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            let line = line.trim_end_matches(' ');
            let mut tokens = Vec::new();
            let mut width = 0usize;

            for ch in line.chars() {
                let text = if markdown {
                    match ch {
                        '|' => String::from("\\|"),
                        '\\' => String::from("\\\\"),
                        _ => ch.to_string(),
                    }
                } else {
                    ch.to_string()
                };
                let token_width = text.chars().count();
                width += token_width;
                tokens.push(Token {
                    text,
                    width: token_width,
                });
            }

            LogicalLine { tokens, width }
        })
        .collect()
}

fn fragment_text(tokens: &[Token]) -> String {
    let mut result = String::new();
    for token in tokens {
        result.push_str(&token.text);
    }
    result
}

fn split_line(line: &LogicalLine, width: usize, overflow: &Overflow) -> Vec<String> {
    if line.tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            if line.width <= width {
                return vec![fragment_text(&line.tokens)];
            }

            let mut used = 0usize;
            let mut taken = 0usize;
            let limit = width.saturating_sub(1);
            while taken < line.tokens.len() {
                let token = &line.tokens[taken];
                if used + token.width > limit {
                    break;
                }
                used += token.width;
                taken += 1;
            }

            let mut result = fragment_text(&line.tokens[..taken]);
            result.push('…');
            vec![result]
        }
        Overflow::Wrap => {
            let mut fragments = Vec::new();
            let mut start = 0usize;

            while start < line.tokens.len() {
                let mut end = start;
                let mut used = 0usize;
                while end < line.tokens.len()
                    && used + line.tokens[end].width <= width
                {
                    used += line.tokens[end].width;
                    end += 1;
                }

                // All valid modes have a positive minimum width, and Markdown
                // escapes are at most two scalars wide.
                if end == start {
                    end += 1;
                }

                fragments.push(fragment_text(&line.tokens[start..end]));
                start = end;
            }

            fragments
        }
    }
}

fn align_fragment(text: &str, width: usize, alignment: &Alignment) -> String {
    let content_width = text.chars().count();
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };

    let mut result = String::new();
    result.extend(std::iter::repeat_n(' ', left));
    result.push_str(text);
    result.extend(std::iter::repeat_n(' ', right));
    result
}

fn append_grid_border(output: &mut String, widths: &[usize]) {
    output.push('+');
    for &width in widths {
        output.extend(std::iter::repeat_n('-', width + 2));
        output.push('+');
    }
    output.push('\n');
}

fn append_data_line(
    output: &mut String,
    fragments: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) {
    match mode {
        Mode::Grid | Mode::Markdown => {
            output.push('|');
            for (index, ((fragment, &width), alignment)) in fragments
                .iter()
                .zip(widths)
                .zip(alignments)
                .enumerate()
            {
                let aligned = align_fragment(fragment, width, alignment);
                output.push(' ');
                output.push_str(&aligned);
                output.push(' ');
                output.push('|');
                let _ = index;
            }
            output.push('\n');
        }
        Mode::Compact => {
            for index in 0..fragments.len() {
                if index > 0 {
                    output.push_str("  ");
                }
                let mut aligned = align_fragment(&fragments[index], widths[index], &alignments[index]);
                if index + 1 == fragments.len() {
                    let right_padding = match alignments[index] {
                        Alignment::Left => widths[index].saturating_sub(fragments[index].chars().count()),
                        Alignment::Right | Alignment::Center => 0,
                    };
                    for _ in 0..right_padding {
                        aligned.pop();
                    }
                }
                output.push_str(&aligned);
            }
            output.push('\n');
        }
    }
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let n = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != n {
            return Err(TableError::Ragged {
                row: row_index,
                expected: n,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }

    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let (overhead, minimum_width) = match options.mode {
        Mode::Grid | Mode::Markdown => (3 * n + 1, if options.mode == Mode::Markdown { 3 } else { 1 }),
        Mode::Compact => (2 * (n - 1), 1),
    };
    let minimum = overhead + n * minimum_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                let next = cell[byte + ch.len_utf8()..].chars().next();
                if invalid_char(ch, next) {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let prepared: Vec<Vec<Vec<LogicalLine>>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| prepare_cell(cell, options.mode == Mode::Markdown))
                .collect()
        })
        .collect();

    let mut widths = vec![minimum_width; n];
    for row in &prepared {
        for (column, lines) in row.iter().enumerate() {
            for line in lines {
                widths[column] = widths[column].max(line.width);
            }
        }
    }

    if options.max_width != 0 {
        let mut total: usize = widths.iter().sum::<usize>() + overhead;
        while total > options.max_width {
            let candidate = (0..n)
                .filter(|&column| widths[column] > minimum_width)
                .max_by_key(|&column| (widths[column], column));
            match candidate {
                Some(column) => {
                    widths[column] -= 1;
                    total -= 1;
                }
                None => break,
            }
        }
    }

    let mut output = String::new();
    if options.mode == Mode::Grid {
        append_grid_border(&mut output, &widths);
    }

    for (row_index, row) in prepared.iter().enumerate() {
        let cell_fragments: Vec<Vec<String>> = row
            .iter()
            .enumerate()
            .map(|(column, lines)| {
                let mut fragments = Vec::new();
                for line in lines {
                    fragments.extend(split_line(line, widths[column], &overflows[column]));
                }
                fragments
            })
            .collect();

        let height = cell_fragments
            .iter()
            .map(Vec::len)
            .max()
            .unwrap_or(1);

        for fragment_index in 0..height {
            let fragments: Vec<String> = cell_fragments
                .iter()
                .map(|cell| cell.get(fragment_index).cloned().unwrap_or_default())
                .collect();
            append_data_line(
                &mut output,
                &fragments,
                &widths,
                &alignments,
                &options.mode,
            );
        }

        if options.mode == Mode::Grid && options.header && row_index == 0 && rows.len() > 1 {
            append_grid_border(&mut output, &widths);
        }

        if options.mode == Mode::Markdown && options.header && row_index == 0 {
            output.push('|');
            for column in 0..n {
                let marker = match alignments[column] {
                    Alignment::Left => format!(":{}", "-".repeat(widths[column] - 1)),
                    Alignment::Right => format!("{}:", "-".repeat(widths[column] - 1)),
                    Alignment::Center => format!(
                        ":{}:",
                        "-".repeat(widths[column] - 2)
                    ),
                };
                output.push(' ');
                output.push_str(&marker);
                output.push(' ');
                output.push('|');
            }
            output.push('\n');
        }
    }

    if options.mode == Mode::Grid {
        append_grid_border(&mut output, &widths);
    }

    Ok(output)
}
