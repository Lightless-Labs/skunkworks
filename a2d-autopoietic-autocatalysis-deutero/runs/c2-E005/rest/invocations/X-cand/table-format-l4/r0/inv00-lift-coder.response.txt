#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn valid_cell(cell: &str) -> Option<usize> {
    let bytes = cell.as_bytes();
    let mut offset = 0;

    while offset < bytes.len() {
        let ch = cell[offset..].chars().next()?;
        if ch == '\r' {
            if bytes.get(offset + 1) != Some(&b'\n') {
                return Some(offset);
            }
            offset += 2;
            continue;
        }
        if (ch.is_control() && ch != '\n') || ch == '\u{2028}' || ch == '\u{2029}' {
            return Some(offset);
        }
        offset += ch.len_utf8();
    }
    None
}

fn make_lines(cell: &str, mode: &Mode) -> Vec<Vec<Token>> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            let trimmed = line.trim_end_matches(' ');
            let mut tokens = Vec::new();
            for ch in trimmed.chars() {
                match mode {
                    Mode::Markdown if ch == '|' => tokens.push(Token {
                        text: "\\|".to_owned(),
                        width: 2,
                    }),
                    Mode::Markdown if ch == '\\' => tokens.push(Token {
                        text: "\\\\".to_owned(),
                        width: 2,
                    }),
                    _ => tokens.push(Token {
                        text: ch.to_string(),
                        width: 1,
                    }),
                }
            }
            tokens
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }

            let mut out = String::new();
            let mut used = 0;
            for token in tokens {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                out.push_str(&token.text);
                used += token.width;
            }
            out.push('…');
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                // All modes provide enough minimum width for every token.
                if end == start {
                    end += 1;
                }
                result.push(tokens[start..end].iter().map(|t| t.text.as_str()).collect());
                start = end;
            }
            result
        }
    }
}

fn scalar_len(s: &str) -> usize {
    s.chars().count()
}

fn align(content: &str, width: usize, alignment: &Alignment) -> String {
    let extra = width.saturating_sub(scalar_len(content));
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let mut out = String::new();
    out.extend(std::iter::repeat_n(' ', left));
    out.push_str(content);
    out.extend(std::iter::repeat_n(' ', right));
    out
}

fn border(widths: &[usize]) -> String {
    let mut out = String::from("+");
    for width in widths {
        out.extend(std::iter::repeat_n('-', width + 2));
        out.push('+');
    }
    out.push('\n');
    out
}

fn data_line(
    cells: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    match mode {
        Mode::Compact => {
            let mut out = String::new();
            for i in 0..cells.len() {
                if i > 0 {
                    out.push_str("  ");
                }
                let value = align(&cells[i], widths[i], &alignments[i]);
                if i + 1 == cells.len() {
                    let right_padding = match alignments[i] {
                        Alignment::Left => widths[i].saturating_sub(scalar_len(&cells[i])),
                        Alignment::Center => {
                            (widths[i].saturating_sub(scalar_len(&cells[i])) + 1) / 2
                        }
                        Alignment::Right => 0,
                    };
                    let keep = value.len().saturating_sub(right_padding);
                    out.push_str(&value[..keep]);
                } else {
                    out.push_str(&value);
                }
            }
            out.push('\n');
            out
        }
        Mode::Grid | Mode::Markdown => {
            let mut out = String::from("|");
            for i in 0..cells.len() {
                out.push(' ');
                out.push_str(&align(&cells[i], widths[i], &alignments[i]));
                out.push_str(" |");
            }
            out.push('\n');
            out
        }
    }
}

fn markdown_separator(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut out = String::from("|");
    for (width, alignment) in widths.iter().zip(alignments) {
        out.push(' ');
        match alignment {
            Alignment::Left => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 1));
            }
            Alignment::Right => {
                out.extend(std::iter::repeat_n('-', width - 1));
                out.push(':');
            }
            Alignment::Center => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 2));
                out.push(':');
            }
        }
        out.push_str(" |");
    }
    out.push('\n');
    out
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_column = match options.mode {
        Mode::Markdown => 3,
        Mode::Grid | Mode::Compact => 1,
    };
    let overhead = match options.mode {
        Mode::Compact => 2 * (n - 1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let minimum = overhead + n * minimum_column;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            if let Some(byte) = valid_cell(cell) {
                return Err(TableError::InvalidChar { row, column, byte });
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let logical: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| make_lines(cell, &options.mode))
                .collect()
        })
        .collect();

    let mut widths = vec![minimum_column; n];
    for row in &logical {
        for (column, lines) in row.iter().enumerate() {
            for line in lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let mut total = overhead + widths.iter().sum::<usize>();
        while total > options.max_width {
            let candidate = (0..n)
                .rev()
                .find(|&i| widths[i] > minimum_column)
                .expect("minimum budget check guarantees a candidate");
            widths[candidate] -= 1;
            total -= 1;
        }
    }

    let mut output = String::new();
    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in logical.iter().enumerate() {
        let cell_fragments: Vec<Vec<String>> = row
            .iter()
            .enumerate()
            .map(|(column, lines)| {
                let mut result = Vec::new();
                for line in lines {
                    result.extend(fragments(line, widths[column], &overflows[column]));
                }
                result
            })
            .collect();

        let height = cell_fragments.iter().map(Vec::len).max().unwrap_or(1);
        for fragment_index in 0..height {
            let cells: Vec<String> = cell_fragments
                .iter()
                .map(|frags| frags.get(fragment_index).cloned().unwrap_or_default())
                .collect();
            output.push_str(&data_line(
                &cells,
                &widths,
                &alignments,
                &options.mode,
            ));
        }

        if options.mode == Mode::Markdown && row_index == 0 {
            output.push_str(&markdown_separator(&widths, &alignments));
        } else if options.mode == Mode::Grid
            && options.header
            && row_index == 0
            && rows.len() > 1
        {
            output.push_str(&border(&widths));
        }
    }

    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }
    Ok(output)
}
