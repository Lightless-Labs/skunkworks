#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

#[derive(Debug, Clone)]
struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    variables: Vec<(String, usize)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let start = cursor;
                let Some(relative_end) = bytes[start + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = start + 1 + relative_end;

                if let Some(relative_nested) =
                    bytes[start + 1..end].iter().position(|&b| b == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: start + 1 + relative_nested,
                    });
                }

                let content_start = start + 1;
                let mut spec_start = content_start;
                let mut operator = Operator::Simple;

                if content_start < end {
                    operator = match bytes[content_start] {
                        b'+' => Operator::Reserved,
                        b'#' => Operator::Fragment,
                        b'.' => Operator::Dot,
                        b'/' => Operator::Slash,
                        b';' => Operator::Semicolon,
                        b'?' => Operator::Query,
                        b'&' => Operator::Continuation,
                        b'=' | b',' | b'!' | b'@' | b'|' => {
                            return Err(Error::UnsupportedOperator { pos: content_start });
                        }
                        _ => Operator::Simple,
                    };
                    if !matches!(operator, Operator::Simple) {
                        spec_start += 1;
                    }
                }

                if spec_start == end {
                    return Err(Error::EmptyExpression { pos: end });
                }

                let mut variables = Vec::new();
                let mut seen = std::collections::HashSet::new();
                let mut part_start = spec_start;

                loop {
                    let part_end = bytes[part_start..end]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| part_start + offset)
                        .unwrap_or(end);

                    if part_start == part_end {
                        return Err(Error::EmptyVariable { pos: part_start });
                    }

                    let name_bytes = &bytes[part_start..part_end];
                    let valid_first = name_bytes[0].is_ascii_alphabetic() || name_bytes[0] == b'_';
                    if !valid_first {
                        return Err(Error::InvalidName { pos: part_start });
                    }
                    for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
                        if !(byte.is_ascii_alphanumeric() || byte == b'_') {
                            return Err(Error::InvalidName {
                                pos: part_start + offset,
                            });
                        }
                    }

                    let name = template[part_start..part_end].to_owned();
                    if !seen.insert(name.clone()) {
                        return Err(Error::DuplicateVariable { pos: part_start });
                    }
                    variables.push((name, part_start));

                    if part_end == end {
                        break;
                    }
                    part_start = part_end + 1;
                }

                expressions.push(Expression {
                    start,
                    end,
                    operator,
                    variables,
                });
                cursor = end + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::with_capacity(template.len());
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let mut fields = Vec::new();
        for (name, _) in &expression.variables {
            if let Some(value) = vars.iter().rev().find_map(|(key, value)| {
                if *key == name.as_str() {
                    Some(*value)
                } else {
                    None
                }
            }) {
                let encoded = encode(value, matches!(expression.operator, Operator::Reserved | Operator::Fragment));
                let field = match expression.operator {
                    Operator::Semicolon => {
                        if encoded.is_empty() {
                            name.clone()
                        } else {
                            format!("{name}={encoded}")
                        }
                    }
                    Operator::Query | Operator::Continuation => format!("{name}={encoded}"),
                    _ => encoded,
                };
                fields.push(field);
            }
        }

        if !fields.is_empty() {
            match expression.operator {
                Operator::Simple | Operator::Reserved => {}
                Operator::Fragment => output.push('#'),
                Operator::Dot => output.push('.'),
                Operator::Slash => output.push('/'),
                Operator::Semicolon => output.push(';'),
                Operator::Query => output.push('?'),
                Operator::Continuation => output.push('&'),
            }
            let separator = match expression.operator {
                Operator::Dot => ".",
                Operator::Slash => "/",
                Operator::Semicolon | Operator::Continuation | Operator::Query => "&",
                _ => ",",
            };
            output.push_str(&fields.join(separator));
        }

        literal_start = expression.end + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() && is_hex(bytes[index + 1]) && is_hex(bytes[index + 2]) {
            result.push('%');
            result.push((bytes[index + 1] as char).to_ascii_uppercase());
            result.push((bytes[index + 2] as char).to_ascii_uppercase());
            index += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(hex_digit(byte >> 4));
            result.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }

    result
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn hex_digit(nibble: u8) -> char {
    match nibble {
        0..=9 => (b'0' + nibble) as char,
        _ => (b'A' + nibble - 10) as char,
    }
}
