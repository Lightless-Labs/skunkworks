#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression<'a> {
    start: usize,
    end: usize,
    operator: Option<u8>,
    variables: Vec<(&'a str, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let content_start = open + 1;
    let bytes = template.as_bytes();

    if let Some(inner) = bytes[content_start..close]
        .iter()
        .position(|&byte| byte == b'{')
    {
        return Err(Error::NestedOpen {
            pos: content_start + inner,
        });
    }

    let mut spec_start = content_start;
    let operator = if spec_start < close
        && matches!(
            bytes[spec_start],
            b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&' | b'=' | b'!' | b'@' | b'|'
        )
    {
        let byte = bytes[spec_start];
        if !matches!(byte, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
            return Err(Error::UnsupportedOperator { pos: spec_start });
        }
        spec_start += 1;
        Some(byte)
    } else {
        None
    };

    if spec_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let content = &template[spec_start..close];
    let mut variables = Vec::new();
    let mut relative_start = 0;

    for part in content.split(',') {
        let absolute_start = spec_start + relative_start;
        if part.is_empty() {
            return Err(Error::EmptyVariable { pos: absolute_start });
        }

        let part_bytes = part.as_bytes();
        if !is_name_start(part_bytes[0]) {
            return Err(Error::InvalidName { pos: absolute_start });
        }
        if let Some(index) = part_bytes.iter().position(|&byte| !is_name_byte(byte)) {
            return Err(Error::InvalidName {
                pos: absolute_start + index,
            });
        }

        if variables.iter().any(|(name, _)| *name == part) {
            return Err(Error::DuplicateVariable { pos: absolute_start });
        }

        variables.push((part, absolute_start));
        relative_start += part.len() + 1;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        operator,
        variables,
    })
}

fn parse_template(template: &str) -> Result<Vec<Expression<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let Some(relative_close) = bytes[pos + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos });
                };
                let close = pos + 1 + relative_close;
                expressions.push(parse_expression(template, pos, close)?);
                pos = close + 1;
            }
            _ => pos += 1,
        }
    }

    Ok(expressions)
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn hex_upper(byte: u8) -> u8 {
    byte.to_ascii_uppercase()
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
            output.push('%');
            output.push(hex_upper(bytes[i + 1]) as char);
            output.push(hex_upper(bytes[i + 2]) as char);
            i += 3;
            continue;
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0F) as usize] as char);
        }
        i += 1;
    }

    output
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|&(binding, value)| (binding == name).then_some(value))
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::with_capacity(template.len());
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);

        let operator = expression.operator;
        let reserved = matches!(operator, Some(b'+' | b'#'));
        let mut fields = Vec::new();

        for &(name, _) in &expression.variables {
            if let Some(value) = lookup(vars, name) {
                let encoded = encode(value, reserved);
                let field = match operator {
                    Some(b';') => {
                        if encoded.is_empty() {
                            name.to_owned()
                        } else {
                            format!("{name}={encoded}")
                        }
                    }
                    Some(b'?') | Some(b'&') => format!("{name}={encoded}"),
                    _ => encoded,
                };
                fields.push(field);
            }
        }

        if !fields.is_empty() {
            let separator = match operator {
                Some(b'.') => ".",
                Some(b'/') => "/",
                Some(b';') | Some(b'?') | Some(b'&') => "&".to_owned(),
                _ => ",".to_owned(),
            };
            let separator = match operator {
                Some(b';') => ";",
                Some(b'?') | Some(b'&') => "&",
                _ => separator.as_str(),
            };

            match operator {
                Some(b'#') => output.push('#'),
                Some(b'.') => output.push('.'),
                Some(b'/') => output.push('/'),
                Some(b';') => output.push(';'),
                Some(b'?') => output.push('?'),
                Some(b'&') => output.push('&'),
                _ => {}
            }

            output.push_str(&fields.join(separator));
        }

        cursor = expression.end;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}
