#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    started: bool,
    pause_time: Option<u64>,
    remaining: u64,
}

#[derive(Clone)]
struct Running {
    index: usize,
    work_at: u64,
    finish: u64,
}

fn waiter_better(a: &Job, b: &Job) -> bool {
    let ap = a.customer.priority;
    let bp = b.customer.priority;
    if ap != bp {
        return ap < bp;
    }

    match (a.pause_time, b.pause_time) {
        (Some(at), Some(bt)) => {
            if at != bt {
                at < bt
            } else {
                a.customer.id < b.customer.id
            }
        }
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            if a.customer.arrival != b.customer.arrival {
                a.customer.arrival < b.customer.arrival
            } else {
                a.customer.id > b.customer.id
            }
        }
    }
}

fn best_waiter(waiting: &[usize], jobs: &[Job]) -> Option<usize> {
    waiting
        .iter()
        .copied()
        .min_by(|&a, &b| {
            if waiter_better(&jobs[a], &jobs[b]) {
                std::cmp::Ordering::Less
            } else if waiter_better(&jobs[b], &jobs[a]) {
                std::cmp::Ordering::Greater
            } else {
                std::cmp::Ordering::Equal
            }
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index]
            .iter()
            .any(|previous| previous.id == customer.id)
        {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut jobs: Vec<Job> = customers
        .iter()
        .cloned()
        .map(|customer| Job {
            remaining: customer.service,
            customer,
            started: false,
            pause_time: None,
        })
        .collect();

    let mut arrivals: Vec<usize> = (0..jobs.len()).collect();
    arrivals.sort_by(|&a, &b| {
        jobs[a]
            .customer
            .arrival
            .cmp(&jobs[b].customer.arrival)
            .then_with(|| jobs[a].customer.priority.cmp(&jobs[b].customer.priority))
            .then_with(|| jobs[b].customer.id.cmp(&jobs[a].customer.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_arrival = 0usize;
    let mut running: Option<Running> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut finished_count = 0usize;
    let mut now = arrivals
        .first()
        .map(|&index| jobs[index].customer.arrival)
        .unwrap_or(0);

    while finished_count < jobs.len() {
        if let Some(active) = running.as_ref() {
            if active.finish == now {
                let index = active.index;
                events.push(Event {
                    time: now,
                    customer: jobs[index].customer.id,
                    kind: Kind::Finish,
                });
                running = None;
                finished_count += 1;
            }
        }

        let cohort_start = next_arrival;
        while next_arrival < arrivals.len()
            && jobs[arrivals[next_arrival]].customer.arrival == now
        {
            next_arrival += 1;
        }
        for &index in &arrivals[cohort_start..next_arrival] {
            events.push(Event {
                time: now,
                customer: jobs[index].customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(index);
        }

        let protection_ended = running
            .as_ref()
            .map(|active| now >= active.work_at)
            .unwrap_or(false);

        if protection_ended {
            if let Some(waiter) = best_waiter(&waiting, &jobs) {
                let active = running.as_ref().unwrap();
                if jobs[waiter].customer.priority < jobs[active.index].customer.priority {
                    let active = running.take().unwrap();
                    let paused_index = active.index;
                    let remaining = active.finish - now;
                    let by = jobs[waiter].customer.id;
                    events.push(Event {
                        time: now,
                        customer: jobs[paused_index].customer.id,
                        kind: Kind::Preempt { remaining, by },
                    });
                    jobs[paused_index].remaining = remaining;
                    jobs[paused_index].pause_time = Some(now);
                    waiting.push(paused_index);
                }
            }
        }

        while running.is_none() {
            let Some(index) = best_waiter(&waiting, &jobs) else {
                break;
            };
            let position = waiting.iter().position(|&candidate| candidate == index).unwrap();
            waiting.swap_remove(position);

            let customer_id = jobs[index].customer.id;
            let cost = if last_dispatched == Some(customer_id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: customer_id,
                time: now,
            })?;
            let finish = work_at.checked_add(jobs[index].remaining).ok_or(Error::TimeOverflow {
                customer: customer_id,
                time: now,
            })?;

            let kind = if jobs[index].started {
                Kind::Resume {
                    remaining: jobs[index].remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: jobs[index].remaining,
                    work_at,
                    finish,
                }
            };
            jobs[index].started = true;
            jobs[index].pause_time = None;
            last_dispatched = Some(customer_id);
            events.push(Event {
                time: now,
                customer: customer_id,
                kind,
            });
            running = Some(Running {
                index,
                work_at,
                finish,
            });

            if finish == now {
                events.push(Event {
                    time: now,
                    customer: customer_id,
                    kind: Kind::Finish,
                });
                running = None;
                finished_count += 1;
            }
        }

        if finished_count == jobs.len() {
            break;
        }

        let mut next_time: Option<u64> = None;
        if next_arrival < arrivals.len() {
            next_time = Some(jobs[arrivals[next_arrival]].customer.arrival);
        }
        if let Some(active) = running.as_ref() {
            if active.finish > now {
                next_time = Some(next_time.map_or(active.finish, |t| t.min(active.finish)));
            }
            if active.work_at > now {
                next_time = Some(next_time.map_or(active.work_at, |t| t.min(active.work_at)));
            }
        }

        match next_time {
            Some(time) => now = time,
            None => break,
        }
    }

    Ok(events)
}
