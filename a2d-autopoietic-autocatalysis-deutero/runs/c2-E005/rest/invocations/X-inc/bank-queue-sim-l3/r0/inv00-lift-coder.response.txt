#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    remaining: u64,
    started: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiting_precedes(a: &Waiting, b: &Waiting) -> bool {
    if a.customer.priority != b.customer.priority {
        return a.customer.priority < b.customer.priority;
    }
    if a.started != b.started {
        return a.started;
    }
    if a.started {
        (a.pause_time, a.customer.id) < (b.pause_time, b.customer.id)
    } else {
        (a.customer.arrival, std::cmp::Reverse(a.customer.id))
            < (b.customer.arrival, std::cmp::Reverse(b.customer.id))
    }
}

fn best_waiter_index(waiting: &[Waiting]) -> Option<usize> {
    waiting
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| {
            if waiting_precedes(a, b) {
                std::cmp::Ordering::Less
            } else if waiting_precedes(b, a) {
                std::cmp::Ordering::Greater
            } else {
                std::cmp::Ordering::Equal
            }
        })
        .map(|(i, _)| i)
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut seen = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut now = arrivals[0].arrival;

    loop {
        // A scheduled finish precedes arrivals and boundary decisions.
        if running.as_ref().is_some_and(|r| r.finish == now) {
            let r = running.take().unwrap();
            events.push(Event {
                time: now,
                customer: r.customer.id,
                kind: Kind::Finish,
            });
        }

        // Add the complete cohort before allowing dispatch or preemption.
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: 0,
            });
            next_arrival += 1;
        }

        let protection_ended = running.as_ref().is_some_and(|r| now >= r.work_at);
        if protection_ended {
            if let (Some(r), Some(i)) = (running.as_ref(), best_waiter_index(&waiting)) {
                if waiting[i].customer.priority < r.customer.priority {
                    let by = waiting[i].customer.id;
                    let r = running.take().unwrap();
                    let remaining = r.finish - now;
                    events.push(Event {
                        time: now,
                        customer: r.customer.id,
                        kind: Kind::Preempt { remaining, by },
                    });
                    waiting.push(Waiting {
                        customer: r.customer,
                        remaining,
                        started: true,
                        pause_time: now,
                    });
                }
            }
        }

        // Dispatch repeatedly at this tick, including inline zero-time finishes.
        while running.is_none() {
            let Some(i) = best_waiter_index(&waiting) else {
                break;
            };
            let w = waiting.remove(i);
            let cost = if last_dispatched == Some(w.customer.id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: w.customer.id,
                time: now,
            })?;
            let finish = work_at.checked_add(w.remaining).ok_or(Error::TimeOverflow {
                customer: w.customer.id,
                time: now,
            })?;

            let kind = if w.started {
                Kind::Resume {
                    remaining: w.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: w.remaining,
                    work_at,
                    finish,
                }
            };
            events.push(Event {
                time: now,
                customer: w.customer.id,
                kind,
            });
            last_dispatched = Some(w.customer.id);
            running = Some(Running {
                customer: w.customer,
                remaining: w.remaining,
                work_at,
                finish,
            });

            if finish == now {
                let r = running.take().unwrap();
                events.push(Event {
                    time: now,
                    customer: r.customer.id,
                    kind: Kind::Finish,
                });
                continue;
            }
        }

        if running.is_none() && next_arrival == arrivals.len() && waiting.is_empty() {
            break;
        }

        let mut next_tick: Option<u64> = None;
        if next_arrival < arrivals.len() {
            next_tick = Some(arrivals[next_arrival].arrival);
        }
        if let Some(r) = running.as_ref() {
            for t in [r.work_at, r.finish] {
                if t > now {
                    next_tick = Some(next_tick.map_or(t, |n| n.min(t)));
                }
            }
        }
        match next_tick {
            Some(t) => now = t,
            None => break,
        }
    }

    Ok(events)
}
