#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: i + 1 + offset,
                    });
                }

                let mut content_start = open + 1;
                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let first = bytes[content_start];
                if matches!(first, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
                    return Err(Error::UnsupportedOperator { pos: content_start });
                }

                let mut names = Vec::new();
                let mut seen = Vec::<&str>::new();

                loop {
                    let comma = bytes[content_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| content_start + offset);
                    let end = comma.unwrap_or(close);

                    if content_start == end {
                        return Err(Error::EmptyVariable { pos: content_start });
                    }

                    let name_bytes = &bytes[content_start..end];
                    if !name_bytes[0].is_ascii_alphabetic() && name_bytes[0] != b'_' {
                        return Err(Error::InvalidName { pos: content_start });
                    }
                    if let Some(offset) = name_bytes[1..]
                        .iter()
                        .position(|b| !b.is_ascii_alphanumeric() && **b != b'_')
                    {
                        return Err(Error::InvalidName {
                            pos: content_start + 1 + offset,
                        });
                    }

                    let name = &template[content_start..end];
                    if seen.contains(&name) {
                        return Err(Error::DuplicateVariable { pos: content_start });
                    }
                    seen.push(name);
                    names.push(name);

                    match comma {
                        Some(comma_pos) => content_start = comma_pos + 1,
                        None => break,
                    }
                }

                expressions.push((open, close, names));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;
    for (open, close, names) in expressions {
        output.push_str(&template[literal_start..open]);

        let mut first_field = true;
        for name in names {
            if let Some(value) = vars
                .iter()
                .rev()
                .find_map(|(binding, value)| (*binding == name).then_some(*value))
            {
                if !first_field {
                    output.push(',');
                }
                first_field = false;
                encode_value(value, &mut output);
            }
        }

        literal_start = close + 1;
    }
    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn encode_value(value: &str, output: &mut String) {
    let bytes = value.as_bytes();
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(hex_digit(high));
                output.push(hex_digit(low));
                i += 3;
                continue;
            }
        }

        let byte = bytes[i];
        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
        i += 1;
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
