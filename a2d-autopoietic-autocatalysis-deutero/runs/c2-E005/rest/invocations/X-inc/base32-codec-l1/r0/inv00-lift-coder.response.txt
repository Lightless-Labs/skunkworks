#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

const ALPHABET: &[u8; 32] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";

pub fn encode(input: &[u8]) -> String {
    if input.is_empty() {
        return String::new();
    }

    let data_len = (input.len() * 8 + 4) / 5;
    let total_len = (data_len + 7) / 8 * 8;
    let mut output = String::with_capacity(total_len);

    for value_index in 0..data_len {
        let mut value = 0u8;
        for bit_index in 0..5 {
            let stream_bit = value_index * 5 + bit_index;
            value <<= 1;
            if stream_bit < input.len() * 8 {
                let byte = input[stream_bit / 8];
                value |= (byte >> (7 - stream_bit % 8)) & 1;
            }
        }
        output.push(ALPHABET[value as usize] as char);
    }

    output.extend(std::iter::repeat_n('=', total_len - data_len));
    output
}

fn alphabet_value(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&symbol| symbol == byte).map(|v| v as u8)
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let n = bytes.len();

    // Lexical validation.
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && alphabet_value(byte).is_none() {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    // Padding placement.
    let p = bytes.iter().position(|&byte| byte == b'=').unwrap_or(n);
    if let Some(offset) = bytes[p..].iter().position(|&byte| byte != b'=') {
        return Err(DecodeError::Padding { offset: p + offset });
    }

    // Data length.
    let residue = p % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength { offset: n });
    }

    // Padding shape.
    let required_padding = match residue {
        0 => 0,
        2 => 6,
        4 => 4,
        5 => 3,
        7 => 1,
        _ => unreachable!(),
    };
    if bytes.len() - p != required_padding || n % 8 != 0 {
        return Err(DecodeError::Padding { offset: p });
    }

    // Trailing bits.
    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };
    if unused_bits != 0 {
        let final_value = alphabet_value(bytes[p - 1]).unwrap();
        if final_value & ((1u8 << unused_bits) - 1) != 0 {
            return Err(DecodeError::TrailingBits { offset: p - 1 });
        }
    }

    let mut output = Vec::with_capacity(p * 5 / 8);
    let mut accumulator = 0u32;
    let mut bit_count = 0u8;

    for &byte in &bytes[..p] {
        accumulator = (accumulator << 5) | alphabet_value(byte).unwrap() as u32;
        bit_count += 5;
        while bit_count >= 8 {
            bit_count -= 8;
            output.push((accumulator >> bit_count) as u8);
            accumulator &= (1u32 << bit_count) - 1;
        }
    }

    Ok(output)
}
