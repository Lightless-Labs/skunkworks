```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
    cwd: Vec<String>,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
            cwd: Vec::new(),
        }
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        if let Some(index) = path.as_bytes().iter().position(|&byte| byte == 0) {
            return Err(Error::InvalidPath(index));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            self.cwd.clone()
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                name => components.push(name.to_owned()),
            }
        }

        Ok(components)
    }

    fn entry_at<'a>(&'a self, components: &[String]) -> Result<&'a Entry, Error> {
        let mut entry = &self.root;
        let mut prefix = Vec::new();

        for component in components {
            match entry {
                Entry::Dir(children) => {
                    prefix.push(component.clone());
                    entry = children
                        .get(component)
                        .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical(&prefix)));
                }
            }
        }

        Ok(entry)
    }

    fn parent_at<'a>(
        &'a self,
        components: &[String],
    ) -> Result<(&'a Entry, String), Error> {
        let parent_components = &components[..components.len() - 1];
        let parent_path = Self::canonical(parent_components);
        let parent = self.entry_at(parent_components)?;
        match parent {
            Entry::Dir(_) => Ok((parent, parent_path)),
            Entry::File(_) => Err(Error::NotDir(parent_path)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let final_path = Self::canonical(&components);
        let (parent, _) = self.parent_at(&components)?;
        let Entry::Dir(children) = parent else {
            unreachable!();
        };

        if children.contains_key(components.last().unwrap()) {
            return Err(Error::Exists(final_path));
        }

        let parent_components = &components[..components.len() - 1];
        let mut entry = &mut self.root;
        for component in parent_components {
            let Entry::Dir(children) = entry else {
                unreachable!();
            };
            entry = children.get_mut(component).unwrap();
        }
        let Entry::Dir(children) = entry else {
            unreachable!();
        };
        children.insert(
            components.last().unwrap().clone(),
            Entry::Dir(BTreeMap::new()),
        );
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = self.normalize(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let final_path = Self::canonical(&components);
        let (parent, _) = self.parent_at(&components)?;
        let Entry::Dir(children) = parent else {
            unreachable!();
        };

        if matches!(children.get(components.last().unwrap()), Some(Entry::Dir(_))) {
            return Err(Error::IsDir(final_path));
        }

        let parent_components = &components[..components.len() - 1];
        let mut entry = &mut self.root;
        for component in parent_components {
            let Entry::Dir(children) = entry else {
                unreachable!();
            };
            entry = children.get_mut(component).unwrap();
        }
        let Entry::Dir(children) = entry else {
            unreachable!();
        };
        children.insert(
            components.last().unwrap().clone(),
            Entry::File(data.to_vec()),
        );
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = self.normalize(path)?;
        match self.entry_at(&components)? {
            Entry::File(data) => Ok(data.clone()),
            Entry::Dir(_) => Err(Error::IsDir(Self::canonical(&components))),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = self.normalize(path)?;
        match self.entry_at(&components)? {
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
            Entry::File(_) => Err(Error::NotDir(Self::canonical(&components))),
        }
    }

    pub fn cwd(&self) -> String {
        Self::canonical(&self.cwd)
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        match self.entry_at(&components)? {
            Entry::Dir(_) => {
                self.cwd = components;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(Self::canonical(&components))),
        }
    }
}
```
