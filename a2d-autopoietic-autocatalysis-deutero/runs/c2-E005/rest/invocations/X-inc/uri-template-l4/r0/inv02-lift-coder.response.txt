#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy)]
struct Spec<'a> {
    name: &'a str,
    name_pos: usize,
    start_pos: usize,
    modifier: Modifier,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Clone, Copy)]
struct Expression<'a> {
    operator: Option<char>,
    specs: Vec<Spec<'a>>,
}

fn name_char_valid(byte: u8, first: bool) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_' || (!first && byte.is_ascii_digit())
}

fn parse_spec<'a>(text: &'a str, base: usize, start: usize, end: usize) -> Result<Spec<'a>, Error> {
    let bytes = text.as_bytes();
    let mut modifier_at = end;
    for i in start..end {
        if bytes[i] == b'*' || bytes[i] == b':' {
            modifier_at = i;
            break;
        }
    }

    if modifier_at == start {
        return Err(Error::InvalidName { pos: base + start });
    }

    for i in start..modifier_at {
        if !name_char_valid(bytes[i], i == start) {
            return Err(Error::InvalidName { pos: base + i });
        }
    }

    let modifier = if modifier_at == end {
        Modifier::None
    } else {
        let suffix = &text[modifier_at..end];
        if suffix == "*" {
            Modifier::Explode
        } else if suffix.as_bytes().first() == Some(&b':') {
            let digits = &suffix.as_bytes()[1..];
            if digits.is_empty()
                || digits.len() > 4
                || digits[0] == b'0'
                || !digits.iter().all(u8::is_ascii_digit)
            {
                return Err(Error::InvalidModifier {
                    pos: base + modifier_at,
                });
            }
            let mut n = 0usize;
            for digit in digits {
                n = n * 10 + (digit - b'0') as usize;
            }
            if !(1..=9999).contains(&n) {
                return Err(Error::InvalidModifier {
                    pos: base + modifier_at,
                });
            }
            Modifier::Prefix(n)
        } else {
            return Err(Error::InvalidModifier {
                pos: base + modifier_at,
            });
        }
    };

    Ok(Spec {
        name: &text[start..modifier_at],
        name_pos: base + start,
        start_pos: base + start,
        modifier,
    })
}

fn parse_expression<'a>(content: &'a str, base: usize, close_pos: usize) -> Result<Expression<'a>, Error> {
    let bytes = content.as_bytes();
    let mut operator = None;
    let mut start = 0;

    if let Some(&first) = bytes.first() {
        if matches!(first, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
            operator = Some(first as char);
            start = 1;
        } else if matches!(first, b'!' | b'@' | b'=' | b'<' | b'>' | b'|' | b'^' | b'%' ) {
            return Err(Error::UnsupportedOperator { pos: base });
        }
    }

    if start == bytes.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specs = Vec::new();
    let mut seen = std::collections::HashSet::<&str>::new();
    let mut part_start = start;

    loop {
        let part_end = content[part_start..]
            .find(',')
            .map(|i| part_start + i)
            .unwrap_or(content.len());

        if part_start == part_end {
            return Err(Error::EmptyVariable {
                pos: base + part_start,
            });
        }

        let spec = parse_spec(content, base, part_start, part_end)?;
        if seen.contains(spec.name) {
            return Err(Error::DuplicateVariable {
                pos: spec.start_pos,
            });
        }
        seen.insert(spec.name);
        specs.push(spec);

        if part_end == content.len() {
            break;
        }
        part_start = part_end + 1;
    }

    Ok(Expression { operator, specs })
}

fn validate_template(template: &str) -> Result<Vec<(usize, usize, Expression<'_>)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(relative_close) = template[i + 1..].find('}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + relative_close;

                if let Some(relative_open) = template[i + 1..close].find('{') {
                    return Err(Error::NestedOpen {
                        pos: i + 1 + relative_open,
                    });
                }

                let content_start = open + 1;
                let expression = parse_expression(
                    &template[content_start..close],
                    content_start,
                    close,
                )?;
                expressions.push((open, close + 1, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                const HEX: &[u8; 16] = b"0123456789ABCDEF";
                out.push('%');
                out.push(HEX[high as usize] as char);
                out.push(HEX[low as usize] as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            out.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    out
}

fn selected_binding<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn limited_scalar(value: &str, modifier: Modifier) -> String {
    match modifier {
        Modifier::Prefix(n) => value.chars().take(n).collect(),
        Modifier::None | Modifier::Explode => value.to_owned(),
    }
}

fn named_field(name: &str, value: String, semicolon: bool) -> String {
    if semicolon && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expression.operator;
    let reserved = matches!(operator, Some('+' | '#'));
    let explode = |modifier: Modifier| matches!(modifier, Modifier::Explode);

    let mut fields = Vec::<String>::new();

    for spec in &expression.specs {
        let Some(value) = selected_binding(spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_)) && !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch {
                pos: spec.start_pos,
            });
        }

        match value {
            Value::Scalar(scalar) => {
                let scalar = limited_scalar(scalar, spec.modifier);
                let encoded = encode(&scalar, reserved);
                fields.push(if matches!(operator, Some(';' | '?' | '&')) {
                    named_field(spec.name, encoded, operator == Some(';'))
                } else {
                    encoded
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if explode(spec.modifier) {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if matches!(operator, Some(';' | '?' | '&')) {
                            named_field(spec.name, encoded, operator == Some(';'))
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if matches!(operator, Some(';' | '?' | '&')) {
                        named_field(spec.name, joined, operator == Some(';'))
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                let mut keys = std::collections::HashSet::new();
                if entries.iter().any(|(key, _)| !keys.insert(key.as_str())) {
                    return Err(Error::DuplicateKey {
                        pos: spec.start_pos,
                    });
                }
                if entries.is_empty() {
                    continue;
                }

                if explode(spec.modifier) {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        if operator == Some(';') && encoded_value.is_empty() {
                            fields.push(encoded_key);
                        } else {
                            fields.push(format!("{encoded_key}={encoded_value}"));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(if matches!(operator, Some(';' | '?' | '&')) {
                        named_field(spec.name, joined, operator == Some(';'))
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (separator, prefix) = match operator {
        None | Some('+') => (",", ""),
        Some('#') => (",", "#"),
        Some('.') => (".", "."),
        Some('/') => ("/", "/"),
        Some(';') => (";", ";"),
        Some('?') => ("&", "?"),
        Some('&') => ("&", "&"),
        _ => (",", ""),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::with_capacity(template.len());
    let mut cursor = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[cursor..start]);
        output.push_str(&expand_expression(&expression, vars)?);
        cursor = end;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}
