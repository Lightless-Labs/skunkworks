#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventKind {
    Start,
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub task: usize,
    pub worker: usize,
    pub kind: EventKind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub events: Vec<Event>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before checking workers or doing computation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build successor lists and indegrees. Dependency list order does not affect
    // the deterministic Kahn traversal because available IDs are kept ordered.
    let mut successors = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            successors[pred].push(task_id);
        }
    }

    let mut available = std::collections::BinaryHeap::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(std::cmp::Reverse(id));
        }
    }

    let mut topo = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(id)) = available.pop() {
        topo.push(id);
        for &child in &successors[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if topo.len() != n {
        let removed = {
            let mut was_removed = vec![false; n];
            for &id in &topo {
                was_removed[id] = true;
            }
            was_removed
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in topo.iter().rev() {
        let max_child_tail = successors[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::with_capacity(n.saturating_mul(2));
    let mut completed = vec![false; n];
    let mut started = vec![false; n];
    let mut worker_available = vec![0u64; workers];
    let mut running: Vec<(u64, usize, usize)> = Vec::new(); // (finish, task, worker)
    let mut done_count = 0usize;
    let mut now = 0u64;
    let mut makespan = 0u64;

    while done_count < n {
        // Complete all tasks finishing now, in ascending task ID order.
        let mut finishing: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|&(finish, _, _)| finish == now)
            .collect();
        finishing.sort_by_key(|&(_, task, _)| task);

        for &(finish, task, worker) in &finishing {
            events.push(Event {
                time: finish,
                task,
                worker,
                kind: EventKind::Finish,
            });
            completed[task] = true;
            done_count += 1;
            makespan = makespan.max(finish);
        }
        if !finishing.is_empty() {
            running.retain(|&(finish, _, _)| finish != now);
        }

        // Admit one task at a time, reevaluating the ranking after every start.
        loop {
            let worker = (0..workers).find(|&w| worker_available[w] <= now);
            let Some(worker) = worker else {
                break;
            };

            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now {
                    continue;
                }
                if tasks[id].deps.iter().any(|&pred| !completed[pred]) {
                    continue;
                }
                best = match best {
                    None => Some(id),
                    Some(current) => {
                        let candidate_key = (tails[id], tasks[id].priority, std::cmp::Reverse(id));
                        let current_key = (
                            tails[current],
                            tasks[current].priority,
                            std::cmp::Reverse(current),
                        );
                        if candidate_key > current_key {
                            Some(id)
                        } else {
                            Some(current)
                        }
                    }
                };
            }

            let Some(task) = best else {
                break;
            };

            let finish = now.checked_add(tasks[task].duration).ok_or(Error::Overflow {
                task,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task,
                stage: Stage::Turnaround,
            })?;

            started[task] = true;
            worker_available[worker] = available_at;
            runs[task] = Some(Run {
                start: now,
                finish,
                worker,
            });
            events.push(Event {
                time: now,
                task,
                worker,
                kind: EventKind::Start,
            });

            if finish == now {
                events.push(Event {
                    time: now,
                    task,
                    worker,
                    kind: EventKind::Finish,
                });
                completed[task] = true;
                done_count += 1;
                makespan = makespan.max(now);
            } else {
                running.push((finish, task, worker));
            }
        }

        if done_count == n {
            break;
        }

        // Advance to the earliest future event that can change eligibility.
        let mut next = None;
        let mut consider = |time: u64| {
            if time > now {
                next = Some(next.map_or(time, |current: u64| current.min(time)));
            }
        };

        for &(finish, _, _) in &running {
            consider(finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        now = next.expect("unfinished tasks must have a future event");
    }

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        events,
    })
}
