#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    ContextTooLarge { context: usize },
    CarriageReturn { side: Side, byte: usize },
    MissingFinalLf { side: Side },
    TooLarge { old_len: usize, new_len: usize },
}

fn split_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        Vec::new()
    } else {
        text[..text.len() - 1].split('\n').collect()
    }
}

fn diff_lines(old: &[&str], new: &[&str]) -> Vec<Edit> {
    let m = old.len();
    let n = new.len();
    let mut lcs = vec![vec![0usize; n + 1]; m + 1];

    for i in (0..m).rev() {
        for j in (0..n).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut edits = Vec::new();
    let (mut i, mut j) = (0, 0);

    while i < m && j < n {
        if old[i] == new[j] {
            edits.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old[i].to_owned(),
            });
            i += 1;
            j += 1;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            if delete_score > insert_score
                || (delete_score == insert_score && (i + j) % 2 == 0)
            {
                edits.push(Edit::Delete {
                    old_index: i,
                    line: old[i].to_owned(),
                });
                i += 1;
            } else {
                edits.push(Edit::Insert {
                    new_index: j,
                    line: new[j].to_owned(),
                });
                j += 1;
            }
        }
    }

    while i < m {
        edits.push(Edit::Delete {
            old_index: i,
            line: old[i].to_owned(),
        });
        i += 1;
    }
    while j < n {
        edits.push(Edit::Insert {
            new_index: j,
            line: new[j].to_owned(),
        });
        j += 1;
    }

    edits
}

fn is_change(edit: &Edit) -> bool {
    matches!(edit, Edit::Delete { .. } | Edit::Insert { .. })
}

fn render_hunk(edits: &[Edit], start: usize, end: usize, output: &mut String) {
    let old_count = edits[start..end]
        .iter()
        .filter(|e| matches!(e, Edit::Keep { .. } | Edit::Delete { .. }))
        .count();
    let new_count = edits[start..end]
        .iter()
        .filter(|e| matches!(e, Edit::Keep { .. } | Edit::Insert { .. }))
        .count();

    let old_before = edits[..start]
        .iter()
        .filter(|e| matches!(e, Edit::Keep { .. } | Edit::Delete { .. }))
        .count();
    let new_before = edits[..start]
        .iter()
        .filter(|e| matches!(e, Edit::Keep { .. } | Edit::Insert { .. }))
        .count();

    let old_start = if old_count > 0 {
        old_before + 1
    } else {
        old_before
    };
    let new_start = if new_count > 0 {
        new_before + 1
    } else {
        new_before
    };

    output.push_str(&format!(
        "@@ -{old_start},{old_count} +{new_start},{new_count} @@\n"
    ));

    for edit in &edits[start..end] {
        match edit {
            Edit::Keep { line, .. } => {
                output.push(' ');
                output.push_str(line);
                output.push('\n');
            }
            Edit::Delete { line, .. } => {
                output.push('-');
                output.push_str(line);
                output.push('\n');
            }
            Edit::Insert { line, .. } => {
                output.push('+');
                output.push_str(line);
                output.push('\n');
            }
        }
    }
}

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }

    if let Some(byte) = old.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn {
            side: Side::Old,
            byte,
        });
    }
    if let Some(byte) = new.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn {
            side: Side::New,
            byte,
        });
    }

    if !old.is_empty() && !old.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::Old });
    }
    if !new.is_empty() && !new.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::New });
    }

    let old_lines = split_lines(old);
    let new_lines = split_lines(new);

    if old_lines.len() > 128 || new_lines.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old_lines.len(),
            new_len: new_lines.len(),
        });
    }

    let edits = diff_lines(&old_lines, &new_lines);
    if !edits.iter().any(is_change) {
        return Ok(String::new());
    }

    let mut groups = Vec::new();
    let mut index = 0;
    while index < edits.len() {
        if is_change(&edits[index]) {
            let start = index;
            while index < edits.len() && is_change(&edits[index]) {
                index += 1;
            }
            groups.push((start, index));
        } else {
            index += 1;
        }
    }

    let merge_distance = 2 * context;
    let mut merged = Vec::new();
    for group in groups {
        if let Some(last) = merged.last_mut() {
            let gap = group.0 - last.1;
            if gap <= merge_distance {
                last.1 = group.1;
                continue;
            }
        }
        merged.push(group);
    }

    let mut output = String::from("--- before\n+++ after\n");
    let mut previous_end = 0;
    for (group_start, group_end) in merged {
        let start = group_start.saturating_sub(context).max(previous_end);
        let end = (group_end + context).min(edits.len());
        render_hunk(&edits, start, end, &mut output);
        previous_end = end;
    }

    Ok(output)
}
