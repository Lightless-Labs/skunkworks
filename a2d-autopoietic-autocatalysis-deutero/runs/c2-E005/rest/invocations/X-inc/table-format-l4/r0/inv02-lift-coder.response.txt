#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn is_invalid(ch: char) -> bool {
    ch == '\r' || ch == '\u{2028}' || ch == '\u{2029}' || ch.is_control() && ch != '\n'
}

fn normalize_cell(
    text: &str,
    row: usize,
    column: usize,
    markdown: bool,
) -> Result<Vec<Vec<Token>>, TableError> {
    let mut normalized = String::with_capacity(text.len());
    let mut chars = text.char_indices().peekable();

    while let Some((byte, ch)) = chars.next() {
        if ch == '\r' {
            if matches!(chars.peek(), Some((_, '\n'))) {
                chars.next();
                normalized.push('\n');
            } else {
                return Err(TableError::InvalidChar { row, column, byte });
            }
        } else if is_invalid(ch) {
            return Err(TableError::InvalidChar { row, column, byte });
        } else {
            normalized.push(ch);
        }
    }

    Ok(normalized
        .split('\n')
        .map(|line| {
            let trimmed = line.trim_end_matches(' ');
            trimmed
                .chars()
                .map(|ch| {
                    if markdown && ch == '|' {
                        Token {
                            text: "\\|".to_owned(),
                            width: 2,
                        }
                    } else if markdown && ch == '\\' {
                        Token {
                            text: "\\\\".to_owned(),
                            width: 2,
                        }
                    } else {
                        Token {
                            text: ch.to_string(),
                            width: 1,
                        }
                    }
                })
                .collect()
        })
        .collect())
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<Vec<Token>> {
    if tokens.is_empty() {
        return vec![Vec::new()];
    }

    match overflow {
        Overflow::Clip => {
            if token_width(tokens) <= width {
                return vec![tokens.to_vec()];
            }

            let capacity = width.saturating_sub(1);
            let mut result = Vec::new();
            let mut used = 0;
            for token in tokens {
                if used + token.width > capacity {
                    break;
                }
                used += token.width;
                result.push(token.clone());
            }
            result.push(Token {
                text: "…".to_owned(),
                width: 1,
            });
            vec![result]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut current = Vec::new();
            let mut used = 0;

            for token in tokens {
                if used + token.width > width {
                    result.push(current);
                    current = Vec::new();
                    used = 0;
                }
                current.push(token.clone());
                used += token.width;
            }

            if !current.is_empty() {
                result.push(current);
            }
            result
        }
    }
}

fn render_aligned(
    tokens: &[Token],
    width: usize,
    alignment: &Alignment,
    omit_right_padding: bool,
) -> String {
    let content_width = token_width(tokens);
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };

    let mut result = String::new();
    result.extend(std::iter::repeat_n(' ', left));
    for token in tokens {
        result.push_str(&token.text);
    }
    if !omit_right_padding {
        result.extend(std::iter::repeat_n(' ', right));
    }
    result
}

fn emit_data_line(
    out: &mut String,
    cells: &[Vec<Token>],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) {
    match mode {
        Mode::Grid | Mode::Markdown => {
            out.push('|');
            for (i, tokens) in cells.iter().enumerate() {
                out.push(' ');
                out.push_str(&render_aligned(tokens, widths[i], &alignments[i], false));
                out.push(' ');
                out.push('|');
            }
        }
        Mode::Compact => {
            for (i, tokens) in cells.iter().enumerate() {
                if i > 0 {
                    out.push_str("  ");
                }
                out.push_str(&render_aligned(
                    tokens,
                    widths[i],
                    &alignments[i],
                    i + 1 == cells.len(),
                ));
            }
        }
    }
    out.push('\n');
}

fn emit_grid_border(out: &mut String, widths: &[usize]) {
    out.push('+');
    for &width in widths {
        out.extend(std::iter::repeat_n('-', width + 2));
        out.push('+');
    }
    out.push('\n');
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let column_count = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != column_count {
            return Err(TableError::Ragged {
                row,
                expected: column_count,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != column_count {
        return Err(TableError::AlignmentCount {
            expected: column_count,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != column_count {
        return Err(TableError::OverflowCount {
            expected: column_count,
            actual: options.overflow.len(),
        });
    }

    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_column_width = if matches!(options.mode, Mode::Markdown) {
        3
    } else {
        1
    };
    let overhead = match options.mode {
        Mode::Compact => 2 * (column_count - 1),
        Mode::Grid | Mode::Markdown => 3 * column_count + 1,
    };
    let minimum = overhead + column_count * minimum_column_width;

    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; column_count]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; column_count]
    } else {
        options.overflow.clone()
    };

    let markdown = matches!(options.mode, Mode::Markdown);
    let mut normalized_rows = Vec::with_capacity(rows.len());
    let mut widths = vec![minimum_column_width; column_count];

    for (row_index, row) in rows.iter().enumerate() {
        let mut normalized_row = Vec::with_capacity(column_count);
        for (column_index, cell) in row.iter().enumerate() {
            let lines = normalize_cell(cell, row_index, column_index, markdown)?;
            for line in &lines {
                widths[column_index] = widths[column_index].max(token_width(line));
            }
            normalized_row.push(lines);
        }
        normalized_rows.push(normalized_row);
    }

    if options.max_width != 0 {
        let mut total: usize = widths.iter().sum::<usize>() + overhead;
        while total > options.max_width {
            let eligible = (0..column_count)
                .filter(|&i| widths[i] > minimum_column_width)
                .max_by_key(|&i| (widths[i], i));

            if let Some(column) = eligible {
                widths[column] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    let mut out = String::new();
    if matches!(options.mode, Mode::Grid) {
        emit_grid_border(&mut out, &widths);
    }

    for (row_index, row) in normalized_rows.iter().enumerate() {
        let mut cell_fragments: Vec<Vec<Vec<Token>>> = Vec::with_capacity(column_count);
        let mut height = 1;

        for column_index in 0..column_count {
            let mut all_lines = Vec::new();
            for line in &row[column_index] {
                let line_fragments = fragments(
                    line,
                    widths[column_index],
                    &overflows[column_index],
                );
                height = height.max(line_fragments.len());
                all_lines.extend(line_fragments);
            }
            cell_fragments.push(all_lines);
        }

        // Each cell's logical lines are laid out independently in order. Track
        // the fragment sequence for each cell and pad shorter sequences per row.
        for fragment_index in 0..height {
            let mut physical_cells = Vec::with_capacity(column_count);
            for column_index in 0..column_count {
                physical_cells.push(
                    cell_fragments[column_index]
                        .get(fragment_index)
                        .cloned()
                        .unwrap_or_default(),
                );
            }
            emit_data_line(
                &mut out,
                &physical_cells,
                &widths,
                &alignments,
                &options.mode,
            );
        }

        if matches!(options.mode, Mode::Grid) && options.header && row_index == 0 && rows.len() > 1 {
            emit_grid_border(&mut out, &widths);
        }

        if matches!(options.mode, Mode::Markdown) && options.header && row_index == 0 {
            out.push('|');
            for column_index in 0..column_count {
                let marker = match alignments[column_index] {
                    Alignment::Left => {
                        format!(":{}", "-".repeat(widths[column_index] - 1))
                    }
                    Alignment::Right => {
                        format!("{}:", "-".repeat(widths[column_index] - 1))
                    }
                    Alignment::Center => {
                        format!(":{}:", "-".repeat(widths[column_index] - 2))
                    }
                };
                out.push(' ');
                out.push_str(&marker);
                out.push(' ');
                out.push('|');
            }
            out.push('\n');
        }
    }

    Ok(out)
}
