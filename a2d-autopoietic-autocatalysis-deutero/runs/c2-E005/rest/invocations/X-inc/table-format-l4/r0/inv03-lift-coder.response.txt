#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn invalid_scalar(c: char) -> bool {
    matches!(c, '\r' | '\u{2028}' | '\u{2029}') || (c.is_control() && c != '\n')
}

fn validate_cell(
    cell: &str,
    row: usize,
    column: usize,
) -> Result<(), TableError> {
    let chars: Vec<(usize, char)> = cell.char_indices().collect();
    let mut i = 0;
    while i < chars.len() {
        let (byte, c) = chars[i];
        if c == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 2;
                continue;
            }
            return Err(TableError::InvalidChar { row, column, byte });
        }
        if invalid_scalar(c) {
            return Err(TableError::InvalidChar { row, column, byte });
        }
        i += 1;
    }
    Ok(())
}

fn make_tokens(line: &str, markdown: bool) -> Vec<Token> {
    let trimmed = line.trim_end_matches(' ');
    let mut out = Vec::new();
    for c in trimmed.chars() {
        match (markdown, c) {
            (true, '|') => out.push(Token { text: "\\|".to_owned(), width: 2 }),
            (true, '\\') => out.push(Token { text: "\\\\".to_owned(), width: 2 }),
            _ => out.push(Token { text: c.to_string(), width: 1 }),
        }
    }
    out
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn split_fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }
            let mut s = String::new();
            let mut used = 0;
            for token in tokens {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                used += token.width;
                s.push_str(&token.text);
            }
            s.push('…');
            vec![s]
        }
        Overflow::Wrap => {
            let mut out = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                // Width minima guarantee progress for all tokens.
                if end == start {
                    end += 1;
                }
                out.push(tokens[start..end].iter().map(|t| t.text.as_str()).collect());
                start = end;
            }
            out
        }
    }
}

fn scalar_len(s: &str) -> usize {
    s.chars().count()
}

fn align_content(s: &str, width: usize, alignment: &Alignment, omit_right: bool) -> String {
    let extra = width.saturating_sub(scalar_len(s));
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let mut out = String::new();
    out.extend(std::iter::repeat_n(' ', left));
    out.push_str(s);
    if !omit_right {
        out.extend(std::iter::repeat_n(' ', right));
    }
    out
}

fn border(widths: &[usize]) -> String {
    let mut out = String::from("+");
    for &width in widths {
        out.extend(std::iter::repeat_n('-', width + 2));
        out.push('+');
    }
    out.push('\n');
    out
}

fn data_line(fragments: &[String], widths: &[usize], alignments: &[Alignment], mode: &Mode) -> String {
    let mut out = String::new();
    match mode {
        Mode::Grid | Mode::Markdown => {
            out.push('|');
            for i in 0..widths.len() {
                out.push(' ');
                out.push_str(&align_content(&fragments[i], widths[i], &alignments[i], false));
                out.push(' ');
                out.push('|');
            }
        }
        Mode::Compact => {
            for i in 0..widths.len() {
                if i > 0 {
                    out.push_str("  ");
                }
                out.push_str(&align_content(
                    &fragments[i],
                    widths[i],
                    &alignments[i],
                    i + 1 == widths.len(),
                ));
            }
        }
    }
    out.push('\n');
    out
}

fn markdown_marker(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut out = String::from("|");
    for i in 0..widths.len() {
        let width = widths[i];
        out.push(' ');
        match alignments[i] {
            Alignment::Left => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 1));
            }
            Alignment::Right => {
                out.extend(std::iter::repeat_n('-', width - 1));
                out.push(':');
            }
            Alignment::Center => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 2));
                out.push(':');
            }
        }
        out.push(' ');
        out.push('|');
    }
    out.push('\n');
    out
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let n = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != n {
            return Err(TableError::Ragged {
                row: row_index,
                expected: n,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }

    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_width = if matches!(options.mode, Mode::Markdown) { 3 } else { 1 };
    let overhead = match options.mode {
        Mode::Compact => 2 * n.saturating_sub(1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let minimum = overhead + n * minimum_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row_index, row) in rows.iter().enumerate() {
        for (column, cell) in row.iter().enumerate() {
            validate_cell(cell, row_index, column)?;
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let markdown = matches!(options.mode, Mode::Markdown);
    let mut logical: Vec<Vec<Vec<Vec<Token>>>> = Vec::with_capacity(rows.len());
    let mut widths = vec![minimum_width; n];

    for row in rows {
        let mut logical_row = Vec::with_capacity(n);
        for (column, cell) in row.iter().enumerate() {
            let normalized = cell.replace("\r\n", "\n");
            let lines: Vec<Vec<Token>> = normalized
                .split('\n')
                .map(|line| make_tokens(line, markdown))
                .collect();
            for line in &lines {
                widths[column] = widths[column].max(token_width(line));
            }
            logical_row.push(lines);
        }
        logical.push(logical_row);
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let mut candidate: Option<usize> = None;
            for i in 0..n {
                if widths[i] > minimum_width
                    && candidate.map_or(true, |best| widths[i] >= widths[best])
                {
                    candidate = Some(i);
                }
            }
            if let Some(i) = candidate {
                widths[i] -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    if matches!(options.mode, Mode::Grid) {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in logical.iter().enumerate() {
        let mut cell_fragments: Vec<Vec<String>> = Vec::with_capacity(n);
        let mut height = 1;
        for column in 0..n {
            let mut fragments = Vec::new();
            for line in &row[column] {
                fragments.extend(split_fragments(line, widths[column], &overflows[column]));
            }
            height = height.max(fragments.len());
            cell_fragments.push(fragments);
        }

        for fragment_index in 0..height {
            let fragments: Vec<String> = (0..n)
                .map(|column| {
                    cell_fragments[column]
                        .get(fragment_index)
                        .cloned()
                        .unwrap_or_default()
                })
                .collect();
            output.push_str(&data_line(&fragments, &widths, &alignments, &options.mode));
        }

        if matches!(options.mode, Mode::Markdown) && row_index == 0 {
            output.push_str(&markdown_marker(&widths, &alignments));
        } else if matches!(options.mode, Mode::Grid)
            && options.header
            && row_index == 0
            && rows.len() > 1
        {
            output.push_str(&border(&widths));
        }
    }

    if matches!(options.mode, Mode::Grid) {
        output.push_str(&border(&widths));
    }

    Ok(output)
}
