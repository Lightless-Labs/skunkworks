#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut ids = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut next_arrival = 0;
    let mut current_time = 0u64;
    let mut finish: Option<u64> = None;
    let mut last_finish: Option<u64> = None;
    let mut first_dispatch = true;

    loop {
        let next_arrival_time = arrivals.get(next_arrival).map(|c| c.arrival);
        let next_time = match (finish, next_arrival_time) {
            (Some(f), Some(a)) => f.min(a),
            (Some(f), None) => f,
            (None, Some(a)) => a,
            (None, None) => break,
        };
        current_time = next_time;

        if finish == Some(current_time) {
            let customer_id = match events.last() {
                Some(Event {
                    customer,
                    kind: Kind::Start { finish: f, .. },
                    ..
                }) if *f == current_time => *customer,
                _ => {
                    // The active customer's start event may not be the last event.
                    let start_index = events.iter().rposition(|event| {
                        matches!(event.kind, Kind::Start { finish: f, .. } if f == current_time)
                    });
                    start_index
                        .map(|index| events[index].customer)
                        .unwrap_or(0)
                }
            };
            events.push(Event {
                time: current_time,
                customer: customer_id,
                kind: Kind::Finish,
            });
            finish = None;
            last_finish = Some(current_time);
        }

        while next_arrival < arrivals.len()
            && arrivals[next_arrival].arrival == current_time
        {
            let customer = arrivals[next_arrival];
            events.push(Event {
                time: current_time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            next_arrival += 1;
        }

        if finish.is_none() {
            if let Some(customer) = waiting.pop_front() {
                let wake = first_dispatch
                    || last_finish.is_some_and(|previous_finish| current_time > previous_finish);
                first_dispatch = false;

                let base_finish = current_time.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: current_time,
                    },
                )?;
                let customer_finish = if wake {
                    base_finish.checked_add(wake_cost).ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time: current_time,
                    })?
                } else {
                    base_finish
                };

                events.push(Event {
                    time: current_time,
                    customer: customer.id,
                    kind: Kind::Start {
                        finish: customer_finish,
                        wake,
                    },
                });

                if customer_finish == current_time {
                    events.push(Event {
                        time: current_time,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    finish = None;
                    last_finish = Some(current_time);
                } else {
                    finish = Some(customer_finish);
                }
            }
        }
    }

    Ok(events)
}
