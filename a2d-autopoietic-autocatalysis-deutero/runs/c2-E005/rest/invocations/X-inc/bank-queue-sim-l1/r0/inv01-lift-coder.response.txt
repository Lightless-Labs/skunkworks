#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut ids = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| b.arrival.cmp(&a.arrival).then_with(|| a.id.cmp(&b.id)));

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut next_arrival = 0;
    let mut current_time = 0u64;
    let mut active: Option<(u32, u64)> = None;
    let mut previous_finish: Option<u64> = None;
    let mut first_dispatch = true;

    loop {
        let next_arrival_time = arrivals.get(next_arrival).map(|c| c.arrival);
        let next_finish_time = active.map(|(_, finish)| finish);

        let time = match (next_arrival_time, next_finish_time) {
            (None, None) => break,
            (Some(a), None) => a,
            (None, Some(f)) => f,
            (Some(a), Some(f)) => a.min(f),
        };

        current_time = time;

        if let Some((customer_id, finish)) = active {
            if finish == time {
                events.push(Event {
                    time,
                    customer: customer_id,
                    kind: Kind::Finish,
                });
                active = None;
                previous_finish = Some(time);
            }
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = arrivals[next_arrival];
            events.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            next_arrival += 1;
        }

        if active.is_none() {
            if let Some(customer) = waiting.pop_front() {
                let wake = first_dispatch || previous_finish.is_some_and(|finish| time > finish);
                first_dispatch = false;

                let service_finish = time.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time,
                    },
                )?;
                let finish = if wake {
                    service_finish.checked_add(wake_cost).ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time,
                    })?
                } else {
                    service_finish
                };

                events.push(Event {
                    time,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });

                if finish == time {
                    events.push(Event {
                        time,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    previous_finish = Some(time);
                } else {
                    active = Some((customer.id, finish));
                }

                // Process zero-duration occupations inline, preserving FIFO dispatch order.
                while active.is_none() {
                    let Some(next_customer) = waiting.pop_front() else {
                        break;
                    };

                    let wake = previous_finish.is_some_and(|finish| time > finish);
                    let service_finish = time.checked_add(next_customer.service).ok_or(
                        Error::TimeOverflow {
                            customer: next_customer.id,
                            time,
                        },
                    )?;
                    let finish = if wake {
                        service_finish.checked_add(wake_cost).ok_or(Error::TimeOverflow {
                            customer: next_customer.id,
                            time,
                        })?
                    } else {
                        service_finish
                    };

                    events.push(Event {
                        time,
                        customer: next_customer.id,
                        kind: Kind::Start { finish, wake },
                    });

                    if finish == time {
                        events.push(Event {
                            time,
                            customer: next_customer.id,
                            kind: Kind::Finish,
                        });
                        previous_finish = Some(time);
                    } else {
                        active = Some((next_customer.id, finish));
                    }
                }
            }
        }

        let _ = current_time;
    }

    Ok(events)
}
