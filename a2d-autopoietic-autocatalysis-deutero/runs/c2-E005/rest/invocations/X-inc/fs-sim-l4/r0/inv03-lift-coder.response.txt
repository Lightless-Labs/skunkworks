```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn validate(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    if path == "/" {
        return Vec::new();
    }
    path.trim_start_matches('/')
        .split('/')
        .filter(|part| !part.is_empty() && *part != ".")
        .fold(Vec::new(), |mut acc, part| {
            if part == ".." {
                acc.pop();
            } else {
                acc.push(part.to_owned());
            }
            acc
        })
}

fn normalize(path: &str, cwd: &str) -> String {
    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        components(cwd)
    };
    for part in path.split('/') {
        if part.is_empty() || part == "." {
            continue;
        }
        if part == ".." {
            parts.pop();
        } else {
            parts.push(part.to_owned());
        }
    }
    join(&parts)
}

fn join(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn parent(path: &str) -> String {
    if path == "/" {
        return "/".to_owned();
    }
    match path.rfind('/') {
        Some(0) => "/".to_owned(),
        Some(i) => path[..i].to_owned(),
        None => "/".to_owned(),
    }
}

fn basename(path: &str) -> &str {
    path.rsplit('/').next().unwrap_or("")
}

fn is_within(path: &str, dir: &str) -> bool {
    path == dir || dir == "/" || path.starts_with(&format!("{}/", dir))
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn kind(&self, path: &str) -> Option<Kind> {
        self.entries.get(path).map(|e| match e {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        })
    }

    fn resolve(&self, path: &str, allow_missing_final: bool) -> Result<String, Error> {
        let mut pending = components(path);
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                return Ok(join(&resolved));
            }
            let part = pending.remove(0);
            resolved.push(part);
            let current = join(&resolved);
            let entry = match self.entries.get(&current) {
                Some(e) => e,
                None => {
                    if allow_missing_final && pending.is_empty() {
                        return Ok(current);
                    }
                    return Err(Error::Missing(current));
                }
            };

            if matches!(entry, Entry::Dir) {
                if pending.is_empty() {
                    return Ok(current);
                }
                continue;
            }

            if let Entry::Link(target) = entry {
                if !visited.insert(current.clone()) {
                    return Err(Error::Loop(current));
                }
                if expansions >= 8 {
                    return Err(Error::Depth(current));
                }
                expansions += 1;

                let mut target_parts = if target.starts_with('/') {
                    Vec::new()
                } else {
                    components(&parent(&current))
                };
                for p in target.split('/') {
                    if p.is_empty() || p == "." {
                        continue;
                    }
                    if p == ".." {
                        target_parts.pop();
                    } else {
                        target_parts.push(p.to_owned());
                    }
                }
                let mut expanded = target_parts;
                expanded.extend(pending.into_iter());
                pending = expanded;
                resolved.clear();
                continue;
            }

            if pending.is_empty() {
                return Ok(current);
            }
            return Err(Error::NotDir(current));
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<String, Error> {
        let p = parent(path);
        let resolved = self.resolve(&p, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Ok(resolved),
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    fn check_path(path: &str, cwd: &str) -> Result<String, Error> {
        validate(path)?;
        Ok(normalize(path, cwd))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = Self::check_path(path, &self.cwd)?;
        let p = self.resolve_parent(&path)?;
        if path == "/" {
            return Err(Error::Exists("/".to_owned()));
        }
        let dest = format!("{}/{}", p.trim_end_matches('/'), basename(&path));
        if self.entries.contains_key(&dest) {
            return Err(Error::Exists(dest));
        }
        self.entries.insert(dest, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = Self::check_path(path, &self.cwd)?;
        let dest = self.resolve(&path, true)?;
        if dest == "/" {
            return Err(Error::IsDir("/".to_owned()));
        }

        let existing = self.entries.get(&dest);
        if matches!(existing, Some(Entry::Dir)) {
            return Err(Error::IsDir(dest));
        }
        if existing.is_none() {
            self.resolve_parent(&dest)?;
        }
        self.entries.insert(dest, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = Self::check_path(path, &self.cwd)?;
        let resolved = self.resolve(&path, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = Self::check_path(path, &self.cwd)?;
        let resolved = self.resolve(&path, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }
        let prefix = if resolved == "/" {
            "/".to_owned()
        } else {
            format!("{}/", resolved)
        };
        let mut result = Vec::new();
        for (entry_path, entry) in &self.entries {
            if !entry_path.starts_with(&prefix) || entry_path == &resolved {
                continue;
            }
            let rest = &entry_path[prefix.len()..];
            if rest.contains('/') {
                continue;
            }
            let kind = match entry {
                Entry::Dir => Kind::Dir,
                Entry::File(_) => Kind::File,
                Entry::Link(_) => Kind::Link,
            };
            result.push((rest.to_owned(), kind));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = Self::check_path(path, &self.cwd)?;
        let resolved = self.resolve(&path, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = Self::check_path(from, &self.cwd)?;
        let to = Self::check_path(to, &self.cwd)?;

        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        let src_parent = self.resolve_parent(&from)?;
        let src = format!("{}/{}", src_parent.trim_end_matches('/'), basename(&from));
        let src_entry = self.entries.get(&src).cloned().ok_or_else(|| Error::Missing(src.clone()))?;

        if matches!(src_entry, Entry::Dir) && is_within(&self.cwd, &src) {
            return Err(Error::Busy(src));
        }

        let dst_parent = self.resolve_parent(&to)?;
        let dst = format!("{}/{}", dst_parent.trim_end_matches('/'), basename(&to));

        if matches!(self.entries.get(&dst), Some(Entry::Dir)) && is_within(&self.cwd, &dst) {
            return Err(Error::Busy(dst));
        }

        if src == dst {
            return Ok(());
        }
        if matches!(src_entry, Entry::Dir) && is_within(&dst, &src) {
            return Err(Error::Cycle(dst));
        }

        let dst_entry = self.entries.get(&dst).cloned();
        if dst_entry.is_some() && !overwrite {
            return Err(Error::Exists(dst));
        }

        if let Some(ref de) = dst_entry {
            match (&src_entry, de) {
                (Entry::Dir, Entry::Dir) => {
                    if self.entries.keys().any(|p| p != &dst && is_within(p, &dst)) {
                        return Err(Error::NotEmpty(dst));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst)),
                _ => {}
            }
        }

        let mut moved = Vec::new();
        for (p, e) in &self.entries {
            if is_within(p, &src) {
                let suffix = &p[src.len()..];
                moved.push((format!("{}{}", dst, suffix), e.clone()));
            }
        }

        if dst_entry.is_some() {
            let old: Vec<String> = self.entries.keys()
                .filter(|p| is_within(p, &dst))
                .cloned()
                .collect();
            for p in old {
                self.entries.remove(&p);
            }
        }

        let old: Vec<String> = self.entries.keys()
            .filter(|p| is_within(p, &src))
            .cloned()
            .collect();
        for p in old {
            self.entries.remove(&p);
        }
        for (p, e) in moved {
            self.entries.insert(p, e);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = Self::check_path(path, &self.cwd)?;
        if path == "/" {
            return Err(Error::Root);
        }
        let p = self.resolve_parent(&path)?;
        let dest = format!("{}/{}", p.trim_end_matches('/'), basename(&path));
        let entry = self.entries.get(&dest).cloned().ok_or_else(|| Error::Missing(dest.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &dest) {
            return Err(Error::Busy(dest));
        }

        let children = self.entries.keys().any(|key| key != &dest && is_within(key, &dest));
        if matches!(entry, Entry::Dir) && children && !recursive {
            return Err(Error::NotEmpty(dest));
        }

        let keys: Vec<String> = self.entries.keys()
            .filter(|key| is_within(key, &dest))
            .cloned()
            .collect();
        for key in keys {
            self.entries.remove(&key);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let link = Self::check_path(link, &self.cwd)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let p = self.resolve_parent(&link)?;
        let dest = format!("{}/{}", p.trim_end_matches('/'), basename(&link));
        if self.entries.contains_key(&dest) {
            return Err(Error::Exists(dest));
        }
        self.entries.insert(dest, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let path = Self::check_path(path, &self.cwd)?;
        if path == "/" {
            return Err(Error::InvalidTarget(0));
        }
        let p = self.resolve_parent(&path)?;
        let dest = format!("{}/{}", p.trim_end_matches('/'), basename(&path));
        match self.entries.get(&dest) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(dest)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let path = Self::check_path(path, &self.cwd)?;
        let root = self.resolve(&path, false)?;
        let entry = self.entries.get(&root).ok_or_else(|| Error::Missing(root.clone()))?;
        let mut rows = Vec::new();
        let prefix = if root == "/" {
            "/".to_owned()
        } else {
            format!("{}/", root)
        };
        for (p, e) in &self.entries {
            if p != &root && !p.starts_with(&prefix) {
                continue;
            }
            let size = match e {
                Entry::Dir => 0,
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
            };
            rows.push((p.clone(), size));
        }
        if matches!(entry, Entry::Dir) {
            let total = rows.iter().filter(|(p, _)| p != &root)
                .map(|(_, size)| *size)
                .sum();
            if let Some(row) = rows.iter_mut().find(|(p, _)| p == &root) {
                row.1 = total;
            }
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}
```
