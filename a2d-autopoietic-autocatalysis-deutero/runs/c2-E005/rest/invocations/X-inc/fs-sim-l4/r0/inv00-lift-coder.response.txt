```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn normalize_from(base: &str, path: &str) -> String {
        let mut parts: Vec<&str> = Vec::new();
        if !path.starts_with('/') && base != "/" {
            parts.extend(base[1..].split('/').filter(|s| !s.is_empty()));
        }
        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other),
            }
        }
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        Ok(Self::normalize_from(&self.cwd, path))
    }

    fn parent(path: &str) -> (String, String) {
        if path == "/" {
            return ("/".to_string(), String::new());
        }
        match path.rfind('/') {
            Some(0) => ("/".to_string(), path[1..].to_string()),
            Some(i) => (path[..i].to_string(), path[i + 1..].to_string()),
            None => ("/".to_string(), path.to_string()),
        }
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_string).collect()
        }
    }

    fn build(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn is_within(path: &str, ancestor: &str) -> bool {
        path == ancestor
            || ancestor == "/"
            || path
                .strip_prefix(ancestor)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                if *candidate == "/" || *candidate == path {
                    return false;
                }
                let (parent, _) = Self::parent(candidate);
                parent == path
            })
            .cloned()
            .collect()
    }

    fn expand_target(link_path: &str, target: &str, suffix: &[String]) -> String {
        let base = if target.starts_with('/') {
            "/".to_string()
        } else {
            Self::parent(link_path).0
        };
        let mut parts = Self::components(&Self::normalize_from(&base, target));
        parts.extend(suffix.iter().cloned());
        Self::build(&parts)
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        create_final: bool,
    ) -> Result<String, Error> {
        let mut pending = Self::components(path);
        let mut current: Vec<String> = Vec::new();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                return Ok(Self::build(&current));
            }

            let component = pending.remove(0);
            let candidate_parts = {
                let mut p = current.clone();
                p.push(component);
                p
            };
            let candidate = Self::build(&candidate_parts);
            let is_final = pending.is_empty();
            let should_follow = !is_final || follow_final;

            match self.entries.get(&candidate) {
                None if create_final && is_final => {
                    let parent = Self::build(&current);
                    match self.entries.get(&parent) {
                        None => return Err(Error::Missing(parent)),
                        Some(Entry::Dir) => return Ok(candidate),
                        Some(_) => return Err(Error::NotDir(parent)),
                    }
                }
                None => return Err(Error::Missing(candidate)),
                Some(Entry::Link(target)) if should_follow => {
                    if visited.contains(&candidate) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    visited.insert(candidate.clone());
                    expansions += 1;
                    let expanded = Self::expand_target(&candidate, target, &pending);
                    pending = Self::components(&expanded);
                    current.clear();
                }
                Some(_) if !is_final => {
                    match self.entries.get(&candidate) {
                        Some(Entry::Dir) => current.push(
                            candidate_parts
                                .last()
                                .expect("component exists")
                                .clone(),
                        ),
                        Some(_) => return Err(Error::NotDir(candidate)),
                        None => unreachable!(),
                    }
                }
                Some(_) => return Ok(candidate),
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = Self::parent(path);
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, name)),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn subtree_paths(&self, root: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|path| Self::is_within(path, root))
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let (parent, name) = Self::parent(&canonical);
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }
        let destination = if name.is_empty() {
            resolved_parent
        } else if resolved_parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", resolved_parent, name)
        };
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, true, true)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let (parent, _) = Self::parent(&resolved);
                match self.entries.get(&parent) {
                    Some(Entry::Dir) => {
                        self.entries.insert(resolved, Entry::File(data.to_vec()));
                        Ok(())
                    }
                    Some(_) => Err(Error::NotDir(parent)),
                    None => Err(Error::Missing(parent)),
                }
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                let mut out: Vec<(String, Kind)> = self
                    .children(&resolved)
                    .into_iter()
                    .map(|child| {
                        let name = Self::parent(&child).1;
                        (name, self.entries[&child].kind())
                    })
                    .collect();
                out.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(out)
            }
            _ => Err(Error::NotDir(resolved)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            _ => Err(Error::NotDir(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = Self::parent(&source);
        let source_parent = self.resolve(&source_parent, true, false)?;
        match self.entries.get(&source_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(source_parent)),
            None => return Err(Error::Missing(source_parent)),
        }
        let source_path = if source_parent == "/" {
            format!("/{}", source_name)
        } else {
            format!("{}/{}", source_parent, source_name)
        };
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent, destination_name) = Self::parent(&destination);
        let destination_parent = self.resolve(&destination_parent, true, false)?;
        match self.entries.get(&destination_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(destination_parent)),
            None => return Err(Error::Missing(destination_parent)),
        }
        let destination_path = if destination_parent == "/" {
            format!("/{}", destination_name)
        } else {
            format!("{}/{}", destination_parent, destination_name)
        };

        if let Some(Entry::Dir) = self.entries.get(&destination_path) {
            if Self::is_within(&self.cwd, &destination_path) {
                return Err(Error::Busy(destination_path));
            }
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&destination_path, &source_path) {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(ref existing) = destination_entry {
            match (&source_entry, existing) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&destination_path).is_empty() {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        let moving = self.subtree_paths(&source_path);
        let entries_to_move: Vec<(String, Entry)> = moving
            .iter()
            .map(|path| {
                (
                    path.clone(),
                    self.entries.get(path).expect("subtree entry exists").clone(),
                )
            })
            .collect();

        if destination_entry.is_some() {
            let replacing = self.subtree_paths(&destination_path);
            for path in replacing {
                self.entries.remove(&path);
            }
        }
        for path in &moving {
            self.entries.remove(path);
        }
        for (old_path, entry) in entries_to_move {
            let suffix = if old_path == source_path {
                String::new()
            } else {
                old_path[source_path.len()..].to_string()
            };
            let new_path = format!("{}{}", destination_path, suffix);
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Root);
        }

        let (parent, name) = Self::parent(&canonical);
        let parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }
        let target = if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        let entry = self
            .entries
            .get(&target)
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && !recursive && !self.children(&target).is_empty() {
            return Err(Error::NotEmpty(target));
        }

        for child in self.subtree_paths(&target) {
            self.entries.remove(&child);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical_link = self.normalize(link)?;
        Self::validate(target).map_err(|error| match error {
            Error::InvalidPath(i) => Error::InvalidTarget(i),
            other => other,
        })?;
        let (parent, name) = Self::parent(&canonical_link);
        let parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }
        let destination = if name.is_empty() {
            parent
        } else if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries
            .insert(destination, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = self.normalize(path)?;
        let (parent, name) = Self::parent(&canonical);
        let parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }
        let target_path = if name.is_empty() {
            parent
        } else if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        match self.entries.get(&target_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            _ => Err(Error::InvalidTarget(0)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, true, false)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let paths = self.subtree_paths(&resolved);
        let mut sizes = BTreeMap::new();
        for item in &paths {
            let size = match self.entries.get(item).expect("subtree entry exists") {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            sizes.insert(item.clone(), size);
        }

        let mut result = Vec::new();
        for item in paths {
            let total = if matches!(self.entries.get(&item), Some(Entry::Dir)) {
                sizes
                    .iter()
                    .filter(|(candidate, _)| *candidate != &item && Self::is_within(candidate, &item))
                    .map(|(_, size)| *size)
                    .sum()
            } else {
                *sizes.get(&item).expect("entry size exists")
            };
            result.push((item, total));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
