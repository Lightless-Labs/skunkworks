```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

#[derive(Debug, Clone)]
pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_string).collect()
        }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(&self.cwd)
        };
        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other.to_string()),
            }
        }
        Ok(Self::canonical(&parts))
    }

    fn parent_and_name(path: &str) -> (String, String) {
        if path == "/" {
            return ("/".to_string(), String::new());
        }
        let idx = path.rfind('/').unwrap_or(0);
        let parent = if idx == 0 {
            "/".to_string()
        } else {
            path[..idx].to_string()
        };
        (parent, path[idx + 1..].to_string())
    }

    fn join(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn is_within(path: &str, ancestor: &str) -> bool {
        if ancestor == "/" {
            return path.starts_with('/');
        }
        path == ancestor
            || path
                .strip_prefix(ancestor)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn normalize_from(base: &str, input: &str) -> String {
        let mut parts = if input.starts_with('/') {
            Vec::new()
        } else {
            Self::components(base)
        };
        for part in input.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other.to_string()),
            }
        }
        Self::canonical(&parts)
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        create_final: bool,
    ) -> Result<String, Error> {
        let mut todo = Self::components(path);
        let mut done: Vec<String> = Vec::new();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        loop {
            if todo.is_empty() {
                return Ok(Self::canonical(&done));
            }

            let name = todo.remove(0);
            let prefix = Self::canonical(&done);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(_) => return Err(Error::NotDir(prefix)),
            }

            let candidate = Self::canonical(&[done.clone(), vec![name.clone()]].concat());
            let is_final = todo.is_empty();
            let entry = self.entries.get(&candidate);

            if is_final && !follow_final {
                return match entry {
                    Some(_) => Ok(candidate),
                    None if create_final => Ok(candidate),
                    None => Err(Error::Missing(candidate)),
                };
            }

            match entry {
                Some(Entry::Link(target)) => {
                    if !visited.insert(candidate.clone()) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    expansions += 1;

                    let physical_parent = Self::canonical(&done);
                    let mut expanded = Self::components(&Self::normalize_from(
                        &physical_parent,
                        target,
                    ));
                    expanded.extend(todo);
                    let expanded_path = Self::canonical(&expanded);
                    todo = Self::components(&expanded_path);
                    done.clear();
                }
                Some(_) => {
                    done.push(name);
                    if !is_final {
                        match self.entries.get(&Self::canonical(&done)) {
                            Some(Entry::Dir) => {}
                            Some(_) => {
                                return Err(Error::NotDir(Self::canonical(&done)));
                            }
                            None => {
                                return Err(Error::Missing(Self::canonical(&done)));
                            }
                        }
                    }
                }
                None if is_final && create_final => {
                    done.push(name);
                    return Ok(Self::canonical(&done));
                }
                None => return Err(Error::Missing(candidate)),
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        if path == "/" {
            return Ok(("/".to_string(), String::new()));
        }
        let (parent, name) = Self::parent_and_name(path);
        let physical_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&physical_parent) {
            Some(Entry::Dir) => Ok((physical_parent, name)),
            Some(_) => Err(Error::NotDir(physical_parent)),
            None => Err(Error::Missing(physical_parent)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let (parent, name) = self.resolve_parent(&normalized)?;
        let target = if name.is_empty() {
            "/".to_string()
        } else {
            Self::join(&parent, &name)
        };
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, true)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let (parent, _) = Self::parent_and_name(&resolved);
                match self.entries.get(&parent) {
                    Some(Entry::Dir) => {
                        self.entries
                            .insert(resolved, Entry::File(data.to_vec()));
                        Ok(())
                    }
                    Some(_) => Err(Error::NotDir(parent)),
                    None => Err(Error::Missing(parent)),
                }
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) | None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                let mut result = Vec::new();
                for (name, entry) in &self.entries {
                    if name == "/" {
                        continue;
                    }
                    let (parent, child) = Self::parent_and_name(name);
                    if parent == resolved {
                        result.push((child, entry.kind()));
                    }
                }
                result.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(result)
            }
            _ => Err(Error::NotDir(resolved)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            _ => Err(Error::NotDir(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source)?;
        let source_path = Self::join(&source_parent, &source_name);
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent, destination_name) = self.resolve_parent(&destination)?;
        let destination_path = Self::join(&destination_parent, &destination_name);

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &destination_path)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&destination_path, &source_path) {
            return Err(Error::Cycle(destination_path));
        }

        let existing = self.entries.get(&destination_path).cloned();
        if existing.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(dest_entry) = &existing {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if self.has_children(&destination_path) {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| Self::is_within(path, &source_path))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if existing.is_some() {
            self.remove_subtree(&destination_path);
        }
        self.remove_subtree(&source_path);
        for (old_path, entry) in moving {
            let new_path = if old_path == source_path {
                destination_path.clone()
            } else {
                format!(
                    "{}{}",
                    destination_path,
                    &old_path[source_path.len()..]
                )
            };
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        if normalized == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&normalized)?;
        let target = Self::join(&parent, &name);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && self.has_children(&target) && !recursive {
            return Err(Error::NotEmpty(target));
        }
        self.remove_subtree(&target);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let normalized_link = self.normalize(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let (parent, name) = self.resolve_parent(&normalized_link)?;
        let link_path = if name.is_empty() {
            "/".to_string()
        } else {
            Self::join(&parent, &name)
        };
        if self.entries.contains_key(&link_path) {
            return Err(Error::Exists(link_path));
        }
        self.entries
            .insert(link_path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let normalized = self.normalize(path)?;
        let (parent, name) = if normalized == "/" {
            ("/".to_string(), String::new())
        } else {
            let (p, n) = Self::parent_and_name(&normalized);
            (self.resolve(&p, true, false)?, n)
        };
        let target = if name.is_empty() {
            "/".to_string()
        } else {
            Self::join(&parent, &name)
        };
        match self.entries.get(&target) {
            Some(Entry::Link(value)) => Ok(value.clone()),
            _ => Err(Error::InvalidTarget(0)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }
        let mut rows = Vec::new();
        let mut subtotal = BTreeMap::<String, u64>::new();

        for (entry_path, entry) in self.entries.iter() {
            if !Self::is_within(entry_path, &resolved) {
                continue;
            }
            let size = match entry {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            rows.push((entry_path.clone(), size));
            let mut current = entry_path.clone();
            loop {
                *subtotal.entry(current.clone()).or_default() += size;
                if current == resolved || current == "/" {
                    break;
                }
                current = Self::parent_and_name(&current).0;
            }
        }
        for row in &mut rows {
            if matches!(self.entries.get(&row.0), Some(Entry::Dir)) {
                row.1 = *subtotal.get(&row.0).unwrap_or(&0);
            }
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries.keys().any(|candidate| {
            candidate != path && Self::is_within(candidate, path)
        })
    }

    fn remove_subtree(&mut self, path: &str) {
        let doomed: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| Self::is_within(candidate, path))
            .cloned()
            .collect();
        for candidate in doomed {
            self.entries.remove(&candidate);
        }
    }
}
```
