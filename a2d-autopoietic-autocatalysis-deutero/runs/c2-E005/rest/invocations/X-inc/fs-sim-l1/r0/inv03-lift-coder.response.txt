```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn validate(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut components = Vec::new();
        let mut start = 1;

        for (i, component) in path[1..].split('/').enumerate() {
            let component_start = start;
            if component.is_empty() {
                let offset = if i == path[1..].split('/').count() - 1 {
                    bytes.len() - 1
                } else {
                    component_start
                };
                return Err(Error::InvalidPath(offset));
            }
            if component == "." || component == ".." {
                return Err(Error::InvalidPath(component_start));
            }
            components.push(component);
            start += component.len() + 1;
        }

        Ok(components)
    }

    fn canonical_prefix(parts: &[&str]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn lookup<'a>(&'a self, parts: &[&str]) -> Result<&'a Entry, Error> {
        let mut entry = &self.root;
        let mut prefix = Vec::new();

        for part in parts {
            match entry {
                Entry::Dir(children) => {
                    prefix.push(*part);
                    entry = children
                        .get(*part)
                        .ok_or_else(|| Error::Missing(Self::canonical_prefix(&prefix)))?;
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical_prefix(&prefix)));
                }
            }
        }

        Ok(entry)
    }

    fn lookup_parent_mut<'a>(
        &'a mut self,
        parts: &[&str],
    ) -> Result<(&'a mut BTreeMap<String, Entry>, String), Error> {
        let parent_parts = &parts[..parts.len().saturating_sub(1)];
        let parent_path = Self::canonical_prefix(parent_parts);
        let mut entry = &mut self.root;
        let mut prefix = Vec::new();

        for part in parent_parts {
            match entry {
                Entry::Dir(children) => {
                    prefix.push(*part);
                    entry = children
                        .get_mut(*part)
                        .ok_or_else(|| Error::Missing(Self::canonical_prefix(&prefix)))?;
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical_prefix(&prefix)));
                }
            }
        }

        match entry {
            Entry::Dir(children) => Ok((children, parent_path)),
            Entry::File(_) => Err(Error::NotDir(parent_path)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = Self::validate(path)?;
        if parts.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }

        let canonical = Self::canonical_prefix(&parts);
        let (parent, _) = self.lookup_parent_mut(&parts)?;
        if parent.contains_key(parts[parts.len() - 1]) {
            return Err(Error::Exists(canonical));
        }

        parent.insert(
            parts[parts.len() - 1].to_string(),
            Entry::Dir(BTreeMap::new()),
        );
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let parts = Self::validate(path)?;
        if parts.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }

        let canonical = Self::canonical_prefix(&parts);
        let (parent, _) = self.lookup_parent_mut(&parts)?;
        match parent.get_mut(parts[parts.len() - 1]) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(canonical)),
            Some(Entry::File(bytes)) => {
                bytes.clear();
                bytes.extend_from_slice(data);
                Ok(())
            }
            None => {
                parent.insert(
                    parts[parts.len() - 1].to_string(),
                    Entry::File(data.to_vec()),
                );
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let parts = Self::validate(path)?;
        let entry = self.lookup(&parts)?;
        match entry {
            Entry::File(bytes) => Ok(bytes.clone()),
            Entry::Dir(_) => Err(Error::IsDir(Self::canonical_prefix(&parts))),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let parts = Self::validate(path)?;
        let entry = self.lookup(&parts)?;
        match entry {
            Entry::File(_) => Err(Error::NotDir(Self::canonical_prefix(&parts))),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
