#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let complement = denominator - remainder;
    if remainder > complement || (remainder == complement && quotient % 2 == 1) {
        Some(quotient + 1)
    } else {
        Some(quotient)
    }
}

#[derive(Clone)]
struct CustomerState {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    first_start: bool,
    waited: u64,
    wait_total: u128,
    queue_entry: u64,
    paused: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    remaining: u64,
}

fn add_time(customer: u32, time: u64, value: u64, stage: OverflowStage) -> Result<u64, Error> {
    time.checked_add(value).ok_or(Error::TimeOverflow {
        customer,
        time,
        stage,
    })
}

fn waiting_cmp(a: usize, b: usize, states: &[CustomerState]) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    let sa = &states[a];
    let sb = &states[b];
    sa.customer
        .priority
        .cmp(&sb.customer.priority)
        .then_with(|| match (sa.paused, sb.paused) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (true, true) => sa
                .pause_time
                .cmp(&sb.pause_time)
                .then_with(|| sa.customer.id.cmp(&sb.customer.id)),
            (false, false) => sa
                .customer
                .arrival
                .cmp(&sb.customer.arrival)
                .then_with(|| sb.customer.id.cmp(&sa.customer.id)),
        })
}

fn take_best(waiting: &mut Vec<usize>, states: &[CustomerState]) -> Option<usize> {
    let best = waiting
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| waiting_cmp(**a, **b, states))
        .map(|(pos, _)| pos)?;
    Some(waiting.remove(best))
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index].iter().any(|prior| prior.id == c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow {
                index,
                id: c.id,
            })?),
            None => None,
        };
        ids.push(deadline);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<CustomerState> = customers
        .iter()
        .enumerate()
        .map(|(i, c)| CustomerState {
            customer: c.clone(),
            deadline: ids[i],
            remaining: c.service,
            first_start: false,
            waited: 0,
            wait_total: 0,
            queue_entry: c.arrival,
            paused: false,
            pause_time: 0,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut last_dispatch: Option<u32> = None;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut completion_order = Vec::new();

    if customers.is_empty() {
        return Ok(Report {
            events,
            stats: Stats {
                accepted,
                balked,
                reneged,
                completed,
                started,
                preemptions,
                elapsed: 0,
                busy_ticks,
                wait_ticks,
                turnaround_ticks,
                mean_wait_milli: 0,
                mean_turnaround_milli: 0,
                utilization_bps: 0,
                completion_order,
            },
        });
    }

    loop {
        if let Some(a) = active.as_ref() {
            if a.finish == now {
                let idx = a.index;
                let id = states[idx].customer.id;
                emit(&mut events, now, id, Kind::Finish);
                completion_order.push(id);
                turnaround_ticks += (now - states[idx].customer.arrival) as u128;
                completed += 1;
                active = None;
            }
        }

        let mut expiring: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expiring.sort_by(|&a, &b| {
            states[a]
                .deadline
                .cmp(&states[b].deadline)
                .then_with(|| states[a].customer.id.cmp(&states[b].customer.id))
        });
        for idx in expiring {
            if let Some(pos) = waiting.iter().position(|&i| i == idx) {
                waiting.remove(pos);
                let seg = now - states[idx].queue_entry;
                states[idx].waited += seg;
                states[idx].wait_total += seg as u128;
                wait_ticks += seg as u128;
                emit(
                    &mut events,
                    now,
                    states[idx].customer.id,
                    Kind::Renege {
                        waited: states[idx].waited,
                    },
                );
                reneged += 1;
            }
        }

        while next_arrival < order.len()
            && customers[order[next_arrival]].arrival == now
        {
            let idx = order[next_arrival];
            next_arrival += 1;
            emit(&mut events, now, states[idx].customer.id, Kind::Arrive);
            let limit = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= limit {
                emit(&mut events, now, states[idx].customer.id, Kind::Balk);
                balked += 1;
            } else {
                accepted += 1;
                states[idx].queue_entry = now;
                states[idx].remaining = states[idx].customer.service;
                states[idx].deadline = states[idx]
                    .customer
                    .patience
                    .map(|p| now + p);
                waiting.push(idx);
            }
        }

        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(&best) = waiting
                    .iter()
                    .min_by(|&&x, &&y| waiting_cmp(x, y, &states))
                {
                    if states[best].customer.priority < states[a.index].customer.priority {
                        let best_id = states[best].customer.id;
                        let idx = a.index;
                        let remaining = a.finish - now;
                        emit(
                            &mut events,
                            now,
                            states[idx].customer.id,
                            Kind::Preempt {
                                remaining,
                                by: best_id,
                            },
                        );
                        preemptions += 1;
                        let old = active.take().unwrap();
                        busy_ticks += (now - old_start_time(&old, now)) as u128;
                        states[idx].remaining = remaining;
                        states[idx].queue_entry = now;
                        states[idx].paused = true;
                        states[idx].pause_time = now;
                        if let Some(p) = states[idx].customer.patience {
                            let budget = p.saturating_sub(states[idx].waited);
                            states[idx].deadline =
                                Some(add_time(states[idx].customer.id, now, budget, OverflowStage::Patience)?);
                        } else {
                            states[idx].deadline = None;
                        }
                        waiting.push(idx);
                    }
                }
            }
        }

        if active.is_none() {
            loop {
                let Some(idx) = take_best(&mut waiting, &states) else {
                    break;
                };
                let id = states[idx].customer.id;
                let wait = now - states[idx].queue_entry;
                states[idx].wait_total += wait as u128;
                wait_ticks += wait as u128;
                states[idx].waited += wait;
                if let Some(p) = states[idx].customer.patience {
                    let remaining_budget = p.saturating_sub(states[idx].waited);
                    states[idx].deadline = Some(now.saturating_add(remaining_budget));
                }
                let remaining = states[idx].remaining;
                let switching = if last_dispatch.is_some_and(|prev| prev != id) {
                    config.switch_cost
                } else {
                    0
                };
                let work_at = add_time(id, now, switching, OverflowStage::WorkAt)?;
                let finish = add_time(id, work_at, remaining, OverflowStage::Finish)?;
                busy_ticks += (finish - now) as u128;
                let first = !states[idx].first_start;
                if first {
                    started += 1;
                    states[idx].first_start = true;
                    emit(
                        &mut events,
                        now,
                        id,
                        Kind::Start {
                            remaining,
                            work_at,
                            finish,
                        },
                    );
                } else {
                    emit(
                        &mut events,
                        now,
                        id,
                        Kind::Resume {
                            remaining,
                            work_at,
                            finish,
                        },
                    );
                }
                last_dispatch = Some(id);
                active = Some(Active {
                    index: idx,
                    work_at,
                    finish,
                    remaining,
                });
                if finish == now {
                    emit(&mut events, now, id, Kind::Finish);
                    completion_order.push(id);
                    turnaround_ticks += (now - states[idx].customer.arrival) as u128;
                    completed += 1;
                    active = None;
                    continue;
                }
                break;
            }
        }

        let mut expiring: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expiring.sort_by(|&a, &b| {
            states[a]
                .deadline
                .cmp(&states[b].deadline)
                .then_with(|| states[a].customer.id.cmp(&states[b].customer.id))
        });
        for idx in expiring {
            if let Some(pos) = waiting.iter().position(|&i| i == idx) {
                waiting.remove(pos);
                let seg = now - states[idx].queue_entry;
                states[idx].waited += seg;
                states[idx].wait_total += seg as u128;
                wait_ticks += seg as u128;
                emit(
                    &mut events,
                    now,
                    states[idx].customer.id,
                    Kind::Renege {
                        waited: states[idx].waited,
                    },
                );
                reneged += 1;
            }
        }

        if next_arrival >= order.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next = u64::MAX;
        if next_arrival < order.len() {
            next = next.min(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = active.as_ref() {
            next = next.min(a.finish);
            if a.work_at > now {
                next = next.min(a.work_at);
            }
        }
        for &idx in &waiting {
            if let Some(d) = states[idx].deadline {
                if d > now {
                    next = next.min(d);
                }
            }
        }
        if next == u64::MAX && now != u64::MAX {
            break;
        }
        if next <= now {
            break;
        }
        now = next;
    }

    let elapsed = events.last().map_or(0, |e| e.time);
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(busy_ticks * 10000, elapsed as u128).unwrap() as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}

fn old_start(_active: &Active, _now: u64) -> u64 {
    0
}
