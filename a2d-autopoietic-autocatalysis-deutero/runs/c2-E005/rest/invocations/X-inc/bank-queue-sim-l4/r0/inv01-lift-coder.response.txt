#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege {
        waited: u64,
    },
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge {
        capacity: usize,
    },
    TooManyCustomers {
        count: usize,
    },
    ZeroId {
        index: usize,
    },
    DuplicateId {
        index: usize,
        id: u32,
    },
    BadPriority {
        index: usize,
        value: u8,
    },
    DeadlineOverflow {
        index: usize,
        id: u32,
    },
    TimeOverflow {
        customer: u32,
        time: u64,
        stage: OverflowStage,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;

    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };

    Some(if round_up { quotient + 1 } else { quotient })
}

#[derive(Clone)]
struct CustomerState {
    customer: Customer,
    remaining: u64,
    has_started: bool,
    waited: u64,
    waiting_since: u64,
    budget_left: Option<u64>,
    deadline: Option<u64>,
    pause_time: Option<u64>,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    protected: bool,
}

fn waiting_cmp(a: &CustomerState, b: &CustomerState) -> std::cmp::Ordering {
    use std::cmp::Ordering;

    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(ap), Some(bp)) => ap.cmp(&bp).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn add_waited(state: &mut CustomerState, now: u64, wait_ticks: &mut u128) {
    let segment = now - state.waiting_since;
    state.waited += segment;
    *wait_ticks += segment as u128;
    if let Some(left) = state.budget_left {
        state.budget_left = Some(left - segment);
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    use std::cmp::Ordering;

    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index]
            .iter()
            .any(|previous| previous.id == customer.id)
        {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
        if let Some(patience) = customer.patience {
            if customer.arrival.checked_add(patience).is_none() {
                return Err(Error::DeadlineOverflow {
                    index,
                    id: customer.id,
                });
            }
        }
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<CustomerState> = customers
        .iter()
        .cloned()
        .map(|customer| CustomerState {
            remaining: customer.service,
            has_started: false,
            waited: 0,
            waiting_since: customer.arrival,
            budget_left: customer.patience,
            deadline: customer
                .patience
                .map(|p| customer.arrival.checked_add(p).unwrap()),
            pause_time: None,
            customer,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;

    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();
    let mut last_event_time = 0u64;
    let mut now = arrivals
        .first()
        .map(|&index| customers[index].arrival)
        .unwrap_or(0);

    while next_arrival < arrivals.len() || active.is_some() || !waiting.is_empty() {
        // Advance only to an event or a server boundary.
        let mut next_time = u64::MAX;
        if let Some(&index) = arrivals.get(next_arrival) {
            next_time = next_time.min(customers[index].arrival);
        }
        if let Some(server) = &active {
            next_time = next_time.min(server.finish);
            if server.protected && server.work_at > now {
                next_time = next_time.min(server.work_at);
            }
        }
        for &index in &waiting {
            if let Some(deadline) = states[index].deadline {
                next_time = next_time.min(deadline);
            }
        }
        now = next_time;

        // Phase A: finish before any other work at this tick.
        if active.as_ref().is_some_and(|server| server.finish == now) {
            let server = active.take().unwrap();
            let id = states[server.index].customer.id;
            events.push(Event {
                time: now,
                customer: id,
                kind: Kind::Finish,
            });
            last_event_time = now;
            completed += 1;
            turnaround_ticks += (now - states[server.index].customer.arrival) as u128;
            completion_order.push(id);
        }

        // Phase B: expire customers who were already waiting.
        let mut expired: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&index| states[index].deadline.is_some_and(|deadline| deadline <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            states[a]
                .deadline
                .cmp(&states[b].deadline)
                .then_with(|| states[a].customer.id.cmp(&states[b].customer.id))
        });
        for index in expired {
            waiting.retain(|&candidate| candidate != index);
            add_waited(&mut states[index], now, &mut wait_ticks);
            events.push(Event {
                time: now,
                customer: states[index].customer.id,
                kind: Kind::Renege {
                    waited: states[index].waited,
                },
            });
            last_event_time = now;
            reneged += 1;
        }

        // Phase C: process the complete arrival cohort without dispatch.
        while next_arrival < arrivals.len()
            && customers[arrivals[next_arrival]].arrival == now
        {
            let index = arrivals[next_arrival];
            next_arrival += 1;
            let id = states[index].customer.id;
            events.push(Event {
                time: now,
                customer: id,
                kind: Kind::Arrive,
            });
            last_event_time = now;

            let idle_slot = usize::from(active.is_none());
            if waiting.len() < config.waiting_capacity + idle_slot {
                accepted += 1;
                waiting.push(index);
            } else {
                balked += 1;
                events.push(Event {
                    time: now,
                    customer: id,
                    kind: Kind::Balk,
                });
                last_event_time = now;
            }
        }

        // A boundary ends protection before preemption is considered.
        if let Some(server) = active.as_mut() {
            if now >= server.work_at {
                server.protected = false;
            }
        }

        // Phase D: preempt once if a strictly better waiter exists.
        let should_preempt = if let Some(server) = &active {
            if !server.protected && !waiting.is_empty() {
                let best = *waiting
                    .iter()
                    .min_by(|&&a, &&b| waiting_cmp(&states[a], &states[b]))
                    .unwrap();
                states[best].customer.priority < states[server.index].customer.priority
            } else {
                false
            }
        } else {
            false
        };

        if should_preempt {
            let best = *waiting
                .iter()
                .min_by(|&&a, &&b| waiting_cmp(&states[a], &states[b]))
                .unwrap();
            let server = active.take().unwrap();
            let remaining = server.finish - now;
            let paused_id = states[server.index].customer.id;
            let by = states[best].customer.id;

            events.push(Event {
                time: now,
                customer: paused_id,
                kind: Kind::Preempt { remaining, by },
            });
            last_event_time = now;
            preemptions += 1;
            busy_ticks += (now - server_start_time_placeholder()) as u128;

            let new_deadline = match states[server.index].budget_left {
                Some(budget) => Some(now.checked_add(budget).ok_or(Error::TimeOverflow {
                    customer: paused_id,
                    time: now,
                    stage: OverflowStage::Patience,
                })?),
                None => None,
            };
            states[server.index].remaining = remaining;
            states[server.index].waiting_since = now;
            states[server.index].deadline = new_deadline;
            states[server.index].pause_time = Some(now);
            waiting.push(server.index);
        }

        // Phase E: dispatch as long as zero-duration jobs finish inline.
        loop {
            if active.is_some() || waiting.is_empty() {
                break;
            }

            let selected_pos = waiting
                .iter()
                .enumerate()
                .min_by(|(_, &a), (_, &b)| waiting_cmp(&states[a], &states[b]))
                .map(|(position, _)| position)
                .unwrap();
            let index = waiting.remove(selected_pos);
            add_waited(&mut states[index], now, &mut wait_ticks);

            let id = states[index].customer.id;
            let switch = if last_dispatched.is_some_and(|previous| previous != id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at
                .checked_add(states[index].remaining)
                .ok_or(Error::TimeOverflow {
                    customer: id,
                    time: now,
                    stage: OverflowStage::Finish,
                })?;

            let remaining = states[index].remaining;
            let is_resume = states[index].has_started;
            if is_resume {
                events.push(Event {
                    time: now,
                    customer: id,
                    kind: Kind::Resume {
                        remaining,
                        work_at,
                        finish,
                    },
                });
            } else {
                states[index].has_started = true;
                started += 1;
                events.push(Event {
                    time: now,
                    customer: id,
                    kind: Kind::Start {
                        remaining,
                        work_at,
                        finish,
                    },
                });
            }
            last_event_time = now;
            last_dispatched = Some(id);

            // Record the interval's start for busy-time accounting.
            active = Some(Active {
                index,
                work_at,
                finish,
                protected: work_at > now,
            });
            ACTIVE_STARTS.with(|starts| starts.borrow_mut().push((id, now)));

            if finish == now {
                active = None;
                events.push(Event {
                    time: now,
                    customer: id,
                    kind: Kind::Finish,
                });
                last_event_time = now;
                completed += 1;
                turnaround_ticks += (now - states[index].customer.arrival) as u128;
                completion_order.push(id);
                continue;
            }
            break;
        }

        // Phase F: remove newly queued zero-budget/deadline waiters.
        let mut expired: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&index| states[index].deadline.is_some_and(|deadline| deadline <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            states[a]
                .deadline
                .cmp(&states[b].deadline)
                .then_with(|| states[a].customer.id.cmp(&states[b].customer.id))
        });
        for index in expired {
            waiting.retain(|&candidate| candidate != index);
            add_waited(&mut states[index], now, &mut wait_ticks);
            events.push(Event {
                time: now,
                customer: states[index].customer.id,
                kind: Kind::Renege {
                    waited: states[index].waited,
                },
            });
            last_event_time = now;
            reneged += 1;
        }

        // Account occupied time when the interval ends by preemption or finish.
        // This is deferred until the next loop observes the interval end.
        if let Some(server) = &active {
            if server.finish == now {
                // Phase A already removed it; this branch is only defensive.
            }
        }
    }

    // Reconstruct occupied intervals from dispatch and terminal/preemption events.
    // The per-dispatch events contain the exact work and finish boundaries; the
    // event trace also identifies any preemptions.
    busy_ticks = calculate_busy_ticks(&events);

    let elapsed = if events.is_empty() { 0 } else { last_event_time };
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(busy_ticks * 10_000, elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}

// The active interval start is retained per dispatch for exact busy accounting.
thread_local! {
    static ACTIVE_STARTS: std::cell::RefCell<Vec<(u32, u64)>> =
        const { std::cell::RefCell::new(Vec::new()) };
}

fn server_start_time_placeholder() -> u64 {
    0
}

fn calculate_busy_ticks(events: &[Event]) -> u128 {
    let mut total = 0u128;
    let mut active_start: Option<u64> = None;

    for event in events {
        match &event.kind {
            Kind::Start { .. } | Kind::Resume { .. } => active_start = Some(event.time),
            Kind::Preempt { .. } | Kind::Finish => {
                if let Some(start) = active_start.take() {
                    total += (event.time - start) as u128;
                }
            }
            _ => {}
        }
    }
    total
}
