#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<char>,
    variables: Vec<(String, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn validate_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;
    let operator = match bytes.get(content_start).copied() {
        Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') => {
            let op = bytes[content_start] as char;
            content_start += 1;
            Some(op)
        }
        Some(b'=' | b'!' | b'@' | b'|' | b'$' | b'^') => {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        _ => None,
    };

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut item_start = content_start;

    loop {
        let item_end = bytes[item_start..end]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| item_start + offset)
            .unwrap_or(end);

        if item_start == item_end {
            return Err(Error::EmptyVariable { pos: item_start });
        }

        let item = &template[item_start..item_end];
        let item_bytes = item.as_bytes();
        if !is_name_start(item_bytes[0]) {
            return Err(Error::InvalidName { pos: item_start });
        }

        for (offset, &byte) in item_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName {
                    pos: item_start + offset,
                });
            }
        }

        if seen.contains(&item) {
            return Err(Error::DuplicateVariable { pos: item_start });
        }
        seen.push(item);
        variables.push((item.to_owned(), item_start));

        if item_end == end {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(Expression {
        start,
        end,
        operator,
        variables,
    })
}

fn parse(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let start = pos;
                let Some(relative_end) = bytes[pos + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = pos + 1 + relative_end;

                if let Some(relative_open) = bytes[pos + 1..end].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: pos + 1 + relative_open,
                    });
                }

                expressions.push(validate_expression(template, start, end)?);
                pos = end + 1;
            }
            _ => pos += 1,
        }
    }

    Ok(expressions)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut pos = 0;

    while pos < bytes.len() {
        let byte = bytes[pos];
        if byte == b'%' && pos + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[pos + 1]), hex_value(bytes[pos + 2])) {
                output.push('%');
                output.push(HEX[high as usize] as char);
                output.push(HEX[low as usize] as char);
                pos += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        pos += 1;
    }

    output
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|&(binding, value)| (binding == name).then_some(value))
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let allow_reserved = matches!(operator, Some('+' | '#'));
    let named = matches!(operator, Some(';' | '?' | '&'));
    let separator = match operator {
        Some('.') => '.',
        Some('/') => '/',
        Some(';') | Some('?') | Some('&') => '&',
        _ => ',',
    };

    let mut fields = Vec::new();
    for (name, _) in &expression.variables {
        if let Some(value) = lookup(vars, name) {
            let encoded = encode(value, allow_reserved);
            let field = if named {
                if operator == Some(';') && encoded.is_empty() {
                    name.clone()
                } else {
                    format!("{name}={encoded}")
                }
            } else {
                encoded
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let mut output = String::new();
    match operator {
        Some('#' | '.') => output.push(operator.unwrap()),
        Some('/') => output.push('/'),
        Some(';') => output.push(';'),
        Some('?') => output.push('?'),
        Some('&') => output.push('&'),
        _ => {}
    }
    for (index, field) in fields.iter().enumerate() {
        if index > 0 {
            output.push(separator);
        }
        output.push_str(field);
    }
    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse(template)?;
    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);
        output.push_str(&expand_expression(&expression, vars));
        literal_start = expression.end + 1;
    }
    output.push_str(&template[literal_start..]);

    Ok(output)
}
