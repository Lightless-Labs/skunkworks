#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    names: Vec<(usize, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let content_start = open + 1;
    let mut start = content_start;
    let mut operator = Operator::Simple;

    if start < close {
        operator = match bytes[start] {
            b'+' => Operator::Reserved,
            b'#' => Operator::Fragment,
            b'.' => Operator::Dot,
            b'/' => Operator::Slash,
            b';' => Operator::Semicolon,
            b'?' => Operator::Question,
            b'&' => Operator::Ampersand,
            b'=' | b'!' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator { pos: start });
            }
            _ => Operator::Simple,
        };

        if !matches!(operator, Operator::Simple) {
            start += 1;
        }
    }

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut used: Vec<&str> = Vec::new();
    let mut spec_start = start;

    loop {
        let spec_end = (spec_start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        if !is_name_start(bytes[spec_start]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        let mut i = spec_start + 1;
        while i < spec_end {
            if !is_name_continue(bytes[i]) {
                return Err(Error::InvalidName { pos: i });
            }
            i += 1;
        }

        let name = &template[spec_start..spec_end];
        if used.contains(&name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        used.push(name);
        names.push((spec_start, spec_end));

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: content_start,
        end: close,
        operator,
        names,
    })
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> u8 {
    if value < 10 {
        b'0' + value
    } else {
        b'A' + (value - 10)
    }
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(hex_digit(high) as char);
                output.push(hex_digit(low) as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4) as char);
            output.push(hex_digit(byte & 0x0f) as char);
        }
        i += 1;
    }

    output
}

fn expand_expression(template: &str, expression: &Expression, vars: &[(&str, &str)]) -> String {
    let allow_reserved = matches!(expression.operator, Operator::Reserved | Operator::Fragment);
    let mut fields: Vec<String> = Vec::new();

    for &(start, end) in &expression.names {
        let name = &template[start..end];
        if let Some(value) = lookup(name, vars) {
            let encoded = encode(value, allow_reserved);
            let field = match expression.operator {
                Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                    if matches!(expression.operator, Operator::Semicolon) && encoded.is_empty() {
                        name.to_string()
                    } else {
                        format!("{name}={encoded}")
                    }
                }
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Reserved => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Question => ("?", "&"),
        Operator::Ampersand => ("&", "&"),
    };

    let mut output = String::from(prefix);
    output.push_str(&fields.join(separator));
    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match (i + 1..bytes.len()).find(|&j| bytes[j] == b'}') {
                    Some(close) => close,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(inner) = (open + 1..close).find(|&j| bytes[j] == b'{') {
                    return Err(Error::NestedOpen { pos: inner });
                }

                expressions.push((open, close, parse_expression(template, open, close)?));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;

    for (open, close, expression) in &expressions {
        output.push_str(&template[cursor..*open]);
        output.push_str(&expand_expression(template, expression, vars));
        cursor = close + 1;
    }
    output.push_str(&template[cursor..]);

    Ok(output)
}
