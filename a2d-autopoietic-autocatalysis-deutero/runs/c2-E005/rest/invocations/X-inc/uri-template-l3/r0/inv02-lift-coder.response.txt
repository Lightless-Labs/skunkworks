#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<u8>,
    variables: Vec<Variable>,
}

#[derive(Debug, Clone)]
struct Variable {
    name: String,
    start: usize,
}

fn is_supported_operator(byte: u8) -> bool {
    matches!(byte, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn validate_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;

    if let Some(relative) = bytes[content_start..end]
        .iter()
        .position(|&byte| byte == b'{')
    {
        return Err(Error::NestedOpen {
            pos: content_start + relative,
        });
    }

    let operator = if content_start < end && is_supported_operator(bytes[content_start]) {
        let operator = bytes[content_start];
        content_start += 1;
        Some(operator)
    } else if content_start < end && matches!(bytes[content_start], b'!' | b'$' | b'=' | b'%' | b'^' | b'|' | b'\\' | b':' | b'*') {
        return Err(Error::UnsupportedOperator {
            pos: content_start,
        });
    } else {
        None
    };

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut variables = Vec::new();
    let mut seen = std::collections::HashSet::<String>::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..end]
            .iter()
            .position(|&byte| byte == b',')
            .map(|relative| spec_start + relative)
            .unwrap_or(end);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let spec = &bytes[spec_start..spec_end];
        let mut name_len = 0;

        if !is_name_start(spec[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        name_len += 1;
        while name_len < spec.len() && is_name_continue(spec[name_len]) {
            name_len += 1;
        }

        if name_len != spec.len() {
            return Err(Error::InvalidName {
                pos: spec_start + name_len,
            });
        }

        let name = template[spec_start..spec_end].to_owned();
        if !seen.insert(name.clone()) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }

        variables.push(Variable {
            name,
            start: spec_start,
        });

        if spec_end == end {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start,
        end,
        operator,
        variables,
    })
}

fn validate_template(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let start = index;
                let Some(relative_end) = bytes[index + 1..].iter().position(|&byte| byte == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = index + 1 + relative_end;
                let expression = validate_expression(template, start, end)?;
                expressions.push(expression);
                index = end + 1;
            }
            _ => index += 1,
        }
    }

    Ok(expressions)
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn upper_hex(byte: u8) -> u8 {
    byte.to_ascii_uppercase()
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut out = String::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() && is_hex(bytes[index + 1]) && is_hex(bytes[index + 2]) {
            out.push('%');
            out.push(upper_hex(bytes[index + 1]) as char);
            out.push(upper_hex(bytes[index + 2]) as char);
            index += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            out.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0F) as usize] as char);
        }
        index += 1;
    }

    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let allow_reserved = matches!(operator, Some(b'+') | Some(b'#'));
    let mut fields = Vec::<String>::new();

    for variable in &expression.variables {
        let Some(value) = lookup(&variable.name, vars) else {
            continue;
        };

        let encoded = encode(value, allow_reserved);
        match operator {
            Some(b';') => {
                if encoded.is_empty() {
                    fields.push(variable.name.clone());
                } else {
                    fields.push(format!("{}={}", variable.name, encoded));
                }
            }
            Some(b'?') | Some(b'&') => {
                fields.push(format!("{}={}", variable.name, encoded));
            }
            _ => fields.push(encoded),
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    match operator {
        Some(b'.') => format!(".{}", fields.join(".")),
        Some(b'/') => format!("/{}", fields.join("/")),
        Some(b';') => format!(";{}", fields.join(";")),
        Some(b'?') => format!("?{}", fields.join("&")),
        Some(b'&') => format!("&{}", fields.join("&")),
        Some(b'#') => format!("#{}", fields.join(",")),
        _ => fields.join(","),
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut result = String::with_capacity(template.len());
    let mut cursor = 0;

    for expression in expressions {
        result.push_str(&template[cursor..expression.start]);
        result.push_str(&expand_expression(&expression, vars));
        cursor = expression.end + 1;
    }

    result.push_str(&template[cursor..]);
    Ok(result)
}
