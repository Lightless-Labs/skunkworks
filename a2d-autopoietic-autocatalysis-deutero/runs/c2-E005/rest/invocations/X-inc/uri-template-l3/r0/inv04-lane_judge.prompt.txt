pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression<'a> {
    operator: Option<char>,
    specs: Vec<Specifier<'a>>,
}

#[derive(Debug, Clone)]
struct Specifier<'a> {
    name: &'a str,
    start: usize,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let content_start = open + 1;
    let content = &template[content_start..close];

    let operator = content
        .as_bytes()
        .first()
        .copied()
        .filter(|b| matches!(b, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'))
        .map(char::from);

    let body_start = content_start + usize::from(operator.is_some());
    let body = &template[body_start..close];

    if body.is_empty() {
        return Err(Error::EmptyExpression { pos: close });
    }

    if let Some(op) = operator {
        if !matches!(op, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
    }

    let mut specs = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut start = body_start;

    for part in body.split(',') {
        if part.is_empty() {
            return Err(Error::EmptyVariable { pos: start });
        }

        let bytes = part.as_bytes();
        let mut invalid = None;
        if !is_name_start(bytes[0]) {
            invalid = Some(start);
        } else {
            for (i, byte) in bytes.iter().copied().enumerate().skip(1) {
                if !is_name_continue(byte) {
                    invalid = Some(start + i);
                    break;
                }
            }
        }

        if let Some(pos) = invalid {
            return Err(Error::InvalidName { pos });
        }

        if seen.contains(&part) {
            return Err(Error::DuplicateVariable { pos: start });
        }
        seen.push(part);
        specs.push(Specifier { name: part, start });
        start += part.len() + 1;
    }

    Ok(Expression { operator, specs })
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            let h1 = bytes[i + 1];
            let h2 = bytes[i + 2];
            if h1.is_ascii_hexdigit() && h2.is_ascii_hexdigit() {
                out.push('%');
                out.push((h1 as char).to_ascii_uppercase());
                out.push((h2 as char).to_ascii_uppercase());
                i += 3;
                continue;
            }
        }

        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                    | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || reserved_byte {
            out.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator.unwrap_or('\0');
    let reserved = matches!(operator, '+' | '#');
    let mut fields = Vec::new();

    for spec in &expression.specs {
        if let Some(value) = lookup(spec.name, vars) {
            let encoded = encode(value, reserved);
            let field = match operator {
                ';' if encoded.is_empty() => spec.name.to_owned(),
                ';' | '?' | '&' => format!("{}={}", spec.name, encoded),
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match operator {
        '#' => ("#", ","),
        '.' => (".", "."),
        '/' => ("/", "/"),
        ';' => (";", ";"),
        '?' => ("?", "&"),
        '&' => ("&", "&"),
        _ => ("", ","),
    };

    let mut out = String::from(prefix);
    out.push_str(&fields.join(separator));
    out
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                let expression = parse_expression(template, open, close)?;
                expressions.push((open, close, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut cursor = 0;
    for (open, close, expression) in &expressions {
        out.push_str(&template[cursor..*open]);
        out.push_str(&expand_expression(expression, vars));
        cursor = close + 1;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}

pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    names: Vec<(usize, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let content_start = open + 1;
    let mut start = content_start;
    let mut operator = Operator::Simple;

    if start < close {
        operator = match bytes[start] {
            b'+' => Operator::Reserved,
            b'#' => Operator::Fragment,
            b'.' => Operator::Dot,
            b'/' => Operator::Slash,
            b';' => Operator::Semicolon,
            b'?' => Operator::Question,
            b'&' => Operator::Ampersand,
            b'=' | b'!' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator { pos: start });
            }
            _ => Operator::Simple,
        };

        if !matches!(operator, Operator::Simple) {
            start += 1;
        }
    }

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut used: Vec<&str> = Vec::new();
    let mut spec_start = start;

    loop {
        let spec_end = (spec_start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        if !is_name_start(bytes[spec_start]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        let mut i = spec_start + 1;
        while i < spec_end {
            if !is_name_continue(bytes[i]) {
                return Err(Error::InvalidName { pos: i });
            }
            i += 1;
        }

        let name = &template[spec_start..spec_end];
        if used.contains(&name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        used.push(name);
        names.push((spec_start, spec_end));

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: content_start,
        end: close,
        operator,
        names,
    })
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> u8 {
    if value < 10 {
        b'0' + value
    } else {
        b'A' + (value - 10)
    }
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(hex_digit(high) as char);
                output.push(hex_digit(low) as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4) as char);
            output.push(hex_digit(byte & 0x0f) as char);
        }
        i += 1;
    }

    output
}

fn expand_expression(template: &str, expression: &Expression, vars: &[(&str, &str)]) -> String {
    let allow_reserved = matches!(expression.operator, Operator::Reserved | Operator::Fragment);
    let mut fields: Vec<String> = Vec::new();

    for &(start, end) in &expression.names {
        let name = &template[start..end];
        if let Some(value) = lookup(name, vars) {
            let encoded = encode(value, allow_reserved);
            let field = match expression.operator {
                Operator::Semicolon | Operator::Question | Operator::Ampersand => {
                    if matches!(expression.operator, Operator::Semicolon) && encoded.is_empty() {
                        name.to_string()
                    } else {
                        format!("{name}={encoded}")
                    }
                }
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Reserved => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Question => ("?", "&"),
        Operator::Ampersand => ("&", "&"),
    };

    let mut output = String::from(prefix);
    output.push_str(&fields.join(separator));
    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match (i + 1..bytes.len()).find(|&j| bytes[j] == b'}') {
                    Some(close) => close,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(inner) = (open + 1..close).find(|&j| bytes[j] == b'{') {
                    return Err(Error::NestedOpen { pos: inner });
                }

                expressions.push((open, close, parse_expression(template, open, close)?));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;

    for (open, close, expression) in &expressions {
        output.push_str(&template[cursor..*open]);
        output.push_str(&expand_expression(template, expression, vars));
        cursor = close + 1;
    }
    output.push_str(&template[cursor..]);

    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<char>,
    variables: Vec<(String, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn validate_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;
    let operator = match bytes.get(content_start).copied() {
        Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') => {
            let op = bytes[content_start] as char;
            content_start += 1;
            Some(op)
        }
        Some(b'=' | b'!' | b'@' | b'|' | b'$' | b'^') => {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        _ => None,
    };

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut item_start = content_start;

    loop {
        let item_end = bytes[item_start..end]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| item_start + offset)
            .unwrap_or(end);

        if item_start == item_end {
            return Err(Error::EmptyVariable { pos: item_start });
        }

        let item = &template[item_start..item_end];
        let item_bytes = item.as_bytes();
        if !is_name_start(item_bytes[0]) {
            return Err(Error::InvalidName { pos: item_start });
        }

        for (offset, &byte) in item_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName {
                    pos: item_start + offset,
                });
            }
        }

        if seen.contains(&item) {
            return Err(Error::DuplicateVariable { pos: item_start });
        }
        seen.push(item);
        variables.push((item.to_owned(), item_start));

        if item_end == end {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(Expression {
        start,
        end,
        operator,
        variables,
    })
}

fn parse(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let start = pos;
                let Some(relative_end) = bytes[pos + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = pos + 1 + relative_end;

                if let Some(relative_open) = bytes[pos + 1..end].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: pos + 1 + relative_open,
                    });
                }

                expressions.push(validate_expression(template, start, end)?);
                pos = end + 1;
            }
            _ => pos += 1,
        }
    }

    Ok(expressions)
}

    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut pos = 0;

    while pos < bytes.len() {
        let byte = bytes[pos];
        if byte == b'%' && pos + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[pos + 1]), hex_value(bytes[pos + 2])) {
                output.push('%');
                output.push(HEX[high as usize] as char);
                output.push(HEX[low as usize] as char);
                pos += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        pos += 1;
    }

    output
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|&(binding, value)| (binding == name).then_some(value))
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let allow_reserved = matches!(operator, Some('+' | '#'));
    let named = matches!(operator, Some(';' | '?' | '&'));
    let separator = match operator {
        Some('.') => '.',
        Some('/') => '/',
        Some(';') | Some('?') | Some('&') => '&',
        _ => ',',
    };

    let mut fields = Vec::new();
    for (name, _) in &expression.variables {
        if let Some(value) = lookup(vars, name) {
            let encoded = encode(value, allow_reserved);
            let field = if named {
                if operator == Some(';') && encoded.is_empty() {
                    name.clone()
                } else {
                    format!("{name}={encoded}")
                }
            } else {
                encoded
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let mut output = String::new();
    match operator {
        Some('#' | '.') => output.push(operator.unwrap()),
        Some('/') => output.push('/'),
        Some(';') => output.push(';'),
        Some('?') => output.push('?'),
        Some('&') => output.push('&'),
        _ => {}
    }
    for (index, field) in fields.iter().enumerate() {
        if index > 0 {
            output.push(separator);
        }
        output.push_str(field);
    }
    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse(template)?;
    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);
        output.push_str(&expand_expression(&expression, vars));
        literal_start = expression.end + 1;
    }
    output.push_str(&template[literal_start..]);

    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<u8>,
    variables: Vec<Variable>,
}

#[derive(Debug, Clone)]
struct Variable {
    name: String,
    start: usize,
}

fn is_supported_operator(byte: u8) -> bool {
    matches!(byte, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn validate_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;

    if let Some(relative) = bytes[content_start..end]
        .iter()
        .position(|&byte| byte == b'{')
    {
        return Err(Error::NestedOpen {
            pos: content_start + relative,
        });
    }

    let operator = if content_start < end && is_supported_operator(bytes[content_start]) {
        let operator = bytes[content_start];
        content_start += 1;
        Some(operator)
    } else if content_start < end && matches!(bytes[content_start], b'!' | b'$' | b'=' | b'%' | b'^' | b'|' | b'\\' | b':' | b'*') {
        return Err(Error::UnsupportedOperator {
            pos: content_start,
        });
    } else {
        None
    };

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut variables = Vec::new();
    let mut seen = std::collections::HashSet::<String>::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..end]
            .iter()
            .position(|&byte| byte == b',')
            .map(|relative| spec_start + relative)
            .unwrap_or(end);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let spec = &bytes[spec_start..spec_end];
        let mut name_len = 0;

        if !is_name_start(spec[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        name_len += 1;
        while name_len < spec.len() && is_name_continue(spec[name_len]) {
            name_len += 1;
        }

        if name_len != spec.len() {
            return Err(Error::InvalidName {
                pos: spec_start + name_len,
            });
        }

        let name = template[spec_start..spec_end].to_owned();
        if !seen.insert(name.clone()) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }

        variables.push(Variable {
            name,
            start: spec_start,
        });

        if spec_end == end {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start,
        end,
        operator,
        variables,
    })
}

fn validate_template(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let start = index;
                let Some(relative_end) = bytes[index + 1..].iter().position(|&byte| byte == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = index + 1 + relative_end;
                let expression = validate_expression(template, start, end)?;
                expressions.push(expression);
                index = end + 1;
            }
            _ => index += 1,
        }
    }

    Ok(expressions)
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn upper_hex(byte: u8) -> u8 {
    byte.to_ascii_uppercase()
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut out = String::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() && is_hex(bytes[index + 1]) && is_hex(bytes[index + 2]) {
            out.push('%');
            out.push(upper_hex(bytes[index + 1]) as char);
            out.push(upper_hex(bytes[index + 2]) as char);
            index += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            out.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0F) as usize] as char);
        }
        index += 1;
    }

    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let allow_reserved = matches!(operator, Some(b'+') | Some(b'#'));
    let mut fields = Vec::<String>::new();

    for variable in &expression.variables {
        let Some(value) = lookup(&variable.name, vars) else {
            continue;
        };

        let encoded = encode(value, allow_reserved);
        match operator {
            Some(b';') => {
                if encoded.is_empty() {
                    fields.push(variable.name.clone());
                } else {
                    fields.push(format!("{}={}", variable.name, encoded));
                }
            }
            Some(b'?') | Some(b'&') => {
                fields.push(format!("{}={}", variable.name, encoded));
            }
            _ => fields.push(encoded),
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    match operator {
        Some(b'.') => format!(".{}", fields.join(".")),
        Some(b'/') => format!("/{}", fields.join("/")),
        Some(b';') => format!(";{}", fields.join(";")),
        Some(b'?') => format!("?{}", fields.join("&")),
        Some(b'&') => format!("&{}", fields.join("&")),
        Some(b'#') => format!("#{}", fields.join(",")),
        _ => fields.join(","),
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut result = String::with_capacity(template.len());
    let mut cursor = 0;

    for expression in expressions {
        result.push_str(&template[cursor..expression.start]);
        result.push_str(&expand_expression(&expression, vars));
        cursor = expression.end + 1;
    }

    result.push_str(&template[cursor..]);
    Ok(result)
}
