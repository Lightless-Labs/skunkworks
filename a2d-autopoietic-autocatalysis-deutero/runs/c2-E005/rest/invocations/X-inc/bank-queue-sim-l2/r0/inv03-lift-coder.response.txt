```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    busy: bool,
    idle_since: u64,
    finish_at: Option<u64>,
    customer: Option<u32>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|s| s.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index].iter().any(|c| c.id == customer.id) {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy: false,
            finish_at: None,
            customer: None,
        })
        .collect();

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| a.arrival.cmp(&b.arrival).then_with(|| b.id.cmp(&a.id)));

    let mut waiting: Vec<usize> = Vec::new();
    let mut next_arrival = 0usize;
    let mut events = Vec::new();
    let mut now = 0u64;

    loop {
        let mut finished: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (s.busy && s.finish_at == Some(now)).then_some(i))
            .collect();
        finished.sort_by_key(|&i| states[i].server.id);

        for i in finished {
            let state = &mut states[i];
            let customer = state.customer.take().unwrap();
            events.push(Event::Finish {
                time: now,
                customer,
                server: state.server.id,
            });
            state.busy = false;
            state.finish_at = None;
            state.idle_since = now;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (!s.opened && s.server.opens == now).then_some(i))
            .collect();
        opening.sort_by(|&a, &b| states[b].server.id.cmp(&states[a].server.id));

        for i in opening {
            let state = &mut states[i];
            state.opened = true;
            state.idle_since = now;
            events.push(Event::Open {
                time: now,
                server: state.server.id,
            });
        }

        let cohort_start = next_arrival;
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            next_arrival += 1;
        }
        for customer in &arrivals[cohort_start..next_arrival] {
            events.push(Event::Arrive {
                time: now,
                customer: customer.id,
            });
            waiting.push(cohort_start + (customer as *const Customer as usize
                - arrivals[cohort_start..next_arrival].as_ptr() as usize)
                / std::mem::size_of::<Customer>());
        }

        loop {
            let Some(&customer_index) = waiting.first() else {
                break;
            };

            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, s)| s.opened && !s.busy)
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };

            let customer = &arrivals[customer_index];
            let server_id = states[server_index].server.id;
            let first = now.checked_add(customer.service).ok_or(Error::TimeOverflow {
                customer: customer.id,
                server: server_id,
                time: now,
            })?;
            let finish = first.checked_add(states[server_index].server.overhead).ok_or(
                Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time: now,
                },
            )?;

            waiting.remove(0);
            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = now;
            } else {
                states[server_index].busy = true;
                states[server_index].finish_at = Some(finish);
                states[server_index].customer = Some(customer.id);
            }
        }

        if waiting.is_empty() && next_arrival == arrivals.len()
            && states.iter().all(|s| s.opened)
            && states.iter().all(|s| !s.busy)
        {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now {
                next_time = Some(next_time.map_or(time, |current| current.min(time)));
            }
        };

        if next_arrival < arrivals.len() {
            consider(arrivals[next_arrival].arrival);
        }
        for state in &states {
            if !state.opened {
                consider(state.server.opens);
            }
            if let Some(finish) = state.finish_at {
                consider(finish);
            }
        }

        if let Some(time) = next_time {
            now = time;
        } else {
            break;
        }
    }

    Ok(events)
}
```
