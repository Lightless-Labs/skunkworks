#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName { site: String, name: String },
    DuplicatePackage { package: String },
    DuplicateVersion { package: String, version: Version },
    DuplicateFeature { package: String, version: Version, feature: String },
    DuplicateLock { package: String },
    Missing { package: String, evidence: Vec<Evidence> },
    NoMatch { package: String, requirements: Vec<Constraint>, evidence: Vec<Evidence> },
    Conflict { package: String, selected: Version, requirements: Vec<Constraint>, evidence: Vec<Evidence> },
    MissingFeature { package: String, selected: Version, feature: String, path: Vec<String> },
    Limit { attempts: usize },
}

fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort_by_key(|c| constraint_key(*c));
    out.dedup();
    out
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b as char, '.' | '_' | '-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(ev: &[Evidence]) -> Vec<Evidence> {
    let mut out = ev.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn evidence_lines(ev: &[Evidence]) -> String {
    let mut out = String::new();
    for e in evidence_sorted(ev) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => {
            format!("duplicate-version|package={}|version={}", encode(package), version_text(*version))
        }
        ResolveError::DuplicateFeature { package, version, feature } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            format!("missing|package={}{}", encode(package), evidence_lines(evidence))
        }
        ResolveError::NoMatch { package, requirements, evidence } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict { package, selected, requirements, evidence } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature { package, selected, feature, path } => format!(
            "missing-feature|package={}|selected={}|feature={}{}",
            encode(package),
            version_text(*selected),
            encode(feature),
            format!("\n  F:{} via {}", encode(feature), path.join(" -> "))
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn validate_names(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName { site: format!("root[{i}]"), name: root.package.clone() });
        }
        for (f, name) in root.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName { site: format!("root[{i}].feature[{f}]"), name: name.clone() });
            }
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName { site: format!("catalog[{i}]"), name: package.name.clone() });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{i}].release[{j}].dep[{k}]");
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName { site: base, name: dep.package.clone() });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName { site: format!("{base}.gate"), name: gate.clone() });
                    }
                }
                for (f, name) in dep.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName { site: format!("{base}.feature[{f}]"), name: name.clone() });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let base = format!("catalog[{i}].release[{j}].feature[{f}]");
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName { site: base.clone(), name: feature.name.clone() });
                }
                for (e, name) in feature.enables.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName { site: format!("{base}.enable[{e}]"), name: name.clone() });
                    }
                }
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName { site: format!("lock[{i}]"), name: lock.package.clone() });
        }
    }
    Ok(())
}

fn target_matches(platform: Platform, target: Target) -> bool {
    matches!(platform, Platform::Any)
        || matches!((platform, target),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows))
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: BTreeMap<String, Vec<Origin>>,
    features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn active(&self, selected: &BTreeMap<String, Version>) -> Active {
        let mut active = Active::default();
        for (i, root) in self.roots.iter().enumerate() {
            let fs = root.features.iter().cloned().collect::<BTreeSet<_>>().into_iter().collect::<Vec<_>>().join(",");
            let token = format!(
                "root[{}]:{}@{}+{{{}}}",
                i, root.package, constraint_text(root.constraint), fs
            );
            active.origins.entry(root.package.clone()).or_default().push(Origin {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
            active.features.entry(root.package.clone()).or_default().extend(root.features.iter().cloned());
            active.paths.entry(root.package.clone()).or_insert(vec![token]);
        }

        loop {
            let mut changed = false;
            let packages: Vec<String> = selected.keys().cloned().collect();
            for package_name in packages {
                let version = selected[&package_name];
                let Some(package) = self.catalog.get(&package_name) else { continue };
                let Some(release) = package.releases.iter().find(|r| r.version == version) else { continue };
                let feature_names = active.features.entry(package_name.clone()).or_default().clone();
                for feature_name in feature_names {
                    if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                        for enabled in &feature.enables {
                            if active.features.entry(package_name.clone()).or_default().insert(enabled.clone()) {
                                changed = true;
                            }
                        }
                    }
                }
                let source_features = active.features.get(&package_name).cloned().unwrap_or_default();
                let source_path = active.paths.get(&package_name).cloned().unwrap_or_default();
                for (j, dep) in release.deps.iter().enumerate() {
                    if !target_matches(dep.platform, self.target)
                        || dep.gate.as_ref().is_some_and(|g| !source_features.contains(g))
                    {
                        continue;
                    }
                    let feature_list = dep.features.iter().cloned().collect::<BTreeSet<_>>().into_iter().collect::<Vec<_>>().join(",");
                    let gate = dep.gate.as_deref().unwrap_or("*");
                    let token = format!(
                        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        package_name,
                        version_text(version),
                        j,
                        dep.package,
                        constraint_text(dep.constraint),
                        feature_list,
                        gate,
                        match dep.platform {
                            Platform::Any => "any",
                            Platform::Linux => "linux",
                            Platform::Mac => "mac",
                            Platform::Windows => "windows",
                        }
                    );
                    let mut path = source_path.clone();
                    path.push(token);
                    let origins = active.origins.entry(dep.package.clone()).or_default();
                    if !origins.iter().any(|o| o.constraint == dep.constraint && o.path == path) {
                        origins.push(Origin { constraint: dep.constraint, path: path.clone() });
                        changed = true;
                    }
                    let old_path = active.paths.get(&dep.package).cloned();
                    if old_path.as_ref().is_none_or(|p| {
                        (path.len(), &path) < (p.len(), p)
                    }) {
                        active.paths.insert(dep.package.clone(), path);
                        changed = true;
                    }
                    let dest_features = active.features.entry(dep.package.clone()).or_default();
                    for feature in &dep.features {
                        if dest_features.insert(feature.clone()) {
                            changed = true;
                        }
                    }
                }
            }
            if !changed {
                break;
            }
        }
        active
    }

    fn error_at(&self, selected: &BTreeMap<String, Version>, active: &Active) -> Option<ResolveError> {
        let active_names: BTreeSet<String> = active.origins.keys().cloned().collect();
        for name in &active_names {
            if !self.catalog.contains_key(name) {
                return Some(ResolveError::Missing { package: name.clone(), evidence: evidence_for(active, name) });
            }
        }
        for name in selected.keys() {
            if let Some(features) = active.features.get(name) {
                let version = selected[name];
                let release = self.catalog[name].releases.iter().find(|r| r.version == version).unwrap();
                let declared: BTreeSet<&str> = release.features.iter().map(|f| f.name.as_str()).collect();
                for feature in features {
                    if !declared.contains(feature.as_str()) {
                        return Some(ResolveError::MissingFeature {
                            package: name.clone(),
                            selected: version,
                            feature: feature.clone(),
                            path: active.paths.get(name).cloned().unwrap_or_default(),
                        });
                    }
                }
            }
        }
        for (name, version) in selected {
            let cs = active.origins.get(name).map(|v| v.iter().map(|o| o.constraint).collect::<Vec<_>>()).unwrap_or_default();
            if !cs.iter().all(|c| matches(*c, *version)) {
                return Some(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: canonical_constraints(&cs),
                    evidence: evidence_for(active, name),
                });
            }
        }
        for name in active_names {
            if selected.contains_key(&name) {
                continue;
            }
            let cs = active.origins.get(&name).map(|v| v.iter().map(|o| o.constraint).collect::<Vec<_>>()).unwrap_or_default();
            let package = self.catalog[&name];
            if !package.releases.iter().any(|r| cs.iter().all(|c| matches(*c, r.version))) {
                return Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: canonical_constraints(&cs),
                    evidence: evidence_for(active, &name),
                });
            }
        }
        None
    }

    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let active = self.active(selected);
        if let Some(error) = self.error_at(selected, &active) {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        let next = active.origins.keys().find(|name| !selected.contains_key(*name)).cloned();
        let Some(name) = next else {
            let cost = self.locks.iter().filter(|(p, v)| selected.get(*p) != Some(*v)).count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best)) if cost < *best_cost => true,
                Some((best_cost, best)) if cost == *best_cost => {
                    let names: BTreeSet<String> = self.catalog.keys().cloned().collect();
                    let mut cmp = std::cmp::Ordering::Equal;
                    for p in names {
                        let a = selected.get(&p);
                        let b = best.get(&p);
                        cmp = match (a, b) {
                            (None, None) => std::cmp::Ordering::Equal,
                            (None, Some(_)) => std::cmp::Ordering::Less,
                            (Some(_), None) => std::cmp::Ordering::Greater,
                            (Some(a), Some(b)) => a.cmp(b),
                        };
                        if cmp != std::cmp::Ordering::Equal { break; }
                    }
                    cmp == std::cmp::Ordering::Greater
                }
                _ => false,
            };
            if replace {
                self.best = Some((cost, selected.clone()));
            }
            return Ok(());
        };

        let cs = active.origins.get(&name).unwrap().iter().map(|o| o.constraint).collect::<Vec<_>>();
        let mut candidates: Vec<Version> = self.catalog[&name].releases.iter()
            .filter(|r| cs.iter().all(|c| matches(*c, r.version)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(locked) = self.locks.get(&name) {
            if let Some(i) = candidates.iter().position(|v| v == locked) {
                let v = candidates.remove(i);
                candidates.insert(0, v);
            }
        }
        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit { attempts: self.max_attempts });
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            let child = self.visit(selected);
            selected.remove(&name);
            child?;
        }
        Ok(())
    }
}

fn evidence_for(active: &Active, package: &str) -> Vec<Evidence> {
    active.origins.get(package).into_iter().flatten()
        .map(|o| Evidence { constraint: o.constraint, path: o.path.clone() })
        .collect()
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_names(catalog, roots, locks)?;

    let mut package_names = BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage { package: package.name.clone() });
        }
    }
    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion { package: package.name.clone(), version: release.version });
            }
        }
    }
    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock { package: lock.package.clone() });
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let lock_map: BTreeMap<String, Version> = locks.iter().map(|l| (l.package.clone(), l.version)).collect();
    let mut search = Search {
        catalog: &catalog_map,
        roots,
        locks: lock_map.clone(),
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.visit(&mut BTreeMap::new())?;
    let Some((_, selected_map)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| {
            ResolveError::NoMatch { package: String::new(), requirements: vec![], evidence: vec![] }
        }));
    };

    let selected: Vec<Selection> = selected_map.iter().map(|(package, version)| Selection {
        package: package.clone(),
        version: *version,
    }).collect();
    let final_active = search.active(&selected_map);
    let mut changes = Vec::new();
    let names: BTreeSet<String> = lock_map.keys().chain(selected_map.keys()).cloned().collect();
    for package in names {
        let before = lock_map.get(&package).copied();
        let after = selected_map.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map.get(&package).is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let cs = final_active.origins.get(&package)
                        .map(|origins| origins.iter().map(|o| o.constraint).collect::<Vec<_>>())
                        .unwrap_or_default();
                    if !cs.iter().all(|c| matches(*c, old)) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            _ => unreachable!(),
        };
        changes.push(Change { package, before, after, reason });
    }
    Ok(Resolution { selected, changes })
}
