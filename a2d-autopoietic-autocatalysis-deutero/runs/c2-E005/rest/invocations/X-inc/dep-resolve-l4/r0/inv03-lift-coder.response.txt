pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateFeature {
        package: String,
        version: Version,
        feature: String,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
        evidence: Vec<Evidence>,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,
    },
    MissingFeature {
        package: String,
        selected: Version,
        feature: String,
        path: Vec<String>,
    },
    Limit {
        attempts: usize,
    },
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort_by_key(|c| constraint_key(*c));
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn sorted_evidence(mut e: Vec<Evidence>) -> Vec<Evidence> {
    e.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    e
}

fn evidence_lines(e: &[Evidence]) -> String {
    let mut out = String::new();
    for item in sorted_evidence(e.to_vec()) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(item.constraint));
        out.push_str(" via ");
        out.push_str(&item.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            encode(package),
            evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn invalid(site: String, name: &str) -> Option<ResolveError> {
    if valid_name(name) {
        None
    } else {
        Some(ResolveError::InvalidName {
            site,
            name: name.to_string(),
        })
    }
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn target_matches(platform: Platform, target: Target) -> bool {
    matches!(
        (platform, target),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn root_token(i: usize, r: &Requirement) -> String {
    let mut fs = r.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        fs.join(",")
    )
}

fn dependency_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    let mut fs = dep.features.clone();
    fs.sort();
    fs.dedup();
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        fs.join(","),
        gate,
        platform
    )
}

#[derive(Clone, Debug)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Debug, Default)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    activated: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn better_path(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn compute_closure(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let package_map: BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut out = Closure::default();

    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(i, root)];
        out.constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        out.origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: path.clone(),
            });
        out.requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        match out.paths.get(&root.package) {
            Some(old) if !better_path(&path, old) => {}
            _ => {
                out.paths.insert(root.package.clone(), path);
            }
        }
    }

    loop {
        let mut changed = false;

        for (name, version) in selected {
            let Some(package) = package_map.get(name.as_str()) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
                continue;
            };
            let requested = out.requests.get(name).cloned().unwrap_or_default();
            let activated = out.activated.entry(name.clone()).or_default();
            let before = activated.len();
            activated.extend(requested);
            if activated.len() != before {
                changed = true;
            }

            let now = out.activated.get(name).cloned().unwrap_or_default();
            for feature_name in now {
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    let req = out.requests.entry(name.clone()).or_default();
                    let before = req.len();
                    req.extend(feature.enables.iter().cloned());
                    if req.len() != before {
                        changed = true;
                    }
                }
            }

            let activated = out.activated.get(name).cloned().unwrap_or_default();
            let source_path = out.paths.get(name).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !activated.contains(gate))
                {
                    continue;
                }
                out.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                let mut path = source_path.clone();
                path.push(dependency_token(name, *version, j, dep));
                out.origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                match out.paths.get(&dep.package) {
                    Some(old) if !better_path(&path, old) => {}
                    _ => {
                        out.paths.insert(dep.package.clone(), path);
                        changed = true;
                    }
                }
                let req = out.requests.entry(dep.package.clone()).or_default();
                let before = req.len();
                req.extend(dep.features.iter().cloned());
                if req.len() != before {
                    changed = true;
                }
            }
        }

        if !changed {
            break;
        }
    }

    for cs in out.constraints.values_mut() {
        *cs = canonical_constraints(std::mem::take(cs));
    }
    out
}

fn collect_evidence(origins: &BTreeMap<String, Vec<Origin>>, name: &str) -> Vec<Evidence> {
    origins
        .get(name)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect()
}

fn selected_constraints_match(
    constraints: &BTreeMap<String, Vec<Constraint>>,
    name: &str,
    version: Version,
) -> bool {
    constraints
        .get(name)
        .into_iter()
        .flatten()
        .all(|c| matches(version, *c))
}

fn find_dynamic_error(
    catalog: &[Package],
    selected: &BTreeMap<String, Version>,
    closure: &Closure,
) -> Option<ResolveError> {
    let names: BTreeSet<String> = closure.constraints.keys().cloned().collect();
    let catalog_names: BTreeSet<&str> = catalog.iter().map(|p| p.name.as_str()).collect();

    for name in names.iter() {
        if !catalog_names.contains(name.as_str()) {
            return Some(ResolveError::Missing {
                package: name.clone(),
                evidence: collect_evidence(&closure.origins, name),
            });
        }
    }

    let package_map: BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();

    for (name, version) in selected {
        let requested = closure.activated.get(name).cloned().unwrap_or_default();
        let Some(package) = package_map.get(name.as_str()) else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
            continue;
        };
        for feature in requested {
            if !release.features.iter().any(|f| f.name == feature) {
                return Some(ResolveError::MissingFeature {
                    package: name.clone(),
                    selected: *version,
                    feature,
                    path: closure.paths.get(name).cloned().unwrap_or_default(),
                });
            }
        }
    }

    for (name, version) in selected {
        let cs = closure.constraints.get(name).cloned().unwrap_or_default();
        if !cs.iter().all(|c| matches(*version, *c)) {
            return Some(ResolveError::Conflict {
                package: name.clone(),
                selected: *version,
                requirements: cs,
                evidence: collect_evidence(&closure.origins, name),
            });
        }
    }

    for name in names {
        if selected.contains_key(&name) {
            continue;
        }
        let cs = closure.constraints.get(&name).cloned().unwrap_or_default();
        let package = package_map[name.as_str()];
        if !package
            .releases
            .iter()
            .any(|r| cs.iter().all(|c| matches(r.version, *c)))
        {
            return Some(ResolveError::NoMatch {
                package: name.clone(),
                requirements: cs,
                evidence: collect_evidence(&closure.origins, &name),
            });
        }
    }
    None
}

#[derive(Clone)]
struct SearchResult {
    selected: BTreeMap<String, Version>,
    closure: Closure,
}

struct SearchState {
    attempts: usize,
    budget: usize,
    first_error: Option<ResolveError>,
    best: Option<SearchResult>,
    best_cost: usize,
    catalog: Vec<Package>,
    roots: Vec<Requirement>,
    locks: BTreeMap<String, Version>,
    target: Target,
    catalog_names: Vec<String>,
}

impl SearchState {
    fn solution_cost(&self, selected: &BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|(name, version)| selected.get(*name) != Some(*version))
            .count()
    }

    fn better_solution(
        &self,
        candidate: &BTreeMap<String, Version>,
        current: &BTreeMap<String, Version>,
    ) -> bool {
        for name in &self.catalog_names {
            match (candidate.get(name), current.get(name)) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let closure = compute_closure(&self.catalog, &self.roots, selected, self.target);
        if let Some(error) = find_dynamic_error(&self.catalog, selected, &closure) {
            self.record_error(error);
            return Ok(());
        }

        let active: BTreeSet<String> = closure.constraints.keys().cloned().collect();
        let next = active.into_iter().find(|name| !selected.contains_key(name));
        let Some(name) = next else {
            let cost = self.solution_cost(selected);
            let replace = match &self.best {
                None => true,
                Some(best) if cost < self.best_cost => true,
                Some(best) if cost == self.best_cost => {
                    self.better_solution(selected, &best.selected)
                }
                _ => false,
            };
            if replace {
                self.best_cost = cost;
                self.best = Some(SearchResult {
                    selected: selected.clone(),
                    closure,
                });
            }
            return Ok(());
        };

        let package = self
            .catalog
            .iter()
            .find(|p| p.name == name)
            .expect("active package was validated");
        let cs = closure.constraints.get(&name).cloned().unwrap_or_default();
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| cs.iter().all(|c| matches(*v, *c)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();
        if let Some(locked) = self.locks.get(&name) {
            if let Some(pos) = candidates.iter().position(|v| v == locked) {
                candidates.remove(pos);
                candidates.insert(0, *locked);
            }
        }

        for version in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            let result = self.visit(selected);
            selected.remove(&name);
            result?;
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if let Some(e) = invalid(format!("root[{}]", i), &root.package) {
            return Err(e);
        }
        for (f, feature) in root.features.iter().enumerate() {
            if let Some(e) = invalid(format!("root[{}].feature[{}]", i, f), feature) {
                return Err(e);
            }
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if let Some(e) = invalid(format!("catalog[{}]", i), &package.name) {
            return Err(e);
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if let Some(e) = invalid(
                    format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                    &dep.package,
                ) {
                    return Err(e);
                }
                if let Some(gate) = &dep.gate {
                    if let Some(e) = invalid(
                        format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                        gate,
                    ) {
                        return Err(e);
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if let Some(e) = invalid(
                        format!("catalog[{}].release[{}].dep[{}].feature[{}]", i, j, k, f),
                        feature,
                    ) {
                        return Err(e);
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if let Some(e) = invalid(
                    format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                    &feature.name,
                ) {
                    return Err(e);
                }
                for (eidx, enable) in feature.enables.iter().enumerate() {
                    if let Some(e) = invalid(
                        format!(
                            "catalog[{}].release[{}].feature[{}].enable[{}]",
                            i, j, f, eidx
                        ),
                        enable,
                    ) {
                        return Err(e);
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen_features = BTreeSet::new();
            for feature in &release.features {
                if !seen_features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if let Some(e) = invalid(format!("lock[{}]", i), &lock.package) {
            return Err(e);
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut lock_map = BTreeMap::new();
    for lock in locks {
        lock_map.insert(lock.package.clone(), lock.version);
    }
    let mut catalog_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    catalog_names.sort();

    let mut state = SearchState {
        attempts: 0,
        budget: max_attempts,
        first_error: None,
        best: None,
        best_cost: usize::MAX,
        catalog: catalog.to_vec(),
        roots: roots.to_vec(),
        locks: lock_map,
        target,
        catalog_names,
    };
    state.visit(&mut BTreeMap::new())?;

    let Some(best) = state.best else {
        return Err(state.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: roots
                .first()
                .map(|r| r.package.clone())
                .unwrap_or_default(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = best
        .selected
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut all_names: BTreeSet<String> = best.selected.keys().cloned().collect();
    all_names.extend(state.locks.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = state.locks.get(&package).copied();
        let after = best.selected.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !selected_constraints_match(
                    &best.closure.constraints,
                    &package,
                    old,
                ) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
