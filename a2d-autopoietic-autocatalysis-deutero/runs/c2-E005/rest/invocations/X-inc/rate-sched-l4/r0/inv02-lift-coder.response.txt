#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
enum InternalState {
    Waiting {
        ready_at: u64,
        attempt: u8,
        remaining: Option<u64>,
    },
    Running {
        attempt: u8,
        finish_at: u64,
        started_at: u64,
    },
    Succeeded,
    Failed,
    Blocked,
}

pub struct Pool {
    jobs: Vec<Job>,
    states: Vec<InternalState>,
    dependencies: Vec<Vec<usize>>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base < 1
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job.tag.as_ref().is_some_and(|tag| !valid_token(tag, 1, 8)) {
                return Err(Error::BadTag(i));
            }
            if ids.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            ids.insert(job.id.clone(), i);
        }

        let mut dependencies = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                let Some(&index) = ids.get(dep) else {
                    return Err(Error::MissingDependency(i, j));
                };
                dependencies[i].push(index);
            }
        }

        fn visit(
            node: usize,
            deps: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[node] == 1 {
                return false;
            }
            if marks[node] == 2 {
                return true;
            }
            marks[node] = 1;
            for &dep in &deps[node] {
                if !visit(dep, deps, marks) {
                    return false;
                }
            }
            marks[node] = 2;
            true
        }

        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if !visit(i, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let states = jobs
            .iter()
            .map(|job| InternalState::Waiting {
                ready_at: job.release_at,
                attempt: 0,
                remaining: None,
            })
            .collect();

        Ok(Self {
            jobs,
            states,
            dependencies,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let Some(index) = self.jobs.iter().position(|job| job.id == id) else {
            return Err(Error::Unknown);
        };
        Ok(match self.states[index] {
            InternalState::Waiting {
                ready_at, attempt, ..
            } => State::Waiting { ready_at, attempt },
            InternalState::Running {
                attempt, finish_at, ..
            } => State::Running { attempt, finish_at },
            InternalState::Succeeded => State::Succeeded,
            InternalState::Failed => State::Failed,
            InternalState::Blocked => State::Blocked,
        })
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }
        if until == self.horizon {
            return Ok(Vec::new());
        }

        let mut states = self.states.clone();
        let mut lines = Vec::new();
        let mut now = if self.horizon == 0 { 0 } else { self.horizon };

        loop {
            self.process_timestamp(now, &mut states, &mut lines);

            let next = states
                .iter()
                .filter_map(|state| match state {
                    InternalState::Waiting { ready_at, .. } if *ready_at > now => Some(*ready_at),
                    InternalState::Running { finish_at, .. } if *finish_at > now => Some(*finish_at),
                    _ => None,
                })
                .min();

            let Some(next_time) = next else {
                break;
            };
            if next_time > until {
                break;
            }
            now = next_time;
        }

        self.states = states;
        self.horizon = until;
        Ok(lines)
    }

    fn process_timestamp(
        &self,
        now: u64,
        states: &mut [InternalState],
        lines: &mut Vec<String>,
    ) {
        let mut completions: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| match state {
                InternalState::Running { finish_at, .. } if *finish_at == now => Some(i),
                _ => None,
            })
            .collect();
        completions.sort_by(|&a, &b| self.jobs[a].id.cmp(&self.jobs[b].id));

        for index in completions {
            let attempt = match states[index] {
                InternalState::Running { attempt, .. } => attempt,
                _ => continue,
            };
            let job = &self.jobs[index];
            if attempt > job.failures {
                lines.push(format!("{} finish {} {} ok", now, job.id, attempt));
                states[index] = InternalState::Succeeded;
            } else {
                lines.push(format!("{} finish {} {} fail", now, job.id, attempt));
                if attempt >= job.max_attempts {
                    states[index] = InternalState::Failed;
                } else {
                    let shift = u32::from(attempt.saturating_sub(1));
                    let delay = self
                        .retry_base
                        .checked_shl(shift)
                        .unwrap_or(u64::MAX)
                        .min(self.retry_cap);
                    let ready_at = now.saturating_add(delay);
                    lines.push(format!("{} retry {} {} {}", now, job.id, attempt, ready_at));
                    states[index] = InternalState::Waiting {
                        ready_at,
                        attempt,
                        remaining: None,
                    };
                }
            }
        }

        loop {
            let mut newly_blocked = Vec::new();
            for i in 0..states.len() {
                if matches!(states[i], InternalState::Waiting { .. })
                    && self.dependencies[i].iter().any(|&dep| {
                        matches!(states[dep], InternalState::Failed | InternalState::Blocked)
                    })
                {
                    newly_blocked.push(i);
                }
            }
            if newly_blocked.is_empty() {
                break;
            }
            for &i in &newly_blocked {
                states[i] = InternalState::Blocked;
            }
            newly_blocked.sort_by(|&a, &b| self.jobs[a].id.cmp(&self.jobs[b].id));
            for i in newly_blocked {
                let dependency = self.dependencies[i]
                    .iter()
                    .filter(|&&dep| {
                        matches!(states[dep], InternalState::Failed | InternalState::Blocked)
                    })
                    .map(|&dep| self.jobs[dep].id.as_str())
                    .min()
                    .unwrap_or("");
                lines.push(format!("{} skip {} {}", now, self.jobs[i].id, dependency));
            }
        }

        loop {
            let mut candidates: Vec<usize> = states
                .iter()
                .enumerate()
                .filter_map(|(i, state)| match state {
                    InternalState::Waiting { ready_at, .. }
                        if *ready_at <= now
                            && self.dependencies[i]
                                .iter()
                                .all(|&dep| matches!(states[dep], InternalState::Succeeded)) =>
                    {
                        Some(i)
                    }
                    _ => None,
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, now, states);
                let pb = self.effective_priority(b, now, states);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut changed = false;
            for candidate in candidates {
                let running: Vec<usize> = states
                    .iter()
                    .enumerate()
                    .filter_map(|(i, state)| {
                        matches!(state, InternalState::Running { .. }).then_some(i)
                    })
                    .collect();

                let slot_free = running.len() < self.capacity;
                let tag_free = self.tag_free(candidate, &running);

                let victim = if slot_free && tag_free {
                    None
                } else {
                    self.choose_victim(candidate, &running, states)
                };

                if !slot_free && victim.is_none() {
                    continue;
                }
                if !tag_free && victim.is_none() {
                    continue;
                }

                if let Some(victim_index) = victim {
                    let (attempt, finish_at) = match states[victim_index] {
                        InternalState::Running {
                            attempt, finish_at, ..
                        } => (attempt, finish_at),
                        _ => continue,
                    };
                    let remaining = finish_at.saturating_sub(now);
                    lines.push(format!(
                        "{} preempt {} by {}",
                        now, self.jobs[victim_index].id, self.jobs[candidate].id
                    ));
                    states[victim_index] = InternalState::Waiting {
                        ready_at: now,
                        attempt,
                        remaining: Some(remaining),
                    };
                }

                let (old_attempt, remaining) = match states[candidate] {
                    InternalState::Waiting {
                        attempt, remaining, ..
                    } => (attempt, remaining),
                    _ => continue,
                };
                let (attempt, duration) = match remaining {
                    Some(rem) => (old_attempt, rem),
                    None => (old_attempt.saturating_add(1), self.jobs[candidate].duration),
                };
                let finish_at = now.saturating_add(duration);
                lines.push(format!(
                    "{} start {} {} {}",
                    now, self.jobs[candidate].id, attempt, duration
                ));
                states[candidate] = InternalState::Running {
                    attempt,
                    finish_at,
                    started_at: now,
                };
                changed = true;
                break;
            }

            if !changed {
                break;
            }
        }
    }

    fn effective_priority(&self, index: usize, now: u64, states: &[InternalState]) -> u8 {
        let ready_at = match states[index] {
            InternalState::Waiting { ready_at, .. } => ready_at,
            _ => now,
        };
        let base = u64::from(self.jobs[index].priority);
        if self.aging_ms == 0 {
            return base as u8;
        }
        let bonus = now.saturating_sub(ready_at) / self.aging_ms;
        base.saturating_add(bonus).min(9) as u8
    }

    fn tag_free(&self, candidate: usize, running: &[usize]) -> bool {
        let Some(tag) = self.jobs[candidate].tag.as_ref() else {
            return true;
        };
        running.iter().all(|&i| self.jobs[i].tag.as_ref() != Some(tag))
    }

    fn choose_victim(
        &self,
        candidate: usize,
        running: &[usize],
        states: &[InternalState],
    ) -> Option<usize> {
        let candidate_priority = self.jobs[candidate].priority;
        let candidate_tag = self.jobs[candidate].tag.as_ref();

        let mut eligible: Vec<usize> = running
            .iter()
            .copied()
            .filter(|&i| {
                let victim_priority = self.jobs[i].priority;
                candidate_priority >= victim_priority.saturating_add(2)
                    && candidate_tag.map_or(true, |tag| self.jobs[i].tag.as_ref() == Some(tag))
            })
            .collect();

        eligible.sort_by(|&a, &b| {
            let pa = self.jobs[a].priority;
            let pb = self.jobs[b].priority;
            let sa = match states[a] {
                InternalState::Running { started_at, .. } => started_at,
                _ => 0,
            };
            let sb = match states[b] {
                InternalState::Running { started_at, .. } => started_at,
                _ => 0,
            };
            pa.cmp(&pb)
                .then_with(|| sb.cmp(&sa))
                .then_with(|| self.jobs[b].id.cmp(&self.jobs[a].id))
        });
        eligible.into_iter().next()
    }
}

fn valid_token(value: &str, min_len: usize, max_len: usize) -> bool {
    let bytes = value.as_bytes();
    (min_len..=max_len).contains(&bytes.len())
        && bytes.iter().all(|&b| {
            b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-'
        })
}
