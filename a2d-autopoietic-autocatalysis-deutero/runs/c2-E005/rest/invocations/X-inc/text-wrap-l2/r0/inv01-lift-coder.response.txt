#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let normalized: String = input
        .chars()
        .map(|ch| if ch == '\r' { '\n' } else { ch })
        .collect();

    let chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut i = 0;

    while i < chars.len() {
        if chars[i] == '\n' {
            let run_start = i;
            while i < chars.len() && chars[i] == '\n' {
                i += 1;
            }

            if i - run_start >= 2 {
                wrap_segment(&chars[segment_start..run_start], options, &mut output);
                for _ in 1..(i - run_start) {
                    output.push(String::new());
                }
                segment_start = i;
            }
        } else {
            i += 1;
        }
    }

    wrap_segment(&chars[segment_start..], options, &mut output);
    Ok(output)
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn wrap_segment(chars: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words: Vec<Vec<char>> = Vec::new();
    let mut word = Vec::new();

    for &ch in chars {
        if is_separator(ch) {
            if !word.is_empty() {
                words.push(std::mem::take(&mut word));
            }
        } else {
            word.push(ch);
        }
    }
    if !word.is_empty() {
        words.push(word);
    }
    if words.is_empty() {
        return;
    }

    let mut first_line = true;
    let mut indent = options.first_indent;
    let mut content = Vec::<char>::new();

    for word in words {
        let mut offset = 0;

        while offset < word.len() {
            let capacity = options.width - indent;
            let remaining = word.len() - offset;

            if remaining <= capacity {
                if content.is_empty() {
                    content.extend_from_slice(&word[offset..]);
                    offset = word.len();
                } else if content.len() + 1 + remaining <= capacity {
                    content.push(' ');
                    content.extend_from_slice(&word[offset..]);
                    offset = word.len();
                } else {
                    emit_line(&content, indent, output);
                    content.clear();
                    first_line = false;
                    indent = options.rest_indent;
                }
            } else if content.is_empty() {
                content.extend_from_slice(&word[offset..offset + capacity]);
                offset += capacity;
                emit_line(&content, indent, output);
                content.clear();
                first_line = false;
                indent = options.rest_indent;
            } else {
                let available = capacity.saturating_sub(content.len() + 1);
                if available >= 2 {
                    content.push(' ');
                    content.extend_from_slice(&word[offset..offset + available]);
                    offset += available;
                    emit_line(&content, indent, output);
                    content.clear();
                    first_line = false;
                    indent = options.rest_indent;
                } else {
                    emit_line(&content, indent, output);
                    content.clear();
                    first_line = false;
                    indent = options.rest_indent;
                }
            }
        }

        // Keep the paragraph's line state independent of whether a word ended
        // exactly at a line boundary.
        if first_line {
            indent = options.first_indent;
        }
    }

    if !content.is_empty() {
        emit_line(&content, indent, output);
    }
}

fn emit_line(content: &[char], indent: usize, output: &mut Vec<String>) {
    let mut line = String::with_capacity(indent + content.len());
    line.extend(std::iter::repeat_n(' ', indent));
    line.extend(content.iter());
    output.push(line);
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn public_unicode_words() {
        assert_eq!(
            wrap(
                "猫犬 鳥魚",
                &WrapOptions {
                    width: 5,
                    first_indent: 0,
                    rest_indent: 0
                }
            ),
            Ok(vec!["猫犬 鳥魚".to_string()])
        );
    }

    #[test]
    fn public_hanging() {
        assert_eq!(
            wrap(
                "a bb ccc dd",
                &WrapOptions {
                    width: 7,
                    first_indent: 1,
                    rest_indent: 2
                }
            ),
            Ok(vec![
                " a bb".to_string(),
                "  ccc".to_string(),
                "  dd".to_string()
            ])
        );
    }

    #[test]
    fn public_paragraph() {
        assert_eq!(
            wrap(
                "a\n\nb",
                &WrapOptions {
                    width: 4,
                    first_indent: 0,
                    rest_indent: 1
                }
            ),
            Ok(vec!["a".to_string(), "".to_string(), "b".to_string()])
        );
    }

    #[test]
    fn public_bad_indent() {
        assert_eq!(
            wrap(
                "x",
                &WrapOptions {
                    width: 3,
                    first_indent: 3,
                    rest_indent: 0
                }
            ),
            Err(WrapError::FirstIndentTooWide { indent: 3 })
        );
    }
}
