#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let high = hex_value(bytes[i + 1]);
            let low = hex_value(bytes[i + 2]);
            match (high, low) {
                (Some(high), Some(low)) => {
                    decoded.push((high << 4) | low);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let decoded_offset = error.valid_up_to();
            let original_offset = offsets.get(decoded_offset).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_offset })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for &byte in text.as_bytes() {
        let common_safe = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let component_safe = match component {
            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => matches!(byte, b'/' | b'?'),
            Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
        };

        if common_safe || component_safe {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn is_scheme_first(byte: u8) -> bool {
    byte.is_ascii_alphabetic()
}

fn is_scheme_rest(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')
}

fn map_codec_error(error: CodecError, base: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

fn validate_component(component: Component, value: &str, base: usize) -> Result<(), ParseError> {
    decode_component(component, value)
        .map(|_| ())
        .map_err(|error| map_codec_error(error, base))
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(colon) => colon,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !is_scheme_first(bytes[0]) {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !is_scheme_rest(byte) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(bytes.len());

    let authority = &input[authority_start..authority_end];
    let authority_bytes = authority.as_bytes();

    let at_positions: Vec<usize> = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'@').then_some(index))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, host_start_local) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(&authority[..at]), at + 1)
    } else {
        (None, 0)
    };

    let host_start = authority_start + host_start_local;
    let hostport = &authority[host_start..authority_end];
    let hostport_bytes = hostport.as_bytes();

    let (host, port, ipv6) = if hostport_bytes.first() == Some(&b'[') {
        let closing = hostport_bytes[1..]
            .iter()
            .position(|&byte| byte == b']')
            .map(|offset| offset + 1)
            .ok_or(ParseError::InvalidIpv6 { at: host_start })?;

        let interior = &hostport[1..closing];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }

        let suffix_start = closing + 1;
        let port = if suffix_start == hostport.len() {
            None
        } else if hostport_bytes[suffix_start] == b':' {
            let port_start = host_start + suffix_start + 1;
            Some(parse_port(&hostport[suffix_start + 1..], port_start)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + suffix_start,
            });
        };

        (interior.to_ascii_lowercase(), port, true)
    } else {
        let split = hostport_bytes.iter().position(|&byte| byte == b':');
        let (host_text, port) = if let Some(split) = split {
            let port_start = host_start + split + 1;
            (
                &hostport[..split],
                Some(parse_port(&hostport[split + 1..], port_start)?),
            )
        } else {
            (hostport, None)
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (offset, &byte) in host_text.as_bytes().iter().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + offset,
                });
            }
        }
        (host_text.to_ascii_lowercase(), port, false)
    };

    let fragment_delimiter = bytes[authority_end..]
        .iter()
        .position(|&byte| byte == b'#')
        .map(|offset| authority_end + offset);

    let before_fragment_end = fragment_delimiter.unwrap_or(bytes.len());
    let query_delimiter = bytes[authority_end..before_fragment_end]
        .iter()
        .position(|&byte| byte == b'?')
        .map(|offset| authority_end + offset);

    let path_end = query_delimiter.unwrap_or(before_fragment_end);
    let path = if path_end == authority_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };

    let query = query_delimiter.map(|start| {
        let end = before_fragment_end;
        input[start + 1..end].to_owned()
    });
    let fragment = fragment_delimiter.map(|start| input[start + 1..].to_owned());

    if let Some(value) = userinfo {
        validate_component(Component::UserInfo, value, authority_start)?;
    }
    if path_end != authority_end {
        validate_component(Component::Path, &path, authority_end)?;
    }
    if let Some(start) = query_delimiter {
        let value = query.as_deref().unwrap_or("");
        validate_component(Component::Query, value, start + 1)?;
    }
    if let Some(start) = fragment_delimiter {
        let value = fragment.as_deref().unwrap_or("");
        validate_component(Component::Fragment, value, start + 1)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo: userinfo.map(str::to_owned),
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(value: &str, port_start: usize) -> Result<u16, ParseError> {
    if value.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    for (offset, byte) in value.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: port_start + offset,
            });
        }
    }

    if value.len() >= 2 && value.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut port: u32 = 0;
    for byte in value.bytes() {
        port = match port
            .checked_mul(10)
            .and_then(|number| number.checked_add((byte - b'0') as u32))
        {
            Some(number) if number <= 65535 => number,
            _ => return Err(ParseError::InvalidPort { at: port_start }),
        };
    }

    if port == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    Ok(port as u16)
}
