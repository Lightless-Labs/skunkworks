#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-')
}

fn validate_name(site: String, name: &str) -> Result<(), ResolveError> {
    if valid_name(name) {
        Ok(())
    } else {
        Err(ResolveError::InvalidName {
            site,
            name: name.to_owned(),
        })
    }
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_rank(constraint: Constraint) -> (u8, Version) {
    match constraint {
        Constraint::Exact(version) => (0, version),
        Constraint::AtLeast(version) => (1, version),
        Constraint::Before(version) => (2, version),
        Constraint::Caret(version) => (3, version),
    }
}

fn canonical_constraints(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort_by_key(|constraint| constraint_rank(*constraint));
    constraints.dedup();
    constraints
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(version) => format!("={}", version_text(version)),
        Constraint::AtLeast(version) => format!(">={}", version_text(version)),
        Constraint::Before(version) => format!("<{}", version_text(version)),
        Constraint::Caret(version) => format!("^{}", version_text(version)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical_constraints(constraints.to_vec())
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(target) => version == target,
        Constraint::AtLeast(target) => version >= target,
        Constraint::Before(target) => version < target,
        Constraint::Caret(target) => {
            if version < target {
                false
            } else if target.major > 0 {
                version.major == target.major
            } else if target.minor > 0 {
                version.major == 0 && version.minor == target.minor
            } else {
                version == target
            }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| matches_constraint(version, *constraint))
}

fn percent_encode(text: &str) -> String {
    let mut encoded = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            encoded.push(*byte as char);
        } else {
            use std::fmt::Write;
            let _ = write!(encoded, "%{:02X}", byte);
        }
    }
    encoded
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            percent_encode(site),
            percent_encode(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", percent_encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            percent_encode(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", percent_encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            percent_encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            percent_encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn active_constraints(
    roots: &[Requirement],
    selected: &[(String, usize)],
    catalog: &[Package],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_insert_with(Vec::new)
            .push(root.constraint);
    }

    for (package_name, release_index) in selected {
        let package_index = catalog
            .iter()
            .position(|package| package.name == *package_name)
            .expect("selected package must exist");
        let release = &catalog[package_index].releases[*release_index];
        for dependency in &release.deps {
            active
                .entry(dependency.package.clone())
                .or_insert_with(Vec::new)
                .push(dependency.constraint);
        }
    }

    for constraints in active.values_mut() {
        *constraints = canonical_constraints(std::mem::take(constraints));
    }

    active
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &mut Vec<(String, usize)>,
    attempts: &mut usize,
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    let active = active_constraints(roots, selected, catalog);

    if let Some(package) = active
        .keys()
        .filter(|name| !catalog.iter().any(|entry| entry.name == ***name))
        .next()
    {
        return Err(ResolveError::Missing {
            package: package.clone(),
        });
    }

    let mut selected_by_name = std::collections::BTreeMap::new();
    for (name, release_index) in selected.iter() {
        selected_by_name.insert(name.as_str(), *release_index);
    }

    for (package_name, constraints) in &active {
        if let Some(release_index) = selected_by_name.get(package_name.as_str()) {
            let package = catalog
                .iter()
                .find(|entry| entry.name == *package_name)
                .expect("selected package must exist");
            let version = package.releases[*release_index].version;
            if !matches_all(version, constraints) {
                return Err(ResolveError::Conflict {
                    package: package_name.clone(),
                    selected: version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    for (package_name, constraints) in &active {
        if selected_by_name.contains_key(package_name.as_str()) {
            continue;
        }
        let package = catalog
            .iter()
            .find(|entry| entry.name == *package_name)
            .expect("missing packages were checked");
        if !package
            .releases
            .iter()
            .any(|release| matches_all(release.version, constraints))
        {
            return Err(ResolveError::NoMatch {
                package: package_name.clone(),
                requirements: constraints.clone(),
            });
        }
    }

    let next_package = active
        .keys()
        .find(|name| !selected_by_name.contains_key(name.as_str()))
        .cloned();

    let Some(package_name) = next_package else {
        let mut result = selected
            .iter()
            .map(|(name, release_index)| {
                let package = catalog
                    .iter()
                    .find(|entry| entry.name == *name)
                    .expect("selected package must exist");
                Selection {
                    package: name.clone(),
                    version: package.releases[*release_index].version,
                }
            })
            .collect::<Vec<_>>();
        result.sort_by(|left, right| left.package.cmp(&right.package));
        return Ok(result);
    };

    let constraints = active
        .get(&package_name)
        .expect("active package must have constraints");
    let package = catalog
        .iter()
        .find(|entry| entry.name == package_name)
        .expect("missing packages were checked");

    let mut candidates = package
        .releases
        .iter()
        .enumerate()
        .filter(|(_, release)| matches_all(release.version, constraints))
        .map(|(index, release)| (index, release.version))
        .collect::<Vec<_>>();
    candidates.sort_by(|left, right| right.1.cmp(&left.1));

    let mut first_error = None;
    for (release_index, _) in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.push((package_name.clone(), release_index));
        let result = search(catalog, roots, selected, attempts, max_attempts);
        selected.pop();

        match result {
            Ok(solution) => return Ok(solution),
            Err(error @ ResolveError::Limit { .. }) => return Err(error),
            Err(error) => {
                if first_error.is_none() {
                    first_error = Some(error);
                }
            }
        }
    }

    Err(first_error.expect("an active unselected package has a candidate"))
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    for (index, root) in roots.iter().enumerate() {
        validate_name(format!("root[{index}]"), &root.package)?;
    }

    for (package_index, package) in catalog.iter().enumerate() {
        validate_name(format!("catalog[{package_index}]"), &package.name)?;
        for (release_index, release) in package.releases.iter().enumerate() {
            for (dependency_index, dependency) in release.deps.iter().enumerate() {
                validate_name(
                    format!(
                        "catalog[{package_index}].release[{release_index}].dep[{dependency_index}]"
                    ),
                    &dependency.package,
                )?;
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut attempts = 0;
    search(
        catalog,
        roots,
        &mut Vec::new(),
        &mut attempts,
        max_attempts,
    )
}
