#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    MissingCheck { offset: usize },
    InvalidByte { offset: usize, byte: u8 },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    CheckMismatch {
        offset: usize,
        expected: u8,
        found: u8,
    },
}

const DATA_ALPHABET: &[u8; 32] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ";
const CHECK_ALPHABET: &[u8; 37] = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ*~$=U";

pub fn encode(input: &[u8], checked: bool) -> String {
    let mut output = String::new();
    let mut remainder = 0u8;

    for chunk in input.chunks(5) {
        let mut bits = 0u64;
        for &byte in chunk {
            bits = (bits << 8) | u64::from(byte);
        }
        let bit_count = chunk.len() * 8;
        let symbol_count = (bit_count + 4) / 5;
        bits <<= symbol_count * 5 - bit_count;

        for i in 0..symbol_count {
            let shift = (symbol_count - 1 - i) * 5;
            let value = ((bits >> shift) & 31) as u8;
            output.push(DATA_ALPHABET[value as usize] as char);
            remainder = ((u16::from(remainder) * 32 + u16::from(value)) % 37) as u8;
        }
    }

    if checked {
        output.push(CHECK_ALPHABET[remainder as usize] as char);
    }
    output
}

fn data_value(byte: u8) -> Option<u8> {
    match byte {
        b'0' | b'O' | b'o' => Some(0),
        b'1' | b'I' | b'i' | b'L' | b'l' => Some(1),
        b'2'..=b'9' => Some(byte - b'0'),
        b'A'..=b'H' => Some(byte - b'A' + 10),
        b'a'..=b'h' => Some(byte - b'a' + 10),
        b'J'..=b'K' => Some(byte - b'J' + 18),
        b'j'..=b'k' => Some(byte - b'j' + 18),
        b'M'..=b'N' => Some(byte - b'M' + 20),
        b'm'..=b'n' => Some(byte - b'm' + 20),
        b'P'..=b'T' => Some(byte - b'P' + 22),
        b'p'..=b't' => Some(byte - b'p' + 22),
        b'V'..=b'Z' => Some(byte - b'V' + 27),
        b'v'..=b'z' => Some(byte - b'v' + 27),
        _ => None,
    }
}

fn check_value(byte: u8) -> Option<u8> {
    match byte {
        b'*' => Some(32),
        b'~' => Some(33),
        b'$' => Some(34),
        b'=' => Some(35),
        b'U' | b'u' => Some(36),
        _ => data_value(byte),
    }
}

pub fn decode(input: &str, checked: bool) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let significant: Vec<(usize, u8)> = bytes
        .iter()
        .copied()
        .enumerate()
        .filter(|&(_, byte)| byte != b'-')
        .collect();

    if checked && significant.is_empty() {
        return Err(DecodeError::MissingCheck {
            offset: input.len(),
        });
    }

    let (data, check) = if checked {
        significant.split_at(significant.len() - 1)
    } else {
        (significant.as_slice(), &[][..])
    };

    let mut values = Vec::with_capacity(data.len());
    for &(offset, byte) in data {
        match data_value(byte) {
            Some(value) => values.push(value),
            None => return Err(DecodeError::InvalidByte { offset, byte }),
        }
    }

    let parsed_check = if checked {
        let (offset, byte) = check[0];
        match check_value(byte) {
            Some(value) => Some((offset, value)),
            None => return Err(DecodeError::InvalidByte { offset, byte }),
        }
    } else {
        None
    };

    let residue = values.len() % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength {
            offset: input.len(),
        });
    }

    let unused_bits = match residue {
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => 0,
    };
    if unused_bits != 0 {
        let last_offset = data.last().map(|&(offset, _)| offset).unwrap_or(0);
        let mask = (1u8 << unused_bits) - 1;
        if values.last().copied().unwrap_or(0) & mask != 0 {
            return Err(DecodeError::TrailingBits {
                offset: last_offset,
            });
        }
    }

    let mut remainder = 0u8;
    let mut accumulator = 0u32;
    let mut bit_count = 0u8;
    let mut output = Vec::with_capacity(values.len() * 5 / 8);

    for &value in &values {
        remainder = ((u16::from(remainder) * 32 + u16::from(value)) % 37) as u8;
        accumulator = (accumulator << 5) | u32::from(value);
        bit_count += 5;

        while bit_count >= 8 {
            bit_count -= 8;
            output.push((accumulator >> bit_count) as u8);
            accumulator &= (1u32 << bit_count) - 1;
        }
    }

    if let Some((offset, found)) = parsed_check {
        if found != remainder {
            return Err(DecodeError::CheckMismatch {
                offset,
                expected: remainder,
                found,
            });
        }
    }

    Ok(output)
}
