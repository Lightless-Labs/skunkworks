use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn global_scan(input: &str) -> Result<(), ParseError> {
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

fn parse_port(port: &str, start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + i });
        }
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut value: u32 = 0;
    for b in port.bytes() {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: start }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

fn map_codec_to_parse(e: CodecError, offset: usize) -> ParseError {
    match e {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    global_scan(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || b == b'+' || b == b'-' || b == b'.') {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..]
        .iter()
        .position(|b| matches!(*b, b'/' | b'?' | b'#'))
        .map(|p| auth_start + p)
        .unwrap_or(input.len());
    let authority = &input[auth_start..auth_end];
    let ats: Vec<usize> = authority.match_indices('@').map(|(i, _)| i).collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo { at: auth_start + ats[1] });
    }
    let (userinfo, hostport, hostport_start) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_start });
        }
        (
            Some(authority[..at].to_string()),
            &authority[at + 1..],
            auth_start + at + 1,
        )
    } else {
        (None, authority, auth_start)
    };

    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let open = hostport_start;
        let close = match hostport.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(i) => i + 1,
            None => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open });
        }
        ipv6 = true;
        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], hostport_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost { at: hostport_start + close + 1 });
        };
        (interior.to_ascii_lowercase(), port)
    } else {
        let split = hostport.find(':');
        let (hostpart, portpart) = match split {
            Some(i) => (&hostport[..i], Some((&hostport[i + 1..], hostport_start + i + 1))),
            None => (hostport, None),
        };
        if hostpart.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (i, b) in hostpart.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: hostport_start + i });
            }
        }
        let port = match portpart {
            Some((p, at)) => Some(parse_port(p, at)?),
            None => None,
        };
        (hostpart.to_ascii_lowercase(), port)
    };

    let hash = bytes[auth_end..].iter().position(|&b| b == b'#').map(|i| auth_end + i);
    let pre_fragment_end = hash.unwrap_or(input.len());
    let question = bytes[auth_end..pre_fragment_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|i| auth_end + i);

    let (path, query) = match question {
        Some(q) => (
            &input[auth_end..q],
            Some(input[q + 1..pre_fragment_end].to_string()),
        ),
        None => (&input[auth_end..pre_fragment_end], None),
    };
    let fragment = hash.map(|h| input[h + 1..].to_string());
    let stored_path = if path.is_empty() { "/".to_string() } else { path.to_string() };

    if let Some(ui) = &userinfo {
        decode_component(Component::UserInfo, ui)
            .map_err(|e| map_codec_to_parse(e, auth_start))?;
    }
    if !path.is_empty() {
        decode_component(Component::Path, path)
            .map_err(|e| map_codec_to_parse(e, auth_end))?;
    }
    if let Some(q) = &query {
        let q_start = question.unwrap() + 1;
        decode_component(Component::Query, q).map_err(|e| map_codec_to_parse(e, q_start))?;
    }
    if let Some(f) = &fragment {
        let f_start = hash.unwrap() + 1;
        decode_component(Component::Fragment, f).map_err(|e| map_codec_to_parse(e, f_start))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path: stored_path,
        query,
        fragment,
        ipv6,
    })
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(h), Some(l)) => {
                    decoded.push(h * 16 + l);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }
    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_string()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: offsets.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::new();
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(hex_digit(b >> 4));
            out.push(hex_digit(b & 0x0f));
        }
    }
    out
}

fn hex_digit(b: u8) -> char {
    b"0123456789ABCDEF"[b as usize] as char
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn codec_to_path(e: CodecError) -> PathError {
    match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(codec_to_path)?;
    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    for (i, raw) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, raw).map_err(codec_to_path)?;
        if decoded == "." {
            if i + 1 == segments.len() {
                stack.push("");
            }
        } else if decoded == ".." {
            if !stack.is_empty() {
                stack.pop();
            }
            if i + 1 == segments.len() {
                stack.push("");
            }
        } else {
            stack.push(raw);
        }
    }
    Ok(format!("/{}", stack.join("/")))
}

fn first_component_end(s: &str) -> usize {
    s.bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(s.len())
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    global_scan(reference).map_err(ResolveError::Reference)?;

    if reference[..first_component_end(reference)].contains(':') {
        let mut u = parse_url(reference).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }
    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut u = parse_url(&constructed).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }

    let hash = reference.find('#');
    let before_fragment = hash.map(|i| &reference[..i]).unwrap_or(reference);
    let fragment = hash.map(|i| reference[i + 1..].to_string());
    let question = before_fragment.find('?');
    let ref_path = question
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let ref_query = question.map(|i| before_fragment[i + 1..].to_string());

    let (selected_path, selected_query) = if ref_path.is_empty() {
        (
            base_path.clone(),
            if question.is_none() {
                base_url.query.clone()
            } else {
                ref_query
            },
        )
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_string()
        } else {
            let slash = base_path.rfind('/').map(|i| i + 1).unwrap_or(0);
            format!("{}{}", &base_path[..slash], ref_path)
        };
        (merged, ref_query)
    };

    let mut raw = format!("{}://", base_url.scheme);
    if let Some(ui) = &base_url.userinfo {
        raw.push_str(ui);
        raw.push('@');
    }
    if base_url.ipv6 {
        raw.push('[');
        raw.push_str(&base_url.host);
        raw.push(']');
    } else {
        raw.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        raw.push(':');
        raw.push_str(&port.to_string());
    }
    raw.push_str(&selected_path);
    if let Some(q) = &selected_query {
        raw.push('?');
        raw.push_str(q);
    }
    if let Some(f) = &fragment {
        raw.push('#');
        raw.push_str(f);
    }
    let mut u = parse_url(&raw).map_err(ResolveError::Reference)?;
    u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
    Ok(u)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct QueryMap {
    pub entries: Vec<(String, Vec<String>)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CanonicalError {
    Parse(ParseError),
    Path(PathError),
    Query(QueryError),
}

fn form_decode(s: &str, offset: usize) -> Result<String, QueryError> {
    let replaced = s.replace('+', " ");
    decode_component(Component::Query, &replaced).map_err(|e| match e {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    })
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut field_start = 0;
    for field in input.split('&') {
        let len = field.len();
        if !field.is_empty() {
            let eq = field.find('=');
            let (key_raw, value_raw, value_offset) = match eq {
                Some(i) => (&field[..i], &field[i + 1..], field_start + i + 1),
                None => (field, "", field_start + len),
            };
            let key = form_decode(key_raw, field_start)?;
            if key.is_empty() {
                return Err(QueryError::EmptyKey { at: field_start });
            }
            let value = form_decode(value_raw, value_offset)?;
            if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
                values.push(value);
            } else {
                entries.push((key, vec![value]));
            }
        }
        field_start += len + 1;
    }
    Ok(QueryMap { entries })
}

fn form_encode(s: &str) -> String {
    let mut out = String::new();
    for &b in s.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(hex_digit(b >> 4));
            out.push(hex_digit(b & 0x0f));
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, target)) = merged.iter_mut().find(|(k, _)| k == key) {
            target.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }
    let mut fields = Vec::new();
    for (key, values) in merged {
        for value in values {
            fields.push(format!("{}={}", form_encode(&key), form_encode(&value)));
        }
    }
    fields.join("&")
}

fn encode_unreserved(text: &str) -> String {
    let mut out = String::new();
    for &b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(hex_digit(b >> 4));
            out.push(hex_digit(b & 0x0f));
        }
    }
    out
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query = match &url.query {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(ui) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, ui)
            .map_err(|e| CanonicalError::Parse(map_codec_to_parse(e, 0)))?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }
    if url.ipv6 {
        let addr = Ipv6Addr::from_str(&url.host).unwrap();
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    let omit_port = matches!((url.scheme.as_str(), url.port), ("http", Some(80)) | ("https", Some(443)));
    if let Some(port) = url.port.filter(|_| !omit_port) {
        out.push(':');
        out.push_str(&port.to_string());
    }

    let mut canonical_path_segments = Vec::new();
    for segment in path.split('/') {
        let decoded = decode_component(Component::Path, segment)
            .map_err(|e| CanonicalError::Path(codec_to_path(e)))?;
        canonical_path_segments.push(encode_unreserved(&decoded));
    }
    out.push_str(&canonical_path_segments.join("/"));

    if let Some(q) = query {
        out.push('?');
        out.push_str(&serialise_query(&q));
    }
    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|e| CanonicalError::Parse(map_codec_to_parse(CodecError::InvalidUtf8 { at: 0 }, 0)))?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(out)
}
