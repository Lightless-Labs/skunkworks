fn hex_value(b: u8) -> u8 {

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() || !is_hex(bytes[i + 1]) || !is_hex(bytes[i + 2]) {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]));
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = origins.get(e.valid_up_to()).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

fn component_safe(component: Component, b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')

            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => matches!(b, b'/' | b'?'),
            Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
        }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {

    }

}

fn parse_port(input: &str, port_start: usize) -> Result<u16, ParseError> {
    if input.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    if let Some(i) = input.as_bytes().iter().position(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: port_start + i });
    }
    if input.len() >= 2 && input.starts_with('0') {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    let mut n: u32 = 0;
    for b in input.bytes() {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at: port_start })?;
    }
    if !(1..=65535).contains(&n) {
        return Err(ParseError::InvalidPort { at: port_start });
    }

}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    if let Some(at) = global_byte_error(input) {
        return Err(ParseError::InvalidByte { at });
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| authority_start + i)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];
    let auth_bytes = authority.as_bytes();

    let at_positions: Vec<usize> = auth_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }
    let (userinfo, host_start_rel) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

            None => return Err(ParseError::InvalidIpv6 { at: host_start }),

        let interior = &hostport[1..close];
        if interior.contains('%') || interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }
        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(
                &suffix[1..],
                host_start + close + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let split = hostport.find(':');
        let (raw_host, raw_port) = match split {
            Some(i) => (&hostport[..i], Some((&hostport[i + 1..], i))),
            None => (hostport, None),
        };
        if raw_host.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        if let Some(i) = raw_host.as_bytes().iter().position(|&b| {
            !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_'))
        }) {
            return Err(ParseError::InvalidHost { at: host_start + i });
        }
        let port = if let Some((p, i)) = raw_port {
            Some(parse_port(p, host_start + i + 1)?)
        } else {
            None
        };
        (raw_host.to_ascii_lowercase(), port, false)
    };

    let remainder = &input[authority_end..];
    let fragment_rel = remainder.find('#');
    let before_fragment_end = fragment_rel.unwrap_or(remainder.len());
    let before_fragment = &remainder[..before_fragment_end];
    let query_rel = before_fragment.find('?');
    let path_end = query_rel.unwrap_or(before_fragment.len());
    let raw_path = &before_fragment[..path_end];
    let query = query_rel.map(|q| before_fragment[q + 1..].to_owned());
    let fragment = fragment_rel.map(|f| remainder[f + 1..].to_owned());
    let path = if raw_path.is_empty() {
        "/".to_owned()
    } else {
        raw_path.to_owned()
    };

    if let Some(ref u) = userinfo {
        let u_offset = authority_start;
        validate_component(u, u_offset, Component::UserInfo)?;
    }
    if !raw_path.is_empty() {
        validate_component(raw_path, authority_end, Component::Path)?;
    }
    if let Some(q) = query.as_deref() {
        let q_offset = authority_end + query_rel.unwrap() + 1;
        validate_component(q, q_offset, Component::Query)?;
    }
    if let Some(f) = fragment.as_deref() {
        let f_offset = authority_end + fragment_rel.unwrap() + 1;
        validate_component(f, f_offset, Component::Fragment)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn map_codec_path(e: CodecError) -> PathError {
    match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(map_codec_path)?;

    let mut stack: Vec<&str> = Vec::new();
    let parts: Vec<&str> = path[1..].split('/').collect();
    let mut terminal_navigation = false;

    for (i, segment) in parts.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(map_codec_path)?;
        if decoded == "." {
            terminal_navigation = i + 1 == parts.len();
        } else if decoded == ".." {
            stack.pop();
            terminal_navigation = i + 1 == parts.len();
        } else {
            stack.push(segment);
            terminal_navigation = false;
        }
    }

    if terminal_navigation {
        stack.push("");
    }
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn scan_reference(reference: &str) -> Option<usize> {
    global_byte_error(reference)
}

fn construct_url(url: &Url, path: &str, query: Option<&str>, fragment: Option<&str>) -> String {
    let mut out = String::new();
    out.push_str(&url.scheme);
    out.push_str("://");
    if let Some(userinfo) = &url.userinfo {
        out.push_str(userinfo);
        out.push('@');
    }
    if url.ipv6 {
        out.push('[');
        out.push_str(&url.host);
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        out.push(':');
        out.push_str(&port.to_string());
    }
    out.push_str(path);
    if let Some(q) = query {
        out.push('?');
        out.push_str(q);
    }
    if let Some(f) = fragment {
        out.push('#');
        out.push_str(f);
    }
    out
}

fn normalize_url_path(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

fn split_reference_components(input: &str) -> (&str, Option<&str>, Option<&str>) {
    let frag_at = input.find('#');
    let before_frag = frag_at.map(|i| &input[..i]).unwrap_or(input);
    let fragment = frag_at.map(|i| &input[i + 1..]);
    let query_at = before_frag.find('?');
    let path = query_at.map(|i| &before_frag[..i]).unwrap_or(before_frag);
    let query = query_at.map(|i| &before_frag[i + 1..]);
    (path, query, fragment)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let normalized_base_path =
        normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    if let Some(at) = scan_reference(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    if let Some(colon) = reference.find(':') {
        let first_delim = reference
            .find(|c| matches!(c, '/' | '?' | '#'))
            .unwrap_or(reference.len());
        if colon < first_delim {
            let url = parse_url(reference).map_err(ResolveError::Reference)?;
            return normalize_url_path(url);
        }
    }

    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let url = parse_url(&constructed).map_err(ResolveError::Reference)?;
        return normalize_url_path(url);
    }

    let (ref_path, ref_query, ref_fragment) = split_reference_components(reference);
    let selected_path = if ref_path.is_empty() {
        normalized_base_path
    } else if ref_path.starts_with('/') {
        ref_path.to_owned()
    } else {
        let base_dir_end = normalized_base_path.rfind('/').unwrap_or(0);
        let mut merged = normalized_base_path[..=base_dir_end].to_owned();
        merged.push_str(ref_path);
        merged
    };

    let query = if ref_path.is_empty() && ref_query.is_none() {
        base_url.query.as_deref()
    } else {
        ref_query
    };
    let raw = construct_url(&base_url, &selected_path, query, ref_fragment);
    let parsed = parse_url(&raw).map_err(ResolveError::Reference)?;
    normalize_url_path(parsed)
}

fn replace_plus_with_space(s: &str) -> String {
    s.replace('+', " ")
}

fn map_codec_query(e: CodecError, offset: usize) -> QueryError {
    match e {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    }
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut field_start = 0usize;

    for field in input.split('&') {
        let field_len = field.len();
        if !field.is_empty() {
            let eq = field.find('=');
            let (raw_key, raw_value, value_rel) = match eq {
                Some(i) => (&field[..i], &field[i + 1..], i + 1),
                None => (field, "", field.len()),
            };
            let key_input = replace_plus_with_space(raw_key);
            let key = decode_component(Component::Query, &key_input)
                .map_err(|e| map_codec_query(e, field_start))?;
            if key.is_empty() {
                return Err(QueryError::EmptyKey { at: field_start });
            }
            let value_input = replace_plus_with_space(raw_value);
            let value = decode_component(Component::Query, &value_input)
                .map_err(|e| map_codec_query(e, field_start + value_rel))?;

            if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
                values.push(value);
            } else {
                entries.push((key, vec![value]));
            }
        }
        field_start += field_len + 1;
    }

    Ok(QueryMap { entries })
}

fn form_encode(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(char::from(b"0123456789ABCDEF"[(b >> 4) as usize]));
            out.push(char::from(b"0123456789ABCDEF"[(b & 15) as usize]));
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing)) = merged.iter_mut().find(|(k, _)| k == key) {
            existing.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        for value in values {
            fields.push(format!("{}={}", form_encode(&key), form_encode(&value)));
        }
    }
    fields.join("&")
}

fn encode_unreserved(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(char::from(b"0123456789ABCDEF"[(b >> 4) as usize]));
            out.push(char::from(b"0123456789ABCDEF"[(b & 15) as usize]));
        }
    }
    out
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match url.query.as_deref() {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = String::new();
    out.push_str(&url.scheme);
    out.push_str("://");

    if let Some(userinfo) = &url.userinfo {
        let decoded =
            decode_component(Component::UserInfo, userinfo).map_err(|e| {
                CanonicalError::Parse(map_codec_parse(e, 0))
            })?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }

    if url.ipv6 {
        let addr: std::net::Ipv6Addr = url
            .host
            .parse()
            .map_err(|_| CanonicalError::Parse(ParseError::InvalidIpv6 { at: 0 }))?;
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }

    let omit_port = match (url.scheme.as_str(), url.port) {
        ("http", Some(80)) | ("https", Some(443)) => true,
        _ => false,
    };
    if let Some(port) = url.port {
        if !omit_port {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let mut encoded_segments = Vec::new();
    for segment in path.split('/') {
        let decoded =
            decode_component(Component::Path, segment).map_err(|e| {
                CanonicalError::Path(match e {
                    CodecError::BadEscape { at } => PathError::BadEscape { at },
                    CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
                })
            })?;
        encoded_segments.push(encode_unreserved(&decoded));
    }
    out.push_str(&encoded_segments.join("/"));

    if let Some(map) = query_map {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }

    if let Some(fragment) = url.fragment.as_deref() {
        let decoded = decode_component(Component::Fragment, fragment).map_err(|e| {
            CanonicalError::Parse(map_codec_parse(e, 0))
        })?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(out)
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn component_safe(component: Component, b: u8) -> bool {
    b.is_ascii_alphanumeric()
        || matches!(b, b'-' | b'.' | b'_' | b'~')
        || match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => matches!(b, b'/' | b'?'),
            Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
        }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(h), Some(l)) => {
                    decoded.push((h << 4) | l);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let offset = e.valid_up_to();
            Err(CodecError::InvalidUtf8 {
                at: origins.get(offset).copied().unwrap_or(input.len()),
            })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if component_safe(component, b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn check_global_bytes(input: &str) -> Result<(), ParseError> {
    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn parse_port(raw: &str, start: usize) -> Result<u16, ParseError> {
    if raw.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (i, b) in raw.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + i });
        }
    }
    if raw.len() >= 2 && raw.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut n: u32 = 0;
    for b in raw.bytes() {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at: start })?;
        if n > 65535 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }
    if n == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(n as u16)
}

fn map_codec_parse(err: CodecError, base: usize) -> ParseError {
    match err {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    check_global_bytes(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| auth_start + i)
        .unwrap_or(bytes.len());
    let authority = &input[auth_start..auth_end];
    let ats: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(auth_start + i))
        .collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo { at: ats[1] });
    }

    let (userinfo, hostport_start) = if let Some(&at) = ats.first() {
        if at == auth_start {
            return Err(ParseError::InvalidUserInfo { at });
        }
        (Some(input[auth_start..at].to_owned()), at + 1)
    } else {
        (None, auth_start)
    };

    let mut ipv6 = false;
    let (host, port) = if hostport_start < auth_end && bytes[hostport_start] == b'[' {
        let close = bytes[hostport_start + 1..auth_end]
            .iter()
            .position(|&b| b == b']')
            .map(|i| hostport_start + 1 + i);
        let close = match close {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: hostport_start }),
        };
        let interior = &input[hostport_start + 1..close];
        if interior.contains('%') || interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: hostport_start });
        }
        let port = if close + 1 == auth_end {
            None
        } else if bytes[close + 1] == b':' {
            Some(parse_port(&input[close + 2..auth_end], close + 2)?)
        } else {
            return Err(ParseError::InvalidHost { at: close + 1 });
        };
        ipv6 = true;
        (interior.to_ascii_lowercase(), port)
    } else {
        let hp = &input[hostport_start..auth_end];
        let split = hp.find(':');
        let host_end = split.map(|i| hostport_start + i).unwrap_or(auth_end);
        if host_end == hostport_start {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (i, b) in input[hostport_start..host_end].bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_start + i,
                });
            }
        }
        let port = if let Some(i) = split {
            Some(parse_port(&input[hostport_start + i + 1..auth_end], hostport_start + i + 1)?)
        } else {
            None
        };
        (
            input[hostport_start..host_end].to_ascii_lowercase(),
            port,
        )
    };

    let fragment_pos = bytes[auth_end..]
        .iter()
        .position(|&b| b == b'#')
        .map(|i| auth_end + i);
    let before_fragment_end = fragment_pos.unwrap_or(bytes.len());
    let query_pos = bytes[auth_end..before_fragment_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|i| auth_end + i);
    let path_end = query_pos.unwrap_or(before_fragment_end);
    let path = if path_end == auth_end {
        "/".to_owned()
    } else {
        input[auth_end..path_end].to_owned()
    };
    let query = query_pos.map(|q| input[q + 1..before_fragment_end].to_owned());
    let fragment = fragment_pos.map(|f| input[f + 1..].to_owned());

    if let Some(ref raw) = userinfo {
        let start = auth_start;
        decode_component(Component::UserInfo, raw).map_err(|e| map_codec_parse(e, start))?;
    }
    if path != "/" || path_end != auth_end {
        decode_component(Component::Path, &path)
            .map_err(|e| map_codec_parse(e, auth_end))?;
    }
    if let (Some(q), Some(qpos)) = (&query, query_pos) {
        decode_component(Component::Query, q)
            .map_err(|e| map_codec_parse(e, qpos + 1))?;
    }
    if let (Some(f), Some(fpos)) = (&fragment, fragment_pos) {
        decode_component(Component::Fragment, f)
            .map_err(|e| map_codec_parse(e, fpos + 1))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn map_codec_path(err: CodecError) -> PathError {
    match err {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(map_codec_path)?;

    let mut stack: Vec<&str> = Vec::new();
    let segments: Vec<&str> = path[1..].split('/').collect();
    for &segment in &segments {
        let decoded = decode_component(Component::Path, segment).map_err(map_codec_path)?;
        if decoded == "." {
            continue;
        } else if decoded == ".." {
            stack.pop();
        } else {
            stack.push(segment);
        }
    }
    let final_decoded = segments
        .last()
        .map(|s| decode_component(Component::Path, s).unwrap_or_default());
    if matches!(final_decoded.as_deref(), Some(".") | Some("..")) {
        stack.push("");
    }

    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn make_absolute_url(url: &Url, path: &str, query: Option<&str>, fragment: Option<&str>) -> String {
    let mut out = format!("{}://", url.scheme);
    if let Some(u) = &url.userinfo {
        out.push_str(u);
        out.push('@');
    }
    if url.ipv6 {
        out.push('[');
        out.push_str(&url.host);
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        out.push(':');
        out.push_str(&port.to_string());
    }
    out.push_str(path);
    if let Some(q) = query {
        out.push('?');
        out.push_str(q);
    }
    if let Some(f) = fragment {
        out.push('#');
        out.push_str(f);
    }
    out
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    check_global_bytes(reference).map_err(ResolveError::Reference)?;

    let first_delim = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delim].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }
    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let frag_pos = reference.find('#');
    let pre_frag_end = frag_pos.unwrap_or(reference.len());
    let query_pos = reference[..pre_frag_end].find('?');
    let path_end = query_pos.unwrap_or(pre_frag_end);
    let ref_path = &reference[..path_end];
    let ref_query = query_pos.map(|q| &reference[q + 1..pre_frag_end]);
    let ref_fragment = frag_pos.map(|f| &reference[f + 1..]);

    let (selected_path, selected_query) = if ref_path.is_empty() {
        (base_path.clone(), if query_pos.is_none() {
            base_url.query.as_deref()
        } else {
            ref_query
        })
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = base_path.rfind('/').map(|i| &base_path[..=i]).unwrap_or("/");
            format!("{slash}{ref_path}")
        };
        (merged, ref_query)
    };

    let absolute = make_absolute_url(
        &base_url,
        &selected_path,
        selected_query,
        ref_fragment,
    );
    let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

fn form_encode(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn replace_plus(s: &str) -> String {
    s.replace('+', " ")
}

fn map_codec_query(err: CodecError, base: usize) -> QueryError {
    match err {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut field_start = 0;

    for field in input.split('&') {
        let start = field_start;
        field_start += field.len() + 1;
        if field.is_empty() {
            continue;
        }
        let eq = field.find('=');
        let (raw_key, raw_value, value_offset) = match eq {
            Some(i) => (&field[..i], &field[i + 1..], start + i + 1),
            None => (field, "", start + field.len()),
        };

        let key_input = replace_plus(raw_key);
        let key = decode_component(Component::Query, &key_input)
            .map_err(|e| map_codec_query(e, start))?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: start });
        }

        let value_input = replace_plus(raw_value);
        let value = decode_component(Component::Query, &value_input)
            .map_err(|e| map_codec_query(e, value_offset))?;

        if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing)) = merged.iter_mut().find(|(k, _)| k == key) {
            existing.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = form_encode(&key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, form_encode(&value)));
        }
    }
    fields.join("&")
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match &url.query {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(raw) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, raw)
            .map_err(|e| CanonicalError::Parse(map_codec_parse(e, 0)))?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }
    if url.ipv6 {
        let address = url
            .host
            .parse::<std::net::Ipv6Addr>()
            .map_err(|_| CanonicalError::Parse(ParseError::InvalidIpv6 { at: 0 }))?;
        out.push('[');
        out.push_str(&address.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        if !((url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443)) {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let mut canonical_segments = Vec::new();
    for segment in path.split('/') {
        let decoded = decode_component(Component::Path, segment)
            .map_err(|e| CanonicalError::Path(map_codec_path(e)))?;
        canonical_segments.push(encode_component_unreserved(&decoded));
    }
    out.push_str(&canonical_segments.join("/"));

    if let Some(map) = query_map {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }
    if let Some(raw) = &url.fragment {
        let decoded = decode_component(Component::Fragment, raw)
            .map_err(|e| CanonicalError::Parse(map_codec_parse(e, 0)))?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(out)
}

fn encode_component_unreserved(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },

fn global_scan(input: &str) -> Result<(), ParseError> {
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

fn parse_port(port: &str, start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + i });
        }
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut value: u32 = 0;
    for b in port.bytes() {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: start }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

fn map_codec_to_parse(e: CodecError, offset: usize) -> ParseError {
    match e {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    global_scan(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || b == b'+' || b == b'-' || b == b'.') {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..]
        .iter()
        .position(|b| matches!(*b, b'/' | b'?' | b'#'))
        .map(|p| auth_start + p)
        .unwrap_or(input.len());
    let authority = &input[auth_start..auth_end];
    let ats: Vec<usize> = authority.match_indices('@').map(|(i, _)| i).collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo { at: auth_start + ats[1] });
    }
    let (userinfo, hostport, hostport_start) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_start });
        }
        (
            Some(authority[..at].to_string()),
            &authority[at + 1..],
            auth_start + at + 1,
        )
    } else {
        (None, authority, auth_start)
    };

    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let open = hostport_start;
        let close = match hostport.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(i) => i + 1,
            None => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open });
        }
        ipv6 = true;
        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], hostport_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost { at: hostport_start + close + 1 });
        };
        (interior.to_ascii_lowercase(), port)
    } else {
        let split = hostport.find(':');
        let (hostpart, portpart) = match split {
            Some(i) => (&hostport[..i], Some((&hostport[i + 1..], hostport_start + i + 1))),
            None => (hostport, None),
        };
        if hostpart.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (i, b) in hostpart.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: hostport_start + i });
            }
        }
        let port = match portpart {
            Some((p, at)) => Some(parse_port(p, at)?),
            None => None,
        };
        (hostpart.to_ascii_lowercase(), port)
    };

    let hash = bytes[auth_end..].iter().position(|&b| b == b'#').map(|i| auth_end + i);
    let pre_fragment_end = hash.unwrap_or(input.len());
    let question = bytes[auth_end..pre_fragment_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|i| auth_end + i);

    let (path, query) = match question {
        Some(q) => (
            &input[auth_end..q],
            Some(input[q + 1..pre_fragment_end].to_string()),
        ),
        None => (&input[auth_end..pre_fragment_end], None),
    };
    let fragment = hash.map(|h| input[h + 1..].to_string());
    let stored_path = if path.is_empty() { "/".to_string() } else { path.to_string() };

    if let Some(ui) = &userinfo {
        decode_component(Component::UserInfo, ui)
            .map_err(|e| map_codec_to_parse(e, auth_start))?;
    }
    if !path.is_empty() {
        decode_component(Component::Path, path)
            .map_err(|e| map_codec_to_parse(e, auth_end))?;
    }
    if let Some(q) = &query {
        let q_start = question.unwrap() + 1;
        decode_component(Component::Query, q).map_err(|e| map_codec_to_parse(e, q_start))?;
    }
    if let Some(f) = &fragment {
        let f_start = hash.unwrap() + 1;
        decode_component(Component::Fragment, f).map_err(|e| map_codec_to_parse(e, f_start))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path: stored_path,
        query,
        fragment,
        ipv6,
    })
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(h), Some(l)) => {
                    decoded.push(h * 16 + l);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }
    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_string()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: offsets.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::new();
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(hex_digit(b >> 4));
            out.push(hex_digit(b & 0x0f));
        }
    }
    out
}

fn hex_digit(b: u8) -> char {
    b"0123456789ABCDEF"[b as usize] as char
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn codec_to_path(e: CodecError) -> PathError {
    match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(codec_to_path)?;
    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    for (i, raw) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, raw).map_err(codec_to_path)?;
        if decoded == "." {
            if i + 1 == segments.len() {
                stack.push("");
            }
        } else if decoded == ".." {
            if !stack.is_empty() {
                stack.pop();
            }
            if i + 1 == segments.len() {
                stack.push("");
            }
        } else {
            stack.push(raw);
        }
    }
    Ok(format!("/{}", stack.join("/")))
}

fn first_component_end(s: &str) -> usize {
    s.bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(s.len())
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    global_scan(reference).map_err(ResolveError::Reference)?;

    if reference[..first_component_end(reference)].contains(':') {
        let mut u = parse_url(reference).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }
    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut u = parse_url(&constructed).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }

    let hash = reference.find('#');
    let before_fragment = hash.map(|i| &reference[..i]).unwrap_or(reference);
    let fragment = hash.map(|i| reference[i + 1..].to_string());
    let question = before_fragment.find('?');
    let ref_path = question
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let ref_query = question.map(|i| before_fragment[i + 1..].to_string());

    let (selected_path, selected_query) = if ref_path.is_empty() {
        (
            base_path.clone(),
            if question.is_none() {
                base_url.query.clone()
            } else {
                ref_query
            },
        )
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_string()
        } else {
            let slash = base_path.rfind('/').map(|i| i + 1).unwrap_or(0);
            format!("{}{}", &base_path[..slash], ref_path)
        };
        (merged, ref_query)
    };

    let mut raw = format!("{}://", base_url.scheme);
    if let Some(ui) = &base_url.userinfo {
        raw.push_str(ui);
        raw.push('@');
    }
    if base_url.ipv6 {
        raw.push('[');
        raw.push_str(&base_url.host);
        raw.push(']');
    } else {
        raw.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        raw.push(':');
        raw.push_str(&port.to_string());
    }
    raw.push_str(&selected_path);
    if let Some(q) = &selected_query {
        raw.push('?');
        raw.push_str(q);
    }
    if let Some(f) = &fragment {
        raw.push('#');
        raw.push_str(f);
    }
    let mut u = parse_url(&raw).map_err(ResolveError::Reference)?;
    u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
    Ok(u)
}

fn form_decode(s: &str, offset: usize) -> Result<String, QueryError> {
    let replaced = s.replace('+', " ");
    decode_component(Component::Query, &replaced).map_err(|e| match e {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    })
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut field_start = 0;
    for field in input.split('&') {
        let len = field.len();
        if !field.is_empty() {
            let eq = field.find('=');
            let (key_raw, value_raw, value_offset) = match eq {
                Some(i) => (&field[..i], &field[i + 1..], field_start + i + 1),
                None => (field, "", field_start + len),
            };
            let key = form_decode(key_raw, field_start)?;
            if key.is_empty() {
                return Err(QueryError::EmptyKey { at: field_start });
            }
            let value = form_decode(value_raw, value_offset)?;
            if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
                values.push(value);
            } else {
                entries.push((key, vec![value]));
            }
        }
        field_start += len + 1;
    }
    Ok(QueryMap { entries })
}

fn form_encode(s: &str) -> String {
    let mut out = String::new();
    for &b in s.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(hex_digit(b >> 4));
            out.push(hex_digit(b & 0x0f));
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, target)) = merged.iter_mut().find(|(k, _)| k == key) {
            target.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }
    let mut fields = Vec::new();
    for (key, values) in merged {
        for value in values {
            fields.push(format!("{}={}", form_encode(&key), form_encode(&value)));
        }
    }
    fields.join("&")
}

fn encode_unreserved(text: &str) -> String {
    let mut out = String::new();
    for &b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(hex_digit(b >> 4));
            out.push(hex_digit(b & 0x0f));
        }
    }
    out
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query = match &url.query {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(ui) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, ui)
            .map_err(|e| CanonicalError::Parse(map_codec_to_parse(e, 0)))?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }
    if url.ipv6 {
        let addr = Ipv6Addr::from_str(&url.host).unwrap();
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    let omit_port = matches!((url.scheme.as_str(), url.port), ("http", Some(80)) | ("https", Some(443)));
    if let Some(port) = url.port.filter(|_| !omit_port) {
        out.push(':');
        out.push_str(&port.to_string());
    }

    let mut canonical_path_segments = Vec::new();
    for segment in path.split('/') {
        let decoded = decode_component(Component::Path, segment)
            .map_err(|e| CanonicalError::Path(codec_to_path(e)))?;
        canonical_path_segments.push(encode_unreserved(&decoded));
    }
    out.push_str(&canonical_path_segments.join("/"));

    if let Some(q) = query {
        out.push('?');
        out.push_str(&serialise_query(&q));
    }
    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|e| CanonicalError::Parse(map_codec_to_parse(CodecError::InvalidUtf8 { at: 0 }, 0)))?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(out)
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]).ok_or(CodecError::BadEscape { at: i })?;
            let lo = hex_value(bytes[i + 2]).ok_or(CodecError::BadEscape { at: i })?;
            decoded.push((hi << 4) | lo);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = origins.get(e.valid_up_to()).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

fn component_safe(component: Component, b: u8) -> bool {
    b.is_ascii_alphanumeric()
        || matches!(b, b'-' | b'.' | b'_' | b'~')
        || match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => matches!(b, b'/' | b'?'),
            Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
        }
}

pub fn encode_component(component: Component, text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if component_safe(component, b) {

            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn parse_port(raw: &str, at: usize) -> Result<u16, ParseError> {
    if raw.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }
    for (i, b) in raw.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: at + i });
        }
    }
    if raw.len() >= 2 && raw.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at });
    }
    let mut value: u32 = 0;
    for b in raw.bytes() {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((b - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at })?;
        if value > 65535 {
            return Err(ParseError::InvalidPort { at });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(c) => c,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|n| auth_start + n)
        .unwrap_or(bytes.len());
    let authority = &input[auth_start..auth_end];

    let mut at_positions = authority.match_indices('@').map(|(i, _)| i);
    let first_at = at_positions.next();
    if let Some(second) = at_positions.next() {
        return Err(ParseError::InvalidUserInfo { at: auth_start + second });
    }
    let (userinfo, host_start_rel) = match first_at {
        Some(i) if i == 0 => return Err(ParseError::InvalidUserInfo { at: auth_start }),
        Some(i) => (Some(authority[..i].to_owned()), i + 1),
        None => (None, 0),
    };
    let host_start = auth_start + host_start_rel;
    let hostport = &authority[host_start_rel..];

    let (host, port, ipv6) = if hostport.starts_with('[') {
        let open_at = host_start;
        let close_rel = match hostport.find(']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: open_at }),
        };
        let interior = &hostport[1..close_rel];
        if interior.contains('%') || interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: open_at });
        }
        let suffix = &hostport[close_rel + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], host_start + close_rel + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + close_rel + 1,
            });
        };
        (ascii_lower(interior), parsed_port, true)
    } else {
        let split = hostport.find(':');
        let (host_raw, port) = match split {
            Some(i) => {
                let p = parse_port(&hostport[i + 1..], host_start + i + 1)?;
                (&hostport[..i], Some(p))
            }
            None => (hostport, None),
        };
        if host_raw.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (i, b) in host_raw.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: host_start + i });
            }
        }
        (ascii_lower(host_raw), port, false)
    };

    let remainder = &input[auth_end..];
    let (before_fragment, fragment) = match remainder.find('#') {
        Some(i) => (&remainder[..i], Some(remainder[i + 1..].to_owned())),
        None => (remainder, None),
    };
    let (raw_path, query) = match before_fragment.find('?') {
        Some(i) => (
            &before_fragment[..i],
            Some(before_fragment[i + 1..].to_owned()),
        ),
        None => (before_fragment, None),
    };
    let path = if raw_path.is_empty() {
        "/".to_owned()
    } else {
        raw_path.to_owned()
    };

    if let Some(ref u) = userinfo {
        let offset = auth_start;
        decode_component(Component::UserInfo, u).map_err(|e| map_codec_parse(e, offset))?;
    }
    if !raw_path.is_empty() {
        decode_component(Component::Path, raw_path)
            .map_err(|e| map_codec_parse(e, auth_end))?;
    }
    if let Some(q) = query.as_ref() {
        let q_at = auth_end + before_fragment.find('?').unwrap() + 1;
        decode_component(Component::Query, q).map_err(|e| map_codec_parse(e, q_at))?;
    }
    if let Some(f) = fragment.as_ref() {
        let f_at = auth_end + remainder.find('#').unwrap() + 1;
        decode_component(Component::Fragment, f).map_err(|e| map_codec_parse(e, f_at))?;
    }

    Ok(Url {
        scheme: ascii_lower(&input[..colon]),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    let decoded = decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let decoded_segments: Vec<&str> = decoded[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let last_navigation = decoded_segments
        .last()
        .map(|s| *s == "." || *s == "..")
        .unwrap_or(false);

    for (raw, dec) in raw_segments.iter().zip(decoded_segments.iter()) {
        match *dec {
            "." => {}
            ".." => {
                stack.pop();
            }
            _ => stack.push(*raw),
        }
    }
    if last_navigation {
        stack.push("");
    }
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn scan_reference(input: &str) -> Result<(), ParseError> {
    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    scan_reference(reference).map_err(ResolveError::Reference)?;

    let first_delim = reference
        .find(|c| matches!(c, '/' | '?' | '#'))
        .unwrap_or(reference.len());
    if reference[..first_delim].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }
    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let (pre_fragment, fragment) = match reference.find('#') {
        Some(i) => (&reference[..i], Some(reference[i + 1..].to_owned())),
        None => (reference, None),
    };
    let (ref_path, ref_query) = match pre_fragment.find('?') {
        Some(i) => (&pre_fragment[..i], Some(pre_fragment[i + 1..].to_owned())),
        None => (pre_fragment, None),
    };

    let selected_path = if ref_path.is_empty() {
        base_path.clone()
    } else if ref_path.starts_with('/') {
        ref_path.to_owned()
    } else {
        let slash = base_path.rfind('/').unwrap_or(0);
        format!("{}{}", &base_path[..=slash], ref_path)
    };
    let selected_query = if ref_path.is_empty() && ref_query.is_none() {
        base_url.query.clone()
    } else {
        ref_query
    };

    let host = if base_url.ipv6 {
        format!("[{}]", base_url.host)
    } else {
        base_url.host.clone()
    };
    let mut constructed = format!("{}://", base_url.scheme);
    if let Some(u) = &base_url.userinfo {
        constructed.push_str(u);
        constructed.push('@');
    }
    constructed.push_str(&host);
    if let Some(p) = base_url.port {
        constructed.push(':');
        constructed.push_str(&p.to_string());
    }
    constructed.push_str(&selected_path);
    if let Some(q) = &selected_query {
        constructed.push('?');
        constructed.push_str(q);
    }
    if let Some(f) = &fragment {
        constructed.push('#');
        constructed.push_str(f);
    }
    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

fn map_codec_query(e: CodecError, offset: usize) -> QueryError {
    match e {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: at + offset },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: at + offset },
    }
}

fn form_plus_to_space(s: &str) -> String {
    s.replace('+', " ")
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut offset = 0;

    for field in input.split('&') {
        let field_start = offset;
        offset += field.len() + 1;
        if field.is_empty() {
            continue;
        }
        let (raw_key, raw_value, value_offset) = match field.find('=') {
            Some(i) => (&field[..i], &field[i + 1..], field_start + i + 1),
            None => (field, "", field_start + field.len()),
        };
        let key_pre = form_plus_to_space(raw_key);
        let key = decode_component(Component::Query, &key_pre)
            .map_err(|e| map_codec_query(e, field_start))?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }
        let value_pre = form_plus_to_space(raw_value);
        let value = decode_component(Component::Query, &value_pre)
            .map_err(|e| map_codec_query(e, value_offset))?;

        if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }
    Ok(QueryMap { entries })
}

fn form_encode(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::new();
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0xf) as usize] as char);
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing)) = merged.iter_mut().find(|(k, _)| k == key) {
            existing.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = form_encode(&key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, form_encode(&value)));
        }
    }
    fields.join("&")
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let mut url = parse_url(input).map_err(CanonicalError::Parse)?;
    url.path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match url.query.as_deref() {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(raw_userinfo) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, raw_userinfo)
            .map_err(|e| CanonicalError::Parse(map_codec_parse(e, input.find('@').unwrap_or(0))))?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }
    if url.ipv6 {
        let addr = url.host.parse::<std::net::Ipv6Addr>().map_err(|_| {
            CanonicalError::Parse(ParseError::InvalidIpv6 { at: 0 })
        })?;
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        if !((url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443)) {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let mut encoded_segments = Vec::new();
    for segment in url.path.split('/') {
        let decoded = decode_component(Component::Path, segment)
            .map_err(|e| CanonicalError::Path(match e {
                CodecError::BadEscape { at } => PathError::BadEscape { at },
                CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
            }))?;
        encoded_segments.push(encode_unreserved(&decoded));
    }
    out.push_str(&encoded_segments.join("/"));

    if let Some(map) = query_map {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }
    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|e| CanonicalError::Parse(map_codec_parse(e, input.find('#').unwrap_or(0) + 1)))?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(out)
}

fn encode_unreserved(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::new();
    for &b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0xf) as usize] as char);
        }
    }
    out
}
