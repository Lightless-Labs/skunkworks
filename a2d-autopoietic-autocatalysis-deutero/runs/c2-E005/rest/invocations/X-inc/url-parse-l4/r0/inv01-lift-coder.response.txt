fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn component_safe(component: Component, b: u8) -> bool {
    b.is_ascii_alphanumeric()
        || matches!(b, b'-' | b'.' | b'_' | b'~')
        || match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => matches!(b, b'/' | b'?'),
            Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
        }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(h), Some(l)) => {
                    decoded.push((h << 4) | l);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let offset = e.valid_up_to();
            Err(CodecError::InvalidUtf8 {
                at: origins.get(offset).copied().unwrap_or(input.len()),
            })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if component_safe(component, b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn check_global_bytes(input: &str) -> Result<(), ParseError> {
    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn parse_port(raw: &str, start: usize) -> Result<u16, ParseError> {
    if raw.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (i, b) in raw.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + i });
        }
    }
    if raw.len() >= 2 && raw.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut n: u32 = 0;
    for b in raw.bytes() {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at: start })?;
        if n > 65535 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }
    if n == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(n as u16)
}

fn map_codec_parse(err: CodecError, base: usize) -> ParseError {
    match err {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    check_global_bytes(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| auth_start + i)
        .unwrap_or(bytes.len());
    let authority = &input[auth_start..auth_end];
    let ats: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(auth_start + i))
        .collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo { at: ats[1] });
    }

    let (userinfo, hostport_start) = if let Some(&at) = ats.first() {
        if at == auth_start {
            return Err(ParseError::InvalidUserInfo { at });
        }
        (Some(input[auth_start..at].to_owned()), at + 1)
    } else {
        (None, auth_start)
    };

    let mut ipv6 = false;
    let (host, port) = if hostport_start < auth_end && bytes[hostport_start] == b'[' {
        let close = bytes[hostport_start + 1..auth_end]
            .iter()
            .position(|&b| b == b']')
            .map(|i| hostport_start + 1 + i);
        let close = match close {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: hostport_start }),
        };
        let interior = &input[hostport_start + 1..close];
        if interior.contains('%') || interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: hostport_start });
        }
        let port = if close + 1 == auth_end {
            None
        } else if bytes[close + 1] == b':' {
            Some(parse_port(&input[close + 2..auth_end], close + 2)?)
        } else {
            return Err(ParseError::InvalidHost { at: close + 1 });
        };
        ipv6 = true;
        (interior.to_ascii_lowercase(), port)
    } else {
        let hp = &input[hostport_start..auth_end];
        let split = hp.find(':');
        let host_end = split.map(|i| hostport_start + i).unwrap_or(auth_end);
        if host_end == hostport_start {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (i, b) in input[hostport_start..host_end].bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_start + i,
                });
            }
        }
        let port = if let Some(i) = split {
            Some(parse_port(&input[hostport_start + i + 1..auth_end], hostport_start + i + 1)?)
        } else {
            None
        };
        (
            input[hostport_start..host_end].to_ascii_lowercase(),
            port,
        )
    };

    let fragment_pos = bytes[auth_end..]
        .iter()
        .position(|&b| b == b'#')
        .map(|i| auth_end + i);
    let before_fragment_end = fragment_pos.unwrap_or(bytes.len());
    let query_pos = bytes[auth_end..before_fragment_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|i| auth_end + i);
    let path_end = query_pos.unwrap_or(before_fragment_end);
    let path = if path_end == auth_end {
        "/".to_owned()
    } else {
        input[auth_end..path_end].to_owned()
    };
    let query = query_pos.map(|q| input[q + 1..before_fragment_end].to_owned());
    let fragment = fragment_pos.map(|f| input[f + 1..].to_owned());

    if let Some(ref raw) = userinfo {
        let start = auth_start;
        decode_component(Component::UserInfo, raw).map_err(|e| map_codec_parse(e, start))?;
    }
    if path != "/" || path_end != auth_end {
        decode_component(Component::Path, &path)
            .map_err(|e| map_codec_parse(e, auth_end))?;
    }
    if let (Some(q), Some(qpos)) = (&query, query_pos) {
        decode_component(Component::Query, q)
            .map_err(|e| map_codec_parse(e, qpos + 1))?;
    }
    if let (Some(f), Some(fpos)) = (&fragment, fragment_pos) {
        decode_component(Component::Fragment, f)
            .map_err(|e| map_codec_parse(e, fpos + 1))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn map_codec_path(err: CodecError) -> PathError {
    match err {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(map_codec_path)?;

    let mut stack: Vec<&str> = Vec::new();
    let segments: Vec<&str> = path[1..].split('/').collect();
    for &segment in &segments {
        let decoded = decode_component(Component::Path, segment).map_err(map_codec_path)?;
        if decoded == "." {
            continue;
        } else if decoded == ".." {
            stack.pop();
        } else {
            stack.push(segment);
        }
    }
    let final_decoded = segments
        .last()
        .map(|s| decode_component(Component::Path, s).unwrap_or_default());
    if matches!(final_decoded.as_deref(), Some(".") | Some("..")) {
        stack.push("");
    }

    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn make_absolute_url(url: &Url, path: &str, query: Option<&str>, fragment: Option<&str>) -> String {
    let mut out = format!("{}://", url.scheme);
    if let Some(u) = &url.userinfo {
        out.push_str(u);
        out.push('@');
    }
    if url.ipv6 {
        out.push('[');
        out.push_str(&url.host);
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        out.push(':');
        out.push_str(&port.to_string());
    }
    out.push_str(path);
    if let Some(q) = query {
        out.push('?');
        out.push_str(q);
    }
    if let Some(f) = fragment {
        out.push('#');
        out.push_str(f);
    }
    out
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    check_global_bytes(reference).map_err(ResolveError::Reference)?;

    let first_delim = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delim].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }
    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let frag_pos = reference.find('#');
    let pre_frag_end = frag_pos.unwrap_or(reference.len());
    let query_pos = reference[..pre_frag_end].find('?');
    let path_end = query_pos.unwrap_or(pre_frag_end);
    let ref_path = &reference[..path_end];
    let ref_query = query_pos.map(|q| &reference[q + 1..pre_frag_end]);
    let ref_fragment = frag_pos.map(|f| &reference[f + 1..]);

    let (selected_path, selected_query) = if ref_path.is_empty() {
        (base_path.clone(), if query_pos.is_none() {
            base_url.query.as_deref()
        } else {
            ref_query
        })
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = base_path.rfind('/').map(|i| &base_path[..=i]).unwrap_or("/");
            format!("{slash}{ref_path}")
        };
        (merged, ref_query)
    };

    let absolute = make_absolute_url(
        &base_url,
        &selected_path,
        selected_query,
        ref_fragment,
    );
    let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

fn form_encode(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn replace_plus(s: &str) -> String {
    s.replace('+', " ")
}

fn map_codec_query(err: CodecError, base: usize) -> QueryError {
    match err {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut field_start = 0;

    for field in input.split('&') {
        let start = field_start;
        field_start += field.len() + 1;
        if field.is_empty() {
            continue;
        }
        let eq = field.find('=');
        let (raw_key, raw_value, value_offset) = match eq {
            Some(i) => (&field[..i], &field[i + 1..], start + i + 1),
            None => (field, "", start + field.len()),
        };

        let key_input = replace_plus(raw_key);
        let key = decode_component(Component::Query, &key_input)
            .map_err(|e| map_codec_query(e, start))?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: start });
        }

        let value_input = replace_plus(raw_value);
        let value = decode_component(Component::Query, &value_input)
            .map_err(|e| map_codec_query(e, value_offset))?;

        if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing)) = merged.iter_mut().find(|(k, _)| k == key) {
            existing.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = form_encode(&key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, form_encode(&value)));
        }
    }
    fields.join("&")
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match &url.query {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(raw) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, raw)
            .map_err(|e| CanonicalError::Parse(map_codec_parse(e, 0)))?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }
    if url.ipv6 {
        let address = url
            .host
            .parse::<std::net::Ipv6Addr>()
            .map_err(|_| CanonicalError::Parse(ParseError::InvalidIpv6 { at: 0 }))?;
        out.push('[');
        out.push_str(&address.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        if !((url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443)) {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let mut canonical_segments = Vec::new();
    for segment in path.split('/') {
        let decoded = decode_component(Component::Path, segment)
            .map_err(|e| CanonicalError::Path(map_codec_path(e)))?;
        canonical_segments.push(encode_component_unreserved(&decoded));
    }
    out.push_str(&canonical_segments.join("/"));

    if let Some(map) = query_map {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }
    if let Some(raw) = &url.fragment {
        let decoded = decode_component(Component::Fragment, raw)
            .map_err(|e| CanonicalError::Parse(map_codec_parse(e, 0)))?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(out)
}

fn encode_component_unreserved(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}
