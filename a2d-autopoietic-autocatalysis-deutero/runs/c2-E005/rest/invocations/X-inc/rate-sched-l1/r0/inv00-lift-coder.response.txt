```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u64,
    jobs: Vec<StoredJob>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum State {
    Queued,
    Cancelled,
    Completed,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct StoredJob {
    job: Job,
    sequence: u64,
    state: State,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            jobs: Vec::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        self.check_time(now)?;

        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|stored| stored.job.id == job.id) {
            return Err(Error::Duplicate);
        }

        self.jobs.push(StoredJob {
            job,
            sequence: self.next_sequence,
            state: State::Queued,
        });
        self.next_sequence += 1;
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        let Some(index) = self.jobs.iter().position(|stored| stored.job.id == id) else {
            return Err(Error::Unknown);
        };

        match self.jobs[index].state {
            State::Queued => self.jobs[index].state = State::Cancelled,
            State::Cancelled => return Err(Error::Cancelled),
            State::Completed => return Err(Error::Completed),
        }

        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        self.check_time(now)?;

        let selected = self
            .jobs
            .iter()
            .enumerate()
            .filter(|(_, stored)| {
                stored.state == State::Queued && stored.job.ready_at <= now
            })
            .max_by(|(_, a), (_, b)| {
                let a_priority = effective_priority(&a.job, now, self.aging_ms);
                let b_priority = effective_priority(&b.job, now, self.aging_ms);
                a_priority
                    .cmp(&b_priority)
                    .then_with(|| b.sequence.cmp(&a.sequence))
            })
            .map(|(index, _)| index);

        self.last_time = now;

        let Some(index) = selected else {
            return Ok(None);
        };

        self.jobs[index].state = State::Completed;
        Ok(Some(Dispatch {
            id: self.jobs[index].job.id.clone(),
            effective_priority: effective_priority(&self.jobs[index].job, now, self.aging_ms),
        }))
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes
            .iter()
            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-')
}

fn effective_priority(job: &Job, now: u64, aging_ms: u64) -> u8 {
    if aging_ms == 0 {
        return job.priority;
    }

    let age_steps = (now - job.ready_at) / aging_ms;
    let room = (9 - job.priority) as u64;
    job.priority + age_steps.min(room) as u8
}
```
