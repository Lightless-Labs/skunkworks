#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum JobState {
    Queued,
    Cancelled,
    Completed,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct StoredJob {
    ready_at: u64,
    priority: u8,
    sequence: u64,
    state: JobState,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u64,
    jobs: std::collections::HashMap<String, StoredJob>,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            jobs: std::collections::HashMap::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.contains_key(&job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        self.next_sequence = self.next_sequence.saturating_add(1);
        self.jobs.insert(
            job.id,
            StoredJob {
                ready_at: job.ready_at,
                priority: job.priority,
                sequence,
                state: JobState::Queued,
            },
        );
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }

        let Some(job) = self.jobs.get(id) else {
            return Err(Error::Unknown);
        };
        match job.state {
            JobState::Queued => {}
            JobState::Cancelled => return Err(Error::Cancelled),
            JobState::Completed => return Err(Error::Completed),
        }

        self.jobs.get_mut(id).unwrap().state = JobState::Cancelled;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }

        let selected = self
            .jobs
            .iter()
            .filter(|(_, job)| job.state == JobState::Queued && job.ready_at <= now)
            .map(|(id, job)| {
                (
                    id.clone(),
                    effective_priority(job.priority, job.ready_at, now, self.aging_ms),
                    job.sequence,
                )
            })
            .max_by(|a, b| a.1.cmp(&b.1).then_with(|| b.2.cmp(&a.2)));

        self.last_time = now;
        let Some((id, effective_priority, _)) = selected else {
            return Ok(None);
        };

        self.jobs.get_mut(&id).unwrap().state = JobState::Completed;
        Ok(Some(Dispatch {
            id,
            effective_priority,
        }))
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes
            .iter()
            .all(|&b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
}

fn effective_priority(priority: u8, ready_at: u64, now: u64, aging_ms: u64) -> u8 {
    if aging_ms == 0 {
        return priority;
    }

    let age_steps = (now - ready_at) / aging_ms;
    let room = (9 - priority) as u64;
    if age_steps >= room {
        9
    } else {
        priority + age_steps as u8
    }
}
