#[derive(Debug, Clone, PartialEq, Eq)]
enum State {
    Queued,
    Cancelled,
    Completed,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct StoredJob {
    job: Job,
    sequence: u64,
    state: State,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            jobs: Vec::new(),
        }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        self.check_time(now)?;

        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|stored| stored.job.id == job.id) {
            return Err(Error::Duplicate);
        }

        self.jobs.push(StoredJob {
            job,
            sequence: self.next_sequence,
            state: State::Queued,
        });
        self.next_sequence += 1;
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        let Some(index) = self.jobs.iter().position(|stored| stored.job.id == id) else {
            return Err(Error::Unknown);
        };

        match self.jobs[index].state {
            State::Queued => self.jobs[index].state = State::Cancelled,
            State::Cancelled => return Err(Error::Cancelled),
            State::Completed => return Err(Error::Completed),
        }

        self.last_time = now;
        Ok(())
    }

        self.check_time(now)?;

        let selected = self
            .jobs
            .iter()

            .filter(|(_, stored)| {
                stored.state == State::Queued && stored.job.ready_at <= now
            })
            .max_by(|(_, a), (_, b)| {
                let a_priority = effective_priority(&a.job, now, self.aging_ms);
                let b_priority = effective_priority(&b.job, now, self.aging_ms);
                a_priority
                    .cmp(&b_priority)
                    .then_with(|| b.sequence.cmp(&a.sequence))
            })
            .map(|(index, _)| index);

        self.last_time = now;

        let Some(index) = selected else {
            return Ok(None);
        };

        self.jobs[index].state = State::Completed;
        Ok(Some(Dispatch {
            id: self.jobs[index].job.id.clone(),
            effective_priority: effective_priority(&self.jobs[index].job, now, self.aging_ms),

    }

        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }
}

    Completed,

#[derive(Debug, Clone, PartialEq, Eq)]

    Cancelled,
    Completed,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct StoredJob {
    ready_at: u64,
    priority: u8,
    sequence: u64,
    state: JobState,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u64,
    jobs: std::collections::HashMap<String, StoredJob>,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            jobs: std::collections::HashMap::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {

            return Err(Error::TimeReversed);

        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.contains_key(&job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        self.next_sequence = self.next_sequence.saturating_add(1);
        self.jobs.insert(
            job.id,
            StoredJob {
                ready_at: job.ready_at,
                priority: job.priority,
                sequence,
                state: JobState::Queued,
            },
        );
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }

        let Some(job) = self.jobs.get(id) else {
            return Err(Error::Unknown);
        };
        match job.state {
            JobState::Queued => {}
            JobState::Cancelled => return Err(Error::Cancelled),
            JobState::Completed => return Err(Error::Completed),
        }

        self.jobs.get_mut(id).unwrap().state = JobState::Cancelled;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }

        let selected = self
            .jobs
            .iter()
            .filter(|(_, job)| job.state == JobState::Queued && job.ready_at <= now)
            .map(|(id, job)| {
                (
                    id.clone(),
                    effective_priority(job.priority, job.ready_at, now, self.aging_ms),
                    job.sequence,
                )
            })
            .max_by(|a, b| a.1.cmp(&b.1).then_with(|| b.2.cmp(&a.2)));

        self.last_time = now;
        let Some((id, effective_priority, _)) = selected else {
            return Ok(None);
        };

        self.jobs.get_mut(&id).unwrap().state = JobState::Completed;
        Ok(Some(Dispatch {
            id,
            effective_priority,
        }))
    }
}
