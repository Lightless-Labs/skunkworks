#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
    AlignmentCount { expected: usize, actual: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let column_count = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != column_count {
            return Err(TableError::Ragged {
                row,
                expected: column_count,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != column_count {
        return Err(TableError::AlignmentCount {
            expected: column_count,
            actual: options.alignments.len(),
        });
    }

    let mut trimmed_rows = Vec::with_capacity(rows.len());
    let mut widths = vec![1usize; column_count];

    for (row_index, row) in rows.iter().enumerate() {
        let mut trimmed_row = Vec::with_capacity(column_count);

        for (column_index, cell) in row.iter().enumerate() {
            for (byte_index, ch) in cell.char_indices() {
                if !(('\u{20}'..='\u{7e}').contains(&ch)) {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte: byte_index,
                    });
                }
            }

            let trimmed = cell.trim_end_matches(' ').to_owned();
            widths[column_index] = widths[column_index].max(trimmed.len());
            trimmed_row.push(trimmed);
        }

        trimmed_rows.push(trimmed_row);
    }

    for (column, &width) in widths.iter().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let mut output = String::new();
    append_border(&mut output, &widths);

    for (row_index, row) in trimmed_rows.iter().enumerate() {
        output.push('|');

        for (column_index, cell) in row.iter().enumerate() {
            let width = widths[column_index];
            let extra = width - cell.len();
            let alignment = options
                .alignments
                .get(column_index)
                .unwrap_or(&Alignment::Left);

            let (left, right) = match alignment {
                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),
            };

            output.push(' ');
            output.extend(std::iter::repeat_n(' ', left));
            output.push_str(cell);
            output.extend(std::iter::repeat_n(' ', right));
            output.push_str(" |");
        }

        output.push('\n');

        if options.header && row_index == 0 && trimmed_rows.len() > 1 {
            append_border(&mut output, &widths);
        }
    }

    append_border(&mut output, &widths);
    Ok(output)
}

fn append_border(output: &mut String, widths: &[usize]) {
    output.push('+');
    for &width in widths {
        output.extend(std::iter::repeat_n('-', width + 2));
        output.push('+');
    }
    output.push('\n');
}
