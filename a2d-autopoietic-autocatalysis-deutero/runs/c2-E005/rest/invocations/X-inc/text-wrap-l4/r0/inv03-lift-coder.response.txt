#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Layout {
    cost: u64,
    marker_lines: usize,
    consumed: Vec<usize>,
    lines: Vec<String>,
    pieces: Vec<usize>,
    gaps: Vec<usize>,
}

#[derive(Clone)]
struct Candidate {
    next_word: usize,
    next_offset: usize,
    marker_end: bool,
    raw: bool,
    displayed: String,
    consumed: usize,
    pieces: usize,
    gaps: usize,
    marker_lines: usize,
    extra_cost: u64,
}

fn better(a: &Layout, b: &Layout) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    for (x, y) in a.consumed.iter().zip(&b.consumed) {
        if x != y {
            return x > y;
        }
    }
    for (x, y) in a.lines.iter().zip(&b.lines) {
        if x != y {
            return x < y;
        }
    }
    false
}

fn solve(
    words: &[Word],
    wi: usize,
    offset: usize,
    first: bool,
    options: &WrapOptions,
    memo: &mut std::collections::HashMap<(usize, usize, bool), Layout>,
) -> Layout {
    if wi == words.len() {
        return Layout {
            cost: 0,
            marker_lines: 0,
            consumed: Vec::new(),
            lines: Vec::new(),
            pieces: Vec::new(),
            gaps: Vec::new(),
        };
    }
    if let Some(result) = memo.get(&(wi, offset, first)) {
        return result.clone();
    }

    let indent = if first {
        options.first_indent
    } else {
        options.rest_indent
    };
    let capacity = options.width - indent;
    let mut candidates = Vec::new();
    let mut line_text = String::new();
    let mut pieces = 0usize;
    let mut gaps = 0usize;
    let mut consumed = 0usize;

    for word_index in wi..words.len() {
        let start = if word_index == wi { offset } else { 0 };
        if word_index > wi {
            line_text.push(' ');
            gaps += 1;
        }
        let word = &words[word_index];
        let remaining = word.chars.len() - start;

        for end in start + 1..word.chars.len() {
            if !word.markers.contains(&end) {
                continue;
            }
            let needed = (end - start) + 1 + gaps + pieces * 0;
            if needed > capacity {
                continue;
            }
            let mut display = line_text.clone();
            display.extend(word.chars[start..end].iter());
            display.push('-');
            let piece_count = pieces + 1;
            let used = (end - start) + consumed;
            candidates.push(Candidate {
                next_word: word_index,
                next_offset: end,
                marker_end: true,
                raw: false,
                displayed: display,
                consumed: used,
                pieces: piece_count,
                gaps,
                marker_lines: 1,
                extra_cost: options.hyphen_cost as u64,
            });
        }

        if (remaining + gaps) <= capacity {
            line_text.extend(word.chars[start..].iter());
            consumed += remaining;
            pieces += 1;
            candidates.push(Candidate {
                next_word: word_index + 1,
                next_offset: 0,
                marker_end: false,
                raw: false,
                displayed: line_text.clone(),
                consumed,
                pieces,
                gaps,
                marker_lines: 0,
                extra_cost: 0,
            });
            continue;
        }

        if word_index == wi && pieces == 0 && remaining > capacity {
            let marker_fits = word.markers.iter().any(|&m| {
                m > start && m < word.chars.len() && (m - start + 1) <= capacity
            });
            if !marker_fits {
                let mut display = String::new();
                display.extend(word.chars[start..start + capacity].iter());
                candidates.push(Candidate {
                    next_word: word_index,
                    next_offset: start + capacity,
                    marker_end: false,
                    raw: true,
                    displayed: display,
                    consumed: capacity,
                    pieces: 1,
                    gaps: 0,
                    marker_lines: 0,
                    extra_cost: (2 * options.width + 1) as u64,
                });
            }
        }
        break;
    }

    let mut best: Option<Layout> = None;
    for candidate in candidates {
        let is_final = candidate.next_word == words.len();
        let gap = options.width - indent - candidate.displayed.chars().count();
        let mut line_cost = if is_final {
            (gap * gap / 4) as u64
        } else {
            (gap * gap) as u64
        };
        line_cost += candidate.extra_cost;
        if !is_final && candidate.pieces == 1 {
            line_cost += 3;
        }

        let suffix = solve(
            words,
            candidate.next_word,
            candidate.next_offset,
            false,
            options,
            memo,
        );
        let mut full_line = " ".repeat(indent);
        full_line.push_str(&candidate.displayed);

        let mut result = suffix;
        result.cost += line_cost;
        result.marker_lines += candidate.marker_lines;
        result.consumed.insert(0, candidate.consumed);
        result.lines.insert(0, full_line);
        result.pieces.insert(0, candidate.pieces);
        result.gaps.insert(0, candidate.gaps);

        if best.as_ref().map_or(true, |current| better(&result, current)) {
            best = Some(result);
        }
    }

    let result = best.expect("raw splitting guarantees a legal layout");
    memo.insert((wi, offset, first), result.clone());
    result
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn layout_paragraph(text: &[char], options: &WrapOptions) -> Vec<String> {
    let mut words = Vec::new();
    let mut current = Word {
        chars: Vec::new(),
        markers: Vec::new(),
    };

    for &c in text {
        if is_separator(c) {
            if !current.chars.is_empty() {
                words.push(current);
                current = Word {
                    chars: Vec::new(),
                    markers: Vec::new(),
                };
            }
        } else if c == '~' {
            current.markers.push(current.chars.len());
        } else {
            current.chars.push(c);
        }
    }
    if !current.chars.is_empty() {
        words.push(current);
    }
    if words.is_empty() {
        return Vec::new();
    }

    let mut memo = std::collections::HashMap::new();
    let mut result = solve(&words, 0, 0, true, options, &mut memo);
    if options.justify {
        for i in 0..result.lines.len() {
            if i + 1 == result.lines.len() || result.pieces[i] < 2 {
                continue;
            }
            let indent = if i == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            let current_width = result.lines[i].chars().count();
            let extra = options.width - current_width;
            let gaps = result.gaps[i];
            if gaps == 0 {
                continue;
            }
            let each = extra / gaps;
            let remainder = extra % gaps;
            let start = i % gaps;
            let mut rebuilt = String::new();
            let mut gap_index = 0usize;
            let mut chars = result.lines[i].chars().peekable();
            while let Some(c) = chars.next() {
                rebuilt.push(c);
                if c == ' ' && rebuilt.chars().count() > indent {
                    let bonus = if (gap_index + gaps - start) % gaps < remainder {
                        1
                    } else {
                        0
                    };
                    rebuilt.push_str(&" ".repeat(each + bonus));
                    gap_index += 1;
                }
            }
            result.lines[i] = rebuilt;
        }
    }
    result.lines
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    if original.len() > 512 {
        return Err(WrapError::InputTooLong {
            chars: original.len(),
        });
    }
    for &(index, c) in &original {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }
    for i in 0..original.len() {
        if original[i].1 != '~' {
            continue;
        }
        let valid_side = |j: isize| -> bool {
            if j < 0 || j as usize >= original.len() {
                return false;
            }
            let c = original[j as usize].1;
            c != '~' && !is_separator(c)
        };
        if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
            return Err(WrapError::InvalidMarker {
                index: original[i].0,
            });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < original.len() {
        let c = original[i].1;
        if c == '\r' {
            normalized.push('\n');
            if i + 1 < original.len() && original[i + 1].1 == '\n' {
                i += 2;
                continue;
            }
        } else {
            normalized.push(c);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut j = 0;
    while j < normalized.len() {
        if normalized[j] == '\n' {
            let mut end = j;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            let run = end - j;
            if run >= 2 {
                output.extend(layout_paragraph(&segment, options));
                segment.clear();
                output.extend(std::iter::repeat(String::new()).take(run - 1));
            } else {
                segment.push('\n');
            }
            j = end;
        } else {
            segment.push(normalized[j]);
            j += 1;
        }
    }
    output.extend(layout_paragraph(&segment, options));
    Ok(output)
}
