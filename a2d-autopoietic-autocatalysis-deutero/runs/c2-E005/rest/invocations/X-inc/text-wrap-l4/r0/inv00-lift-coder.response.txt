#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    text: String,
    gap: usize,
    marker_ended: bool,
    raw: bool,
    piece_count: usize,
    consumed: usize,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_count: usize,
    lines: Vec<Line>,
    consumed: Vec<usize>,
    texts: Vec<String>,
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_count != b.marker_count {
        return a.marker_count < b.marker_count;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    for (x, y) in a.consumed.iter().zip(&b.consumed) {
        if x != y {
            return x > y;
        }
    }
    for (x, y) in a.texts.iter().zip(&b.texts) {
        if x != y {
            return x < y;
        }
    }
    false
}

fn parse_words(s: &str) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for ch in s.chars() {
        if matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00a0}') {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if ch == '~' {
            markers.push(chars.len());
        } else {
            chars.push(ch);
        }
    }

    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }
    words
}

fn render_base(indent: usize, pieces: &[String]) -> String {
    let mut out = " ".repeat(indent);
    for (i, piece) in pieces.iter().enumerate() {
        if i > 0 {
            out.push(' ');
        }
        out.push_str(piece);
    }
    out
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    let n = words.len();
    let mut memo: Vec<Vec<[Option<Solution>; 2]>> =
        vec![vec![[None, None]; n + 1]; 1];

    fn dp(
        wi: usize,
        offset: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut Vec<Vec<[Option<Solution>; 2]>>,
    ) -> Solution {
        if wi == words.len() {
            return Solution {
                cost: 0,
                marker_count: 0,
                lines: Vec::new(),
                consumed: Vec::new(),
                texts: Vec::new(),
            };
        }

        // Offsets are bounded by the current word length; the memo is sized lazily.
        if memo[0].len() <= wi {
            memo[0].resize_with(wi + 1, || [None, None]);
        }
        let slot = usize::from(!first);
        if offset == 0 {
            if let Some(cached) = &memo[0][wi][slot] {
                return cached.clone();
            }
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let capacity = options.width - indent;
        let word = &words[wi];
        let len = word.chars.len();
        let remaining = len - offset;
        let mut candidates = Vec::new();

        // Enumerate lines that finish at a word boundary, optionally including later words.
        let mut pieces = Vec::new();
        let mut used = 0usize;
        let mut idx = wi;
        let mut start = offset;
        while idx < words.len() {
            let w = &words[idx];
            let suffix_len = w.chars.len() - start;
            let sep = usize::from(!pieces.is_empty());
            if used + sep + suffix_len > capacity {
                break;
            }
            let piece: String = w.chars[start..].iter().collect();
            used += sep + suffix_len;
            pieces.push(piece);
            let next = dp(idx + 1, 0, false, words, options, memo);
            let base = render_base(indent, &pieces);
            let gap = options.width - (indent + used);
            let mut line = Line {
                indent,
                pieces: pieces.clone(),
                text: base,
                gap,
                marker_ended: false,
                raw: false,
                piece_count: pieces.len(),
                consumed: used - (pieces.len() - 1),
            };
            let line_cost = if next.lines.is_empty() {
                (gap * gap / 4) as u64
            } else {
                (gap * gap) as u64
            } + if !next.lines.is_empty() && pieces.len() == 1 {
                3
            } else {
                0
            };
            let mut sol = next;
            sol.cost += line_cost;
            sol.lines.insert(0, line.clone());
            sol.consumed.insert(0, line.consumed);
            sol.texts.insert(0, line.text.clone());
            candidates.push(sol);
            idx += 1;
            start = 0;
        }

        // Marker-ended line within the starting word.
        for &mark in &word.markers {
            if mark <= offset || mark >= len {
                continue;
            }
            let prefix_len = mark - offset;
            if prefix_len + 1 > capacity {
                continue;
            }
            let mut piece: String = word.chars[offset..mark].iter().collect();
            piece.push('-');
            let mut line_text = " ".repeat(indent);
            line_text.push_str(&piece);
            let next = dp(wi, mark, false, words, options, memo);
            let gap = options.width - (indent + prefix_len + 1);
            let line_cost = if next.lines.is_empty() {
                (gap * gap / 4) as u64
            } else {
                (gap * gap) as u64
            } + options.hyphen_cost as u64
                + if !next.lines.is_empty() { 3 } else { 0 };
            let mut sol = next;
            sol.cost += line_cost;
            sol.marker_count += 1;
            sol.lines.insert(
                0,
                Line {
                    indent,
                    pieces: vec![piece],
                    text: line_text,
                    gap,
                    marker_ended: true,
                    raw: false,
                    piece_count: 1,
                    consumed: prefix_len,
                },
            );
            sol.consumed.insert(0, prefix_len);
            sol.texts.insert(0, sol.lines[0].text.clone());
            candidates.push(sol);
        }

        // Raw split is available only on an otherwise empty line, and only if no marker fits.
        let marker_fits = word.markers.iter().any(|&mark| {
            mark > offset && mark < len && mark - offset + 1 <= capacity
        });
        if remaining > capacity && !marker_fits {
            let piece: String = word.chars[offset..offset + capacity].iter().collect();
            let mut line_text = " ".repeat(indent);
            line_text.push_str(&piece);
            let mut next_offset = offset + capacity;
            while word.markers.iter().any(|&m| m > offset && m <= next_offset) {
                // Markers passed by a raw split are discarded.
                break;
            }
            let next = if next_offset == len {
                dp(wi + 1, 0, false, words, options, memo)
            } else {
                dp(wi, next_offset, false, words, options, memo)
            };
            let gap = options.width - (indent + capacity);
            let line_cost = if next.lines.is_empty() {
                (gap * gap / 4) as u64
            } else {
                (gap * gap) as u64
            } + (2 * options.width + 1) as u64
                + if !next.lines.is_empty() { 3 } else { 0 };
            let mut sol = next;
            sol.cost += line_cost;
            sol.lines.insert(
                0,
                Line {
                    indent,
                    pieces: vec![piece],
                    text: line_text,
                    gap,
                    marker_ended: false,
                    raw: true,
                    piece_count: 1,
                    consumed: capacity,
                },
            );
            sol.consumed.insert(0, capacity);
            sol.texts.insert(0, sol.lines[0].text.clone());
            candidates.push(sol);
        }

        let best = candidates
            .into_iter()
            .reduce(|a, b| if better(&a, &b) { a } else { b })
            .unwrap_or(Solution {
                cost: 0,
                marker_count: 0,
                lines: Vec::new(),
                consumed: Vec::new(),
                texts: Vec::new(),
            });

        if offset == 0 {
            memo[0][wi][slot] = Some(best.clone());
        }
        best
    }

    if words.is_empty() {
        Vec::new()
    } else {
        let result = dp(0, 0, true, words, options, &mut memo);
        result.lines
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }
    for &(index, ch) in &chars {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let is_sep = |c: char| matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}');
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let valid_left = i > 0 && chars[i - 1].1 != '~' && !is_sep(chars[i - 1].1);
            let valid_right =
                i + 1 < chars.len() && chars[i + 1].1 != '~' && !is_sep(chars[i + 1].1);
            if !valid_left || !valid_right {
                return Err(WrapError::InvalidMarker {
                    index: chars[i].0,
                });
            }
        }
    }

    let mut normalized = String::new();
    let mut it = input.chars().peekable();
    while let Some(ch) = it.next() {
        if ch == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let scalars: Vec<char> = normalized.chars().collect();
    let mut paragraphs: Vec<String> = Vec::new();
    let mut current = String::new();
    let mut i = 0;
    while i < scalars.len() {
        if scalars[i] == '\n' {
            let mut j = i;
            while j < scalars.len() && scalars[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                paragraphs.push(std::mem::take(&mut current));
                for _ in 1..(j - i) {
                    paragraphs.push(String::new());
                }
                i = j;
                continue;
            }
        }
        current.push(scalars[i]);
        i += 1;
    }
    paragraphs.push(current);

    let mut output = Vec::new();
    for paragraph in paragraphs {
        let words = parse_words(&paragraph);
        let mut lines = solve_paragraph(&words, options);
        for (line_index, line) in lines.iter_mut().enumerate() {
            if options.justify
                && line_index + 1 < lines.len()
                && line.piece_count >= 2
            {
                let gaps = line.piece_count - 1;
                let extra = line.gap;
                let each = extra / gaps;
                let remainder = extra % gaps;
                let mut text = " ".repeat(line.indent);
                for (i, piece) in line.pieces.iter().enumerate() {
                    if i > 0 {
                        text.push(' ');
                        let gap_index = i - 1;
                        let start = line_index % gaps;
                        let cyclic_rank = (gap_index + gaps - start) % gaps;
                        text.push_str(&" ".repeat(each + usize::from(cyclic_rank < remainder)));
                    }
                    text.push_str(piece);
                }
                line.text = text;
            }
            output.push(line.text.clone());
        }
    }
    Ok(output)
}
