#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("new encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                }
                self.tail_len += 1;
                index += 1;

                if self.tail_len == 3 {
                    append_triplet(&mut output, self.tail[0], self.tail[1], input[index - 1]);
                    self.tail_len = 0;
                    break;
                }
            }
        }

        while index + 3 <= input.len() {
            append_triplet(
                &mut output,
                input[index],
                input[index + 1],
                input[index + 2],
            );
            index += 3;
        }

        while index < input.len() {
            self.tail[self.tail_len] = input[index];
            self.tail_len += 1;
            index += 1;
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            1 => {
                let a = self.tail[0];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                output.push_str("==");
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => {}
        }
        self.tail_len = 0;
        Ok(output)
    }
}

fn append_triplet(output: &mut String, a: u8, b: u8, c: u8) {
    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    output.push(ALPHABET[(c & 0x3f) as usize] as char);
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    group_len: usize,
    next_offset: usize,
    pending_second_padding: Option<usize>,
    terminal_padding: bool,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            group_len: 0,
            next_offset: 0,
            pending_second_padding: None,
            terminal_padding: false,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let offset = self.next_offset;
            self.next_offset = self.next_offset.saturating_add(1);

            let error = if self.terminal_padding {
                Some(Error::BadPadding { offset })
            } else if self.pending_second_padding.is_some() && byte != b'=' {
                Some(Error::BadPadding { offset })
            } else if byte == b'=' {
                if self.group_len <= 1 {
                    Some(Error::BadPadding { offset })
                } else if self.group_len == 2 {
                    self.group[2] = byte;
                    self.offsets[2] = offset;
                    self.pending_second_padding = Some(offset);
                    self.group_len = 3;
                    None
                } else {
                    self.group[3] = byte;
                    self.offsets[3] = offset;
                    self.group_len = 4;
                    self.terminal_padding = true;
                    output.extend(decode_group(&self.group));
                    self.group_len = 0;
                    None
                }
            } else if let Some(value) = digit(byte) {
                self.group[self.group_len] = value;
                self.offsets[self.group_len] = offset;
                self.group_len += 1;

                if self.group_len == 4 {
                    output.extend(decode_group(&self.group));
                    self.group_len = 0;
                }
                None
            } else {
                Some(Error::InvalidByte { offset, byte })
            };

            if let Some(error) = error {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            let error = Error::BadPadding { offset };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        if self.group_len != 0 {
            let error = Error::Truncated {
                offset: self.offsets[0],
            };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

fn digit(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&candidate| candidate == byte).map(|n| n as u8)
}

fn decode_group(group: &[u8; 4]) -> Vec<u8> {
    let mut output = Vec::with_capacity(3);
    output.push((group[0] << 2) | (group[1] >> 4));

    if group[2] != 64 {
        output.push((group[1] << 4) | (group[2] >> 2));
    }
    if group[3] != 64 {
        output.push((group[2] << 6) | group[3]);
    }
    output
}
