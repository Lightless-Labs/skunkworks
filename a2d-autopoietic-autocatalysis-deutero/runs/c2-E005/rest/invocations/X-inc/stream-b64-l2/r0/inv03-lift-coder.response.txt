#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn encode_group(bytes: &[u8]) -> String {
    let a = bytes[0];
    let b = if bytes.len() > 1 { bytes[1] } else { 0 };
    let c = if bytes.len() > 2 { bytes[2] } else { 0 };

    let mut out = String::with_capacity(4);
    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);

    if bytes.len() > 1 {
        out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    } else {
        out.push('=');
    }

    if bytes.len() > 2 {
        out.push(ALPHABET[(c & 0x3f) as usize] as char);
    } else {
        out.push('=');
    }

    out
}

pub fn encode(input: &[u8]) -> String {
    let mut out = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);
    for chunk in &mut chunks {
        out.push_str(&encode_group(chunk));
    }
    let tail = chunks.remainder();
    if !tail.is_empty() {
        out.push_str(&encode_group(tail));
    }
    out
}

pub struct Encoder {
    tail: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut combined = Vec::with_capacity(self.tail.len() + input.len());
        combined.extend_from_slice(&self.tail);
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 3 * 3;
        let mut out = String::with_capacity(complete_len / 3 * 4);
        for chunk in combined[..complete_len].chunks_exact(3) {
            out.push_str(&encode_group(chunk));
        }

        self.tail.clear();
        self.tail.extend_from_slice(&combined[complete_len..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let out = if self.tail.is_empty() {
            String::new()
        } else {
            encode_group(&self.tail)
        };
        self.tail.clear();
        self.finished = true;
        Ok(out)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

pub struct Decoder {
    group: [u8; 4],
    group_offsets: [usize; 4],
    group_len: usize,
    offset: usize,
    terminal: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            group_offsets: [0; 4],
            group_len: 0,
            offset: 0,
            terminal: false,
            pending_second_padding: None,
            finished: false,
            error: None,
        }
    }

    fn fail(&mut self, error: Error) -> Error {
        self.error = Some(error.clone());
        error
    }

    fn emit_group(&mut self, out: &mut Vec<u8>) {
        let a = digit(self.group[0]).unwrap();
        let b = digit(self.group[1]).unwrap();

        out.push((a << 2) | (b >> 4));

        if self.group[2] != b'=' {
            let c = digit(self.group[2]).unwrap();
            out.push((b << 4) | (c >> 2));

            if self.group[3] != b'=' {
                let d = digit(self.group[3]).unwrap();
                out.push((c << 6) | d);
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();

        for &byte in input {
            let current_offset = self.offset;
            self.offset += 1;

            if self.terminal {
                return Err(self.fail(Error::BadPadding {
                    offset: current_offset,
                }));
            }

            if self.pending_second_padding.is_some() && byte != b'=' {
                return Err(self.fail(Error::BadPadding {
                    offset: current_offset,
                }));
            }

            if byte == b'=' {
                match self.group_len {
                    0 | 1 => {
                        return Err(self.fail(Error::BadPadding {
                            offset: current_offset,
                        }));
                    }
                    2 => {
                        self.pending_second_padding = Some(current_offset);
                        self.group[self.group_len] = byte;
                        self.group_offsets[self.group_len] = current_offset;
                        self.group_len += 1;
                    }
                    3 => {
                        self.group[self.group_len] = byte;
                        self.group_offsets[self.group_len] = current_offset;
                        self.group_len += 1;

                        self.emit_group(&mut out);
                        self.group_len = 0;
                        self.pending_second_padding = None;
                        self.terminal = true;
                    }
                    _ => unreachable!(),
                }
                continue;
            }

            let Some(_) = digit(byte) else {
                return Err(self.fail(Error::InvalidByte {
                    offset: current_offset,
                    byte,
                }));
            };

            self.group[self.group_len] = byte;
            self.group_offsets[self.group_len] = current_offset;
            self.group_len += 1;

            if self.group_len == 4 {
                self.emit_group(&mut out);
                self.group_len = 0;
            }
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            let error = self.fail(Error::BadPadding { offset });
            return Err(error);
        }

        if self.group_len != 0 {
            let error = self.fail(Error::Truncated {
                offset: self.group_offsets[0],
            });
            return Err(error);
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}
