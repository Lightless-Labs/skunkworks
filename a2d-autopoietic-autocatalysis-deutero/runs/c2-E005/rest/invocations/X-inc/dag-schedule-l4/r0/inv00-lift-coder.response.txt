#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

#[derive(Clone)]
struct Running {
    task: usize,
    worker: usize,
    finish: u64,
    available_at: u64,
}

fn resource_snapshot(used: &[u64]) -> String {
    if used.is_empty() {
        return "-".to_string();
    }

    let mut result = String::new();
    for (index, value) in used.iter().enumerate() {
        if index > 0 {
            result.push(',');
        }
        result.push_str(&value.to_string());
    }
    result
}

fn fits(task: &Task, used: &[u64], capacity: &[u32]) -> bool {
    task.needs
        .iter()
        .enumerate()
        .all(|(r, &need)| u64::from(need) <= u64::from(capacity[r]) - used[r])
}

fn schedule_order(tasks: &[Task]) -> Result<(Vec<usize>, Vec<Vec<usize>>), Error> {
    let count = tasks.len();

    // Validate every dependency entry before performing any other work.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let mut indegree = vec![0usize; count];
    let mut children = vec![Vec::new(); count];

    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    let mut available = vec![false; count];
    for id in 0..count {
        available[id] = indegree[id] == 0;
    }

    let mut order = Vec::with_capacity(count);
    while let Some(id) = available.iter().position(|&is_available| is_available) {
        available[id] = false;
        order.push(id);

        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available[child] = true;
            }
        }
    }

    if order.len() != count {
        let removed = {
            let mut bits = vec![false; count];
            for &id in &order {
                bits[id] = true;
            }
            bits
        };
        let blocked = (0..count).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    Ok((order, children))
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    // Dependency validation must precede worker and resource validation.
    let (order, children) = {
        let count = tasks.len();

        for (task_id, task) in tasks.iter().enumerate() {
            for (position, &value) in task.deps.iter().enumerate() {
                if value >= count {
                    return Err(Error::BadDependency {
                        task: task_id,
                        position,
                        value,
                    });
                }
                if value == task_id {
                    return Err(Error::SelfDependency {
                        task: task_id,
                        position,
                    });
                }
                if task.deps[..position].contains(&value) {
                    return Err(Error::DuplicateDependency {
                        task: task_id,
                        position,
                        value,
                    });
                }
            }
        }

        if cooldowns.is_empty() || cooldowns.len() > 64 {
            return Err(Error::Workers {
                count: cooldowns.len(),
            });
        }

        for (task_id, task) in tasks.iter().enumerate() {
            if task.needs.len() != capacity.len() {
                return Err(Error::ResourceShape {
                    task: task_id,
                    expected: capacity.len(),
                    actual: task.needs.len(),
                });
            }
            for (resource, &need) in task.needs.iter().enumerate() {
                if need > capacity[resource] {
                    return Err(Error::ResourceExceeded {
                        task: task_id,
                        resource,
                        need,
                        capacity: capacity[resource],
                    });
                }
            }
        }

        // Kahn traversal uses the smallest currently available ID.
        let mut indegree = vec![0usize; count];
        let mut children = vec![Vec::new(); count];
        for (task_id, task) in tasks.iter().enumerate() {
            indegree[task_id] = task.deps.len();
            for &predecessor in &task.deps {
                children[predecessor].push(task_id);
            }
        }

        let mut available = vec![false; count];
        for id in 0..count {
            available[id] = indegree[id] == 0;
        }

        let mut order = Vec::with_capacity(count);
        while let Some(id) = available.iter().position(|&is_available| is_available) {
            available[id] = false;
            order.push(id);
            for &child in &children[id] {
                indegree[child] -= 1;
                if indegree[child] == 0 {
                    available[child] = true;
                }
            }
        }

        if order.len() != count {
            let mut removed = vec![false; count];
            for &id in &order {
                removed[id] = true;
            }
            return Err(Error::Cycle {
                blocked: (0..count).filter(|&id| !removed[id]).collect(),
            });
        }

        (order, children)
    };

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; tasks.len()];
    for &id in order.iter().rev() {
        let successor_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);

        tails[id] = tasks[id]
            .duration
            .checked_add(successor_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let count = tasks.len();
    let worker_count = cooldowns.len();
    let mut trace = format!(
        "workers={};resources={}\n",
        worker_count,
        capacity.len()
    );
    let mut runs = vec![
        Run {
            start: 0,
            finish: 0,
            worker: 0,
        };
        count
    ];
    let mut started = vec![false; count];
    let mut completed = vec![false; count];
    let mut completed_predecessors = vec![0usize; count];
    let mut worker_available = vec![0u64; worker_count];
    let mut running: Vec<Running> = Vec::new();
    let mut used = vec![0u64; capacity.len()];
    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < count {
        // Complete all tasks finishing now, in ascending task ID order.
        let mut finishing: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(index, item)| (item.finish == now).then_some(index))
            .collect();
        finishing.sort_by_key(|&index| running[index].task);

        for index in finishing.into_iter().rev() {
            let item = running.remove(index);
            let task_id = item.task;

            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] -= u64::from(need);
            }

            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(item.finish);

            trace.push_str(&format!(
                "{}|end|{}|{}|{}\n",
                now,
                task_id,
                item.worker,
                resource_snapshot(&used)
            ));

            for &child in &children[task_id] {
                completed_predecessors[child] += 1;
            }
        }

        loop {
            let worker = (0..worker_count)
                .find(|&worker| worker_available[worker] <= now);
            let Some(worker) = worker else {
                break;
            };

            let mut best: Option<usize> = None;
            for id in 0..count {
                if started[id]
                    || tasks[id].release > now
                    || completed_predecessors[id] != tasks[id].deps.len()
                    || !fits(&tasks[id], &used, capacity)
                {
                    continue;
                }

                let is_better = match best {
                    None => true,
                    Some(current) => {
                        tails[id] > tails[current]
                            || (tails[id] == tails[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority
                                        && id < current)))
                    }
                };
                if is_better {
                    best = Some(id);
                }
            }

            let Some(task_id) = best else {
                break;
            };

            // Select the task first, then its preferred available worker.
            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&candidate| worker_available[candidate] <= now)
                .expect("an available worker was found");

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            runs[task_id] = Run {
                start: now,
                finish,
                worker,
            };

            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }

            trace.push_str(&format!(
                "{}|start|{}|{}|{}\n",
                now,
                task_id,
                worker,
                resource_snapshot(&used)
            ));

            if finish == now {
                // Zero-duration tasks complete immediately before another admission.
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }

                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);

                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    now,
                    task_id,
                    worker,
                    resource_snapshot(&used)
                ));

                for &child in &children[task_id] {
                    completed_predecessors[child] += 1;
                }
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                    available_at,
                });
            }
        }

        if completed_count == count {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now && next_time.is_none_or(|current| time < current) {
                next_time = Some(time);
            }
        };

        for item in &running {
            consider(item.finish);
        }
        for id in 0..count {
            if !started[id]
                && completed_predecessors[id] == tasks[id].deps.len()
                && tasks[id].release > now
            {
                consider(tasks[id].release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        let Some(next) = next_time else {
            // With acyclic dependencies and finite releases, this is unreachable.
            break;
        };
        now = next;
    }

    trace.push_str(&format!("makespan={}\n", makespan));

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}
