use std::fmt::Write;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

fn resource_string(used: &[u128]) -> String {
    if used.is_empty() {
        return "-".to_string();
    }

    let mut out = String::new();
    for (i, value) in used.iter().enumerate() {
        if i > 0 {
            out.push(',');
        }
        write!(&mut out, "{value}").unwrap();
    }
    out
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Dependency validation has precedence over every other check.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    // Resource validation is task-major, then resource-major.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &limit)) in task.needs.iter().zip(capacity).enumerate() {
            if need > limit {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: limit,
                });
            }
        }
    }

    // Build graph edges from predecessor to dependent task.
    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Kahn traversal always removes the smallest available task ID.
    let mut available = vec![false; n];
    for id in 0..n {
        if indegree[id] == 0 {
            available[id] = true;
        }
    }

    let mut topo = Vec::with_capacity(n);
    while let Some(id) = (0..n).find(|&id| available[id]) {
        available[id] = false;
        topo.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available[child] = true;
            }
        }
    }

    if topo.len() != n {
        let mut blocked = Vec::new();
        let mut removed = vec![false; n];
        for &id in &topo {
            removed[id] = true;
        }
        for id in 0..n {
            if !removed[id] {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in topo.iter().rev() {
        let mut max_child = 0u64;
        for &child in &children[id] {
            max_child = max_child.max(tails[child]);
        }
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let resource_count = capacity.len();
    let mut used = vec![0u128; resource_count];
    let mut worker_available = vec![0u64; worker_count];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_deps = tasks.iter().map(|t| t.deps.len()).collect::<Vec<_>>();
    let mut running = Vec::<(usize, usize, u64)>::new(); // task, worker, finish
    let mut runs = vec![
        Run {
            start: 0,
            finish: 0,
            worker: 0,
        };
        n
    ];

    let mut trace = String::new();
    writeln!(
        &mut trace,
        "workers={worker_count};resources={resource_count}"
    )
    .unwrap();

    let mut now = 0u64;
    let mut finished_count = 0usize;
    let mut makespan = 0u64;

    while finished_count < n {
        // Complete all tasks ending now, in ascending task ID order.
        running.sort_by_key(|&(task_id, _, _)| task_id);
        let mut kept = Vec::with_capacity(running.len());
        for (task_id, worker, finish) in running.drain(..) {
            if finish == now {
                for r in 0..resource_count {
                    used[r] -= tasks[task_id].needs[r] as u128;
                }
                completed[task_id] = true;
                finished_count += 1;
                makespan = makespan.max(finish);
                writeln!(
                    &mut trace,
                    "{now}|end|{task_id}|{worker}|{}",
                    resource_string(&used)
                )
                .unwrap();

                for &child in &children[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                kept.push((task_id, worker, finish));
            }
        }
        running = kept;

        // Start one task at a time, reconsidering eligibility after each start.
        loop {
            let mut chosen = None;
            for id in 0..n {
                if started[id]
                    || tasks[id].release > now
                    || remaining_deps[id] != 0
                {
                    continue;
                }
                let fits = (0..resource_count).all(|r| {
                    tasks[id].needs[r] as u128 <= capacity[r] as u128 - used[r]
                });
                if !fits {
                    continue;
                }
                let better = match chosen {
                    None => true,
                    Some(best) => {
                        tails[id] > tails[best]
                            || (tails[id] == tails[best]
                                && (tasks[id].priority > tasks[best].priority
                                    || (tasks[id].priority == tasks[best].priority && id < best)))
                    }
                };
                if better {
                    chosen = Some(id);
                }
            }

            let Some(task_id) = chosen else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&w| worker_available[w] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            runs[task_id] = Run {
                start: now,
                finish,
                worker,
            };

            for r in 0..resource_count {
                used[r] += tasks[task_id].needs[r] as u128;
            }

            writeln!(
                &mut trace,
                "{now}|start|{task_id}|{worker}|{}",
                resource_string(&used)
            )
            .unwrap();

            if finish == now {
                for r in 0..resource_count {
                    used[r] -= tasks[task_id].needs[r] as u128;
                }
                completed[task_id] = true;
                finished_count += 1;
                makespan = makespan.max(finish);
                writeln!(
                    &mut trace,
                    "{now}|end|{task_id}|{worker}|{}",
                    resource_string(&used)
                )
                .unwrap();

                for &child in &children[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.push((task_id, worker, finish));
            }
        }

        if finished_count == n {
            break;
        }

        // Move to the earliest future event that can change eligibility.
        let mut next = None::<u64>;
        let mut consider = |t: u64| {
            if t > now {
                next = Some(next.map_or(t, |old| old.min(t)));
            }
        };

        for &(_, _, finish) in &running {
            consider(finish);
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                consider(tasks[id].release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        now = next.expect("unfinished tasks always have a future event");
    }

    writeln!(&mut trace, "makespan={makespan}").unwrap();

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}
