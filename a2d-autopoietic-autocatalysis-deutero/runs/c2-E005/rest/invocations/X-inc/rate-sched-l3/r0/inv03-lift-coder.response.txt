#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobData {
    job: Job,
    dependencies: Vec<usize>,
    state: Option<State>,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<JobData>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if job.max_attempts == 0 || job.max_attempts > 32 {
                return Err(Error::BadAttempts(i));
            }
            if ids.insert(job.id.as_str(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut dependencies = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (j, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(dependency.as_str()) {
                    Some(&index) => dependencies[i].push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
        }

        if has_cycle(&dependencies) {
            return Err(Error::Cycle);
        }

        let jobs = jobs
            .into_iter()
            .zip(dependencies)
            .map(|(job, dependencies)| JobData {
                job,
                dependencies,
                state: None,
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;
        if self.cycle_started
            && self.jobs.iter().any(|job| {
                matches!(job.state, Some(State::Waiting { .. } | State::Running { .. }))
            })
        {
            return Err(Error::Busy);
        }

        for job in &mut self.jobs {
            job.state = Some(State::Waiting { ready_at: now });
            job.attempts = 0;
        }
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut selected: Option<(usize, u8)> = None;
        for (i, job) in self.jobs.iter().enumerate() {
            let Some(State::Waiting { ready_at }) = job.state else {
                continue;
            };
            if ready_at > now
                || !job.dependencies.iter().all(|&dep| {
                    matches!(self.jobs[dep].state, Some(State::Succeeded))
                })
            {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                job.job.priority
            } else {
                let gained = (now - ready_at) / self.aging_ms;
                (u64::from(job.job.priority) + gained).min(9) as u8
            };

            if selected.map_or(true, |(_, priority)| effective > priority) {
                selected = Some((i, effective));
            }
        }

        self.last_time = now;
        let Some((index, _)) = selected else {
            return Ok(None);
        };

        let job = &mut self.jobs[index];
        job.attempts += 1;
        let attempt = job.attempts;
        job.state = Some(State::Running { attempt });
        Ok(Some(Ticket {
            id: job.job.id.clone(),
            attempt,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let index = self
            .jobs
            .iter()
            .position(|job| job.job.id == id)
            .ok_or(Error::Unknown)?;

        if !matches!(self.jobs[index].state, Some(State::Running { .. })) {
            return Err(Error::NotRunning);
        }

        self.last_time = now;
        if success {
            self.jobs[index].state = Some(State::Succeeded);
        } else if self.jobs[index].attempts == self.jobs[index].job.max_attempts {
            self.jobs[index].state = Some(State::Failed);
            self.propagate_blocked();
        } else {
            let delay = retry_delay(
                self.retry_base,
                self.retry_cap,
                self.jobs[index].attempts,
            );
            let ready_at = now.saturating_add(delay);
            self.jobs[index].state = Some(State::Waiting { ready_at });
        }
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let job = self.jobs.iter().find(|job| job.job.id == id).ok_or(Error::Unknown)?;
        job.state.clone().ok_or(Error::NoCycle)
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut changed = false;
            for i in 0..self.jobs.len() {
                if !matches!(self.jobs[i].state, Some(State::Waiting { .. })) {
                    continue;
                }
                let blocked = self.jobs[i].dependencies.iter().any(|&dep| {
                    matches!(
                        self.jobs[dep].state,
                        Some(State::Failed | State::Blocked)
                    )
                });
                if blocked {
                    self.jobs[i].state = Some(State::Blocked);
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes.iter().all(|&b| {
            b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-'
        })
}

fn has_cycle(dependencies: &[Vec<usize>]) -> bool {
    fn visit(node: usize, dependencies: &[Vec<usize>], marks: &mut [u8]) -> bool {
        match marks[node] {
            1 => return true,
            2 => return false,
            _ => {}
        }
        marks[node] = 1;
        for &dep in &dependencies[node] {
            if visit(dep, dependencies, marks) {
                return true;
            }
        }
        marks[node] = 2;
        false
    }

    let mut marks = vec![0; dependencies.len()];
    (0..dependencies.len()).any(|node| visit(node, dependencies, &mut marks))
}

fn retry_delay(base: u64, cap: u64, attempt: u8) -> u64 {
    let mut delay = base;
    for _ in 1..attempt {
        if delay >= cap || delay > cap / 2 {
            return cap;
        }
        delay *= 2;
    }
    delay.min(cap)
}
