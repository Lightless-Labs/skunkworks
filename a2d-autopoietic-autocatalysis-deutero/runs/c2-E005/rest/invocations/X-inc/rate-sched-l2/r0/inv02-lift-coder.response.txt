#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(text, 0, split, 60, &mut minutes)?;
        parse_field(text, split + 1, bytes.len(), 24, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut item_start = start;

    for pos in start..=end {
        if pos == end || bytes[pos] == b',' {
            if item_start == pos {
                return Err(CronError::Syntax(item_start));
            }
            parse_item(text, item_start, pos, limit, selected)?;
            item_start = pos + 1;
        }
    }
    Ok(())
}

fn parse_item(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut slash = None;
    for pos in start..end {
        if bytes[pos] == b'/' {
            if slash.is_some() {
                return Err(CronError::Syntax(pos));
            }
            slash = Some(pos);
        }
    }

    let base_end = slash.unwrap_or(end);
    let step = if let Some(slash_pos) = slash {
        let base_is_star = base_end == start + 1 && bytes[start] == b'*';
        let base_is_range = bytes[start..base_end].contains(&b'-');
        if !base_is_star && !base_is_range {
            return Err(CronError::Syntax(slash_pos));
        }
        Some(parse_number(text, slash_pos + 1, end, limit, true)?)
    } else {
        None
    };

    let (range_start, range_end, dash) = if start + 1 == base_end && bytes[start] == b'*' {
        (0, limit - 1, None)
    } else {
        let mut dash_pos = None;
        for pos in start..base_end {
            if bytes[pos] == b'-' {
                dash_pos = Some(pos);
                break;
            }
        }

        if let Some(dash_pos) = dash_pos {
            let left = parse_number(text, start, dash_pos, limit, false)?;
            let right = parse_number(text, dash_pos + 1, base_end, limit, false)?;
            if left > right {
                return Err(CronError::Descending(dash_pos));
            }
            (left, right, Some(dash_pos))
        } else {
            let value = parse_number(text, start, base_end, limit, false)?;
            (value, value, None)
        }
    };

    let increment = step.unwrap_or(1);
    let _ = dash;
    let mut value = range_start;
    while value <= range_end {
        selected[value] = true;
        match value.checked_add(increment) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_number(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
    is_step: bool,
) -> Result<usize, CronError> {
    let bytes = text.as_bytes();
    if start == end {
        return Err(CronError::Syntax(start));
    }

    for pos in start..end {
        if !bytes[pos].is_ascii_digit() {
            return Err(CronError::Syntax(pos));
        }
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for pos in start..end {
        let digit = (bytes[pos] - b'0') as usize;
        value = match value.checked_mul(10).and_then(|v| v.checked_add(digit)) {
            Some(v) => v,
            None => {
                return Err(if is_step {
                    CronError::Step(start)
                } else {
                    CronError::Bounds(start)
                });
            }
        };
    }

    if is_step {
        if value == 0 || value > limit {
            return Err(CronError::Step(start));
        }
    } else if value >= limit {
        return Err(CronError::Bounds(start));
    }
    Ok(value)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
    eligible_after: Option<u64>,
}

pub struct Calendar {
    jobs: Vec<Job>,
    reserved: Vec<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            reserved: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let order = self.jobs.len();
        self.reserved.push(id.to_owned());
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order,
            eligible_after: self.last_poll,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower = self.last_poll;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let effective_lower = match (lower, job.eligible_after) {
                (Some(a), Some(b)) => Some(a.max(b)),
                (Some(a), None) => Some(a),
                (None, Some(b)) => Some(b),
                (None, None) => None,
            };

            let first_boundary = match effective_lower {
                Some(t) => t / 60_000 + 1,
                None => 0,
            };
            let last_boundary = now / 60_000;
            if first_boundary > last_boundary {
                continue;
            }

            let matches_at = |boundary: u64| job.cron.matches(boundary);
            match job.policy {
                Policy::All => {
                    for minute in first_boundary..=last_boundary {
                        if matches_at(minute) {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
                Policy::Once => {
                    if let Some(minute) = (first_boundary..=last_boundary)
                        .rev()
                        .find(|&minute| matches_at(minute))
                    {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0 {
                        let minute = now / 60_000;
                        if minute >= first_boundary
                            && minute <= last_boundary
                            && matches_at(minute)
                        {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|entry| entry.0).collect())
    }
}
