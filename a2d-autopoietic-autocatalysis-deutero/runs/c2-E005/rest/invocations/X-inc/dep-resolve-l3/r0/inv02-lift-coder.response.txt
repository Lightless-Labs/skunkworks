#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    (1..=24).contains(&bytes.len())
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_names = std::collections::BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    Ok(())
}

fn constraint_matches(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn canonicalize(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn active_constraints(
    roots: &[Requirement],
    catalog: &[Package],
    assigned: &std::collections::BTreeMap<String, Version>,
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (name, version) in assigned {
        if let Some(package) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        *constraints = canonicalize(std::mem::take(constraints));
    }
    active
}

fn candidates<'a>(
    package: &'a Package,
    constraints: &[Constraint],
    lock: Option<Version>,
) -> Vec<&'a Release> {
    let mut matches: Vec<&Release> = package
        .releases
        .iter()
        .filter(|release| {
            constraints
                .iter()
                .all(|constraint| constraint_matches(*constraint, release.version))
        })
        .collect();
    matches.sort_by(|a, b| b.version.cmp(&a.version));
    if let Some(locked) = lock {
        if let Some(index) = matches.iter().position(|r| r.version == locked) {
            let preferred = matches.remove(index);
            matches.insert(0, preferred);
        }
    }
    matches
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a std::collections::BTreeMap<String, Version>,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, std::collections::BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn record_leaf_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn lock_cost(&self, assigned: &std::collections::BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|(name, version)| assigned.get(*name) != Some(*version))
            .count()
    }

    fn better(
        &self,
        cost: usize,
        candidate: &std::collections::BTreeMap<String, Version>,
    ) -> bool {
        let Some((best_cost, best)) = &self.best else {
            return true;
        };
        if cost != *best_cost {
            return cost < *best_cost;
        }

        let mut names: Vec<&str> = self.catalog.iter().map(|p| p.name.as_str()).collect();
        names.sort_unstable();
        for name in names {
            match (candidate.get(name), best.get(name)) {
                (Some(a), Some(b)) if a != b => return a > b,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                _ => {}
            }
        }
        false
    }

    fn inspect(
        &mut self,
        assigned: &mut std::collections::BTreeMap<String, Version>,
    ) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, self.catalog, assigned);

        if let Some(name) = active
            .keys()
            .filter(|name| !self.catalog.iter().any(|p| p.name == ***name))
            .next()
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in assigned.iter() {
            if let Some(constraints) = active.get(name) {
                if !constraints
                    .iter()
                    .all(|constraint| constraint_matches(*constraint, *version))
                {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if !assigned.contains_key(name) {
                let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
                if candidates(package, constraints, self.locks.get(name).copied()).is_empty() {
                    return Err(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        let next = active.keys().find(|name| !assigned.contains_key(*name)).cloned();
        let Some(name) = next else {
            let cost = self.lock_cost(assigned);
            if self.better(cost, assigned) {
                self.best = Some((cost, assigned.clone()));
            }
            return Ok(());
        };

        let constraints = &active[&name];
        let package = self.catalog.iter().find(|p| p.name == name).unwrap();
        let options = candidates(package, constraints, self.locks.get(&name).copied());

        for release in options {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assigned.insert(name.clone(), release.version);
            match self.inspect(assigned) {
                Ok(()) => {}
                Err(error @ ResolveError::Limit { .. }) => {
                    assigned.remove(&name);
                    return Err(error);
                }
                Err(error) => self.record_leaf_error(error),
            }
            assigned.remove(&name);
        }
        Ok(())
    }
}

fn encode_text(value: &str) -> String {
    let mut encoded = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            encoded.push(*byte as char);
        } else {
            use std::fmt::Write;
            let _ = write!(encoded, "%{byte:02X}");
        }
    }
    encoded
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonicalize(constraints.to_vec())
        .iter()
        .map(|constraint| constraint_text(*constraint))
        .collect::<Vec<_>>()
        .join(",")
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode_text(site), encode_text(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode_text(package))
        }
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode_text(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode_text(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let lock_map: std::collections::BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();

    let mut search = Search {
        catalog,
        roots,
        locks: &lock_map,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };

    let mut assigned = std::collections::BTreeMap::new();
    search.inspect(&mut assigned)?;

    let Some((_, selected_map)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| {
            ResolveError::NoMatch {
                package: String::new(),
                requirements: Vec::new(),
            }
        }));
    };

    let final_active = active_constraints(roots, catalog, &selected_map);
    let selected: Vec<Selection> = selected_map
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut changes = Vec::new();
    for (package, before) in &lock_map {
        let after = selected_map.get(package).copied();
        if after == Some(*before) {
            continue;
        }
        let reason = match after {
            None => ChangeReason::Removed,
            Some(_) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == *package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == *before));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active.get(package).is_some_and(|constraints| {
                    !constraints
                        .iter()
                        .all(|constraint| constraint_matches(*constraint, *before))
                }) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
        };
        changes.push(Change {
            package: package.clone(),
            before: Some(*before),
            after,
            reason,
        });
    }

    for (package, after) in &selected_map {
        if !lock_map.contains_key(package) {
            changes.push(Change {
                package: package.clone(),
                before: None,
                after: Some(*after),
                reason: ChangeReason::Added,
            });
        }
    }
    changes.sort_by(|a, b| a.package.cmp(&b.package));

    Ok(Resolution { selected, changes })
}
