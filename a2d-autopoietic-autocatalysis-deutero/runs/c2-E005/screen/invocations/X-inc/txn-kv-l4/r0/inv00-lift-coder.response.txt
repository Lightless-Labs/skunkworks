```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
struct Entry {
    value: Option<String>,
    deadline: Option<u64>,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    entry: Entry,
}

struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Entry>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Entry>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn tx_exists(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn visible_entry(&self, key: &str, tx: Option<u64>) -> Option<Entry> {
        if let Some(id) = tx {
            let transaction = self.transactions.get(&id)?;
            if let Some(entry) = transaction.writes.get(key) {
                return Some(entry.clone());
            }
            self.data.get(key)?.iter()
                .rev()
                .find(|v| v.revision <= transaction.begin_revision)
                .map(|v| v.entry.clone())
        } else {
            self.data.get(key)?.last().map(|v| v.entry.clone())
        }
    }

    fn live(entry: &Entry, now: u64) -> Option<&str> {
        if entry.deadline.is_some_and(|d| d <= now) {
            return None;
        }
        entry.value.as_deref()
    }

    fn matching(&self, tx: Option<u64>, now: u64, start: &str, end: Option<&str>, prefix: Option<&str>) -> Vec<(String, String)> {
        let mut result = Vec::new();
        for key in self.data.keys() {
            if key.as_str() < start
                || end.is_some_and(|e| key.as_str() >= e)
                || prefix.is_some_and(|p| !key.starts_with(p))
            {
                continue;
            }
            if let Some(entry) = self.visible_entry(key, tx) {
                if let Some(value) = Self::live(&entry, now) {
                    result.push((key.clone(), value.to_owned()));
                }
            }
        }
        if let Some(id) = tx {
            if let Some(transaction) = self.transactions.get(&id) {
                for (key, entry) in &transaction.writes {
                    if self.data.contains_key(key)
                        || key.as_str() < start
                        || end.is_some_and(|e| key.as_str() >= e)
                        || prefix.is_some_and(|p| !key.starts_with(p))
                    {
                        continue;
                    }
                    if let Some(value) = Self::live(entry, now) {
                        result.push((key.clone(), value.to_owned()));
                    }
                }
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        result
    }

    fn put_committed(&mut self, key: String, entry: Entry) {
        self.revision += 1;
        self.data.entry(key).or_default().push(Version {
            revision: self.revision,
            entry,
        });
    }

    fn validate_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (i, b) in name.bytes().enumerate() {
            let valid = if i == 0 {
                b.is_ascii_lowercase()
            } else {
                b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_'
            };
            if !valid {
                return Err(Error::NameInvalid { at: i });
            }
        }
        Ok(())
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(id, Transaction {
            begin_revision: self.revision,
            read_only,
            writes: BTreeMap::new(),
        });
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let tx = self.transactions.get(&id).ok_or(Error::UnknownTransaction(id))?;
        if !tx.writes.is_empty() {
            for key in tx.writes.keys() {
                if self.data.get(key).and_then(|v| v.last())
                    .is_some_and(|v| v.revision > tx.begin_revision)
                {
                    return Err(Error::Conflict { tx: id, key: key.clone() });
                }
            }
            let writes = tx.writes.clone();
            self.revision += 1;
            let revision = self.revision;
            for (key, entry) in writes {
                self.data.entry(key).or_default().push(Version { revision, entry });
            }
            self.transactions.remove(&id);
            Ok(revision)
        } else {
            self.transactions.remove(&id);
            Ok(self.revision)
        }
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.remove(&id).is_none() {
            return Err(Error::UnknownTransaction(id));
        }
        Ok(())
    }

    pub fn set(&mut self, now: u64, tx: Option<u64>, key: &str, value: &str, ttl: Option<u64>) -> Result<(), Error> {
        self.admit(now)?;
        self.tx_exists(tx)?;
        if key.is_empty() { return Err(Error::KeyEmpty); }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let entry = match ttl {
            Some(0) => Entry { value: None, deadline: None },
            Some(d) => Entry {
                value: Some(value.to_owned()),
                deadline: Some(now.checked_add(d).ok_or(Error::DeadlineOverflow)?),
            },
            None => Entry { value: Some(value.to_owned()), deadline: None },
        };
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_owned(), entry);
        } else {
            self.put_committed(key.to_owned(), entry);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.tx_exists(tx)?;
        if key.is_empty() { return Err(Error::KeyEmpty); }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only { return Err(Error::ReadOnly(id)); }
        }
        let was_live = self.visible_entry(key, tx)
            .and_then(|e| Self::live(&e, now).map(str::to_owned)).is_some();
        let tombstone = Entry { value: None, deadline: None };
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_owned(), tombstone);
        } else {
            self.put_committed(key.to_owned(), tombstone);
        }
        Ok(was_live)
    }

    pub fn get(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.tx_exists(tx)?;
        if key.is_empty() { return Err(Error::KeyEmpty); }
        Ok(self.visible_entry(key, tx).and_then(|e| Self::live(&e, now).map(str::to_owned)))
    }

    pub fn count(&mut self, now: u64, tx: Option<u64>, value: &str) -> Result<usize, Error> {
        self.admit(now)?;
        self.tx_exists(tx)?;
        Ok(self.matching(tx, now, "", None, None).iter().filter(|(_, v)| v == value).count())
    }

    pub fn scan(&mut self, now: u64, tx: Option<u64>, start: &str, end: Option<&str>, prefix: Option<&str>, offset: usize, limit: usize, cursor: Option<&Cursor>) -> Result<Page, Error> {
        self.admit(now)?;
        self.tx_exists(tx)?;
        if end.is_some_and(|e| e <= start) { return Err(Error::InvalidRange); }
        if let Some(c) = cursor {
            if c.start != start || c.end.as_deref() != end || c.prefix.as_deref() != prefix || c.offset != offset
                || c.last.is_empty() || c.last.as_str() < start
                || end.is_some_and(|e| c.last.as_str() >= e)
                || prefix.is_some_and(|p| !c.last.starts_with(p))
            {
                return Err(Error::CursorMismatch);
            }
        }
        if limit == 0 { return Ok(Page { items: Vec::new(), next: None }); }
        let mut keys = self.matching(tx, now, start, end, prefix);
        if let Some(c) = cursor {
            keys.retain(|(k, _)| k > &c.last);
        } else {
            let skip = offset.min(keys.len());
            keys.drain(..skip);
        }
        let has_more = keys.len() > limit;
        keys.truncate(limit);
        let next = if has_more {
            Some(Cursor {
                start: start.to_owned(),
                end: end.map(str::to_owned),
                prefix: prefix.map(str::to_owned),
                offset,
                last: keys.last().unwrap().0.clone(),
            })
        } else { None };
        Ok(Page { items: keys, next })
    }

    pub fn aggregate(&mut self, now: u64, tx: Option<u64>, start: &str, end: Option<&str>, prefix: Option<&str>) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.tx_exists(tx)?;
        if end.is_some_and(|e| e <= start) { return Err(Error::InvalidRange); }
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in self.matching(tx, now, start, end, prefix) {
            let number = parse_canonical_i64(&value).ok_or_else(|| Error::InvalidNumber { key: key.clone(), value: value.clone() })?;
            sum = sum.checked_add(number).ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 { return Err(Error::SnapshotLimit); }
        let mut saved = BTreeMap::new();
        for (key, versions) in &self.data {
            if let Some(entry) = versions.last().map(|v| &v.entry) {
                if Self::live(entry, now).is_some() {
                    saved.insert(key.clone(), entry.clone());
                }
            }
        }
        self.snapshots.insert(name.to_owned(), saved);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        let saved = self.snapshots.get(name).ok_or_else(|| Error::SnapshotMissing(name.to_owned()))?;
        if !self.transactions.is_empty() { return Err(Error::ActiveTransactions); }
        let copied: BTreeMap<String, Entry> = saved.iter()
            .filter(|(_, e)| Self::live(e, now).is_some())
            .map(|(k, e)| (k.clone(), e.clone())).collect();
        let mut live_current = BTreeSet::new();
        for (k, versions) in &self.data {
            if versions.last().is_some_and(|v| Self::live(&v.entry, now).is_some()) {
                live_current.insert(k.clone());
            }
        }
        let union: BTreeSet<String> = live_current.union(&copied.keys().cloned().collect()).cloned().collect();
        if union.is_empty() { return Ok(self.revision); }
        self.revision += 1;
        let rev = self.revision;
        for key in union {
            let entry = copied.get(&key).cloned().unwrap_or(Entry { value: None, deadline: None });
            self.data.entry(key).or_default().push(Version { revision: rev, entry });
        }
        Ok(rev)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_owned()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let mut removed = 0;
        for versions in self.data.values_mut() {
            let mut keep = BTreeSet::new();
            if !versions.is_empty() { keep.insert(versions.len() - 1); }
            for tx in self.transactions.values() {
                if let Some(index) = versions.iter().rposition(|v| v.revision <= tx.begin_revision) {
                    keep.insert(index);
                }
            }
            let mut i = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&i);
                i += 1;
                if !retain { removed += 1; }
                retain
            });
        }
        removed
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let keys = self.data.len();
        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self.data.values().flat_map(|v| v.iter()).filter(|v| v.entry.value.is_none()).count();
        let live_keys = self.data.values().filter(|v| v.last().is_some_and(|x| Self::live(&x.entry, now).is_some())).count();
        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys,
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions: self.transactions.values().filter(|t| t.read_only).count(),
            staged_writes: self.transactions.values().map(|t| t.writes.len()).sum(),
            snapshots: self.snapshots.len(),
        })
    }
}

fn parse_canonical_i64(s: &str) -> Option<i64> {
    if s == "0" { return Some(0); }
    let digits = if let Some(rest) = s.strip_prefix('-') {
        if rest.is_empty() || rest == "0" || !rest.bytes().all(|b| b.is_ascii_digit()) || rest.starts_with('0') {
            return None;
        }
        rest
    } else {
        if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) || s.starts_with('0') {
            return None;
        }
        s
    };
    if digits.is_empty() { return None; }
    s.parse::<i64>().ok()
}
```
