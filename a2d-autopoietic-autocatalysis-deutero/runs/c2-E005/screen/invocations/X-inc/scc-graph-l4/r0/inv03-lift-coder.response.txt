pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EdgeStatus {
    Duplicate,
    Internal,
    Link,
    Merge,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Step {
    pub edge_index: usize,
    pub status: EdgeStatus,
    pub before: (usize, usize),
    pub after: (usize, usize),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub edge_index: usize,
    pub before: Vec<usize>,
    pub after: usize,
    pub members: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Batch {
    pub steps: Vec<Step>,
    pub merges: Vec<Merge>,
    pub graph: Graph,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

fn validate(n: usize, edges: &[(usize, usize)]) -> Result<(), GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
    }
    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }
    Ok(())
}

fn build_graph(n: usize, edges: &[(usize, usize)]) -> Graph {
    let mut adjacency = vec![Vec::<usize>::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
    }

    // Iterative first pass of Kosaraju's algorithm.
    let mut seen = vec![false; n];
    let mut finish_order = Vec::with_capacity(n);
    for root in 0..n {
        if seen[root] {
            continue;
        }
        seen[root] = true;
        let mut stack = vec![(root, 0usize)];
        while let Some((vertex, next_index)) = stack.pop() {
            if next_index < adjacency[vertex].len() {
                stack.push((vertex, next_index + 1));
                let next = adjacency[vertex][next_index];
                if !seen[next] {
                    seen[next] = true;
                    stack.push((next, 0));
                }
            } else {
                finish_order.push(vertex);
            }
        }
    }

    let mut reverse_adjacency = vec![Vec::<usize>::new(); n];
    for &(from, to) in edges {
        reverse_adjacency[to].push(from);
    }

    let mut assigned = vec![false; n];
    let mut components = Vec::<Vec<usize>>::new();
    for &root in finish_order.iter().rev() {
        if assigned[root] {
            continue;
        }
        assigned[root] = true;
        let mut stack = vec![root];
        let mut component = Vec::new();
        while let Some(vertex) = stack.pop() {
            component.push(vertex);
            for &next in &reverse_adjacency[vertex] {
                if !assigned[next] {
                    assigned[next] = true;
                    stack.push(next);
                }
            }
        }
        component.sort_unstable();
        components.push(component);
    }

    components.sort_by(|left, right| {
        right
            .len()
            .cmp(&left.len())
            .then_with(|| left[0].cmp(&right[0]))
    });

    let mut component_of = vec![0usize; n];
    for (id, component) in components.iter().enumerate() {
        for &vertex in component {
            component_of[vertex] = id;
        }
    }

    let count = components.len();
    let mut cyclic = vec![false; count];
    let mut internal_edges = vec![0usize; count];
    let mut arc_data = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let source = component_of[from];
        let target = component_of[to];
        if source == target {
            internal_edges[source] += 1;
            if from == to {
                cyclic[source] = true;
            }
        } else {
            let entry = arc_data.entry((source, target)).or_insert((0, edge_index));
            entry.0 += 1;
        }
    }

    for (id, component) in components.iter().enumerate() {
        if component.len() > 1 {
            cyclic[id] = true;
        }
    }

    let mut arcs: Vec<Arc> = arc_data
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|left, right| {
        left.from
            .cmp(&right.from)
            .then_with(|| right.to.cmp(&left.to))
    });

    let mut indegree = vec![0usize; count];
    let mut outgoing = vec![Vec::<usize>::new(); count];
    for arc in &arcs {
        indegree[arc.to] += 1;
        outgoing[arc.from].push(arc.to);
    }

    let mut ready = Vec::<usize>::new();
    for id in 0..count {
        if indegree[id] == 0 {
            ready.push(id);
        }
    }

    let mut topo = Vec::with_capacity(count);
    let mut depth = vec![0usize; count];
    while !ready.is_empty() {
        let mut best_index = 0;
        for index in 1..ready.len() {
            let candidate = ready[index];
            let best = ready[best_index];
            if internal_edges[candidate] > internal_edges[best]
                || (internal_edges[candidate] == internal_edges[best]
                    && components[candidate][0] > components[best][0])
            {
                best_index = index;
            }
        }

        let vertex = ready.swap_remove(best_index);
        topo.push(vertex);
        for &next in &outgoing[vertex] {
            depth[next] = depth[next].max(depth[vertex] + 1);
            indegree[next] -= 1;
            if indegree[next] == 0 {
                ready.push(next);
            }
        }
    }

    Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    }
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    validate(n, edges)?;
    Ok(build_graph(n, edges))
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        let n = self.component_of.len();
        if from >= n {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let source = self.component_of[from];
        let target = self.component_of[to];
        if source == target {
            return Ok(if self.cyclic[source] {
                Query {
                    kind: Relation::Cycle,
                    route: vec![source, source],
                }
            } else {
                Query {
                    kind: Relation::Trivial,
                    route: vec![source],
                }
            });
        }

        fn shortest_route(graph: &Graph, source: usize, target: usize) -> Option<Vec<usize>> {
            let count = graph.components.len();
            let mut adjacency = vec![Vec::<usize>::new(); count];
            for arc in &graph.arcs {
                adjacency[arc.from].push(arc.to);
            }
            for neighbors in &mut adjacency {
                neighbors.sort_unstable();
            }

            let mut parent = vec![usize::MAX; count];
            let mut queue = std::collections::VecDeque::new();
            parent[source] = source;
            queue.push_back(source);

            while let Some(vertex) = queue.pop_front() {
                if vertex == target {
                    break;
                }
                for &next in &adjacency[vertex] {
                    if parent[next] == usize::MAX {
                        parent[next] = vertex;
                        queue.push_back(next);
                    }
                }
            }

            if parent[target] == usize::MAX {
                return None;
            }

            let mut route = vec![target];
            let mut vertex = target;
            while vertex != source {
                vertex = parent[vertex];
                route.push(vertex);
            }
            route.reverse();
            Some(route)
        }

        if let Some(route) = shortest_route(self, source, target) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_route(self, target, source) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

#[derive(Clone, Debug)]
pub struct Incremental {
    n: usize,
    edges: Vec<(usize, usize)>,
    graph: Graph,
    history: Vec<Merge>,
}

impl Incremental {
    pub fn new(n: usize, edges: &[(usize, usize)]) -> Result<Self, GraphError> {
        let graph = analyze(n, edges)?;
        Ok(Self {
            n,
            edges: edges.to_vec(),
            graph,
            history: Vec::new(),
        })
    }

    pub fn snapshot(&self) -> Graph {
        self.graph.clone()
    }

    pub fn history(&self) -> Vec<Merge> {
        self.history.clone()
    }

    pub fn insert(&mut self, batch_edges: &[(usize, usize)]) -> Result<Batch, GraphError> {
        if self.edges.len() + batch_edges.len() > MAX_EDGES {
            return Err(GraphError::TooManyEdges { limit: MAX_EDGES });
        }

        for (index, &(from, to)) in batch_edges.iter().enumerate() {
            if from >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: index,
                    side: 0,
                    vertex: from,
                });
            }
            if to >= self.n {
                return Err(GraphError::BadEndpoint {
                    edge: index,
                    side: 1,
                    vertex: to,
                });
            }
        }

        let mut steps = Vec::with_capacity(batch_edges.len());
        let mut merges = Vec::new();

        for &(from, to) in batch_edges {
            let edge_index = self.edges.len();
            let old_graph = self.graph.clone();
            let before = (
                old_graph.component_of[from],
                old_graph.component_of[to],
            );
            let duplicate = self.edges.iter().any(|&(u, v)| u == from && v == to);

            self.edges.push((from, to));
            self.graph = build_graph(self.n, &self.edges);

            let after = (self.graph.component_of[from], self.graph.component_of[to]);
            let status = if duplicate {
                EdgeStatus::Duplicate
            } else if before.0 == before.1 {
                EdgeStatus::Internal
            } else if after.0 == after.1 {
                EdgeStatus::Merge
            } else {
                EdgeStatus::Link
            };

            if status == EdgeStatus::Merge {
                let after_id = after.0;
                let members = self.graph.components[after_id].clone();
                let mut before_ids = Vec::new();
                for &vertex in &members {
                    let id = old_graph.component_of[vertex];
                    if !before_ids.contains(&id) {
                        before_ids.push(id);
                    }
                }
                before_ids.sort_unstable();

                merges.push(Merge {
                    edge_index,
                    before: before_ids,
                    after: after_id,
                    members,
                });
            }

            steps.push(Step {
                edge_index,
                status,
                before,
                after,
            });
        }

        self.history.extend(merges.iter().cloned());
        Ok(Batch {
            steps,
            merges,
            graph: self.graph.clone(),
        })
    }
}
