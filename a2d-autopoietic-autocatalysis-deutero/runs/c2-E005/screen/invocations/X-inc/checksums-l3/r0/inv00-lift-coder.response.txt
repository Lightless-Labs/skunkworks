#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;

        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }

        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        update_base(data, &mut self.crc, &mut self.adler_a, &mut self.adler_b);
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut crc = 0u32;
        for (offset, &byte) in bytes[1..9].iter().enumerate() {
            crc = (crc << 4) | u32::from(hex_digit(byte, offset + 1)?);
        }

        let mut adler = 0u32;
        for (index, &byte) in bytes[11..19].iter().enumerate() {
            adler = (adler << 4) | u32::from(hex_digit(byte, index + 11)?);
        }

        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn hex_digit(byte: u8, offset: usize) -> Result<u8, SnapshotError> {
    match byte {
        b'0'..=b'9' => Ok(byte - b'0'),
        b'a'..=b'f' => Ok(byte - b'a' + 10),
        _ => Err(SnapshotError::Hex { offset }),
    }
}

fn update_base(data: &[u8], crc: &mut u32, a: &mut u32, b: &mut u32) {
    for &byte in data {
        *crc ^= u32::from(byte);
        for _ in 0..8 {
            if *crc & 1 != 0 {
                *crc = (*crc >> 1) ^ 0xedb8_8320;
            } else {
                *crc >>= 1;
            }
        }

        *a = (*a + u32::from(byte)) % 65_521;
        *b = (*b + *a) % 65_521;
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    update_base(data, &mut crc, &mut a, &mut b);
    crc ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 65_521;
        b = (b + a) % 65_521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }

    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for span in spans {
        update_base(
            &data[span.start..span.end],
            &mut crc,
            &mut a,
            &mut b,
        );
    }

    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;

    for &byte in data {
        a = (a + u32::from(byte)) % 255;
        b = (b + a) % 255;
    }

    ((b as u16) << 8) | a as u16
}

fn fletcher32_update_word(a: &mut u32, b: &mut u32, word: u16) {
    *a = (*a + u32::from(word)) % 65_535;
    *b = (*b + *a) % 65_535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut chunks = data.chunks_exact(2);

    for pair in &mut chunks {
        let word = (u16::from(pair[0]) << 8) | u16::from(pair[1]);
        fletcher32_update_word(&mut a, &mut b, word);
    }

    if let Some(&last) = chunks.remainder().first() {
        fletcher32_update_word(&mut a, &mut b, u16::from(last) << 8);
    }

    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut a = 0u32;
    let mut b = 0u32;
    let mut pending_high = None;

    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending_high.take() {
                let word = (u16::from(high) << 8) | u16::from(byte);
                fletcher32_update_word(&mut a, &mut b, word);
            } else {
                pending_high = Some(byte);
            }
        }
    }

    if let Some(high) = pending_high {
        fletcher32_update_word(&mut a, &mut b, u16::from(high) << 8);
    }

    Ok((b << 16) | a)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            let mut crc = match variant {
                Crc16::Xmodem => 0u16,
                Crc16::False => 0xffffu16,
                Crc16::X25 => unreachable!(),
            };

            for &byte in data {
                crc ^= u16::from(byte) << 8;
                for _ in 0..8 {
                    let top = crc & 0x8000 != 0;
                    crc = crc.wrapping_shl(1);
                    if top {
                        crc ^= 0x1021;
                    }
                }
            }
            crc
        }
        Crc16::X25 => {
            let mut crc = 0xffffu16;
            for &byte in data {
                crc ^= u16::from(byte);
                for _ in 0..8 {
                    let low = crc & 1 != 0;
                    crc >>= 1;
                    if low {
                        crc ^= 0x8408;
                    }
                }
            }
            crc ^ 0xffff
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut crc = 0xffff_ffffu32;
    let mut adler_a = 1u32;
    let mut adler_b = 0u32;

    let mut f16_a = 0u32;
    let mut f16_b = 0u32;

    let mut f32_a = 0u32;
    let mut f32_b = 0u32;
    let mut pending_high = None;

    let mut crc16_state = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffffu16,
    };

    for span in spans {
        for &byte in &data[span.start..span.end] {
            update_base(
                std::slice::from_ref(&byte),
                &mut crc,
                &mut adler_a,
                &mut adler_b,
            );

            f16_a = (f16_a + u32::from(byte)) % 255;
            f16_b = (f16_b + f16_a) % 255;

            if let Some(high) = pending_high.take() {
                fletcher32_update_word(
                    &mut f32_a,
                    &mut f32_b,
                    (u16::from(high) << 8) | u16::from(byte),
                );
            } else {
                pending_high = Some(byte);
            }

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    crc16_state ^= u16::from(byte) << 8;
                    for _ in 0..8 {
                        let top = crc16_state & 0x8000 != 0;
                        crc16_state = crc16_state.wrapping_shl(1);
                        if top {
                            crc16_state ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    crc16_state ^= u16::from(byte);
                    for _ in 0..8 {
                        let low = crc16_state & 1 != 0;
                        crc16_state >>= 1;
                        if low {
                            crc16_state ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    if let Some(high) = pending_high {
        fletcher32_update_word(&mut f32_a, &mut f32_b, u16::from(high) << 8);
    }

    if variant == Crc16::X25 {
        crc16_state ^= 0xffff;
    }

    Ok(Multi {
        base: Hashes {
            crc32: crc ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: (f32_b << 16) | f32_a,
        crc16: crc16_state,
    })
}
