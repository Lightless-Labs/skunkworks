#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

const ADLER_MOD: u64 = 65_521;

#[derive(Debug, Clone, Copy)]
struct State {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
}

impl State {
    fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    fn from_seed(crc32: u32, adler32: u32) -> Self {
        Self {
            crc: crc32 ^ 0xffff_ffff,
            adler_a: adler32 & 0xffff,
            adler_b: adler32 >> 16,
        }
    }

    fn update_byte(&mut self, byte: u8) {
        self.crc ^= u32::from(byte);
        for _ in 0..8 {
            let low = self.crc & 1;
            self.crc >>= 1;
            if low != 0 {
                self.crc ^= 0xedb8_8320;
            }
        }

        self.adler_a = ((u64::from(self.adler_a) + u64::from(byte)) % ADLER_MOD) as u32;
        self.adler_b = ((u64::from(self.adler_b) + u64::from(self.adler_a)) % ADLER_MOD) as u32;
    }

    fn checksums(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut state = State::new();
    for &byte in data {
        state.update_byte(byte);
    }
    state.checksums().crc32
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut state = State::new();
    for &byte in data {
        state.update_byte(byte);
    }
    state.checksums().adler32
}

pub fn checksums(data: &[u8]) -> Hashes {
    let mut state = State::new();
    for &byte in data {
        state.update_byte(byte);
    }
    state.checksums()
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }

    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = State::new();
    for span in spans {
        for &byte in &data[span.start..span.end] {
            state.update_byte(byte);
        }
    }
    Ok(state.checksums())
}

#[derive(Debug, Clone)]
pub struct Stream {
    state: State,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

impl Stream {
    pub fn new() -> Self {
        Self { state: State::new() }
    }

    pub fn checksum(&self) -> Hashes {
        self.state.checksums()
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= ADLER_MOD as u32 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= ADLER_MOD as u32 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            state: State::from_seed(crc32, adler32),
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.state.update_byte(byte);
        }
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            for &byte in &data[span.start..span.end] {
                self.state.update_byte(byte);
            }
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        self.state = State::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut values = [0u32; 2];
        for (i, &byte) in bytes[1..9].iter().chain(bytes[11..19].iter()).enumerate() {
            let digit = match byte {
                b'0'..=b'9' => u32::from(byte - b'0'),
                b'a'..=b'f' => u32::from(byte - b'a' + 10),
                _ => {
                    let offset = if i < 8 { i + 1 } else { i + 3 };
                    return Err(SnapshotError::Hex { offset });
                }
            };
            let slot = usize::from(i >= 8);
            values[slot] = (values[slot] << 4) | digit;
        }

        Self::from_seed(values[0], values[1])
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
struct Fletcher32State {
    a: u32,
    b: u32,
    pending_high: Option<u8>,
}

impl Fletcher32State {
    fn new() -> Self {
        Self {
            a: 0,
            b: 0,
            pending_high: None,
        }
    }

    fn update_byte(&mut self, byte: u8) {
        if let Some(high) = self.pending_high.take() {
            self.update_word((u16::from(high) << 8) | u16::from(byte));
        } else {
            self.pending_high = Some(byte);
        }
    }

    fn update_word(&mut self, word: u16) {
        const MOD: u64 = 65_535;
        self.a = ((u64::from(self.a) + u64::from(word)) % MOD) as u32;
        self.b = ((u64::from(self.b) + u64::from(self.a)) % MOD) as u32;
    }

    fn finish(mut self) -> u32 {
        if let Some(high) = self.pending_high.take() {
            self.update_word(u16::from(high) << 8);
        }
        (self.b << 16) | self.a
    }
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut state = Fletcher32State::new();
    for &byte in data {
        state.update_byte(byte);
    }
    state.finish()
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = Fletcher32State::new();
    for span in spans {
        for &byte in &data[span.start..span.end] {
            state.update_byte(byte);
        }
    }
    Ok(state.finish())
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

fn crc16_update_byte(mut crc: u16, byte: u8, variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            crc ^= u16::from(byte) << 8;
            for _ in 0..8 {
                if crc & 0x8000 != 0 {
                    crc = (crc << 1) ^ 0x1021;
                } else {
                    crc <<= 1;
                }
            }
            crc
        }
        Crc16::X25 => {
            crc ^= u16::from(byte);
            for _ in 0..8 {
                if crc & 1 != 0 {
                    crc = (crc >> 1) ^ 0x8408;
                } else {
                    crc >>= 1;
                }
            }
            crc
        }
    }
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut crc = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    for &byte in data {
        crc = crc16_update_byte(crc, byte, variant);
    }
    if variant == Crc16::X25 {
        crc ^ 0xffff
    } else {
        crc
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut state = State::new();
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32_state = Fletcher32State::new();
    let mut c16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        for &byte in &data[span.start..span.end] {
            state.update_byte(byte);

            f16_a = (f16_a + u32::from(byte)) % 255;
            f16_b = (f16_b + f16_a) % 255;

            f32_state.update_byte(byte);
            c16 = crc16_update_byte(c16, byte, variant);
        }
    }

    if variant == Crc16::X25 {
        c16 ^= 0xffff;
    }

    Ok(Multi {
        base: state.checksums(),
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: f32_state.finish(),
        crc16: c16,
    })
}
