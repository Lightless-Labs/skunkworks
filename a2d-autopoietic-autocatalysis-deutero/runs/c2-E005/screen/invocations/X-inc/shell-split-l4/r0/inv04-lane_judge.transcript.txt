fn append_text(word: &mut Word, text: &str, wildcard_active: bool) {
    for c in text.chars() {
        let offset = word.text.len();
        if wildcard_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(offset);
        }
        word.text.push(c);
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

fn parse_expansion(input: &str, dollar: usize, env: &[(&str, &str)]) -> Result<(String, usize), SplitError> {
    let bytes = input.as_bytes();
    let mut close = None;
    let mut i = dollar + 2;

    while i < bytes.len() {
        if bytes[i] == b'\\' {
            i += 1;
            if i < bytes.len() {
                let c = input[i..].chars().next().unwrap();
                i += c.len_utf8();
            }
        } else if bytes[i] == b'}' {
            close = Some(i);
            break;
        } else {
            let c = input[i..].chars().next().unwrap();
            i += c.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[dollar + 2..close];
    let (name, default) = if let Some(split) = contents.find(":-") {
        (&contents[..split], Some(&contents[split + 2..]))
    } else {
        (contents, None)
    };

    let mut name_chars = name.chars();
    let valid_name = name_chars.next().is_some_and(is_name_start)
        && name_chars.all(is_name_char);

    if !valid_name {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let value = match default {
        Some(text) => {
            let found = lookup(name, env);
            if found.is_empty() {
                decode_default(text)
            } else {
                found.to_string()
            }
        }

    };

    Ok((value, close + 1))

fn scan_name(input: &str, start: usize) -> Option<(&str, usize)> {
    let mut iter = input[start..].char_indices();
    let (_, first) = iter.next()?;
    if !is_name_start(first) {
        return None;
    }
    let mut end = start + first.len_utf8();
    for (offset, c) in iter {
        if !is_name_char(c) {
            break;
        }
        end = start + offset + c.len_utf8();
    }
    Some((&input[start..end], end))
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{word}'"))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

fn fallback_candidate(word: &str) -> String {
    let mut out = String::from("'");
    for c in word.chars() {
        if c == '\'' {
            out.push_str("'\\''");
        } else {
            out.push(c);
        }
    }
    out.push('\'');
    out
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let candidates = [
            bare_candidate(word),
            single_candidate(word),
            double_candidate(word),
        ];

        let best = candidates
            .into_iter()
            .flatten()
            .min_by_key(|candidate| candidate.len())
            .unwrap_or_else(|| fallback_candidate(word));
        encoded.push(best);
    }

    Ok(encoded.join(" "))
}

fn find_braced_end(input: &str, mut index: usize) -> Option<usize> {
    while index < input.len() {
        let c = input[index..].chars().next()?;
        if c == '\\' {
            index += c.len_utf8();
            if index < input.len() {
                let next = input[index..].chars().next()?;
                index += next.len_utf8();
            }
        } else if c == '}' {
            return Some(index);
        } else {
            index += c.len_utf8();
        }
    }
    None
}

    let default = if name_end == contents.len() {
        None
    } else if contents[name_end..].starts_with(":-") {
        Some(&contents[name_end + 2..])
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    let value = match (lookup(env, name), default) {
        (Some(value), None) => value.to_string(),
        (Some(value), Some(_)) if !value.is_empty() => value.to_string(),
        (_, Some(text)) => decode_default(text),
        (None, None) => String::new(),
    };

    append_expansion(word, &value, active_glob);
    Ok(close + 1)
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_open = 0;
    let mut index = 0;

    while index < input.len() {
        let c = input[index..].chars().next().unwrap();
        let next_index = index + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_literal(&mut current, c, false);
                }
                index = next_index;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    index = next_index;
                } else if c == '\\' {
                    let escape_pos = index;
                    if next_index >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    let next = input[next_index..].chars().next().unwrap();
                    let after_next = next_index + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next == '$' {
                            append_literal(&mut current, '$', false);
                            index = after_next;
                        } else if next != '\n' {
                            append_literal(&mut current, next, false);
                            index = after_next;
                        } else {
                            index = after_next;
                        }
                    } else {
                        append_literal(&mut current, '\\', false);
                        index = next_index;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: index });
                } else if c == '$' {
                    if input[next_index..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: index });
                    }
                    if input[next_index..].starts_with('{') {
                        index = expand_braced(input, index, env, &mut current, false)?;
                    } else if next_index < input.len()
                        && input[next_index..]
                            .chars()
                            .next()
                            .is_some_and(is_name_start)
                    {
                        let mut end = next_index;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[next_index..end];
                        let value = lookup(env, name).unwrap_or("");
                        append_expansion(&mut current, value, false);
                        index = end;
                    } else {
                        append_literal(&mut current, '$', false);
                        index = next_index;
                    }
                } else {
                    append_literal(&mut current, c, false);
                    index = next_index;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    index = next_index;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_open = index;
                    active = true;
                    index = next_index;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_open = index;
                    active = true;
                    index = next_index;
                } else if c == '\\' {
                    if next_index >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: index });
                    }
                    let next = input[next_index..].chars().next().unwrap();
                    let after_next = next_index + next.len_utf8();
                    if next != '\n' {
                        active = true;
                        append_literal(&mut current, next, false);
                    }
                    index = after_next;
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: index });
                } else if c == '$' {
                    if input[next_index..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: index });
                    }
                    if input[next_index..].starts_with('{') {
                        active = true;
                        index = expand_braced(input, index, env, &mut current, true)?;
                    } else if next_index < input.len()
                        && input[next_index..]
                            .chars()
                            .next()
                            .is_some_and(is_name_start)
                    {
                        let mut end = next_index;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[next_index..end];
                        let value = lookup(env, name).unwrap_or("");
                        append_expansion(&mut current, value, true);
                        active = true;
                        index = end;
                    } else {
                        append_literal(&mut current, '$', true);
                        active = true;
                        index = next_index;
                    }
                } else {
                    append_literal(&mut current, c, true);
                    active = true;
                    index = next_index;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_open }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_open }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }

    let mut result = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            result.push(c);
        } else {
            result.push('\\');
            result.push(c);
        }
    }
    Some(result)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        return None;
    }
    Some(format!("'{word}'"))
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }

    let mut result = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            result.push('\\');
        }
        result.push(c);
    }
    result.push('"');
    Some(result)
}

fn fallback_candidate(word: &str) -> String {
    let mut result = String::from("'");
    for c in word.chars() {
        if c == '\'' {
            result.push('\'');
            result.push('\\');
            result.push('\'');
            result.push('\'');
        } else {
            result.push(c);
        }
    }
    result.push('\'');
    result
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates = Vec::new();
        if let Some(candidate) = bare_candidate(word) {
            candidates.push((candidate, 0usize));
        }
        if let Some(candidate) = single_candidate(word) {
            candidates.push((candidate, 1usize));
        }
        if let Some(candidate) = double_candidate(word) {
            candidates.push((candidate, 2usize));
        }

        let candidate = if candidates.is_empty() {
            fallback_candidate(word)
        } else {
            candidates
                .into_iter()
                .min_by_key(|(candidate, priority)| (candidate.len(), *priority))
                .unwrap()
                .0
        };
        encoded.push(candidate);
    }

    Ok(encoded.join(" "))
}

fn scalar_at(input: &str, pos: usize) -> Option<(char, usize)> {
    input[pos..]
        .chars()
        .next()
        .map(|c| (c, pos + c.len_utf8()))
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let content_start = dollar + 2;
    let mut cursor = content_start;
    let mut closing = None;

    while cursor < input.len() {
        let (c, next) = scalar_at(input, cursor).expect("valid UTF-8 boundary");
        if c == '\\' {
            if next < input.len() {
                let (_, after) = scalar_at(input, next).expect("valid UTF-8 boundary");
                cursor = after;
            } else {
                cursor = next;
            }
        } else if c == '}' {
            closing = Some(cursor);
            break;
        } else {
            cursor = next;
        }
    }

    let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[content_start..close];

    let name_len = contents
        .char_indices()
        .take_while(|(_, c)| is_name_continue(*c))
        .map(|(i, c)| i + c.len_utf8())
        .last()
        .unwrap_or(0);

    if name_len == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let mut chars = contents[..name_len].chars();
    if !chars.next().is_some_and(is_name_start) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_len];
    let rest = &contents[name_len..];

    let value = if rest.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default_text) = rest.strip_prefix(":-") {
        match lookup(name, env) {
            Some(v) if !v.is_empty() => v.to_string(),
            _ => decode_default(default_text),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close + 1, value))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut cursor = 0;

    while cursor < text.len() {
        let (c, next) = scalar_at(text, cursor).expect("valid UTF-8 boundary");
        if c == '\\' && next < text.len() {
            let (escaped, after) = scalar_at(text, next).expect("valid UTF-8 boundary");
            result.push(escaped);
            cursor = after;
        } else {
            result.push(c);
            cursor = next;
        }
    }

    result
}

            }

        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),

    }

        words.push(current);
    }

fn encode_word(word: &str) -> String {
    let candidates = [
        bare_candidate(word),
        single_candidate(word),
        double_candidate(word),
    ];

    let mut best: Option<(usize, String)> = None;
    for candidate in candidates.into_iter().flatten() {
        let len = candidate.len();
        if best.as_ref().is_none_or(|(best_len, _)| len < *best_len) {
            best = Some((len, candidate));
        }
    }

    best.map(|(_, value)| value)
        .unwrap_or_else(|| fallback_candidate(word))
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.as_bytes().iter().position(|byte| *byte == 0) {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut result = String::new();
    for (index, word) in words.iter().enumerate() {
        if index != 0 {
            result.push(' ');
        }
        let encoded = encode_word(word);
        let _ = write!(&mut result, "{encoded}");
    }
    Ok(result)
}

struct Parser<'a> {
    input: &'a str,
    env: &'a [(&'a str, &'a str)],
    pos: usize,
    quote: Quote,
    quote_start: usize,
    text: String,
    glob: Vec<usize>,
    active: bool,
}

impl<'a> Parser<'a> {
    fn new(input: &'a str, env: &'a [(&'a str, &'a str)]) -> Self {
        Self {
            input,
            env,
            pos: 0,
            quote: Quote::None,
            quote_start: 0,
            text: String::new(),
            glob: Vec::new(),
            active: false,
        }
    }

    fn peek(&self) -> Option<char> {
        self.input[self.pos..].chars().next()
    }

    fn peek_after(&self, byte_offset: usize) -> Option<char> {
        self.input[self.pos + byte_offset..].chars().next()
    }

    fn append(&mut self, c: char, wildcard_active: bool) {
        if wildcard_active && matches!(c, '*' | '?' | '[') {
            self.glob.push(self.text.len());
        }
        self.text.push(c);
        self.active = true;
    }

    fn append_str(&mut self, s: &str, wildcard_active: bool) {
        for c in s.chars() {
            self.append(c, wildcard_active);
        }
        self.active = true;
    }

    fn finish_word(&mut self, words: &mut Vec<Word>) {
        if self.active {
            words.push(Word {
                text: std::mem::take(&mut self.text),
                glob: std::mem::take(&mut self.glob),
            });
            self.active = false;
        }
    }

    fn env_value(&self, name: &str) -> Option<&'a str> {
        self.env
            .iter()
            .find_map(|(key, value)| (*key == name).then_some(*value))
    }

    fn scan_braced(&self, dollar: usize) -> Result<(usize, String), SplitError> {
        let content_start = dollar + 2;
        let mut i = content_start;
        let mut closing = None;

        while i < self.input.len() {
            let c = self.input[i..].chars().next().unwrap();
            if c == '\\' {
                i += c.len_utf8();
                if i < self.input.len() {
                    i += self.input[i..].chars().next().unwrap().len_utf8();
                }
            } else if c == '}' {
                closing = Some(i);
                break;
            } else {
                i += c.len_utf8();
            }
        }

        let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let content = &self.input[content_start..close];

        let (name, default) = if let Some(operator) = content.find(":-") {
            (&content[..operator], Some(&content[operator + 2..]))
        } else {
            (content, None)
        };

        let mut chars = name.chars();
        let valid_start = chars
            .next()
            .is_some_and(|c| c == '_' || c.is_ascii_alphabetic());
        let valid_rest = chars.all(|c| c == '_' || c.is_ascii_alphanumeric());
        if !valid_start || !valid_rest {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let value = match default {
            Some(text) => match self.env_value(name) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(text),
            },
            None => self.env_value(name).unwrap_or("").to_string(),
        };

        Ok((close + 1, value))
    }

    fn parse(mut self) -> Result<Vec<Word>, SplitError> {
        if let Some((pos, _)) = self.input.char_indices().find(|(_, c)| *c == '\0') {
            return Err(SplitError::Nul { pos });
        }

        let mut words = Vec::new();

        while let Some(c) = self.peek() {
            match self.quote {
                Quote::Single => {
                    self.pos += c.len_utf8();
                    if c == '\'' {
                        self.quote = Quote::None;
                    } else {
                        self.append(c, false);
                    }
                }
                Quote::Double => {
                    if c == '"' {
                        self.pos += 1;
                        self.quote = Quote::None;
                    } else if c == '\\' {
                        let slash = self.pos;
                        self.pos += 1;
                        let Some(next) = self.peek() else {
                            return Err(SplitError::TrailingEscape { pos: slash });
                        };
                        if matches!(next, '"' | '\\' | '$' | '\n') {
                            self.pos += next.len_utf8();
                            if next != '\n' {
                                self.append(next, false);
                            }
                        } else {
                            self.append('\\', false);
                        }
                    } else if c == '`' {
                        return Err(SplitError::CommandSubstitution { pos: self.pos });
                    } else if c == '$' {
                        let dollar = self.pos;
                        let next_pos = dollar + 1;
                        if self.input[next_pos..].starts_with('(') {
                            return Err(SplitError::CommandSubstitution { pos: dollar });
                        }
                        if self.input[next_pos..].starts_with('{') {
                            let (end, value) = self.scan_braced(dollar)?;
                            self.pos = end;
                            self.append_str(&value, false);
                        } else if self.peek_after(1).is_some_and(is_name_start) {
                            let start = next_pos;
                            let mut end = start;
                            while end < self.input.len() {
                                let ch = self.input[end..].chars().next().unwrap();
                                if !is_name_char(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &self.input[start..end];
                            let value = self.env_value(name).unwrap_or("");
                            self.pos = end;
                            self.append_str(value, false);
                        } else {
                            self.pos += 1;
                            self.append('$', false);
                        }
                    } else {
                        self.pos += c.len_utf8();
                        self.append(c, false);
                    }
                }
                Quote::None => {
                    if is_separator(c) {
                        self.pos += c.len_utf8();
                        self.finish_word(&mut words);
                    } else if c == '\'' {
                        self.quote = Quote::Single;
                        self.quote_start = self.pos;
                        self.active = true;
                        self.pos += 1;
                    } else if c == '"' {
                        self.quote = Quote::Double;
                        self.quote_start = self.pos;
                        self.active = true;
                        self.pos += 1;
                    } else if c == '\\' {
                        let slash = self.pos;
                        self.pos += 1;
                        let Some(next) = self.peek() else {
                            return Err(SplitError::TrailingEscape { pos: slash });
                        };
                        self.pos += next.len_utf8();
                        if next != '\n' {
                            self.append(next, false);
                        }
                    } else if c == '`' {
                        return Err(SplitError::CommandSubstitution { pos: self.pos });
                    } else if c == '$' {
                        let dollar = self.pos;
                        let next_pos = dollar + 1;
                        if self.input[next_pos..].starts_with('(') {
                            return Err(SplitError::CommandSubstitution { pos: dollar });
                        }
                        if self.input[next_pos..].starts_with('{') {
                            let (end, value) = self.scan_braced(dollar)?;
                            self.pos = end;
                            self.append_str(&value, true);
                        } else if self.peek_after(1).is_some_and(is_name_start) {
                            let start = next_pos;
                            let mut end = start;
                            while end < self.input.len() {
                                let ch = self.input[end..].chars().next().unwrap();
                                if !is_name_char(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &self.input[start..end];
                            let value = self.env_value(name).unwrap_or("");
                            self.pos = end;
                            self.append_str(value, true);
                        } else {
                            self.pos += 1;
                            self.append('$', false);
                        }
                    } else {
                        self.pos += c.len_utf8();
                        self.append(c, true);
                    }
                }
            }
        }

        match self.quote {
            Quote::Single => {
                return Err(SplitError::UnterminatedSingle {
                    pos: self.quote_start,
                });
            }
            Quote::Double => {
                return Err(SplitError::UnterminatedDouble {
                    pos: self.quote_start,
                });
            }
            Quote::None => {}
        }

        self.finish_word(&mut words);
        Ok(words)
    }
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_char(c: char) -> bool {
    c == '_' || c.is_ascii_alphanumeric()
}

fn decode_default(text: &str) -> String {
    let mut output = String::new();
    let mut chars = text.chars();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                output.push(next);
            } else {
                output.push('\\');
            }
        } else {
            output.push(c);
        }
    }
    output
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    Parser::new(input, env).parse()
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric()
            || matches!(c, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%')
        {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

fn fallback_candidate(word: &str) -> String {
    let mut out = String::from("'");
    for c in word.chars() {
        if c == '\'' {
            out.push('\'');
            out.push('\\');
            out.push('\'');
            out.push('\'');
        } else {
            out.push(c);
        }
    }
    out.push('\'');
    out
}

fn encode_word(word: &str) -> String {
    let mut candidates: Vec<(usize, usize, String)> = Vec::new();

    if let Some(candidate) = bare_candidate(word) {
        candidates.push((candidate.len(), 0, candidate));
    }
    if let Some(candidate) = single_candidate(word) {
        candidates.push((candidate.len(), 1, candidate));
    }
    if let Some(candidate) = double_candidate(word) {
        candidates.push((candidate.len(), 2, candidate));
    }

    if candidates.is_empty() {
        return fallback_candidate(word);
    }

    candidates.sort_by_key(|(len, priority, _)| (*len, *priority));
    candidates.remove(0).2
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    Ok(words
        .iter()
        .map(|word| encode_word(word))
        .collect::<Vec<_>>()
        .join(" "))
}
