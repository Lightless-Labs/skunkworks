#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    wildcard: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_static(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || is_unreserved(value) {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(base + i);
            }
            decoded.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b[0].is_ascii_lowercase()
        && b[1..]
            .iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_')
}

fn parse_path<'a>(input: &'a str, pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let err = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };
    if !input.starts_with('/') {
        return Err(err(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut pieces: Vec<(usize, &str)> = Vec::new();
    let bytes = input.as_bytes();
    let mut start = 1;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            pieces.push((start, &input[start..i]));
            start = i + 1;
        }
    }
    let trailing = input.ends_with('/');
    if trailing {
        // The final empty piece represents the permitted trailing slash.
        if start < input.len() {
            pieces.push((start, &input[start..]));
        }
    } else {
        pieces.push((start, &input[start..]));
    }

    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();
    for (idx, (at, raw)) in pieces.iter().copied().enumerate() {
        if raw.is_empty() {
            return Err(err(at));
        }
        let first = raw.as_bytes()[0];
        if pattern && (first == b':' || first == b'*') {
            let name = &raw[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(err(at));
            }
            let wildcard = first == b'*';
            if wildcard && idx + 1 != pieces.len() {
                return Err(err(at));
            }
            segments.push(if wildcard {
                Segment::Wildcard(name.to_owned())
            } else {
                Segment::Param(name.to_owned())
            });
        } else {
            let decoded = decode_static(raw, at).map_err(err)?;
            segments.push(Segment::Static(decoded));
        }
    }
    Ok((segments, trailing))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing) = parse_path(pattern, true)?;
        let wildcard = matches!(segments.last(), Some(Segment::Wildcard(_)));
        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_owned(),
            segments,
            trailing,
            wildcard,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request, request_trailing) = parse_path(head, false)?;

        let mut candidates = Vec::new();
        for (id, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            if let Some(params) = match_segments(&route.segments, &request) {
                candidates.push((id, params));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let allowed_methods = {
            let mut methods: Vec<String> = candidates
                .iter()
                .map(|(id, _)| self.routes[*id].method.clone())
                .collect();
            if methods.iter().any(|m| m == "GET") {
                methods.push("HEAD".to_owned());
            }
            methods.sort();
            methods.dedup();
            methods
        };

        let effective_method = if method == "HEAD"
            && candidates
                .iter()
                .any(|(id, _)| self.routes[*id].method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = candidates
            .into_iter()
            .filter(|(id, _)| self.routes[*id].method == effective_method)
            .collect();
        if eligible.is_empty() {
            return Err(Error::MethodNotAllowed {
                allowed: allowed_methods,
            });
        }

        let mut best: Option<(usize, Vec<(String, String)>, Vec<u8>)> = None;
        for (id, params) in eligible {
            let route = &self.routes[id];
            let rank = rank(&route.segments, route.wildcard);
            if best.as_ref().map_or(true, |(_, _, r)| rank > *r) {
                best = Some((id, params, rank));
            }
        }
        let (id, params, _) = best.expect("eligible candidates are nonempty");
        let redirect = if self.policy == SlashPolicy::Redirect
            && self.routes[id].trailing != request_trailing
            && !head.is_empty()
            && head != "/"
        {
            if self.routes[id].trailing {
                Some(format!("{head}/"))
            } else {
                Some(head[..head.len() - 1].to_owned())
            }
        } else {
            None
        };
        Ok(Match {
            route: id,
            params,
            redirect,
        })
    }
}

fn rank(segments: &[Segment], wildcard: bool) -> Vec<u8> {
    let mut out = Vec::new();
    for segment in segments {
        match segment {
            Segment::Static(_) => out.push(2),
            Segment::Param(_) => out.push(1),
            Segment::Wildcard(_) => {
                out.push(0);
                return out;
            }
        }
    }
    if !wildcard {
        out.push(3);
    }
    out
}

fn match_segments(pattern: &[Segment], path: &[Segment]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut pi = 0;
    for segment in pattern {
        match segment {
            Segment::Wildcard(name) => {
                let value = path[pi..]
                    .iter()
                    .map(|s| match s {
                        Segment::Static(v) => v.as_str(),
                        _ => unreachable!(),
                    })
                    .collect::<Vec<_>>()
                    .join("/");
                params.push((name.clone(), value));
                return Some(params);
            }
            Segment::Static(expected) => {
                if !matches!(path.get(pi), Some(Segment::Static(actual)) if actual == expected) {
                    return None;
                }
            }
            Segment::Param(name) => {
                let Some(Segment::Static(value)) = path.get(pi) else {
                    return None;
                };
                params.push((name.clone(), value.clone()));
            }
        }
        pi += 1;
    }
    (pi == path.len()).then_some(params)
}
