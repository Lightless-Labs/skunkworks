#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
    id: usize,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

#[derive(Clone)]
struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn invalid_pattern() -> Error {
    Error::InvalidPattern { at: 0 }
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool, Vec<u8>), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false, vec![3]));
    }

    let bytes = pattern.as_bytes();
    let mut segments = Vec::new();
    let mut starts = Vec::new();
    let mut pos = 1;
    while pos <= bytes.len() {
        let end = bytes[pos..]
            .iter()
            .position(|&b| b == b'/')
            .map(|i| pos + i)
            .unwrap_or(bytes.len());
        starts.push(pos);
        segments.push(&pattern[pos..end]);
        if end == bytes.len() {
            break;
        }
        pos = end + 1;
    }

    let trailing = segments.last().is_some_and(|s| s.is_empty());
    if trailing {
        segments.pop();
        starts.pop();
    }

    let mut parsed = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut rank = Vec::new();

    for (i, raw) in segments.iter().enumerate() {
        let at = starts[i];
        if raw.is_empty() {
            return Err(Error::InvalidPattern { at });
        }

        if let Some(name) = raw.strip_prefix(':') {
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at });
            }
            parsed.push(Segment::Param(name.to_owned()));
            rank.push(1);
        } else if let Some(name) = raw.strip_prefix('*') {
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at });
            }
            if i + 1 != segments.len() {
                return Err(Error::InvalidPattern { at });
            }
            parsed.push(Segment::Wildcard(name.to_owned()));
            rank.push(0);
        } else {
            let decoded = decode_segment(raw, at).map_err(|at| Error::InvalidPattern { at })?;
            parsed.push(Segment::Static(decoded));
            rank.push(2);
        }
    }

    if !parsed.last().is_some_and(|s| matches!(s, Segment::Wildcard(_))) {
        rank.push(3);
    }

    Ok((parsed, trailing, rank))
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b[0].is_ascii_lowercase()
        && b[1..]
            .iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_')
}

fn decode_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !bytes[i + 1].is_ascii_hexdigit()
                || !bytes[i + 2].is_ascii_hexdigit()
                || bytes[i + 1].is_ascii_lowercase()
                || bytes[i + 2].is_ascii_lowercase()
            {
                return Err(base + i);
            }
            let value = (hex_value(bytes[i + 1]).unwrap() << 4) | hex_value(bytes[i + 2]).unwrap();
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            out.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(base + i);
            }
            out.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        b'a'..=b'f' => Some(b - b'a' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn parse_path(path: &str) -> Result<(ParsedPath, String), Error> {
    let cut = path
        .char_indices()
        .find(|(_, c)| *c == '?' || *c == '#')
        .map(|(i, _)| i)
        .unwrap_or(path.len());
    let head = &path[..cut];

    if !head.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if head == "/" {
        return Ok((
            ParsedPath {
                segments: Vec::new(),
                trailing: false,
            },
            head.to_owned(),
        ));
    }

    let bytes = head.as_bytes();
    let mut segments = Vec::new();
    let mut pos = 1;
    while pos <= bytes.len() {
        let end = bytes[pos..]
            .iter()
            .position(|&b| b == b'/')
            .map(|i| pos + i)
            .unwrap_or(bytes.len());

        if end == pos {
            if end == bytes.len() {
                break;
            }
            return Err(Error::InvalidPath { at: pos });
        }

        let decoded =
            decode_segment(&head[pos..end], pos).map_err(|at| Error::InvalidPath { at })?;
        segments.push(decoded);

        if end == bytes.len() {
            break;
        }
        pos = end + 1;
        if pos == bytes.len() {
            break;
        }
    }

    let trailing = head.len() > 1 && head.ends_with('/');
    Ok((
        ParsedPath { segments, trailing },
        head.to_owned(),
    ))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing, rank) = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_owned(),
            segments,
            trailing,
            rank,
            id,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (parsed, head) = parse_path(path)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = self
            .routes
            .iter()
            .filter_map(|route| {
                if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                    return None;
                }
                match_route(route, &parsed.segments).map(|params| (route, params))
            })
            .collect();

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<usize> = if method == "HEAD" {
            if candidates.iter().any(|(r, _)| r.method == "HEAD") {
                candidates
                    .iter()
                    .enumerate()
                    .filter_map(|(i, (r, _))| (r.method == "HEAD").then_some(i))
                    .collect()
            } else {
                candidates
                    .iter()
                    .enumerate()
                    .filter_map(|(i, (r, _))| (r.method == "GET").then_some(i))
                    .collect()
            }
        } else {
            candidates
                .iter()
                .enumerate()
                .filter_map(|(i, (r, _))| (r.method == method).then_some(i))
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(r, _)| r.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let best_index = *eligible
            .iter()
            .max_by(|&&a, &&b| {
                candidates[a]
                    .0
                    .rank
                    .cmp(&candidates[b].0.rank)
                    .then_with(|| candidates[b].0.id.cmp(&candidates[a].0.id))
            })
            .unwrap();

        let (route, params) = candidates.swap_remove(best_index);
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != parsed.trailing
            && !route.segments.is_empty()
        {
            if route.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head.trim_end_matches('/').to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_at = route
        .segments
        .iter()
        .position(|s| matches!(s, Segment::Wildcard(_)));
    let fixed_len = wildcard_at.unwrap_or(route.segments.len());

    if wildcard_at.is_none() && path.len() != fixed_len {
        return None;
    }
    if wildcard_at.is_some() && path.len() < fixed_len {
        return None;
    }

    let mut params = Vec::new();
    for (i, segment) in route.segments.iter().take(fixed_len).enumerate() {
        match segment {
            Segment::Static(value) if value != &path[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), path[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if let Some(Segment::Wildcard(name)) = route.segments.get(fixed_len) {
        params.push((name.clone(), path[fixed_len..].join("/")));
    }
    Some(params)
}
