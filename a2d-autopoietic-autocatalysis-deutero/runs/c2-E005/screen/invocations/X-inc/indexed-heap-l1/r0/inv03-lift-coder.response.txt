```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

pub struct IndexedHeap {
    items: Vec<StoredItem>,
}

struct StoredItem {
    item: Item,
    age: u64,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self { items: Vec::new() }
    }

    pub fn len(&self) -> usize {
        self.items.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<(), Error> {
        let canonical = canonicalize_key(key)?;

        if self.items.iter().any(|stored| stored.item.key == canonical) {
            return Err(Error::Duplicate { key: canonical });
        }

        let age = self.items.len() as u64;
        self.items.push(StoredItem {
            item: Item {
                key: canonical,
                priority,
                value: value.to_owned(),
            },
            age,
        });
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.items
            .iter()
            .min_by(|a, b| compare_stored(a, b))
            .map(|stored| stored.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let best = self
            .items
            .iter()
            .enumerate()
            .min_by(|(_, a), (_, b)| compare_stored(a, b))
            .map(|(index, _)| index)?;

        Some(self.items.swap_remove(best).item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let canonical = canonicalize_key(key)?;
        Ok(self
            .items
            .iter()
            .find(|stored| stored.item.key == canonical)
            .map(|stored| stored.item.clone()))
    }
}

fn compare_stored(a: &StoredItem, b: &StoredItem) -> std::cmp::Ordering {
    a.item
        .priority
        .cmp(&b.item.priority)
        .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
        .then_with(|| a.age.cmp(&b.age))
}

fn canonicalize_key(key: &str) -> Result<String, Error> {
    let bytes = key.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }

    if start == end {
        return Err(Error::EmptyKey);
    }

    for (offset, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
        };

        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }

    Ok(key[start..end].to_ascii_lowercase())
}
```
