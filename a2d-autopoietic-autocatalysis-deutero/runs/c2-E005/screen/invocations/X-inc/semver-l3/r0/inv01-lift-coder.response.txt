use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_terminator(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if start == *pos {
        return Err(if is_core_terminator(bytes.get(*pos).copied()) {
            VersionError::MissingCore { at: *pos }
        } else {
            VersionError::InvalidChar { at: *pos }
        });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn identifier_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    numeric_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end && bytes[p] != b'.' {
            if !identifier_char(bytes[p]) {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if id_start == p {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if numeric_pre
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.len() > 1
            && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(id.to_string());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    if s.as_bytes().get(p) != Some(&b'.') {
        return Err(if is_core_terminator(s.as_bytes().get(p).copied()) {
            VersionError::MissingCore { at: p }
        } else {
            VersionError::InvalidChar { at: p }
        });
    }
    p += 1;
    let minor = parse_core(s, &mut p)?;
    if s.as_bytes().get(p) != Some(&b'.') {
        return Err(if is_core_terminator(s.as_bytes().get(p).copied()) {
            VersionError::MissingCore { at: p }
        } else {
            VersionError::InvalidChar { at: p }
        });
    }
    p += 1;
    let patch = parse_core(s, &mut p)?;
    if p < s.len() && s.as_bytes()[p] != b'-' && s.as_bytes()[p] != b'+' {
        return Err(VersionError::InvalidChar { at: p });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < s.len() && s.as_bytes()[p] == b'-' {
        let pre_start = p + 1;
        let plus = s[pre_start..].find('+').map(|x| pre_start + x);
        let pre_end = plus.unwrap_or(s.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        p = pre_end;
    }
    if p < s.len() && s.as_bytes()[p] == b'+' {
        let build_start = p + 1;
        build = parse_identifiers(s, build_start, s.len(), false)?;
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn is_numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_id(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    if a.is_empty() {
        return if b.is_empty() { Ordering::Equal } else { Ordering::Greater };
    }
    if b.is_empty() {
        return Ordering::Less;
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric_id(x);
        let yn = is_numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => compare_numeric_id(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Bound {
    version: Version,
    inclusive: bool,
}

#[derive(Clone, Debug)]
enum Constraint {
    Lower(Bound),
    Upper(Bound),
    Equal(Version),
}

#[derive(Clone, Debug)]
struct Arm {
    constraints: Vec<Constraint>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,

fn cmp_tuple(a: (u64, u64, u64), b: (u64, u64, u64)) -> Ordering {
    a.cmp(&b)
}

fn tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn zero_pre(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major, minor, patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    }
}

fn stable(major: u64, minor: u64, patch: u64) -> Version {
    Version { major, minor, patch, pre: Vec::new(), build: Vec::new() }
}

fn strip_build(mut v: Version) -> Version {
    v.build.clear();
    v
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

#[derive(Clone, Debug)]
struct Token {
    text: String,
    start: usize,
}

fn tokenize(s: &str) -> Vec<Token> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = 0;
    while p < bytes.len() {
        while p < bytes.len() && whitespace(bytes[p]) { p += 1; }
        if p == bytes.len() { break; }
        let start = p;
        while p < bytes.len() && !whitespace(bytes[p]) { p += 1; }
        out.push(Token { text: s[start..p].to_string(), start });
    }
    out
}

fn next_major(major: u64, at: usize) -> Result<Version, RangeError> {
    Ok(stable(major.checked_add(1).ok_or(RangeError::Syntax { at })?, 0, 0))
}

fn next_minor(major: u64, minor: u64, at: usize) -> Result<Version, RangeError> {
    Ok(stable(major, minor.checked_add(1).ok_or(RangeError::Syntax { at })?, 0))
}

fn next_patch(major: u64, minor: u64, patch: u64, at: usize) -> Result<Version, RangeError> {
    Ok(stable(major, minor, patch.checked_add(1).ok_or(RangeError::Syntax { at })?))
}

fn next_core(v: &Version, k: usize, at: usize) -> Result<Version, RangeError> {
    match k {
        1 => next_major(v.major, at),
        2 => next_minor(v.major, v.minor, at),
        _ => next_patch(v.major, v.minor, v.patch, at),
    }
}

#[derive(Clone, Debug)]
enum Operand {
    Full(Version),
    Partial { values: Vec<u64>, k: usize },
}

fn parse_operand(s: &str, token: &Token, op_start: usize) -> Result<Operand, RangeError> {
    let operand = &token.text[op_start..];
    let absolute = token.start + op_start;
    if operand.is_empty() {
        return Err(RangeError::Syntax { at: token.start });
    }
    let has_suffix = operand.contains('-') || operand.contains('+');
    let fields: Vec<&str> = operand.split('.').collect();
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    let parse_full = has_suffix || (fields.len() >= 3 && !has_wild);
    if parse_full {
        if !operand.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token.start });
        }
        return parse_version(operand)
            .map(Operand::Full)
            .map_err(|error| RangeError::Version { at: absolute, error });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at: token.start });
    }
    let mut values = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at: token.start });
        }
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen || !f.bytes().all(|b| b.is_ascii_digit()) {
                return Err(RangeError::Syntax { at: token.start });
            }
            if f.len() > 1 && f.as_bytes()[0] == b'0' {
                return Err(RangeError::Syntax { at: token.start });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: token.start })?;
            values.push(n);
        }
    }
    if values.len() != 0 && wildcard_seen {
        // Valid numeric prefix then wildcard suffix.
    }
    if !wildcard_seen && values.len() != fields_len_count(operand) {
        return Err(RangeError::Syntax { at: token.start });
    }
    Ok(Operand::Partial { k: values.len(), values })
}

fn fields_len_count(s: &str) -> usize { s.split('.').count() }

fn padded(values: &[u64]) -> Version {
    stable(
        *values.get(0).unwrap_or(&0),
        *values.get(1).unwrap_or(&0),
        *values.get(2).unwrap_or(&0),
    )
}

fn add_lower(arm: &mut Arm, version: Version, inclusive: bool) {
    arm.constraints.push(Constraint::Lower(Bound { version, inclusive }));
}

fn add_upper(arm: &mut Arm, version: Version, inclusive: bool) {
    arm.constraints.push(Constraint::Upper(Bound { version, inclusive }));
}

fn add_operand_constraints(
    arm: &mut Arm,
    op: &str,
    token: &Token,
) -> Result<(), RangeError> {
    let op_len = if matches!(op, "^" | "~" | "-") { op.len() } else { 0 };
    let operand = parse_operand("", token, op_len)?;
    match op {
        "^" | "~" => {
            let (lower, k) = match operand {
                Operand::Full(v) => {
                    if v.pre.len() > 0 { arm.anchors.push(tuple(&v)); }
                    (v.clone(), 3)
                }
                Operand::Partial { values, k } => {
                    if k == 0 { return Err(RangeError::Syntax { at: token.start }); }
                    (padded(&values), k)
                }
            };
            add_lower(arm, lower.clone(), true);
            let upper = match op {
                "^" => {
                    let (a,b,c) = tuple(&lower);
                    if a > 0 || k == 1 { next_major(a, token.start)? }
                    else if b > 0 || k == 2 { next_minor(a,b,token.start)? }
                    else { next_patch(a,b,c,token.start)? }
                }
                _ => {
                    let (a,b,_) = tuple(&lower);
                    if k == 1 { next_major(a, token.start)? } else { next_minor(a,b,token.start)? }
                }
            };
            add_upper(arm, upper, false);
        }
        "-" => unreachable!(),
        _ => match operand {
            Operand::Full(v) => {
                if !v.pre.is_empty() { arm.anchors.push(tuple(&v)); }
                add_lower(arm, v.clone(), true);
                add_upper(arm, v, true);
            }
            Operand::Partial { values, k } => {
                if k == 0 { return Ok(()); }
                let lower = padded(&values);
                add_lower(arm, lower.clone(), true);
                if k < 3 {
                    let upper = next_core(&lower, k, token.start)?;
                    add_upper(arm, upper, false);
                } else {
                    add_lower(arm, lower.clone(), true);
                    add_upper(arm, lower, true);
                }
            }
        },
    }
    Ok(())
}

fn parse_comparator(arm: &mut Arm, token: &Token) -> Result<(), RangeError> {
    let t = token.text.as_str();
    let (op, n) = if t.starts_with(">=") { (">=", 2) }
        else if t.starts_with("<=") { ("<=", 2) }
        else if t.starts_with('>') { (">", 1) }
        else if t.starts_with('<') { ("<", 1) }
        else if t.starts_with('=') { ("=", 1) }
        else { ("", 0) };
    if n > 0 {
        if n == t.len() || !t.as_bytes()[n].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token.start });
        }
        let v = parse_version(&t[n..]).map_err(|error| RangeError::Version { at: token.start + n, error })?;
        if !v.pre.is_empty() { arm.anchors.push(tuple(&v)); }
        match op {
            ">" => add_lower(arm, v, false),
            ">=" => add_lower(arm, v, true),
            "<" => add_upper(arm, v, false),
            "<=" => add_upper(arm, v, true),
            _ => {
                add_lower(arm, v.clone(), true);
                add_upper(arm, v, true);
            }
        }
        return Ok(());
    }
    add_operand_constraints(arm, "", token)
}

fn split_arms(s: &str) -> Result<Vec<(usize, Vec<Token>)>, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut pending_empty = None;
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].text == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: tokens[i].start });
            }
            arms.push((current[0].start, std::mem::take(&mut current)));
            pending_empty = Some(tokens[i].start);
        } else {
            current.push(tokens[i].clone());
            pending_empty = None;
        }
        i += 1;
    }
    if current.is_empty() {
        let at = if pending_empty.is_some() { s.len() } else { 0 };
        return Err(RangeError::Empty { at });
    }
    arms.push((current[0].start, current));
    Ok(arms)
}

fn parse_arm(tokens: &[Token]) -> Result<Arm, RangeError> {
    let mut arm = Arm { constraints: Vec::new(), anchors: Vec::new() };
    let hyphens: Vec<usize> = tokens.iter().enumerate().filter(|(_,t)| t.text == "-").map(|(i,_)| i).collect();
    if !hyphens.is_empty() {
        if hyphens.len() != 1 || tokens.len() != 3 || hyphens[0] != 1 {
            return Err(RangeError::Syntax { at: tokens[hyphens[0]].start });
        }
        let lo = parse_operand("", &tokens[0], 0)?;
        let hi = parse_operand("", &tokens[2], 0)?;
        let (lv, lk) = match lo {
            Operand::Full(v) => (v, 3),
            Operand::Partial { values, k } => {
                if k == 0 { return Err(RangeError::Syntax { at: tokens[0].start }); }
                (padded(&values), k)
            }
        };
        let (hv, hk, full) = match hi {
            Operand::Full(v) => (v, 3, true),
            Operand::Partial { values, k } => {
                if k == 0 { return Err(RangeError::Syntax { at: tokens[2].start }); }
                (padded(&values), k, false)
            }
        };
        if !lv.pre.is_empty() { arm.anchors.push(tuple(&lv)); }
        if !hv.pre.is_empty() { arm.anchors.push(tuple(&hv)); }
        add_lower(&mut arm, lv, true);
        if full || hk == 3 {
            add_upper(&mut arm, hv, true);
        } else {
            add_upper(&mut arm, next_core(&hv, hk, tokens[2].start)?, false);
        }
        let _ = lk;
        return Ok(arm);
    }

    for token in tokens {
        let t = token.text.as_str();
        if t.starts_with('^') {
            add_operand_constraints(&mut arm, "^", token)?;
        } else if t.starts_with('~') {
            add_operand_constraints(&mut arm, "~", token)?;
        } else {
            let first = t.as_bytes().first().copied();
            if !matches!(first, Some(b'0'..=b'9' | b'*' | b'x' | b'X' | b'<' | b'>' | b'=')) {
                return Err(RangeError::Syntax { at: token.start });
            }
            parse_comparator(&mut arm, token)?;
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let pieces = split_arms(s)?;
    let mut arms = Vec::new();
    for (_, tokens) in pieces {
        arms.push(parse_arm(&tokens)?);
    }
    Ok(Range { arms })
}

fn strongest_bounds(arm: &Arm) -> (Option<Bound>, Option<Bound>, Option<Version>) {
    let mut lower: Option<Bound> = None;
    let mut upper: Option<Bound> = None;
    let mut equal: Option<Version> = None;
    for c in &arm.constraints {
        match c {
            Constraint::Lower(b) => {
                if lower.as_ref().map_or(true, |x| {
                    compare_versions(&b.version, &x.version) == Ordering::Greater
                        || (compare_versions(&b.version, &x.version) == Ordering::Equal && !b.inclusive && x.inclusive)
                }) { lower = Some(b.clone()); }
            }
            Constraint::Upper(b) => {
                if upper.as_ref().map_or(true, |x| {
                    compare_versions(&b.version, &x.version) == Ordering::Less
                        || (compare_versions(&b.version, &x.version) == Ordering::Equal && !b.inclusive && x.inclusive)
                }) { upper = Some(b.clone()); }
            }
            Constraint::Equal(v) => equal = Some(v.clone()),
        }
    }
    if let Some(v) = equal {
        let b = Bound { version: v.clone(), inclusive: true };
        lower = Some(b.clone());
        upper = Some(b);
        return (lower, upper, Some(v));
    }
    (lower, upper, None)
}

fn anchor_match(v: &Version, arm: &Arm) -> bool {
    arm.anchors.iter().any(|a| *a == tuple(v))
}

fn arm_matches(v: &Version, arm: &Arm, include_pre: bool) -> bool {
    for c in &arm.constraints {
        match c {
            Constraint::Lower(b) => {
                let o = compare_versions(v, &b.version);
                if o == Ordering::Less || (o == Ordering::Equal && !b.inclusive) { return false; }
            }
            Constraint::Upper(b) => {
                let o = compare_versions(v, &b.version);
                if o == Ordering::Greater || (o == Ordering::Equal && !b.inclusive) { return false; }
            }
            Constraint::Equal(x) => if compare_versions(v, x) != Ordering::Equal { return false; },
        }
    }
    v.pre.is_empty() || include_pre || anchor_match(v, arm)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) { continue; }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (max && ord == Ordering::Greater) || (!max && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn canonical_core(v: &Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn canonical_version(v: &Version) -> String {
    let mut s = canonical_core(v);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

pub fn canonical(r: &Range) -> String {
    let mut strings = Vec::new();
    for arm in &r.arms {
        let (lo, hi, _) = strongest_bounds(arm);
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anchor_text = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
        };
        let lower = lo.map_or("*".to_string(), |b| {
            format!("{}{}", if b.inclusive { ">=" } else { ">" }, canonical_version(&strip_build(b.version)))
        });
        let upper = hi.map_or("*".to_string(), |b| {
            format!("{}{}", if b.inclusive { "<=" } else { "<" }, canonical_version(&strip_build(b.version)))
        });
        strings.push(format!("[{lower};{upper};pre={anchor_text}]"));
    }
    strings.sort();
    strings.dedup();
    if strings.is_empty() { "EMPTY".to_string() } else { strings.join(" || ") }
}

fn successor(v: &Version) -> Option<Version> {
    if !v.pre.is_empty() {
        let mut n = v.clone();
        n.pre.push("0".to_string());
        n.build.clear();
        return Some(n);
    }
    let (a,b,c) = tuple(v);
    if c < u64::MAX { Some(zero_pre(a,b,c+1)) }
    else if b < u64::MAX { Some(zero_pre(a,b+1,0)) }
    else if a < u64::MAX { Some(zero_pre(a+1,0,0)) }
    else { None }
}

fn predecessor(v: &Version) -> Option<Version> {
    if v.pre.is_empty() {
        let (a,b,c) = tuple(v);
        if c > 0 { return Some(stable(a,b,c-1)); }
        if b > 0 { return Some(stable(a,b-1,u64::MAX)); }
        if a > 0 { return Some(stable(a-1,u64::MAX,u64::MAX)); }
        return None;
    }
    None
}

fn arm_bounds_allow(v: &Version, arm: &Arm) -> bool {
    arm.constraints.iter().all(|c| match c {
        Constraint::Lower(b) => {
            let o = compare_versions(v, &b.version);
            o == Ordering::Greater || (o == Ordering::Equal && b.inclusive)
        }
        Constraint::Upper(b) => {
            let o = compare_versions(v, &b.version);
            o == Ordering::Less || (o == Ordering::Equal && b.inclusive)
        }
        Constraint::Equal(x) => compare_versions(v, x) == Ordering::Equal,
    })
}

fn interval_nonempty(arm: &Arm) -> bool {
    let (lo, hi, _) = strongest_bounds(arm);
    match (lo, hi) {
        (Some(l), Some(u)) => {
            let ord = compare_versions(&l.version, &u.version);
            if ord == Ordering::Greater || (ord == Ordering::Equal && (!l.inclusive || !u.inclusive)) {
                return false;
            }
            if ord == Ordering::Equal { return true; }
            let candidate = if l.inclusive { l.version.clone() } else {
                successor(&l.version).unwrap_or_else(|| l.version.clone())
            };
            arm_bounds_allow(&candidate, arm)
        }
        _ => true,
    }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let mut combined = Arm {
                constraints: aa.constraints.clone(),
                anchors: aa.anchors.iter().copied().filter(|x| bb.anchors.contains(x)).collect(),
            };
            combined.constraints.extend(bb.constraints.clone());
            if !interval_nonempty(&combined) { continue; }
            if include_prerelease {
                return true;
            }
            // Stable witnesses are checked by selecting the first boundary-adjacent stable candidate.
            let (lo, hi, _) = strongest_bounds(&combined);
            let mut candidates = vec![stable(0,0,0), stable(u64::MAX,u64::MAX,u64::MAX)];
            if let Some(l) = lo { candidates.push(if l.inclusive { strip_build(l.version) } else { successor(&l.version).unwrap_or_else(|| l.version) }); }
            if let Some(u) = hi { candidates.push(if u.inclusive { strip_build(u.version) } else { predecessor(&u.version).unwrap_or_else(|| u.version) }); }
            for v in candidates {
                if v.pre.is_empty() && arm_bounds_allow(&v, &combined) { return true; }
            }
            // Anchored prereleases can witness only at their exact tuples.
            for anchor in &combined.anchors {
                for source in [aa, bb] {
                    for c in &source.constraints {
                        let ver = match c {
                            Constraint::Lower(x) => &x.version,
                            Constraint::Upper(x) => &x.version,
                            Constraint::Equal(x) => x,
                        };
                        if tuple(ver) == *anchor && !ver.pre.is_empty() && arm_bounds_allow(ver, &combined) {
                            return true;
                        }
                    }
                }
            }
        }
    }
    false
}
