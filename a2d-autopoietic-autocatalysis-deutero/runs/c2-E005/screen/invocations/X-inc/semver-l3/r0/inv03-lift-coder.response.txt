use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if start == *pos {
        return Err(if *pos == bytes.len()
            || matches!(bytes[*pos], b'.' | b'-' | b'+')
        {
            VersionError::MissingCore { at: *pos }
        } else {
            VersionError::InvalidChar { at: *pos }
        });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut p = start;
    loop {
        let ident_start = p;
        while p < end && bytes[p] != b'.' {
            if !bytes[p].is_ascii_alphanumeric() && bytes[p] != b'-' {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if ident_start == p {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let ident = &s[ident_start..p];
        if pre && ident.bytes().all(is_digit) && ident.len() > 1 && ident.starts_with('0') {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        result.push(ident.to_string());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for expected in 0..2 {
        if pos >= s.len() || s.as_bytes()[pos] != b'.' {
            return Err(if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        pos += 1;
        let value = parse_core(s, &mut pos)?;
        if expected == 0 {
            let minor = value;
            if pos >= s.len() || s.as_bytes()[pos] != b'.' {
                return Err(if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
                    VersionError::MissingCore { at: pos }
                } else {
                    VersionError::InvalidChar { at: pos }
                });
            }
            pos += 1;
            let patch = parse_core(s, &mut pos)?;
            return parse_version_tail(s, pos, major, minor, patch);
        }
    }
    unreachable!()
}

fn parse_version_tail(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() {
        match bytes[pos] {
            b'-' => {
                pos += 1;
                let plus = bytes[pos..]
                    .iter()
                    .position(|&b| b == b'+')
                    .map(|i| pos + i)
                    .unwrap_or(bytes.len());
                pre = parse_identifiers(s, pos, plus, true)?;
                pos = plus;
            }
            b'+' => {}
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }
    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(s, pos, bytes.len(), false)?;
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn cmp_numeric_ident(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let xn = x.bytes().all(is_digit);
                    let yn = y.bytes().all(is_digit);
                    let ord = match (xn, yn) {
                        (true, true) => cmp_numeric_ident(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Op { Eq, Lt, Le, Gt, Ge }

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn version_core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn version_core_only(t: (u64, u64, u64), pre: Vec<String>) -> Version {
    Version { major: t.0, minor: t.1, patch: t.2, pre, build: Vec::new() }
}

fn add_cmp(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() {
        arm.anchors.push(version_core(&v));
    }
    arm.comparators.push(Comparator { op, version: v, anchor });
}

fn cmp_matches(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include_pre: bool) -> bool {
    if !include_pre && !v.pre.is_empty() && !arm.anchors.contains(&version_core(v)) {
        return false;
    }
    arm.comparators.iter().all(|c| cmp_matches(v, c))
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str) -> Vec<(usize, &str)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = 0;
    while p < bytes.len() {
        while p < bytes.len() && whitespace(bytes[p]) { p += 1; }
        if p == bytes.len() { break; }
        let start = p;
        while p < bytes.len() && !whitespace(bytes[p]) { p += 1; }
        out.push((start, &s[start..p]));
    }
    out
}

#[derive(Clone, Debug)]
enum Operand {
    Full(Version),
    Partial(Vec<Option<u64>>),
}

fn parse_partial_or_full(s: &str, start: usize) -> Result<Operand, RangeError> {
    if s.contains('-') || s.contains('+') {
        return parse_version(s)
            .map(Operand::Full)
            .map_err(|error| RangeError::Version { at: start, error });
    }
    let fields: Vec<&str> = s.split('.').collect();
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if fields.len() >= 3 && !has_wild {
        return parse_version(s)
            .map(Operand::Full)
            .map_err(|error| RangeError::Version { at: start, error });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at: start });
    }
    let mut vals = Vec::new();
    let mut wildcard = false;
    for f in fields {
        if f.is_empty() { return Err(RangeError::Syntax { at: start }); }
        if matches!(f, "*" | "x" | "X") {
            wildcard = true;
            vals.push(None);
        } else {
            if wildcard || !f.bytes().all(is_digit) || (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at: start });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: start })?;
            vals.push(Some(n));
        }
    }
    if vals.iter().skip_while(|v| v.is_some()).any(|v| v.is_some()) {
        return Err(RangeError::Syntax { at: start });
    }
    while vals.len() < 3 { vals.push(None); }
    Ok(Operand::Partial(vals))
}

fn expand_partial(v: &[Option<u64>], start: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let nums: Vec<u64> = v.iter().take_while(|x| x.is_some()).map(|x| x.unwrap()).collect();
    let k = nums.len();
    if k == 0 { return Ok(()); }
    let low = version_core_only(
        (nums[0], *nums.get(1).unwrap_or(&0), *nums.get(2).unwrap_or(&0)),
        Vec::new(),
    );
    add_cmp(arm, Op::Ge, low, false);
    let mut t = (0, 0, 0);
    let upper;
    if k == 1 {
        if nums[0] == u64::MAX { return Err(RangeError::Syntax { at: start }); }
        t = (nums[0] + 1, 0, 0);
        upper = t;
    } else if k == 2 {
        if nums[1] == u64::MAX { return Err(RangeError::Syntax { at: start }); }
        t = (nums[0], nums[1] + 1, 0);
        upper = t;
    } else {
        return Ok(());
    }
    let _ = t;
    add_cmp(arm, Op::Lt, version_core_only(upper, vec!["0".into()]), false);
    Ok(())
}

fn padded(op: &Operand) -> (Version, usize) {
    match op {
        Operand::Full(v) => (v.clone(), 3),
        Operand::Partial(fields) => {
            let nums: Vec<u64> = fields.iter().take_while(|x| x.is_some()).map(|x| x.unwrap()).collect();
            let k = nums.len();
            (version_core_only(
                (*nums.get(0).unwrap_or(&0), *nums.get(1).unwrap_or(&0), *nums.get(2).unwrap_or(&0)),
                Vec::new(),
            ), k)
        }
    }
}

fn bump(t: (u64, u64, u64), which: usize, start: usize) -> Result<(u64,u64,u64), RangeError> {
    match which {
        0 if t.0 < u64::MAX => Ok((t.0 + 1, 0, 0)),
        1 if t.1 < u64::MAX => Ok((t.0, t.1 + 1, 0)),
        2 if t.2 < u64::MAX => Ok((t.0, t.1, t.2 + 1)),
        _ => Err(RangeError::Syntax { at: start }),
    }
}

fn parse_expansion(token: &str, start: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let (prefix, operand, os) = if token.starts_with('^') || token.starts_with('~') {
        (&token[..1], &token[1..], start + 1)
    } else { ("", token, start) };
    let op = parse_partial_or_full(operand, os)?;
    if prefix.is_empty() {
        match op {
            Operand::Full(v) => add_cmp(arm, Op::Eq, v, true),
            Operand::Partial(fields) => expand_partial(&fields, start, arm)?,
        }
        return Ok(());
    }
    let (low, k) = padded(&op);
    if k == 0 { return Err(RangeError::Syntax { at: start }); }
    add_cmp(arm, Op::Ge, low.clone(), true);
    let t = version_core(&low);
    let upper = if prefix == "^" {
        let nums = match &op { Operand::Partial(f) => f.iter().take_while(|x| x.is_some()).count(), _ => 3 };
        let which = if t.0 > 0 || nums == 1 { 0 } else if t.1 > 0 || nums == 2 { 1 } else { 2 };
        bump(t, which, start)?
    } else {
        bump(t, if k == 1 { 0 } else { 1 }, start)?
    };
    add_cmp(arm, Op::Lt, version_core_only(upper, vec!["0".into()]), false);
    Ok(())
}

fn parse_arm(tokens: &[(usize, &str)]) -> Result<Arm, RangeError> {
    let mut arm = Arm { comparators: Vec::new(), anchors: Vec::new() };
    if let Some((idx, _)) = tokens.iter().find(|(_, t)| *t == "-") {
        if tokens.len() != 3 || tokens[1].1 != "-" {
            return Err(RangeError::Syntax { at: *idx });
        }
        let (lo_s, lo_at) = (tokens[0].1, tokens[0].0);
        let (hi_s, hi_at) = (tokens[2].1, tokens[2].0);
        let lo = parse_partial_or_full(lo_s, lo_at)?;
        let hi = parse_partial_or_full(hi_s, hi_at)?;
        let (lower, lk) = padded(&lo);
        let (upper, uk) = padded(&hi);
        if lk == 0 { return Err(RangeError::Syntax { at: lo_at }); }
        if uk == 0 { return Err(RangeError::Syntax { at: hi_at }); }
        add_cmp(&mut arm, Op::Ge, lower, true);
        match hi {
            Operand::Full(v) => add_cmp(&mut arm, Op::Le, v, true),
            Operand::Partial(_) if uk == 1 => {
                let t = bump(version_core(&upper), 0, hi_at)?;
                add_cmp(&mut arm, Op::Lt, version_core_only(t, vec!["0".into()]), false);
            }
            Operand::Partial(_) if uk == 2 => {
                let t = bump(version_core(&upper), 1, hi_at)?;
                add_cmp(&mut arm, Op::Lt, version_core_only(t, vec!["0".into()]), false);
            }
            Operand::Partial(_) => add_cmp(&mut arm, Op::Le, upper, false),
        }
        return Ok(arm);
    }
    for (start, token) in tokens {
        if token == "||" { return Err(RangeError::Syntax { at: *start }); }
        if let Some(opchar) = token.as_bytes().first().copied() {
            if matches!(opchar, b'=' | b'<' | b'>') {
                let (op, n) = if token.starts_with("<=") { (Op::Le, 2) }
                    else if token.starts_with(">=") { (Op::Ge, 2) }
                    else if token.starts_with('<') { (Op::Lt, 1) }
                    else if token.starts_with('>') { (Op::Gt, 1) }
                    else { (Op::Eq, 1) };
                let operand = &token[n..];
                let at = start + n;
                if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                    return Err(RangeError::Syntax { at: *start });
                }
                let v = parse_version(operand).map_err(|error| RangeError::Version { at, error })?;
                add_cmp(&mut arm, op, v, true);
            } else if token.starts_with('^') || token.starts_with('~')
                || token.starts_with('*') || token.starts_with('x') || token.starts_with('X')
                || token.as_bytes()[0].is_ascii_digit()
            {
                parse_expansion(token, *start, &mut arm)?;
            } else {
                return Err(RangeError::Syntax { at: *start });
            }
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let mut arms = Vec::new();
    let mut begin = 0;
    let mut i = 0;
    while i <= tokens.len() {
        if i == tokens.len() || tokens[i].1 == "||" {
            if begin == i {
                let at = if i < tokens.len() { tokens[i].0 } else { s.len() };
                return Err(RangeError::Empty { at });
            }
            arms.push(parse_arm(&tokens[begin..i])?);
            begin = i + 1;
        }
        i += 1;
    }
    Ok(Range { arms })
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (index, s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index, error })?);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) { continue; }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (max && ord == Ordering::Greater) || (!max && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn cmp_to_string(c: &Comparator, lower: bool) -> Option<String> {
    let op = match (lower, c.op) {
        (true, Op::Ge) => ">=",
        (true, Op::Gt) => ">",
        (false, Op::Le) => "<=",
        (false, Op::Lt) => "<",
        (true, Op::Eq) => ">=",
        (false, Op::Eq) => "<=",
        _ => return None,
    };
    Some(format!("{op}{}", display_version(&c.version)))
}

fn display_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() { s.push('-'); s.push_str(&v.pre.join(".")); }
    s
}

pub fn canonical(r: &Range) -> String {
    let mut out = Vec::new();
    for arm in &r.arms {
        let lower = arm.comparators.iter().filter(|c| matches!(c.op, Op::Gt | Op::Ge | Op::Eq))
            .max_by(|a,b| compare_versions(&a.version,&b.version).then_with(|| (a.op == Op::Gt).cmp(&(b.op == Op::Gt))));
        let upper = arm.comparators.iter().filter(|c| matches!(c.op, Op::Lt | Op::Le | Op::Eq))
            .min_by(|a,b| compare_versions(&a.version,&b.version).then_with(|| (a.op == Op::Lt).cmp(&(b.op == Op::Lt))));
        let lo = lower.and_then(|c| cmp_to_string(c,true)).unwrap_or_else(|| "*".into());
        let hi = upper.and_then(|c| cmp_to_string(c,false)).unwrap_or_else(|| "*".into());
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let a = if anchors.is_empty() { "-".into() } else {
            anchors.iter().map(|(x,y,z)| format!("{x}.{y}.{z}")).collect::<Vec<_>>().join(",")
        };
        out.push(format!("[{lo};{hi};pre={a}]"));
    }
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".into() } else { out.join(" || ") }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let mut candidates = Vec::new();
            for c in aa.comparators.iter().chain(bb.comparators.iter()) {
                candidates.push(c.version.clone());
                let (ma,mi,pa) = version_core(&c.version);
                candidates.push(version_core_only((ma,mi,pa), vec!["0".into()]));
                if pa < u64::MAX { candidates.push(version_core_only((ma,mi,pa+1), Vec::new())); }
            }
            candidates.push(version_core_only((0,0,0), Vec::new()));
            for v in &candidates {
                if arm_matches(v, aa, include_prerelease) && arm_matches(v, bb, include_prerelease) {
                    return true;
                }
            }
        }
    }
    false
}
