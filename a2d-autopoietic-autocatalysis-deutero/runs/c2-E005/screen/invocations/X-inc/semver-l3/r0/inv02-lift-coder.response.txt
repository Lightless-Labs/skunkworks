use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: *pos });
        }
        return Err(VersionError::InvalidChar { at: *pos });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let n = s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })?;
    Ok(n)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end && bytes[p] != b'.' {
            let b = bytes[p];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if pre && id.bytes().all(is_digit) && id.len() > 1 && id.as_bytes()[0] == b'0' {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_string());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    for _ in 0..2 {
        if p >= s.len() || s.as_bytes()[p] != b'.' {
            if p >= s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: p });
            }
            return Err(VersionError::InvalidChar { at: p });
        }
        p += 1;
        if _ == 0 {}
    }
    // Re-scan straightforwardly to preserve field error ordering.
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        if p >= s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p });
        }
        return Err(VersionError::InvalidChar { at: p });
    }
    p += 1;
    let minor = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        if p >= s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p });
        }
        return Err(VersionError::InvalidChar { at: p });
    }
    p += 1;
    let patch = parse_core(s, &mut p)?;

    if p < s.len() && !matches!(s.as_bytes()[p], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: p });
    }
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < s.len() && s.as_bytes()[p] == b'-' {
        let begin = p + 1;
        let plus = s[begin..].find('+').map(|i| begin + i).unwrap_or(s.len());
        pre = parse_identifiers(s, begin, plus, true)?;
        p = plus;
    }
    if p < s.len() && s.as_bytes()[p] == b'+' {
        let begin = p + 1;
        build = parse_identifiers(s, begin, s.len(), false)?;
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    if a.is_empty() && b.is_empty() { return Ordering::Equal; }
    if a.is_empty() { return Ordering::Greater; }
    if b.is_empty() { return Ordering::Less; }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let c = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal { return c; }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op { Eq, Lt, Le, Gt, Ge }

#[derive(Clone, Debug)]
struct Cmp {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Debug)]
struct Arm {
    cmps: Vec<Cmp>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range { arms: Vec<Arm> }

fn ws(b: u8) -> bool { matches!(b, b' ' | b'\t' | b'\r' | b'\n') }

fn tokens(s: &str, base: usize) -> Vec<(String, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = 0;
    while p < bytes.len() {
        while p < bytes.len() && ws(bytes[p]) { p += 1; }
        if p == bytes.len() { break; }
        let st = p;
        while p < bytes.len() && !ws(bytes[p]) { p += 1; }
        out.push((s[st..p].to_string(), base + st));
    }
    out
}

fn full_cmp(tok: &str, at: usize) -> Result<Cmp, RangeError> {
    let (op, n) = if tok.starts_with(">=") { (Op::Ge, 2) }
        else if tok.starts_with("<=") { (Op::Le, 2) }
        else if tok.starts_with(">") { (Op::Gt, 1) }
        else if tok.starts_with("<") { (Op::Lt, 1) }
        else if tok.starts_with("=") { (Op::Eq, 1) }
        else { (Op::Eq, 0) };
    if n == tok.len() || !tok.as_bytes()[n].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    let v = parse_version(&tok[n..]).map_err(|error| RangeError::Version { at: at+n, error })?;
    Ok(Cmp { op, anchor: !v.pre.is_empty(), version: v })
}

#[derive(Clone)]
enum Operand { Full(Version), Partial(Vec<Option<u64>>) }

fn parse_operand(s: &str, at: usize) -> Result<Operand, RangeError> {
    let fields: Vec<&str> = s.split('.').collect();
    let full = s.contains('-') || s.contains('+')
        || (fields.len() >= 3 && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X")));
    if full {
        return parse_version(s)
            .map(Operand::Full)
            .map_err(|error| RangeError::Version { at, error });
    }
    if fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut values = Vec::new();
    let mut wild = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wild = true;
            values.push(None);
        } else {
            if wild || !f.bytes().all(is_digit) || (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            values.push(Some(n));
        }
    }
    while values.len() < 3 { values.push(None); }
    Ok(Operand::Partial(values))
}

fn version_tuple(v: &Version) -> (u64,u64,u64) { (v.major,v.minor,v.patch) }
fn vtuple(t:(u64,u64,u64), pre: Vec<String>) -> Version {
    Version { major:t.0, minor:t.1, patch:t.2, pre, build:Vec::new() }
}
fn partial_version(p:&[Option<u64>]) -> Version {
    vtuple((p[0].unwrap_or(0),p[1].unwrap_or(0),p[2].unwrap_or(0)), Vec::new())
}
fn cmp(op: Op, v: Version, anchor: bool, arm: &mut Arm) {
    if anchor { arm.anchors.push(version_tuple(&v)); }
    arm.cmps.push(Cmp { op, version:v, anchor });
}
fn inc(x:u64, at:usize)->Result<u64,RangeError>{
    x.checked_add(1).ok_or(RangeError::Syntax{at})
}
fn upper_partial(p:&[Option<u64>], at:usize)->Result<Option<Version>,RangeError>{
    let k=p.iter().take_while(|x| x.is_some()).count();
    let mut t=(p[0].unwrap_or(0),p[1].unwrap_or(0),p[2].unwrap_or(0));
    match k {
        0 => Ok(None),
        1 => { t.0=inc(t.0,at)?; t.1=0;t.2=0; }
        2 => { t.1=inc(t.1,at)?;t.2=0; }
        _ => { t.2=inc(t.2,at)?; }
    }
    Ok(Some(vtuple(t,vec!["0".into()])))
}
fn add_operand(arm:&mut Arm,tok:&str,at:usize,kind:u8)->Result<(),RangeError>{
    let operand=parse_operand(tok,at)?;
    match kind {
        0 => match operand {
            Operand::Full(v)=>cmp(Op::Eq,v,true,arm),
            Operand::Partial(p)=>{
                let k=p.iter().take_while(|x|x.is_some()).count();
                if k==0 { return Ok(()) }
                let lo=partial_version(&p);
                cmp(Op::Ge,lo,false,arm);
                if let Some(hi)=upper_partial(&p,at)? { cmp(Op::Lt,hi,false,arm); }
            }
        },
        1 | 2 => {
            let (lo,p,k,full)=match operand {
                Operand::Full(v)=>(v.clone(),vec![Some(v.major),Some(v.minor),Some(v.patch)],3,Some(v)),
                Operand::Partial(p)=>{
                    let k=p.iter().take_while(|x|x.is_some()).count();
                    (partial_version(&p),p,k,None)
                }
            };
            if k==0 { return Err(RangeError::Syntax{at}); }
            let mut up=(lo.major,lo.minor,lo.patch);
            if kind==1 {
                if lo.major>0 || k==1 { up.0=inc(up.0,at)?;up.1=0;up.2=0; }
                else if lo.minor>0 || k==2 { up.1=inc(up.1,at)?;up.2=0; }
                else { up.2=inc(up.2,at)?; }
            } else if k==1 { up.0=inc(up.0,at)?;up.1=0;up.2=0; }
            else { up.1=inc(up.1,at)?;up.2=0; }
            cmp(Op::Ge,lo.clone(),full.as_ref().is_some_and(|v|!v.pre.is_empty()),arm);
            cmp(Op::Lt,vtuple(up,vec!["0".into()]),false,arm);
        },
        _=>{}
    }
    Ok(())
}

fn parse_arm(ts:&[(String,usize)]) -> Result<Arm,RangeError> {
    let mut arm=Arm{cmps:Vec::new(),anchors:Vec::new()};
    if let Some((i,(_,at)))=ts.iter().enumerate().find(|(_,x)|x.0=="-") {
        if i!=1 || ts.len()!=3 { return Err(RangeError::Syntax{at:*at}); }
        let (lt,la)=(&ts[0].0,ts[0].1);
        let (ut,ua)=(&ts[2].0,ts[2].1);
        let lo=parse_operand(lt,*la)?;
        let hi=parse_operand(ut,*ua)?;
        let (lv,lk)=match lo { Operand::Full(v)=>(v,3),Operand::Partial(p)=>{let k=p.iter().take_while(|x|x.is_some()).count();(partial_version(&p),k)} };
        let (uv,uk,ufull)=match hi { Operand::Full(v)=>(v,3,true),Operand::Partial(p)=>{let k=p.iter().take_while(|x|x.is_some()).count();(partial_version(&p),k,false)} };
        if lk==0 { return Err(RangeError::Syntax{at:*la}); }
        if uk==0 { return Err(RangeError::Syntax{at:*ua}); }
        cmp(Op::Ge,lv.clone(),!lv.pre.is_empty(),&mut arm);
        if ufull { cmp(Op::Le,uv.clone(),!uv.pre.is_empty(),&mut arm); }
        else if let Some(up)=upper_partial(&[Some(uv.major),if uk>=2{Some(uv.minor)}else{None},None],*ua)? {
            cmp(Op::Lt,up,false,&mut arm);
        }
        return Ok(arm);
    }
    for (tok,at) in ts {
        if tok.starts_with('^') || tok.starts_with('~') {
            add_operand(&mut arm,&tok[1..],at+1,if tok.starts_with('^'){1}else{2})?;
        } else if tok.starts_with(['<','>','=']) {
            let c=full_cmp(tok,*at)?;
            if c.anchor { arm.anchors.push(version_tuple(&c.version)); }
            arm.cmps.push(c);
        } else if tok.starts_with(|c:char| c.is_ascii_digit() || matches!(c,'*'|'x'|'X')) {
            add_operand(&mut arm,tok,*at,0)?;
        } else { return Err(RangeError::Syntax{at:*at}); }
    }
    Ok(arm)
}

pub fn parse_range(s:&str)->Result<Range,RangeError>{
    let ts=tokens(s,0);
    if ts.is_empty(){return Err(RangeError::Empty{at:0});}
    let mut groups:Vec<Vec<(String,usize)>>=vec![Vec::new()];
    let mut sep_at=None;
    for t in ts {
        if t.0=="||" {
            let last=groups.last().unwrap();
            if last.is_empty(){return Err(RangeError::Empty{at:t.1});}
            sep_at=Some(t.1);
            groups.push(Vec::new());
        }else{groups.last_mut().unwrap().push(t);}
    }
    if groups.last().unwrap().is_empty(){
        return Err(RangeError::Empty{at:s.len()});
    }
    let mut arms=Vec::new();
    for g in groups { arms.push(parse_arm(&g)?); }
    let _=sep_at;
    Ok(Range{arms})
}

fn admitted(v:&Version,a:&Arm)->bool{
    a.anchors.iter().any(|t|*t==version_tuple(v))
}
fn cmp_matches(v:&Version,c:&Cmp)->bool{
    let o=compare_versions(v,&c.version);
    match c.op {Op::Eq=>o==Ordering::Equal,Op::Lt=>o==Ordering::Less,Op::Le=>o!=Ordering::Greater,Op::Gt=>o==Ordering::Greater,Op::Ge=>o!=Ordering::Less}
}
fn arm_matches(v:&Version,a:&Arm,include:bool)->bool{
    if !v.pre.is_empty() && !include && !admitted(v,a){return false;}
    a.cmps.iter().all(|c|cmp_matches(v,c))
}
pub fn matches_with(v:&Version,r:&Range,include_prerelease:bool)->bool{
    r.arms.iter().any(|a|arm_matches(v,a,include_prerelease))
}
pub fn matches(v:&Version,r:&Range)->bool{matches_with(v,r,false)}

fn extrema(candidates:&[String],r:&Range,max:bool)->Result<Option<String>,CandidateError>{
    let mut parsed=Vec::new();
    for (i,s) in candidates.iter().enumerate(){
        match parse_version(s){Ok(v)=>parsed.push(v),Err(error)=>return Err(CandidateError{index:i,error})}
    }
    let mut best:Option<usize>=None;
    for i in 0..parsed.len(){
        if !matches(&parsed[i],r){continue;}
        if let Some(j)=best{
            let o=compare_versions(&parsed[i],&parsed[j]);
            if (max&&o==Ordering::Greater)||(!max&&o==Ordering::Less){best=Some(i);}
        }else{best=Some(i);}
    }
    Ok(best.map(|i|candidates[i].clone()))
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index:usize, pub error:VersionError }
pub fn max_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extrema(candidates,r,true)}
pub fn min_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extrema(candidates,r,false)}

#[derive(Clone,Copy)]
struct Bound { v: Version, inclusive: bool }
fn arm_bounds(a:&Arm)->(Option<Bound>,Option<Bound>){
    let mut lo=None;let mut hi=None;
    for c in &a.cmps{
        match c.op{
            Op::Eq=>{
                let b=Bound{v:c.version.clone(),inclusive:true};
                lo=Some(b);hi=Some(b);
            }
            Op::Gt|Op::Ge=>{
                let b=Bound{v:c.version.clone(),inclusive:matches!(c.op,Op::Ge)};
                lo=Some(match lo{Some(old)=>if compare_versions(&b.v,&old.v)==Ordering::Greater || (compare_versions(&b.v,&old.v)==Ordering::Equal&&!b.inclusive){b}else{old},None=>b});
            }
            Op::Lt|Op::Le=>{
                let b=Bound{v:c.version.clone(),inclusive:matches!(c.op,Op::Le)};
                hi=Some(match hi{Some(old)=>if compare_versions(&b.v,&old.v)==Ordering::Less || (compare_versions(&b.v,&old.v)==Ordering::Equal&&!b.inclusive){b}else{old},None=>b});
            }
        }
    }
    (lo,hi)
}
fn core_string(v:&Version)->String{format!("{}.{}.{}",v.major,v.minor,v.patch)}
fn version_string(v:&Version)->String{
    let mut s=core_string(v);
    if !v.pre.is_empty(){s.push('-');s.push_str(&v.pre.join("."));}
    s
}
pub fn canonical(r:&Range)->String{
    let mut out=Vec::new();
    for a in &r.arms{
        let (lo,hi)=arm_bounds(a);
        let mut anchors=a.anchors.clone();anchors.sort();
        anchors.dedup();
        let anc=if anchors.is_empty(){"-".to_string()}else{anchors.iter().map(|t|format!("{}.{}.{}",t.0,t.1,t.2)).collect::<Vec<_>>().join(",")};
        let l=lo.map(|b|format!("{}{}",if b.inclusive{">="}else{">"},version_string(&b.v))).unwrap_or("*".into());
        let h=hi.map(|b|format!("{}{}",if b.inclusive{"<="}else{"<"},version_string(&b.v))).unwrap_or("*".into());
        out.push(format!("[{};{};pre={}]",l,h,anc));
    }
    out.sort();out.dedup();
    if out.is_empty(){"EMPTY".into()}else{out.join(" || ")}
}
pub fn intersects(a:&Range,b:&Range,include_prerelease:bool)->bool{
    for aa in &a.arms{for bb in &b.arms{
        let (al,ah)=arm_bounds(aa);let (bl,bh)=arm_bounds(bb);
        let lower=match (al,bl){(Some(x),Some(y))=>Some(if compare_versions(&x.v,&y.v)==Ordering::Greater || (compare_versions(&x.v,&y.v)==Ordering::Equal&&!x.inclusive){x}else{y}),(x,None)|(None,x)=>x};
        let upper=match (ah,bh){(Some(x),Some(y))=>Some(if compare_versions(&x.v,&y.v)==Ordering::Less || (compare_versions(&x.v,&y.v)==Ordering::Equal&&!x.inclusive){x}else{y}),(x,None)|(None,x)=>x};
        if let (Some(l),Some(h))=(lower,upper){
            let c=compare_versions(&l.v,&h.v);
            if c==Ordering::Greater || (c==Ordering::Equal&&(!l.inclusive||!h.inclusive)){continue;}
        }
        if include_prerelease || (!aa.anchors.is_empty() && !bb.anchors.is_empty() && aa.anchors.iter().any(|x|bb.anchors.contains(x))) { return true; }
        if lower.map(|x|x.v.pre.is_empty()).unwrap_or(true) { return true; }
    }}
    false
}
