use std::cmp::Ordering;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted: [bool; 2],
}

#[derive(Clone, Copy)]
struct FieldSpec {
    field: Field,
    min: u8,
    max: u8,
}

const SPECS: [FieldSpec; 5] = [
    FieldSpec { field: Field::Minute, min: 0, max: 59 },
    FieldSpec { field: Field::Hour, min: 0, max: 23 },
    FieldSpec { field: Field::DayOfMonth, min: 1, max: 31 },
    FieldSpec { field: Field::Month, min: 1, max: 12 },
    FieldSpec { field: Field::DayOfWeek, min: 0, max: 6 },
];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_length(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > month_length(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u8> {
    if !bytes.is_ascii() {
        return None;
    }
    let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
    let names: &[&[u8]] = match field {
        Field::Month => &[
            b"JAN", b"FEB", b"MAR", b"APR", b"MAY", b"JUN",
            b"JUL", b"AUG", b"SEP", b"OCT", b"NOV", b"DEC",
        ],
        Field::DayOfWeek => &[b"SUN", b"MON", b"TUE", b"WED", b"THU", b"FRI", b"SAT"],
        _ => return None,
    };
    names.iter().position(|name| *name == upper.as_slice()).map(|i| {
        if field == Field::Month { (i + 1) as u8 } else { i as u8 }
    })
}

fn parse_number(
    bytes: &[u8],
    offset: usize,
    spec: FieldSpec,
    step: bool,
) -> Result<u8, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field: spec.field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field: spec.field, offset })?;
    }
    if step {
        let width = (spec.max - spec.min + 1) as u32;
        if value == 0 || value > width {
            return Err(ParseError::BadStep { field: spec.field, offset });
        }
    } else if value < spec.min as u32 || value > spec.max as u32 {
        return Err(ParseError::OutOfRange { field: spec.field, offset });
    }
    Ok(value as u8)
}

fn parse_endpoint(
    bytes: &[u8],
    offset: usize,
    spec: FieldSpec,
) -> Result<u8, ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) && !bytes.is_empty() {
        parse_number(bytes, offset, spec, false)
    } else if let Some(v) = name_value(spec.field, bytes) {
        Ok(v)
    } else {
        Err(ParseError::InvalidSyntax { field: spec.field, offset })
    }
}

fn parse_item(
    item: &[u8],
    item_offset: usize,
    spec: FieldSpec,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem { field: spec.field, offset: item_offset });
    }

    let slash_positions: Vec<usize> = item.iter().enumerate()
        .filter_map(|(i, b)| if *b == b'/' { Some(i) } else { None })
        .collect();
    let hyphen_positions: Vec<usize> = item.iter().enumerate()
        .filter_map(|(i, b)| if *b == b'-' { Some(i) } else { None })
        .collect();

    if slash_positions.len() > 1 || hyphen_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax { field: spec.field, offset: item_offset });
    }

    let slash = slash_positions.first().copied();
    let base_end = slash.unwrap_or(item.len());
    let base = &item[..base_end];
    let range = hyphen_positions.first().copied().filter(|p| *p < base_end);

    if base.is_empty()
        || (slash.is_some() && slash.unwrap() + 1 == item.len())
        || (hyphen_positions.first().is_some_and(|p| *p >= base_end))
    {
        return Err(ParseError::InvalidSyntax { field: spec.field, offset: item_offset });
    }

    if base == b"*" {
        // valid wildcard
    } else if let Some(h) = range {
        if h == 0 || h + 1 == base.len() {
            return Err(ParseError::InvalidSyntax { field: spec.field, offset: item_offset });
        }
        let left = &base[..h];
        let right = &base[h + 1..];
        if left == b"*" || right == b"*" {
            return Err(ParseError::InvalidSyntax { field: spec.field, offset: item_offset });
        }
        let _ = parse_endpoint(left, item_offset, spec)
            .or_else(|e| match e {
                ParseError::InvalidSyntax { .. } => Err(ParseError::InvalidSyntax { field: spec.field, offset: item_offset }),
                other => Err(other),
            })?;
        let _ = parse_endpoint(right, item_offset + h + 1, spec)
            .or_else(|e| match e {
                ParseError::InvalidSyntax { .. } => Err(ParseError::InvalidSyntax { field: spec.field, offset: item_offset }),
                other => Err(other),
            })?;
    } else {
        let _ = parse_endpoint(base, item_offset, spec)
            .or_else(|e| match e {
                ParseError::InvalidSyntax { .. } => Err(ParseError::InvalidSyntax { field: spec.field, offset: item_offset }),
                other => Err(other),
            })?;
    }

    let mut start;
    let mut end;
    if base == b"*" {
        start = spec.min;
        end = spec.max;
    } else if let Some(h) = range {
        start = parse_endpoint(&base[..h], item_offset, spec)?;
        end = parse_endpoint(&base[h + 1..], item_offset + h + 1, spec)?;
        if start > end {
            return Err(ParseError::ReversedRange {
                field: spec.field,
                offset: item_offset + h + 1,
            });
        }
    } else {
        start = parse_endpoint(base, item_offset, spec)?;
        end = if slash.is_some() { spec.max } else { start };
    }

    let step = if let Some(s) = slash {
        let part = &item[s + 1..];
        if !part.iter().all(u8::is_ascii_digit) || part.is_empty() {
            return Err(ParseError::InvalidSyntax { field: spec.field, offset: item_offset });
        }
        parse_number(part, item_offset + s + 1, spec, true)?
    } else {
        1
    };

    if slash.is_none() && base != b"*" && range.is_none() {
        selected[start as usize] = true;
        return Ok(());
    }

    let mut v = start;
    while v <= end {
        selected[v as usize] = true;
        match v.checked_add(step) {
            Some(next) => v = next,
            None => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.as_bytes().first() == Some(&b'@') {
        match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        input
    };

    let parts: Vec<&str> = expanded.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|i| {
        vec![false; SPECS[i].max as usize + 1]
    });
    let mut unrestricted = [false; 2];

    for (i, part) in parts.iter().enumerate() {
        let spec = SPECS[i];
        let bytes = part.as_bytes();
        for (offset, &b) in bytes.iter().enumerate() {
            if !(b.is_ascii_digit()
                || b.is_ascii_alphabetic()
                || matches!(b, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field: spec.field, offset });
            }
        }

        if (i == 2 || i == 4) && *part == "*" {
            unrestricted[if i == 2 { 0 } else { 1 }] = true;
        }

        let mut start = 0usize;
        for (pos, item) in bytes.iter().enumerate() {
            if *item == b',' {
                parse_item(&bytes[start..pos], start, spec, &mut selected[i])?;
                start = pos + 1;
            }
        }
        parse_item(&bytes[start..], start, spec, &mut selected[i])?;
    }

    Ok(Cron { selected, unrestricted })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let i = field_index(field);
        self.selected[i]
            .iter()
            .enumerate()
            .filter_map(|(value, yes)| if *yes { Some(value as u8) } else { None })
            .collect()
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.selected[0][at.minute as usize]
            || !self.selected[1][at.hour as usize]
            || !self.selected[3][at.month as usize]
        {
            return false;
        }

        let dom = self.selected[2][at.day as usize];
        let dow = self.selected[4][weekday(at.year, at.month, at.day) as usize];
        match (self.unrestricted[0], self.unrestricted[1]) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == month_length(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    fn first_in_date(&self, at: CivilDateTime, min_time: u16) -> Option<CivilDateTime> {
        if !self.matches_valid(CivilDateTime { hour: 0, minute: 0, ..at }) {
            return None;
        }
        for hour in self.allowed(Field::Hour) {
            for minute in self.allowed(Field::Minute) {
                let time = hour as u16 * 60 + minute as u16;
                if time >= min_time {
                    return Some(CivilDateTime { hour, minute, ..at });
                }
            }
        }
        None
    }

    fn last_in_date(&self, at: CivilDateTime, max_time: i32) -> Option<CivilDateTime> {
        if !self.matches_valid(CivilDateTime { hour: 0, minute: 0, ..at }) {
            return None;
        }
        let hours = self.allowed(Field::Hour);
        let minutes = self.allowed(Field::Minute);
        for &hour in hours.iter().rev() {
            for &minute in minutes.iter().rev() {
                let time = hour as i32 * 60 + minute as i32;
                if time <= max_time {
                    return Some(CivilDateTime { hour, minute, ..at });
                }
            }
        }
        None
    }

    fn advance_date(at: CivilDateTime) -> Option<CivilDateTime> {
        if at.day < month_length(at.year, at.month) {
            Some(CivilDateTime { day: at.day + 1, hour: 0, minute: 0, ..at })
        } else if at.month < 12 {
            Some(CivilDateTime { month: at.month + 1, day: 1, hour: 0, minute: 0, ..at })
        } else if at.year < 2399 {
            Some(CivilDateTime { year: at.year + 1, month: 1, day: 1, hour: 0, minute: 0 })
        } else {
            None
        }
    }

    fn retreat_date(at: CivilDateTime) -> Option<CivilDateTime> {
        if at.day > 1 {
            Some(CivilDateTime { day: at.day - 1, hour: 0, minute: 0, ..at })
        } else if at.month > 1 {
            let month = at.month - 1;
            Some(CivilDateTime {
                month,
                day: month_length(at.year, month),
                hour: 0,
                minute: 0,
                ..at
            })
        } else if at.year > 2000 {
            Some(CivilDateTime { year: at.year - 1, month: 12, day: 31, hour: 0, minute: 0 })
        } else {
            None
        }
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut date = CivilDateTime { hour: 0, minute: 0, ..after };
        let after_time = after.hour as u16 * 60 + after.minute as u16;
        let mut first = true;
        loop {
            let min_time = if first { after_time + 1 } else { 0 };
            if let Some(found) = self.first_in_date(date, min_time) {
                return Ok(Some(found));
            }
            match Self::advance_date(date) {
                Some(next) => date = next,
                None => return Ok(None),
            }
            first = false;
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(before)?;
        let mut date = CivilDateTime { hour: 0, minute: 0, ..before };
        let before_time = before.hour as i32 * 60 + before.minute as i32;
        let mut first = true;
        loop {
            let max_time = if first { before_time - 1 } else { 1439 };
            if let Some(found) = self.last_in_date(date, max_time) {
                return Ok(Some(found));
            }
            match Self::retreat_date(date) {
                Some(prev) => date = prev,
                None => return Ok(None),
            }
            first = false;
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate_datetime(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate_datetime(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        if start == end {
            return Ok(FireWindow { fires: Vec::new(), truncated: false });
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        loop {
            let Some(found) = self.next_fire(cursor).expect("validated cursor") else {
                break;
            };
            if found >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow { fires, truncated: true });
            }
            fires.push(found);
            cursor = found;
        }
        Ok(FireWindow { fires, truncated: false })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::new();
        for i in 0..5 {
            let spec = SPECS[i];
            let values: Vec<u8> = self.allowed(spec.field);
            let full: Vec<u8> = (spec.min..=spec.max).collect();
            let unrestricted_day = (i == 2 && self.unrestricted[0]) || (i == 4 && self.unrestricted[1]);
            if unrestricted_day {
                fields.push("*".to_string());
            } else if values == full {
                if i == 2 || i == 4 {
                    fields.push("*/1".to_string());
                } else {
                    fields.push("*".to_string());
                }
            } else {
                fields.push(values.iter().map(u8::to_string).collect::<Vec<_>>().join(","));
            }
        }
        fields.join(" ")
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn c(s: &str) -> Cron {
        parse_cron(s).unwrap()
    }

    fn d(year: u16, month: u8, day: u8, hour: u8, minute: u8) -> CivilDateTime {
        CivilDateTime { year, month, day, hour, minute }
    }

    #[test]
    fn p_parse_canonical_allowed() {
        let q = c("15 5 * * *");
        assert_eq!(q.canonical(), "15 5 * * *");
        assert_eq!(q.allowed(Field::Minute), vec![15]);
        assert_eq!(c("4,2 3 * MAR MON").canonical(), "2,4 * 3 1");
    }

    #[test]
    fn p_next_and_match() {
        let q = c("0 0 12 * MON");
        assert!(q.matches(d(2000, 1, 12, 0, 0)).unwrap());
        assert_eq!(q.next_fire(d(2000, 1, 12, 1, 0)).unwrap(), Some(d(2000, 1, 17, 0, 0)));
    }

    #[test]
    fn p_previous_daily() {
        assert_eq!(c("15 5 * * *").previous_fire(d(2020, 6, 9, 8, 0)).unwrap(), Some(d(2020, 6, 9, 5, 15)));
    }

    #[test]
    fn p_window_capped() {
        assert_eq!(
            c("10 * * * *").fires_between(d(2020, 6, 9, 1, 10), d(2020, 6, 9, 4, 10), 2).unwrap(),
            FireWindow {
                fires: vec![d(2020, 6, 9, 1, 10), d(2020, 6, 9, 2, 10)],
                truncated: true,
            }
        );
    }
}
