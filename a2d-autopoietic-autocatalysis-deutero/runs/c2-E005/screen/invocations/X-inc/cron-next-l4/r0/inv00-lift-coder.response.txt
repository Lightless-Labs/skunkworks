use std::fmt::Write;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_days: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &yes)| yes.then_some(value as u8 + field_min(field)))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut date = (after.year, after.month, after.day);
        while let Some(current_date) = date {
            if self.date_matches(current_date.0, current_date.1, current_date.2) {
                let start_hour = if current_date == (after.year, after.month, after.day) {
                    after.hour
                } else {
                    0
                };
                for hour in self.allowed(Field::Hour) {
                    if hour < start_hour {
                        continue;
                    }
                    let start_minute =
                        if current_date == (after.year, after.month, after.day) && hour == after.hour {
                            after.minute
                        } else {
                            0
                        };
                    for minute in self.allowed(Field::Minute) {
                        if minute < start_minute {
                            continue;
                        }
                        let candidate = CivilDateTime {
                            year: current_date.0,
                            month: current_date.1,
                            day: current_date.2,
                            hour,
                            minute,
                        };
                        if candidate > after {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            date = next_date(current_date.0, current_date.1, current_date.2);
        }
        Ok(None)
    }

    pub fn previous_fire(
        &self,
        before: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(before)?;
        let mut date = (before.year, before.month, before.day);
        while let Some(current_date) = date {
            if self.date_matches(current_date.0, current_date.1, current_date.2) {
                let mut hours = self.allowed(Field::Hour);
                hours.reverse();
                let start_hour = if current_date == (before.year, before.month, before.day) {
                    before.hour
                } else {
                    23
                };
                for hour in hours {
                    if hour > start_hour {
                        continue;
                    }
                    let start_minute =
                        if current_date == (before.year, before.month, before.day) && hour == before.hour {
                            before.minute
                        } else {
                            59
                        };
                    let mut minutes = self.allowed(Field::Minute);
                    minutes.reverse();
                    for minute in minutes {
                        if minute > start_minute {
                            continue;
                        }
                        let candidate = CivilDateTime {
                            year: current_date.0,
                            month: current_date.1,
                            day: current_date.2,
                            hour,
                            minute,
                        };
                        if candidate < before {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            date = previous_date(current_date.0, current_date.1, current_date.2);
        }
        Ok(None)
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate_datetime(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate_datetime(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        if start == end {
            return Ok(FireWindow {
                fires: Vec::new(),
                truncated: false,
            });
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while let Some(candidate) = self.next_fire(cursor).expect("validated datetime") {
            if candidate >= end {
                break;
            }
            if fires.len() == cap {
                return Ok(FireWindow {
                    fires,
                    truncated: true,
                });
            }
            fires.push(candidate);
            cursor = candidate;
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::with_capacity(5);
        for field in [
            Field::Minute,
            Field::Hour,
            Field::DayOfMonth,
            Field::Month,
            Field::DayOfWeek,
        ] {
            let i = field_index(field);
            let full = self.selected[i].iter().all(|&x| x);
            if field == Field::DayOfMonth && self.unrestricted_days[0] {
                fields.push("*".to_string());
            } else if field == Field::DayOfWeek && self.unrestricted_days[1] {
                fields.push("*".to_string());
            } else if full
                && matches!(field, Field::Minute | Field::Hour | Field::Month)
            {
                fields.push("*".to_string());
            } else if full && matches!(field, Field::DayOfMonth | Field::DayOfWeek) {
                fields.push("*/1".to_string());
            } else {
                let mut out = String::new();
                for (n, value) in self.allowed(field).iter().enumerate() {
                    if n != 0 {
                        out.push(',');
                    }
                    let _ = write!(out, "{value}");
                }
                fields.push(out);
            }
        }
        fields.join(" ")
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        self.selected[field_index(Field::Minute)][at.minute as usize]
            && self.selected[field_index(Field::Hour)][at.hour as usize]
            && self.selected[field_index(Field::Month)][at.month as usize - 1]
            && self.date_matches(at.year, at.month, at.day)
    }

    fn date_matches(&self, year: u16, month: u8, day: u8) -> bool {
        let dom_any = self.unrestricted_days[0];
        let dow_any = self.unrestricted_days[1];
        let dom = self.selected[field_index(Field::DayOfMonth)][day as usize - 1];
        let dow = self.selected[field_index(Field::DayOfWeek)][weekday(year, month, day) as usize];

        match (dom_any, dow_any) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if day == days_in_month(year, month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    if input.as_bytes().first() == Some(&b'@') {
        let expansion = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        return parse_regular(expansion);
    }
    parse_regular(input)
}

fn parse_regular(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|i| {
        vec![false; (field_max(fields[i]) - field_min(fields[i]) + 1) as usize]
    });
    let mut unrestricted_days = [false; 2];

    for i in 0..5 {
        let field = fields[i];
        let text = parts[i];
        for (offset, byte) in text.bytes().enumerate() {
            if !(byte.is_ascii_digit()
                || byte.is_ascii_alphabetic()
                || matches!(byte, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        if field == Field::DayOfMonth && text == "*" {
            unrestricted_days[0] = true;
        }
        if field == Field::DayOfWeek && text == "*" {
            unrestricted_days[1] = true;
        }

        let mut start = 0;
        for item in text.split(',') {
            if item.is_empty() {
                return Err(ParseError::EmptyItem {
                    field,
                    offset: start,
                });
            }
            parse_item(item, start, field, &mut selected[i])?;
            start += item.len() + 1;
        }
    }

    Ok(Cron {
        selected,
        unrestricted_days,
    })
}

fn parse_item(
    item: &str,
    offset: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let bytes = item.as_bytes();
    let slashes: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slashes.len() > 1 {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    let (base, step_text, step_offset) = if let Some(&slash) = slashes.first() {
        (&item[..slash], Some(&item[slash + 1..]), offset + slash + 1)
    } else {
        (item, None, 0)
    };
    if base.is_empty() || step_text == Some("") {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    let hyphens: Vec<usize> = base
        .as_bytes()
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();
    if hyphens.len() > 1 || base == "*" && !hyphens.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    let (left, right) = if let Some(&hyphen) = hyphens.first() {
        let l = &base[..hyphen];
        let r = &base[hyphen + 1..];
        if l.is_empty() || r.is_empty() || l == "*" || r == "*" {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        (l, Some((r, hyphen + 1)))
    } else {
        (base, None)
    };

    let base_is_star = base == "*";
    let start_value = if base_is_star {
        None
    } else {
        Some(parse_atom(left, field, offset)?)
    };
    let end_value = if let Some((right_text, right_rel)) = right {
        Some(parse_atom(right_text, field, offset + right_rel)?)
    } else {
        None
    };

    if let (Some(start), Some(end)) = (start_value, end_value) {
        if start > end {
            return Err(ParseError::ReversedRange {
                field,
                offset: offset + right.unwrap().1,
            });
        }
    }

    let step = if let Some(step_text) = step_text {
        let step_value = parse_step(step_text, field, step_offset)?;
        Some(step_value)
    } else {
        None
    };

    let min = field_min(field);
    let max = field_max(field);
    let (from, to) = if base_is_star {
        (min, max)
    } else if let Some(end) = end_value {
        (start_value.unwrap(), end)
    } else if step.is_some() {
        (start_value.unwrap(), max)
    } else {
        let value = start_value.unwrap();
        (value, value)
    };

    let increment = step.unwrap_or(1);
    let mut value = from;
    loop {
        selected[(value - min) as usize] = true;
        if value.saturating_add(increment) > to {
            break;
        }
        value += increment;
    }
    Ok(())
}

fn parse_atom(text: &str, field: Field, offset: usize) -> Result<u8, ParseError> {
    if text.bytes().all(|b| b.is_ascii_digit()) {
        return parse_number(text, field, offset, false).map(|v| v as u8);
    }
    if let Some(value) = name_value(text, field) {
        return Ok(value);
    }
    Err(ParseError::InvalidSyntax { field, offset })
}

fn parse_step(text: &str, field: Field, offset: usize) -> Result<u8, ParseError> {
    if !text.bytes().all(|b| b.is_ascii_digit()) || text.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset: offset.saturating_sub(text.len()) });
    }
    let value = parse_number(text, field, offset, true)?;
    let width = field_max(field) - field_min(field) + 1;
    if value == 0 || value > width as u32 {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(value as u8)
}

fn parse_number(
    text: &str,
    field: Field,
    offset: usize,
    _step: bool,
) -> Result<u32, ParseError> {
    if text.len() > 1 && text.as_bytes()[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for byte in text.bytes() {
        let digit = (byte - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(digit))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    if value < field_min(field) as u32 || value > field_max(field) as u32 {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(value)
}

fn name_value(text: &str, field: Field) -> Option<u8> {
    let upper = text.to_ascii_uppercase();
    match field {
        Field::Month => [
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC",
        ]
        .iter()
        .position(|&name| name == upper)
        .map(|i| i as u8 + 1),
        Field::DayOfWeek => ["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"]
            .iter()
            .position(|&name| name == upper)
            .map(|i| i as u8),
        _ => None,
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_min(field: Field) -> u8 {
    match field {
        Field::Minute | Field::Hour | Field::DayOfWeek => 0,
        Field::DayOfMonth | Field::Month => 1,
    }
}

fn field_max(field: Field) -> u8 {
    match field {
        Field::Minute => 59,
        Field::Hour => 23,
        Field::DayOfMonth => 31,
        Field::Month => 12,
        Field::DayOfWeek => 6,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn next_date(year: u16, month: u8, day: u8) -> Option<(u16, u8, u8)> {
    if day < days_in_month(year, month) {
        Some((year, month, day + 1))
    } else if month < 12 {
        Some((year, month + 1, 1))
    } else if year < 2399 {
        Some((year + 1, 1, 1))
    } else {
        None
    }
}

fn previous_date(year: u16, month: u8, day: u8) -> Option<(u16, u8, u8)> {
    if day > 1 {
        Some((year, month, day - 1))
    } else if month > 1 {
        let m = month - 1;
        Some((year, m, days_in_month(year, m)))
    } else if year > 2000 {
        Some((year - 1, 12, 31))
    } else {
        None
    }
}
