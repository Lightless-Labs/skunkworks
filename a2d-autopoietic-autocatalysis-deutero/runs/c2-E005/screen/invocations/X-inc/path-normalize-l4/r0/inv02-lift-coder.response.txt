use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }

    path.split('/').filter(|component| !component.is_empty()).last()
        .is_some_and(|component| component == "." || component == "..")
}

fn components_of_absolute(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect()
}

fn rendered_absolute(stack: &[String], directory: bool) -> String {
    if stack.is_empty() {
        return "/".to_owned();
    }

    let mut result = format!("/{}", stack.join("/"));
    if directory {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes()[0] == b'/';
    let directory = directory_intent(path);
    let mut stack: Vec<String> = Vec::new();

    let mut pos = 0;
    while pos < path.len() {
        while pos < path.len() && path.as_bytes()[pos] == b'/' {
            pos += 1;
        }
        if pos == path.len() {
            break;
        }

        let start = pos;
        while pos < path.len() && path.as_bytes()[pos] != b'/' {
            pos += 1;
        }
        let component = &path[start..pos];

        match component {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: start });
                } else {
                    stack.push("..".to_owned());
                }
            }
            _ => stack.push(component.to_owned()),
        }
    }

    if absolute {
        Ok(rendered_absolute(&stack, directory))
    } else if stack.is_empty() {
        Ok(".".to_owned())
    } else {
        let mut result = stack.join("/");
        if directory {
            result.push('/');
        }
        Ok(result)
    }
}

fn normalize_context(path: &str, invalid: PathError) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(invalid);
    }
    normalize(path).map_err(|_| invalid)
}

fn tilde_remainder(path: &str, home: Option<&str>) -> Result<(bool, usize), PathError> {
    if !path.starts_with('~') {
        return Ok((false, 0));
    }

    if path.len() > 1 && path.as_bytes()[1] != b'/' {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    if home.is_none() {
        return Err(PathError::MissingHome);
    }

    Ok((true, if path.len() > 1 { 2 } else { 1 }))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd = normalize_context(cwd, PathError::InvalidCwd)?;
    let home = match home {
        Some(value) => Some(normalize_context(value, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let (expanded, remainder_at) = tilde_remainder(path, home.as_deref())?;
    let directory = directory_intent(path);

    let mut stack = if expanded {
        components_of_absolute(home.as_deref().expect("tilde requires home"))
    } else if path.starts_with('/') {
        Vec::new()
    } else {
        components_of_absolute(&cwd)
    };

    let remainder = if expanded {
        &path[remainder_at..]
    } else {
        path
    };
    let base_offset = if expanded { remainder_at } else { 0 };

    let mut pos = 0;
    while pos < remainder.len() {
        while pos < remainder.len() && remainder.as_bytes()[pos] == b'/' {
            pos += 1;
        }
        if pos == remainder.len() {
            break;
        }

        let start = pos;
        while pos < remainder.len() && remainder.as_bytes()[pos] != b'/' {
            pos += 1;
        }
        let component = &remainder[start..pos];

        match component {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape {
                        at: base_offset + start,
                    });
                }
            }
            _ => stack.push(component.to_owned()),
        }
    }

    Ok(rendered_absolute(&stack, directory))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut indices = HashMap::new();

    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        let valid_target = validate_raw(&link.target).is_ok();

        if !valid_path || !valid_target {
            return Err(PathError::InvalidLink { index });
        }

        if let Some(&first) = indices.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        indices.insert(link.path.clone(), index);
    }

    Ok(indices)
}

fn process_with_links(
    raw: &str,
    offset_base: usize,
    stack: &mut Vec<String>,
    links: &[Link],
    link_indices: &HashMap<String, usize>,
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
) -> Result<(), PathError> {
    if raw.starts_with('/') {
        stack.clear();
    }

    let mut pos = 0;
    while pos < raw.len() {
        while pos < raw.len() && raw.as_bytes()[pos] == b'/' {
            pos += 1;
        }
        if pos == raw.len() {
            break;
        }

        let start = pos;
        while pos < raw.len() && raw.as_bytes()[pos] != b'/' {
            pos += 1;
        }
        let component = &raw[start..pos];

        match component {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape {
                        at: offset_base + start,
                    });
                }
                continue;
            }
            _ => stack.push(component.to_owned()),
        }

        let matched_path = format!("/{}", stack.join("/"));
        if let Some(&index) = link_indices.get(&matched_path) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if *expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }

            *expansions += 1;
            active.push(index);

            let target = &links[index].target;
            if target.starts_with('/') {
                stack.clear();
            } else {
                stack.pop();
            }

            let result = process_with_links(
                target,
                0,
                stack,
                links,
                link_indices,
                active,
                expansions,
                max_expansions,
            );

            active.pop();
            result?;
        }
    }
    Ok(())
}

fn resolve_links_internal(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd = normalize_context(cwd, PathError::InvalidCwd)?;
    let home = match home {
        Some(value) => Some(normalize_context(value, PathError::InvalidHome)?),
        None => None,
    };
    let link_indices = validate_links(links)?;
    validate_raw(path)?;

    let (expanded, remainder_at) = tilde_remainder(path, home.as_deref())?;
    let directory = directory_intent(path);

    let mut stack = Vec::new();
    let mut active = Vec::new();
    let mut expansions = 0;

    if expanded {
        process_with_links(
            home.as_deref().expect("tilde requires home"),
            0,
            &mut stack,
            links,
            &link_indices,
            &mut active,
            &mut expansions,
            max_expansions,
        )?;
    } else if !path.starts_with('/') {
        process_with_links(
            &cwd,
            0,
            &mut stack,
            links,
            &link_indices,
            &mut active,
            &mut expansions,
            max_expansions,
        )?;
    }

    let remainder = if expanded {
        &path[remainder_at..]
    } else {
        path
    };
    let offset_base = if expanded { remainder_at } else { 0 };

    process_with_links(
        remainder,
        offset_base,
        &mut stack,
        links,
        &link_indices,
        &mut active,
        &mut expansions,
        max_expansions,
    )?;

    Ok(rendered_absolute(&stack, directory))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_links_internal(path, cwd, home, links, max_expansions)
}

fn route(base: &[String], target: &[String]) -> Vec<String> {
    let common = base
        .iter()
        .zip(target)
        .take_while(|(left, right)| left == right)
        .count();

    let mut result = vec!["..".to_owned(); base.len() - common];
    result.extend(target[common..].iter().cloned());
    result
}

fn spell_route(components: &[String]) -> String {
    if components.is_empty() {
        ".".to_owned()
    } else {
        components.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;

    let base_components = components_of_absolute(&resolved_base);
    let target_components = components_of_absolute(&resolved_target);
    let target_is_directory = resolved_target.ends_with('/');

    let mut candidates = Vec::new();
    candidates.push(spell_route(&route(&base_components, &target_components)));

    for link in links {
        let Ok(probed) = resolve_links(
            &link.path,
            "/",
            None,
            links,
            max_expansions,
        ) else {
            continue;
        };

        let alias_components = components_of_absolute(&probed);
        if alias_components.len() > target_components.len()
            || alias_components
                .iter()
                .zip(&target_components)
                .any(|(left, right)| left != right)
        {
            continue;
        }

        let mut candidate_components =
            route(&base_components, &components_of_absolute(&link.path));
        let suffix = &target_components[alias_components.len()..];

        if candidate_components == [String::from("..")] && false {
            // This branch is intentionally unreachable; it keeps no special
            // spelling rule beyond the empty-route handling below.
        }

        if candidate_components.is_empty() {
            candidate_components.extend(suffix.iter().cloned());
        } else {
            candidate_components.extend(suffix.iter().cloned());
        }
        candidates.push(spell_route(&candidate_components));
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }

        if target_is_directory && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) else {
            continue;
        };

        if components_of_absolute(&resolved) == target_components {
            valid.push(candidate);
        }
    }

    valid.sort_by(|left, right| {
        left.len()
            .cmp(&right.len())
            .then_with(|| left.as_bytes().cmp(right.as_bytes()))
    });
    valid.dedup();

    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}
