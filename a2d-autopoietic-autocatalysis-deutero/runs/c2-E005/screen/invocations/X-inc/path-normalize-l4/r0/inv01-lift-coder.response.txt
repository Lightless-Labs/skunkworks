#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|part| !part.is_empty()).collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|part| *part == "." || *part == "..")
}

fn normalized_parts(path: &str) -> Result<(bool, Vec<String>), PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let mut stack: Vec<String> = Vec::new();

    for part in components(path) {
        match part {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    let at = path.find(part).unwrap_or(0);
                    return Err(PathError::Escape { at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            _ => stack.push(part.to_owned()),
        }
    }

    Ok((absolute, stack))
}

fn render_parts(absolute: bool, parts: &[String], want_dir: bool) -> String {
    let mut result = String::new();
    if absolute {
        result.push('/');
    }
    result.push_str(&parts.join("/"));

    if result.is_empty() {
        result.push('.');
    }
    if want_dir && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    let want_dir = directory_intent(path);
    let (absolute, parts) = normalized_parts(path)?;
    Ok(render_parts(absolute, &parts, want_dir))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let (_, parts) = normalized_parts(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(parts)
}

fn validate_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    match home {
        None => Ok(None),
        Some(path) => {
            if !path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            let (_, parts) = normalized_parts(path).map_err(|_| PathError::InvalidHome)?;
            Ok(Some(parts))
        }
    }
}

fn tilde_remainder<'a>(
    path: &'a str,
    home: Option<&[String]>,
) -> Result<Option<(&'a str, Vec<String>)>, PathError> {
    if !path.starts_with('~') {
        return Ok(None);
    }
    let bytes = path.as_bytes();
    if bytes.len() > 1 && bytes[1] != b'/' {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    let home = home.ok_or(PathError::MissingHome)?.to_vec();
    let remainder = if bytes.len() == 1 { "" } else { &path[2..] };
    Ok(Some((remainder, home)))
}

fn push_part(stack: &mut Vec<String>, part: &str, absolute: bool, at: usize) -> Result<(), PathError> {
    match part {
        "." => Ok(()),
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
                Ok(())
            } else if absolute {
                Err(PathError::Escape { at })
            } else {
                stack.push("..".to_owned());
                Ok(())
            }
        }
        _ => {
            stack.push(part.to_owned());
            Ok(())
        }
    }
}

fn resolve_core(path: &str, cwd: &[String], home: Option<&[String]>) -> Result<String, PathError> {
    validate_raw(path)?;

    let want_dir = directory_intent(path);
    let (remainder, mut stack, absolute) = match tilde_remainder(path, home)? {
        Some((remainder, home_parts)) => (remainder, home_parts, true),
        None if path.starts_with('/') => (&path[1..], Vec::new(), true),
        None => (path, cwd.to_vec(), true),
    };

    let base_offset = path.len() - remainder.len();
    let mut cursor = 0;
    for part in remainder.split('/') {
        if part.is_empty() {
            cursor += 1;
            continue;
        }
        let at = base_offset + cursor;
        push_part(&mut stack, part, absolute, at)?;
        cursor += part.len() + 1;
    }

    Ok(render_parts(true, &stack, want_dir))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validate_home(home)?;
    resolve_core(path, &cwd_parts, home_parts.as_deref())
}

fn path_from_parts(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    for (index, link) in links.iter().enumerate() {
        let path_ok = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path && link.path != "/");
        let target_ok = validate_raw(&link.target).is_ok();
        if !path_ok || !target_ok {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(first) = links[..index]
            .iter()
            .position(|prior| prior.path == link.path)
        {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
    }
    Ok(())
}

fn link_index(links: &[Link], stack: &[String]) -> Option<usize> {
    let current = path_from_parts(stack);
    links.iter().position(|link| link.path == current)
}

fn process_stream(
    raw: &str,
    offset_base: usize,
    stack: &mut Vec<String>,
    links: &[Link],
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
) -> Result<(), PathError> {
    let mut cursor = 0;
    for part in raw.split('/') {
        if part.is_empty() {
            cursor += 1;
            continue;
        }
        let at = offset_base + cursor;
        match part {
            "." => {}
            ".." => {
                if stack.is_empty() {
                    return Err(PathError::Escape { at });
                }
                stack.pop();
            }
            _ => {
                stack.push(part.to_owned());
                if let Some(index) = link_index(links, stack) {
                    if active.contains(&index) {
                        return Err(PathError::Loop { index });
                    }
                    if *expansions >= max_expansions {
                        return Err(PathError::Limit { index });
                    }
                    *expansions += 1;

                    let link_parent_len = stack.len() - 1;
                    let saved_stack = stack.clone();
                    stack.truncate(link_parent_len);
                    active.push(index);
                    let target = &links[index].target;
                    if target.starts_with('/') {
                        stack.clear();
                        process_stream(
                            target,
                            0,
                            stack,
                            links,
                            active,
                            expansions,
                            max_expansions,
                        )?;
                    } else {
                        process_stream(
                            target,
                            0,
                            stack,
                            links,
                            active,
                            expansions,
                            max_expansions,
                        )?;
                    }
                    active.pop();
                    let _ = saved_stack;
                }
            }
        }
        cursor += part.len() + 1;
    }
    Ok(())
}

fn resolve_links_parts(
    path: &str,
    cwd: &[String],
    home: Option<&[String]>,
    links: &[Link],
    max_expansions: usize,
) -> Result<(Vec<String>, bool), PathError> {
    validate_raw(path)?;
    let want_dir = directory_intent(path);
    let (remainder, mut stack, is_absolute, offset_base) = match tilde_remainder(path, home)? {
        Some((remainder, home_parts)) => {
            let start = path.len() - remainder.len();
            (remainder, home_parts, true, start)
        }
        None if path.starts_with('/') => (&path[1..], Vec::new(), true, 1),
        None => (path, cwd.to_vec(), true, 0),
    };

    let mut active = Vec::new();
    let mut expansions = 0;
    if !is_absolute {
        return unreachable!();
    }
    process_stream(
        remainder,
        offset_base,
        &mut stack,
        links,
        &mut active,
        &mut expansions,
        max_expansions,
    )?;
    Ok((stack, want_dir))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validate_home(home)?;
    validate_links(links)?;
    let (parts, want_dir) =
        resolve_links_parts(path, &cwd_parts, home_parts.as_deref(), links, max_expansions)?;
    Ok(render_parts(true, &parts, want_dir))
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target)
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = vec!["..".to_owned(); base.len() - common];
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

fn result_parts(path: &str) -> Vec<String> {
    components(path).into_iter().map(str::to_owned).collect()
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let base_result = resolve_links(base, cwd, home, links, max_expansions)?;
    let target_result = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = result_parts(&base_result);
    let target_parts = result_parts(&target_result);
    let target_dir = target_result.ends_with('/');

    let mut candidates = vec![route(&base_parts, &target_parts)];
    for link in links {
        if let Ok(probe) = resolve_links(&link.path, "/", None, links, max_expansions) {
            let probe_parts = result_parts(&probe);
            if target_parts.starts_with(&probe_parts) {
                let mut candidate = route(&base_parts, &components(&link.path).into_iter().map(str::to_owned).collect::<Vec<_>>());
                let suffix = &target_parts[probe_parts.len()..];
                if !suffix.is_empty() {
                    if candidate == "." {
                        candidate = suffix.join("/");
                    } else {
                        candidate.push('/');
                        candidate.push_str(&suffix.join("/"));
                    }
                }
                candidates.push(candidate);
            }
        }
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_dir && candidate != "." {
            candidate.push('/');
        }

        if let Ok(resolved) =
            resolve_links(&candidate, &base_result, home, links, max_expansions)
        {
            if result_parts(&resolved) == target_parts {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Empty)
}
