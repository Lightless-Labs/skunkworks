#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|part| !part.is_empty())
            .is_some_and(|part| part == "." || part == "..")
}

fn raw_components(path: &str, offset_base: usize) -> Vec<(String, usize)> {
    let bytes = path.as_bytes();
    let mut components = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }

        components.push((path[start..i].to_owned(), offset_base + start));
    }

    components
}

fn reduce_components(
    stack: &mut Vec<String>,
    components: impl IntoIterator<Item = (String, usize)>,
    absolute: bool,
) -> Result<(), PathError> {
    for (component, at) in components {
        match component.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(component);
                }
            }
            _ => stack.push(component),
        }
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if absolute && stack.is_empty() {
        return "/".to_owned();
    }
    if !absolute && stack.is_empty() {
        return ".".to_owned();
    }

    if intent {
        result.push('/');
    }
    result
}

fn canonical_absolute(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::Escape { at: 0 });
    }
    validate_raw(path)?;
    let mut stack = Vec::new();
    reduce_components(&mut stack, raw_components(path, 0), true)?;
    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    reduce_components(&mut stack, raw_components(path, 0), absolute)?;
    Ok(render(&stack, absolute, intent))
}

fn validate_contexts(
    cwd: &str,
    home: Option<&str>,
) -> Result<(Vec<String>, Option<Vec<String>>), PathError> {
    let cwd_parts = canonical_absolute(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(home) => Some(canonical_absolute(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };
    Ok((cwd_parts, home_parts))
}

fn request_start<'a>(
    path: &'a str,
    home: Option<&'a [String]>,
) -> Result<(bool, Vec<String>, &'a str, usize), PathError> {
    if path.starts_with('/') {
        return Ok((true, Vec::new(), path, 0));
    }

    if path.starts_with('~') {
        if path.len() == 1 {
            let home = home.ok_or(PathError::MissingHome)?;
            return Ok((true, home.to_vec(), "", path.len()));
        }

        if path.as_bytes()[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }

        let home = home.ok_or(PathError::MissingHome)?;
        return Ok((true, home.to_vec(), &path[2..], 2));
    }

    Ok((true, Vec::new(), path, 0))
}

fn resolve_plain_after_validation(
    path: &str,
    cwd_parts: &[String],
    home_parts: Option<&[String]>,
) -> Result<String, PathError> {
    let intent = directory_intent(path);
    let (absolute, mut stack, remainder, offset) = request_start(path, home_parts)?;

    if !path.starts_with('/') && !path.starts_with('~') {
        stack = cwd_parts.to_vec();
    }

    reduce_components(&mut stack, raw_components(remainder, offset), absolute)?;
    Ok(render(&stack, true, intent))
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validate_contexts(cwd, home)?;
    validate_raw(path)?;
    resolve_plain_after_validation(path, &cwd_parts, home_parts.as_deref())
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();

    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        let valid_target = validate_raw(&link.target).is_ok();

        if !valid_path || !valid_target {
            return Err(PathError::InvalidLink { index });
        }

        if let Some((_, first)) = seen.iter().find(|(path, _)| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }

    Ok(())
}

#[derive(Clone)]
struct PendingComponent {
    value: String,
    at: usize,
    absolute_offset: bool,
}

fn push_raw_components(
    queue: &mut Vec<PendingComponent>,
    raw: &str,
    offset_base: usize,
    absolute_offset: bool,
) {
    queue.extend(raw_components(raw, offset_base).into_iter().map(|(value, at)| {
        PendingComponent {
            value,
            at,
            absolute_offset,
        }
    }));
}

fn absolute_stack_path(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn traverse_links(
    mut stack: Vec<String>,
    initial: Vec<PendingComponent>,
    links: &[Link],
    max_expansions: usize,
) -> Result<Vec<String>, PathError> {
    let mut queue = initial;
    let mut active = Vec::<usize>::new();
    let mut expansion_count = 0usize;

    while let Some(component) = queue.first().cloned() {
        queue.remove(0);

        match component.value.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    if component.absolute_offset {
                        return Err(PathError::Escape { at: component.at });
                    }
                    return Err(PathError::Escape { at: component.at });
                }
                continue;
            }
            _ => {}
        }

        stack.push(component.value);
        let matched_path = absolute_stack_path(&stack);
        let Some((index, link)) = links.iter().enumerate().find(|(_, link)| link.path == matched_path)
        else {
            continue;
        };

        if active.contains(&index) {
            return Err(PathError::Loop { index });
        }
        if expansion_count >= max_expansions {
            return Err(PathError::Limit { index });
        }
        expansion_count += 1;

        stack.pop();
        active.push(index);

        let target = &link.target;
        let target_absolute = target.starts_with('/');
        let mut target_stack = if target_absolute {
            Vec::new()
        } else {
            stack.clone()
        };

        let mut target_queue = Vec::new();
        push_raw_components(&mut target_queue, target, 0, false);
        target_queue.extend(queue);
        queue = target_queue;

        // A completion marker keeps the link active until its target, including
        // nested targets, has been fully traversed.
        queue.insert(
            0,
            PendingComponent {
                value: format!("\u{0}LINK_END:{index}"),
                at: 0,
                absolute_offset: false,
            },
        );

        let mut nested_queue = Vec::new();
        while let Some(item) = queue.first().cloned() {
            if item.value.starts_with("\u{0}LINK_END:") {
                let marker = queue.remove(0);
                if let Some(end) = marker.value.strip_prefix("\u{0}LINK_END:") {
                    if let Ok(done) = end.parse::<usize>() {
                        if active.last() == Some(&done) {
                            active.pop();
                        }
                    }
                }
                break;
            }
            nested_queue.push(queue.remove(0));
        }

        // Store the target's components ahead of the outer remainder. The
        // marker is reinserted after processing through a stack of frames.
        let _ = &mut target_stack;
        queue = nested_queue
            .into_iter()
            .chain(std::iter::once(PendingComponent {
                value: format!("\u{0}DEFER:{index}"),
                at: 0,
                absolute_offset: false,
            }))
            .chain(queue.into_iter())
            .collect();

        // Defer marker handling is performed at the top of the next iteration.
        if let Some(marker_pos) = queue.iter().position(|c| c.value.starts_with("\u{0}DEFER:")) {
            let deferred = queue.remove(marker_pos);
            if let Some(raw_index) = deferred.value.strip_prefix("\u{0}DEFER:") {
                if let Ok(done) = raw_index.parse::<usize>() {
                    if active.last() == Some(&done) {
                        active.pop();
                    }
                }
            }
        }
    }

    Ok(stack)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let (cwd_parts, home_parts) = validate_contexts(cwd, home)?;
    validate_links(links)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (absolute, mut stack, remainder, offset) = request_start(path, home_parts.as_deref())?;
    if absolute && !path.starts_with('/') && !path.starts_with('~') {
        stack = cwd_parts;
    }
    if path.starts_with('/') {
        stack.clear();
    }

    let mut initial = Vec::new();

    if !path.starts_with('/') && !path.starts_with('~') {
        push_raw_components(&mut initial, &stack.join("/"), 0, false);
        stack.clear();
    } else if path.starts_with('~') {
        push_raw_components(&mut initial, &stack.join("/"), 0, false);
        stack.clear();
    }

    push_raw_components(&mut initial, remainder, offset, true);

    let mut result = traverse_links(stack, initial, links, max_expansions)?;
    // Context components were traversed before request components; starting
    // paths are already normalized, so their offsets do not affect errors.
    if !path.starts_with('/') && !path.starts_with('~') && result.is_empty() {
        result = cwd_parts;
    }

    Ok(render(&result, true, intent))
}

fn result_components(path: &str) -> Vec<String> {
    path.trim_start_matches('/')
        .trim_end_matches('/')
        .split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();

    let mut parts = vec!["..".to_owned(); base.len() - common];
    parts.extend(target[common..].iter().cloned());

    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_parts = result_components(&resolved_base);
    let target_parts = result_components(&resolved_target);
    let target_directory = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        let Ok(probe) = resolve_links(&link.path, "/", None, links, max_expansions) else {
            continue;
        };
        let probe_parts = result_components(&probe);
        if probe_parts.len() > target_parts.len()
            || probe_parts
                .iter()
                .zip(target_parts.iter())
                .any(|(a, b)| a != b)
        {
            continue;
        }

        let mut candidate = route(&base_parts, &result_components(&link.path));
        let suffix = &target_parts[probe_parts.len()..];
        if !suffix.is_empty() {
            if candidate == "." {
                candidate = suffix.join("/");
            } else {
                candidate.push('/');
                candidate.push_str(&suffix.join("/"));
            }
        }
        candidates.push(candidate);
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_directory && candidate != "." {
            candidate.push('/');
        }

        if resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        )
        .is_ok_and(|resolved| result_components(&resolved) == target_parts)
        {
            valid.push(candidate);
        }
    }

    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}
