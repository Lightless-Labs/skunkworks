use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let mut core = [0u64; 3];

    for (index, number) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if start == pos {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut value = 0u64;
        for &digit in &bytes[start..pos] {
            let d = (digit - b'0') as u64;
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add(d))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        *number = value;

        if index < 2 {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() {
        match bytes[pos] {
            b'-' => {
                pos += 1;
                let pre_start = pos;
                while pos < bytes.len() && bytes[pos] != b'+' {
                    pos += 1;
                }
                parse_identifiers(&bytes[pre_start..pos], pre_start, true, &mut pre)?;

                if pos < bytes.len() {
                    pos += 1;
                    parse_identifiers(&bytes[pos..], pos, false, &mut build)?;
                }
            }
            b'+' => {
                pos += 1;
                parse_identifiers(&bytes[pos..], pos, false, &mut build)?;
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    bytes: &[u8],
    base: usize,
    check_numeric_leading_zero: bool,
    output: &mut Vec<String>,
) -> Result<(), VersionError> {
    let mut start = 0;

    loop {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'.')
            .map(|offset| start + offset)
            .unwrap_or(bytes.len());

        if start == end {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }

        for (offset, &byte) in bytes[start..end].iter().enumerate() {
            if !(byte.is_ascii_alphanumeric() || byte == b'-') {
                return Err(VersionError::InvalidChar {
                    at: base + start + offset,
                });
            }
        }

        let identifier = &bytes[start..end];
        if check_numeric_leading_zero
            && identifier.len() > 1
            && identifier[0] == b'0'
            && identifier.iter().all(u8::is_ascii_digit)
        {
            return Err(VersionError::LeadingZero { at: base + start });
        }

        output.push(String::from_utf8(identifier.to_vec()).expect("ASCII identifier"));

        if end == bytes.len() {
            break;
        }
        start = end + 1;
    }

    Ok(())
}

fn is_numeric_identifier(identifier: &str) -> bool {
    !identifier.is_empty() && identifier.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_identifiers(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core_order = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core_order != Ordering::Equal {
        return core_order;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (a_id, b_id) in a.pre.iter().zip(&b.pre) {
        let a_numeric = is_numeric_identifier(a_id);
        let b_numeric = is_numeric_identifier(b_id);
        let order = match (a_numeric, b_numeric) {
            (true, true) => compare_numeric_identifiers(a_id, b_id),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => a_id.as_bytes().cmp(b_id.as_bytes()),
        };
        if order != Ordering::Equal {
            return order;
        }
    }

    a.pre.len().cmp(&b.pre.len())
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
    admits_prerelease_at: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Copy, Debug)]
enum Operator {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
struct Comparator {
    operator: Operator,
    version: Version,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut comparators = Vec::new();
    let mut admits_prerelease_at = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
        pos += 1;
    }
    if pos == bytes.len() {
        return Err(RangeError::Empty { at: 0 });
    }

    while pos < bytes.len() {
        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];

        let (operator, operator_len) = if token.starts_with(">=") {
            (Operator::GreaterEqual, 2)
        } else if token.starts_with("<=") {
            (Operator::LessEqual, 2)
        } else if token.starts_with('>') {
            (Operator::Greater, 1)
        } else if token.starts_with('<') {
            (Operator::Less, 1)
        } else if token.starts_with('=') {
            (Operator::Equal, 1)
        } else {
            (Operator::Equal, 0)
        };

        let operand_start = token_start + operator_len;
        if operand_start >= token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        if !version.pre.is_empty() {
            let tuple = (version.major, version.minor, version.patch);
            if !admits_prerelease_at.contains(&tuple) {
                admits_prerelease_at.push(tuple);
            }
        }

        comparators.push(Comparator { operator, version });

        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
    }

    Ok(Range {
        comparators,
        admits_prerelease_at,
    })
}

fn is_range_whitespace(byte: u8) -> bool {
    matches!(byte, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn matches(v: &Version, r: &Range) -> bool {
    if !v.pre.is_empty()
        && !r
            .admits_prerelease_at
            .contains(&(v.major, v.minor, v.patch))
    {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let order = compare_versions(v, &comparator.version);
        match comparator.operator {
            Operator::Equal => order == Ordering::Equal,
            Operator::Less => order == Ordering::Less,
            Operator::LessEqual => order != Ordering::Greater,
            Operator::Greater => order == Ordering::Greater,
            Operator::GreaterEqual => order != Ordering::Less,
        }
    })
}
