use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for (field, value) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if start == pos {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        let digits = &bytes[start..pos];
        if digits.len() > 1 && digits[0] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let mut n = 0u64;
        for &digit in digits {
            let d = (digit - b'0') as u64;
            n = match n.checked_mul(10).and_then(|x| x.checked_add(d)) {
                Some(n) => n,
                None => return Err(VersionError::Overflow { at: start }),
            };
        }
        *value = n;

        if field < 2 {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() {
        match bytes[pos] {
            b'-' => {
                pos += 1;
                let pre_start = pos;
                let plus = bytes[pos..]
                    .iter()
                    .position(|&b| b == b'+')
                    .map(|i| pos + i);
                let pre_end = plus.unwrap_or(bytes.len());
                pre = parse_identifiers(bytes, pre_start, pre_end, true)?;
                pos = pre_end;
                if pos < bytes.len() {
                    pos += 1;
                    build = parse_identifiers(bytes, pos, bytes.len(), false)?;
                    pos = bytes.len();
                }
            }
            b'+' => {
                pos += 1;
                build = parse_identifiers(bytes, pos, bytes.len(), false)?;
                pos = bytes.len();
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    bytes: &[u8],
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let mut result = Vec::new();
    let mut id_start = start;

    loop {
        let id_end = bytes[id_start..end]
            .iter()
            .position(|&b| b == b'.')
            .map(|i| id_start + i)
            .unwrap_or(end);

        if id_start == id_end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        for i in id_start..id_end {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
        }

        let identifier = &bytes[id_start..id_end];
        if is_pre
            && identifier.len() > 1
            && identifier[0] == b'0'
            && identifier.iter().all(u8::is_ascii_digit)
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }

        result.push(
            std::str::from_utf8(identifier)
                .expect("validated ASCII identifier")
                .to_owned(),
        );

        if id_end == end {
            break;
        }
        id_start = id_end + 1;
    }

    Ok(result)
}

fn core_cmp(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch))
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(u8::is_ascii_digit)
}

fn compare_identifier(a: &str, b: &str) -> Ordering {
    let a_numeric = is_numeric_identifier(a);
    let b_numeric = is_numeric_identifier(b);

    match (a_numeric, b_numeric) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = core_cmp(a, b);
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (left, right) in a.pre.iter().zip(&b.pre) {
                let cmp = compare_identifier(left, right);
                if cmp != Ordering::Equal {
                    return cmp;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        let (op, op_len) = match bytes[pos] {
            b'=' => (Op::Equal, 1),
            b'<' if pos + 1 < bytes.len() && bytes[pos + 1] == b'=' => (Op::LessEqual, 2),
            b'<' => (Op::Less, 1),
            b'>' if pos + 1 < bytes.len() && bytes[pos + 1] == b'=' => (Op::GreaterEqual, 2),
            b'>' => (Op::Greater, 1),
            _ => (Op::Equal, 0),
        };

        let operand_start = pos + op_len;
        if operand_start >= bytes.len() || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let mut token_end = operand_start;
        while token_end < bytes.len() && !is_range_whitespace(bytes[token_end]) {
            token_end += 1;
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        comparators.push(Comparator { op, version });
        pos = token_end;
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range { comparators })
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn matches(v: &Version, r: &Range) -> bool {
    let admits_prerelease = v.pre.is_empty()
        || r.comparators.iter().any(|comparator| {
            !comparator.version.pre.is_empty() && core_cmp(v, &comparator.version) == Ordering::Equal
        });

    if !admits_prerelease {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let cmp = compare_versions(v, &comparator.version);
        match comparator.op {
            Op::Equal => cmp == Ordering::Equal,
            Op::Less => cmp == Ordering::Less,
            Op::LessEqual => cmp != Ordering::Greater,
            Op::Greater => cmp == Ordering::Greater,
            Op::GreaterEqual => cmp != Ordering::Less,
        }
    })
}
