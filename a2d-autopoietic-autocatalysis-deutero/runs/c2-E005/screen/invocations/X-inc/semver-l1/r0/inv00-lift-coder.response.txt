use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, bytes: &[u8], pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }

    if *pos == start {
        return Err(match bytes.get(start) {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: start },
            _ => VersionError::InvalidChar { at: start },
        });
    }

    let digits = &s[start..*pos];
    if digits.len() > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    bytes: &[u8],
    pos: &mut usize,
    stop_at_plus: bool,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let mut identifiers = Vec::new();

    loop {
        let start = *pos;
        while *pos < bytes.len()
            && bytes[*pos] != b'.'
            && !(stop_at_plus && bytes[*pos] == b'+')
        {
            let b = bytes[*pos];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: *pos });
            }
            *pos += 1;
        }

        if *pos == start {
            return Err(VersionError::EmptyIdentifier { at: start });
        }

        let ident = &s[start..*pos];
        if is_pre
            && ident.len() > 1
            && ident.bytes().all(is_digit)
            && ident.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: start });
        }
        identifiers.push(ident.to_owned());

        if *pos < bytes.len() && bytes[*pos] == b'.' {
            *pos += 1;
            continue;
        }
        break;
    }

    Ok(identifiers)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0;

    let major = parse_core(s, bytes, &mut pos)?;
    for _ in 0..2 {
        if pos >= bytes.len() || bytes[pos] != b'.' {
            return Err(match bytes.get(pos) {
                None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
                _ => VersionError::InvalidChar { at: pos },
            });
        }
        pos += 1;
        let value = parse_core(s, bytes, &mut pos)?;
        if pos == 1 {
            unreachable!();
        }
        if _ == 0 {
            let minor = value;
            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
                let patch = parse_core(s, bytes, &mut pos)?;
                return finish_version(s, bytes, pos, major, minor, patch);
            }
            return Err(match bytes.get(pos) {
                None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
                _ => VersionError::InvalidChar { at: pos },
            });
        }
    }

    unreachable!()
}

fn finish_version(
    s: &str,
    bytes: &[u8],
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        pre = parse_identifiers(s, bytes, &mut pos, true, true)?;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(s, bytes, &mut pos, false, false)?;
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_numeric_identifiers(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (left, right) in a.pre.iter().zip(&b.pre) {
                let left_numeric = numeric_identifier(left);
                let right_numeric = numeric_identifier(right);
                let ordering = match (left_numeric, right_numeric) {
                    (true, true) => compare_numeric_identifiers(left, right),
                    (true, false) => Ordering::Less,
                    (false, true) => Ordering::Greater,
                    (false, false) => left.as_bytes().cmp(right.as_bytes()),
                };
                if ordering != Ordering::Equal {
                    return ordering;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Operator,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Operator {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

fn is_range_space(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_space(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        let (op, op_len) = match bytes.get(pos) {
            Some(b'=') => (Operator::Equal, 1),
            Some(b'<') if bytes.get(pos + 1) == Some(&b'=') => (Operator::LessEqual, 2),
            Some(b'<') => (Operator::Less, 1),
            Some(b'>') if bytes.get(pos + 1) == Some(&b'=') => (Operator::GreaterEqual, 2),
            Some(b'>') => (Operator::Greater, 1),
            _ => (Operator::Equal, 0),
        };
        let operand_start = pos + op_len;

        if operand_start >= bytes.len() || !is_digit(bytes[operand_start]) {
            return Err(RangeError::Syntax { at: token_start });
        }

        let token_end = (operand_start..bytes.len())
            .find(|&i| is_range_space(bytes[i]))
            .unwrap_or(bytes.len());
        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        comparators.push(Comparator { op, version });
        pos = token_end;
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range { comparators })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    let admitted_pre = v.pre.is_empty()
        || r.comparators.iter().any(|comparator| {
            !comparator.version.pre.is_empty()
                && (comparator.version.major, comparator.version.minor, comparator.version.patch)
                    == (v.major, v.minor, v.patch)
        });
    if !admitted_pre {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let ordering = compare_versions(v, &comparator.version);
        match comparator.op {
            Operator::Equal => ordering == Ordering::Equal,
            Operator::Less => ordering == Ordering::Less,
            Operator::LessEqual => ordering != Ordering::Greater,
            Operator::Greater => ordering == Ordering::Greater,
            Operator::GreaterEqual => ordering != Ordering::Less,
        }
    })
}
