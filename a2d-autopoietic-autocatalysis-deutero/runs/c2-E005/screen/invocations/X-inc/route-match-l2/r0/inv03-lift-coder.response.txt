#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
    id: usize,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .last()
            .is_some_and(|s| matches!(s, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { segments, rank, id });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map_or(path.len(), |(i, _)| i);
        let head = &path[..head_end];
        let request = parse_path(head)?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            let Some(params) = match_route(route, &request) else {
                continue;
            };

            let replace = best.as_ref().is_none_or(|(current, _)| {
                route.rank > current.rank
                    || (route.rank == current.rank && route.id < current.id)
            });
            if replace {
                best = Some((route, params));
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let parts = split_segments(input);
    let mut result = Vec::with_capacity(parts.len());
    let mut names = std::collections::HashSet::new();

    for (index, (start, raw)) in parts.iter().enumerate() {
        if raw.is_empty() {
            return Err(Error::InvalidPattern { at: *start });
        }

        let is_last = index + 1 == parts.len();
        if let Some(name) = raw.strip_prefix(':') {
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            result.push(Segment::Param(name.to_owned()));
        } else if let Some(name) = raw.strip_prefix('*') {
            if !is_last || !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            result.push(Segment::Wildcard(name.to_owned()));
        } else {
            result.push(Segment::Static(decode_segment(raw, *start, true)?));
        }
    }

    if input.ends_with('/') {
        return Err(Error::InvalidPattern {
            at: input.len() - 1,
        });
    }

    Ok(result)
}

struct ParsedPath {
    segments: Vec<String>,
}

fn parse_path(input: &str) -> Result<ParsedPath, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
        });
    }

    let parts = split_segments(input);
    let mut segments = Vec::with_capacity(parts.len());

    for (start, raw) in &parts {
        if raw.is_empty() {
            return Err(Error::InvalidPath { at: *start });
        }
        segments.push(decode_segment(raw, *start, false)?);
    }

    if input.ends_with('/') {
        return Err(Error::InvalidPath {
            at: input.len() - 1,
        });
    }

    Ok(ParsedPath { segments })
}

fn split_segments(input: &str) -> Vec<(usize, &str)> {
    let mut parts = Vec::new();
    let mut start = 1;
    for (offset, byte) in input.as_bytes().iter().enumerate().skip(1) {
        if *byte == b'/' {
            parts.push((start, &input[start..offset]));
            start = offset + 1;
        }
    }
    parts.push((start, &input[start..]));
    parts
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, base: usize, pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            let fail = || {
                if pattern {
                    Error::InvalidPattern { at: base + i }
                } else {
                    Error::InvalidPath { at: base + i }
                }
            };
            if i + 2 >= bytes.len() {
                return Err(fail());
            }
            let Some(high) = uppercase_hex(bytes[i + 1]) else {
                return Err(fail());
            };
            let Some(low) = uppercase_hex(bytes[i + 2]) else {
                return Err(fail());
            };
            let value = high * 16 + low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(fail());
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return if pattern {
                Err(Error::InvalidPattern { at: base + i })
            } else {
                Err(Error::InvalidPath { at: base + i })
            };
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn uppercase_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, request: &ParsedPath) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if request.segments.get(path_index)? != value {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                let value = request.segments.get(path_index)?;
                params.push((name.clone(), value.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = request.segments[path_index..].join("/");
                params.push((name.clone(), value));
                path_index = request.segments.len();
            }
        }
    }

    if path_index != request.segments.len() {
        return None;
    }

    Some(params)
}
