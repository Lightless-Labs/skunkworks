#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let rank = segments
            .iter()
            .map(|segment| match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            })
            .chain(
                (!matches!(segments.last(), Some(Segment::Wildcard(_))))
                    .then_some(3),
            )
            .collect();

        let id = self.routes.len();
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .as_bytes()
            .iter()
            .position(|&byte| byte == b'?' || byte == b'#')
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request_segments = parse_path(head)?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            if let Some(params) = match_route(&route.segments, &request_segments) {
                let replace = match &best {
                    None => true,
                    Some((current, _)) => route.rank > current.rank,
                };
                if replace {
                    best = Some((route, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    let bytes = pattern.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let mut pieces = split_segments(bytes);
    let trailing_slash = pieces.last().is_some_and(|piece| piece.0.is_empty());
    if trailing_slash {
        pieces.pop();
    }

    let mut segments = Vec::with_capacity(pieces.len());
    let mut names = Vec::<String>::new();

    for (start, end) in pieces {
        let raw = &bytes[start..end];
        if raw.is_empty() {
            return Err(Error::InvalidPattern { at: start });
        }

        if raw[0] == b':' || raw[0] == b'*' {
            let name_bytes = &raw[1..];
            if !valid_name(name_bytes) {
                return Err(Error::InvalidPattern { at: start });
            }
            let name = String::from_utf8(name_bytes.to_vec()).unwrap();
            if names.iter().any(|existing| existing == &name) {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.clone());

            if raw[0] == b'*' {
                if trailing_slash || end != bytes.len() {
                    return Err(Error::InvalidPattern { at: start });
                }
                segments.push(Segment::Wildcard(name));
            } else {
                segments.push(Segment::Param(name));
            }
        } else {
            let decoded = decode_segment(raw, start)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }

    if trailing_slash {
        return Err(Error::InvalidPattern {
            at: bytes.len() - 1,
        });
    }

    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    let bytes = path.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let mut pieces = split_segments(bytes);
    let trailing_slash = pieces.last().is_some_and(|piece| piece.0.is_empty());
    if trailing_slash {
        pieces.pop();
    }

    let mut segments = Vec::with_capacity(pieces.len());
    for (start, end) in pieces {
        if start == end {
            return Err(Error::InvalidPath { at: start });
        }
        segments.push(
            decode_segment(&bytes[start..end], start)
                .map_err(|at| Error::InvalidPath { at })?,
        );
    }

    if trailing_slash {
        return Err(Error::InvalidPath {
            at: bytes.len() - 1,
        });
    }

    Ok(segments)
}

// Returns byte ranges between separators, excluding the initial slash.
fn split_segments(bytes: &[u8]) -> Vec<(usize, usize)> {
    let mut pieces = Vec::new();
    let mut start = 1;
    for index in 1..bytes.len() {
        if bytes[index] == b'/' {
            pieces.push((start, index));
            start = index + 1;
        }
    }
    pieces.push((start, bytes.len()));
    pieces
}

fn valid_name(name: &[u8]) -> bool {
    if name.is_empty() || !name[0].is_ascii_lowercase() {
        return false;
    }
    name[1..]
        .iter()
        .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'_')
}

fn decode_segment(raw: &[u8], base: usize) -> Result<String, usize> {
    let mut decoded = Vec::with_capacity(raw.len());
    let mut index = 0;

    while index < raw.len() {
        let byte = raw[index];
        if byte == b'%' {
            if index + 2 >= raw.len() {
                return Err(base + index);
            }
            let high = hex_upper(raw[index + 1]).ok_or(base + index)?;
            let low = hex_upper(raw[index + 2]).ok_or(base + index)?;
            let value = high * 16 + low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + index);
            }
            decoded.push(value);
            index += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            index += 1;
        } else {
            return Err(base + index);
        }
    }

    Ok(String::from_utf8(decoded).unwrap())
}

fn hex_upper(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn match_route(
    pattern: &[Segment],
    request: &[String],
) -> Option<Vec<(String, String)>> {
    let has_wildcard = matches!(pattern.last(), Some(Segment::Wildcard(_)));
    let fixed_len = pattern.len() - usize::from(has_wildcard);

    if (!has_wildcard && request.len() != pattern.len())
        || (has_wildcard && request.len() < fixed_len)
    {
        return None;
    }

    let mut params = Vec::new();
    for index in 0..fixed_len {
        match &pattern[index] {
            Segment::Static(expected) if expected != &request[index] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), request[index].clone())),
            Segment::Wildcard(_) => return None,
        }
    }

    if let Some(Segment::Wildcard(name)) = pattern.last() {
        params.push((name.clone(), request[fixed_len..].join("/")));
    }

    Some(params)
}
