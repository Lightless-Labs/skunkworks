#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut state = State::Bare;
    let mut quote_pos = 0;

    let mut iter = input.char_indices().peekable();

    while let Some((pos, ch)) = iter.next() {
        match state {
            State::Bare => match ch {
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                '\'' => {
                    active = true;
                    quote_pos = pos;
                    state = State::Single;
                }
                '"' => {
                    active = true;
                    quote_pos = pos;
                    state = State::Double;
                }
                '\\' => {
                    let Some((_, next)) = iter.next() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if next != '\n' {
                        word.push(next);
                        active = true;
                    }
                }
                _ => {
                    word.push(ch);
                    active = true;
                }
            },
            State::Single => {
                if ch == '\'' {
                    state = State::Bare;
                } else {
                    word.push(ch);
                }
            }
            State::Double => match ch {
                '"' => state = State::Bare,
                '\\' => {
                    let Some((_, next)) = iter.next() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    match next {
                        '"' | '\\' | '$' => word.push(next),
                        '\n' => {}
                        _ => {
                            word.push('\\');
                            word.push(next);
                        }
                    }
                }
                _ => word.push(ch),
            },
        }
    }

    match state {
        State::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        State::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        State::Bare => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum State {
    Bare,
    Single,
    Double,
}
