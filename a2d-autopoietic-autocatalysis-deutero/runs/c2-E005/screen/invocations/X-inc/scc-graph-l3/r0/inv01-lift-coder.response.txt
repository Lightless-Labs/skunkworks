pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }
    for (i, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: i,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adj = vec![Vec::<usize>::new(); n];
    let mut rev = vec![Vec::<usize>::new(); n];
    for &(from, to) in edges {
        adj[from].push(to);
        rev[to].push(from);
    }

    // Iterative Kosaraju first pass.
    let mut seen = vec![false; n];
    let mut finish = Vec::with_capacity(n);
    for start in 0..n {
        if seen[start] {
            continue;
        }
        seen[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((v, next)) = stack.last_mut() {
            if *next < adj[*v].len() {
                let w = adj[*v][*next];
                *next += 1;
                if !seen[w] {
                    seen[w] = true;
                    stack.push((w, 0));
                }
            } else {
                let (v, _) = stack.pop().unwrap();
                finish.push(v);
            }
        }
    }

    let mut raw_id = vec![usize::MAX; n];
    let mut raw_components = Vec::<Vec<usize>>::new();
    for &start in finish.iter().rev() {
        if raw_id[start] != usize::MAX {
            continue;
        }
        let id = raw_components.len();
        let mut members = Vec::new();
        let mut stack = vec![start];
        raw_id[start] = id;
        while let Some(v) = stack.pop() {
            members.push(v);
            for &w in &rev[v] {
                if raw_id[w] == usize::MAX {
                    raw_id[w] = id;
                    stack.push(w);
                }
            }
        }
        members.sort_unstable();
        raw_components.push(members);
    }

    let mut order: Vec<usize> = (0..raw_components.len()).collect();
    order.sort_by(|&a, &b| {
        raw_components[b]
            .len()
            .cmp(&raw_components[a].len())
            .then_with(|| raw_components[a][0].cmp(&raw_components[b][0]))
    });

    let mut remap = vec![0usize; order.len()];
    let mut components = Vec::with_capacity(order.len());
    for (new_id, &old_id) in order.iter().enumerate() {
        remap[old_id] = new_id;
        components.push(raw_components[old_id].clone());
    }
    let component_of: Vec<usize> = raw_id.iter().map(|&id| remap[id]).collect();
    let count = components.len();

    let mut cyclic: Vec<bool> = components.iter().map(|c| c.len() >= 2).collect();
    let mut internal_edges = vec![0usize; count];
    let mut arc_data = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (i, &(from, to)) in edges.iter().enumerate() {
        let a = component_of[from];
        let b = component_of[to];
        if a == b {
            internal_edges[a] += 1;
            if from == to {
                cyclic[a] = true;
            }
        } else {
            let entry = arc_data.entry((a, b)).or_insert((0, i));
            entry.0 += 1;
            entry.1 = entry.1.min(i);
        }
    }

    let mut arcs: Vec<Arc> = arc_data
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    let mut outgoing = vec![Vec::<usize>::new(); count];
    let mut indegree = vec![0usize; count];
    for arc in &arcs {
        outgoing[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }

    // Priority Kahn traversal. Heap entries are reversed to make BinaryHeap
    // prioritize the desired greatest internal edge count and smallest member.
    let mut ready = std::collections::BinaryHeap::<(usize, usize, usize)>::new();
    for c in 0..count {
        if indegree[c] == 0 {
            ready.push((internal_edges[c], components[c][0], c));
        }
    }

    let mut topo = Vec::with_capacity(count);
    let mut depth = vec![0usize; count];
    while let Some((_, _, c)) = ready.pop() {
        topo.push(c);
        for &to in &outgoing[c] {
            depth[to] = depth[to].max(depth[c] + 1);
            indegree[to] -= 1;
            if indegree[to] == 0 {
                ready.push((internal_edges[to], components[to][0], to));
            }
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];
        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        let mut outgoing = vec![Vec::<usize>::new(); self.components.len()];
        for arc in &self.arcs {
            outgoing[arc.from].push(arc.to);
        }
        for neighbors in &mut outgoing {
            neighbors.sort_unstable();
            neighbors.dedup();
        }

        fn shortest_lex_path(
            outgoing: &[Vec<usize>],
            start: usize,
            goal: usize,
        ) -> Option<Vec<usize>> {
            let n = outgoing.len();
            let mut distance = vec![usize::MAX; n];
            let mut queue = std::collections::VecDeque::new();
            distance[goal] = 0;
            queue.push_back(goal);

            // Walk the reversed graph implicitly to find distances to goal.
            while let Some(v) = queue.pop_front() {
                let d = distance[v];
                for u in 0..n {
                    if outgoing[u].binary_search(&v).is_ok() && distance[u] == usize::MAX {
                        distance[u] = d + 1;
                        queue.push_back(u);
                    }
                }
            }
            if distance[start] == usize::MAX {
                return None;
            }

            let mut route = vec![start];
            let mut current = start;
            while current != goal {
                let next = outgoing[current]
                    .iter()
                    .copied()
                    .find(|&v| distance[v] + 1 == distance[current])?;
                route.push(next);
                current = next;
            }
            Some(route)
        }

        if let Some(route) = shortest_lex_path(&outgoing, a, b) {
            return Ok(Query {
                kind: Relation::Forward,
                route,
            });
        }
        if let Some(mut route) = shortest_lex_path(&outgoing, b, a) {
            route.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route,
            });
        }
        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}
