#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&i| (jobs[i].arrival, i));

    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut ready: std::collections::VecDeque<usize> = std::collections::VecDeque::new();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut event_index = 0usize;
    let mut last_id: Option<u32> = None;
    let mut completed = 0usize;

    while completed < jobs.len() {
        while event_index < events.len()
            && u64::from(jobs[events[event_index]].arrival) <= time
        {
            let i = events[event_index];
            if jobs[i].ticks == 0 {
                completion[i] = Some(u64::from(jobs[i].arrival));
                completed += 1;
            } else {
                ready.push_back(i);
            }
            event_index += 1;
        }

        if completed == jobs.len() {
            break;
        }

        let Some(i) = ready.pop_front() else {
            if event_index < events.len() {
                let next_time = u64::from(jobs[events[event_index]].arrival);
                if next_time > time {
                    if let Some(last) = idle.last_mut() {
                        if last.1 == time {
                            last.1 = next_time;
                        } else {
                            idle.push((time, next_time));
                        }
                    } else {
                        idle.push((time, next_time));
                    }
                    time = next_time;
                    last_id = None;
                }
                continue;
            }
            break;
        };

        let id = jobs[i].id;
        if last_id.is_some_and(|previous| previous != id) {
            time += u64::from(switch_cost);
        }

        while event_index < events.len()
            && u64::from(jobs[events[event_index]].arrival) <= time
        {
            let arrived = events[event_index];
            if jobs[arrived].ticks == 0 {
                completion[arrived] = Some(u64::from(jobs[arrived].arrival));
                completed += 1;
            } else {
                ready.push_back(arrived);
            }
            event_index += 1;
        }

        let start = time;
        let consumed = remaining[i].min(quantum);
        time += u64::from(consumed);
        remaining[i] -= consumed;
        let finished = remaining[i] == 0;

        runs.push(Run {
            id,
            start,
            end: time,
            finished,
        });

        while event_index < events.len()
            && u64::from(jobs[events[event_index]].arrival) <= time
        {
            let arrived = events[event_index];
            if jobs[arrived].ticks == 0 {
                completion[arrived] = Some(u64::from(jobs[arrived].arrival));
                completed += 1;
            } else {
                ready.push_back(arrived);
            }
            event_index += 1;
        }

        if finished {
            completion[i] = Some(time);
            completed += 1;
        } else {
            ready.push_back(i);
        }
        last_id = Some(id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(i, job)| (job.id, completion[i].unwrap_or(time)))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
