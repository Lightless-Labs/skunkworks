#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut remaining: Vec<u64> = jobs.iter().map(|job| u64::from(job.ticks)).collect();
    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut ready = std::collections::VecDeque::new();
    let mut event_index = 0usize;
    let mut time = 0u64;
    let mut last_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();

    while completion.iter().any(Option::is_none) {
        while event_index < events.len()
            && u64::from(jobs[events[event_index]].arrival) <= time
        {
            let index = events[event_index];
            if jobs[index].ticks == 0 {
                completion[index] = Some(u64::from(jobs[index].arrival));
            } else {
                ready.push_back(index);
            }
            event_index += 1;
        }

        if ready.is_empty() {
            if event_index == events.len() {
                break;
            }
            let next_time = u64::from(jobs[events[event_index]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_id = None;
            }
            continue;
        }

        let index = ready.pop_front().expect("ready queue is nonempty");
        let job = &jobs[index];

        if last_id.is_some_and(|last| last != job.id) {
            time += u64::from(switch_cost);
        }

        while event_index < events.len()
            && u64::from(jobs[events[event_index]].arrival) <= time
        {
            let arrived = events[event_index];
            if jobs[arrived].ticks == 0 {
                completion[arrived] = Some(u64::from(jobs[arrived].arrival));
            } else {
                ready.push_back(arrived);
            }
            event_index += 1;
        }

        let consumed = remaining[index].min(u64::from(quantum));
        let start = time;
        time += consumed;
        remaining[index] -= consumed;

        runs.push(Run {
            id: job.id,
            start,
            end: time,
            finished: remaining[index] == 0,
        });

        while event_index < events.len()
            && u64::from(jobs[events[event_index]].arrival) <= time
        {
            let arrived = events[event_index];
            if jobs[arrived].ticks == 0 {
                completion[arrived] = Some(u64::from(jobs[arrived].arrival));
            } else {
                ready.push_back(arrived);
            }
            event_index += 1;
        }

        if remaining[index] == 0 {
            completion[index] = Some(time);
        } else {
            ready.push_back(index);
        }
        last_id = Some(job.id);
    }

    let end = completion
        .iter()
        .filter_map(|&value| value)
        .max()
        .unwrap_or(0);
    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].unwrap_or(end)))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end,
    })
}
