#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn push_scalar(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn push_string(word: &mut Word, s: &str, active_glob: bool) {
    for c in s.chars() {
        push_scalar(word, c, active_glob);
    }
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            }
        } else {
            result.push(c);
        }
    }

    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    push_scalar(&mut word, c, false);
                }
                i = next_i;
            }
            Quote::Double => {
                match c {
                    '"' => {
                        quote = Quote::None;
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let escaped = input[next_i..].chars().next().unwrap();
                        let escaped_end = next_i + escaped.len_utf8();
                        if matches!(escaped, '"' | '\\' | '$') {
                            push_scalar(&mut word, escaped, false);
                            active = true;
                            i = escaped_end;
                        } else if escaped == '\n' {
                            i = escaped_end;
                        } else {
                            push_scalar(&mut word, '\\', false);
                            active = true;
                            i = next_i;
                        }
                    }
                    '$' => {
                        if input[next_i..].starts_with('(') {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                        let (consumed, value) = match_expansion(input, i, env)?;
                        if let Some((end, value)) = value {
                            push_string(&mut word, &value, false);
                            active = true;
                            i = end;
                        } else {
                            push_scalar(&mut word, '$', false);
                            active = true;
                            i = next_i;
                        }
                        let _ = consumed;
                    }
                    '`' => return Err(SplitError::CommandSubstitution { pos: i }),
                    _ => {
                        push_scalar(&mut word, c, false);
                        active = true;
                        i = next_i;
                    }
                }
            }
            Quote::None => {
                match c {
                    ' ' | '\t' | '\r' | '\n' => {
                        if active {
                            words.push(word);
                            word = Word {
                                text: String::new(),
                                glob: Vec::new(),
                            };
                            active = false;
                        }
                        i = next_i;
                    }
                    '\'' => {
                        quote = Quote::Single;
                        quote_pos = i;
                        active = true;
                        i = next_i;
                    }
                    '"' => {
                        quote = Quote::Double;
                        quote_pos = i;
                        active = true;
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let escaped = input[next_i..].chars().next().unwrap();
                        let escaped_end = next_i + escaped.len_utf8();
                        if escaped == '\n' {
                            i = escaped_end;
                        } else {
                            push_scalar(&mut word, escaped, false);
                            active = true;
                            i = escaped_end;
                        }
                    }
                    '$' => {
                        if input[next_i..].starts_with('(') {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                        let (_, value) = match_expansion(input, i, env)?;
                        if let Some((end, value)) = value {
                            push_string(&mut word, &value, true);
                            active = true;
                            i = end;
                        } else {
                            push_scalar(&mut word, '$', false);
                            active = true;
                            i = next_i;
                        }
                    }
                    '`' => return Err(SplitError::CommandSubstitution { pos: i }),
                    _ => {
                        push_scalar(&mut word, c, true);
                        active = true;
                        i = next_i;
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

// Returns (consumed length, optional (end offset, expanded value)).
fn match_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, Option<(usize, String)>), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((1, None));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut cursor = content_start;
        let mut close = None;

        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            let end = cursor + c.len_utf8();
            if c == '\\' {
                if end < input.len() {
                    let escaped = input[end..].chars().next().unwrap();
                    cursor = end + escaped.len_utf8();
                } else {
                    cursor = end;
                }
            } else if c == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor = end;
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let name_end = contents
            .char_indices()
            .find_map(|(idx, c)| if is_name_char(c) { None } else { Some(idx) })
            .unwrap_or(contents.len());

        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &contents[..name_end];
        if !name.chars().next().is_some_and(is_name_start) {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let default = if name_end == contents.len() {
            None
        } else if contents[name_end..].starts_with(":-") {
            Some(&contents[name_end + 2..])
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        let found = env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value);
        let value = match (default, found) {
            (Some(text), Some(value)) if !value.is_empty() => value.to_string(),
            (Some(text), _) => decode_default(text),
            (None, Some(value)) => value.to_string(),
            (None, None) => String::new(),
        };

        let end = close + 1;
        Ok((end - dollar, Some((end, value))))
    } else if is_name_start(first) {
        let mut end = after_dollar + first.len_utf8();
        while end < input.len() {
            let c = input[end..].chars().next().unwrap();
            if !is_name_char(c) {
                break;
            }
            end += c.len_utf8();
        }
        let name = &input[after_dollar..end];
        let value = env
            .iter()
            .find(|(key, _)| *key == name)
            .map(|(_, value)| (*value).to_string())
            .unwrap_or_default();
        Ok((end - dollar, Some((end, value))))
    } else {
        Ok((1, None))
    }
}
