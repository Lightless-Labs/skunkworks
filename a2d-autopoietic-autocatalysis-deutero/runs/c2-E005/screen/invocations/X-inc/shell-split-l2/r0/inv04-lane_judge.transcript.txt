fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()

fn scalar_at(input: &str, pos: usize) -> Option<(char, usize)> {
    let c = input.get(pos..)?.chars().next()?;
    Some((c, pos + c.len_utf8()))
}

fn expand(

    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {

        return Ok(("$".to_string(), after_dollar));
    };

    if next != '{' {
        if !is_name_start(next) {
            return Ok(("$".to_string(), after_dollar));
        }

            if !is_name_continue(c) {
                break;
            }
            end = next_end;
        }

        return Ok((lookup(name, env).unwrap_or("").to_string(), end));

    let mut close = None;

    while let Some((c, next_end)) = scalar_at(input, cursor) {

            if let Some((_, escaped_end)) = scalar_at(input, next_end) {
                cursor = escaped_end;
            } else {
                cursor = next_end;
            }
            continue;
        }
        if c == '}' {
            close = Some(cursor);
            break;
        }
        cursor = next_end;
    }

        return Err(SplitError::UnterminatedExpansion { pos: dollar });
    };

        .char_indices()

        .map(|(i, c)| i + c.len_utf8())
        .last()

        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let suffix = &contents[name_end..];

        }

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut iter = text.char_indices().peekable();

    while let Some((_, ch)) = iter.next() {
        if ch == '\\' {
            if let Some((_, next)) = iter.next() {
                out.push(next);
            }
        } else {
            out.push(ch);
        }
    }

    out
}

    dollar: usize,

) -> Result<(String, usize), SplitError> {

                let ch = input[end..].chars().next().unwrap();
                if !is_name_continue(ch) {
                    break;
                }
                end += ch.len_utf8();
            }
            let name = &input[after_dollar..end];
            return Ok((lookup(name, env).unwrap_or("").to_string(), end));
        }
    }

    let contents_start = after_dollar + 1;

        let ch = input[cursor..].chars().next().unwrap();
        if ch == '\\' {
            cursor += ch.len_utf8();
            if cursor < input.len() {
                let next = input[cursor..].chars().next().unwrap();
                cursor += next.len_utf8();
            }

            close = Some(cursor);
            break;
        } else {
            cursor += ch.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[contents_start..close];

    let mut name_end = 0;
    let mut chars = contents.char_indices();
    if let Some((_, first)) = chars.next() {
        if !is_name_start(first) {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        name_end = first.len_utf8();
        for (offset, ch) in chars {
            if !is_name_continue(ch) {
                break;
            }
            name_end = offset + ch.len_utf8();
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_end];
    let suffix = &contents[name_end..];

fn expand(input: &str, dollar: usize, env: &[(&str, &str)]) -> Result<(String, usize), SplitError> {
    let after_dollar = dollar + 1;
    let rest = &input[after_dollar..];
    let mut chars = rest.char_indices();

    match chars.next() {
        Some((_, '{')) => {
            let content_start = after_dollar + 1;
            let mut cursor = content_start;
            let mut closing = None;

            while cursor < input.len() {
                let ch = input[cursor..].chars().next().unwrap();
                let next = cursor + ch.len_utf8();

                if ch == '\\' {
                    if next < input.len() {
                        let escaped = input[next..].chars().next().unwrap();
                        cursor = next + escaped.len_utf8();
                    } else {
                        cursor = next;
                    }
                } else if ch == '}' {
                    closing = Some(cursor);
                    break;
                } else {
                    cursor = next;
                }
            }

            let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
            let contents = &input[content_start..close];
            let mut split_at = None;
            let mut scan = 0;

            while scan < contents.len() {
                let ch = contents[scan..].chars().next().unwrap();
                if ch == ':' && contents[scan..].starts_with(":-") {
                    split_at = Some(scan);
                    break;
                }
                scan += ch.len_utf8();
            }

            let (name, default) = if let Some(index) = split_at {
                (&contents[..index], Some(&contents[index + 2..]))
            } else {
                (contents, None)
            };

            let mut name_chars = name.chars();
            let valid_name = match name_chars.next() {
                Some(first) if is_name_start(first) => {
                    name_chars.all(is_name_continue)
                }
                _ => false,
            };

            if !valid_name {
                return Err(SplitError::InvalidExpansion { pos: dollar });
            }

            let value = match default {
                Some(default_text) => match find_env(env, name) {
                    Some(value) if !value.is_empty() => value.to_string(),
                    _ => decode_default(default_text),
                },
                None => find_env(env, name).unwrap_or("").to_string(),
            };

            Ok((value, close + 1))
        }
        Some((_, ch)) if is_name_start(ch) => {
            let mut end = after_dollar + ch.len_utf8();
            while end < input.len() {
                let next = input[end..].chars().next().unwrap();
                if is_name_continue(next) {
                    end += next.len_utf8();
                } else {
                    break;
                }
            }
            let name = &input[after_dollar..end];
            Ok((find_env(env, name).unwrap_or("").to_string(), end))
        }
        _ => Ok(("$".to_string(), after_dollar)),
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        let next_cursor = cursor + ch.len_utf8();

        match quote {
            Quote::Single { .. } => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
                cursor = next_cursor;
            }
            Quote::Double { opening } => {
                if ch == '"' {
                    quote = Quote::None;
                    cursor = next_cursor;
                } else if ch == '\\' {
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next_cursor..].chars().next().unwrap();
                    let after_escaped = next_cursor + escaped.len_utf8();
                    if escaped == '"' || escaped == '\\' || escaped == '$' {
                        word.push(escaped);
                        cursor = after_escaped;
                    } else if escaped == '\n' {
                        cursor = after_escaped;
                    } else {
                        word.push('\\');
                        cursor = next_cursor;
                    }
                } else if ch == '$' {
                    let (value, after) = expand(input, cursor, env)?;
                    if after == next_cursor {
                        word.push('$');
                    } else {
                        word.push_str(&value);
                        active = true;
                    }
                    cursor = after;
                } else {
                    word.push(ch);
                    cursor = next_cursor;
                }
                let _ = opening;
            }
            Quote::None => {
                if ch == '\'' {
                    quote = Quote::Single { opening: cursor };
                    active = true;
                    cursor = next_cursor;
                } else if ch == '"' {
                    quote = Quote::Double { opening: cursor };
                    active = true;
                    cursor = next_cursor;
                } else if ch == '\\' {
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next_cursor..].chars().next().unwrap();
                    let after_escaped = next_cursor + escaped.len_utf8();
                    if escaped != '\n' {
                        word.push(escaped);
                        active = true;
                    }
                    cursor = after_escaped;
                } else if ch == ' ' || ch == '\t' || ch == '\r' || ch == '\n' {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    cursor = next_cursor;
                } else if ch == '$' {
                    let (value, after) = expand(input, cursor, env)?;
                    if after == next_cursor {
                        word.push('$');
                        active = true;
                    } else {
                        word.push_str(&value);
                        active = true;
                    }
                    cursor = after;
                } else {
                    word.push(ch);
                    active = true;
                    cursor = next_cursor;
                }
            }
        }
    }

    match quote {
        Quote::Single { opening } => {
            return Err(SplitError::UnterminatedSingle { pos: opening });
        }
        Quote::Double { opening } => {
            let _ = opening;
            return Err(SplitError::UnterminatedDouble {
                pos: match quote {
                    Quote::Double { opening } => opening,
                    _ => unreachable!(),
                },
            });
        }
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}

pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },

    UnterminatedExpansion { pos: usize },

enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double(open_pos) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if n == '"' || n == '\\' || n == '$' {
                        word.push(n);
                        active = true;
                        i = after_n;
                    } else if n == '\n' {
                        i = after_n;
                    } else {
                        word.push('\\');
                        active = true;
                        i = next_i;
                    }
                } else if c == '$' {
                    let (consumed, result) = expand(input, i, env)?;
                    if consumed == 0 {
                        word.push('$');
                        active = true;
                        i = next_i;
                    } else {
                        word.push_str(&result);
                        active = true;
                        i += consumed;
                    }
                } else {
                    word.push(c);
                    active = true;
                    i = next_i;
                }
                let _ = open_pos;
            }
            Quote::None => {
                if c == ' ' || c == '\t' || c == '\r' || c == '\n' {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if n == '\n' {
                        i = after_n;
                    } else {
                        word.push(n);
                        active = true;
                        i = after_n;
                    }
                } else if c == '$' {
                    let (consumed, result) = expand(input, i, env)?;
                    if consumed == 0 {
                        word.push('$');
                        active = true;
                        i = next_i;
                    } else {
                        word.push_str(&result);
                        active = true;
                        i += consumed;
                    }
                } else {
                    word.push(c);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand(input: &str, dollar: usize, env: &[(&str, &str)]) -> Result<(usize, String), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((0, String::new()));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut j = content_start;
        let mut closing = None;
        while j < input.len() {
            let c = input[j..].chars().next().unwrap();
            let next = j + c.len_utf8();
            if c == '\\' {
                if next < input.len() {
                    let escaped = input[next..].chars().next().unwrap();
                    j = next + escaped.len_utf8();
                } else {
                    j = next;
                }
            } else if c == '}' {
                closing = Some(j);
                break;
            } else {
                j = next;
            }
        }

        let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let content = &input[content_start..close];

        let mut name_end = 0;
        for (offset, c) in content.char_indices() {
            if offset == 0 {
                if !is_name_start(c) {
                    return Err(SplitError::InvalidExpansion { pos: dollar });
                }
            } else if !is_name_continue(c) {
                name_end = offset;
                break;
            }
            name_end = offset + c.len_utf8();
        }

        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &content[..name_end];
        let rest = &content[name_end..];
        let value = if rest.is_empty() {
            env_value(env, name).unwrap_or("").to_string()
        } else if let Some(default) = rest.strip_prefix(":-") {
            match env_value(env, name) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        Ok((close + 1 - dollar, value))
    } else if is_name_start(first) {
        let mut end = after_dollar;
        for c in input[after_dollar..].chars() {
            if !is_name_continue(c) {
                break;
            }
            end += c.len_utf8();
        }
        let name = &input[after_dollar..end];
        Ok((end - dollar, env_value(env, name).unwrap_or("").to_string()))
    } else {
        Ok((0, String::new()))
    }
}
