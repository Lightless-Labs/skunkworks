#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single { opening: usize },
    Double { opening: usize },
}

fn is_name_start(ch: char) -> bool {
    ch.is_ascii_alphabetic() || ch == '_'
}

fn is_name_continue(ch: char) -> bool {
    ch.is_ascii_alphanumeric() || ch == '_'
}

fn find_env<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, ch)) = chars.next() {
        if ch == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(ch);
        }
    }

    result
}

fn expand(input: &str, dollar: usize, env: &[(&str, &str)]) -> Result<(String, usize), SplitError> {
    let after_dollar = dollar + 1;
    let rest = &input[after_dollar..];
    let mut chars = rest.char_indices();

    match chars.next() {
        Some((_, '{')) => {
            let content_start = after_dollar + 1;
            let mut cursor = content_start;
            let mut closing = None;

            while cursor < input.len() {
                let ch = input[cursor..].chars().next().unwrap();
                let next = cursor + ch.len_utf8();

                if ch == '\\' {
                    if next < input.len() {
                        let escaped = input[next..].chars().next().unwrap();
                        cursor = next + escaped.len_utf8();
                    } else {
                        cursor = next;
                    }
                } else if ch == '}' {
                    closing = Some(cursor);
                    break;
                } else {
                    cursor = next;
                }
            }

            let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
            let contents = &input[content_start..close];
            let mut split_at = None;
            let mut scan = 0;

            while scan < contents.len() {
                let ch = contents[scan..].chars().next().unwrap();
                if ch == ':' && contents[scan..].starts_with(":-") {
                    split_at = Some(scan);
                    break;
                }
                scan += ch.len_utf8();
            }

            let (name, default) = if let Some(index) = split_at {
                (&contents[..index], Some(&contents[index + 2..]))
            } else {
                (contents, None)
            };

            let mut name_chars = name.chars();
            let valid_name = match name_chars.next() {
                Some(first) if is_name_start(first) => {
                    name_chars.all(is_name_continue)
                }
                _ => false,
            };

            if !valid_name {
                return Err(SplitError::InvalidExpansion { pos: dollar });
            }

            let value = match default {
                Some(default_text) => match find_env(env, name) {
                    Some(value) if !value.is_empty() => value.to_string(),
                    _ => decode_default(default_text),
                },
                None => find_env(env, name).unwrap_or("").to_string(),
            };

            Ok((value, close + 1))
        }
        Some((_, ch)) if is_name_start(ch) => {
            let mut end = after_dollar + ch.len_utf8();
            while end < input.len() {
                let next = input[end..].chars().next().unwrap();
                if is_name_continue(next) {
                    end += next.len_utf8();
                } else {
                    break;
                }
            }
            let name = &input[after_dollar..end];
            Ok((find_env(env, name).unwrap_or("").to_string(), end))
        }
        _ => Ok(("$".to_string(), after_dollar)),
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        let next_cursor = cursor + ch.len_utf8();

        match quote {
            Quote::Single { .. } => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
                cursor = next_cursor;
            }
            Quote::Double { opening } => {
                if ch == '"' {
                    quote = Quote::None;
                    cursor = next_cursor;
                } else if ch == '\\' {
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next_cursor..].chars().next().unwrap();
                    let after_escaped = next_cursor + escaped.len_utf8();
                    if escaped == '"' || escaped == '\\' || escaped == '$' {
                        word.push(escaped);
                        cursor = after_escaped;
                    } else if escaped == '\n' {
                        cursor = after_escaped;
                    } else {
                        word.push('\\');
                        cursor = next_cursor;
                    }
                } else if ch == '$' {
                    let (value, after) = expand(input, cursor, env)?;
                    if after == next_cursor {
                        word.push('$');
                    } else {
                        word.push_str(&value);
                        active = true;
                    }
                    cursor = after;
                } else {
                    word.push(ch);
                    cursor = next_cursor;
                }
                let _ = opening;
            }
            Quote::None => {
                if ch == '\'' {
                    quote = Quote::Single { opening: cursor };
                    active = true;
                    cursor = next_cursor;
                } else if ch == '"' {
                    quote = Quote::Double { opening: cursor };
                    active = true;
                    cursor = next_cursor;
                } else if ch == '\\' {
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next_cursor..].chars().next().unwrap();
                    let after_escaped = next_cursor + escaped.len_utf8();
                    if escaped != '\n' {
                        word.push(escaped);
                        active = true;
                    }
                    cursor = after_escaped;
                } else if ch == ' ' || ch == '\t' || ch == '\r' || ch == '\n' {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    cursor = next_cursor;
                } else if ch == '$' {
                    let (value, after) = expand(input, cursor, env)?;
                    if after == next_cursor {
                        word.push('$');
                        active = true;
                    } else {
                        word.push_str(&value);
                        active = true;
                    }
                    cursor = after;
                } else {
                    word.push(ch);
                    active = true;
                    cursor = next_cursor;
                }
            }
        }
    }

    match quote {
        Quote::Single { opening } => {
            return Err(SplitError::UnterminatedSingle { pos: opening });
        }
        Quote::Double { opening } => {
            let _ = opening;
            return Err(SplitError::UnterminatedDouble {
                pos: match quote {
                    Quote::Double { opening } => opening,
                    _ => unreachable!(),
                },
            });
        }
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
