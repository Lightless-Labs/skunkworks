#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(ch: char) -> bool {
    ch.is_ascii_alphabetic() || ch == '_'
}

fn is_name_continue(ch: char) -> bool {
    is_name_start(ch) || ch.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut iter = text.char_indices().peekable();

    while let Some((_, ch)) = iter.next() {
        if ch == '\\' {
            if let Some((_, next)) = iter.next() {
                out.push(next);
            }
        } else {
            out.push(ch);
        }
    }

    out
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    after_dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let rest = &input[after_dollar..];

    if let Some(first) = rest.chars().next() {
        if is_name_start(first) {
            let mut end = after_dollar + first.len_utf8();
            while end < input.len() {
                let ch = input[end..].chars().next().unwrap();
                if !is_name_continue(ch) {
                    break;
                }
                end += ch.len_utf8();
            }
            let name = &input[after_dollar..end];
            return Ok((lookup(name, env).unwrap_or("").to_string(), end));
        }
    }

    // The caller invokes this helper only for a dollar followed by '{'.
    let contents_start = after_dollar + 1;
    let mut cursor = contents_start;
    let mut close = None;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        if ch == '\\' {
            cursor += ch.len_utf8();
            if cursor < input.len() {
                let next = input[cursor..].chars().next().unwrap();
                cursor += next.len_utf8();
            }
        } else if ch == '}' {
            close = Some(cursor);
            break;
        } else {
            cursor += ch.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[contents_start..close];

    let mut name_end = 0;
    let mut chars = contents.char_indices();
    if let Some((_, first)) = chars.next() {
        if !is_name_start(first) {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        name_end = first.len_utf8();
        for (offset, ch) in chars {
            if !is_name_continue(ch) {
                break;
            }
            name_end = offset + ch.len_utf8();
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_end];
    let suffix = &contents[name_end..];

    let value = if suffix.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default) = suffix.strip_prefix(":-") {
        match lookup(name, env) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        let next = cursor + ch.len_utf8();

        match quote {
            Quote::Single(_) => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
                cursor = next;
            }
            Quote::Double(opening) => {
                if ch == '"' {
                    quote = Quote::None;
                    cursor = next;
                } else if ch == '\\' {
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next..].chars().next().unwrap();
                    let escaped_end = next + escaped.len_utf8();
                    if matches!(escaped, '"' | '\\' | '$' | '\n') {
                        if escaped != '\n' {
                            word.push(escaped);
                        }
                        cursor = escaped_end;
                    } else {
                        word.push('\\');
                        cursor = next;
                    }
                } else if ch == '$' {
                    if next < input.len() {
                        let following = input[next..].chars().next().unwrap();
                        if following == '{' {
                            let (value, end) = parse_expansion(input, cursor, next, env)?;
                            word.push_str(&value);
                            active = true;
                            cursor = end;
                        } else if is_name_start(following) {
                            let (value, end) = parse_expansion(input, cursor, next, env)?;
                            word.push_str(&value);
                            active = true;
                            cursor = end;
                        } else {
                            word.push('$');
                            cursor = next;
                        }
                    } else {
                        word.push('$');
                        cursor = next;
                    }
                } else {
                    word.push(ch);
                    cursor = next;
                }

                // Keep the opening position available for the EOF error.
                let _ = opening;
            }
            Quote::None => {
                if is_separator(ch) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    cursor = next;
                } else if ch == '\'' {
                    active = true;
                    quote = Quote::Single(cursor);
                    cursor = next;
                } else if ch == '"' {
                    active = true;
                    quote = Quote::Double(cursor);
                    cursor = next;
                } else if ch == '\\' {
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next..].chars().next().unwrap();
                    let escaped_end = next + escaped.len_utf8();
                    if escaped == '\n' {
                        cursor = escaped_end;
                    } else {
                        word.push(escaped);
                        active = true;
                        cursor = escaped_end;
                    }
                } else if ch == '$' {
                    if next < input.len() {
                        let following = input[next..].chars().next().unwrap();
                        if following == '{' {
                            let (value, end) = parse_expansion(input, cursor, next, env)?;
                            word.push_str(&value);
                            active = true;
                            cursor = end;
                        } else if is_name_start(following) {
                            let (value, end) = parse_expansion(input, cursor, next, env)?;
                            word.push_str(&value);
                            active = true;
                            cursor = end;
                        } else {
                            word.push('$');
                            active = true;
                            cursor = next;
                        }
                    } else {
                        word.push('$');
                        active = true;
                        cursor = next;
                    }
                } else {
                    word.push(ch);
                    active = true;
                    cursor = next;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
