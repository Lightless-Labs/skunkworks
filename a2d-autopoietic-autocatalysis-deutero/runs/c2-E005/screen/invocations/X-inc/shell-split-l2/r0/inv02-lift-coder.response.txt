#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double(open_pos) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if n == '"' || n == '\\' || n == '$' {
                        word.push(n);
                        active = true;
                        i = after_n;
                    } else if n == '\n' {
                        i = after_n;
                    } else {
                        word.push('\\');
                        active = true;
                        i = next_i;
                    }
                } else if c == '$' {
                    let (consumed, result) = expand(input, i, env)?;
                    if consumed == 0 {
                        word.push('$');
                        active = true;
                        i = next_i;
                    } else {
                        word.push_str(&result);
                        active = true;
                        i += consumed;
                    }
                } else {
                    word.push(c);
                    active = true;
                    i = next_i;
                }
                let _ = open_pos;
            }
            Quote::None => {
                if c == ' ' || c == '\t' || c == '\r' || c == '\n' {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if n == '\n' {
                        i = after_n;
                    } else {
                        word.push(n);
                        active = true;
                        i = after_n;
                    }
                } else if c == '$' {
                    let (consumed, result) = expand(input, i, env)?;
                    if consumed == 0 {
                        word.push('$');
                        active = true;
                        i = next_i;
                    } else {
                        word.push_str(&result);
                        active = true;
                        i += consumed;
                    }
                } else {
                    word.push(c);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand(input: &str, dollar: usize, env: &[(&str, &str)]) -> Result<(usize, String), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((0, String::new()));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut j = content_start;
        let mut closing = None;
        while j < input.len() {
            let c = input[j..].chars().next().unwrap();
            let next = j + c.len_utf8();
            if c == '\\' {
                if next < input.len() {
                    let escaped = input[next..].chars().next().unwrap();
                    j = next + escaped.len_utf8();
                } else {
                    j = next;
                }
            } else if c == '}' {
                closing = Some(j);
                break;
            } else {
                j = next;
            }
        }

        let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let content = &input[content_start..close];

        let mut name_end = 0;
        for (offset, c) in content.char_indices() {
            if offset == 0 {
                if !is_name_start(c) {
                    return Err(SplitError::InvalidExpansion { pos: dollar });
                }
            } else if !is_name_continue(c) {
                name_end = offset;
                break;
            }
            name_end = offset + c.len_utf8();
        }

        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &content[..name_end];
        let rest = &content[name_end..];
        let value = if rest.is_empty() {
            env_value(env, name).unwrap_or("").to_string()
        } else if let Some(default) = rest.strip_prefix(":-") {
            match env_value(env, name) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        Ok((close + 1 - dollar, value))
    } else if is_name_start(first) {
        let mut end = after_dollar;
        for c in input[after_dollar..].chars() {
            if !is_name_continue(c) {
                break;
            }
            end += c.len_utf8();
        }
        let name = &input[after_dollar..end];
        Ok((end - dollar, env_value(env, name).unwrap_or("").to_string()))
    } else {
        Ok((0, String::new()))
    }
}
