#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct CfwsRun {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn parse_email_inner(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut clean = Vec::<u8>::with_capacity(bytes.len());
    let mut positions = Vec::<usize>::with_capacity(bytes.len());
    let mut runs = Vec::<CfwsRun>::new();

    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_comment = 0usize;
    let mut run_start: Option<usize> = None;
    let mut quote_protected = false;
    let mut quote_open = 0usize;
    let mut quote_escape = false;

    while i < bytes.len() {
        let b = bytes[i];

        if quote_protected {
            clean.push(b);
            positions.push(i);
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                quote_protected = false;
            }
            i += 1;
            continue;
        }

        if depth > 0 {
            if b == b'\\' {
                if run_start.is_none() {
                    run_start = Some(i);
                }
                let Some(&next) = bytes.get(i + 1) else {
                    return Err(EmailError::CommentEscape { at: i });
                };
                if !(0x20..=0x7e).contains(&next) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if run_start.is_none() {
                    run_start = Some(i);
                }
                let Some(end) = fold_end(bytes, i) else {
                    return Err(EmailError::BadFold { at: i });
                };
                i = end;
                continue;
            }
            if is_hws(b) {
                if run_start.is_none() {
                    run_start = Some(i);
                }
                i += 1;
                continue;
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                if depth == 0 {
                    if let Some(start) = run_start.take() {
                        runs.push(CfwsRun { start, end: i });
                    }
                }
                continue;
            }
            if !(0x20..=0x7e).contains(&b) {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        if b == b'(' {
            if run_start.is_none() {
                run_start = Some(i);
            }
            depth = 1;
            outer_comment = i;
            i += 1;
            continue;
        }
        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if b == b'\r' || b == b'\n' {
            if run_start.is_none() {
                run_start = Some(i);
            }
            let Some(end) = fold_end(bytes, i) else {
                return Err(EmailError::BadFold { at: i });
            };
            i = end;
            continue;
        }
        if is_hws(b) {
            if run_start.is_none() {
                run_start = Some(i);
            }
            i += 1;
            continue;
        }

        if let Some(start) = run_start.take() {
            runs.push(CfwsRun { start, end: i });
        }

        if clean.is_empty() && b == b'"' {
            quote_protected = true;
            quote_open = i;
        }
        clean.push(b);
        positions.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }
    if let Some(start) = run_start {
        runs.push(CfwsRun {
            start,
            end: bytes.len(),
        });
    }

    for run in &runs {
        let first_retained = positions.first().copied().unwrap_or(bytes.len());
        let last_retained = positions.last().copied().unwrap_or(bytes.len());
        let at_before = clean
            .iter()
            .zip(positions.iter())
            .any(|(&c, &p)| c == b'@' && p + 1 == run.start);
        let at_after = clean
            .iter()
            .zip(positions.iter())
            .any(|(&c, &p)| c == b'@' && p == run.end);
        if run.end <= first_retained || run.start > last_retained || at_before || at_after {
            continue;
        }
        return Err(EmailError::CfwsBoundary { at: run.start });
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut protected_end = 0usize;
    if clean.first() == Some(&b'"') {
        let mut j = 1usize;
        let mut closed = false;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                protected_end = j + 1;
                closed = true;
                break;
            } else {
                j += 1;
            }
        }
        if !closed {
            return Err(EmailError::UnclosedQuote {
                at: positions.first().copied().unwrap_or(quote_open),
            });
        }
    }

    let mut separators = Vec::new();
    for idx in protected_end..clean.len() {
        if clean[idx] == b'@' {
            separators.push(idx);
        }
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: positions[separators[1]],
        });
    }
    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: positions[sep],
        });
    }

    let local_bytes = &clean[..sep];
    let local_positions = &positions[..sep];
    let quoted = local_bytes.first() == Some(&b'"');
    let local = if !quoted {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_positions[64],
            });
        }
        for idx in 0..local_bytes.len() {
            let b = local_bytes[idx];
            if b == b'.' {
                if idx == 0
                    || idx + 1 == local_bytes.len()
                    || local_bytes[idx - 1] == b'.'
                {
                    return Err(EmailError::LocalDot {
                        at: local_positions[idx],
                    });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar {
                    at: local_positions[idx],
                });
            }
        }
        String::from_utf8(local_bytes.to_vec()).unwrap()
    } else {
        let mut closing = None;
        let mut j = 1usize;
        while j < local_bytes.len() {
            if local_bytes[j] == b'\\' {
                j += 2;
            } else if local_bytes[j] == b'"' {
                closing = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        let Some(close) = closing else {
            return Err(EmailError::UnclosedQuote {
                at: local_positions[0],
            });
        };
        if close + 1 != local_bytes.len() {
            return Err(EmailError::TrailingLocal {
                at: local_positions[close + 1],
            });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1usize;
        while j < close {
            let b = local_bytes[j];
            if b == b'\\' {
                let Some(&next) = local_bytes.get(j + 1) else {
                    return Err(EmailError::BadEscape {
                        at: local_positions[j],
                    });
                };
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape {
                        at: local_positions[j],
                    });
                }
                decoded.push(next);
                origins.push(local_positions[j]);
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: local_positions[j],
                    });
                }
                decoded.push(b);
                origins.push(local_positions[j]);
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: local_positions[close],
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        String::from_utf8(decoded).unwrap()
    };

    let domain_start = sep + 1;
    let mut domain_end = clean.len();
    let root_dot = if domain_end > domain_start && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(positions[domain_end])
    } else {
        None
    };
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: positions.get(domain_start).copied().unwrap_or(bytes.len()),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start + 253],
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for idx in domain_start..=domain_end {
        if idx == domain_end || clean[idx] == b'.' {
            if label_start == idx {
                let at = if idx == domain_end {
                    root_dot.unwrap_or_else(|| {
                        positions.get(idx).copied().unwrap_or(bytes.len())
                    })
                } else {
                    positions[idx]
                };
                return Err(EmailError::EmptyLabel { at });
            }
            labels.push((label_start, idx));
            label_start = idx + 1;
        }
    }

    for &(start, end) in &labels {
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: positions[start + 63],
            });
        }
        if clean[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[start],
            });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[end - 1],
            });
        }
        for idx in start..end {
            let b = clean[idx];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: positions[idx],
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: positions[domain_start],
        });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: positions[last_start],
        });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_email_inner(input)
}
