#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct CfwsRun {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn is_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(bytes: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<CfwsRun>), EmailError> {
    let mut clean = Vec::with_capacity(bytes.len());
    let mut positions = Vec::with_capacity(bytes.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut in_quote = false;
    let mut quote_escape = false;
    let mut depth = 0usize;
    let mut outer_comment = 0usize;
    let mut run_start: Option<usize> = None;

    while i < bytes.len() {
        let b = bytes[i];

        if in_quote {
            clean.push(b);
            positions.push(i);
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                in_quote = false;
            }
            i += 1;
            continue;
        }

        if depth > 0 {
            if let Some(start) = run_start {
                runs.push(CfwsRun { start, end: i });
                run_start = None;
            }

            if b == b'\\' {
                let Some(&next) = bytes.get(i + 1) else {
                    return Err(EmailError::CommentEscape { at: i });
                };
                if !is_printable(next) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }

            if b == b'\r' || b == b'\n' {
                if b == b'\r' {
                    if let Some(end) = is_fold(bytes, i) {
                        i = end;
                        continue;
                    }
                }
                return Err(EmailError::BadFold { at: i });
            }

            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                if depth == 0 {
                    if let Some(start) = run_start {
                        runs.push(CfwsRun { start, end: i });
                        run_start = None;
                    }
                }
                continue;
            }
            if is_printable(b) || b == b'\t' {
                i += 1;
                continue;
            }
            return Err(EmailError::CommentChar { at: i });
        }

        if b == b' ' || b == b'\t' {
            run_start.get_or_insert(i);
            i += 1;
            continue;
        }
        if b == b'\r' || b == b'\n' {
            if b == b'\r' {
                if let Some(end) = is_fold(bytes, i) {
                    run_start.get_or_insert(i);
                    i = end;
                    continue;
                }
            }
            return Err(EmailError::BadFold { at: i });
        }
        if b == b'(' {
            run_start.get_or_insert(i);
            depth = 1;
            outer_comment = i;
            i += 1;
            continue;
        }
        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if let Some(start) = run_start {
            runs.push(CfwsRun { start, end: i });
            run_start = None;
        }

        let first_retained = clean.is_empty();
        clean.push(b);
        positions.push(i);
        if first_retained && b == b'"' {
            in_quote = true;
        }
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }
    if let Some(start) = run_start {
        runs.push(CfwsRun {
            start,
            end: bytes.len(),
        });
    }

    Ok((clean, positions, runs))
}

fn mapped(positions: &[usize], clean_len: usize, input_len: usize, at: usize) -> usize {
    if at < clean_len {
        positions[at]
    } else {
        input_len
    }
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, positions, runs) = cleanup(bytes)?;
    let mut ats = Vec::new();
    let mut quote_end = None;

    if clean.first() == Some(&b'"') {
        let mut i = 1;
        let mut escaped = false;
        while i < clean.len() {
            let b = clean[i];
            if escaped {
                escaped = false;
            } else if b == b'\\' {
                escaped = true;
            } else if b == b'"' {
                quote_end = Some(i);
                break;
            }
            i += 1;
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: mapped(&positions, clean.len(), bytes.len(), 0),
            });
        }
    }

    for (i, &b) in clean.iter().enumerate() {
        if b == b'@' && !quote_end.is_some_and(|end| i <= end) {
            ats.push(i);
        }
    }

    for run in runs {
        let left_at = run.start == 0
            || positions
                .iter()
                .position(|&p| p >= run.start)
                .and_then(|idx| idx.checked_sub(1))
                .is_some_and(|idx| clean[idx] == b'@');
        let right_at = positions
            .iter()
            .position(|&p| p >= run.end)
            .is_some_and(|idx| clean[idx] == b'@');
        let at_end = run.end == bytes.len() || !positions.iter().any(|&p| p >= run.end);
        if !(left_at || right_at || at_end) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: mapped(&positions, clean.len(), bytes.len(), ats[1]),
        });
    }

    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: mapped(&positions, clean.len(), bytes.len(), sep),
        });
    }

    let local_bytes = &clean[..sep];
    let local_positions = &positions[..sep];
    let quoted = local_bytes.first() == Some(&b'"');
    let local = if !quoted {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_positions[64],
            });
        }
        let mut previous_dot = false;
        for (i, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || previous_dot {
                    return Err(EmailError::LocalDot {
                        at: local_positions[i],
                    });
                }
                previous_dot = true;
            } else {
                previous_dot = false;
                let allowed = b.is_ascii_alphanumeric()
                    || b"!#$%&'*+-/=?^_`{|}~".contains(&b);
                if !allowed {
                    return Err(EmailError::LocalChar {
                        at: local_positions[i],
                    });
                }
            }
        }
        String::from_utf8(local_bytes.to_vec()).unwrap()
    } else {
        let end_quote = quote_end.unwrap();
        if end_quote + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: mapped(&positions, clean.len(), bytes.len(), end_quote + 1),
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < end_quote {
            let b = clean[i];
            if b == b'\\' {
                if i + 1 >= end_quote || !matches!(clean[i + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: local_positions[i],
                    });
                }
                decoded.push(clean[i + 1]);
                origins.push(local_positions[i]);
                i += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: local_positions[i],
                    });
                }
                decoded.push(b);
                origins.push(local_positions[i]);
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: local_positions[end_quote],
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        String::from_utf8(decoded).unwrap()
    };

    let domain_start = sep + 1;
    let mut domain_end = clean.len();
    let mut root_dot = None;
    if domain_end > domain_start && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_dot = Some(domain_end);
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: mapped(&positions, clean.len(), bytes.len(), domain_start),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start + 253],
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || clean[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }

    for (label_index, &(start, end)) in labels.iter().enumerate() {
        if start == end {
            let at = if start == domain_end {
                root_dot
                    .map(|idx| positions[idx])
                    .unwrap_or_else(|| mapped(&positions, clean.len(), bytes.len(), start))
            } else {
                mapped(&positions, clean.len(), bytes.len(), start)
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: positions[start + 63],
            });
        }
        if clean[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[start],
            });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[end - 1],
            });
        }
        for i in start..end {
            if !(clean[i].is_ascii_alphanumeric() || clean[i] == b'-') {
                return Err(EmailError::DomainChar { at: positions[i] });
            }
        }
        let _ = label_index;
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: mapped(&positions, clean.len(), bytes.len(), domain_start),
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: positions[last_start],
        });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(|&b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
