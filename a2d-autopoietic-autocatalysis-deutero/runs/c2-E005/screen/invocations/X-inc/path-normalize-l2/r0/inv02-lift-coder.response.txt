#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.bytes().position(|byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.bytes().position(|byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .bytes()
        .position(|byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str) -> Vec<(usize, &str)> {
    path.split('/')
        .scan(0usize, |offset, part| {
            let start = *offset;
            *offset += part.len() + 1;
            Some((start, part))
        })
        .filter(|(_, part)| !part.is_empty())
        .collect()
}

fn directory_intent(path: &str, parts: &[(usize, &str)]) -> bool {
    path.ends_with('/')
        || parts
            .last()
            .is_some_and(|(_, part)| *part == "." || *part == "..")
}

fn render(absolute: bool, parts: &[String], directory: bool) -> String {
    let mut result = if absolute {
        format!("/{}", parts.join("/"))
    } else if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    };

    if directory && result != "/" && result != "." {
        result.push('/');
    }

    result
}

fn reduce(
    stack: &mut Vec<String>,
    parts: &[(usize, &str)],
    absolute: bool,
) -> Result<(), PathError> {
    for &(at, part) in parts {
        match part {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|component| component != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push("..".to_string());
                }
            }
            _ => stack.push(part.to_string()),
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.starts_with('/');
    let parts = components(path);
    let directory = directory_intent(path, &parts);
    let mut stack = Vec::new();

    reduce(&mut stack, &parts, absolute)?;
    Ok(render(absolute, &stack, directory))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }

    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(normalized
        .split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_string)
        .collect())
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;

    let home_parts = match home {
        Some(home) => Some(validate_context(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let requested_parts = components(path);
    let directory = directory_intent(path, &requested_parts);

    let mut absolute = false;
    let mut stack = Vec::new();
    let mut remainder = path;

    if path.starts_with('/') {
        absolute = true;
    } else if path.starts_with('~') {
        if path == "~" || path.starts_with("~/") {
            stack = home_parts.ok_or(PathError::MissingHome)?;
            remainder = if path == "~" {
                ""
            } else {
                &path[2..]
            };
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else {
        stack = cwd_parts;
    }

    let remainder_parts = components(remainder);
    reduce(&mut stack, &remainder_parts, true)?;

    Ok(render(true || absolute, &stack, directory))
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn public_normalize_again() {
        assert_eq!(normalize("//p/./q"), Ok("/p/q".into()));
    }

    #[test]
    fn public_cwd() {
        assert_eq!(resolve("../file", "/work/source", None), Ok("/work/file".into()));
        assert_eq!(resolve("/log/", "/work", None), Ok("/log/".into()));
    }

    #[test]
    fn public_home() {
        assert_eq!(resolve("~/notes", "/work", Some("/users/me")), Ok("/users/me/notes".into()));
        assert_eq!(resolve("~", "/work", Some("/users/me")), Ok("/users/me".into()));
    }

    #[test]
    fn public_context_error() {
        assert_eq!(resolve("file", "work", None), Err(PathError::InvalidCwd));
        assert_eq!(resolve("~", "/", None), Err(PathError::MissingHome));
    }
}
