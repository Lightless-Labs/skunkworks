#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<(usize, &str)> {
    path.split('/')
        .scan(0usize, |offset, component| {
            let start = *offset;
            *offset += component.len() + 1;
            Some((start, component))
        })
        .filter(|(_, component)| !component.is_empty())
        .collect()
}

fn reduce(
    stack: &mut Vec<String>,
    path: &str,
    start: usize,
    absolute: bool,
) -> Result<(), PathError> {
    for (at, component) in components(path) {
        if at < start {
            continue;
        }
        match component {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push("..".to_string());
                }
            }
            ordinary => stack.push(ordinary.to_string()),
        }
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, intent: bool) -> String {
    let joined = stack.join("/");
    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_string()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_string()
    } else {
        joined
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    reduce(&mut stack, path, 0, absolute)?;
    Ok(render(&stack, absolute, intent))
}

fn validated_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    Ok(normalized
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_string)
        .collect())
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_components = validated_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(validated_context(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let tilde = path.starts_with('~');
    let (mut stack, remainder, remainder_offset) = if tilde {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home_components = home_components.ok_or(PathError::MissingHome)?;
        let offset = if bytes.len() > 1 { 2 } else { 1 };
        (home_components, &path[offset..], offset)
    } else if path.starts_with('/') {
        (Vec::new(), path, 0)
    } else {
        (cwd_components, path, 0)
    };

    let intent = directory_intent(path);
    reduce(&mut stack, remainder, remainder_offset, true)?;
    Ok(render(&stack, true, intent))
}```
