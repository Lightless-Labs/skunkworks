#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn has_directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn reduce_components<'a, I>(
    components: I,
    absolute: bool,
    base: &mut Vec<&'a str>,
) -> Result<(), usize>
where
    I: IntoIterator<Item = (&'a str, usize)>,
{
    for (component, offset) in components {
        match component {
            "" | "." => {}
            ".." => {
                if base.last().is_some_and(|last| *last != "..") {
                    base.pop();
                } else if absolute {
                    return Err(offset);
                } else {
                    base.push("..");
                }
            }
            _ => base.push(component),
        }
    }
    Ok(())
}

fn render(components: &[&str], absolute: bool, directory_intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", components.join("/"))
    } else if components.is_empty() {
        ".".to_string()
    } else {
        components.join("/")
    };

    if absolute && components.is_empty() {
        return "/".to_string();
    }

    if directory_intent && result != "." && result != "/" {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let directory_intent = has_directory_intent(path);
    let mut components = Vec::new();
    let mut offset = 0;

    for component in path.split('/') {
        let component_offset = offset;
        offset += component.len() + 1;
        components.push((component, component_offset));
    }

    if let Err(at) = reduce_components(components, absolute, &mut Vec::new()) {
        return Err(PathError::Escape { at });
    }

    let mut reduced = Vec::new();
    let mut offset = 0;
    for component in path.split('/') {
        let component_offset = offset;
        offset += component.len() + 1;
        if component == "" || component == "." {
            continue;
        }
        if component == ".." {
            if reduced.last().is_some_and(|last: &&str| **last != "..") {
                reduced.pop();
            } else if absolute {
                return Err(PathError::Escape {
                    at: component_offset,
                });
            } else {
                reduced.push(component);
            }
        } else {
            reduced.push(component);
        }
    }

    Ok(render(&reduced, absolute, directory_intent))
}

fn validate_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') || validate_raw(path).is_err() {
        return Err(());
    }

    let normalized = normalize(path).map_err(|_| ())?;
    Ok(normalized
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_string)
        .collect())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(validate_context(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde_request = bytes.first() == Some(&b'~');
    let (absolute, start, remainder, remainder_offset, tilde_slash) = if tilde_request {
        if path == "~" {
            let components = home_components.ok_or(PathError::MissingHome)?;
            let mut base: Vec<String> = components;
            let result: Vec<&str> = base.iter().map(String::as_str).collect();
            return Ok(render(&result, true, false));
        }

        if bytes.get(1) == Some(&b'/') {
            let base = home_components.ok_or(PathError::MissingHome)?;
            (true, base, &path[2..], 2usize, true)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if bytes.first() == Some(&b'/') {
        (true, Vec::new(), path, 0usize, false)
    } else {
        (true, cwd_components, path, 0usize, false)
    };

    let directory_intent = if tilde_request {
        tilde_slash || has_directory_intent(remainder)
    } else {
        has_directory_intent(path)
    };

    let mut base: Vec<String> = start;
    let mut component_offset = remainder_offset;
    for component in remainder.split('/') {
        let at = component_offset;
        component_offset += component.len() + 1;

        match component {
            "" | "." => {}
            ".." => {
                if base.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => base.push(component.to_string()),
        }
    }

    let result: Vec<&str> = base.iter().map(String::as_str).collect();
    Ok(render(&result, absolute, directory_intent))
}
