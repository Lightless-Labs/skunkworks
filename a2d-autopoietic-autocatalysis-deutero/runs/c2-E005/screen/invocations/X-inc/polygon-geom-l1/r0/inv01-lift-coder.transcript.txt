pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = (b.x as i128) - (a.x as i128);
    let dy = (b.y as i128) - (a.y as i128);

    // Coordinates are range-checked before this function is called.
    let scaled_squared = ((dx * dx + dy * dy) as u128) * 1_000_000u128;

    // Integer square root by binary search. The result is at most 2.83e9.
    let mut low = 0u128;
    let mut high = 3_000_000_001u128;
    while low + 1 < high {
        let mid = low + (high - low) / 2;
        if mid * mid <= scaled_squared {
            low = mid;
        } else {
            high = mid;
        }
    }

    // Round to nearest integer, with exact half ties upward. Compare
    // scaled_squared against (low + 1/2)^2 without floating point:
    // 4*S >= (2*low + 1)^2.
    let midpoint_twice = 2 * low + 1;
    if 4 * scaled_squared >= midpoint_twice * midpoint_twice {
        (low + 1) as u64
    } else {
        low as u64
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, point) in points.iter().enumerate() {
        if point.x < -COORD_LIMIT
            || point.x > COORD_LIMIT
            || point.y < -COORD_LIMIT
            || point.y > COORD_LIMIT
        {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if points.len() >= 2 && points.last() == points.first() {
        points.len() - 1
    } else {
        points.len()
    };

    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    let polygon = &points[..effective_len];
    let mut signed_area2 = 0i128;
    let mut edge_milli = Vec::with_capacity(effective_len);
    let mut perimeter_milli = 0u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..effective_len {
        let a = polygon[i];
        let b = polygon[(i + 1) % effective_len];

        signed_area2 += (a.x as i128) * (b.y as i128) - (a.y as i128) * (b.x as i128);

        let length = rounded_edge_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let replace = match longest_edge {
            None => true,
            Some(previous) => {
                let previous_length = edge_milli[previous];
                if length != previous_length {
                    length > previous_length

                    let (low, high) = if a <= b { (a, b) } else { (b, a) };
                    let pa = polygon[previous];
                    let pb = polygon[(previous + 1) % effective_len];

                        if pa <= pb { (pa, pb) } else { (pb, pa) };

                    (low, high, i) < (previous_low, previous_high, previous)
                }
            }
        };

        if replace {

        }
    }

        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}
