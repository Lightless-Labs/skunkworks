    end: usize,

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut parsed = Vec::with_capacity(5);
    for (index, field) in FIELDS.iter().copied().enumerate() {
        parsed.push(parse_field(parts[index], field)?);
    }

    Ok(Cron {
        selected: parsed.try_into().ok().unwrap(),
        day_of_month_is_star: parts[2] == "*",
        day_of_week_is_star: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &yes)| yes.then_some(value as u8))
            .collect()
    }

        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize]
            && self.selected[3][at.month as usize]
            && self.selected[4][weekday as usize])
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut date = CivilDateTime {
            year: after.year,
            month: after.month,
            day: after.day,
            hour: 0,
            minute: 0,
        };

        loop {
            if self.selected[3][date.month as usize]
                && self.selected[2][date.day as usize]
                && self.selected[4][weekday(date.year, date.month, date.day) as usize]
            {
                for hour in 0..24u8 {
                    if !self.selected[1][hour as usize] {
                        continue;
                    }
                    for minute in 0..60u8 {
                        if !self.selected[0][minute as usize] {
                            continue;
                        }
                        let candidate = CivilDateTime { hour, minute, ..date };
                        if candidate > after {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }

            if date.year == 2399 && date.month == 12 && date.day == 31 {
                break;
            }
            advance_day(&mut date);
        }

        Ok(None)
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn advance_day(date: &mut CivilDateTime) {
    if date.day < days_in_month(date.year, date.month) {
        date.day += 1;
    } else if date.month < 12 {
        date.month += 1;
        date.day = 1;
    } else {
        date.year += 1;
        date.month = 1;
        date.day = 1;
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    let error = |kind: fn(Field, usize) -> ParseError| kind(field, start);

    if end == start {
        return Err(error(|field, offset| ParseError::InvalidSyntax { field, offset }));
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(error(|field, offset| ParseError::LeadingZero { field, offset }));
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        let digit = (byte - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(digit))
            .ok_or_else(|| error(|field, offset| ParseError::NumberOverflow { field, offset }))?;
    }

    if is_step {
        let (_, _, width) = domain(field);
        if value == 0 || value > width {
            return Err(error(|field, offset| ParseError::BadStep { field, offset }));
        }
    } else {
        let (min, max, _) = domain(field);
        if value < min || value > max {
            return Err(error(|field, offset| ParseError::OutOfRange { field, offset }));
        }
    }

    Ok(value)
}

        };

        let (start_value, end_value) = if base == b"*" {
            (min, max)
        } else if hyphens.is_empty() {
            let value = parse_number(item, 0, base.len(), field, false)?;
            (value, if step.is_some() { max } else { value })
        } else {
            let h = hyphens[0];
            let start_value = parse_number(item, 0, h, field, false)?;
            let end_value = parse_number(item, h + 1, base.len(), field, false)?;
            if start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_start + h + 1,
                });
            }
            (start_value, end_value)
        };

        let increment = step.unwrap_or(1);
        let mut value = start_value;
        while value <= end_value {
            selected[value as usize] = true;
            match value.checked_add(increment) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok((min..=max)
        .filter(|&value| selected[value as usize])
        .map(|value| value as u8)
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut parsed = Vec::with_capacity(5);
    for (piece, field) in pieces.iter().zip(fields) {
        parsed.push(parse_field(piece, field)?);
    }

    Ok(Cron {
        selected: parsed.try_into().ok().expect("five parsed fields"),
        day_of_month_is_star: pieces[2].as_bytes() == b"*",
        day_of_week_is_star: pieces[4].as_bytes() == b"*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)].clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.selected[0].contains(&at.minute)
            && self.selected[1].contains(&at.hour)
            && self.selected[2].contains(&at.day)
            && self.selected[3].contains(&at.month)
            && self.selected[4].contains(&weekday))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut year = after.year;
        let mut month = after.month;
        let mut day = after.day;
        let mut hour = after.hour;
        let mut minute = after.minute;

        loop {
            if minute < 59 {
                minute += 1;
            } else {
                minute = 0;
                if hour < 23 {
                    hour += 1;
                } else {
                    hour = 0;
                    if day < days_in_month(year, month) {
                        day += 1;
                    } else {
                        day = 1;
                        if month < 12 {
                            month += 1;
                        } else if year < 2399 {
                            year += 1;
                            month = 1;
                        } else {
                            return Ok(None);
                        }
                    }
                }
            }

            let candidate = CivilDateTime {
                year,
                month,
                day,
                hour,
                minute,
            };
            if self.matches(candidate)? {
                return Ok(Some(candidate));
            }
        }
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    let error = |kind: fn(Field, usize) -> ParseError| kind(field, start);
    if end - start > 1 && bytes[start] == b'0' {
        return Err(error(|field, offset| ParseError::LeadingZero { field, offset }));
    }
    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or_else(|| error(|field, offset| ParseError::NumberOverflow { field, offset }))?;
    }
    Ok(value)

fn parse_field(input: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = input.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max, width) = domain(field);
    let mut selected = vec![false; (max + 1) as usize];
    let mut item_start = 0;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|n| item_start + n)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        let (base_end, step_start) = match slash_positions.as_slice() {
            [] => (item.len(), None),
            [slash] => (*slash, Some(slash + 1)),
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        };

        let base = &item[..base_end];
        let star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_kind = if star && hyphens.is_empty() {
            0 // star
        } else if !star && hyphens.is_empty() && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            1 // number
        } else if !star
            && hyphens.len() == 1
            && hyphens[0] > 0
            && hyphens[0] + 1 < base.len()
            && base[..hyphens[0]].iter().all(u8::is_ascii_digit)
            && base[hyphens[0] + 1..].iter().all(u8::is_ascii_digit)
        {
            2 // range
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        };

        if let Some(step_start) = step_start {
            if step_start == item.len() || !item[step_start..].iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }

        let (start_value, end_value, end_offset) = match base_kind {
            0 => (min, max, item_start),
            1 => {
                let n = parse_number(base, 0, base.len(), field)?;
                if n < min || n > max {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start,
                    });
                }
                (n, n, item_start)
            }
            _ => {
                let hyphen = hyphens[0];
                let start = parse_number(base, 0, hyphen, field)?;
                if start < min || start > max {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start,
                    });
                }
                let end = parse_number(base, hyphen + 1, base.len(), field)?;
                let end_offset = item_start + hyphen + 1;
                if end < min || end > max {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: end_offset,
                    });
                }
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_offset,
                    });
                }
                (start, end, end_offset)
            }
        };

        let step = if let Some(step_start) = step_start {
            let step = parse_number(item, step_start, item.len(), field)?;
            if step == 0 || step > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: item_start + step_start,
                });
            }
            step
        } else {
            1
        };

        let actual_end = if step_start.is_some() && base_kind == 1 {
            max
        } else {
            end_value
        };
        let mut value = start_value;
        while value <= actual_end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let item = &bytes[item_start..];
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
        .collect();

    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let (base_end_rel, step_range) = if let Some(&slash) = slash_positions.first() {
        (slash, Some((slash + 1, item.len())))
    } else {
        (item.len(), None)
    };

    let base = &item[..base_end_rel];
    let (min, max) = bounds(field);

    let base_kind = if base == b"*" {
        (0u8, 0usize, 0usize)
    } else {
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
            .collect();
        if hyphens.len() > 1 || (hyphens.len() == 1 && (base[0] == b'*' || base[hyphens[0] + 1..].contains(&b'*'))) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        if base.contains(&b'*') {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        if let Some(&hyphen) = hyphens.first() {
            if hyphen == 0 || hyphen + 1 == base.len() {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            (2u8, hyphen, hyphen + 1)
        } else {
            if base.is_empty() || !base.iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            (1u8, 0, 0)
        }
    };

    if base_kind == (0, 0, 0) {
        // Star base.
    } else if base_kind.0 == 2 {
        if !base[..base_kind.1].iter().all(u8::is_ascii_digit)
            || !base[base_kind.2..].iter().all(u8::is_ascii_digit)
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
    }

    if let Some((step_start_rel, step_end_rel)) = step_range {
        if step_start_rel == step_end_rel
            || !item[step_start_rel..step_end_rel]
                .iter()
                .all(u8::is_ascii_digit)
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
    }

    let mut range_start = min;
    let mut range_end = max;
    let mut range_end_offset = item_start;

    match base_kind.0 {
        0 => {}
        1 => {
            let n = parse_number(bytes, item_start, item_start + base.len(), field)?;
            if n < min || n > max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_start,
                });
            }
            range_start = n;
            range_end = if step_range.is_some() { max } else { n };
        }
        2 => {
            let hyphen = base_kind.1;
            let first = parse_number(bytes, item_start, item_start + hyphen, field)?;
            let second_start = item_start + hyphen + 1;
            let second = parse_number(bytes, second_start, item_start + base.len(), field)?;
            if first < min || first > max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_start,
                });
            }
            if second < min || second > max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: second_start,
                });
            }
            if first > second {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: second_start,
                });
            }
            range_start = first;
            range_end = second;
            range_end_offset = second_start;
        }
        _ => unreachable!(),
    }

    let step = if let Some((step_start_rel, step_end_rel)) = step_range {
        let step_offset = item_start + step_start_rel;
        let value = parse_number(bytes, step_offset, item_start + step_end_rel, field)?;
        let width = max - min + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: step_offset,
            });
        }
        value
    } else {
        1
    };

    let _ = range_end_offset;
    let mut value = range_start;
    while value <= range_end {
        selected[(value - min) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_field(field_text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let (min, max) = bounds(field);
    let mut selected = vec![false; (max - min + 1) as usize];
    let bytes = field_text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut start = 0usize;
    loop {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| start + p)
            .unwrap_or(bytes.len());
        if start == end {
            return Err(ParseError::EmptyItem {
                field,
                offset: start,
            });
        }
        parse_item(&bytes[..end], start, field, &mut selected)?;
        if end == bytes.len() {
            break;
        }
        start = end + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for (i, field) in FIELDS.iter().copied().enumerate() {
        selected[i] = parse_field(parts[i], field)?;
    }

    Ok(Cron {
        selected,
        day_of_month_exact_star: parts[2].as_bytes() == b"*",
        day_of_week_exact_star: parts[4].as_bytes() == b"*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, _) = bounds(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| {
                if yes {
                    Some(min as u8 + i as u8)
                } else {
                    None
                }
            })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        if after.year == 2399 && after.month == 12 && after.day == 31
            && after.hour == 23 && after.minute == 59
        {
            return Ok(None);
        }

        let mut current = after;
        loop {
            current = advance_minute(current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
            if current.year == 2399 && current.month == 12 && current.day == 31
                && current.hour == 23 && current.minute == 59
            {
                return Ok(None);
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        let weekday = weekday(at.year, at.month, at.day);
        self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize - 1]
            && self.selected[3][at.month as usize - 1]
            && self.selected[4][weekday as usize]
    }
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > month_length(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_length(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn advance_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < month_length(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}
