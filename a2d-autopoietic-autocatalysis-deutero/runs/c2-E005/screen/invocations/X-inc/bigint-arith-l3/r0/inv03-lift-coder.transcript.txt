#[derive(Clone, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented by [0] and is never negative.
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

impl fmt::Debug for Big {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_struct("Big")
            .field("negative", &self.negative)
            .field("digits", &self.digits)
            .finish()
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],

    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn normalize(&mut self) {
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        if let Some(index) = (start..bytes.len())
            .find(|&i| bytes[i] != b'_' && !bytes[i].is_ascii_digit())
        {
            return Err(Error::InvalidDigit { index });
        }

                    || i + 1 == bytes.len()

                    || !bytes[i + 1].is_ascii_digit())

                return Err(Error::Separator { index: i });
            }

            .find(|&i| bytes[i] != b'_')

            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            let index = (start..bytes.len())
                .find(|&i| {
                    if bytes[i] == b'_' {
                        return false;
                    }
                    seen += 1;
                    seen == excess_number
                })
                .expect("digit count exceeds limit");
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });

            .iter()
            .filter(|&&b| b != b'_')
            .rev()
            .map(|&b| b - b'0')
            .collect::<Vec<_>>();

        let mut result = Self { negative, digits };
        result.normalize();
        Ok(result)
    }

                        other => return other,

            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => rhs.cmp_abs(self),

                borrow = 0;

                    result

        }

        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut accum = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            for (j, &b) in rhs.digits.iter().enumerate() {
                accum[i + j] += (a as u32) * (b as u32);
            }
        }

        let mut digits = Vec::with_capacity(accum.len() + 1);
        let mut carry = 0u32;
        for value in accum {
            let total = value + carry;
            digits.push((total % 10) as u8);
            carry = total / 10;
        }
        while carry != 0 {
            digits.push((carry % 10) as u8);
            carry /= 10;
        }

        let mut result = Self {
            negative: self.negative ^ rhs.negative,
            digits,
        };
        result.normalize();
        result
    }

            carry = product / 10;

            digits.push((carry % 10) as u8);
            carry /= 10;
        }
        let mut result = Self {
            negative: false,
            digits,
        };
        result.normalize();
        result
    }

            self.digits[0] = digit;
        } else {
            self.digits.insert(0, digit);
        }
        self.normalize();
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let mut dividend = self.clone();
        dividend.negative = false;
        let mut divisor = rhs.clone();
        divisor.negative = false;

            return Ok((Self::zero(), self.clone()));
        }

        let mut remainder = Self::zero();

            digits: quotient_digits,
        };
