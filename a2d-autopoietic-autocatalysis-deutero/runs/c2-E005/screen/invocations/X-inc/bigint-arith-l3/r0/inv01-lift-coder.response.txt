```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented by [0] and is never negative.
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let sign_offset = if bytes[0] == b'+' || bytes[0] == b'-' {
            1
        } else {
            0
        };
        if sign_offset == bytes.len() {
            return Err(Error::SignOnly);
        }

        let magnitude = &bytes[sign_offset..];

        if let Some(index) = magnitude
            .iter()
            .position(|&byte| byte != b'_' && !byte.is_ascii_digit())
        {
            return Err(Error::InvalidDigit {
                index: sign_offset + index,
            });
        }

        for i in 0..magnitude.len() {
            if magnitude[i] == b'_'
                && (i == 0
                    || i + 1 == magnitude.len()
                    || !magnitude[i - 1].is_ascii_digit()
                    || !magnitude[i + 1].is_ascii_digit())
            {
                return Err(Error::Separator {
                    index: sign_offset + i,
                });
            }
        }

        let digit_count = magnitude.iter().filter(|&&b| b != b'_').count();
        let first_digit_index = magnitude
            .iter()
            .position(|&b| b != b'_')
            .expect("nonempty magnitude contains a digit");

        if digit_count > 1 && magnitude[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: sign_offset + first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_digit = max_digits;
            let mut seen = 0;
            let byte_index = magnitude
                .iter()
                .position(|&b| {
                    if b == b'_' {
                        false
                    } else {
                        let is_excess = seen == excess_digit;
                        seen += 1;
                        is_excess
                    }
                })
                .expect("digit count exceeds limit");
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: sign_offset + byte_index,
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in magnitude.iter().rev() {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }

        let mut result = Self {
            negative: bytes[0] == b'-',
            digits,
        };
        result.normalize();
        Ok(result)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let natural_len = self.digits.len().max(1);
        let width = natural_len.max(min_digits);
        let padding = width - natural_len;

        let mut out = String::with_capacity(width.saturating_add(width / 2) + 1);
        if self.negative && !self.is_zero() {
            out.push('-');
        }

        for position in (0..width).rev() {
            if group > 0 && position + 1 < width && (width - position - 1) % group == 0 {
                out.push('_');
            }
            let digit = if position >= padding {
                self.digits[position - padding]
            } else {
                0
            };
            out.push((b'0' + digit) as char);
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.signum(), rhs.signum()) {
            (a, b) if a != b => a.cmp(&b),
            (0, 0) => Ordering::Equal,
            (-1, -1) => cmp_magnitude(&rhs.digits, &self.digits),
            _ => cmp_magnitude(&self.digits, &rhs.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        match (self.negative, rhs.negative) {
            (a, b) if a == b => {
                let mut result = Self {
                    negative: a,
                    digits: add_magnitude(&self.digits, &rhs.digits),
                };
                result.normalize();
                result
            }
            _ => match cmp_magnitude(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self {
                        negative: self.negative,
                        digits: sub_magnitude(&self.digits, &rhs.digits),
                    };
                    result.normalize();
                    result
                }
                Ordering::Less => {
                    let mut result = Self {
                        negative: rhs.negative,
                        digits: sub_magnitude(&rhs.digits, &self.digits),
                    };
                    result.normalize();
                    result
                }
            },
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut result = Self {
            negative: self.negative ^ rhs.negative,
            digits: mul_magnitude(&self.digits, &rhs.digits),
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let (quotient_digits, remainder_digits) =
            div_rem_magnitude(&self.digits, &rhs.digits);

        let mut quotient = Self {
            negative: self.negative ^ rhs.negative,
            digits: quotient_digits,
        };
        quotient.normalize();

        let mut remainder = Self {
            negative: self.negative,
            digits: remainder_digits,
        };
        remainder.normalize();

        Ok((quotient, remainder))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn normalize(&mut self) {
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.is_empty() {
            self.digits.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }
}

fn cmp_magnitude(a: &[u8], b: &[u8]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => {
            for (&x, &y) in a.iter().rev().zip(b.iter().rev()) {
                match x.cmp(&y) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u8;

    for i in 0..a.len().max(b.len()) {
        let sum = a.get(i).copied().unwrap_or(0) + b.get(i).copied().unwrap_or(0) + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }
    if carry != 0 {
        result.push(carry);
    }
    result
}

// Requires a >= b.
fn sub_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len());
    let mut borrow = 0i8;

    for i in 0..a.len() {
        let mut digit = a[i] as i8 - borrow - b.get(i).copied().unwrap_or(0) as i8;
        if digit < 0 {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push(digit as u8);
    }

    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

fn mul_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = vec![0u8; a.len() + b.len() + 1];

    for (i, &x) in a.iter().enumerate() {
        let mut carry = 0u32;
        for (j, &y) in b.iter().enumerate() {
            let index = i + j;
            let value = result[index] as u32 + x as u32 * y as u32 + carry;
            result[index] = (value % 10) as u8;
            carry = value / 10;
        }

        let mut index = i + b.len();
        while carry != 0 {
            let value = result[index] as u32 + carry;
            result[index] = (value % 10) as u8;
            carry = value / 10;
            index += 1;
        }
    }

    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

// Returns magnitude quotient and remainder, each in little-endian decimal digits.
fn div_rem_magnitude(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
    if cmp_magnitude(dividend, divisor) == Ordering::Less {
        return (vec![0], dividend.to_vec());
    }

    let mut quotient_be = Vec::with_capacity(dividend.len());
    let mut remainder = vec![0];

    for &digit in dividend.iter().rev() {
        if remainder.len() == 1 && remainder[0] == 0 {
            remainder[0] = digit;
        } else {
            remainder.insert(0, digit);
        }
        trim_magnitude(&mut remainder);

        let mut quotient_digit = 0u8;
        while cmp_magnitude(&remainder, divisor) != Ordering::Less {
            remainder = sub_magnitude(&remainder, divisor);
            quotient_digit += 1;
        }
        quotient_be.push(quotient_digit);
    }

    quotient_be.reverse();
    trim_magnitude(&mut quotient_be);
    trim_magnitude(&mut remainder);
    (quotient_be, remainder)
}

fn trim_magnitude(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    if digits.is_empty() {
        digits.push(0);
    }
}
```
