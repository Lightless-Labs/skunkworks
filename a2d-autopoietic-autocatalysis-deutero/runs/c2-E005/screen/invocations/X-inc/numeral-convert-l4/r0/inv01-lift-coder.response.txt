#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 7] = [
    "quintillion",
    "quadrillion",
    "trillion",
    "billion",
    "million",
    "thousand",
    "",
];

const SCALE_VALUES: [u128; 7] = [
    1_000_000_000_000_000_000,
    1_000_000_000_000_000,
    1_000_000_000_000,
    1_000_000_000,
    1_000_000,
    1_000,
    1,
];

fn append_group(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            out.push_str(" and ");
            append_group(remainder, out);
        }
    } else if n >= 20 {
        out.push_str(TENS[(n / 10 - 2) as usize]);
        let remainder = n % 10;
        if remainder != 0 {
            out.push('-');
            out.push_str(SMALL[remainder as usize]);
        }
    } else if n > 0 {
        out.push_str(SMALL[n as usize]);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for index in (0..7).rev() {
        groups[index] = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let units = groups[6];
    let has_higher = groups[..6].iter().any(|&g| g != 0);
    let cross_and = has_higher && units > 0 && units < 100;

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }

    let mut wrote_group = false;
    for index in 0..7 {
        let group = groups[index];
        if group == 0 {
            continue;
        }
        if wrote_group {
            if index == 6 && cross_and {
                result.push_str(" and ");
            } else {
                result.push(' ');
            }
        }
        append_group(group, &mut result);
        if !SCALES[index].is_empty() {
            result.push(' ');
            result.push_str(SCALES[index]);
        }
        wrote_group = true;
    }
    Ok(result)
}

fn word_value(word: &str) -> Option<u128> {
    if let Some(index) = SMALL.iter().position(|&candidate| candidate == word) {
        return Some(index as u128);
    }
    if let Some(index) = TENS.iter().position(|&candidate| candidate == word) {
        return Some(((index + 2) * 10) as u128);
    }
    match word {
        "hundred" => Some(100),
        "thousand" => Some(1_000),
        "million" => Some(1_000_000),
        "billion" => Some(1_000_000_000),
        "trillion" => Some(1_000_000_000_000),
        "quadrillion" => Some(1_000_000_000_000_000),
        "quintillion" => Some(1_000_000_000_000_000_000),
        "and" | "minus" => Some(0),
        _ => None,
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();

    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (position, &(token, index)) in tokens.iter().enumerate() {
        if token == "minus" {
            if position != 0 {
                return Err(WordsError::InvalidWord { index });
            }
        } else if word_value(token).is_none() {
            return Err(WordsError::InvalidWord { index });
        }
    }

    let negative = tokens.first().is_some_and(|(word, _)| *word == "minus");
    let arithmetic_start = usize::from(negative);
    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for &(token, _) in &tokens[arithmetic_start..] {
        match token {
            "and" => {}
            "hundred" => group *= 100,
            "thousand" | "million" | "billion" | "trillion" | "quadrillion" | "quintillion" => {
                let scale = match token {
                    "thousand" => 1_000,
                    "million" => 1_000_000,
                    "billion" => 1_000_000_000,
                    "trillion" => 1_000_000_000_000,
                    "quadrillion" => 1_000_000_000_000_000,
                    _ => 1_000_000_000_000_000_000,
                };
                total += group * scale;
                group = 0;
            }
            _ => group += word_value(token).unwrap_or(0),
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical != input {
        let mismatch = canonical
            .bytes()
            .zip(bytes.iter().copied())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: mismatch });
    }
    Ok(value)
}
