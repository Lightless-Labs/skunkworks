fn encode_under_thousand(n: u16) -> String {
    if n < 20 {
        return SMALL[n as usize].to_string();
    }
    if n < 100 {
        let tens = TENS[(n / 10) as usize];
        let units = n % 10;
        return if units == 0 {
            tens.to_string()
        } else {
            format!("{tens}-{}", SMALL[units as usize])
        };
    }

    let hundreds = n / 100;
    let remainder = n % 100;
    if remainder == 0 {
        format!("{} hundred", SMALL[hundreds as usize])
    } else {
        format!(
            "{} hundred and {}",
            SMALL[hundreds as usize],
            encode_under_thousand(remainder)
        )
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut remaining = magnitude;
    let mut parts: Vec<(String, bool)> = Vec::new();

    for &(scale, name) in &SCALES[..5] {
        let group = (remaining / scale) as u16;
        remaining %= scale;
        if group != 0 {
            let mut spelling = encode_under_thousand(group);
            spelling.push(' ');
            spelling.push_str(name);
            parts.push((spelling, false));
        }
    }

    if remaining != 0 {
        let units = remaining as u16;
        parts.push((encode_under_thousand(units), units < 100));
    }

    let mut result = String::new();
    let last_index = parts.len().saturating_sub(1);
    for (i, (part, is_small_units)) in parts.into_iter().enumerate() {
        if i != 0 {
            if i == last_index && is_small_units {
                result.push_str(" and ");
            } else {
                result.push(' ');
            }
        }
        result.push_str(&part);
    }

    if value < 0 {
        Ok(format!("minus {result}"))
    } else {
        Ok(result)
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !byte.is_ascii_lowercase() && byte != b' ' && byte != b'-' {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, byte) in bytes.iter().enumerate() {
        if *byte == b' ' || *byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    let mut kinds = Vec::with_capacity(tokens.len());
    for &(word, index) in &tokens {
        if word == "minus" && !kinds.is_empty() {
            return Err(WordsError::InvalidWord { index });
        }
        match dictionary_word(word) {
            Some(kind) => kinds.push(kind),
            None => return Err(WordsError::InvalidWord { index }),
        }
    }

    let negative = matches!(kinds.first(), Some(TokenKind::Minus));
    let arithmetic = if negative { &kinds[1..] } else { &kinds[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for kind in arithmetic {
        match *kind {
            TokenKind::Minus => {}
            TokenKind::And => {}
            TokenKind::Add(value) => group += value,
            TokenKind::Hundred => group *= 100,
            TokenKind::Scale(scale) => {
                total += group * scale;
                group = 0;
            }
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let signed = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    if canonical.as_bytes() != input.as_bytes() {
        let differing = canonical
            .bytes()
            .zip(input.bytes())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }

    Ok(signed)
}

pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 10] = [
    "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty",
    "ninety",
];

const SCALES: [&str; 7] = [
    "", "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = Vec::new();
    let mut remaining = magnitude;
    for scale in SCALES {
        let group = (remaining % 1000) as u16;
        groups.push(group);
        remaining /= 1000;
    }

    let mut parts: Vec<String> = Vec::new();
    let mut higher_present = false;

    for index in (0..groups.len()).rev() {
        let group = groups[index];
        if group == 0 {
            continue;
        }

        let spelling = spell_group(group);
        if index != 0 {
            parts.push(format!("{} {}", spelling, SCALES[index]));
            higher_present = true;
        } else if higher_present && group < 100 {
            parts.push(format!("and {}", spelling));
        } else {
            parts.push(spelling);
        }
    }

    let mut result = parts.join(" ");
    if value < 0 {
        result.insert_str(0, "minus ");
    }
    Ok(result)
}

fn spell_group(number: u16) -> String {
    let mut result = String::new();
    if number >= 100 {
        result.push_str(SMALL[(number / 100) as usize]);
        result.push_str(" hundred");
        let remainder = number % 100;
        if remainder != 0 {
            result.push_str(" and ");
            result.push_str(&spell_under_hundred(remainder));
        }
    } else {
        result.push_str(&spell_under_hundred(number));
    }
    result
}

fn spell_under_hundred(number: u16) -> String {
    if number < 20 {
        return SMALL[number as usize].to_string();
    }
    let tens = (number / 10) as usize;
    let units = number % 10;
    if units == 0 {
        TENS[tens].to_string()
    } else {
        format!("{}-{}", TENS[tens], SMALL[units as usize])
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            let before_is_letter = index > 0 && bytes[index - 1].is_ascii_lowercase();
            let after_is_letter =
                index + 1 < bytes.len() && bytes[index + 1].is_ascii_lowercase();
            if !before_is_letter || !after_is_letter {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(usize, &str)> = Vec::new();
    let mut start = 0;
    for index in 0..=bytes.len() {
        if index == bytes.len() || bytes[index] == b' ' || bytes[index] == b'-' {
            tokens.push((start, &input[start..index]));
            start = index + 1;
        }
    }

    for (index, token) in &tokens {
        if *token == "minus" {
            if *index != tokens[0].0 {
                return Err(WordsError::InvalidWord { index: *index });
            }
        } else if !is_dictionary_word(token) {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let negative = tokens.first().map(|(_, word)| *word == "minus").unwrap_or(false);
    let arithmetic_tokens = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for (_, token) in arithmetic_tokens {
        match *token {
            "zero" => {}
            "one" => group += 1,
            "two" => group += 2,
            "three" => group += 3,
            "four" => group += 4,
            "five" => group += 5,
            "six" => group += 6,
            "seven" => group += 7,
            "eight" => group += 8,
            "nine" => group += 9,
            "ten" => group += 10,
            "eleven" => group += 11,
            "twelve" => group += 12,
            "thirteen" => group += 13,
            "fourteen" => group += 14,
            "fifteen" => group += 15,
            "sixteen" => group += 16,
            "seventeen" => group += 17,
            "eighteen" => group += 18,
            "nineteen" => group += 19,
            "twenty" => group += 20,
            "thirty" => group += 30,
            "forty" => group += 40,
            "fifty" => group += 50,
            "sixty" => group += 60,
            "seventy" => group += 70,
            "eighty" => group += 80,
            "ninety" => group += 90,
            "hundred" => group *= 100,
            "and" => {}
            "thousand" => apply_scale(&mut total, &mut group, 1_000),
            "million" => apply_scale(&mut total, &mut group, 1_000_000),
            "billion" => apply_scale(&mut total, &mut group, 1_000_000_000),
            "trillion" => apply_scale(&mut total, &mut group, 1_000_000_000_000),

            _ => unreachable!(),

        if total + group > LIMIT {

    }

        magnitude as i128

            .iter()

            index: differing_index,

}

fn apply_scale(total: &mut u128, group: &mut u128, scale: u128) {
    *total += *group * scale;

}
