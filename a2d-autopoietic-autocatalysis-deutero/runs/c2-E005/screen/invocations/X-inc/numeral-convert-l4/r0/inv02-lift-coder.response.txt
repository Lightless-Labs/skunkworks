#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 10] = [
    "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty",
    "ninety",
];

const SCALES: [&str; 7] = [
    "", "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = Vec::new();
    let mut remaining = magnitude;
    for scale in SCALES {
        let group = (remaining % 1000) as u16;
        groups.push(group);
        remaining /= 1000;
    }

    let mut parts: Vec<String> = Vec::new();
    let mut higher_present = false;

    for index in (0..groups.len()).rev() {
        let group = groups[index];
        if group == 0 {
            continue;
        }

        let spelling = spell_group(group);
        if index != 0 {
            parts.push(format!("{} {}", spelling, SCALES[index]));
            higher_present = true;
        } else if higher_present && group < 100 {
            parts.push(format!("and {}", spelling));
        } else {
            parts.push(spelling);
        }
    }

    let mut result = parts.join(" ");
    if value < 0 {
        result.insert_str(0, "minus ");
    }
    Ok(result)
}

fn spell_group(number: u16) -> String {
    let mut result = String::new();
    if number >= 100 {
        result.push_str(SMALL[(number / 100) as usize]);
        result.push_str(" hundred");
        let remainder = number % 100;
        if remainder != 0 {
            result.push_str(" and ");
            result.push_str(&spell_under_hundred(remainder));
        }
    } else {
        result.push_str(&spell_under_hundred(number));
    }
    result
}

fn spell_under_hundred(number: u16) -> String {
    if number < 20 {
        return SMALL[number as usize].to_string();
    }
    let tens = (number / 10) as usize;
    let units = number % 10;
    if units == 0 {
        TENS[tens].to_string()
    } else {
        format!("{}-{}", TENS[tens], SMALL[units as usize])
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            let before_is_letter = index > 0 && bytes[index - 1].is_ascii_lowercase();
            let after_is_letter =
                index + 1 < bytes.len() && bytes[index + 1].is_ascii_lowercase();
            if !before_is_letter || !after_is_letter {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(usize, &str)> = Vec::new();
    let mut start = 0;
    for index in 0..=bytes.len() {
        if index == bytes.len() || bytes[index] == b' ' || bytes[index] == b'-' {
            tokens.push((start, &input[start..index]));
            start = index + 1;
        }
    }

    for (index, token) in &tokens {
        if *token == "minus" {
            if *index != tokens[0].0 {
                return Err(WordsError::InvalidWord { index: *index });
            }
        } else if !is_dictionary_word(token) {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let negative = tokens.first().map(|(_, word)| *word == "minus").unwrap_or(false);
    let arithmetic_tokens = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for (_, token) in arithmetic_tokens {
        match *token {
            "zero" => {}
            "one" => group += 1,
            "two" => group += 2,
            "three" => group += 3,
            "four" => group += 4,
            "five" => group += 5,
            "six" => group += 6,
            "seven" => group += 7,
            "eight" => group += 8,
            "nine" => group += 9,
            "ten" => group += 10,
            "eleven" => group += 11,
            "twelve" => group += 12,
            "thirteen" => group += 13,
            "fourteen" => group += 14,
            "fifteen" => group += 15,
            "sixteen" => group += 16,
            "seventeen" => group += 17,
            "eighteen" => group += 18,
            "nineteen" => group += 19,
            "twenty" => group += 20,
            "thirty" => group += 30,
            "forty" => group += 40,
            "fifty" => group += 50,
            "sixty" => group += 60,
            "seventy" => group += 70,
            "eighty" => group += 80,
            "ninety" => group += 90,
            "hundred" => group *= 100,
            "and" => {}
            "thousand" => apply_scale(&mut total, &mut group, 1_000),
            "million" => apply_scale(&mut total, &mut group, 1_000_000),
            "billion" => apply_scale(&mut total, &mut group, 1_000_000_000),
            "trillion" => apply_scale(&mut total, &mut group, 1_000_000_000_000),
            "quadrillion" => apply_scale(&mut total, &mut group, 1_000_000_000_000_000),
            "quintillion" => apply_scale(&mut total, &mut group, 1_000_000_000_000_000_000),
            _ => unreachable!(),
        }

        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != bytes {
        let differing_index = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical {
            index: differing_index,
        });
    }
    Ok(value)
}

fn apply_scale(total: &mut u128, group: &mut u128, scale: u128) {
    *total += *group * scale;
    *group = 0;
}

fn is_dictionary_word(word: &str) -> bool {
    word == "zero"
        || word == "hundred"
        || word == "thousand"
        || word == "million"
        || word == "billion"
        || word == "trillion"
        || word == "quadrillion"
        || word == "quintillion"
        || word == "and"
        || SMALL[1..].contains(&word)
        || TENS[2..].contains(&word)
}
