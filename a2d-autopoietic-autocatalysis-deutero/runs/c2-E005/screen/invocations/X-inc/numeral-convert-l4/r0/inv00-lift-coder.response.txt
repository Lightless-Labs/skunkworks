#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 10] = [
    "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty",
    "ninety",
];

const SCALES: [(u128, &str); 6] = [
    (1_000_000_000_000_000, "quadrillion"),
    (1_000_000_000_000, "trillion"),
    (1_000_000_000, "billion"),
    (1_000_000, "million"),
    (1_000, "thousand"),
    (1, ""),
];

fn encode_under_thousand(n: u16) -> String {
    if n < 20 {
        return SMALL[n as usize].to_string();
    }
    if n < 100 {
        let tens = TENS[(n / 10) as usize];
        let units = n % 10;
        return if units == 0 {
            tens.to_string()
        } else {
            format!("{tens}-{}", SMALL[units as usize])
        };
    }

    let hundreds = n / 100;
    let remainder = n % 100;
    if remainder == 0 {
        format!("{} hundred", SMALL[hundreds as usize])
    } else {
        format!(
            "{} hundred and {}",
            SMALL[hundreds as usize],
            encode_under_thousand(remainder)
        )
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut remaining = magnitude;
    let mut parts: Vec<(String, bool)> = Vec::new();

    for &(scale, name) in &SCALES[..5] {
        let group = (remaining / scale) as u16;
        remaining %= scale;
        if group != 0 {
            let mut spelling = encode_under_thousand(group);
            spelling.push(' ');
            spelling.push_str(name);
            parts.push((spelling, false));
        }
    }

    if remaining != 0 {
        let units = remaining as u16;
        parts.push((encode_under_thousand(units), units < 100));
    }

    let mut result = String::new();
    let last_index = parts.len().saturating_sub(1);
    for (i, (part, is_small_units)) in parts.into_iter().enumerate() {
        if i != 0 {
            if i == last_index && is_small_units {
                result.push_str(" and ");
            } else {
                result.push(' ');
            }
        }
        result.push_str(&part);
    }

    if value < 0 {
        Ok(format!("minus {result}"))
    } else {
        Ok(result)
    }
}

#[derive(Clone, Copy)]
enum TokenKind {
    Minus,
    And,
    Hundred,
    Add(u128),
    Scale(u128),
}

fn dictionary_word(word: &str) -> Option<TokenKind> {
    match word {
        "minus" => Some(TokenKind::Minus),
        "and" => Some(TokenKind::And),
        "hundred" => Some(TokenKind::Hundred),
        "thousand" => Some(TokenKind::Scale(1_000)),
        "million" => Some(TokenKind::Scale(1_000_000)),
        "billion" => Some(TokenKind::Scale(1_000_000_000)),
        "trillion" => Some(TokenKind::Scale(1_000_000_000_000)),
        "quadrillion" => Some(TokenKind::Scale(1_000_000_000_000_000)),
        "quintillion" => Some(TokenKind::Scale(1_000_000_000_000_000_000)),
        "zero" => Some(TokenKind::Add(0)),
        "one" => Some(TokenKind::Add(1)),
        "two" => Some(TokenKind::Add(2)),
        "three" => Some(TokenKind::Add(3)),
        "four" => Some(TokenKind::Add(4)),
        "five" => Some(TokenKind::Add(5)),
        "six" => Some(TokenKind::Add(6)),
        "seven" => Some(TokenKind::Add(7)),
        "eight" => Some(TokenKind::Add(8)),
        "nine" => Some(TokenKind::Add(9)),
        "ten" => Some(TokenKind::Add(10)),
        "eleven" => Some(TokenKind::Add(11)),
        "twelve" => Some(TokenKind::Add(12)),
        "thirteen" => Some(TokenKind::Add(13)),
        "fourteen" => Some(TokenKind::Add(14)),
        "fifteen" => Some(TokenKind::Add(15)),
        "sixteen" => Some(TokenKind::Add(16)),
        "seventeen" => Some(TokenKind::Add(17)),
        "eighteen" => Some(TokenKind::Add(18)),
        "nineteen" => Some(TokenKind::Add(19)),
        "twenty" => Some(TokenKind::Add(20)),
        "thirty" => Some(TokenKind::Add(30)),
        "forty" => Some(TokenKind::Add(40)),
        "fifty" => Some(TokenKind::Add(50)),
        "sixty" => Some(TokenKind::Add(60)),
        "seventy" => Some(TokenKind::Add(70)),
        "eighty" => Some(TokenKind::Add(80)),
        "ninety" => Some(TokenKind::Add(90)),
        _ => None,
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !byte.is_ascii_lowercase() && byte != b' ' && byte != b'-' {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, byte) in bytes.iter().enumerate() {
        if *byte == b' ' || *byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    let mut kinds = Vec::with_capacity(tokens.len());
    for &(word, index) in &tokens {
        if word == "minus" && !kinds.is_empty() {
            return Err(WordsError::InvalidWord { index });
        }
        match dictionary_word(word) {
            Some(kind) => kinds.push(kind),
            None => return Err(WordsError::InvalidWord { index }),
        }
    }

    let negative = matches!(kinds.first(), Some(TokenKind::Minus));
    let arithmetic = if negative { &kinds[1..] } else { &kinds[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for kind in arithmetic {
        match *kind {
            TokenKind::Minus => {}
            TokenKind::And => {}
            TokenKind::Add(value) => group += value,
            TokenKind::Hundred => group *= 100,
            TokenKind::Scale(scale) => {
                total += group * scale;
                group = 0;
            }
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let signed = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    if canonical.as_bytes() != input.as_bytes() {
        let differing = canonical
            .bytes()
            .zip(input.bytes())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }

    Ok(signed)
}
