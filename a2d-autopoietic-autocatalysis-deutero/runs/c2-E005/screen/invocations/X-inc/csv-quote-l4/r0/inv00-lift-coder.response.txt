use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|v| v == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|v| v == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if pos == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0 unquoted, 1 quoted, 2 after closing quote
    let mut opening = 0usize;
    let mut record_index = 0usize;
    let mut field_started = false;
    let mut record_started = false;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if state != 1 && b == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: pos,
            });
        }

        if state == 1 {
            if b == quote {
                if bytes.get(pos + 1) == Some(&quote) {
                    field.push(quote as char);
                    pos += 2;
                    continue;
                } else {
                    state = 2;
                    pos += 1;
                    continue;
                }
            }
            if b < 0x80 {
                field.push(b as char);
                pos += 1;
            } else {
                let s = std::str::from_utf8(&bytes[pos..]).unwrap();
                let ch = s.chars().next().unwrap();
                field.push(ch);
                pos += ch.len_utf8();
            }
            continue;
        }

        if state == 2 {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                state = 0;
                field_started = false;
                record_started = true;
                pos += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                state = 0;
                field_started = false;
                record_started = false;
                record_index += 1;
                pos += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: pos,
            });
        }

        if b == delimiter {
            record.push(std::mem::take(&mut field));
            field_started = false;
            record_started = true;
            pos += 1;
        } else if b == b'\n' || b == b'\r' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            field_started = false;
            record_started = false;
            record_index += 1;
            pos += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: pos,
                });
            }
            state = 1;
            opening = pos;
            field_started = true;
            record_started = true;
            pos += 1;
        } else {
            field_started = true;
            record_started = true;
            if b < 0x80 {
                field.push(b as char);
                pos += 1;
            } else {
                let s = std::str::from_utf8(&bytes[pos..]).unwrap();
                let ch = s.chars().next().unwrap();
                field.push(ch);
                pos += ch.len_utf8();
            }
        }
    }

    if state == 1 {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    if record_started || field_started || !record.is_empty() || state == 2 {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

fn duplicate_error(header: &[String]) -> Option<CsvError> {
    let mut seen: HashMap<&str, usize> = HashMap::new();
    for (i, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first,
                second: i,
            });
        }
        seen.insert(name, i);
    }
    None
}

fn table_from_records(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut iter = records.into_iter();
            (iter.next(), iter.collect())
        }
    } else {
        (None, records)
    };

    if let Some(header) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(e) = duplicate_error(header) {
                return Err(e);
            }
        }
        for (i, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    finished: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            finished: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.finished {
            return result.clone();
        }
        let result = parse_with(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

    MissingCell,
    Empty,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    if !(b'1'..=b'9').contains(&b[start]) {
        return false;
    }
    b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if start >= b.len() {
        return false;
    }
    let dot = match b[start..].iter().position(|&x| x == b'.') {
        Some(n) => start + n,
        None => return false,
    };
    let whole = &b[start..dot];
    let frac = &b[dot + 1..];
    if frac.len() != 2 || !frac.iter().all(u8::is_ascii_digit) || whole.is_empty() {
        return false;
    }
    if whole.len() == 1 && whole[0] == b'0' {
        return true;
    }
    (b'1'..=b'9').contains(&whole[0]) && whole[1..].iter().all(u8::is_ascii_digit)
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) || s == "-0.00" {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let neg = s.starts_with('-');
    let body = if neg { &s[1..] } else { s };
    let dot = body.find('.').unwrap();
    let whole = body[..dot]
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let frac = body[dot + 1..].parse::<i128>().unwrap();
    let mut value = whole * 100 + frac;
    if neg {
        value = -value;
    }
    i64::try_from(value).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate(cell: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if spec.nullable && cell.is_empty() {
        return Ok(Value::Null);
    }
    if cell.is_empty() && spec.kind != ColumnType::Text {
        return Err(CellErrorKind::Empty);
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut out = Vec::with_capacity(columns.len());
        for (spec_index, spec) in columns.iter().enumerate() {
            let index = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let Some(index) = index else {
                out.push(None);
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                if policy != ErrorPolicy::Collect {
                    while out.len() < columns.len() {
                        out.push(None);
                    }
                    result.rows.push(out);
                    if policy == ErrorPolicy::Stop {
                        break 'rows;
                    }
                    continue 'rows;
                }
                continue;
            };
            let Some(cell) = row.get(index) else {
                out.push(None);
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(index),
                    kind: CellErrorKind::MissingCell,
                });
                if policy != ErrorPolicy::Collect {
                    while out.len() < columns.len() {
                        out.push(None);
                    }
                    result.rows.push(out);
                    if policy == ErrorPolicy::Stop {
                        break 'rows;
                    }
                    continue 'rows;
                }
                continue;
            };
            match evaluate(cell, spec) {
                Ok(value) => out.push(Some(value)),
                Err(kind) => {
                    out.push(None);
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(index),
                        kind,
                    });
                    if policy != ErrorPolicy::Collect {
                        while out.len() < columns.len() {
                            out.push(None);
                        }
                        result.rows.push(out);
                        if policy == ErrorPolicy::Stop {
                            break 'rows;
                        }
                        continue 'rows;
                    }
                }
            }
        }
        result.rows.push(out);
    }
    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {

    pub line_ending: LineEnding,

    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns { record: usize },
    DuplicateHeader {
        name: String,
        first: usize,
        second: usize,
    },
}

fn write_duplicate_error(header: &[String]) -> Option<WriteError> {
    let mut seen: HashMap<&str, usize> = HashMap::new();
    for (i, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(WriteError::DuplicateHeader {
                name: name.clone(),
                first,
                second: i,
            });
        }
        seen.insert(name, i);
    }
    None
}

fn emit_field(out: &mut String, field: &str, delimiter: u8, quote: u8, first: bool, sole: bool) {
    let bytes = field.as_bytes();
    let edge_space = bytes.first().is_some_and(|&b| b == b' ' || b == b'\t')
        || bytes.last().is_some_and(|&b| b == b' ' || b == b'\t');
    let starts_bom = first && bytes.starts_with(&[0xef, 0xbb, 0xbf]);
    let needs_quote = field.is_empty() && sole
        || edge_space
        || starts_bom
        || bytes
            .iter()
            .any(|&b| b == delimiter || b == quote || b == b'\r' || b == b'\n');

    if !needs_quote {
        out.push_str(field);
        return;
    }

    out.push(quote as char);
    for &b in bytes {
        if b == quote {
            out.push(quote as char);
        }
        out.push(b as char);
    }
    out.push(quote as char);
}

fn emit_record(out: &mut String, row: &[String], delimiter: u8, quote: u8, first_record: bool) {
    for (i, field) in row.iter().enumerate() {
        if i > 0 {
            out.push(delimiter as char);
        }
        emit_field(out, field, delimiter, quote, first_record && i == 0, row.len() == 1);
    }
}

fn row_cmp(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let cmp = x.as_bytes().cmp(y.as_bytes());
        if cmp != std::cmp::Ordering::Equal {
            return cmp;
        }
    }
    a.len().cmp(&b.len())
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original: Vec<(usize, &Vec<String>)> = Vec::new();
    if let Some(header) = &table.header {
        original.push((0, header));
    }
    for (i, row) in table.rows.iter().enumerate() {
        original.push((if table.header.is_some() { i + 1 } else { i }, row));
    }

    for &(record, row) in &original {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for &(record, row) in &original {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(err) = write_duplicate_error(header) {
                    return Err(err);
                }
            }
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                output_records.push(header.clone());
            }
            output_records.extend(table.rows.iter().cloned());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut retained: HashMap<&str, usize> = HashMap::new();
                for (i, name) in header.iter().enumerate() {
                    match table.duplicates {
                        DuplicatePolicy::KeepLast => {
                            retained.insert(name, i);
                        }
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                            retained.entry(name).or_insert(i);
                        }
                    }
                }
                let mut indexes: Vec<usize> = retained.values().copied().collect();
                indexes.sort_by(|&a, &b| {
                    header[a]
                        .as_bytes()
                        .cmp(header[b].as_bytes())
                });
                output_records.push(indexes.iter().map(|&i| header[i].clone()).collect());
                let mut projected: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| indexes.iter().map(|&i| row[i].clone()).collect())
                    .collect();
                projected.sort_by(|a, b| row_cmp(a, b));
                output_records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| row_cmp(a, b));
                output_records = rows;
            }
        }
    }

    if output_records.is_empty() {
        return Ok(String::new());
    }

    let mut out = String::new();
    let canonical = config.mode == WriteMode::Canonical;
    let line_ending: &[u8] = if canonical {
        b"\n"
    } else {
        match config.line_ending {
            LineEnding::Lf => b"\n",
            LineEnding::CrLf => b"\r\n",
        }
    };

    for (i, row) in output_records.iter().enumerate() {
        if i > 0 {
            out.push_str(std::str::from_utf8(line_ending).unwrap());
        }
        emit_record(&mut out, row, config.delimiter, config.quote, i == 0);
    }

    if canonical || config.final_terminator {
        out.push_str(std::str::from_utf8(line_ending).unwrap());
    }
    Ok(out)
}
