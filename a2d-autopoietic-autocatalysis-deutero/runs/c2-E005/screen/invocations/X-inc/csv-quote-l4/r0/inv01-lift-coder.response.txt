    Quote,

    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    if pos == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = Vec::new();
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_start = 0usize;
    let mut record_index = 0usize;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if !in_quotes && b == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: pos,
            });
        }

        if in_quotes {
            if b == quote {
                if bytes.get(pos + 1) == Some(&quote) {
                    field.push(quote);
                    pos += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    pos += 1;
                }
            } else {
                field.push(b);
                pos += 1;
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                record.push(String::from_utf8(field).unwrap());
                field = Vec::new();
                after_quote = false;
                pos += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                record.push(String::from_utf8(field).unwrap());
                records.push(record);
                record = Vec::new();
                field = Vec::new();
                after_quote = false;
                if b == b'\r' {
                    pos += 2;
                } else {
                    pos += 1;
                }
                record_index += 1;
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: pos,
            });
        }

        if b == quote {
            if field.is_empty() {
                in_quotes = true;
                quote_start = pos;
                pos += 1;
            } else {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: pos,
                });
            }
        } else if b == delimiter {
            record.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            pos += 1;
        } else if b == b'\n' || b == b'\r' {
            record.push(String::from_utf8(field).unwrap());
            records.push(record);
            record = Vec::new();
            field = Vec::new();
            if b == b'\r' {
                pos += 2;
            } else {
                pos += 1;
            }
            record_index += 1;
        } else {
            field.push(b);
            pos += 1;
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if after_quote || !field.is_empty() || !record.is_empty() || bytes.last() == Some(&delimiter) {
        record.push(String::from_utf8(field).unwrap());
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|n| n == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|n| n == name)
            }
        }
    }
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    for second in 0..header.len() {
        if let Some(first) = header[..second].iter().position(|n| n == &header[second]) {
            return Some(CsvError::DuplicateHeader {
                name: header[second].clone(),
                first,
                second,
            });
        }
    }
    None
}

fn table_from_parsed(
    parsed: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if parsed.is_empty() {
            (None, Vec::new())
        } else {
            let mut iter = parsed.into_iter();
            (iter.next(), iter.collect::<Vec<_>>())
        }
    } else {
        (None, parsed)
    };

    if let Some(ref header) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header(header) {
                return Err(error);
            }
        }
        for (i, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|parsed| table_from_parsed(parsed, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    if !(b'1'..=b'9').contains(&b[start]) {
        return false;
    }
    b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if start >= b.len() {
        return false;
    }
    let dot = match b[start..].iter().position(|&x| x == b'.') {
        Some(i) => start + i,
        None => return false,
    };
    if b.len() != dot + 3 {
        return false;
    }
    let integral = &b[start..dot];
    if integral.is_empty() || !b[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    if integral[0] == b'0' {
        if integral.len() != 1 {
            return false;
        }
    } else if !(b'1'..=b'9').contains(&integral[0])
        || !integral[1..].iter().all(u8::is_ascii_digit)
    {
        return false;
    }
    !(start == 1 && integral == b"0" && &b[dot + 1..] == b"00")
}

fn evaluate_cell(
    text: &str,
    spec: &ColumnSpec,
    row: usize,
    spec_index: usize,
    column: Option<usize>,
) -> Result<Value, CellError> {
    let fail = |kind| CellError {
        row,
        spec: spec_index,
        column,
        kind,
    };

    if spec.nullable && text.is_empty() {
        return Ok(Value::Null);
    }
    if text.is_empty() && spec.kind != ColumnType::Text {
        return Err(fail(CellErrorKind::Empty));
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(text.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(text) {
                return Err(fail(CellErrorKind::InvalidInteger));
            }
            text.parse::<i64>()
                .map(Value::Integer)
                .map_err(|_| fail(CellErrorKind::IntegerOverflow))
        }
        ColumnType::Decimal => {
            if !decimal_syntax(text) {
                return Err(fail(CellErrorKind::InvalidDecimal));
            }
            let negative = text.starts_with('-');
            let unsigned = if negative { &text[1..] } else { text };
            let mut parts = unsigned.split('.');
            let whole = parts.next().unwrap_or("");
            let fraction = parts.next().unwrap_or("");
            let magnitude_text = format!("{}{}", whole, fraction);
            let magnitude = magnitude_text
                .parse::<i128>()
                .map_err(|_| fail(CellErrorKind::DecimalOverflow))?;
            let signed = if negative { -magnitude } else { magnitude };
            i64::try_from(signed)
                .map(Value::Decimal)
                .map_err(|_| fail(CellErrorKind::DecimalOverflow))
        }
        ColumnType::Boolean => match text {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(fail(CellErrorKind::InvalidBoolean)),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };
    let mut stop = false;

    for (row_index, row) in table.rows.iter().enumerate() {
        if stop {
            break;
        }
        let mut output = Vec::with_capacity(columns.len());
        let mut failed_this_row = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if failed_this_row {
                output.push(None);
                continue;
            }

            let index = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let index = match index {
                Some(index) => index,
                None => {
                    output.push(None);
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: None,
                        kind: CellErrorKind::UnknownColumn,
                    });
                    if policy != ErrorPolicy::Collect {
                        failed_this_row = true;
                    }
                    if policy == ErrorPolicy::Stop {
                        stop = true;
                    }
                    continue;
                }
            };

            let text = match row.get(index) {
                Some(text) => text,
                None => {
                    output.push(None);
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(index),
                        kind: CellErrorKind::MissingCell,
                    });
                    if policy != ErrorPolicy::Collect {
                        failed_this_row = true;
                    }
                    if policy == ErrorPolicy::Stop {
                        stop = true;
                    }
                    continue;
                }
            };

            match evaluate_cell(text, spec, row_index, spec_index, Some(index)) {
                Ok(value) => output.push(Some(value)),
                Err(error) => {
                    output.push(None);
                    result.errors.push(error);
                    if policy != ErrorPolicy::Collect {
                        failed_this_row = true;
                    }
                    if policy == ErrorPolicy::Stop {
                        stop = true;
                    }
                }
            }
        }
        result.rows.push(output);
    }

    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original_records: Vec<(usize, &Vec<String>)> = Vec::new();
    if let Some(header) = &table.header {
        original_records.push((0, header));
    }
    let row_offset = usize::from(table.header.is_some());
    for (i, row) in table.rows.iter().enumerate() {
        original_records.push((i + row_offset, row));
    }

    for (record, fields) in &original_records {
        for (column, field) in fields.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record: *record,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, fields) in &original_records {
        if fields.is_empty() {
            return Err(WriteError::ZeroColumns { record: *record });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let (Some(header), DuplicatePolicy::Reject) = (&table.header, table.duplicates) {
            if let Some(CsvError::DuplicateHeader { name, first, second }) =
                duplicate_header(header)
            {
                return Err(WriteError::DuplicateHeader {
                    name,
                    first,
                    second,
                });
            }
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();

    if config.mode == WriteMode::Preserve {
        if let Some(header) = &table.header {
            output_records.push(header.clone());
        }
        output_records.extend(table.rows.iter().cloned());
    } else if let Some(header) = &table.header {
        let mut retained = Vec::new();
        for (i, name) in header.iter().enumerate() {
            let keep = match table.duplicates {
                DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                    !header[..i].iter().any(|n| n == name)
                }
                DuplicatePolicy::KeepLast => !header[i + 1..].iter().any(|n| n == name),
            };
            if keep {
                retained.push((name.clone(), i));
            }
        }
        retained.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));

        output_records.push(retained.iter().map(|(name, _)| name.clone()).collect());
        let mut rows: Vec<Vec<String>> = table
            .rows
            .iter()
            .map(|row| retained.iter().map(|(_, i)| row[*i].clone()).collect())
            .collect();
        rows.sort_by(|a, b| compare_rows(a, b));
        output_records.extend(rows);
    } else {
        let mut rows = table.rows.clone();
        rows.sort_by(|a, b| compare_rows(a, b));
        output_records = rows;
    }

    if output_records.is_empty() {
        return Ok(String::new());
    }

    let canonical = config.mode == WriteMode::Canonical;
    let separator = if canonical {
        "\n"
    } else {
        match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        }
    };
    let mut output = String::new();
    for (record_index, record) in output_records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(separator);
        }
        for (column_index, field) in record.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter as char);
            }
            let starts_bom = record_index == 0
                && column_index == 0
                && field.starts_with('\u{feff}');
            let sole_empty = record.len() == 1 && field.is_empty();
            let bytes = field.as_bytes();
            let needs_quote = bytes.contains(&config.delimiter)
                || bytes.contains(&config.quote)
                || bytes.contains(&b'\r')
                || bytes.contains(&b'\n')
                || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
                || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
                || starts_bom
                || sole_empty;
            if needs_quote {
                output.push(config.quote as char);
                for ch in field.chars() {
                    if ch == config.quote as char {
                        output.push(ch);
                        output.push(ch);
                    } else {
                        output.push(ch);
                    }
                }
                output.push(config.quote as char);
            } else {
                output.push_str(field);
            }
        }
    }

    let final_terminator = canonical || config.final_terminator;
    if final_terminator {
        output.push_str(separator);
    }

    Ok(output)
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        let order = left.as_bytes().cmp(right.as_bytes());
        if order != std::cmp::Ordering::Equal {
            return order;
        }
    }
    a.len().cmp(&b.len())
}
