#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_prefix = bytes.first() == Some(&b'"');
    let closing_quote = if quoted_prefix {
        let mut i = 1;
        let mut found = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    found = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match found {
            Some(i) => Some(i),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let protected_end = closing_quote.map_or(0, |i| i + 1);
    for i in 0..bytes.len() {
        if bytes[i] == b'@' && (!quoted_prefix || i >= protected_end) {
            separators.push(i);
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let separator = separators[0];
    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local = if quoted_prefix {
        let close = closing_quote.expect("quoted prefix has a closing quote");
        if close + 1 != separator {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < close {
            if bytes[i] == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape { at: i });
                }
                let escaped = bytes[i + 1];
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: i });
                }
                decoded.push(escaped);
                origins.push(i);
                i += 2;
            } else {
                let b = bytes[i];
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: i });
                }
                decoded.push(b);
                origins.push(i);
                i += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).expect("ASCII bytes"), true)
    } else {
        if separator > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for i in 0..separator {
            let b = bytes[i];
            if b == b'.' {
                if i == 0 || i + 1 == separator || bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: i });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: i });
            }
        }

        (String::from_utf8(bytes[..separator].to_vec()).expect("ASCII bytes"), false)
    };

    let domain_start = separator + 1;
    let mut domain_end = bytes.len();
    let removed_root_dot = if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain_bytes = &bytes[domain_start..domain_end];
    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = 0;

    for i in 0..=domain_bytes.len() {
        if i == domain_bytes.len() || domain_bytes[i] == b'.' {
            let label_end = i;
            if label_start == label_end {
                let at = if label_start == domain_bytes.len() {
                    removed_root_dot.unwrap_or(domain_end)
                } else {
                    domain_start + label_start
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let label_len = label_end - label_start;
            if label_len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: domain_start + label_start + 63,
                });
            }

            if domain_bytes[label_start] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: domain_start + label_start,
                });
            }
            if domain_bytes[label_end - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: domain_start + label_end - 1,
                });
            }

            for j in label_start..label_end {
                let b = domain_bytes[j];
                if !b.is_ascii_alphanumeric() && b != b'-' {
                    return Err(EmailError::DomainChar {
                        at: domain_start + j,
                    });
                }
            }

            labels.push((label_start, label_end));
            label_start = i + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = *labels.last().expect("validated labels are nonempty");
    if domain_bytes[last_start..last_end]
        .iter()
        .all(u8::is_ascii_digit)
    {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let normalized_domain: Vec<u8> = domain_bytes
        .iter()
        .map(u8::to_ascii_lowercase)
        .collect();

    Ok(Address {
        local: local.0,
        domain: String::from_utf8(normalized_domain).expect("ASCII bytes"),
        quoted: local.1,
    })
}
