#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_close = if bytes[0] == b'"' {
        let mut i = 1;
        let mut close = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    close = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match close {
            Some(i) => Some(i),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let protected_through = quoted_close.unwrap_or(0);
    for i in 0..bytes.len() {
        if bytes[i] == b'@' && !(quoted_close.is_some() && i <= protected_through) {
            separators.push(i);
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let (local, is_quoted) = if let Some(close) = quoted_close {
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < close {
            if bytes[i] == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape { at: i });
                }
                let escaped = bytes[i + 1];
                match escaped {
                    b'"' | b'\\' | b'@' => {
                        decoded.push(escaped);
                        origins.push(i);
                        i += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: i }),
                }
            } else {
                if !(0x20..=0x7e).contains(&bytes[i]) {
                    return Err(EmailError::QuotedChar { at: i });
                }
                decoded.push(bytes[i]);
                origins.push(i);
                i += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        let local_bytes = &bytes[..sep];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        let specials = b"!#$%&'*+-/=?^_`{|}~";
        for i in 0..local_bytes.len() {
            let b = local_bytes[i];
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || local_bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: i });
                }
            } else if !b.is_ascii_alphanumeric() && !specials.contains(&b) {
                return Err(EmailError::LocalChar { at: i });
            }
        }
        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start = sep + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(has_root_dot);

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain = &raw_domain[..domain_len];
    let mut label_starts = Vec::new();
    let mut label_ends = Vec::new();
    let mut start = 0;

    for i in 0..=domain.len() {
        if i == domain.len() || domain[i] == b'.' {
            if i == start {
                let at = if i == domain.len() && has_root_dot {
                    input.len() - 1
                } else {
                    domain_start + start
                };
                return Err(EmailError::EmptyLabel { at });
            }
            label_starts.push(start);
            label_ends.push(i);
            start = i + 1;
        }
    }

    for (&label_start, &label_end) in label_starts.iter().zip(label_ends.iter()) {
        let len = label_end - label_start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: domain_start + label_start + 63,
            });
        }

        if domain[label_start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + label_start,
            });
        }
        if domain[label_end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + label_end - 1,
            });
        }

        for i in label_start..label_end {
            let b = domain[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: domain_start + i,
                });
            }
        }
    }

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let last_start = *label_starts.last().unwrap();
    let last_end = *label_ends.last().unwrap();
    if domain[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let lower_domain = domain
        .iter()
        .map(u8::to_ascii_lowercase)
        .collect::<Vec<_>>();
    Ok(Address {
        local,
        domain: String::from_utf8(lower_domain).unwrap(),
        quoted: is_quoted,
    })
}
