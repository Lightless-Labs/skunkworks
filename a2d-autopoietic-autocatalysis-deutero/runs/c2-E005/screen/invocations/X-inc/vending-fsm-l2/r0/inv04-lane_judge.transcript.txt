    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();

        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product
                    .id
                    .bytes()
                    .all(|byte| byte.is_ascii_lowercase());

            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }

            if seen.iter().any(|id| *id == product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),

                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let denomination_index = DENOMINATIONS.iter().position(|&d| d == coin);
        if denomination_index.is_none() {
            return Err(Error::BadCoin { coin });
        }

        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let product_exists = self.products.iter().any(|product| product.id == id);
        if !product_exists {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }

        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = match self.selected.as_ref() {
            Some(id) => id,
            None => return Err(Error::NoSelection),
        };

        let product = match self.products.iter().find(|p| p.id == *selected_id) {
            Some(product) => product,
            None => return Err(Error::NoSelection),
        };

        let id = product.id.clone();
        let price = product.price;

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = match choose_change(amount, &self.tubes) {
            Some(change) => change,
            None => return Err(Error::NoChange { amount }),
        };

        let mut proposed_tubes = self.tubes;

            let dispensed = change
                .iter()
                .filter(|&&coin| coin == *denomination)
                .count() as u16;
            proposed_tubes[index] -= dispensed;
        }

        for &coin in &self.escrow {
            let index = DENOMINATIONS
                .iter()
                .position(|&denomination| denomination == coin)
                .expect("escrow only contains accepted denominations");
            proposed_tubes[index] += 1;
        }

        for (index, count) in proposed_tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let refunded: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(refunded)
    }
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let descending_indices = [3usize, 2, 1, 0];
    let mut greedy_counts = [0u16; 4];
    let mut remaining = amount;

    for &index in &descending_indices {
        let denomination = DENOMINATIONS[index];
        let count = (remaining / denomination).min(tubes[index]);
        greedy_counts[index] = count;
        remaining -= count * denomination;
    }

    if remaining == 0 {
        return Some(counts_to_descending_coins(&greedy_counts));
    }

    let target = amount as usize;
    let mut best: Vec<Option<[u16; 4]>> = vec![None; target + 1];
    best[0] = Some([0; 4]);

    for value in 0..=target {
        let current = match best[value] {
            Some(counts) => counts,
            None => continue,
        };

        for index in 0..4 {
            let denomination = DENOMINATIONS[index] as usize;
            let next_value = value + denomination;
            if next_value > target || current[index] >= tubes[index] {
                continue;
            }

            let mut candidate = current;
            candidate[index] += 1;

            let replace = match best[next_value] {
                None => true,
                Some(existing) => {
                    let candidate_total: u16 = candidate.iter().sum();
                    let existing_total: u16 = existing.iter().sum();

                    candidate_total < existing_total
                        || (candidate_total == existing_total
                            && lexicographically_greater_descending(&candidate, &existing))
                }
            };

            if replace {
                best[next_value] = Some(candidate);
            }
        }
    }

    best[target].map(|counts| counts_to_descending_coins(&counts))
}

fn lexicographically_greater_descending(left: &[u16; 4], right: &[u16; 4]) -> bool {
    for index in [3usize, 2, 1, 0] {
        if left[index] != right[index] {
            return left[index] > right[index];
        }
    }
    false
}

fn counts_to_descending_coins(counts: &[u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for index in [3usize, 2, 1, 0] {
        for _ in 0..counts[index] {
            coins.push(DENOMINATIONS[index]);
        }
    }
    coins
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    if amount == 0 {
        return Some([0; 4]);
    }

    let mut greedy = [0u16; 4];
    let mut remainder = amount;

    for index in (0..DENOMINATIONS.len()).rev() {
        let coin = DENOMINATIONS[index];
        let take = tubes[index].min(remainder / coin);
        greedy[index] = take;
        remainder -= take * coin;
    }

    if remainder == 0 {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u32::MAX;
    search_change(
        3,
        amount,
        tubes,
        [0; 4],
        0,
        &mut best,
        &mut best_count,
    );
    best
}

fn search_change(
    index: isize,
    remainder: u16,
    tubes: &[u16; 4],
    current: [u16; 4],
    coin_count: u32,
    best: &mut Option<[u16; 4]>,
    best_count: &mut u32,
) {
    if remainder == 0 {
        let replace = match *best {
            None => true,
            Some(previous) => {
                coin_count < *best_count
                    || (coin_count == *best_count
                        && lexicographically_greater_descending(&current, &previous))
            }
        };
        if replace {
            *best = Some(current);
            *best_count = coin_count;
        }
        return;
    }

    if index < 0 || coin_count >= *best_count {
        return;
    }

    let index = index as usize;
    let coin = DENOMINATIONS[index];
    let maximum = tubes[index].min(remainder / coin);

    for take in (0..=maximum).rev() {
        let next_count = coin_count + u32::from(take);
        if next_count > *best_count {
            continue;
        }

        let mut next = current;
        next[index] = take;
        search_change(
            index as isize - 1,
            remainder - take * coin,
            tubes,
            next,
            next_count,
            best,
            best_count,
        );
    }
}

fn lexicographically_greater_descending(left: &[u16; 4], right: &[u16; 4]) -> bool {
    for index in (0..4).rev() {
        if left[index] != right[index] {
            return left[index] > right[index];
        }
    }
    false
}

                return Err(Error::BadId { id: product.id });
            }
            if catalog.contains_key(&product.id) {
                return Err(Error::DuplicateId { id: product.id });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice { id: product.id });
            }
            catalog.insert(product.id, product.price);
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products: catalog,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.contains_key(id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = self.selected.as_ref().ok_or(Error::NoSelection)?;
        let price = *self.products.get(id).ok_or(Error::NoSelection)?;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = choose_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for coin in &change {
            let index = denomination_index(*coin);
            proposed_tubes[index] -= 1;
        }
        for coin in &self.escrow {
            let index = denomination_index(*coin);
            proposed_tubes[index] += 1;
        }
        for (index, count) in proposed_tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        let receipt = Receipt {
            id: id.clone(),
            price,
            change,
        };
        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }
}

fn denomination_index(coin: u16) -> usize {
    match coin {
        1 => 0,
        3 => 1,
        4 => 2,
        10 => 3,
        _ => unreachable!(),
    }
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut remaining = amount;
    let mut greedy_counts = [0u16; 4];
    for index in (0..4).rev() {
        let coin = DENOMINATIONS[index];
        let take = tubes[index].min(remaining / coin);
        greedy_counts[index] = take;
        remaining -= take * coin;
    }
    if remaining == 0 {
        return Some(counts_to_descending_vec(&greedy_counts));
    }

    let target = usize::from(amount);
    let mut best: Vec<Option<[u16; 4]>> = vec![None; target + 1];
    best[0] = Some([0; 4]);

    for value in 0..=target {
        let Some(current) = best[value] else {
            continue;
        };
        for index in 0..4 {
            let coin = usize::from(DENOMINATIONS[index]);
            if current[index] >= tubes[index] || value + coin > target {
                continue;
            }
            let mut candidate = current;
            candidate[index] += 1;
            let next_value = value + coin;
            let replace = match best[next_value] {
                None => true,
                Some(existing) => better_combination(&candidate, &existing),
            };
            if replace {
                best[next_value] = Some(candidate);
            }
        }
    }

    best[target].map(|counts| counts_to_descending_vec(&counts))
}

fn better_combination(candidate: &[u16; 4], existing: &[u16; 4]) -> bool {
    let candidate_total: u32 = candidate.iter().map(|&n| u32::from(n)).sum();
    let existing_total: u32 = existing.iter().map(|&n| u32::from(n)).sum();

    if candidate_total != existing_total {
        return candidate_total < existing_total;
    }

    for index in (0..4).rev() {
        if candidate[index] != existing[index] {
            return candidate[index] > existing[index];
        }
    }
    false
}

fn counts_to_descending_vec(counts: &[u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for index in (0..4).rev() {
        for _ in 0..counts[index] {
            coins.push(DENOMINATIONS[index]);
        }
    }
    coins
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let descending_indices = [3usize, 2, 1, 0];
    let mut remaining = amount;
    let mut greedy_counts = [0u16; 4];

    for &index in &descending_indices {
        let denomination = DENOMINATIONS[index];
        let take = tubes[index].min(remaining / denomination);
        greedy_counts[index] = take;
        remaining -= take * denomination;

    if remaining == 0 {
        return Some(counts_to_descending_coins(&greedy_counts));
    }

        0,
        amount,
        tubes,
        &mut current,
        &mut best,
    );

}

fn search_change(
    position: usize,
    remaining: u16,
    tubes: &[u16; 4],
    current: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if position == 4 {
        if remaining == 0 && is_better_change(current, best.as_ref()) {
            *best = Some(*current);
        }
        return;
    }

    let index = [3usize, 2, 1, 0][position];
    let denomination = DENOMINATIONS[index];
    let max_count = tubes[index].min(remaining / denomination);

    for count in (0..=max_count).rev() {
        current[index] = count;
        search_change(
            position + 1,
            remaining - count * denomination,
            tubes,
            current,
            best,
        );
    }
    current[index] = 0;
}

fn is_better_change(candidate: &[u16; 4], current_best: Option<&[u16; 4]>) -> bool {
    let Some(best) = current_best else {
        return true;
    };

    let candidate_total: u32 = candidate.iter().map(|&n| u32::from(n)).sum();
    let best_total: u32 = best.iter().map(|&n| u32::from(n)).sum();

    if candidate_total != best_total {
        return candidate_total < best_total;
    }

    for index in [3usize, 2, 1, 0] {
        if candidate[index] != best[index] {
            return candidate[index] > best[index];
        }
    }

    false
}

        }
    }
    coins
