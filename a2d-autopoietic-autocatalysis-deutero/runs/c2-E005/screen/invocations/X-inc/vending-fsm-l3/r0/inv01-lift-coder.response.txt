```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    events: Vec<Event>,
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = std::collections::HashSet::new();
        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if !seen.insert(product.id.clone()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for (i, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            events: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        let next = u32::from(self.credit) + u32::from(coin);
        if next > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit = next as u16;
        self.escrow.push(coin);
        self.log(EventKind::Inserted {
            coin,
            credit: self.credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let index = self.find_product(id).ok_or_else(|| Error::UnknownProduct {
            id: id.to_owned(),
        })?;
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        let selected_id = self.products[index].id.clone();
        self.selected = Some(selected_id.clone());
        self.log(EventKind::Selected { id: selected_id });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = self.selected.clone().ok_or(Error::NoSelection)?;
        let index = self
            .find_product(&id)
            .expect("selected product must be in the catalog");

        let product = &self.products[index];
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = Self::make_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for (i, &coin) in DENOMINATIONS.iter().enumerate() {
            let used = change.iter().filter(|&&c| c == coin).count() as u16;
            proposed_tubes[i] -= used;
        }
        for &coin in &self.escrow {
            let i = DENOMINATIONS
                .iter()
                .position(|&d| d == coin)
                .expect("escrow contains only accepted denominations");
            let Some(count) = proposed_tubes[i].checked_add(1) else {
                return Err(Error::TubeLimit { coin });
            };
            if count > 100 {
                return Err(Error::TubeLimit { coin });
            }
            proposed_tubes[i] = count;
        }

        self.tubes = proposed_tubes;
        self.products[index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        let receipt = Receipt {
            id: id.clone(),
            price,
            change: change.clone(),
        };
        self.log(EventKind::Vended { id, price, change });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let index = self.find_product(id).ok_or_else(|| Error::UnknownProduct {
            id: id.to_owned(),
        })?;
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let index = self.find_product(id).ok_or_else(|| Error::UnknownProduct {
            id: id.to_owned(),
        })?;
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let proposed = u32::from(self.products[index].stock) + u32::from(count);
        if proposed > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        self.products[index].stock = proposed as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: proposed as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    fn find_product(&self, id: &str) -> Option<usize> {
        self.products.iter().position(|p| p.id == id)
    }

    fn log(&mut self, kind: EventKind) {
        self.events.push(Event {
            seq: self.events.len(),
            kind,
        });
    }

    fn make_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
        if amount == 0 {
            return Some(Vec::new());
        }

        let mut remaining = amount;
        let mut greedy = Vec::new();
        for i in (0..4).rev() {
            let coin = DENOMINATIONS[i];
            let take = tubes[i].min(remaining / coin);
            greedy.extend(std::iter::repeat_n(coin, take as usize));
            remaining -= take * coin;
        }
        if remaining == 0 {
            return Some(greedy);
        }

        let mut best: Option<Vec<u16>> = None;
        let mut counts = [0u16; 4];
        Self::search_change(3, amount, tubes, &mut counts, &mut best);
        best
    }

    fn search_change(
        index: isize,
        remaining: u16,
        tubes: &[u16; 4],
        counts: &mut [u16; 4],
        best: &mut Option<Vec<u16>>,
    ) {
        if remaining == 0 {
            let mut candidate = Vec::new();
            for i in (0..4).rev() {
                candidate.extend(std::iter::repeat_n(
                    DENOMINATIONS[i],
                    counts[i] as usize,
                ));
            }
            let replace = match best {
                None => true,
                Some(current) => {
                    candidate.len() < current.len()
                        || (candidate.len() == current.len()
                            && candidate.as_slice() > current.as_slice())
                }
            };
            if replace {
                *best = Some(candidate);
            }
            return;
        }
        if index < 0 {
            return;
        }

        let i = index as usize;
        let coin = DENOMINATIONS[i];
        let max_take = tubes[i].min(remaining / coin);
        for take in (0..=max_take).rev() {
            let next_remaining = remaining - take * coin;
            if let Some(current) = best.as_ref() {
                let used: usize = counts.iter().map(|&n| n as usize).sum();
                if used + take as usize > current.len() {
                    continue;
                }
            }
            counts[i] = take;
            Self::search_change(index - 1, next_remaining, tubes, counts, best);
        }
        counts[i] = 0;
    }
}
```
