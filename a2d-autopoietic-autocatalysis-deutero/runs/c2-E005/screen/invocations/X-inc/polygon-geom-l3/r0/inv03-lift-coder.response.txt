#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective[0] == *effective.last().unwrap() {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let acx = c.x as i128 - a.x as i128;
    let acy = c.y as i128 - a.y as i128;
    abx * acy - aby * acx
}

fn signed_area2(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn edge_length_milli(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let scaled_squared = (dx * dx + dy * dy) as u128 * 1_000_000u128;

    let mut lo = 0u128;
    let mut hi = 3_000_000_000_000_000u128;
    while lo < hi {
        let mid = lo + (hi - lo + 1) / 2;
        if mid <= scaled_squared / mid.max(1) && mid * mid <= scaled_squared {
            lo = mid;
        } else {
            hi = mid - 1;
        }
    }

    let floor = lo;
    let remainder = scaled_squared - floor * floor;
    let threshold = 2 * floor + 1;
    if 4 * remainder >= threshold * threshold {
        (floor + 1) as u64
    } else {
        floor as u64
    }
}

fn perimeter_and_edges(points: &[Point]) -> (Vec<u64>, u64, Option<usize>) {
    if points.is_empty() {
        return (Vec::new(), 0, None);
    }

    let mut edge_milli = Vec::with_capacity(points.len());
    let mut perimeter = 0u64;
    let mut longest: Option<usize> = None;

    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        let length = edge_length_milli(a, b);
        edge_milli.push(length);
        perimeter += length;

        let endpoints = if a <= b { (a, b) } else { (b, a) };
        if let Some(current) = longest {
            let current_length = edge_milli[current];
            let ca = points[current];
            let cb = points[(current + 1) % points.len()];
            let current_endpoints = if ca <= cb { (ca, cb) } else { (cb, ca) };
            if length > current_length
                || (length == current_length
                    && (endpoints < current_endpoints
                        || (endpoints == current_endpoints && i < current)))
            {
                longest = Some(i);
            }
        } else {
            longest = Some(i);
        }
    }

    (edge_milli, perimeter, longest)
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepare_polygon(points)?;
    let signed = signed_area2(&polygon);
    let orientation = if signed > 0 {
        Orientation::CounterClockwise
    } else if signed < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };
    let (edge_milli, perimeter_milli, longest_edge) = perimeter_and_edges(&polygon);

    Ok(Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

fn locate(polygon: &[Point], q: Point) -> Location {
    for i in 0..polygon.len() {
        if on_segment(polygon[i], polygon[(i + 1) % polygon.len()], q) {
            return Location::Boundary { edge: i };
        }
    }

    let mut odd = false;
    for i in 0..polygon.len() {
        let a = polygon[i];
        let b = polygon[(i + 1) % polygon.len()];
        if (a.y > q.y) != (b.y > q.y) {
            let c = cross(a, b, q);
            if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                odd = !odd;
            }
        }
    }

    if odd {
        Location::Inside
    } else {
        Location::Outside
    }
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }
    Ok(queries.iter().map(|&q| locate(&polygon, q)).collect())
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut sorted: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if let Some(existing) = sorted.iter_mut().find(|entry| entry.0 == point) {
            existing.1 = existing.1.min(index);
        } else {
            sorted.push((point, index));
        }
    }
    sorted.sort_by_key(|entry| entry.0);

    if sorted.is_empty() {
        return Ok(Hull {
            vertices: Vec::new(),
            source_indices: Vec::new(),
            area2: 0,
            perimeter_milli: 0,
        });
    }

    if sorted.len() == 1 {
        return Ok(Hull {
            vertices: vec![sorted[0].0],
            source_indices: vec![sorted[0].1],
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let all_collinear = sorted
        .iter()
        .skip(2)
        .all(|entry| cross(sorted[0].0, sorted[1].0, entry.0) == 0);

    let hull_entries: Vec<(Point, usize)> = if all_collinear {
        match mode {
            Collinear::Include => sorted,
            Collinear::Exclude => vec![sorted[0], *sorted.last().unwrap()],
        }
    } else {
        let mut lower = Vec::new();
        for &entry in &sorted {
            while lower.len() >= 2 {
                let turn = cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, entry.0);
                if turn > 0 || (mode == Collinear::Include && turn == 0) {
                    break;
                }
                lower.pop();
            }
            lower.push(entry);
        }

        let mut upper = Vec::new();
        for &entry in sorted.iter().rev() {
            while upper.len() >= 2 {
                let turn = cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, entry.0);
                if turn > 0 || (mode == Collinear::Include && turn == 0) {
                    break;
                }
                upper.pop();
            }
            upper.push(entry);
        }

        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower
    };

    let vertices: Vec<Point> = hull_entries.iter().map(|entry| entry.0).collect();
    let source_indices: Vec<usize> = hull_entries.iter().map(|entry| entry.1).collect();
    let area = signed_area2(&vertices).unsigned_abs();
    let (_, perimeter_milli, _) = perimeter_and_edges(&vertices);

    Ok(Hull {
        vertices,
        source_indices,
        area2: area,
        perimeter_milli,
    })
}
