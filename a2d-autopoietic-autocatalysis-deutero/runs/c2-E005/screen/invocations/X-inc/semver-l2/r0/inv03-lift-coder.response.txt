use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_terminator(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    if *pos >= bytes.len() || !bytes[*pos].is_ascii_digit() {
        return if is_core_terminator(bytes.get(*pos).copied()) {
            Err(VersionError::MissingCore { at: *pos })
        } else {
            Err(VersionError::InvalidChar { at: *pos })
        };
    }

    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    let field = &s[start..*pos];
    if field.len() > 1 && field.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    field
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    numeric_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut id_start = start;

    loop {
        let mut i = id_start;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }

        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..i];
        if numeric_pre
            && id.len() > 1
            && id.as_bytes()[0] == b'0'
            && id.bytes().all(|b| b.is_ascii_digit())
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(id.to_string());

        if i == end {
            break;
        }
        id_start = i + 1;
        if id_start > end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    for _ in 0..2 {
        match s.as_bytes().get(pos).copied() {
            Some(b'.') => pos += 1,
            Some(b'-' | b'+') | None => return Err(VersionError::MissingCore { at: pos }),
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
        let value = parse_core_number(s, &mut pos)?;
        if _ == 0 {
            // Kept below in the ordered tuple.
            let _ = value;
        }
    }
    // Parse explicitly to retain the two remaining fields while preserving
    // the required field-by-field error ordering.
    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    if s.as_bytes().get(pos) != Some(&b'.') {
        return match s.as_bytes().get(pos).copied() {
            Some(b'-' | b'+') | None => Err(VersionError::MissingCore { at: pos }),
            _ => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;
    let minor = parse_core_number(s, &mut pos)?;
    if s.as_bytes().get(pos) != Some(&b'.') {
        return match s.as_bytes().get(pos).copied() {
            Some(b'-' | b'+') | None => Err(VersionError::MissingCore { at: pos }),
            _ => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;
    let patch = parse_core_number(s, &mut pos)?;

    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = bytes[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|x| pre_start + x)
            .unwrap_or(bytes.len());
        pre = parse_identifiers(s, pre_start, plus, true)?;
        pos = plus;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, bytes.len(), false)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric_identifier(x);
        let yn = is_numeric_identifier(y);
        let ord = match (xn, yn) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Comparator {
    Eq(Version),
    Lt(Version),
    Le(Version),
    Gt(Version),
    Ge(Version),
}

impl Comparator {
    fn test(&self, candidate: &Version) -> bool {
        let (op, version) = match self {
            Comparator::Eq(v) => (0, v),
            Comparator::Lt(v) => (1, v),
            Comparator::Le(v) => (2, v),
            Comparator::Gt(v) => (3, v),
            Comparator::Ge(v) => (4, v),
        };
        let ord = compare_versions(candidate, version);
        match op {
            0 => ord == Ordering::Equal,
            1 => ord == Ordering::Less,
            2 => ord != Ordering::Greater,
            3 => ord == Ordering::Greater,
            _ => ord != Ordering::Less,
        }
    }
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

impl Arm {
    fn matches(&self, candidate: &Version) -> bool {
        if !candidate.pre.is_empty()
            && !self
                .anchors
                .iter()
                .any(|&core| core == (candidate.major, candidate.minor, candidate.patch))
        {
            return false;
        }
        self.comparators.iter().all(|c| c.test(candidate))
    }
}

fn is_range_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens_with_offsets(s: &str, start: usize, end: usize) -> Vec<(usize, &str)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && is_range_ws(bytes[i]) {
            i += 1;
        }
        if i >= end {
            break;
        }
        let token_start = i;
        while i < end && !is_range_ws(bytes[i]) {
            i += 1;
        }
        out.push((token_start, &s[token_start..i]));
    }
    out
}

fn parse_full_at(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

#[derive(Clone)]
struct Partial {
    fields: Vec<Option<u64>>,
    full: Option<Version>,
}

impl Partial {
    fn numeric_count(&self) -> usize {
        self.fields.iter().take_while(|x| x.is_some()).count()
    }

    fn padded(&self) -> Version {
        Version {
            major: self.fields.first().and_then(|x| *x).unwrap_or(0),
            minor: self.fields.get(1).and_then(|x| *x).unwrap_or(0),
            patch: self.fields.get(2).and_then(|x| *x).unwrap_or(0),
            pre: self.full.as_ref().map(|v| v.pre.clone()).unwrap_or_default(),
            build: self.full.as_ref().map(|v| v.build.clone()).unwrap_or_default(),
        }
    }
}

fn parse_partial(token: &str, at: usize) -> Result<Partial, RangeError> {
    if token.contains('-') || token.contains('+') {
        return Ok(Partial {
            fields: Vec::new(),
            full: Some(parse_full_at(token, at)?),
        });
    }

    let fields: Vec<&str> = token.split('.').collect();
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if fields.len() >= 3 && !has_wild {
        return Ok(Partial {
            fields: Vec::new(),
            full: Some(parse_full_at(token, at)?),
        });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }

    let mut parsed = Vec::new();
    let mut wildcard_seen = false;
    for field in fields {
        if field.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(field, "*" | "x" | "X") {
            wildcard_seen = true;
            parsed.push(None);
        } else {
            if wildcard_seen
                || !field.bytes().all(|b| b.is_ascii_digit())
                || (field.len() > 1 && field.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            let value = field.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            parsed.push(Some(value));
        }
    }
    if parsed.is_empty() {
        return Err(RangeError::Syntax { at });
    }
    Ok(Partial {
        fields: parsed,
        full: None,
    })
}

fn zero_pre(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    }
}

fn stable_version(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn increment_bound(p: &Partial, kind: &str, at: usize) -> Result<Version, RangeError> {
    let k = p.numeric_count();
    let lower = p.padded();
    let (mut major, mut minor, mut patch) = (lower.major, lower.minor, lower.patch);
    match kind {
        "bare" => {
            if k == 1 {
                major = major.checked_add(1).ok_or(RangeError::Syntax { at })?;
                minor = 0;
                patch = 0;
            } else {
                minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
                patch = 0;
            }
        }
        "caret" => {
            if major > 0 || k == 1 {
                major = major.checked_add(1).ok_or(RangeError::Syntax { at })?;
                minor = 0;
                patch = 0;
            } else if minor > 0 || k == 2 {
                minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
                patch = 0;
            } else {
                patch = patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
            }
        }
        "tilde" => {
            if k == 1 {
                major = major.checked_add(1).ok_or(RangeError::Syntax { at })?;
                minor = 0;
                patch = 0;
            } else {
                minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
                patch = 0;
            }
        }
        "hyphen" => {
            if k == 1 {
                major = major.checked_add(1).ok_or(RangeError::Syntax { at })?;
                minor = 0;
                patch = 0;
            } else {
                minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
                patch = 0;
            }
        }
        _ => unreachable!(),
    }
    Ok(zero_pre(major, minor, patch))
}

fn add_anchor(arm: &mut Arm, v: &Version) {
    if !v.pre.is_empty() {
        arm.anchors.push((v.major, v.minor, v.patch));
    }
}

fn add_partial_token(
    arm: &mut Arm,
    token: &str,
    at: usize,
    mode: &str,
) -> Result<(), RangeError> {
    let partial = parse_partial(token, at)?;
    let k = partial.numeric_count();
    if mode != "bare" && k == 0 {
        return Err(RangeError::Syntax { at });
    }

    if let Some(full) = partial.full {
        if mode == "bare" {
            arm.comparators.push(Comparator::Eq(full.clone()));
            add_anchor(arm, &full);
            return Ok(());
        }
        let lower = full.clone();
        arm.comparators.push(Comparator::Ge(lower.clone()));
        add_anchor(arm, &lower);
        let p = Partial {
            fields: vec![Some(full.major), Some(full.minor), Some(full.patch)],
            full: Some(full),
        };
        let upper = increment_bound(&p, mode, at)?;
        arm.comparators.push(Comparator::Lt(upper));
        return Ok(());
    }

    if k == 0 {
        return Ok(());
    }
    let lower = partial.padded();
    arm.comparators.push(Comparator::Ge(lower));
    if mode == "bare" || mode == "caret" || mode == "tilde" {
        let upper = increment_bound(&partial, mode, at)?;
        arm.comparators.push(Comparator::Lt(upper));
    }
    Ok(())
}

fn parse_comparator_token(
    arm: &mut Arm,
    token: &str,
    at: usize,
) -> Result<(), RangeError> {
    let (op_len, op) = if token.starts_with(">=") {
        (2, 4)
    } else if token.starts_with("<=") {
        (2, 2)
    } else if token.starts_with('>') {
        (1, 3)
    } else if token.starts_with('<') {
        (1, 1)
    } else if token.starts_with('=') {
        (1, 0)
    } else {
        (0, 0)
    };
    let operand_at = at + op_len;
    let operand = &token[op_len..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    let v = parse_full_at(operand, operand_at)?;
    add_anchor(arm, &v);
    arm.comparators.push(match op {
        1 => Comparator::Lt(v),
        2 => Comparator::Le(v),
        3 => Comparator::Gt(v),
        4 => Comparator::Ge(v),
        _ => Comparator::Eq(v),
    });
    Ok(())
}

fn parse_arm(s: &str, tokens: &[(usize, &str)]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };

    let standalone_minus: Vec<usize> = tokens
        .iter()
        .filter(|(_, t)| *t == "-")
        .map(|(at, _)| *at)
        .collect();

    if let Some(&minus_at) = standalone_minus.first() {
        if tokens.len() != 3 || tokens[1].1 != "-" {
            return Err(RangeError::Syntax { at: minus_at });
        }
        let (lower_at, lower_token) = tokens[0];
        let (upper_at, upper_token) = tokens[2];
        let lower = parse_partial(lower_token, lower_at)?;
        let upper = parse_partial(upper_token, upper_at)?;
        if lower.numeric_count() == 0 {
            return Err(RangeError::Syntax { at: lower_at });
        }
        if upper.numeric_count() == 0 {
            return Err(RangeError::Syntax { at: upper_at });
        }
        let low_version = lower.padded();
        add_anchor(&mut arm, &low_version);
        arm.comparators.push(Comparator::Ge(low_version));

        if let Some(full) = upper.full {
            add_anchor(&mut arm, &full);
            arm.comparators.push(Comparator::Le(full));
        } else {
            let high = increment_bound(&upper, "hyphen", upper_at)?;
            arm.comparators.push(Comparator::Lt(high));
        }
        return Ok(arm);
    }

    for &(at, token) in tokens {
        if token.starts_with('^') || token.starts_with('~') {
            let mode = if token.starts_with('^') { "caret" } else { "tilde" };
            add_partial_token(&mut arm, &token[1..], at + 1, mode)?;
        } else if token.starts_with('<')
            || token.starts_with('>')
            || token.starts_with('=')
        {
            parse_comparator_token(&mut arm, token, at)?;
        } else if token.starts_with('*')
            || token.starts_with('x')
            || token.starts_with('X')
            || token.as_bytes().first().is_some_and(|b| b.is_ascii_digit())
        {
            add_partial_token(&mut arm, token, at, "bare")?;
        } else {
            return Err(RangeError::Syntax { at });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut separators = Vec::new();
    let mut i = 0;
    while i + 1 < bytes.len() {
        if bytes[i] == b'|' && bytes[i + 1] == b'|' {
            let left_ok = i == 0 || is_range_ws(bytes[i - 1]);
            let right_ok = i + 2 == bytes.len() || is_range_ws(bytes[i + 2]);
            if left_ok && right_ok {
                separators.push(i);
                i += 2;
                continue;
            }
        }
        i += 1;
    }

    let mut bounds = Vec::new();
    let mut start = 0;
    for &sep in &separators {
        bounds.push((start, sep));
        start = sep + 2;
    }
    bounds.push((start, s.len()));

    let mut arms = Vec::new();
    for (arm_start, arm_end) in bounds {
        let tokens = tokens_with_offsets(s, arm_start, arm_end);
        if tokens.is_empty() {
            let at = if arm_start >= s.len() { s.len() } else { arm_start };
            return Err(RangeError::Empty { at });
        }
        arms.push(parse_arm(s, &tokens)?);
    }

    if arms.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| arm.matches(v))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(
    candidates: &[String],
    r: &Range,
    maximum: bool,
) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates
        .iter()
        .enumerate()
        .map(|(index, value)| {
            parse_version(value).map_err(|error| CandidateError { index, error })
        })
        .collect::<Result<_, _>>()?;

    let mut best: Option<(usize, Ordering)> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        if let Some((best_index, _)) = best {
            let ord = compare_versions(version, &parsed[best_index]);
            if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                best = Some((index, ord));
            }
        } else {
            best = Some((index, Ordering::Equal));
        }
    }
    Ok(best.map(|(index, _)| candidates[index].clone()))
}
