#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone, Debug, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone, Debug)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing_slash: bool,
    has_wildcard: bool,
    rank: Vec<u8>,
}

#[derive(Debug)]
pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
    b'0'..=b'9' => Some(b - b'0'),
    b'A'..=b'F' => Some(b - b'A' + 10),
    _ => None,
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || matches!(b, b'_' | b'-'))
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_static_segment(segment: &str, base: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let high = hex_value(bytes[i + 1]).ok_or(base + i)?;
            let low = hex_value(bytes[i + 2]).ok_or(base + i)?;
            let value = (high << 4) | low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing_slash = input.ends_with('/');
    let end = if trailing_slash {
        input.len() - 1
    } else {
        input.len()
    };

    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut start = 1;
    let mut index = 1;

    while index <= end {
        if index == end || bytes[index] == b'/' {
            if index == start {
                return Err(invalid(start));
            }

            let raw = &input[start..index];
            if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
                let marker = raw.as_bytes()[0];
                let name = &raw[1..];
                if !valid_capture_name(name) {
                    return Err(invalid(start));
                }
                if marker == b'*' && index != end {
                    return Err(invalid(start));
                }
                if !names.insert(name.to_owned()) {
                    return Err(invalid(start));
                }
                segments.push(if marker == b':' {
                    Segment::Param(name.to_owned())
                } else {
                    Segment::Wildcard(name.to_owned())
                });
            } else {
                let decoded = parse_static_segment(raw, start).map_err(invalid)?;
                segments.push(Segment::Static(decoded));
            }

            start = index + 1;
        }
        index += 1;
    }

    Ok((segments, trailing_slash))
}

fn path_head(path: &str) -> &str {
    let query = path.find('?').unwrap_or(path.len());
    let fragment = path.find('#').unwrap_or(path.len());
    &path[..query.min(fragment)]
}

fn encode_segment(value: &str) -> String {
    let mut result = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            result.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            result.push('%');
            result.push(HEX[(b >> 4) as usize] as char);
            result.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    result
}

fn conflict_shape(a: &Route, b: &Route) -> bool {
    if a.segments.len() != b.segments.len() {
        return false;
    }
    a.segments.iter().zip(&b.segments).all(|(left, right)| {
        match (left, right) {
            (Segment::Static(x), Segment::Static(y)) => x == y,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        }
    })
}

fn route_matches(route: &Route, segments: &[Segment]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut i = 0;

    for segment in &route.segments {
        match segment {
            Segment::Static(expected) => match segments.get(i) {
                Some(Segment::Static(actual)) if expected == actual => i += 1,
                _ => return None,
            },
            Segment::Param(name) => match segments.get(i) {
                Some(Segment::Static(value)) => {
                    params.push((name.clone(), value.clone()));
                    i += 1;
                }
                _ => return None,
            },
            Segment::Wildcard(name) => {
                let tail = segments[i..]
                    .iter()
                    .map(|segment| match segment {
                        Segment::Static(value) => Some(value.as_str()),
                        _ => None,
                    })
                    .collect::<Option<Vec<_>>>()?
                    .join("/");
                params.push((name.clone(), tail));
                i = segments.len();
            }
        }
    }

    if i == segments.len() {
        Some(params)
    } else {
        None
    }
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing_slash) = parse_path(pattern, true)?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));
        let mut rank = segments
            .iter()
            .map(|segment| match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            })
            .collect::<Vec<_>>();
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        let candidate = Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing_slash,
            has_wildcard,
            rank,
        };

        if let Some(existing) = self.routes.iter().find(|route| {
            route.method == candidate.method
                && conflict_shape(route, &candidate)
                && (self.policy != SlashPolicy::Strict
                    || route.trailing_slash == candidate.trailing_slash)
        }) {
            return Err(Error::Conflict { with: existing.id });
        }

        self.routes.push(candidate);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head = path_head(path);
        let (request_segments, request_trailing) = parse_path(head, false)?;

        let mut path_candidates = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing_slash != request_trailing {
                continue;
            }
            if let Some(params) = route_matches(route, &request_segments) {
                path_candidates.push((route, params));
            }
        }

        if path_candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let head_has_route = method == "HEAD"
            && path_candidates.iter().any(|(route, _)| route.method == "HEAD");
        let effective_method = if method == "HEAD" && !head_has_route {
            "GET"
        } else {
            method
        };

        let mut eligible = path_candidates
            .iter()
            .filter(|(route, _)| route.method == effective_method)
            .collect::<Vec<_>>();

        if eligible.is_empty() {
            let mut allowed = path_candidates
                .iter()
                .map(|(route, _)| route.method.clone())
                .collect::<Vec<_>>();
            if allowed.iter().any(|method| method == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|(a, _), (b, _)| b.rank.cmp(&a.rank).then_with(|| a.id.cmp(&b.id)));
        let (route, params) = eligible[0];

        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing_slash != request_trailing
            && !request_segments.is_empty()
        {
            if route.trailing_slash {
                Some(format!("{head}/"))
            } else {
                Some(head[..head.len() - 1].to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
            supplied.insert(*key, *value);
        }

        let mut required = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(key) | Segment::Wildcard(key) => {
                    required.push(key.as_str());
                    if !supplied.contains_key(key.as_str()) {
                        return Err(Error::MissingParam { name: key.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        let mut extras = supplied
            .keys()
            .filter(|key| !required.contains(key))
            .copied()
            .collect::<Vec<_>>();
        extras.sort();
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Param(key) => {
                    let value = supplied[key.as_str()];
                    if value.is_empty()
                        || !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || value.contains('/')
                    {
                        return Err(Error::InvalidValue { name: key.clone() });
                    }
                }
                Segment::Wildcard(key) => {
                    let value = supplied[key.as_str()];
                    let valid_bytes = value.bytes().all(|b| (0x20..=0x7e).contains(&b));
                    let valid_parts = value.is_empty()
                        || value
                            .split('/')
                            .all(|part| !part.is_empty());
                    if !valid_bytes || !valid_parts {
                        return Err(Error::InvalidValue { name: key.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        let mut output = String::from("/");
        let mut emitted = false;

        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if emitted {
                        output.push('/');
                    }
                    output.push_str(&encode_segment(value));
                    emitted = true;
                }
                Segment::Param(key) => {
                    if emitted {
                        output.push('/');
                    }
                    output.push_str(&encode_segment(supplied[key.as_str()]));
                    emitted = true;
                }
                Segment::Wildcard(key) => {
                    let value = supplied[key.as_str()];
                    if !value.is_empty() {
                        if emitted {
                            output.push('/');
                        }
                        let mut first = true;
                        for part in value.split('/') {
                            if !first {
                                output.push('/');
                            }
                            output.push_str(&encode_segment(part));
                            first = false;
                        }
                        emitted = true;
                    }
                }
            }
        }

        if route.trailing_slash && emitted {
            output.push('/');
        }

        Ok(output)
    }
}
