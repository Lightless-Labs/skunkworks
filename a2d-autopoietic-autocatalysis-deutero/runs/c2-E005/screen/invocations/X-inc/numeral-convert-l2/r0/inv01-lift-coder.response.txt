#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let payload_start = if let Some(tag_end) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..tag_end] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        tag_end + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];

    for (i, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => {}
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + i,
                });
            }
        }
    }

    for (i, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (i == 0
                || i + 1 == payload.len()
                || payload[i - 1] == b'_'
                || payload[i + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + i,
            });
        }
    }

    let digits: Vec<u8> = payload
        .iter()
        .filter_map(|&byte| if byte == b'_' { None } else { digit_value(byte) })
        .collect();

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, radix: u8, digit: u8) {
    let mut carry = digit as u16;
    for limb in value.iter_mut() {
        let n = (*limb as u16) * radix as u16 + carry;
        *limb = (n % 1_000_000_000) as u32;
        carry = n / 1_000_000_000;
    }
    while carry != 0 {
        value.push((carry % 1_000_000_000) as u32);
        carry /= 1_000_000_000;
    }
}

fn parse_value(digits: &[u8], radix: u8) -> Vec<u32> {
    let mut value = Vec::new();
    for &digit in digits {
        multiply_add(&mut value, radix, digit);
    }
    value
}

fn add_values(left: &[u32], right: &[u32]) -> Vec<u32> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u64;
    for i in 0..left.len().max(right.len()) {
        let a = left.get(i).copied().unwrap_or(0) as u64;
        let b = right.get(i).copied().unwrap_or(0) as u64;
        let sum = a + b + carry;
        result.push((sum % 1_000_000_000) as u32);
        carry = sum / 1_000_000_000;
    }
    if carry != 0 {
        result.push(carry as u32);
    }
    result
}

fn divide_value(value: &mut Vec<u32>, divisor: u8) -> u8 {
    let mut remainder = 0u64;
    for limb in value.iter_mut().rev() {
        let current = remainder * 1_000_000_000 + *limb as u64;
        *limb = (current / divisor as u64) as u32;
        remainder = current % divisor as u64;
    }
    while value.last() == Some(&0) {
        value.pop();
    }
    remainder as u8
}

fn render_value(mut value: Vec<u32>, radix: u8, style: BaseStyle) -> String {
    let mut digits = Vec::new();
    if value.is_empty() {
        digits.push(0);
    } else {
        while !value.is_empty() {
            digits.push(divide_value(&mut value, radix));
        }
        digits.reverse();
    }

    while digits.len() < style.min_width {
        digits.insert(0, 0);
    }

    let mut output = String::new();
    let group = style.group;
    for (i, &digit) in digits.iter().enumerate() {
        if group != 0 && i != 0 && (digits.len() - i) % group == 0 {
            output.push('_');
        }
        let character = match digit {
            0..=9 => b'0' + digit,
            10..=35 if style.upper => b'A' + (digit - 10),
            10..=35 => b'a' + (digit - 10),
            _ => unreachable!(),
        };
        output.push(character as char);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;
    let digits = parse_operand(input, from, 0)?;
    Ok(render_value(parse_value(&digits, from), to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;
    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_value = parse_value(&left_digits, base);
    let right_value = parse_value(&right_digits, base);
    Ok(render_value(add_values(&left_value, &right_value), base, style))
}
