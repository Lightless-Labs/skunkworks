```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy)]
enum EventKind {
    Arrival,
    Wake,
}

#[derive(Clone, Copy)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
}

struct State {
    level: usize,
    burst_pos: usize,
    remaining: u32,
    io_level: usize,
}

fn round_mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        return 0;
    }
    let numerator = total * 100;
    let denominator = count as u64;
    (numerator + denominator / 2) / denominator
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let mut events = Vec::with_capacity(jobs.len() * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: job.arrival as u64,
            kind: EventKind::Arrival,
            index,
        });
    }
    events.sort_by_key(|event| {
        let kind_order = match event.kind {
            EventKind::Arrival => 0,
            EventKind::Wake => 1,
        };
        (event.time, kind_order, event.index)
    });

    let mut queues: Vec<VecDeque<usize>> =
        (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut states: Vec<State> = jobs
        .iter()
        .map(|job| State {
            level: job.level,
            burst_pos: 0,
            remaining: job.bursts[0],
            io_level: job.level,
        })
        .collect();
    let mut completions = vec![None; jobs.len()];
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut event_cursor = 0usize;
    let mut last_dispatch_id: Option<u32> = None;
    let mut completed_count = 0usize;

    let admit_through = |through: u64,
                         event_cursor: &mut usize,
                         events: &mut Vec<Event>,
                         queues: &mut Vec<VecDeque<usize>>,
                         states: &mut Vec<State>| {
        while *event_cursor < events.len() && events[*event_cursor].time <= through {
            let event = events[*event_cursor];
            match event.kind {
                EventKind::Arrival => {
                    queues[states[event.index].level].push_back(event.index);
                }
                EventKind::Wake => {
                    let level = states[event.index].io_level.saturating_sub(1);
                    states[event.index].level = level;
                    queues[level].push_back(event.index);
                }
            }
            *event_cursor += 1;
        }
    };

    while completed_count < jobs.len() {
        admit_through(time, &mut event_cursor, &mut events, &mut queues, &mut states);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = events[event_cursor].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let job_id = jobs[index].id;

        if last_dispatch_id.is_some_and(|last_id| last_id != job_id) {
            time += switch_cost as u64;
        }
        admit_through(time, &mut event_cursor, &mut events, &mut queues, &mut states);

        let quantum = quanta[level];
        let consumed = states[index].remaining.min(quantum);
        let start = time;
        time += consumed as u64;
        states[index].remaining -= consumed;

        let burst_finished = states[index].remaining == 0;
        let final_burst = burst_finished && states[index].burst_pos + 1 == jobs[index].bursts.len();
        runs.push(Run {
            id: job_id,
            start,
            end: time,
            finished: final_burst,
            level,
        });

        if final_burst {
            completions[index] = Some(time);
            completed_count += 1;
        } else if burst_finished {
            states[index].io_level = level;
            states[index].burst_pos += 2;
            states[index].remaining = jobs[index].bursts[states[index].burst_pos];
            let wake_time = time + jobs[index].bursts[states[index].burst_pos - 1] as u64;
            events.push(Event {
                time: wake_time,
                kind: EventKind::Wake,
                index,
            });
            events[event_cursor..].sort_by_key(|event| {
                let kind_order = match event.kind {
                    EventKind::Arrival => 0,
                    EventKind::Wake => 1,
                };
                (event.time, kind_order, event.index)
            });
        }

        admit_through(time, &mut event_cursor, &mut events, &mut queues, &mut states);

        if !burst_finished {
            states[index].level = (level + 1).min(quanta.len() - 1);
            queues[states[index].level].push_back(index);
        }

        last_dispatch_id = Some(job_id);
    }

    let end = time;
    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = completions[index].unwrap_or(0);
        let turnaround = completion - job.arrival as u64;
        let cpu_total: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&burst| burst as u64)
            .sum();
        let io_total: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&delay| delay as u64)
            .sum();
        let waiting = turnaround - cpu_total - io_total;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: round_mean_x100(total_waiting, jobs.len()),
        mean_turnaround_x100: round_mean_x100(total_turnaround, jobs.len()),
        end,
    })
}

use std::collections::VecDeque;
```
