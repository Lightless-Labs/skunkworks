#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum EventKind {
    Arrival,
    Wake { level: usize },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
}

fn insert_event(events: &mut Vec<Event>, event: Event) {
    let position = events
        .iter()
        .position(|existing| {
            (existing.time, event_kind_order(existing.kind), existing.index)
                > (event.time, event_kind_order(event.kind), event.index)
        })
        .unwrap_or(events.len());
    events.insert(position, event);
}

fn event_kind_order(kind: EventKind) -> u8 {
    match kind {
        EventKind::Arrival => 0,
        EventKind::Wake { .. } => 1,
    }
}

pub fn schedule(
    jobs: &[Job],
    quanta: &[u32],
    switch_cost: u32,
) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|previous| previous.id == job.id) {
            return Err(Error::DuplicateId { index });
        }

        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }

        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }

        if job.bursts.is_empty()
            || job.bursts.len() > 31
            || job.bursts.len() % 2 == 0
        {
            return Err(Error::BurstCount { index });
        }

        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut queues: Vec<Vec<usize>> = (0..quanta.len()).map(|_| Vec::new()).collect();
    let mut events = Vec::with_capacity(jobs.len() * 16);
    for (index, job) in jobs.iter().enumerate() {
        insert_event(
            &mut events,
            Event {
                time: u64::from(job.arrival),
                kind: EventKind::Arrival,
                index,
            },
        );
    }

    let mut burst_positions = vec![0usize; jobs.len()];
    let mut remaining_cpu: Vec<u64> = jobs
        .iter()
        .map(|job| u64::from(job.bursts[0]))
        .collect();
    let mut levels: Vec<usize> = jobs.iter().map(|job| job.level).collect();
    let mut completions = vec![0u64; jobs.len()];
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut completed_count = 0usize;

    while completed_count < jobs.len() {
        while events.first().is_some_and(|event| event.time <= time) {
            let event = events.remove(0);
            match event.kind {
                EventKind::Arrival => queues[jobs[event.index].level].push(event.index),
                EventKind::Wake { level } => queues[level].push(event.index),
            }
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());

        let Some(level) = selected_level else {
            if let Some(event) = events.first().copied() {
                if event.time > time {
                    let start = time;
                    time = event.time;
                    if let Some(last) = idle.last_mut() {
                        if last.1 == start {
                            last.1 = time;
                        } else {
                            idle.push((start, time));
                        }
                    } else {
                        idle.push((start, time));
                    }
                    last_dispatch_id = None;
                }
                continue;
            }
            break;
        };

        let job_index = queues[level].remove(0);
        let job = &jobs[job_index];
        if last_dispatch_id.is_some_and(|id| id != job.id) {
            time += u64::from(switch_cost);
        }

        while events.first().is_some_and(|event| event.time <= time) {
            let event = events.remove(0);
            match event.kind {
                EventKind::Arrival => queues[jobs[event.index].level].push(event.index),
                EventKind::Wake { level } => queues[level].push(event.index),
            }
        }

        let run_start = time;
        let quantum = u64::from(quanta[level]);
        let consumed = remaining_cpu[job_index].min(quantum);
        time += consumed;
        remaining_cpu[job_index] -= consumed;

        let burst_finished = remaining_cpu[job_index] == 0;
        let final_burst = burst_finished
            && burst_positions[job_index] + 1 == job.bursts.len();
        runs.push(Run {
            id: job.id,
            start: run_start,
            end: time,
            finished: final_burst,
            level,
        });

        while events.first().is_some_and(|event| event.time <= time) {
            let event = events.remove(0);
            match event.kind {
                EventKind::Arrival => queues[jobs[event.index].level].push(event.index),
                EventKind::Wake { level } => queues[level].push(event.index),
            }
        }

        last_dispatch_id = Some(job.id);

        if !burst_finished {
            levels[job_index] = (level + 1).min(quanta.len() - 1);
            queues[levels[job_index]].push(job_index);
        } else if final_burst {
            completions[job_index] = time;
            completed_count += 1;
        } else {
            let io_position = burst_positions[job_index] + 1;
            let io_delay = u64::from(job.bursts[io_position]);
            burst_positions[job_index] += 2;
            remaining_cpu[job_index] = u64::from(job.bursts[burst_positions[job_index]]);
            let wake_level = level.saturating_sub(1);
            insert_event(
                &mut events,
                Event {
                    time: time + io_delay,
                    kind: EventKind::Wake { level: wake_level },
                    index: job_index,
                },
            );
        }
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = completions[index];
        let turnaround = completion - u64::from(job.arrival);
        let cpu_and_io: u64 = job.bursts.iter().map(|&length| u64::from(length)).sum();
        let waiting = turnaround - cpu_and_io;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = (total_waiting * 100 + count / 2) / count;
    let mean_turnaround_x100 = (total_turnaround * 100 + count / 2) / count;

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end: time,
    })
}
