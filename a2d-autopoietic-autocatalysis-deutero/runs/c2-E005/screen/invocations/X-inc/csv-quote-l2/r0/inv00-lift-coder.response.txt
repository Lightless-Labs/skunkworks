#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|column| column == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|column| column == name)
            }
        }
    }
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut position = 0usize;
    if input.starts_with('\u{feff}') {
        position = '\u{feff}'.len_utf8();
    }
    if position == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;
    let mut field_started = false;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_opening = 0usize;

    while position < bytes.len() {
        let byte = bytes[position];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: position,
            });
        }

        if in_quotes {
            if byte == quote {
                if position + 1 < bytes.len() && bytes[position + 1] == quote {
                    field.push(quote as char);
                    position += 2;
                    continue;
                }
                in_quotes = false;
                after_quote = true;
                position += 1;
                continue;
            }

            let start = position;
            position += 1;
            while position < bytes.len() && bytes[position] != quote && bytes[position] != 0 {
                position += 1;
            }
            field.push_str(&input[start..position]);
            continue;
        }

        if byte == b'\r'
            && (position + 1 >= bytes.len() || bytes[position + 1] != b'\n')
        {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: position,
            });
        }

        if after_quote {
            if byte == delimiter {
                fields.push(std::mem::take(&mut field));
                field_started = false;
                after_quote = false;
                position += 1;
                continue;
            }
            if byte == b'\n' || byte == b'\r' {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                field_started = false;
                after_quote = false;
                record_index += 1;
                position += if byte == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: position,
            });
        }

        if byte == delimiter {
            fields.push(std::mem::take(&mut field));
            field_started = false;
            position += 1;
        } else if byte == b'\n' || byte == b'\r' {
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            field_started = false;
            record_index += 1;
            position += if byte == b'\r' { 2 } else { 1 };
        } else if byte == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: position,
                });
            }
            in_quotes = true;
            quote_opening = position;
            field_started = true;
            position += 1;
        } else {
            let start = position;
            field_started = true;
            position += 1;
            while position < bytes.len()
                && bytes[position] != delimiter
                && bytes[position] != quote
                && bytes[position] != b'\n'
                && bytes[position] != b'\r'
                && bytes[position] != 0
            {
                position += 1;
            }
            field.push_str(&input[start..position]);
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_opening,
        });
    }

    if field_started || after_quote || !fields.is_empty() {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn build_table(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let mut records = records.into_iter();
    let header = if config.headers {
        records.next()
    } else {
        None
    };

    if let (Some(header_values), DuplicatePolicy::Reject) = (&header, config.duplicates) {
        let mut seen: Vec<(&str, usize)> = Vec::new();
        for (second, name) in header_values.iter().enumerate() {
            if let Some((_, first)) = seen.iter().find(|(seen_name, _)| *seen_name == name) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first: *first,
                    second,
                });
            }
            seen.push((name.as_str(), second));
        }
    }

    let mut rows = Vec::new();
    for (record, row) in records.enumerate() {
        if let Some(header_values) = &header {
            if row.len() != header_values.len() {
                return Err(CsvError::HeaderWidth {
                    record: record + 1,
                    expected: header_values.len(),
                    actual: row.len(),
                });
            }
        }
        rows.push(row);
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }

        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}
