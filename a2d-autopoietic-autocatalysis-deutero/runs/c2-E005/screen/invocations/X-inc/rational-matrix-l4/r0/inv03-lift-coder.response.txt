use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = n.checked_neg().ok_or(RationalError::Overflow)?;
        d = d.checked_neg().ok_or(RationalError::Overflow)?;
    }

    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let g = gcd_u128(n.unsigned_abs(), d as u128);
    let reduced_n = n / g as i128;
    let reduced_d = d / g as i128;

    if reduced_n < i64::MIN as i128 || reduced_n > i64::MAX as i128
        || reduced_d < 1 || reduced_d > i64::MAX as i128
    {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: reduced_n as i64,
        den: reduced_d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = rhs.den as i128 / g;
        let b = self.den as i128 / g;
        let n = (self.num as i128)
            .checked_mul(a)
            .and_then(|x| {
                (rhs.num as i128)
                    .checked_mul(b)
                    .and_then(|y| x.checked_add(y))
            })
            .ok_or(RationalError::Overflow)?;
        let d = (self.den as i128)
            .checked_mul(a)
            .ok_or(RationalError::Overflow)?;
        rational_from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = rhs.den as i128 / g;
        let b = self.den as i128 / g;
        let n = (self.num as i128)
            .checked_mul(a)
            .and_then(|x| {
                (rhs.num as i128)
                    .checked_mul(b)
                    .and_then(|y| x.checked_sub(y))
            })
            .ok_or(RationalError::Overflow)?;
        let d = (self.den as i128)
            .checked_mul(a)
            .ok_or(RationalError::Overflow)?;
        rational_from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.den as u128);
        let g2 = gcd_u128(rhs.num.unsigned_abs() as u128, self.den as u128);
        let n1 = (self.num as i128) / g1 as i128;
        let n2 = (rhs.num as i128) / g2 as i128;
        let d1 = (self.den as i128) / g2 as i128;
        let d2 = (rhs.den as i128) / g1 as i128;

        rational_from_i128(
            n1.checked_mul(n2).ok_or(RationalError::Overflow)?,
            d1.checked_mul(d2).ok_or(RationalError::Overflow)?,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }

        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.num.unsigned_abs() as u128);
        let g2 = gcd_u128(self.den as u128, rhs.den as u128);
        let n1 = (self.num as i128) / g1 as i128;
        let n2 = (rhs.den as i128) / g2 as i128;
        let d1 = (self.den as i128) / g2 as i128;
        let d2 = (rhs.num as i128) / g1 as i128;

        rational_from_i128(
            n1.checked_mul(n2).ok_or(RationalError::Overflow)?,
            d1.checked_mul(d2).ok_or(RationalError::Overflow)?,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = self.num.checked_neg().ok_or(RationalError::Overflow)?;
        Ok(Rational { num: n, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0usize;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }

        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            if i == bytes.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            i += 1;
            let start = i;
            den_start = Some(start);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }

        let num_digits = num_end - num_start;
        if num_digits >= 2 && bytes[num_start] == b'0' {
            return Err(parse_err(num_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == 1 && bytes[num_start] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        if let (Some(start), Some(end)) = (den_start, den_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_err(start + 1, ParseKind::NonCanonical));
            }
        }

        let num_limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let mut num_mag = 0u128;
        for &byte in &bytes[num_start..num_end] {
            num_mag = num_mag
                .saturating_mul(10)
                .saturating_add((byte - b'0') as u128);
            if num_mag > num_limit {
                return Err(parse_err(0, ParseKind::OutOfRange));
            }
        }

        let den_value = if let (Some(start), Some(end)) = (den_start, den_end) {
            let mut value = 0u128;
            for &byte in &bytes[start..end] {
                value = value
                    .saturating_mul(10)
                    .saturating_add((byte - b'0') as u128);
                if value > i64::MAX as u128 {
                    return Err(parse_err(start, ParseKind::OutOfRange));
                }
            }
            Some(value as u64)
        } else {
            None
        };

        if let (Some(start), Some(_)) = (den_start, den_end) {
            if den_value == Some(0) {
                return Err(parse_err(start, ParseKind::ZeroDenominator));
            }
        }

        if let (Some(start), Some(_), Some(value)) = (den_start, den_end, den_value) {
            if value == 1 {
                return Err(parse_err(start - 1, ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_err(start - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, value as u128) > 1 {
                return Err(parse_err(start - 1, ParseKind::NonCanonical));
            }
        }

        let signed_num = if negative {
            if num_mag == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };

        match den_value {
            Some(d) => Ok(Rational { num: signed_num, den: d as i64 }),
            None => Ok(Rational { num: signed_num, den: 1 }),
        }
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }

        let output_len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(output_len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    fn eliminate(&self, candidate_cols: usize) -> Result<Reduction, MatrixError> {
        let mut work = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..candidate_cols {
            if p == work.rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for row in p..work.rows {
                let value = work.data[row * work.cols + col];
                if value.num == 0 {
                    continue;
                }
                let key = (value.den, value.num.unsigned_abs(), row);
                if chosen.map_or(true, |old| {
                    let v = work.data[old * work.cols + col];
                    key < (v.den, v.num.unsigned_abs(), old)
                }) {
                    chosen = Some(row);
                }
            }
            let Some(q) = chosen else { continue };

            if q != p {
                for c in 0..work.cols {
                    work.data.swap(p * work.cols + c, q * work.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = work.data[p * work.cols + col];
            for c in 0..work.cols {
                let value = work.data[p * work.cols + c];
                work.data[p * work.cols + c] = value.div(&pivot).map_err(|source| {
                    MatrixError::Arithmetic { row: p, col: c, step: p, source }
                })?;
            }

            for row in 0..work.rows {
                if row == p {
                    continue;
                }
                let factor = work.data[row * work.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..work.cols {
                    let pivot_entry = work.data[p * work.cols + c];
                    let product = factor.mul(&pivot_entry).map_err(|source| {
                        MatrixError::Arithmetic { row, col: c, step: p, source }
                    })?;
                    let current = work.data[row * work.cols + c];
                    work.data[row * work.cols + c] =
                        current.sub(&product).map_err(|source| {
                            MatrixError::Arithmetic { row, col: c, step: p, source }
                        })?;
                }
            }

            pivots.push(col);
            p += 1;
        }

        Ok(Reduction { matrix: work, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);

        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(if r == c {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }

        let augmented = Self { rows: n, cols: width, data };
        let reduction = augmented.eliminate(n)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular { rank: reduction.pivots.len() });
        }

        let mut inverse_data = Vec::with_capacity(total / 2);
        for r in 0..n {
            for c in 0..n {
                inverse_data.push(reduction.matrix.data[r * width + n + c]);
            }
        }
        Ok(Self { rows: n, cols: n, data: inverse_data })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }

        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let total = self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut data = Vec::with_capacity(total);
        for r in 0..self.rows {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..k {
                data.push(rhs.data[r * k + c]);
            }
        }

        let augmented = Self { rows: self.rows, cols: width, data };
        let reduction = augmented.eliminate(n).map_err(SolveError::Matrix)?;

        for row in 0..self.rows {
            let coeff_zero = (0..n).all(|c| reduction.matrix.data[row * width + c].num == 0);
            if coeff_zero {
                for rhs_col in 0..k {
                    if reduction.matrix.data[row * width + n + rhs_col].num != 0 {
                        let witness = reduction.matrix.data[row * width..(row + 1) * width].to_vec();
                        return Ok(Solution::None { row, rhs_col, witness });
                    }
                }
            }
        }

        let rank = reduction.pivots.len();
        let mut free = Vec::new();
        let mut pivot_mask = vec![false; n];
        for &pivot in &reduction.pivots {
            pivot_mask[pivot] = true;
        }
        for c in 0..n {
            if !pivot_mask[c] {
                free.push(c);
            }
        }

        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![Rational { num: 0, den: 1 }; particular_len];
        for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
            for rhs_col in 0..k {
                particular_data[pivot_col * k + rhs_col] =
                    reduction.matrix.data[pivot_row * width + n + rhs_col];
            }
        }
        let particular = Matrix { rows: n, cols: k, data: particular_data };

        if rank == n {
            return Ok(Solution::Unique { values: particular, pivots: reduction.pivots });
        }

        let basis_len = n.checked_mul(free.len()).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![Rational { num: 0, den: 1 }; basis_len];
        for (basis_col, &free_col) in free.iter().enumerate() {
            basis_data[free_col * free.len() + basis_col] = Rational { num: 1, den: 1 };
            for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
                let coefficient = reduction.matrix.data[pivot_row * width + free_col];
                basis_data[pivot_col * free.len() + basis_col] =
                    coefficient.neg().map_err(|source| {
                        SolveError::Matrix(MatrixError::Arithmetic {
                            row: pivot_col,
                            col: basis_col,
                            step: rank,
                            source,
                        })
                    })?;
            }
        }

        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix { rows: n, cols: free.len(), data: basis_data },
            pivots: reduction.pivots,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}
