use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn abs_i64(n: i64) -> u128 {
    if n < 0 {
        (-(n as i128)) as u128
    } else {
        n as u128
    }
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let mut n = num as i128;
        let mut d = den as i128;
        if d < 0 {
            n = -n;
            d = -d;
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let ad = (self.den as i128) / g;
        let bd = (rhs.den as i128) / g;
        let n = (self.num as i128) * bd + (rhs.num as i128) * ad;
        let d = ad * rhs.den as i128;
        Self::from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let ad = (self.den as i128) / g;
        let bd = (rhs.den as i128) / g;
        let n = (self.num as i128) * bd - (rhs.num as i128) * ad;
        let d = ad * rhs.den as i128;
        Self::from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(abs_i64(self.num), rhs.den as u128) as i128;
        let g2 = gcd_u128(abs_i64(rhs.num), self.den as u128) as i128;
        let n = (self.num as i128 / g1) * (rhs.num as i128 / g2);
        let d = (self.den as i128 / g2) * (rhs.den as i128 / g1);
        Self::from_i128(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let g1 = gcd_u128(abs_i64(self.num), abs_i64(rhs.num)) as i128;
        let g2 = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let n = (self.num as i128 / g1) * (rhs.den as i128 / g2);
        let mut d = (self.den as i128 / g2) * (rhs.num as i128 / g1);
        let mut n = n;
        if d < 0 {
            d = -d;
            n = -n;
        }
        Self::from_i128(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        Self::from_i128(n, self.den as i128)
    }

    fn from_i128(mut n: i128, mut d: i128) -> Result<Self, RationalError> {
        if d == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let slash = if i < bytes.len() && bytes[i] == b'/' {
            let s = i;
            i += 1;
            let den_start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if den_start == i {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            Some((s, den_start, i))
        } else {
            None
        };
        if i < bytes.len() {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let num_end = slash.map_or(bytes.len(), |x| x.0);
        let neg = bytes[0] == b'-';
        let digits_start = if neg { 1 } else { 0 };
        let num_digits = num_end - digits_start;
        if num_digits >= 2 && bytes[digits_start] == b'0' {
            return Err(parse_err(digits_start + 1, ParseKind::NonCanonical));
        }
        if neg && num_digits == 1 && bytes[1] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        if let Some((_, ds, de)) = slash {
            if de - ds >= 2 && bytes[ds] == b'0' {
                return Err(parse_err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let nmag = parse_magnitude(&bytes[digits_start..num_end]);
        let nlimit = if neg { 1u128 << 63 } else { i64::MAX as u128 };
        if nmag > nlimit {
            return Err(parse_err(0, ParseKind::OutOfRange));
        }
        let n = if neg {
            if nmag == 1u128 << 63 {
                i64::MIN
            } else {
                -(nmag as i64)
            }
        } else {
            nmag as i64
        };

        let Some((slash_at, ds, de)) = slash else {
            return Ok(Self { num: n, den: 1 });
        };
        let dmag = parse_magnitude(&bytes[ds..de]);
        if dmag > i64::MAX as u128 {
            return Err(parse_err(ds, ParseKind::OutOfRange));
        }
        if dmag == 0 {
            return Err(parse_err(ds, ParseKind::ZeroDenominator));
        }
        if dmag == 1 {
            return Err(parse_err(slash_at, ParseKind::NonCanonical));
        }
        if n == 0 {
            return Err(parse_err(slash_at, ParseKind::NonCanonical));
        }
        if gcd_u128(abs_i64(n), dmag) > 1 {
            return Err(parse_err(slash_at, ParseKind::NonCanonical));
        }
        Ok(Self {
            num: n,
            den: dmag as i64,
        })
    }
}

fn parse_magnitude(bytes: &[u8]) -> u128 {
    bytes.iter().fold(0u128, |acc, b| {
        acc.saturating_mul(10).saturating_add((b - b'0') as u128)
    })
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols), right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut out = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                out.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data: out })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let (matrix, pivots, swaps) = eliminate(self.clone(), self.cols, self.cols)?;
        Ok(Reduction { matrix, pivots, swaps })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c { Rational { num: 1, den: 1 } } else { Rational { num: 0, den: 1 } });
            }
        }
        let aug = Self { rows: n, cols: width, data };
        let (red, pivots, _) = eliminate(aug, n, width)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }
        let mut right = Vec::with_capacity(n * n);
        for r in 0..n {
            right.extend_from_slice(&red.data[r * width + n..r * width + width]);
        }
        Ok(Self { rows: n, cols: n, data: right })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients: self.rows, rhs: rhs.rows });
        }
        let m = self.rows;
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let total = m.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        for r in 0..m {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            data.extend_from_slice(&rhs.data[r * k..(r + 1) * k]);
        }
        let aug = Self { rows: m, cols: width, data };
        let (red, pivots, _) = eliminate(aug, n, width).map_err(SolveError::Matrix)?;

        for r in 0..m {
            let coeff_zero = (0..n).all(|c| red.data[r * width + c].num == 0);
            if coeff_zero {
                if let Some(c) = (0..k).find(|&c| red.data[r * width + n + c].num != 0) {
                    let witness = red.data[r * width..r * width + width].to_vec();
                    return Ok(Solution::None { row: r, rhs_col: c, witness });
                }
            }
        }

        let free: Vec<usize> = (0..n).filter(|c| !pivots.contains(c)).collect();
        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular = vec![Rational { num: 0, den: 1 }; particular_len];
        for (j, &pcol) in pivots.iter().enumerate() {
            for c in 0..k {
                particular[pcol * k + c] = red.data[j * width + n + c];
            }
        }
        let particular = Self { rows: n, cols: k, data: particular };

        if pivots.len() == n {
            return Ok(Solution::Unique { values: particular, pivots });
        }

        let basis_len = n.checked_mul(free.len()).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![Rational { num: 0, den: 1 }; basis_len];
        for (j, &free_col) in free.iter().enumerate() {
            basis[free_col * free.len() + j] = Rational { num: 1, den: 1 };
            for (i, &pivot_col) in pivots.iter().enumerate() {
                let value = red.data[i * width + free_col].neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col, col: j, step: pivots.len(), source,
                    })
                })?;
                basis[pivot_col * free.len() + j] = value;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Self { rows: n, cols: basis_len.checked_div(n.max(1)).unwrap_or(0), data: basis },
            pivots,
        })
    }
}

fn eliminate(
    mut matrix: Matrix,
    candidate_cols: usize,
    update_cols: usize,
) -> Result<(Matrix, Vec<usize>, Vec<(usize, usize)>), MatrixError> {
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0;
    for col in 0..candidate_cols {
        if p == matrix.rows { break; }
        let mut best: Option<usize> = None;
        for r in p..matrix.rows {
            let v = matrix.data[r * matrix.cols + col];
            if v.num == 0 { continue; }
            let key = (v.den, abs_i64(v.num), r);
            if best.map_or(true, |b| {
                let w = matrix.data[b * matrix.cols + col];
                key < (w.den, abs_i64(w.num), b)
            }) {
                best = Some(r);
            }
        }
        let Some(q) = best else { continue };
        if q != p {
            for c in 0..matrix.cols {
                matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
            }
            swaps.push((p, q));
        }
        let pivot = matrix.data[p * matrix.cols + col];
        for c in 0..update_cols {
            let idx = p * matrix.cols + c;
            matrix.data[idx] = matrix.data[idx].div(&pivot)
                .map_err(|source| MatrixError::Arithmetic { row: p, col: c, step: p, source })?;
        }
        for r in 0..matrix.rows {
            if r == p { continue; }
            let factor = matrix.data[r * matrix.cols + col];
            if factor.num == 0 { continue; }
            for c in 0..update_cols {
                let product = factor.mul(&matrix.data[p * matrix.cols + c])
                    .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
                let idx = r * matrix.cols + c;
                matrix.data[idx] = matrix.data[idx].sub(&product)
                    .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
            }
        }
        pivots.push(col);
        p += 1;
    }
    Ok((matrix, pivots, swaps))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize { self.pivots.len() }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}
