use std::cmp::Ordering;
use std::collections::HashSet;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return Err(if start == bytes.len()
            || matches!(bytes[start], b'.' | b'-' | b'+')
        {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn valid_identifier_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut at = start;
    loop {
        let id_start = at;
        while at < end && bytes[at] != b'.' {
            if !valid_identifier_char(bytes[at]) {
                return Err(VersionError::InvalidChar { at });
            }
            at += 1;
        }
        if at == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        if prerelease {
            let id = &s[id_start..at];
            if id.bytes().all(is_digit) && id.len() > 1 && id.starts_with('0') {
                return Err(VersionError::LeadingZero { at: id_start });
            }
        }
        out.push(s[id_start..at].to_string());
        if at == end {
            break;
        }
        at += 1;
        if at == end {
            return Err(VersionError::EmptyIdentifier { at });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    for _ in 0..2 {
        if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        if s.as_bytes()[pos] != b'.' {
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        if _ == 0 {
            break;
        }
    }
    let minor = parse_core_number(s, &mut pos)?;
    if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
        return Err(VersionError::MissingCore { at: pos });
    }
    if s.as_bytes()[pos] != b'.' {
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;
    let patch = parse_core_number(s, &mut pos)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < s.len() {
        match s.as_bytes()[pos] {
            b'-' => {
                let pre_start = pos + 1;
                let plus = s[pre_start..].find('+').map(|i| pre_start + i);
                let pre_end = plus.unwrap_or(s.len());
                pre = parse_identifiers(s, pre_start, pre_end, true)?;
                if let Some(plus_at) = plus {
                    build = parse_identifiers(s, plus_at + 1, s.len(), false)?;
                }
            }
            b'+' => build = parse_identifiers(s, pos + 1, s.len(), false)?,
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }
    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_numeric_id(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let c = match (xn, yn) {
            (true, true) => cmp_numeric_id(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core(a)
        .cmp(&core(b))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => cmp_pre(&a.pre, &b.pre),
        })
}

fn make_version(c: (u64, u64, u64), pre: Vec<String>) -> Version {
    Version {
        major: c.0,
        minor: c.1,
        patch: c.2,
        pre,
        build: Vec::new(),
    }
}

fn fmt_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn bump(c: (u64, u64, u64), field: usize) -> Option<(u64, u64, u64)> {
    match field {
        0 => Some((c.0.checked_add(1)?, 0, 0)),
        1 => Some((c.0, c.1.checked_add(1)?, 0)),
        _ => Some((c.0, c.1, c.2.checked_add(1)?)),
    }
}

#[derive(Clone, Debug)]
enum Op {
    Lt,
    Le,
    Eq,
    Ge,
    Gt,
}

#[derive(Clone, Debug)]
struct Cmp {
    op: Op,
    version: Version,
}

#[derive(Clone, Debug)]
struct Arm {
    cmps: Vec<Cmp>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

fn satisfies_cmp(v: &Version, c: &Cmp) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Eq => ord == Ordering::Equal,
        Op::Ge => ord != Ordering::Less,
        Op::Gt => ord == Ordering::Greater,
    }
}

fn arm_matches(v: &Version, a: &Arm, include_prerelease: bool) -> bool {
    if !a.cmps.iter().all(|c| satisfies_cmp(v, c)) {
        return false;
    }
    v.pre.is_empty()
        || include_prerelease
        || a.anchors.iter().any(|c| *c == core(v))
}

fn words_with_offsets(s: &str) -> Vec<(usize, usize)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && matches!(b[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        if i == b.len() {
            break;
        }
        let start = i;
        while i < b.len() && !matches!(b[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        out.push((start, i));
    }
    out
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    full: Option<Version>,
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix = s.contains('-') || s.contains('+');
    let parts: Vec<&str> = s.split('.').collect();
    let wildcard_free_full = parts.len() >= 3 && !parts.iter().any(|p| matches!(*p, "*" | "x" | "X"));
    if has_suffix || wildcard_free_full {
        return parse_version(s)
            .map(|v| Partial { fields: vec![Some(v.major), Some(v.minor), Some(v.patch)], full: Some(v) })
            .map_err(|error| RangeError::Version { at, error });
    }
    if parts.is_empty() || parts.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    for p in parts {
        if matches!(p, "*" | "x" | "X") {
            wildcard_seen = true;
            fields.push(None);
        } else if !p.is_empty() && p.bytes().all(is_digit) {
            if wildcard_seen || (p.len() > 1 && p.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let n = p.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            fields.push(Some(n));
        } else {
            return Err(RangeError::Syntax { at });
        }
    }
    Ok(Partial { fields, full: None })
}

fn padded(p: &Partial) -> (u64, u64, u64) {
    (
        p.fields.get(0).and_then(|x| *x).unwrap_or(0),
        p.fields.get(1).and_then(|x| *x).unwrap_or(0),
        p.fields.get(2).and_then(|x| *x).unwrap_or(0),
    )
}

fn numeric_count(p: &Partial) -> usize {
    p.fields.iter().take_while(|x| x.is_some()).count()
}

fn add_bound(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(core(&version));
    }
    arm.cmps.push(Cmp { op, version });
}

fn parse_operand_expansion(
    arm: &mut Arm,
    token: &str,
    token_at: usize,
    mode: char,
) -> Result<(), RangeError> {
    let (operand, operand_at) = if mode == '^' || mode == '~' {
        (&token[1..], token_at + 1)
    } else {
        (token, token_at)
    };
    let p = parse_partial(operand, operand_at)?;
    let k = numeric_count(&p);
    let c = padded(&p);
    if mode == '^' || mode == '~' {
        if k == 0 {
            return Err(RangeError::Syntax { at: token_at });
        }
        let lower = p.full.clone().unwrap_or_else(|| make_version(c, Vec::new()));
        add_bound(arm, Op::Ge, lower, true);
        let field = if mode == '~' {
            if k == 1 { 0 } else { 1 }
        } else if c.0 > 0 || k == 1 {
            0
        } else if c.1 > 0 || k == 2 {
            1
        } else {
            2
        };
        let up = bump(c, field).ok_or(RangeError::Syntax { at: operand_at })?;
        arm.cmps.push(Cmp {
            op: Op::Lt,
            version: make_version(up, vec!["0".to_string()]),
        });
    } else if let Some(v) = p.full {
        add_bound(arm, Op::Eq, v, true);
    } else if k == 0 {
        // All wildcard fields match all versions.
    } else if k == 1 || k == 2 {
        arm.cmps.push(Cmp { op: Op::Ge, version: make_version(c, Vec::new()) });
        let up = bump(c, k - 1).ok_or(RangeError::Syntax { at: operand_at })?;
        arm.cmps.push(Cmp { op: Op::Lt, version: make_version(up, vec!["0".to_string()]) });
    } else {
        arm.cmps.push(Cmp { op: Op::Eq, version: make_version(c, Vec::new()) });
    }
    Ok(())
}

fn parse_comparator(token: &str, at: usize) -> Result<Cmp, RangeError> {
    let (op, n) = if token.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.starts_with("<=") {
        (Op::Le, 2)
    } else if token.starts_with('>') {
        (Op::Gt, 1)
    } else if token.starts_with('<') {
        (Op::Lt, 1)
    } else {
        (Op::Eq, 0)
    };
    let operand_at = at + n;
    let operand = &token[n..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    let version = parse_version(operand)
        .map_err(|error| RangeError::Version { at: operand_at, error })?;
    Ok(Cmp { op, version })
}

fn parse_hyphen_arm(words: &[(usize, usize)], s: &str, arm: &mut Arm) -> Result<(), RangeError> {
    let minus = words
        .iter()
        .find(|(a, b)| &s[*a..*b] == "-")
        .map(|(a, _)| *a)
        .unwrap();
    if words.len() != 3 || &s[words[1].0..words[1].1] != "-" {
        return Err(RangeError::Syntax { at: minus });
    }
    let (la, lb) = words[0];
    let (ua, ub) = words[2];
    let lower = parse_partial(&s[la..lb], la)?;
    let upper = parse_partial(&s[ua..ub], ua)?;
    if numeric_count(&lower) == 0 {
        return Err(RangeError::Syntax { at: la });
    }
    if numeric_count(&upper) == 0 {
        return Err(RangeError::Syntax { at: ua });
    }
    let lv = lower.full.clone().unwrap_or_else(|| make_version(padded(&lower), Vec::new()));
    add_bound(arm, Op::Ge, lv, true);
    if let Some(v) = upper.full {
        add_bound(arm, Op::Le, v, true);
    } else {
        let c = padded(&upper);
        let k = numeric_count(&upper);
        let up = bump(c, k - 1).ok_or(RangeError::Syntax { at: ua })?;
        arm.cmps.push(Cmp { op: Op::Lt, version: make_version(up, vec!["0".to_string()]) });
    }
    Ok(())
}

fn parse_arm(s: &str, words: &[(usize, usize)]) -> Result<Arm, RangeError> {
    let mut arm = Arm { cmps: Vec::new(), anchors: Vec::new() };
    if words.iter().any(|(a, b)| &s[*a..*b] == "-") {
        parse_hyphen_arm(words, s, &mut arm)?;
        return Ok(arm);
    }
    for &(a, b) in words {
        let token = &s[a..b];
        if token.starts_with('^') || token.starts_with('~') {
            parse_operand_expansion(&mut arm, token, a, token.as_bytes()[0] as char)?;
        } else if token.starts_with(">") || token.starts_with("<") || token.starts_with("=") {
            arm.cmps.push(parse_comparator(token, a)?);
        } else if token.starts_with(|c: char| c.is_ascii_digit())
            || token.starts_with('*')
            || token.starts_with('x')
            || token.starts_with('X')
        {
            parse_operand_expansion(&mut arm, token, a, ' ')?;
        } else {
            return Err(RangeError::Syntax { at: a });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let words = words_with_offsets(s);
    if words.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < words.len() {
        let (a, b) = words[i];
        if &s[a..b] == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: a });
            }
            arms.push(parse_arm(s, &current)?);
            current.clear();
            if i + 1 == words.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            current.push(words[i]);
        }
        i += 1;
    }
    if current.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(s, &current)?);
    Ok(Range { arms })
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

fn arm_bound(arm: &Arm, lower: bool) -> Option<(&Version, bool)> {
    let mut best: Option<(&Version, bool)> = None;
    for c in &arm.cmps {
        let (is_lower, inclusive) = match c.op {
            Op::Gt => (true, false),
            Op::Ge => (true, true),
            Op::Lt => (false, false),
            Op::Le => (false, true),
            Op::Eq => (true, true),
        };
        if is_lower == lower {
            let replace = match best {
                None => true,
                Some((v, inc)) => {
                    let ord = compare_versions(&c.version, v);
                    if lower {
                        ord == Ordering::Greater || (ord == Ordering::Equal && !inclusive && inc)
                    } else {
                        ord == Ordering::Less || (ord == Ordering::Equal && !inclusive && inc)
                    }
                }
            };
            if replace {
                best = Some((&c.version, inclusive));
            }
        }
        if c.op == Op::Eq && !lower {
            let replace = match best {
                None => true,
                Some((v, inc)) => compare_versions(&c.version, v) == Ordering::Less
                    || (compare_versions(&c.version, v) == Ordering::Equal && !inc),
            };
            if replace {
                best = Some((&c.version, true));
            }
        }
    }
    best
}

fn next_stable_core(c: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if c.2 < u64::MAX {
        Some((c.0, c.1, c.2 + 1))
    } else if c.1 < u64::MAX {
        Some((c.0, c.1 + 1, 0))
    } else if c.0 < u64::MAX {
        Some((c.0 + 1, 0, 0))
    } else {
        None
    }
}

fn simple_witness(arm: &Arm) -> bool {
    let lower = arm_bound(arm, true);
    let upper = arm_bound(arm, false);
    if let (Some((l, li)), Some((u, ui))) = (lower, upper) {
        match compare_versions(l, u) {
            Ordering::Greater => return false,
            Ordering::Equal => {
                if !(li && ui) {
                    return false;
                }
                let v = make_version(core(l), l.pre.clone());
                return arm_matches(&v, arm, true) || v.pre.is_empty();
            }
            Ordering::Less => {}
        }
    }
    // Stable releases occupy every core tuple, so test the strongest core boundary and its successor.
    let start = lower.map(|(v, inc)| {
        if inc && v.pre.is_empty() {
            core(v)
        } else if let Some(c) = next_stable_core(core(v)) {
            c
        } else {
            core(v)
        }
    });
    let mut candidates = Vec::new();
    if let Some(c) = start {
        candidates.push(make_version(c, Vec::new()));
        if let Some(n) = next_stable_core(c) {
            candidates.push(make_version(n, Vec::new()));
        }
    } else {
        candidates.push(make_version((0, 0, 0), Vec::new()));
    }
    candidates.into_iter().any(|v| {
        arm_matches(&v, arm, true)
            && upper
                .map(|(u, inc)| {
                    let o = compare_versions(&v, u);
                    o == Ordering::Less || (inc && o == Ordering::Equal)
                })
                .unwrap_or(true)
    }) || !arm.anchors.is_empty()
        && arm.anchors.iter().any(|c| {
            let v = make_version(*c, vec!["0".to_string()]);
            arm_matches(&v, arm, false)
        })
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let mut merged = Arm { cmps: aa.cmps.clone(), anchors: aa.anchors.clone() };
            merged.cmps.extend(bb.cmps.clone());
            merged.anchors.extend(bb.anchors.clone());
            if include_prerelease {
                if simple_witness(&merged) {
                    return true;
                }
            } else {
                let shared: Vec<_> = aa.anchors.iter().copied()
                    .filter(|x| bb.anchors.contains(x)).collect();
                let pre_ok = shared.iter().any(|c| {
                    arm_matches(&make_version(*c, vec!["0".to_string()]), &merged, false)
                });
                if pre_ok || simple_witness(&merged) {
                    return true;
                }
            }
        }
    }
    false
}

pub fn canonical(r: &Range) -> String {
    let mut out = Vec::new();
    for arm in &r.arms {
        if !simple_witness(arm) {
            continue;
        }
        let bound_string = |lower: bool| -> String {
            match arm_bound(arm, lower) {
                None => "*".to_string(),
                Some((v, inc)) => {
                    let op = match (lower, inc) {
                        (true, true) => ">=",
                        (true, false) => ">",
                        (false, true) => "<=",
                        (false, false) => "<",
                    };
                    format!("{}{}", op, fmt_version(v))
                }
            }
        };
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anchor_s = if anchors.is_empty() {
            "-".to_string()
        } else {
            anchors.iter().map(|c| format!("{}.{}.{}", c.0, c.1, c.2)).collect::<Vec<_>>().join(",")
        };
        out.push(format!("[{};{};pre={}]", bound_string(true), bound_string(false), anchor_s));
    }
    out.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
    out.dedup();
    if out.is_empty() { "EMPTY".to_string() } else { out.join(" || ") }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (index, s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index, error })?);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        best = match best {
            None => Some(i),
            Some(j) => {
                let ord = compare_versions(v, &parsed[j]);
                if (max && ord == Ordering::Greater) || (!max && ord == Ordering::Less) {
                    Some(i)
                } else {
                    Some(j)
                }
            }
        };
    }
    Ok(best.map(|i| candidates[i].clone()))
}

    pub expr: SetExpr,
}

#[derive(Clone, Debug)]
pub struct Package {
    pub name: String,
    pub candidates: Vec<String>,
    pub constraints: Vec<Constraint>,
}

#[derive(Clone, Copy, Debug)]
pub enum Preference {
    Highest,
    Lowest,
    StableFirst,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    ranges: &mut Vec<Vec<Range>>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_string(),
                label: label.to_string(),
                path: path.clone(),
                error,
            })?;
            ranges.push(vec![r]);
        }
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path, ranges)?;
                path.pop();
            }
        }
    }
    Ok(())
}

fn eval_expr(expr: &SetExpr, v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|r| matches_with(v, &r, include)).unwrap_or(false),
        SetExpr::Union(children) => children.iter().any(|x| eval_expr(x, v, include)),
        SetExpr::Intersection(children) => children.iter().all(|x| eval_expr(x, v, include)),
    }
}

fn conflict_labels(
    package: &Package,
    versions: &[Version],
    include: bool,
) -> Vec<String> {
    if versions.is_empty() {
        return Vec::new();
    }
    let n = package.constraints.len();
    let mut best: Option<Vec<String>> = None;
    if n < usize::BITS as usize {
        for mask in 1usize..(1usize << n) {
            let count = mask.count_ones() as usize;
            if best.as_ref().map(|b| b.len() < count).unwrap_or(false) {
                continue;
            }
            let any = versions.iter().any(|v| {
                (0..n).filter(|i| mask & (1 << i) != 0)
                    .all(|i| eval_expr(&package.constraints[i].expr, v, include))
            });
            if !any {
                let mut labels: Vec<String> = (0..n)
                    .filter(|i| mask & (1 << i) != 0)
                    .map(|i| package.constraints[i].label.clone())
                    .collect();
                labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
                let replace = best.as_ref().map(|b| {
                    labels.len() < b.len()
                        || (labels.len() == b.len()
                            && labels.iter().map(String::as_bytes).cmp(b.iter().map(String::as_bytes)) == Ordering::Less)
                }).unwrap_or(true);
                if replace {
                    best = Some(labels);
                }
            }
        }
    } else {
        let mut chosen: Vec<String> = package.constraints.iter().map(|c| c.label.clone()).collect();
        chosen.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        return chosen;
    }
    best.unwrap_or_default()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }
    for p in packages {
        let mut labels = HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel {
                    package: p.name.clone(),
                    label: c.label.clone(),
                });
            }
        }
    }
    let mut parsed_by_package = Vec::new();
    for p in packages {
        let mut vs = Vec::new();
        for (index, s) in p.candidates.iter().enumerate() {
            vs.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(),
                index,
                error,
            })?);
        }
        parsed_by_package.push(vs);
    }
    for p in packages {
        for c in &p.constraints {
            let mut ranges = Vec::new();
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new(), &mut ranges)?;
        }
    }

    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let versions = &parsed_by_package[pi];
        let mut qualifying = Vec::new();
        for (i, v) in versions.iter().enumerate() {
            if p.constraints.iter().all(|c| eval_expr(&c.expr, v, include_prerelease)) {
                qualifying.push(i);
            }
        }
        if qualifying.is_empty() {
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels: conflict_labels(p, versions, include_prerelease),
            });
        }
        let stable_first = matches!(preference, Preference::StableFirst)
            && qualifying.iter().any(|i| versions[*i].pre.is_empty());
        let mut best = qualifying[0];
        for &i in &qualifying[1..] {
            let vi = &versions[i];
            let vb = &versions[best];
            if stable_first && vi.pre.is_empty() != vb.pre.is_empty() {
                if vi.pre.is_empty() {
                    best = i;
                }
                continue;
            }
            let ord = compare_versions(vi, vb);
            let replace = match preference {
                Preference::Lowest => ord == Ordering::Less,
                Preference::Highest | Preference::StableFirst => ord == Ordering::Greater,
            };
            if replace {
                best = i;
            }
        }
        selections.push(Selection {
            package: p.name.clone(),
            version: p.candidates[best].clone(),
        });
    }
    Ok(selections)
}
