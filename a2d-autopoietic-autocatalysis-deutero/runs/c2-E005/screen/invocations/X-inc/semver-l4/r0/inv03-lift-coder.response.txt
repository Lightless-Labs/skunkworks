use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return Err(if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn valid_ident_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut at = start;
    loop {
        let id_start = at;
        while at < end && bytes[at] != b'.' {
            if !valid_ident_byte(bytes[at]) {
                return Err(VersionError::InvalidChar { at });
            }
            at += 1;
        }
        if id_start == at {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let numeric = bytes[id_start..at].iter().all(u8::is_ascii_digit);
        if prerelease && numeric && at - id_start > 1 && bytes[id_start] == b'0' {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(s[id_start..at].to_string());
        if at == end {
            break;
        }
        at += 1;
        if at == end {
            return Err(VersionError::EmptyIdentifier { at });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_number(s, &mut pos)?;
    for _ in 0..2 {
        if pos >= s.len() {
            return Err(VersionError::MissingCore { at: pos });
        }
        if s.as_bytes()[pos] != b'.' {
            return Err(if matches!(s.as_bytes()[pos], b'-' | b'+') {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        pos += 1;
        if _ == 0 {}
    }
    // Parse minor and patch explicitly, preserving numeric error precedence.
    // Rewind to the first separator and consume the two fields.
    pos = s.as_bytes()[..].iter().position(|&b| b == b'.').unwrap() + 1;
    let minor = parse_number(s, &mut pos)?;
    if pos >= s.len() {
        return Err(VersionError::MissingCore { at: pos });
    }
    if s.as_bytes()[pos] != b'.' {
        return Err(if matches!(s.as_bytes()[pos], b'-' | b'+') {
            VersionError::MissingCore { at: pos }
        } else {
            VersionError::InvalidChar { at: pos }
        });
    }
    pos += 1;
    let patch = parse_number(s, &mut pos)?;
    if pos < s.len() && !matches!(s.as_bytes()[pos], b'-' | b'+') {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < s.len() && s.as_bytes()[pos] == b'-' {
        let pstart = pos + 1;
        let plus = s[pstart..].find('+').map(|x| pstart + x);
        let pend = plus.unwrap_or(s.len());
        pre = parse_identifiers(s, pstart, pend, true)?;
        pos = pend;
    }
    if pos < s.len() && s.as_bytes()[pos] == b'+' {
        let bstart = pos + 1;
        build = parse_identifiers(s, bstart, s.len(), false)?;
        pos = s.len();
    }
    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let c = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Cmp {
    op: Op,
    version: Version,
    anchor: bool,
}
#[derive(Clone, Copy, Debug)]
enum Op { Eq, Lt, Le, Gt, Ge }

#[derive(Clone, Debug)]
struct Arm {
    cmps: Vec<Cmp>,
    anchors: Vec<(u64,u64,u64)>,
}
#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn ws(b: u8) -> bool { matches!(b, b' ' | b'\t' | b'\r' | b'\n') }

fn core_version(m: u64, n: u64, p: u64, pre: Vec<String>) -> Version {
    Version { major:m, minor:n, patch:p, pre, build:vec![] }
}
fn bump(v: &Version, field: usize) -> Option<Version> {
    let (mut m, mut n, mut p) = (v.major, v.minor, v.patch);
    match field {
        0 => { m = m.checked_add(1)?; n=0; p=0; }
        1 => { n = n.checked_add(1)?; p=0; }
        _ => p = p.checked_add(1)?,
    }
    Some(core_version(m,n,p,vec!["0".into()]))
}

fn parse_partial(s: &str, offset: usize) -> Result<(Version, usize, bool), RangeError> {
    let full = s.contains(['-', '+'])
        || (s.split('.').count() >= 3 && !s.split('.').any(|x| matches!(x, "*"|"x"|"X")));
    if full {
        let v = parse_version(s).map_err(|error| RangeError::Version { at: offset, error })?;
        return Ok((v, 3, true));
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() > 3 || fields.is_empty() || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at: offset });
    }
    let mut nums = Vec::new();
    let mut wildcard = false;
    for f in &fields {
        if matches!(*f, "*"|"x"|"X") {
            wildcard = true;
        } else {
            if wildcard || !f.bytes().all(is_digit) || (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at: offset });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: offset })?;
            nums.push(n);
        }
    }
    let k = nums.len();
    let mut vals = nums;
    while vals.len() < 3 { vals.push(0); }
    Ok((core_version(vals[0],vals[1],vals[2],vec![]), k, false))
}

fn add_cmp(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() {
        arm.anchors.push((v.major,v.minor,v.patch));
    }
    arm.cmps.push(Cmp { op, version:v, anchor });
}

fn parse_bare(tok: &str, at: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let (v,k,full) = parse_partial(tok, at)?;
    if full {
        add_cmp(arm, Op::Eq, v, true);
    } else if k == 0 {
        return Ok(());
    } else {
        add_cmp(arm, Op::Ge, v.clone(), false);
        if k < 3 {
            let field = if k == 1 {0} else {1};
            let up = bump(&v,field).ok_or(RangeError::Syntax { at })?;
            add_cmp(arm, Op::Lt, up, false);
        }
    }
    Ok(())
}

fn parse_token(tok: &str, at: usize, arm: &mut Arm) -> Result<(), RangeError> {
    if let Some(rest) = tok.strip_prefix('^').or_else(|| tok.strip_prefix('~')) {
        let caret = tok.starts_with('^');
        let off = at + 1;
        let (v,k,full) = parse_partial(rest, off)?;
        if k == 0 { return Err(RangeError::Syntax { at }); }
        add_cmp(arm, Op::Ge, v.clone(), full && !v.pre.is_empty());
        let field = if caret {
            if v.major > 0 || k == 1 { 0 } else if v.minor > 0 || k == 2 { 1 } else { 2 }
        } else if k == 1 { 0 } else { 1 };
        let up = bump(&v,field).ok_or(RangeError::Syntax { at: off })?;
        add_cmp(arm, Op::Lt, up, false);
        return Ok(());
    }
    if tok.starts_with(['<','>','=']) {
        let (op, n) = if tok.starts_with(">=") {(Op::Ge,2)}
            else if tok.starts_with("<=") {(Op::Le,2)}
            else { (match tok.as_bytes()[0] { b'<' => Op::Lt, b'>' => Op::Gt, _ => Op::Eq },1) };
        let operand = &tok[n..];
        let operand_at = at+n;
        if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at });
        }
        let v = parse_version(operand).map_err(|error| RangeError::Version { at: operand_at, error })?;
        add_cmp(arm,op,v.clone(),!v.pre.is_empty());
        return Ok(());
    }
    if tok.as_bytes()[0].is_ascii_digit() || matches!(tok, "*"|"x"|"X") {
        parse_bare(tok,at,arm)
    } else {
        Err(RangeError::Syntax { at })
    }
}

fn tokenize(s: &str, start: usize, end: usize) -> Vec<(String,usize)> {
    let b=s.as_bytes(); let mut out=Vec::new(); let mut i=start;
    while i<end {
        while i<end && ws(b[i]) {i+=1;}
        if i>=end {break;}
        let st=i; while i<end && !ws(b[i]) {i+=1;}
        out.push((s[st..i].to_string(),st));
    }
    out
}

fn parse_arm(s: &str, tokens: &[(String,usize)]) -> Result<Arm,RangeError> {
    let mut arm=Arm{cmps:vec![],anchors:vec![]};
    let hyphens: Vec<usize> = tokens.iter().enumerate().filter(|(_,t)| t.0=="-").map(|(i,_)|i).collect();
    if !hyphens.is_empty() {
        let h=hyphens[0];
        if hyphens.len()!=1 || h!=1 || tokens.len()!=3 {
            return Err(RangeError::Syntax{at:tokens[h].1});
        }
        let (lo,lk,lf)=parse_partial(&tokens[0].0,tokens[0].1)?;
        let (hi,hk,hf)=parse_partial(&tokens[2].0,tokens[2].1)?;
        if lk==0 {return Err(RangeError::Syntax{at:tokens[0].1});}
        if hk==0 {return Err(RangeError::Syntax{at:tokens[2].1});}
        add_cmp(&mut arm,Op::Ge,lo.clone(),lf && !lo.pre.is_empty());
        if hf {
            add_cmp(&mut arm,Op::Le,hi.clone(),!hi.pre.is_empty());
        } else {
            let up=bump(&hi,if hk==1 {0}else{1}).ok_or(RangeError::Syntax{at:tokens[2].1})?;
            add_cmp(&mut arm,Op::Lt,up,false);
        }
    } else {
        for (t,at) in tokens {
            parse_token(t,*at,&mut arm)?;
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b=s.as_bytes(); let mut arms=Vec::new(); let mut i=0; let mut arm_start=0;
    while i < b.len() {
        if i+1<b.len() && b[i]==b'|' && b[i+1]==b'|' && (i==0 || ws(b[i-1])) && (i+2==b.len() || ws(b[i+2])) {
            let toks=tokenize(s,arm_start,i);
            if toks.is_empty() { return Err(RangeError::Empty{at:i}); }
            arms.push(parse_arm(s,&toks)?);
            i+=2;
            while i<b.len() && ws(b[i]) {i+=1;}
            arm_start=i;
        } else { i+=1; }
    }
    let toks=tokenize(s,arm_start,b.len());
    if toks.is_empty() {
        if arms.is_empty() { return Err(RangeError::Empty{at:0}); }
        return Err(RangeError::Empty{at:b.len()});
    }
    arms.push(parse_arm(s,&toks)?);
    Ok(Range{arms})
}

fn cmp_test(v:&Version,c:&Cmp)->bool {
    let o=compare_versions(v,&c.version);
    match c.op { Op::Eq=>o==Ordering::Equal, Op::Lt=>o==Ordering::Less, Op::Le=>o!=Ordering::Greater, Op::Gt=>o==Ordering::Greater, Op::Ge=>o!=Ordering::Less }
}
fn arm_match(v:&Version,a:&Arm,include:bool)->bool {
    if !include && !v.pre.is_empty() && !a.anchors.contains(&(v.major,v.minor,v.patch)) { return false; }
    a.cmps.iter().all(|c|cmp_test(v,c))
}
pub fn matches_with(v:&Version,r:&Range,include_prerelease:bool)->bool {
    r.arms.iter().any(|a|arm_match(v,a,include_prerelease))
}
pub fn matches(v:&Version,r:&Range)->bool { matches_with(v,r,false) }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index:usize, pub error:VersionError }

fn extremum(candidates:&[String],r:&Range,max:bool)->Result<Option<String>,CandidateError>{
    let mut parsed=Vec::new();
    for (i,s) in candidates.iter().enumerate(){
        let v=parse_version(s).map_err(|error|CandidateError{index:i,error})?;
        parsed.push(v);
    }
    let mut best:Option<usize>=None;
    for i in 0..parsed.len(){
        if !matches(&parsed[i],r){continue;}
        if let Some(j)=best {
            let o=compare_versions(&parsed[i],&parsed[j]);
            if (max && o==Ordering::Greater)||(!max&&o==Ordering::Less){best=Some(i);}
        } else {best=Some(i);}
    }
    Ok(best.map(|i|candidates[i].clone()))
}
pub fn max_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extremum(candidates,r,true)}
pub fn min_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extremum(candidates,r,false)}

pub fn intersects(_a:&Range,_b:&Range,_include_prerelease:bool)->bool { false }

pub fn canonical(_r:&Range)->String { "EMPTY".to_string() }

#[derive(Clone, Debug)]
pub enum SetExpr { Range(String), Union(Vec<SetExpr>), Intersection(Vec<SetExpr>) }
#[derive(Clone, Debug)]
pub struct Constraint { pub label:String, pub expr:SetExpr }
#[derive(Clone, Debug)]
pub struct Package { pub name:String, pub candidates:Vec<String>, pub constraints:Vec<Constraint> }
#[derive(Clone, Copy, Debug)]
pub enum Preference { Highest, Lowest, StableFirst }
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection { pub package:String, pub version:String }
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name:String },
    DuplicateLabel { package:String, label:String },
    InvalidCandidate { package:String, index:usize, error:VersionError },
    InvalidRange { package:String, label:String, path:Vec<usize>, error:RangeError },
    NoSolution { package:String, labels:Vec<String> },
}

fn validate_expr(e:&SetExpr,package:&str,label:&str,path:&mut Vec<usize>)->Result<(),SelectError>{
    match e {
        SetExpr::Range(s)=>parse_range(s).map(|_|()).map_err(|error|SelectError::InvalidRange{package:package.into(),label:label.into(),path:path.clone(),error}),
        SetExpr::Union(v)|SetExpr::Intersection(v)=>{
            for (i,x) in v.iter().enumerate(){path.push(i);validate_expr(x,package,label,path)?;path.pop();}
            Ok(())
        }
    }
}
fn expr_match(e:&SetExpr,v:&Version,include:bool)->bool{
    match e {
        SetExpr::Range(s)=>parse_range(s).map(|r|matches_with(v,&r,include)).unwrap_or(false),
        SetExpr::Union(xs)=>xs.iter().any(|x|expr_match(x,v,include)),
        SetExpr::Intersection(xs)=>xs.iter().all(|x|expr_match(x,v,include)),
    }
}
fn conflict_labels(pkg:&Package,vers:&[Version],include:bool)->Vec<String>{
    let n=pkg.constraints.len();
    if vers.is_empty(){return vec![];}
    for size in 1..=n {
        let mut found:Option<Vec<String>>=None;
        fn rec(pkg:&Package,vers:&[Version],include:bool,size:usize,start:usize,chosen:&mut Vec<usize>,found:&mut Option<Vec<String>>){
            if found.is_some(){return;}
            if chosen.len()==size {
                let ok=vers.iter().any(|v|chosen.iter().all(|&i|expr_match(&pkg.constraints[i].expr,v,include)));
                if !ok {
                    let mut ls:Vec<String>=chosen.iter().map(|&i|pkg.constraints[i].label.clone()).collect();
                    ls.sort_by(|a,b|a.as_bytes().cmp(b.as_bytes()));
                    *found=Some(ls);
                }
                return;
            }
            for i in start..pkg.constraints.len(){
                chosen.push(i);rec(pkg,vers,include,size,i+1,chosen,found);chosen.pop();
            }
        }
        rec(pkg,vers,include,size,0,&mut Vec::new(),&mut found);
        if let Some(x)=found{return x;}
    }
    vec![]
}
pub fn select_packages(packages:&[Package],preference:Preference,include_prerelease:bool)->Result<Vec<Selection>,SelectError>{
    let mut names=std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()){return Err(SelectError::DuplicatePackage{name:p.name.clone()});}
    }
    for p in packages {
        let mut labels=std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()){return Err(SelectError::DuplicateLabel{package:p.name.clone(),label:c.label.clone()});}
        }
    }
    let mut all_versions=Vec::new();
    for p in packages {
        let mut vs=Vec::new();
        for (i,s) in p.candidates.iter().enumerate(){
            vs.push(parse_version(s).map_err(|error|SelectError::InvalidCandidate{package:p.name.clone(),index:i,error})?);
        }
        all_versions.push(vs);
    }
    for p in packages {
        for c in &p.constraints { validate_expr(&c.expr,&p.name,&c.label,&mut Vec::new())?; }
    }
    let mut out=Vec::new();
    for (pi,p) in packages.iter().enumerate(){
        let vs=&all_versions[pi];
        let mut eligible:Vec<usize>=vs.iter().enumerate().filter(|(_,v)|p.constraints.iter().all(|c|expr_match(&c.expr,v,include_prerelease))).map(|(i,_)|i).collect();
        if eligible.is_empty(){
            return Err(SelectError::NoSolution{package:p.name.clone(),labels:conflict_labels(p,vs,include_prerelease)});
        }
        if preference==Preference::StableFirst {
            let stable:Vec<usize>=eligible.iter().copied().filter(|&i|vs[i].pre.is_empty()).collect();
            if !stable.is_empty(){eligible=stable;}
        }
        let mut best=eligible[0];
        for &i in eligible.iter().skip(1){
            let o=compare_versions(&vs[i],&vs[best]);
            let better=match preference {Preference::Lowest=>o==Ordering::Less,_=>o==Ordering::Greater};
            if better {best=i;}
        }
        out.push(Selection{package:p.name.clone(),version:p.candidates[best].clone()});
    }
    Ok(out)
}
