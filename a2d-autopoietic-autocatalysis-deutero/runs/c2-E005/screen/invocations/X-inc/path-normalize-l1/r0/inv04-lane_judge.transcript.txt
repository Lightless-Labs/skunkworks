#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = path.as_bytes()[0] == b'/';
    let directory_intent = path.ends_with('/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..");

    let mut components: Vec<(String, usize)> = Vec::new();
    let mut offset = 0;

    for component in path.split('/') {
        let component_at = offset;
        offset += component.len() + 1;

        if component.is_empty() || component == "." {
            continue;
        }

        if component == ".." {
            if components
                .last()
                .is_some_and(|(previous, _)| previous != "..")
            {
                components.pop();
            } else if absolute {
                return Err(PathError::Escape { at: component_at });
            } else {
                components.push(("..".to_owned(), component_at));
            }
        } else {
            components.push((component.to_owned(), component_at));
        }
    }

    let joined = components
        .iter()
        .map(|(component, _)| component.as_str())
        .collect::<Vec<_>>()
        .join("/");

    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_owned()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_owned()
    } else {
        joined
    };

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}

    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = path.as_bytes()[0] == b'/';
    let trailing_slash = path.as_bytes().last() == Some(&b'/');
    let mut components: Vec<(&str, usize)> = Vec::new();
    let mut last_nonempty: Option<&str> = None;

    let mut start = 0;
    while start < path.len() {
        if path.as_bytes()[start] == b'/' {
            start += 1;
            continue;
        }

        let mut end = start;
        while end < path.len() && path.as_bytes()[end] != b'/' {
            end += 1;
        }

        let component = &path[start..end];
        last_nonempty = Some(component);

        match component {
            "." => {}
            ".." => {
                if components.last().is_some_and(|(name, _)| *name != "..") {
                    components.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: start });
                } else {
                    components.push((component, start));
                }
            }
            _ => components.push((component, start)),
        }

        start = end;
    }

    let directory_intent = trailing_slash || matches!(last_nonempty, Some("." | ".."));
    let mut result = String::new();

    if absolute {
        result.push('/');
    }

    for (index, (component, _)) in components.iter().enumerate() {
        if index > 0 {
            result.push('/');
        }
        result.push_str(component);
    }

    if result.is_empty() {
        result.push('.');
    }

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}

    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = path.as_bytes()[0] == b'/';
    let directory_intent = path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .next_back()
            .is_some_and(|component| component == "." || component == "..");

    let mut components: Vec<(String, usize)> = Vec::new();
    let mut offset = 0;

        let component_offset = offset;
        offset += component.len() + 1;

        if component.is_empty() || component == "." {
            continue;
        }

        if component == ".." {
            if components
                .last()
                .is_some_and(|(previous, _)| previous != "..")
            {
                components.pop();
            } else if absolute {
                return Err(PathError::Escape {
                    at: component_offset,
                });
            } else {
                components.push((component.to_owned(), component_offset));
            }
        } else {
            components.push((component.to_owned(), component_offset));
        }
    }

    let mut result = components
        .into_iter()
        .map(|(component, _)| component)
        .collect::<Vec<_>>()
        .join("/");

        result.insert(0, '/');
        if result == "/" {
            return Ok(result);
        }
    } else if result.is_empty() {
        return Ok(".".to_owned());
    }

    if directory_intent {
        result.push('/');
    }

    Ok(result)
