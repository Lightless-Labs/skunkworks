use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = n.checked_neg().ok_or(RationalError::Overflow)?;
        d = d.checked_neg().ok_or(RationalError::Overflow)?;
    }

    let g = gcd_u128(n.unsigned_abs(), d as u128);
    let reduced_n = n / g as i128;
    let reduced_d = d / g as i128;

    if reduced_n < i64::MIN as i128 || reduced_n > i64::MAX as i128
        || reduced_d < 1 || reduced_d > i64::MAX as i128
    {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: reduced_n as i64,
        den: reduced_d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let mut n = num as i128;
        let mut d = den as i128;
        if d < 0 {
            n = -n;
            d = -d;
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128);
        rational_from_i128(n / g as i128, d / g as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = self.den as i128 / g;
        let b = rhs.den as i128 / g;
        let n = self.num as i128 * b + rhs.num as i128 * a;
        let d = a * rhs.den as i128;
        rational_from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = self.den as i128 / g;
        let b = rhs.den as i128 / g;
        let n = self.num as i128 * b - rhs.num as i128 * a;
        let d = a * rhs.den as i128;
        rational_from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.den as u128);
        let g2 = gcd_u128(rhs.num.unsigned_abs() as u128, self.den as u128);
        let n = (self.num as i128 / g1 as i128) * (rhs.num as i128 / g2 as i128);
        let d = (self.den as i128 / g2 as i128) * (rhs.den as i128 / g1 as i128);
        rational_from_i128(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.num.unsigned_abs() as u128);
        let g2 = gcd_u128(self.den as u128, rhs.den as u128);
        let mut n = (self.num as i128 / g1 as i128) * (rhs.den as i128 / g2 as i128);
        let mut d = (self.den as i128 / g2 as i128) * (rhs.num as i128 / g1 as i128);
        if d < 0 {
            n = -n;
            d = -d;
        }
        rational_from_i128(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = self.num.checked_neg().ok_or(RationalError::Overflow)?;
        Ok(Self { num: n, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(parse_err(i, if i == bytes.len() { ParseKind::InvalidByte } else { ParseKind::InvalidByte }));
        }

        let slash = if i < bytes.len() && bytes[i] == b'/' {
            let s = i;
            i += 1;
            Some(s)
        } else {
            None
        };

        let den_start = if slash.is_some() { i } else { bytes.len() };
        if slash.is_some() {
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }
        if i < bytes.len() {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let num_end = slash.unwrap_or(bytes.len());
        let num_token = &bytes[..num_end];
        let digits_start = usize::from(bytes[0] == b'-');
        if num_end - digits_start >= 2 && bytes[digits_start] == b'0' {
            return Err(parse_err(digits_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && num_end == 2 && bytes[1] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        if let Some(slash_at) = slash {
            if bytes.len() - den_start >= 2 && bytes[den_start] == b'0' {
                return Err(parse_err(den_start + 1, ParseKind::NonCanonical));
            }
        }

        let num_digits = &num_token[digits_start..];
        let num_limit: u128 = if bytes[0] == b'-' { 1u128 << 63 } else { i64::MAX as u128 };
        let mut num_mag = 0u128;
        for &b in num_digits {
            num_mag = num_mag.saturating_mul(10).saturating_add((b - b'0') as u128);
            if num_mag > num_limit {
                return Err(parse_err(0, ParseKind::OutOfRange));
            }
        }

        let mut den_val = 1u128;
        if slash.is_some() {
            den_val = 0;
            for &b in &bytes[den_start..] {
                den_val = den_val.saturating_mul(10).saturating_add((b - b'0') as u128);
                if den_val > i64::MAX as u128 {
                    return Err(parse_err(den_start, ParseKind::OutOfRange));
                }
            }
            if den_val == 0 {
                return Err(parse_err(den_start, ParseKind::ZeroDenominator));
            }
            if den_val == 1 {
                return Err(parse_err(slash.unwrap(), ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_err(slash.unwrap(), ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den_val) > 1 {
                return Err(parse_err(slash.unwrap(), ParseKind::NonCanonical));
            }
        }

        let signed_num = if bytes[0] == b'-' {
            if num_mag == (1u128 << 63) { i64::MIN } else { -(num_mag as i64) }
        } else {
            num_mag as i64
        };
        Self::new(signed_num, den_val as i64)
            .map_err(|_| parse_err(0, ParseKind::OutOfRange))
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    fn eliminate(&mut self, candidate_cols: usize) -> Result<(Vec<usize>, Vec<(usize, usize)>), MatrixError> {
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..candidate_cols {
            if p == self.rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for row in p..self.rows {
                let v = self.data[row * self.cols + col];
                if v.num == 0 {
                    continue;
                }
                let key = (v.den, v.num.unsigned_abs(), row);
                if chosen.map_or(true, |old| {
                    let ov = self.data[old * self.cols + col];
                    key < (ov.den, ov.num.unsigned_abs(), old)
                }) {
                    chosen = Some(row);
                }
            }
            let Some(q) = chosen else { continue };
            if q != p {
                for c in 0..self.cols {
                    self.data.swap(p * self.cols + c, q * self.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = self.data[p * self.cols + col];
            for c in 0..self.cols {
                let current = self.data[p * self.cols + c];
                self.data[p * self.cols + c] = current.div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic { row: p, col: c, step: p, source })?;
            }

            for row in 0..self.rows {
                if row == p {
                    continue;
                }
                let factor = self.data[row * self.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..self.cols {
                    let product = factor.mul(&self.data[p * self.cols + c])
                        .map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                    self.data[row * self.cols + c] = self.data[row * self.cols + c].sub(&product)
                        .map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                }
            }

            pivots.push(col);
            p += 1;
        }
        Ok((pivots, swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let (pivots, swaps) = matrix.eliminate(self.cols)?;
        Ok(Reduction { matrix, pivots, swaps })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..width {
                if c < n {
                    data.push(self.data[r * n + c]);
                } else {
                    data.push(if c - n == r { Rational { num: 1, den: 1 } } else { Rational { num: 0, den: 1 } });
                }
            }
        }
        let mut augmented = Self { rows: n, cols: width, data };
        let (pivots, _) = augmented.eliminate(n)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }
        let mut inverse_data = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                inverse_data.push(augmented.data[r * width + n + c]);
            }
        }
        Ok(Self { rows: n, cols: n, data: inverse_data })
    }
}
