fn parse_atom(bytes: &[u8], field: Field, offset: usize) -> Result<u32, ParseError> {
    if bytes.is_empty() {
        return Err(syntax_error(field, offset));
    }

    if bytes.iter().all(u8::is_ascii_digit) {
        if bytes.len() > 1 && bytes[0] == b'0' {
            return Err(ParseError::LeadingZero { field, offset });
        }

        for &byte in bytes {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((byte - b'0') as u32))
                .ok_or(ParseError::NumberOverflow { field, offset })?;
        }
        let (min, max) = domain(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
        return Ok(value);
    }

        let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();

                b"FEB" => Some(2),
                b"MAR" => Some(3),
                b"APR" => Some(4),

                b"AUG" => Some(8),

                b"OCT" => Some(10),

                b"MON" => Some(1),

                b"THU" => Some(4),
                b"FRI" => Some(5),
                b"SAT" => Some(6),
                _ => None,
            },
            _ => None,
        };
        return value.ok_or_else(|| syntax_error(field, offset));
    }

    Err(syntax_error(field, offset))
}

fn parse_step(bytes: &[u8], field: Field, offset: usize) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(syntax_error(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    let (min, max) = domain(field);
    let width = max - min + 1;
    if value == 0 || value > width {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(value)
}

fn parse_item(item: &[u8], field: Field, item_offset: usize, selected: &mut [bool]) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem { field, offset: item_offset });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(syntax_error(field, item_offset));
    }
    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), item_offset + slash + 1)
    } else {
        (item, None, item_offset)
    };

    let (min, max) = domain(field);
    let mut start = min;
    let mut end = max;
    let mut wildcard = false;

    if base == b"*" {
        wildcard = true;
    } else {
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();
        if hyphens.len() > 1 {
            return Err(syntax_error(field, item_offset));
        }
        if let Some(&hyphen) = hyphens.first() {
            let left = &base[..hyphen];
            let right = &base[hyphen + 1..];
            if left.is_empty() || right.is_empty() || left == b"*" || right == b"*" {
                return Err(syntax_error(field, item_offset));
            }
            let left_numeric = left.iter().all(u8::is_ascii_digit);
            let right_numeric = right.iter().all(u8::is_ascii_digit);
            let left_name = left.iter().all(u8::is_ascii_alphabetic);
            let right_name = right.iter().all(u8::is_ascii_alphabetic);
            if !(left_numeric || left_name) || !(right_numeric || right_name) {
                return Err(syntax_error(field, item_offset));
            }
            start = parse_atom(left, field, item_offset)?;
            let end_offset = item_offset + hyphen + 1;
            end = parse_atom(right, field, end_offset)?;
            if start > end {
                return Err(ParseError::ReversedRange { field, offset: end_offset });
            }
        } else {
            let numeric = base.iter().all(u8::is_ascii_digit);
            let name = base.iter().all(u8::is_ascii_alphabetic);
            if !(numeric || name) {
                return Err(syntax_error(field, item_offset));
            }
            start = parse_atom(base, field, item_offset)?;
            end = if step_bytes.is_some() { max } else { start };
        }
    }

    let step = match step_bytes {
        Some(bytes) => Some(parse_step(bytes, field, step_offset)?),
        None => None,
    };
    let stride = step.unwrap_or(1);
    let mut value = start;
    while value <= end {
        selected[(value - min) as usize] = true;
        match value.checked_add(stride) {
            Some(next) => value = next,
            None => break,
        }
    }
    let _ = wildcard;
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = domain(field);
    let mut selected = vec![false; (max - min + 1) as usize];
    let mut offset = 0usize;
    for item in bytes.split(|&b| b == b',') {
        parse_item(item, field, offset, &mut selected)?;
        offset += item.len() + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }

    let mut sets = Vec::with_capacity(5);
    for (index, field) in FIELDS.iter().copied().enumerate() {
        sets.push(parse_field(pieces[index], field)?);
    }

    Ok(Cron {
        sets: sets.try_into().ok().unwrap(),
        unrestricted_days: [pieces[2] == "*", pieces[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, _) = domain(field);
        self.sets[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &selected)| selected.then_some(min as u8 + i as u8))
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, max) = domain(field);
        value as u32 >= min
            && (value as u32) <= max
            && self.sets[index][value as usize - min as usize]
    }

    fn validate(at: CivilDateTime) -> Result<(), DateError> {
        if !(2000..=2399).contains(&at.year) {
            return Err(DateError::InvalidDateTime { component: DatePart::Year });
        }
        if !(1..=12).contains(&at.month) {
            return Err(DateError::InvalidDateTime { component: DatePart::Month });
        }
        let last = days_in_month(at.year, at.month);
        if at.day == 0 || at.day > last {
            return Err(DateError::InvalidDateTime { component: DatePart::Day });
        }
        if at.hour > 23 {
            return Err(DateError::InvalidDateTime { component: DatePart::Hour });
        }
        if at.minute > 59 {
            return Err(DateError::InvalidDateTime { component: DatePart::Minute });
        }
        Ok(())
    }

    fn date_matches(&self, at: CivilDateTime) -> bool {
        if !self.contains(Field::Month, at.month) {
            return false;
        }
        let dom = self.contains(Field::DayOfMonth, at.day);
        let dow = self.contains(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn datetime_matches_valid(&self, at: CivilDateTime) -> bool {
        self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.date_matches(at)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        Self::validate(at)?;
        Ok(self.datetime_matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        Self::validate(after)?;

        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }

            current = advance_minute(current);
            if self.datetime_matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn advance_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
    } else {
        at.minute = 0;
        if at.hour < 23 {
            at.hour += 1;
        } else {
            at.hour = 0;
            if at.day < days_in_month(at.year, at.month) {
                at.day += 1;
            } else {
                at.day = 1;
                if at.month < 12 {
                    at.month += 1;
                } else {
                    at.month = 1;
                    at.year += 1;
                }
            }
        }
    }
    at
}

            (b"JAN", 1), (b"FEB", 2), (b"MAR", 3), (b"APR", 4),
            (b"MAY", 5), (b"JUN", 6), (b"JUL", 7), (b"AUG", 8),
            (b"SEP", 9), (b"OCT", 10), (b"NOV", 11), (b"DEC", 12),
        ],
        Field::DayOfWeek => &[
            (b"SUN", 0), (b"MON", 1), (b"TUE", 2), (b"WED", 3),
            (b"THU", 4), (b"FRI", 5), (b"SAT", 6),
        ],
        _ => return None,
    };

    names.iter().find_map(|(name, value)| {
        if bytes.len() != name.len() {
            return None;
        }
        bytes
            .iter()
            .zip(name.iter())
            .all(|(a, b)| a.to_ascii_uppercase() == *b)
            .then_some(*value)
    })
}

fn atom_syntax_ok(field: Field, bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    if bytes.iter().all(u8::is_ascii_digit) {
        return true;
    }
    name_value(field, bytes).is_some()
}

fn parse_number(field: Field, bytes: &[u8], offset: usize) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn endpoint_value(
    field: Field,
    bytes: &[u8],
    offset: usize,
) -> Result<u32, ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) {
        let value = parse_number(field, bytes, offset)?;
        let (min, max, _) = bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
        Ok(value)
    } else {
        name_value(field, bytes)
            .ok_or(ParseError::InvalidSyntax { field, offset })
    }
}

fn parse_item(
    field: Field,
    item: &[u8],
    item_offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slashes: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    let (base, step_part, step_offset) = match slashes.as_slice() {
        [] => (item, None, 0),
        [slash] => (&item[..*slash], Some(&item[*slash + 1..]), slash + 1),
        _ => {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
    };

    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    let base_shape_valid = match hyphens.as_slice() {
        [] => base == b"*" || atom_syntax_ok(field, base),
        [hyphen] => {
            base[0..*hyphen] != *b"*"
                && base[*hyphen + 1..] != *b"*"
                && atom_syntax_ok(field, &base[..*hyphen])
                && atom_syntax_ok(field, &base[*hyphen + 1..])
        }
        _ => false,
    };

    if !base_shape_valid || base.is_empty() {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }
    if step_part.is_some_and(|part| part.is_empty() || !part.iter().all(u8::is_ascii_digit)) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (start, range_end, end_offset) = match hyphens.as_slice() {
        [] if base == b"*" => {
            let (min, max, _) = bounds(field);
            (min, Some(max), item_offset)
        }
        [] => {
            let value = endpoint_value(field, base, item_offset)?;
            (value, None, item_offset)
        }
        [hyphen] => {
            let start = endpoint_value(field, &base[..*hyphen], item_offset)?;
            let end_at = item_offset + hyphen + 1;
            let end = endpoint_value(field, &base[*hyphen + 1..], end_at)?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_at,
                });
            }
            (start, Some(end), end_at)
        }
        _ => unreachable!(),
    };

    let step = if let Some(part) = step_part {
        let step_at = item_offset + step_offset;
        if !part.iter().all(u8::is_ascii_digit) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        let value = parse_number(field, part, step_at)?;
        let (_, _, width) = bounds(field);
        if value == 0 || value > width as u32 {
            return Err(ParseError::BadStep {
                field,
                offset: step_at,
            });
        }
        Some(value)
    } else {
        None
    };

    let (_, domain_max, _) = bounds(field);
    let end = match (step, range_end) {
        (Some(_), None) => domain_max,
        (_, Some(value)) => value,
        (None, None) => start,
    };

    let stride = step.unwrap_or(1);
    let mut value = start;
    while value <= end {
        selected[value as usize] = true;
        match value.checked_add(stride) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }

    let _ = end_offset;
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = [
        vec![false; 60],
        vec![false; 24],
        vec![false; 32],
        vec![false; 13],
        vec![false; 7],
    ];
    let mut unrestricted_days = [false; 2];

    for (index, field_text) in parts.iter().enumerate() {
        let field = FIELDS[index];
        for (offset, &byte) in field_text.as_bytes().iter().enumerate() {
            if !(byte.is_ascii_digit()
                || byte.is_ascii_alphabetic()
                || matches!(byte, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        if index == 2 {
            unrestricted_days[0] = *field_text == "*";
        } else if index == 4 {
            unrestricted_days[1] = *field_text == "*";
        }

        let mut offset = 0usize;
        for item in field_text.split(',') {
            parse_item(field, item.as_bytes(), offset, &mut selected[index])?;
            offset += item.len() + 1;
        }
    }

    Ok(Cron {
        selected,
        unrestricted_days,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &is_selected)| is_selected.then_some(value as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;

        let weekday = weekday(at.year, at.month, at.day);
        let day_of_month = self.selected[2][at.day as usize];
        let day_of_week = self.selected[4][weekday as usize];

        let day_matches = match self.unrestricted_days {
            [true, true] => true,
            [false, true] => day_of_month,
            [true, false] => day_of_week,
            [false, false] => {
                if at.day == days_in_month(at.year, at.month) {
                    day_of_month && day_of_week
                } else {
                    day_of_month || day_of_week
                }
            }
        };

        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[3][at.month as usize]
            && day_matches)
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }

            current = advance_minute(current);
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if !(1..=days_in_month(at.year, at.month)).contains(&at.day) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn advance_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}

#[derive(Debug, Clone, PartialEq, Eq)]

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,

fn domain(field: Field) -> (u32, u32, usize) {
    match field {
        Field::Minute => (0, 59, 60),
        Field::Hour => (0, 23, 24),
        Field::DayOfMonth => (1, 31, 31),
        Field::Month => (1, 12, 12),
        Field::DayOfWeek => (0, 6, 7),
    }
}

fn name_value(field: Field, text: &str) -> Option<u32> {
    let names: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };

    names
        .iter()
        .position(|name| name.eq_ignore_ascii_case(text))
        .map(|i| i as u32 + if field == Field::Month { 1 } else { 0 })
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(
    field: Field,
    text: &str,
    offset: usize,
    is_step: bool,
) -> Result<u32, ParseError> {
    let bytes = text.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(invalid_syntax(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add(u32::from(byte - b'0')))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    if is_step {
        let width = domain(field).2 as u32;
        if value == 0 || value > width {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (min, max, _) = domain(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }
    Ok(value)
}

fn parse_endpoint(
    field: Field,
    text: &str,
    offset: usize,
) -> Result<u32, ParseError> {
    if text.as_bytes().iter().all(u8::is_ascii_digit) && !text.is_empty() {
        return parse_number(field, text, offset, false);
    }
    name_value(field, text).ok_or_else(|| invalid_syntax(field, offset))
}

fn parse_item(
    field: Field,
    item: &str,
    offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(invalid_syntax(field, offset));
    }

    let (base, step_text, step_offset) = if let Some(&slash) = slash_positions.first() {
        let step = &item[slash + 1..];
        if step.is_empty() {
            return Err(invalid_syntax(field, offset));
        }
        ( &item[..slash], Some(step), offset + slash + 1 )
    } else {
        (item, None, 0)
    };

    let (min, max, _) = domain(field);
    let mut start = min;
    let mut end = max;

    if base != "*" {
        let hyphens: Vec<usize> = base
            .as_bytes()
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        if hyphens.len() > 1 {
            return Err(invalid_syntax(field, offset));
        }

        if let Some(&hyphen) = hyphens.first() {
            let left = &base[..hyphen];
            let right = &base[hyphen + 1..];
            if left.is_empty() || right.is_empty() || left == "*" || right == "*" {
                return Err(invalid_syntax(field, offset));
            }
            let left_valid = left.as_bytes().iter().all(u8::is_ascii_digit)
                || name_value(field, left).is_some();
            let right_valid = right.as_bytes().iter().all(u8::is_ascii_digit)
                || name_value(field, right).is_some();
            if !left_valid || !right_valid {
                return Err(invalid_syntax(field, offset));
            }

            let left_value = parse_endpoint(field, left, offset)?;
            let right_value = parse_endpoint(field, right, offset + hyphen + 1)?;
            if left_value > right_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: offset + hyphen + 1,
                });
            }
            start = left_value;
            end = right_value;
        } else {
            let is_valid = (!base.is_empty()
                && base.as_bytes().iter().all(u8::is_ascii_digit))
                || name_value(field, base).is_some();
            if !is_valid {
                return Err(invalid_syntax(field, offset));
            }
            start = parse_endpoint(field, base, offset)?;
            end = if step_text.is_some() { max } else { start };
        }
    }

    let step = if let Some(text) = step_text {
        Some(parse_number(field, text, step_offset, true)?)
    } else {
        None
    };

    let increment = step.unwrap_or(1);
    let mut value = start;
    while value <= end {
        let index = (value - min) as usize;
        selected[index] = true;
        match value.checked_add(increment) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = [
        vec![false; 60],
        vec![false; 24],
        vec![false; 31],
        vec![false; 12],
        vec![false; 7],
    ];

    let mut unrestricted_days = [false; 2];

    for (field_index, (&field, text)) in FIELDS.iter().zip(parts.iter()).enumerate() {
        for (offset, byte) in text.bytes().enumerate() {
            let legal = byte.is_ascii_digit()
                || byte.is_ascii_alphabetic()
                || matches!(byte, b'*' | b'-' | b'/' | b',');
            if !legal {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        if field == Field::DayOfMonth {
            unrestricted_days[0] = *text == "*";
        } else if field == Field::DayOfWeek {
            unrestricted_days[1] = *text == "*";
        }

        let mut item_offset = 0usize;
        for item in text.split(',') {
            if item.is_empty() {
                return Err(ParseError::EmptyItem {
                    field,
                    offset: item_offset,
                });
            }
            parse_item(field, item, item_offset, &mut selected[field_index])?;
            item_offset += item.len() + 1;
        }
    }

    Ok(Cron {
        selected,
        unrestricted_days,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, _, _) = domain(field);
        let values = &self.selected[field_index(field)];
        values
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(min as u8 + i as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        if after == max_datetime() {
            return Ok(None);
        }

        let mut current = after;
        loop {
            current = advance_minute(current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
            if current == max_datetime() {
                return Ok(None);
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        let minute_ok = self.selected[0][at.minute as usize];
        let hour_ok = self.selected[1][at.hour as usize];
        let month_ok = self.selected[3][at.month as usize - 1];
        if !(minute_ok && hour_ok && month_ok) {
            return false;
        }

        let dom = self.selected[2][at.day as usize - 1];
        let weekday = weekday(at.year, at.month, at.day);
        let dow = self.selected[4][weekday as usize];

        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn days_in_month(year: u16, month: u8) -> u8 {

fn weekday(year: u16, month: u8, day: u8) -> u8 {

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn max_datetime() -> CivilDateTime {
    CivilDateTime {
        year: 2399,
        month: 12,
        day: 31,
        hour: 23,
        minute: 59,
    }
}

fn advance_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32, usize) {
    match field {
        Field::Minute => (0, 59, 60),
        Field::Hour => (0, 23, 24),
        Field::DayOfMonth => (1, 31, 31),
        Field::Month => (1, 12, 12),
        Field::DayOfWeek => (0, 6, 7),
    }
}

fn name_value(field: Field, text: &str) -> Option<u32> {
    let upper = text.to_ascii_uppercase();
    match field {
        Field::Month => match upper.as_str() {
            "JAN" => Some(1), "FEB" => Some(2), "MAR" => Some(3),
            "APR" => Some(4), "MAY" => Some(5), "JUN" => Some(6),
            "JUL" => Some(7), "AUG" => Some(8), "SEP" => Some(9),
            "OCT" => Some(10), "NOV" => Some(11), "DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_str() {
            "SUN" => Some(0), "MON" => Some(1), "TUE" => Some(2),
            "WED" => Some(3), "THU" => Some(4), "FRI" => Some(5),
            "SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

fn parse_atom(
    field: Field,
    text: &str,
    offset: usize,
    is_step: bool,
) -> Result<u32, ParseError> {
    let bytes = text.as_bytes();
    if bytes.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if bytes.iter().all(u8::is_ascii_digit) {
        if bytes.len() > 1 && bytes[0] == b'0' {
            return Err(ParseError::LeadingZero { field, offset });
        }
        let mut value = 0u32;
        for &b in bytes {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or(ParseError::NumberOverflow { field, offset })?;
        }
        if is_step {
            let (_, _, width) = bounds(field);
            if value == 0 || value > width as u32 {
                return Err(ParseError::BadStep { field, offset });
            }
        } else {
            let (min, max, _) = bounds(field);
            if value < min || value > max {
                return Err(ParseError::OutOfRange { field, offset });
            }
        }
        return Ok(value);
    }
    if !is_step {
        if let Some(value) = name_value(field, text) {
            return Ok(value);
        }
    }
    Err(ParseError::InvalidSyntax { field, offset })
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    for (i, &b) in text.as_bytes().iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (min, max, width) = bounds(field);
    let mut selected = vec![false; width];
    let mut item_start = 0usize;

    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field, offset: item_start });
        }

        let slash_count = item.bytes().filter(|&b| b == b'/').count();
        let slash_at = item.find('/');
        let base = slash_at.map_or(item, |i| &item[..i]);
        let step_text = slash_at.map(|i| &item[i + 1..]);

        let hyphen_count = base.bytes().filter(|&b| b == b'-').count();
        let (left, right) = if base == "*" {
            (None, None)
        } else if hyphen_count == 0 {
            (Some(base), None)
        } else if hyphen_count == 1 {
            let i = base.find('-').unwrap();
            let l = &base[..i];
            let r = &base[i + 1..];
            if l.is_empty() || r.is_empty() || l == "*" || r == "*" {
                return Err(ParseError::InvalidSyntax { field, offset: item_start });
            }
            (Some(l), Some(r))
        } else {
            return Err(ParseError::InvalidSyntax { field, offset: item_start });
        };

        if slash_count > 1
            || base.is_empty()
            || step_text.is_some_and(|s| s.is_empty())
            || (base != "*" && left.is_none())
        {
            return Err(ParseError::InvalidSyntax { field, offset: item_start });
        }

        let mut start = min;
        let mut end = max;
        let mut number_offsets = Vec::new();

        if let Some(l) = left {
            let loff = item_start;
            start = parse_atom(field, l, loff, false)?;
            number_offsets.push(loff);
            if let Some(r) = right {
                let roff = item_start + l.len() + 1;
                end = parse_atom(field, r, roff, false)?;
                number_offsets.push(roff);
                if start > end {
                    return Err(ParseError::ReversedRange { field, offset: roff });
                }
            } else if step_text.is_none() {
                end = start;
            }
        }

        let step = if let Some(s) = step_text {
            let off = item_start + base.len() + 1;
            Some(parse_atom(field, s, off, true)?)
        } else {
            None
        };

        let stride = step.unwrap_or(1);
        let mut value = start;
        while value <= end {
            selected[(value - min) as usize] = true;
            match value.checked_add(stride) {
                Some(next) => value = next,
                None => break,
            }
        }

        item_start += item.len() + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected = Vec::with_capacity(5);
    for (field, text) in FIELDS.into_iter().zip(parts.iter().copied()) {
        selected.push(parse_field(field, text)?);
    }

    Ok(Cron {
        selected: selected.try_into().unwrap(),
        dom_unrestricted: parts[2] == "*",
        dow_unrestricted: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, _, _) = bounds(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(min as u8 + i as u8))
            .collect()
    }

    fn has(&self, field: Field, value: u8) -> bool {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, _, _) = bounds(field);
        self.selected[index][(value as u32 - min) as usize]
    }

    fn date_matches(&self, at: CivilDateTime, weekday: u8) -> bool {
        let dom = self.has(Field::DayOfMonth, at.day);
        let dow = self.has(Field::DayOfWeek, weekday);
        match (self.dom_unrestricted, self.dow_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.has(Field::Minute, at.minute)
            && self.has(Field::Hour, at.hour)
            && self.has(Field::Month, at.month)
            && self.date_matches(at, weekday))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
            increment_minute(&mut current);
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn increment_minute(at: &mut CivilDateTime) {
    if at.minute < 59 {
        at.minute += 1;
        return;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
    } else {
        at.month = 1;
        at.year += 1;
    }
}
