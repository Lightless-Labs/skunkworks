use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<Run>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while input.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn clean_cfws(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut out = Cleaned {
        bytes: Vec::with_capacity(input.len()),
        positions: Vec::with_capacity(input.len()),
        runs: Vec::new(),
    };
    let mut i = 0;
    let mut comment_depth = 0usize;
    let mut outer_open = 0usize;
    let mut in_quote = false;
    let mut quote_escape = false;
    let mut run_start: Option<usize> = None;

    while i < input.len() {
        let b = input[i];

        if in_quote {
            out.bytes.push(b);
            out.positions.push(i);
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                in_quote = false;
            }
            i += 1;
            continue;
        }

        if comment_depth > 0 {
            if b == b'\\' {
                if run_start.is_none() {
                    run_start = Some(i);
                }
                if i + 1 >= input.len()
                    || !(0x20..=0x7e).contains(&input[i + 1])
                {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    if run_start.is_none() {
                        run_start = Some(i);
                    }
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                comment_depth -= 1;
                i += 1;
                if comment_depth == 0 {
                    let start = run_start.take().unwrap_or(outer_open);
                    out.runs.push(Run { start, end: i });
                }
                continue;
            }
            if !(b == b'\t' || (0x20..=0x7e).contains(&b)) {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        let is_first_retained_quote = out.bytes.is_empty() && b == b'"';
        if is_first_retained_quote {
            if let Some(start) = run_start.take() {
                out.runs.push(Run { start, end: i });
            }
            in_quote = true;
            out.bytes.push(b);
            out.positions.push(i);
            i += 1;
            continue;
        }

        if b == b' ' || b == b'\t' {
            if run_start.is_none() {
                run_start = Some(i);
            }
            i += 1;
            continue;
        }
        if b == b'\r' || b == b'\n' {
            if let Some(end) = fold_end(input, i) {
                if run_start.is_none() {
                    run_start = Some(i);
                }
                i = end;
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }
        if b == b'(' {
            if run_start.is_none() {
                run_start = Some(i);
            }
            outer_open = i;
            comment_depth = 1;
            i += 1;
            continue;
        }
        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if let Some(start) = run_start.take() {
            out.runs.push(Run { start, end: i });
        }
        out.bytes.push(b);
        out.positions.push(i);
        i += 1;
    }

    if comment_depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }
    if let Some(start) = run_start {
        out.runs.push(Run {
            start,
            end: input.len(),
        });
    }
    Ok(out)
}

fn source_pos(cleaned: &Cleaned, index: usize, input_len: usize) -> usize {
    cleaned.positions.get(index).copied().unwrap_or(input_len)
}

fn local_dot_atom_valid(local: &[u8]) -> bool {
    if local.is_empty() {
        return false;
    }
    let allowed = b"!#$%&'*+-/=?^_`{|}~";
    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return false;
            }
        } else if !b.is_ascii_alphanumeric() && !allowed.contains(&b) {
            return false;
        }
    }
    true
}

fn parse_core(input: &str, allow_single_label: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = clean_cfws(raw)?;
    let bytes = &cleaned.bytes;

    for run in &cleaned.runs {
        let before = cleaned.positions.iter().position(|&p| p >= run.start);
        let after = cleaned.positions.iter().position(|&p| p >= run.end);
        let left_is_at = before
            .and_then(|idx| idx.checked_sub(1))
            .is_some_and(|idx| bytes[idx] == b'@');
        let right_is_at = after.is_some_and(|idx| bytes[idx] == b'@');
        let at_boundary = before.is_none() || after == Some(0);
        if !(left_is_at || right_is_at || at_boundary) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i = i.saturating_add(2);
            } else if bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: source_pos(&cleaned, 0, raw.len()),
            });
        }
    }

    let mut separators = Vec::new();
    let mut i = 0;
    if let Some(end) = quote_end {
        i = end + 1;
    }
    while i < bytes.len() {
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: source_pos(&cleaned, separators[1], raw.len()),
        });
    }
    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: source_pos(&cleaned, sep, raw.len()),
        });
    }

    let mut local = Vec::new();
    let mut local_origins = Vec::new();
    let quoted = bytes[0] == b'"';
    if !quoted {
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: source_pos(&cleaned, 64, raw.len()),
            });
        }
        let allowed = b"!#$%&'*+-/=?^_`{|}~";
        for j in 0..sep {
            let b = bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == sep || bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: source_pos(&cleaned, j, raw.len()),
                    });
                }
            } else if !b.is_ascii_alphanumeric() && !allowed.contains(&b) {
                return Err(EmailError::LocalChar {
                    at: source_pos(&cleaned, j, raw.len()),
                });
            }
            local.push(b);
            local_origins.push(source_pos(&cleaned, j, raw.len()));
        }
    } else {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: source_pos(&cleaned, end + 1, raw.len()),
            });
        }
        let mut j = 1;
        while j < end {
            let b = bytes[j];
            if b == b'\\' {
                if j + 1 >= end {
                    return Err(EmailError::BadEscape {
                        at: source_pos(&cleaned, j, raw.len()),
                    });
                }
                let escaped = bytes[j + 1];
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape {
                        at: source_pos(&cleaned, j, raw.len()),
                    });
                }
                local.push(escaped);
                local_origins.push(source_pos(&cleaned, j, raw.len()));
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: source_pos(&cleaned, j, raw.len()),
                    });
                }
                local.push(b);
                local_origins.push(source_pos(&cleaned, j, raw.len()));
                j += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: source_pos(&cleaned, end, raw.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_origins[64],
            });
        }
    }

    let domain_start = sep + 1;
    let mut domain_end = bytes.len();
    let mut root_dot_pos = None;
    if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_dot_pos = Some(source_pos(&cleaned, domain_end, raw.len()));
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: source_pos(&cleaned, domain_start, raw.len()),
        });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: source_pos(&cleaned, domain_start + 253, raw.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for j in domain_start..domain_end {
        if bytes[j] == b'.' {
            labels.push((label_start, j));
            label_start = j + 1;
        }
    }
    labels.push((label_start, domain_end));

    for (start, end) in &labels {
        if start == end {
            let at = if *start == domain_end {
                root_dot_pos.unwrap_or_else(|| source_pos(&cleaned, *start, raw.len()))
            } else {
                source_pos(&cleaned, *start, raw.len())
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: source_pos(&cleaned, start + 63, raw.len()),
            });
        }
        if bytes[*start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: source_pos(&cleaned, *start, raw.len()),
            });
        }
        if bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: source_pos(&cleaned, end - 1, raw.len()),
            });
        }
        for j in *start..*end {
            let b = bytes[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: source_pos(&cleaned, j, raw.len()),
                });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: source_pos(&cleaned, domain_start, raw.len()),
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: source_pos(&cleaned, last_start, raw.len()),
        });
    }

    let domain = bytes[domain_start..domain_end]
        .iter()
        .map(u8::to_ascii_lowercase)
        .collect::<Vec<_>>();
    let domain = String::from_utf8(domain).unwrap_or_default();

    Ok((
        Address {
            local: String::from_utf8(local).unwrap_or_default(),
            domain,
            quoted,
        },
        local_origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_core(input, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_core(input, policy.allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(plus) = address.local.find('+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.first().copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(plus);
        }
    }

    let local = address.local.as_bytes();
    if !local_dot_atom_valid(local) {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = seen.get(&key) {
                    duplicate_of[index] = Some(first_index);
                } else {
                    seen.insert(key, index);
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
