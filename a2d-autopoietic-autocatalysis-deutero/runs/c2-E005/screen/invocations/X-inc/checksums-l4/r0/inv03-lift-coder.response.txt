fn crc32_update(mut crc: u32, data: &[u8]) -> u32 {
    for &byte in data {
        crc ^= u32::from(byte);
        for _ in 0..8 {
            let low = crc & 1;
            crc >>= 1;
            if low != 0 {
                crc ^= 0xedb88320;
            }
        }
    }
    crc
}

pub fn crc32(data: &[u8]) -> u32 {
    crc32_update(0xffff_ffff, data) ^ 0xffff_ffff
}

fn adler_update(a: &mut u32, b: &mut u32, data: &[u8]) {
    const MOD: u32 = 65_521;
    for &byte in data {
        *a = (*a + u32::from(byte)) % MOD;
        *b = (*b + *a) % MOD;
    }
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1;
    let mut b = 0;
    adler_update(&mut a, &mut b, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for span in spans {
        let selected = &data[span.start..span.end];
        crc = crc32_update(crc, selected);
        adler_update(&mut a, &mut b, selected);
    }
    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc = crc32_update(self.crc, data);
        adler_update(&mut self.a, &mut self.b, data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let checksums = self.checksum();
        format!("c{:08x}:a{:08x}", checksums.crc32, checksums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut values = [0u32; 2];
        for (index, range) in [(0usize, 1usize..9usize), (1, 11usize..19usize)] {
            let mut value = 0u32;
            for offset in range {
                let digit = match bytes[offset] {
                    b'0'..=b'9' => u32::from(bytes[offset] - b'0'),
                    b'a'..=b'f' => u32::from(bytes[offset] - b'a' + 10),
                    _ => return Err(SnapshotError::Hex { offset }),
                };
                value = (value << 4) | digit;
            }
            values[index] = value;
        }
        Self::from_seed(values[0], values[1])
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

fn fletcher32_word(a: &mut u32, b: &mut u32, word: u16) {
    *a = (*a + u32::from(word)) % 65_535;
    *b = (*b + *a) % 65_535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        fletcher32_word(
            &mut a,
            &mut b,
            (u16::from(pair[0]) << 8) | u16::from(pair[1]),
        );
    }
    if let Some(&last) = chunks.remainder().first() {
        fletcher32_word(&mut a, &mut b, u16::from(last) << 8);
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u32;
    let mut b = 0u32;
    let mut pending = None;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending.take() {
                fletcher32_word(
                    &mut a,
                    &mut b,
                    (u16::from(high) << 8) | u16::from(byte),
                );
            } else {
                pending = Some(byte);
            }
        }
    }
    if let Some(high) = pending {
        fletcher32_word(&mut a, &mut b, u16::from(high) << 8);
    }
    Ok((b << 16) | a)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut crc = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                crc ^= u16::from(byte) << 8;
                for _ in 0..8 {
                    let high = crc & 0x8000;
                    crc = crc.wrapping_shl(1);
                    if high != 0 {
                        crc ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                crc ^= u16::from(byte);
                for _ in 0..8 {
                    let low = crc & 1;
                    crc >>= 1;
                    if low != 0 {
                        crc ^= 0x8408;
                    }
                }
            }
        }
    }
    if matches!(variant, Crc16::X25) {
        crc ^ 0xffff
    } else {
        crc
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc = 0xffff_ffff;
    let mut adler_a = 1;
    let mut adler_b = 0;
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32_a = 0u32;
    let mut f32_b = 0u32;
    let mut f32_pending = None;
    let mut crc16_state = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        crc = crc32_update(crc, selected);
        adler_update(&mut adler_a, &mut adler_b, selected);
        for &byte in selected {
            f16_a = (f16_a + u32::from(byte)) % 255;
            f16_b = (f16_b + f16_a) % 255;

            if let Some(high) = f32_pending.take() {
                fletcher32_word(
                    &mut f32_a,
                    &mut f32_b,
                    (u16::from(high) << 8) | u16::from(byte),
                );
            } else {
                f32_pending = Some(byte);
            }

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    crc16_state ^= u16::from(byte) << 8;
                    for _ in 0..8 {
                        let high = crc16_state & 0x8000;
                        crc16_state = crc16_state.wrapping_shl(1);
                        if high != 0 {
                            crc16_state ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    crc16_state ^= u16::from(byte);
                    for _ in 0..8 {
                        let low = crc16_state & 1;
                        crc16_state >>= 1;
                        if low != 0 {
                            crc16_state ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    if let Some(high) = f32_pending {
        fletcher32_word(&mut f32_a, &mut f32_b, u16::from(high) << 8);
    }

    if matches!(variant, Crc16::X25) {
        crc16_state ^= 0xffff;
    }

    Ok(Multi {
        base: Hashes {
            crc32: crc ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: ((f16_b << 8) | f16_a) as u16,
        fletcher32: (f32_b << 16) | f32_a,
        crc16: crc16_state,
    })
}

fn matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut result = 0u32;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            result ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    result
}

fn matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = matrix_times(matrix, matrix[index]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    let mut even = [0u32; 32];

    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }

    matrix_square(&mut even, &odd);
    matrix_square(&mut odd, &even);

    let mut len = right_len;
    let mut crc = left;
    loop {
        matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }

        matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }

    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut length = 0u64;
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for span in spans {
        let selected = &data[span.start..span.end];
        length += selected.len() as u64;
        crc = crc32_update(crc, selected);
        adler_update(&mut a, &mut b, selected);
    }
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        length,
        crc ^ 0xffff_ffff,
        (b << 16) | a
    ))
}

fn parse_hex(bytes: &[u8], start: usize, end: usize) -> Result<u64, VerifyError> {
    let mut value = 0u64;
    for offset in start..end {
        let digit = match bytes[offset] {
            b'0'..=b'9' => u64::from(bytes[offset] - b'0'),
            b'a'..=b'f' => u64::from(bytes[offset] - b'a' + 10),
            _ => return Err(VerifyError::Hex { offset }),
        };
        value = (value << 4) | digit;
    }
    Ok(value)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let length = parse_hex(bytes, 4, 20)?;
    let expected_crc = parse_hex(bytes, 25, 33)? as u32;
    let expected_adler = parse_hex(bytes, 40, 48)? as u32;

    let mut actual_length = 0u64;
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for span in spans {
        let selected = &data[span.start..span.end];
        actual_length += selected.len() as u64;
        crc = crc32_update(crc, selected);
        adler_update(&mut a, &mut b, selected);
    }
    if length != actual_length {
        return Err(VerifyError::Length);
    }

    let actual_crc = crc ^ 0xffff_ffff;
    let actual_adler = (b << 16) | a;
    if expected_crc != actual_crc {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual_crc,
        });
    }
    if expected_adler != actual_adler {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual_adler,
        });
    }
    Ok(())
}
