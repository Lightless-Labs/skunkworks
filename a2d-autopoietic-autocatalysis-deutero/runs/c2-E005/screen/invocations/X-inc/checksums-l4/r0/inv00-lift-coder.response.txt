#[derive(Debug, Clone)]
pub struct Stream {
    crc_raw: u32,
    adler_a: u32,
    adler_b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_raw: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_raw: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_raw ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        update_crc_adler(
            &mut self.crc_raw,
            &mut self.adler_a,
            &mut self.adler_b,
            data,
        );
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        for offset in (1..9).chain(11..19) {
            if hex_value(bytes[offset]).is_none() {
                return Err(SnapshotError::Hex { offset });
            }
        }
        let crc = parse_hex_u32(&bytes[1..9]);
        let adler = parse_hex_u32(&bytes[11..19]);
        Self::from_seed(crc, adler)
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut raw = 0xffff_ffff;
    for &byte in data {
        raw ^= u32::from(byte);
        for _ in 0..8 {
            let low = raw & 1;
            raw >>= 1;
            if low != 0 {
                raw ^= 0xedb8_8320;
            }
        }
    }
    raw ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 65521;
        b = (b + a) % 65521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = Stream::new();
    for span in spans {
        state.update(&data[span.start..span.end]);
    }
    Ok(state.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u64;
    let mut b = 0u64;
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        let word = (u16::from(pair[0]) << 8) | u16::from(pair[1]);
        a = (a + u64::from(word)) % 65535;
        b = (b + a) % 65535;
    }
    if let Some(&last) = chunks.remainder().first() {
        let word = u16::from(last) << 8;
        a = (a + u64::from(word)) % 65535;
        b = (b + a) % 65535;
    }
    ((b as u32) << 16) | a as u32
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u64;
    let mut b = 0u64;
    let mut pending = None;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending.take() {
                let word = (u16::from(high) << 8) | u16::from(byte);
                a = (a + u64::from(word)) % 65535;
                b = (b + a) % 65535;
            } else {
                pending = Some(byte);
            }
        }
    }
    if let Some(high) = pending {
        let word = u16::from(high) << 8;
        a = (a + u64::from(word)) % 65535;
        b = (b + a) % 65535;
    }
    Ok(((b as u32) << 16) | a as u32)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut register = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffffu16,
    };
    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                register ^= u16::from(byte) << 8;
                for _ in 0..8 {
                    let high = register & 0x8000;
                    register = register.wrapping_shl(1);
                    if high != 0 {
                        register ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                register ^= u16::from(byte);
                for _ in 0..8 {
                    let low = register & 1;
                    register >>= 1;
                    if low != 0 {
                        register ^= 0x8408;
                    }
                }
            }
        }
    }
    if matches!(variant, Crc16::X25) {
        register ^ 0xffff
    } else {
        register
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut base = Stream::new();
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32_a = 0u64;
    let mut f32_b = 0u64;
    let mut f32_pending = None;
    let mut crc16_state = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffffu16,
    };

    for span in spans {
        for &byte in &data[span.start..span.end] {
            base.update(std::slice::from_ref(&byte));

            f16_a = (f16_a + u32::from(byte)) % 255;
            f16_b = (f16_b + f16_a) % 255;

            if let Some(high) = f32_pending.take() {
                let word = (u16::from(high) << 8) | u16::from(byte);
                f32_a = (f32_a + u64::from(word)) % 65535;
                f32_b = (f32_b + f32_a) % 65535;
            } else {
                f32_pending = Some(byte);
            }

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    crc16_state ^= u16::from(byte) << 8;
                    for _ in 0..8 {
                        let high = crc16_state & 0x8000;
                        crc16_state = crc16_state.wrapping_shl(1);
                        if high != 0 {
                            crc16_state ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    crc16_state ^= u16::from(byte);
                    for _ in 0..8 {
                        let low = crc16_state & 1;
                        crc16_state >>= 1;
                        if low != 0 {
                            crc16_state ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    if let Some(high) = f32_pending {
        let word = u16::from(high) << 8;
        f32_a = (f32_a + u64::from(word)) % 65535;
        f32_b = (f32_b + f32_a) % 65535;
    }

    if matches!(variant, Crc16::X25) {
        crc16_state ^= 0xffff;
    }

    Ok(Multi {
        base: base.checksum(),
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: ((f32_b as u32) << 16) | f32_a as u32,
        crc16: crc16_state,
    })
}

fn update_crc_adler(
    crc_raw: &mut u32,
    adler_a: &mut u32,
    adler_b: &mut u32,
    data: &[u8],
) {
    for &byte in data {
        *crc_raw ^= u32::from(byte);
        for _ in 0..8 {
            let low = *crc_raw & 1;
            *crc_raw >>= 1;
            if low != 0 {
                *crc_raw ^= 0xedb8_8320;
            }
        }
        *adler_a = (*adler_a + u32::from(byte)) % 65521;
        *adler_b = (*adler_b + *adler_a) % 65521;
    }
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
}

fn zero_byte_operator() -> [u32; 32] {
    let mut matrix = [0u32; 32];
    matrix[0] = 0xedb8_8320;
    let mut row = 1u32;
    for item in matrix.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }

    let mut bit = [0u32; 32];
    gf2_matrix_square(&mut bit, &matrix);
    let mut two_bits = [0u32; 32];
    gf2_matrix_square(&mut two_bits, &bit);
    let mut four_bits = [0u32; 32];
    gf2_matrix_square(&mut four_bits, &two_bits);
    let mut byte = [0u32; 32];
    gf2_matrix_square(&mut byte, &four_bits);
    byte
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut operator = zero_byte_operator();
    let mut len = right_len;
    let mut crc = left;
    while len != 0 {
        if len & 1 != 0 {
            crc = gf2_matrix_times(&operator, crc);
        }
        len >>= 1;
        if len != 0 {
            let mut squared = [0u32; 32];
            gf2_matrix_square(&mut squared, &operator);
            operator = squared;
        }
    }
    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = Stream::new();
    let mut len = 0u64;
    for span in spans {
        state.update(&data[span.start..span.end]);
        len += (span.end - span.start) as u64;
    }
    let hashes = state.checksum();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, hashes.crc32, hashes.adler32
    ))
}

pub fn verify(
    data: &[u8],
    spans: &[Span],
    token: &str,
) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for offset in (4..20).chain(25..33).chain(40..48) {
        if hex_value(bytes[offset]).is_none() {
            return Err(VerifyError::Hex { offset });
        }
    }

    let expected_len = parse_hex_u64(&bytes[4..20]);
    let expected_crc = parse_hex_u32(&bytes[25..33]);
    let expected_adler = parse_hex_u32(&bytes[40..48]);

    let mut state = Stream::new();
    let mut actual_len = 0u64;
    for span in spans {
        state.update(&data[span.start..span.end]);
        actual_len += (span.end - span.start) as u64;
    }
    if expected_len != actual_len {
        return Err(VerifyError::Length);
    }

    let actual = state.checksum();
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

fn hex_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some(u32::from(byte - b'0')),
        b'a'..=b'f' => Some(u32::from(byte - b'a') + 10),
        _ => None,
    }
}

fn parse_hex_u32(bytes: &[u8]) -> u32 {
    let mut value = 0u32;
    for &byte in bytes {
        value = (value << 4) | hex_value(byte).unwrap();
    }
    value
}

fn parse_hex_u64(bytes: &[u8]) -> u64 {
    let mut value = 0u64;
    for &byte in bytes {
        value = (value << 4) | u64::from(hex_value(byte).unwrap());
    }
    value
}
