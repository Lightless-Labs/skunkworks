```rust
use std::cmp::Ordering;
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,
}

#[derive(Debug, Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    indices: HashMap<String, usize>,
    capacity: Option<usize>,
    next_age: u128,
}

fn validate_key(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }

    if start == end {
        return Err(Error::EmptyKey);
    }

    for (offset, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphanumeric() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }

    Ok(input[start..end].to_ascii_lowercase())
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self::with_capacity_internal(None)
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self::with_capacity_internal(Some(capacity))
    }

    fn with_capacity_internal(capacity: Option<usize>) -> Self {
        Self {
            entries: Vec::new(),
            indices: HashMap::new(),
            capacity,
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    fn compare_entries(a: &Entry, b: &Entry) -> Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn compare_proposal(item: &Item, age: u128, entry: &Entry) -> Ordering {
        item.priority
            .cmp(&entry.item.priority)
            .then_with(|| item.value.len().cmp(&entry.item.value.len()))
            .then_with(|| age.cmp(&entry.age))
    }

    fn rebuild_indices(&mut self) {
        self.indices.clear();
        for (index, entry) in self.entries.iter().enumerate() {
            self.indices.insert(entry.item.key.clone(), index);
        }
    }

    fn sorted_indices(&self) -> Vec<usize> {
        let mut order: Vec<usize> = (0..self.entries.len()).collect();
        order.sort_by(|&a, &b| Self::compare_entries(&self.entries[a], &self.entries[b]));
        order
    }

    fn worst_index(&self) -> Option<usize> {
        self.entries
            .iter()
            .enumerate()
            .max_by(|(_, a), (_, b)| Self::compare_entries(a, b))
            .map(|(index, _)| index)
    }

    fn remove_index(&mut self, index: usize) -> Item {
        let removed = self.entries.swap_remove(index).item;
        self.rebuild_indices();
        removed
    }

    fn fresh_age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            let mut order: Vec<usize> = (0..self.entries.len()).collect();
            order.sort_by(|&a, &b| self.entries[a].age.cmp(&self.entries[b].age));
            for (age, index) in order.into_iter().enumerate() {
                self.entries[index].age = age as u128;
            }
            self.next_age = self.entries.len() as u128;
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<Admission, Error> {
        let key = validate_key(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let proposed = Item {
            key,
            priority,
            value: value.to_owned(),
        };

        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(proposed));
        }

        if let Some(limit) = self.capacity {
            if self.entries.len() >= limit {
                let worst_index = self.worst_index().expect("full nonzero queue");
                let age = self.next_age;
                if Self::compare_proposal(&proposed, age, &self.entries[worst_index])
                    != Ordering::Less
                {
                    return Ok(Admission::Discarded(proposed));
                }
                let evicted = self.remove_index(worst_index);
                let age = self.fresh_age();
                let index = self.entries.len();
                self.entries.push(Entry {
                    item: proposed,
                    age,
                });
                self.indices.insert(self.entries[index].item.key.clone(), index);
                return Ok(Admission::Evicted(evicted));
            }
        }

        let age = self.fresh_age();
        let index = self.entries.len();
        self.entries.push(Entry {
            item: proposed,
            age,
        });
        self.indices
            .insert(self.entries[index].item.key.clone(), index);
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.sorted_indices()
            .first()
            .map(|&index| self.entries[index].item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let best = self
            .entries
            .iter()
            .enumerate()
            .min_by(|(_, a), (_, b)| Self::compare_entries(a, b))
            .map(|(index, _)| index)?;
        Some(self.remove_index(best))
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = validate_key(key)?;
        Ok(self
            .indices
            .get(&key)
            .map(|&index| self.entries[index].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = validate_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = validate_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = validate_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        Ok(std::mem::replace(
            &mut self.entries[index].item.value,
            value.to_owned(),
        ))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = validate_key(key)?;
        let Some(&index) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        Ok(self.remove_index(index))
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let Some(limit) = capacity else {
            return Vec::new();
        };
        let mut evicted = Vec::new();
        while self.entries.len() > limit {
            if let Some(index) = self.worst_index() {
                evicted.push(self.remove_index(index));
            }
        }
        evicted
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        self.sorted_indices()
            .into_iter()
            .map(|index| self.entries[index].item.clone())
            .collect::<Vec<_>>()
            .into_iter()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let source_order = other.sorted_indices();
        if policy == CollisionPolicy::Reject {
            for index in &source_order {
                let key = &other.entries[*index].item.key;
                if self.indices.contains_key(key) {
                    return Err(Error::Collision { key: key.clone() });
                }
            }
        }

        let source: Vec<Entry> = source_order
            .into_iter()
            .map(|index| other.entries[index].clone())
            .collect();

        let mut moved = 0;
        let mut replaced = 0;
        let mut discarded = 0;

        for entry in source {
            let key = entry.item.key.clone();
            if let Some(&index) = self.indices.get(&key) {
                match policy {
                    CollisionPolicy::KeepSelf => discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[index].item.priority = entry.item.priority;
                        self.entries[index].item.value = entry.item.value;
                        replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.fresh_age();
                let index = self.entries.len();
                self.entries.push(Entry {
                    item: entry.item,
                    age,
                });
                self.indices
                    .insert(self.entries[index].item.key.clone(), index);
                moved += 1;
            }
        }

        other.entries.clear();
        other.indices.clear();

        let evicted = self.set_capacity(self.capacity);
        Ok(MergeReport {
            moved,
            replaced,
            discarded,
            evicted,
        })
    }
}

impl Default for IndexedHeap {
    fn default() -> Self {
        Self::new()
    }
}
```
