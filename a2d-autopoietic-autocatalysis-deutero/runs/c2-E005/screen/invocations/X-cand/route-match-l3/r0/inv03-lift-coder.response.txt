#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing_slash: bool,
    rank: Vec<u8>,
}

#[derive(Debug, Clone)]
struct ParsedPath {
    segments: Vec<String>,
    trailing_slash: bool,
}

#[derive(Debug, Clone)]
pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;
        let (segments, trailing_slash, rank) = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_owned(),
            segments,
            trailing_slash,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;

        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_path(head)?;

        let mut candidates: Vec<(usize, Vec<(String, String)>)> = Vec::new();
        for (index, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict
                && route.trailing_slash != parsed.trailing_slash
            {
                continue;
            }
            if let Some(params) = match_route(route, &parsed.segments) {
                candidates.push((index, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let candidate_methods: Vec<&str> = candidates
            .iter()
            .map(|(index, _)| self.routes[*index].method.as_str())
            .collect();

        let eligible_method = if method == "HEAD" && candidate_methods.contains(&"HEAD") {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let eligible: Vec<(usize, Vec<(String, String)>)> = candidates
            .into_iter()
            .filter(|(index, _)| self.routes[*index].method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidate_methods
                .into_iter()
                .map(str::to_owned)
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let (winner_index, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| {
                self.routes[*a]
                    .rank
                    .cmp(&self.routes[*b].rank)
                    .then_with(|| b.cmp(a))
            })
            .expect("eligible candidates are nonempty");

        let winner = &self.routes[winner_index];
        let redirect = if self.policy == SlashPolicy::Redirect
            && winner.trailing_slash != parsed.trailing_slash
            && !head.is_empty()
            && head != "/"
        {
            if winner.trailing_slash {
                Some(format!("{head}/"))
            } else {
                Some(head.strip_suffix('/').unwrap_or(head).to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: winner_index,
            params,
            redirect,
        })
    }
}

fn validate_method(method: &str) -> Result<(), Error> {
    if !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase()) {
        Ok(())
    } else {
        Err(Error::InvalidMethod)
    }
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool, Vec<u8>), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false, vec![3]));
    }

    let (parts, trailing_slash) = split_parts(pattern);
    let mut segments = Vec::with_capacity(parts.len());
    let mut names = Vec::<String>::new();
    let mut rank = Vec::new();

    for (index, (part, start)) in parts.iter().enumerate() {
        if part.is_empty() {
            return Err(Error::InvalidPattern { at: *start });
        }

        if let Some(name) = part.strip_prefix(':') {
            let valid = valid_capture_name(name);
            if !valid || names.iter().any(|n| n == name) {
                return Err(Error::InvalidPattern { at: *start });
            }
            names.push(name.to_owned());
            segments.push(Segment::Param(name.to_owned()));
            rank.push(1);
        } else if let Some(name) = part.strip_prefix('*') {
            if index + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            if !valid_capture_name(name) || names.iter().any(|n| n == name) {
                return Err(Error::InvalidPattern { at: *start });
            }
            names.push(name.to_owned());
            segments.push(Segment::Wildcard(name.to_owned()));
            rank.push(0);
        } else {
            let decoded = decode_segment(part, *start, true)?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }
    }

    if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
        rank.push(3);
    }

    Ok((segments, trailing_slash, rank))
}

fn parse_path(path: &str) -> Result<ParsedPath, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing_slash: false,
        });
    }

    let (parts, trailing_slash) = split_parts(path);
    let mut segments = Vec::with_capacity(parts.len());
    for (part, start) in parts {
        if part.is_empty() {
            return Err(Error::InvalidPath { at: start });
        }
        segments.push(decode_segment(part, start, false).map_err(|e| match e {
            Error::InvalidPattern { at } => Error::InvalidPath { at },
            _ => unreachable!(),
        })?);
    }
    Ok(ParsedPath {
        segments,
        trailing_slash,
    })
}

fn split_parts(input: &str) -> (Vec<(&str, usize)>, bool) {
    let trailing_slash = input.ends_with('/');
    let end = if trailing_slash {
        input.len() - 1
    } else {
        input.len()
    };

    let mut parts = Vec::new();
    let mut start = 1;
    let bytes = input.as_bytes();
    for i in 1..end {
        if bytes[i] == b'/' {
            parts.push((&input[start..i], start));
            start = i + 1;
        }
    }
    if start <= end {
        parts.push((&input[start..end], start));
    }
    (parts, trailing_slash)
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(segment: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !bytes[i + 1].is_ascii_digit() && !(b'A'..=b'F').contains(&bytes[i + 1])
                || !bytes[i + 2].is_ascii_digit() && !(b'A'..=b'F').contains(&bytes[i + 2])
            {
                return Err(invalid_segment(offset + i, pattern));
            }
            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid_segment(offset + i, pattern));
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(invalid_segment(offset + i, pattern));
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn invalid_segment(at: usize, pattern: bool) -> Error {
    if pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'A'..=b'F' => b - b'A' + 10,
        _ => 0,
    }
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_index = route
        .segments
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)));
    let fixed_len = wildcard_index.unwrap_or(route.segments.len());

    if wildcard_index.is_none() && path.len() != fixed_len {
        return None;
    }
    if wildcard_index.is_some() && path.len() < fixed_len {
        return None;
    }

    let mut params = Vec::new();
    for (i, segment) in route.segments.iter().take(fixed_len).enumerate() {
        match segment {
            Segment::Static(value) if value != &path[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), path[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if let Some(Segment::Wildcard(name)) = wildcard_index.map(|i| &route.segments[i]) {
        let value = path[fixed_len..].join("/");
        params.push((name.clone(), value));
    }

    Some(params)
}
