fn parse_error(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(parse_error(field, start, 0));
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(parse_error(field, start, 1));
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or_else(|| parse_error(field, start, 2))?;
    }
    Ok(value)
}

    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }

    let mut allowed: [Vec<bool>; 5] = [

        vec![false; 24],
        vec![false; 32],
        vec![false; 13],
        vec![false; 7],

    for (field_idx, piece) in pieces.iter().enumerate() {
        let domain = DOMAINS[field_idx];
        let field = domain.field;
        let bytes = piece.as_bytes();

            if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
                return Err(ParseError::InvalidCharacter { field, offset });
            }

            exact_day_stars[0] = *piece == "*";
        } else if field == Field::DayOfWeek {
            exact_day_stars[1] = *piece == "*";
        }

        loop {
            let item_end = bytes[item_start..]
                .iter()
                .position(|&b| b == b',')
                .map(|n| item_start + n)
                .unwrap_or(bytes.len());

            if item_start == item_end {
                return Err(ParseError::EmptyItem { field, offset: item_start });
            }

            let item = &bytes[item_start..item_end];
            let slash_positions: Vec<usize> = item
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| (b == b'/').then_some(i))
                .collect();

            if slash_positions.len() > 1 {
                return Err(parse_error(field, item_start, 0));
            }

            let step_start = slash_positions.first().map(|&i| i + 1);

            let (base_kind, first_range, second_range) = if base_end == 1 && item[0] == b'*' {
                (0u8, None, None)
            } else {
                let hyphens: Vec<usize> = item[..base_end]
                    .iter()
                    .enumerate()

                    .collect();

                            return Err(parse_error(field, item_start, 0));

                    }
                    _ => return Err(parse_error(field, item_start, 0)),
                }
            };

            if step_start.is_some_and(|s| s == item.len()) {
                return Err(parse_error(field, item_start, 0));
            }

            let mut start_value = domain.min;
            let mut end_value = domain.max;

            if let Some((s, e)) = first_range {
                start_value = parse_number(item, s, e, field)?;
                if start_value < domain.min || start_value > domain.max {
                    return Err(parse_error(field, item_start + s, 3));
                }
                if let Some((s2, e2)) = second_range {
                    end_value = parse_number(item, s2, e2, field)?;
                    if end_value < domain.min || end_value > domain.max {
                        return Err(parse_error(field, item_start + s2, 3));
                    }
                    if start_value > end_value {
                        return Err(parse_error(field, item_start + s2, 4));
                    }
                } else if step_start.is_none() {
                    end_value = start_value;
                } else {
                    end_value = domain.max;
                }
            }

                let value = parse_number(item, s, item.len(), field)?;
                let width = domain.max - domain.min + 1;
                if value == 0 || value > width {
                    return Err(parse_error(field, item_start + s, 5));
                }
                value
            } else {
                1
            };

                    Some(next) => value = next,
                    None => break,
                }
            }

            if item_end == bytes.len() {
                break;
            }
            item_start = item_end + 1;
        }

        allowed,
        day_of_month_was_exact_star: exact_day_stars[0],
        day_of_week_was_exact_star: exact_day_stars[1],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.allowed[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(value, &selected)| selected.then_some(value as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.allowed[0][at.minute as usize]
            && self.allowed[1][at.hour as usize]
            && self.allowed[2][at.day as usize]
            && self.allowed[3][at.month as usize]
            && self.allowed[4][weekday as usize])
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
            current = next_minute(current);
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day < 1 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn next_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }

    let (minimum, maximum, width) = field_domain(field);
    if step {
        if value == 0 || value > width as u32 {
            return Err(ParseError::BadStep { field, offset: start });
        }
    } else if value < minimum || value > maximum {
        return Err(ParseError::OutOfRange { field, offset: start });
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {

    let mut selected = vec![false; width];

    for (offset, &byte) in bytes.iter().enumerate() {

            return Err(ParseError::InvalidCharacter { field, offset });
        }

    loop {

            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());

            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];

            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
            .collect();

        let step_part = if slash_positions.len() == 1 {
            Some((slash_positions[0] + 1, item.len()))

            None
        };

            .iter()

                offset: item_start,

            }
        }

        let stride = step.unwrap_or(1);
        let mut value = start_value;
        while value <= end_value {
            selected[(value - minimum) as usize] = true;
            match value.checked_add(stride) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = Vec::with_capacity(5);
    for (part, field) in parts.iter().zip(fields) {
        selected.push(parse_field(part, field)?);
    }

    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        day_of_month_is_star: parts[2].as_bytes() == b"*",
        day_of_week_is_star: parts[4].as_bytes() == b"*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (minimum, _, _) = field_domain(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &is_selected)| {
                if is_selected {
                    Some(minimum as u8 + i as u8)
                } else {
                    None
                }
            })
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        let (minimum, maximum, _) = field_domain(field);
        if (value as u32) < minimum || (value as u32) > maximum {
            return false;
        }
        self.selected[field_index(field)][(value as u32 - minimum) as usize]
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.contains(Field::DayOfMonth, at.day)
            && self.contains(Field::DayOfWeek, weekday))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if !increment_minute(&mut current) {
                return Ok(None);
            }
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }

    if at.day == 0 || at.day > days_in_month(at.year, at.month) {

    if at.hour > 23 {

            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn increment_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;

    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;

    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;

    if at.month < 12 {
        at.month += 1;
        return true;
    }
    at.month = 1;

    if at.year < 2399 {
        at.year += 1;
        return true;
    }
    false
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(u32::from(byte - b'0')))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    let (minimum, maximum) = domain(field);
    if step {
        if value < 1 || value > maximum - minimum + 1 {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else if value < minimum || value > maximum {
        return Err(ParseError::OutOfRange {
            field,
            offset: start,
        });
    }

    Ok(value)
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = domain(field);
    let mut selected = vec![false; (maximum + 1) as usize];

    let mut item_start = 0usize;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|n| item_start + n)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(invalid_syntax(field, item_start));
        }

        let (base_end, step_part) = if let Some(&slash) = slash_positions.first() {
            if slash == 0 || slash + 1 == item.len() {
                return Err(invalid_syntax(field, item_start));
            }
            (slash, Some(slash + 1))
        } else {
            (item.len(), None)
        };

        let base = &item[..base_end];
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_kind = if base == b"*" {
            0
        } else if hyphens.is_empty() && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            1
        } else if hyphens.len() == 1 {
            let hyphen = hyphens[0];
            if hyphen == 0
                || hyphen + 1 == base.len()
                || !base[..hyphen].iter().all(u8::is_ascii_digit)
                || !base[hyphen + 1..].iter().all(u8::is_ascii_digit)
            {
                return Err(invalid_syntax(field, item_start));
            }
            2
        } else {
            return Err(invalid_syntax(field, item_start));
        };

        let step = if let Some(step_start_in_item) = step_part {
            if item[step_start_in_item..].iter().any(|&b| !b.is_ascii_digit()) {
                return Err(invalid_syntax(field, item_start));
            }
            Some(parse_number(
                item,
                step_start_in_item,
                item.len(),
                field,
                true,
            )?)
        } else {
            None
        };

        let (start, end) = match base_kind {
            0 => (minimum, maximum),
            1 => {
                let value = parse_number(item, 0, base_end, field, false)?;
                (value, if step.is_some() { maximum } else { value })
            }
            _ => {
                let hyphen = hyphens[0];
                let start = parse_number(item, 0, hyphen, field, false)?;
                let end = parse_number(item, hyphen + 1, base_end, field, false)?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + hyphen + 1,
                    });
                }
                (start, end)
            }
        };

        let stride = step.unwrap_or(1);
        let mut value = start;
        loop {
            selected[value as usize] = true;
            match value.checked_add(stride) {
                Some(next) if next <= end => value = next,
                _ => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for (index, (&field, part)) in FIELDS.iter().zip(parts.iter()).enumerate() {
        selected[index] = parse_field(part, field)?;
    }

    Ok(Cron {
        selected,
        full_day_of_month: parts[2] == "*",
        full_day_of_week: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &yes)| yes.then_some(value as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);

        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize]
            && self.selected[3][at.month as usize]
            && self.selected[4][weekday as usize])
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut date = after;
        let mut first_day = true;
        while date.year <= 2399 {
            let day_matches = self.selected[2][date.day as usize]
                && self.selected[3][date.month as usize]
                && self.selected[4][weekday(date.year, date.month, date.day) as usize];

            if day_matches {
                let start_hour = if first_day { after.hour } else { 0 };
                for hour in start_hour..24 {
                    if !self.selected[1][hour as usize] {
                        continue;
                    }
                    let start_minute = if first_day && hour == after.hour {
                        after.minute.saturating_add(1)
                    } else {
                        0
                    };
                    for minute in start_minute..60 {
                        if self.selected[0][minute as usize] {
                            return Ok(Some(CivilDateTime {
                                year: date.year,
                                month: date.month,
                                day: date.day,
                                hour,
                                minute,
                            }));
                        }
                    }
                }
            }

            first_day = false;
            if !advance_day(&mut date) {
                break;
            }
        }

        Ok(None)
    }
}
