#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_domain(field: Field) -> (u32, u32, usize) {
    match field {
        Field::Minute => (0, 59, 60),
        Field::Hour => (0, 23, 24),
        Field::DayOfMonth => (1, 31, 31),
        Field::Month => (1, 12, 12),
        Field::DayOfWeek => (0, 6, 7),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }

    let (minimum, maximum, width) = field_domain(field);
    if step {
        if value == 0 || value > width as u32 {
            return Err(ParseError::BadStep { field, offset: start });
        }
    } else if value < minimum || value > maximum {
        return Err(ParseError::OutOfRange { field, offset: start });
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    let (minimum, maximum, width) = field_domain(field);
    let mut selected = vec![false; width];

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
            .collect();

        let base_end = slash_positions.first().copied().unwrap_or(item.len());
        let step_part = if slash_positions.len() == 1 {
            Some((slash_positions[0] + 1, item.len()))
        } else {
            None
        };

        let base = &item[..base_end];
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
            .collect();

        let syntax_valid = slash_positions.len() <= 1
            && !base.is_empty()
            && if base == b"*" {
                true
            } else if hyphens.is_empty() {
                base.iter().all(u8::is_ascii_digit)
            } else if hyphens.len() == 1 {
                let h = hyphens[0];
                h > 0
                    && h + 1 < base.len()
                    && base[..h].iter().all(u8::is_ascii_digit)
                    && base[h + 1..].iter().all(u8::is_ascii_digit)
            } else {
                false
            };

        let step_syntax_valid = match step_part {
            None => true,
            Some((start, end)) => start < end && item[start..end].iter().all(u8::is_ascii_digit),
        };

        if !syntax_valid || !step_syntax_valid {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let step = match step_part {
            Some((start, end)) => Some(parse_number(
                item,
                start,
                end,
                field,
                true,
            )?),
            None => None,
        };

        let (start_value, end_value, number_offsets) = if base == b"*" {
            (minimum, maximum, None)
        } else if hyphens.is_empty() {
            let value = parse_number(item, 0, base.len(), field, false)?;
            (value, if step.is_some() { maximum } else { value }, Some((0, 0)))
        } else {
            let h = hyphens[0];
            let start_value = parse_number(item, 0, h, field, false)?;
            let end_value = parse_number(item, h + 1, base.len(), field, false)?;
            (start_value, end_value, Some((0, h + 1)))
        };

        if let Some((_, end_offset)) = number_offsets {
            if hyphens.len() == 1 && start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_offset,
                });
            }
        }

        let stride = step.unwrap_or(1);
        let mut value = start_value;
        while value <= end_value {
            selected[(value - minimum) as usize] = true;
            match value.checked_add(stride) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = Vec::with_capacity(5);
    for (part, field) in parts.iter().zip(fields) {
        selected.push(parse_field(part, field)?);
    }

    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        day_of_month_is_star: parts[2].as_bytes() == b"*",
        day_of_week_is_star: parts[4].as_bytes() == b"*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (minimum, _, _) = field_domain(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &is_selected)| {
                if is_selected {
                    Some(minimum as u8 + i as u8)
                } else {
                    None
                }
            })
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        let (minimum, maximum, _) = field_domain(field);
        if (value as u32) < minimum || (value as u32) > maximum {
            return false;
        }
        self.selected[field_index(field)][(value as u32 - minimum) as usize]
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.contains(Field::DayOfMonth, at.day)
            && self.contains(Field::DayOfWeek, weekday))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if !increment_minute(&mut current) {
                return Ok(None);
            }
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn increment_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;

    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;

    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;

    if at.month < 12 {
        at.month += 1;
        return true;
    }
    at.month = 1;

    if at.year < 2399 {
        at.year += 1;
        return true;
    }
    false
}
