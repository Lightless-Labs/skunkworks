#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    allowed: [Vec<bool>; 5],
    day_of_month_was_exact_star: bool,
    day_of_week_was_exact_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Clone, Copy)]
struct Domain {
    field: Field,
    min: u32,
    max: u32,
}

const DOMAINS: [Domain; 5] = [
    Domain { field: Field::Minute, min: 0, max: 59 },
    Domain { field: Field::Hour, min: 0, max: 23 },
    Domain { field: Field::DayOfMonth, min: 1, max: 31 },
    Domain { field: Field::Month, min: 1, max: 12 },
    Domain { field: Field::DayOfWeek, min: 0, max: 6 },
];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_error(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(parse_error(field, start, 0));
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(parse_error(field, start, 1));
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or_else(|| parse_error(field, start, 2))?;
    }
    Ok(value)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }

    let mut allowed: [Vec<bool>; 5] = [
        vec![false; 60],
        vec![false; 24],
        vec![false; 32],
        vec![false; 13],
        vec![false; 7],
    ];
    let mut exact_day_stars = [false; 2];

    for (field_idx, piece) in pieces.iter().enumerate() {
        let domain = DOMAINS[field_idx];
        let field = domain.field;
        let bytes = piece.as_bytes();

        for (offset, &byte) in bytes.iter().enumerate() {
            if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        if field == Field::DayOfMonth {
            exact_day_stars[0] = *piece == "*";
        } else if field == Field::DayOfWeek {
            exact_day_stars[1] = *piece == "*";
        }

        let mut item_start = 0;
        loop {
            let item_end = bytes[item_start..]
                .iter()
                .position(|&b| b == b',')
                .map(|n| item_start + n)
                .unwrap_or(bytes.len());

            if item_start == item_end {
                return Err(ParseError::EmptyItem { field, offset: item_start });
            }

            let item = &bytes[item_start..item_end];
            let slash_positions: Vec<usize> = item
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| (b == b'/').then_some(i))
                .collect();

            if slash_positions.len() > 1 {
                return Err(parse_error(field, item_start, 0));
            }

            let base_end = slash_positions.first().copied().unwrap_or(item.len());
            let step_start = slash_positions.first().map(|&i| i + 1);

            let (base_kind, first_range, second_range) = if base_end == 1 && item[0] == b'*' {
                (0u8, None, None)
            } else {
                let hyphens: Vec<usize> = item[..base_end]
                    .iter()
                    .enumerate()
                    .filter_map(|(i, &b)| (b == b'-').then_some(i))
                    .collect();

                match hyphens.as_slice() {
                    [] => {
                        if base_end == 0
                            || !item[..base_end].iter().all(u8::is_ascii_digit)
                        {
                            return Err(parse_error(field, item_start, 0));
                        }
                        (1, Some((0, base_end)), None)
                    }
                    [hyphen] => {
                        if *hyphen == 0
                            || *hyphen + 1 == base_end
                            || !item[..*hyphen].iter().all(u8::is_ascii_digit)
                            || !item[*hyphen + 1..base_end]
                                .iter()
                                .all(u8::is_ascii_digit)
                        {
                            return Err(parse_error(field, item_start, 0));
                        }
                        (2, Some((0, *hyphen)), Some((*hyphen + 1, base_end)))
                    }
                    _ => return Err(parse_error(field, item_start, 0)),
                }
            };

            if step_start.is_some_and(|s| s == item.len()) {
                return Err(parse_error(field, item_start, 0));
            }

            let mut start_value = domain.min;
            let mut end_value = domain.max;

            if let Some((s, e)) = first_range {
                start_value = parse_number(item, s, e, field)?;
                if start_value < domain.min || start_value > domain.max {
                    return Err(parse_error(field, item_start + s, 3));
                }
                if let Some((s2, e2)) = second_range {
                    end_value = parse_number(item, s2, e2, field)?;
                    if end_value < domain.min || end_value > domain.max {
                        return Err(parse_error(field, item_start + s2, 3));
                    }
                    if start_value > end_value {
                        return Err(parse_error(field, item_start + s2, 4));
                    }
                } else if step_start.is_none() {
                    end_value = start_value;
                } else {
                    end_value = domain.max;
                }
            }

            let step = if let Some(s) = step_start {
                let value = parse_number(item, s, item.len(), field)?;
                let width = domain.max - domain.min + 1;
                if value == 0 || value > width {
                    return Err(parse_error(field, item_start + s, 5));
                }
                value
            } else {
                1
            };

            let mut value = start_value;
            while value <= end_value {
                allowed[field_idx][value as usize] = true;
                match value.checked_add(step) {
                    Some(next) => value = next,
                    None => break,
                }
            }

            if item_end == bytes.len() {
                break;
            }
            item_start = item_end + 1;
        }
    }

    Ok(Cron {
        allowed,
        day_of_month_was_exact_star: exact_day_stars[0],
        day_of_week_was_exact_star: exact_day_stars[1],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.allowed[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(value, &selected)| selected.then_some(value as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.allowed[0][at.minute as usize]
            && self.allowed[1][at.hour as usize]
            && self.allowed[2][at.day as usize]
            && self.allowed[3][at.month as usize]
            && self.allowed[4][weekday as usize])
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
            current = next_minute(current);
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day < 1 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn next_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}
