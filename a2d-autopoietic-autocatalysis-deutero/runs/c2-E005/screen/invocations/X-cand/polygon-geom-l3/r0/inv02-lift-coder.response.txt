#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

const COORD_LIMIT: i64 = 1_000_000_000;
const INPUT_LIMIT: usize = 4096;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > INPUT_LIMIT {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective.last() == effective.first() {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let acx = c.x as i128 - a.x as i128;
    let acy = c.y as i128 - a.y as i128;
    abx * acy - aby * acx
}

fn edge_mill(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let squared = (dx * dx + dy * dy) as u128 * 1_000_000;
    let mut low = 0u128;
    let mut high = 3_000_000_000u128;
    while low < high {
        let mid = (low + high + 1) / 2;
        if mid * mid <= squared {
            low = mid;
        } else {
            high = mid - 1;
        }
    }
    let remainder = squared - low * low;
    if remainder >= low * 2 + 1 {
        (low + 1) as u64
    } else {
        low as u64
    }
}

fn signed_area2(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn perimeter_milli(points: &[Point]) -> u64 {
    if points.is_empty() {
        return 0;
    }
    let mut total = 0u64;
    for i in 0..points.len() {
        total += edge_mill(points[i], points[(i + 1) % points.len()]);
    }
    total
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let points = prepare_polygon(points)?;
    let signed = signed_area2(&points);
    let mut edge_milli = Vec::with_capacity(points.len());
    let mut perimeter = 0u64;
    let mut longest: Option<usize> = None;

    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        let length = edge_mill(a, b);
        edge_milli.push(length);
        perimeter += length;

        let replace = match longest {
            None => true,
            Some(current) if length > edge_milli[current] => true,
            Some(current) if length < edge_milli[current] => false,
            Some(current) => {
                let pair = (a.min(b), a.max(b));
                let previous_a = points[current];
                let previous_b = points[(current + 1) % points.len()];
                pair < (previous_a.min(previous_b), previous_a.max(previous_b))
                    || (pair == (previous_a.min(previous_b), previous_a.max(previous_b))
                        && i < current)
            }
        };
        if replace {
            longest = Some(i);
        }
    }

    Ok(Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation: match signed.cmp(&0) {
            std::cmp::Ordering::Less => Orientation::Clockwise,
            std::cmp::Ordering::Equal => Orientation::Degenerate,
            std::cmp::Ordering::Greater => Orientation::CounterClockwise,
        },
        edge_milli,
        perimeter_milli: perimeter,
        longest_edge: longest,
    })
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > INPUT_LIMIT {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &query in queries {
        let mut boundary = None;
        let mut inside = false;

        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];

            if cross(a, b, query) == 0
                && query.x >= a.x.min(b.x)
                && query.x <= a.x.max(b.x)
                && query.y >= a.y.min(b.y)
                && query.y <= a.y.max(b.y)
            {
                boundary = Some(i);
                break;
            }

            if (a.y > query.y) != (b.y > query.y) {
                let orientation = cross(a, b, query);
                if (b.y > a.y && orientation > 0) || (b.y < a.y && orientation < 0) {
                    inside = !inside;
                }
            }
        }

        results.push(match boundary {
            Some(edge) => Location::Boundary { edge },
            None if inside => Location::Inside,
            None => Location::Outside,
        });
    }

    Ok(results)
}

fn validated_cloud(points: &[Point]) -> Result<Vec<(Point, usize)>, GeomError> {
    if points.len() > INPUT_LIMIT {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique = points
        .iter()
        .copied()
        .enumerate()
        .map(|(index, point)| (point, index))
        .collect::<Vec<_>>();
    unique.sort_by_key(|&(point, index)| (point, index));
    unique.dedup_by(|a, b| a.0 == b.0);
    Ok(unique)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    let unique = validated_cloud(points)?;
    if unique.len() <= 1 {
        let vertices = unique.iter().map(|&(point, _)| point).collect::<Vec<_>>();
        let source_indices = unique.iter().map(|&(_, index)| index).collect::<Vec<_>>();
        return Ok(Hull {
            vertices,
            source_indices,
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let all_collinear = unique
        .iter()
        .skip(2)
        .all(|&(point, _)| cross(unique[0].0, unique[1].0, point) == 0);

    let hull_indices: Vec<usize> = if all_collinear {
        match mode {
            Collinear::Include => (0..unique.len()).collect(),
            Collinear::Exclude => vec![0, unique.len() - 1],
        }
    } else {
        let mut lower = Vec::new();
        for i in 0..unique.len() {
            while lower.len() >= 2 {
                let a = unique[lower[lower.len() - 2]].0;
                let b = unique[lower[lower.len() - 1]].0;
                let c = unique[i].0;
                let turn = cross(a, b, c);
                if match mode {
                    Collinear::Exclude => turn <= 0,
                    Collinear::Include => turn < 0,
                } {
                    lower.pop();
                } else {
                    break;
                }
            }
            lower.push(i);
        }

        let mut upper = Vec::new();
        for i in (0..unique.len()).rev() {
            while upper.len() >= 2 {
                let a = unique[upper[upper.len() - 2]].0;
                let b = unique[upper[upper.len() - 1]].0;
                let c = unique[i].0;
                let turn = cross(a, b, c);
                if match mode {
                    Collinear::Exclude => turn <= 0,
                    Collinear::Include => turn < 0,
                } {
                    upper.pop();
                } else {
                    break;
                }
            }
            upper.push(i);
        }

        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower
    };

    let vertices = hull_indices
        .iter()
        .map(|&index| unique[index].0)
        .collect::<Vec<_>>();
    let source_indices = hull_indices
        .iter()
        .map(|&index| unique[index].1)
        .collect::<Vec<_>>();
    let area = signed_area2(&vertices).unsigned_abs();

    Ok(Hull {
        perimeter_milli: perimeter_milli(&vertices),
        vertices,
        source_indices,
        area2: area,
    })
}
