#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct Byte {
    value: u8,
    at: usize,
}

struct Cleanup {
    bytes: Vec<Byte>,
    runs: Vec<(usize, usize)>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn valid_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleanup, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut quoted_prefix = false;
    let mut quote_closed = false;

    while i < input.len() {
        if quoted_prefix && !quote_closed {
            let b = input[i];
            bytes.push(Byte { value: b, at: i });
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    bytes.push(Byte { value: input[i], at: i });
                    i += 1;
                }
            } else {
                if b == b'"' {
                    quote_closed = true;
                }
                i += 1;
            }
            continue;
        }

        let b = input[i];
        if b == b' ' || b == b'\t' || b == b'\r' || b == b'\n' || b == b'(' {
            let start = i;
            let run_start = bytes.len();

            loop {
                if i >= input.len() {
                    break;
                }
                let c = input[i];
                if is_hws(c) {
                    i += 1;
                } else if c == b'\r' || c == b'\n' {
                    if let Some(end) = valid_fold(input, i) {
                        i = end;
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if c == b'(' {
                    let outer = i;
                    let mut depth = 1usize;
                    i += 1;
                    while i < input.len() {
                        let x = input[i];
                        if x == b'\\' {
                            let esc_at = i;
                            if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                                return Err(EmailError::CommentEscape { at: esc_at });
                            }
                            i += 2;
                        } else if x == b'(' {
                            if depth == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            depth += 1;
                            i += 1;
                        } else if x == b')' {
                            depth -= 1;
                            i += 1;
                            if depth == 0 {
                                break;
                            }
                        } else if x == b'\r' || x == b'\n' {
                            if let Some(end) = valid_fold(input, i) {
                                i = end;
                            } else {
                                return Err(EmailError::BadFold { at: i });
                            }
                        } else if is_printable(x) || x == b'\t' {
                            i += 1;
                        } else {
                            return Err(EmailError::CommentChar { at: i });
                        }
                    }
                    if depth != 0 {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                } else {
                    break;
                }
            }

            if i > start {
                runs.push((run_start, bytes.len(), start));
            }
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if b == b'\r' || b == b'\n' {
            return Err(EmailError::BadFold { at: i });
        }

        if bytes.is_empty() && b == b'"' {
            quoted_prefix = true;
            quote_closed = false;
        }
        bytes.push(Byte { value: b, at: i });
        i += 1;
    }

    Ok(Cleanup { bytes, runs })
}

fn parse_email(input: &str) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    let bytes = &cleaned.bytes;

    for &(start, end, original_at) in &cleaned.runs {
        let left_ok = start == 0 || bytes.get(start.wrapping_sub(1)).is_some_and(|b| b.value == b'@');
        let right_ok = end == bytes.len() || bytes.get(end).is_some_and(|b| b.value == b'@');
        if !left_ok && !right_ok {
            return Err(EmailError::CfwsBoundary { at: original_at });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if bytes[0].value == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i].value == b'\\' {
                i += 2;
            } else if bytes[i].value == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: bytes[0].at });
        }
    }

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quote_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i].value == b'@' {
            separators.push(i);
        }
        i += 1;
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: bytes[separators[1]].at });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: bytes[sep].at });
    }

    let mut local = String::new();
    let quoted = quote_end.is_some();
    if let Some(end) = quote_end {
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal { at: bytes[end + 1].at });
        }
        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            let b = bytes[j].value;
            if b == b'\\' {
                let origin = bytes[j].at;
                if j + 1 >= end {
                    return Err(EmailError::BadEscape { at: origin });
                }
                let escaped = bytes[j + 1].value;
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape { at: origin });
                }
                local.push(escaped as char);
                origins.push(origin);
                j += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar { at: bytes[j].at });
                }
                local.push(b as char);
                origins.push(bytes[j].at);
                j += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: bytes[end].at });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        if sep > 64 {
            return Err(EmailError::LocalTooLong { at: bytes[64].at });
        }
        for j in 0..sep {
            let b = bytes[j].value;
            if b == b'.' {
                if j == 0 || j + 1 == sep || bytes[j - 1].value == b'.' {
                    return Err(EmailError::LocalDot { at: bytes[j].at });
                }
            } else if !(b.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
            {
                return Err(EmailError::LocalChar { at: bytes[j].at });
            }
            local.push(b as char);
        }
    }

    let domain_start = sep + 1;
    if domain_start == bytes.len() {
        return Err(EmailError::EmptyDomain { at: input.len() });
    }

    let mut domain_end = bytes.len();
    let mut root_dot = None;
    if bytes[domain_end - 1].value == b'.' {
        root_dot = Some(bytes[domain_end - 1].at);
        domain_end -= 1;
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: bytes[domain_start].at,
        });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong { at: bytes[domain_start + 253].at });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for j in domain_start..=domain_end {
        if j == domain_end || bytes[j].value == b'.' {
            if label_start == j {
                let at = if j == domain_end {
                    root_dot.unwrap_or(input.len())
                } else {
                    bytes[j].at
                };
                return Err(EmailError::EmptyLabel { at });
            }
            labels.push((label_start, j));
            label_start = j + 1;
        }
    }

    for &(start, end) in &labels {
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong { at: bytes[start + 63].at });
        }
        if bytes[start].value == b'-' {
            return Err(EmailError::LabelHyphen { at: bytes[start].at });
        }
        if bytes[end - 1].value == b'-' {
            return Err(EmailError::LabelHyphen { at: bytes[end - 1].at });
        }
        for j in start..end {
            let b = bytes[j].value;
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar { at: bytes[j].at });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: bytes[domain_start].at });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end].iter().all(|b| b.value.is_ascii_digit()) {
        return Err(EmailError::NumericTld { at: bytes[last_start].at });
    }

    let domain: String = bytes[domain_start..domain_end]
        .iter()
        .map(|b| (b.value as char).to_ascii_lowercase())
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
