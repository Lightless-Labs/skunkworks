#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone)]
struct CfwsRun {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut clean = Vec::<u8>::new();
    let mut positions = Vec::<usize>::new();
    let mut runs = Vec::<CfwsRun>::new();
    let mut i = 0usize;
    let mut comment_depth = 0usize;
    let mut comment_outer = 0usize;
    let mut pending_run: Option<usize> = None;
    let mut quote_protected = false;

    while i < bytes.len() {
        if quote_protected {
            let b = bytes[i];
            clean.push(b);
            positions.push(i);
            i += 1;
            if b == b'\\' {
                if i < bytes.len() {
                    clean.push(bytes[i]);
                    positions.push(i);
                    i += 1;
                }
            } else if b == b'"' {
                quote_protected = false;
            }
            continue;
        }

        if comment_depth > 0 {
            let b = bytes[i];
            if b == b'\\' {
                if i + 1 >= bytes.len() || !(0x20..=0x7e).contains(&bytes[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = is_fold(bytes, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                comment_depth -= 1;
                i += 1;
                continue;
            }
            if b == b'\t' || (0x20..=0x7e).contains(&b) {
                i += 1;
                continue;
            }
            return Err(EmailError::CommentChar { at: i });
        }

        if clean.is_empty() && bytes[i] == b'"' {
            quote_protected = true;
            clean.push(bytes[i]);
            positions.push(i);
            i += 1;
            continue;
        }

        let b = bytes[i];
        if b == b'(' {
            if pending_run.is_none() {
                pending_run = Some(i);
            }
            comment_outer = i;
            comment_depth = 1;
            i += 1;
            continue;
        }
        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if b == b'\r' || b == b'\n' {
            if let Some(end) = is_fold(bytes, i) {
                if pending_run.is_none() {
                    pending_run = Some(i);
                }
                i = end;
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }
        if is_hws(b) {
            if pending_run.is_none() {
                pending_run = Some(i);
            }
            i += 1;
            continue;
        }

        if let Some(start) = pending_run.take() {
            runs.push(CfwsRun { start, end: i });
        }
        clean.push(b);
        positions.push(i);
        i += 1;
    }

    if comment_depth > 0 {
        return Err(EmailError::UnclosedComment { at: comment_outer });
    }
    if let Some(start) = pending_run {
        runs.push(CfwsRun {
            start,
            end: bytes.len(),
        });
    }

    for run in &runs {
        let left_boundary = clean.is_empty()
            || run.start == 0
            || run.end == bytes.len()
            || positions
                .iter()
                .enumerate()
                .any(|(idx, &p)| p < run.start && clean[idx] == b'@');
        let right_boundary = clean.is_empty()
            || run.start == 0
            || run.end == bytes.len()
            || positions
                .iter()
                .enumerate()
                .any(|(idx, &p)| p >= run.end && clean[idx] == b'@');
        if !left_boundary && !right_boundary {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let n = clean.len();
    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut j = 1usize;
        while j < n {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                quote_end = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: positions[0] });
        }
    }

    let protected_end = quote_end.map(|x| x + 1).unwrap_or(0);
    let ats: Vec<usize> = (0..n)
        .filter(|&j| clean[j] == b'@' && j >= protected_end)
        .collect();
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: positions[ats[1]],
        });
    }

    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: positions[sep],
        });
    }

    let mut local = String::new();
    let quoted;
    if clean[0] == b'"' {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: positions[end + 1],
            });
        }
        quoted = true;
        let mut decoded = Vec::<(u8, usize)>::new();
        let mut j = 1usize;
        while j < end {
            let b = clean[j];
            if b == b'\\' {
                if j + 1 >= end || !matches!(clean[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: positions[j],
                    });
                }
                decoded.push((clean[j + 1], positions[j]));
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: positions[j] });
                }
                decoded.push((b, positions[j]));
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: positions[end],
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: decoded[64].1,
            });
        }
        for (b, _) in decoded {
            local.push(b as char);
        }
    } else {
        quoted = false;
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: positions[64],
            });
        }
        for j in 0..sep {
            let b = clean[j];
            if b == b'.' {
                if j == 0 || j + 1 == sep || clean[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: positions[j] });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: positions[j] });
            }
            local.push(b as char);
        }
    }

    let domain_start = sep + 1;
    let mut domain_end = n;
    let root_dot = if domain_end > domain_start && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(positions[domain_end])
    } else {
        None
    };
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: if domain_start < n {
                positions[domain_start]
            } else {
                input.len()
            },
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start + 253],
        });
    }

    let mut label_starts = Vec::<usize>::new();
    let mut label_ends = Vec::<usize>::new();
    let mut start = domain_start;
    for j in domain_start..=domain_end {
        if j == domain_end || clean[j] == b'.' {
            if start == j {
                let at = if j == domain_end {
                    root_dot.unwrap_or(input.len())
                } else {
                    positions[j]
                };
                return Err(EmailError::EmptyLabel { at });
            }
            label_starts.push(start);
            label_ends.push(j);
            let len = j - start;
            if len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: positions[start + 63],
                });
            }
            if clean[start] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: positions[start],
                });
            }
            if clean[j - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: positions[j - 1],
                });
            }
            for k in start..j {
                let b = clean[k];
                if !b.is_ascii_alphanumeric() && b != b'-' {
                    return Err(EmailError::DomainChar { at: positions[k] });
                }
            }
            start = j + 1;
        }
    }

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: positions[domain_start],
        });
    }

    let last_start = *label_starts.last().unwrap();
    let last_end = *label_ends.last().unwrap();
    if clean[last_start..last_end]
        .iter()
        .all(|b| b.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: positions[last_start],
        });
    }

    let domain: String = clean[domain_start..domain_end]
        .iter()
        .map(|b| (b.to_ascii_lowercase()) as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
