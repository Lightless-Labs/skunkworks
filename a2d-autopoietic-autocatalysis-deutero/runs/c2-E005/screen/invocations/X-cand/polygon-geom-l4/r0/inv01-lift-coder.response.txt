#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
    Duplicate { first: usize, second: usize },
    ZeroArea,
    AdjacentOverlap { vertex: usize },
    Intersection { edge_a: usize, edge_b: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Validated {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub measures: Measures,
}

const LIMIT: i64 = 1_000_000_000;

fn in_range(p: Point) -> bool {
    (-LIMIT..=LIMIT).contains(&p.x) && (-LIMIT..=LIMIT).contains(&p.y)
}

fn prepare(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > 4096 {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }
    let mut vertices = points.to_vec();
    if vertices.len() >= 2 && vertices.last() == vertices.first() {
        vertices.pop();
    }
    if vertices.len() < 3 {
        return Err(GeomError::TooFewVertices { len: vertices.len() });
    }
    Ok(vertices)
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn dot_from(vertex: Point, a: Point, b: Point) -> i128 {
    (a.x as i128 - vertex.x as i128) * (b.x as i128 - vertex.x as i128)
        + (a.y as i128 - vertex.y as i128) * (b.y as i128 - vertex.y as i128)
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

fn signed_area2(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn rounded_sqrt(n: u128) -> u128 {
    let mut lo = 0u128;
    let mut hi = 1u128;
    while hi <= n / hi {
        hi *= 2;
    }
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid <= n / mid {
            lo = mid;
        } else {
            hi = mid;
        }
    }
    let lower_sq = lo * lo;
    let upper = lo + 1;
    let upper_sq = upper * upper;
    if n - lower_sq >= upper_sq - n {
        upper
    } else {
        lo
    }
}

fn edge_length_milli(a: Point, b: Point) -> u64 {
    let dx = a.x as i128 - b.x as i128;
    let dy = a.y as i128 - b.y as i128;
    let squared = (dx * dx + dy * dy) as u128 * 1_000_000;
    rounded_sqrt(squared) as u64
}

fn measures_for(points: &[Point]) -> Measures {
    let signed = signed_area2(points);
    let mut edge_milli = Vec::with_capacity(points.len());
    let mut perimeter = 0u64;
    for i in 0..points.len() {
        let length = edge_length_milli(points[i], points[(i + 1) % points.len()]);
        edge_milli.push(length);
        perimeter += length;
    }

    let longest_edge = if points.is_empty() {
        None
    } else {
        let mut best = 0usize;
        for i in 1..points.len() {
            let pair_i = ordered_pair(points[i], points[(i + 1) % points.len()]);
            let pair_best = ordered_pair(points[best], points[(best + 1) % points.len()]);
            if edge_milli[i] > edge_milli[best]
                || (edge_milli[i] == edge_milli[best]
                    && (pair_i < pair_best || (pair_i == pair_best && i < best)))
            {
                best = i;
            }
        }
        Some(best)
    };

    Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation: if signed > 0 {
            Orientation::CounterClockwise
        } else if signed < 0 {
            Orientation::Clockwise
        } else {
            Orientation::Degenerate
        },
        edge_milli,
        perimeter_milli: perimeter,
        longest_edge,
    }
}

fn ordered_pair(a: Point, b: Point) -> (Point, Point) {
    if a <= b { (a, b) } else { (b, a) }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let vertices = prepare(points)?;
    Ok(measures_for(&vertices))
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let vertices = prepare(points)?;
    if queries.len() > 4096 {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut result = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        for i in 0..vertices.len() {
            if on_segment(vertices[i], vertices[(i + 1) % vertices.len()], q) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            result.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..vertices.len() {
            let a = vertices[i];
            let b = vertices[(i + 1) % vertices.len()];
            if (a.y > q.y) != (b.y > q.y) {
                let c = cross(a, b, q);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                    inside = !inside;
                }
            }
        }
        result.push(if inside { Location::Inside } else { Location::Outside });
    }
    Ok(result)
}

fn cloud_points(points: &[Point]) -> Result<Vec<(Point, usize)>, GeomError> {
    if points.len() > 4096 {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }
    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|(p, _)| *p == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|(p, _)| *p);
    Ok(unique)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    let sorted = cloud_points(points)?;
    let m = sorted.len();
    if m == 0 {
        return Ok(Hull { vertices: vec![], source_indices: vec![], area2: 0, perimeter_milli: 0 });
    }
    if m == 1 {
        return Ok(Hull {
            vertices: vec![sorted[0].0],
            source_indices: vec![sorted[0].1],
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let all_collinear = (2..m).all(|i| cross(sorted[0].0, sorted[m - 1].0, sorted[i].0) == 0);
    let hull_indices: Vec<usize> = if all_collinear {
        match mode {
            Collinear::Include => (0..m).collect(),
            Collinear::Exclude => vec![0, m - 1],
        }
    } else if mode == Collinear::Exclude {
        let mut lower = Vec::new();
        for i in 0..m {
            while lower.len() >= 2
                && cross(sorted[lower[lower.len() - 2]].0, sorted[lower[lower.len() - 1]].0, sorted[i].0) <= 0
            {
                lower.pop();
            }
            lower.push(i);
        }
        let mut upper = Vec::new();
        for i in (0..m).rev() {
            while upper.len() >= 2
                && cross(sorted[upper[upper.len() - 2]].0, sorted[upper[upper.len() - 1]].0, sorted[i].0) <= 0
            {
                upper.pop();
            }
            upper.push(i);
        }
        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower
    } else {
        let mut lower = Vec::new();
        for i in 0..m {
            while lower.len() >= 2
                && cross(sorted[lower[lower.len() - 2]].0, sorted[lower[lower.len() - 1]].0, sorted[i].0) < 0
            {
                lower.pop();
            }
            lower.push(i);
        }
        let mut upper = Vec::new();
        for i in (0..m).rev() {
            while upper.len() >= 2
                && cross(sorted[upper[upper.len() - 2]].0, sorted[upper[upper.len() - 1]].0, sorted[i].0) < 0
            {
                upper.pop();
            }
            upper.push(i);
        }
        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower
    };

    let vertices: Vec<Point> = hull_indices.iter().map(|&i| sorted[i].0).collect();
    let source_indices: Vec<usize> = hull_indices.iter().map(|&i| sorted[i].1).collect();
    let area = if vertices.len() >= 3 { signed_area2(&vertices).unsigned_abs() } else { 0 };
    let perimeter = measures_for(&vertices).perimeter_milli;
    Ok(Hull { vertices, source_indices, area2: area, perimeter_milli: perimeter })
}

fn intersects(a: Point, b: Point, c: Point, d: Point) -> bool {
    let ab_c = cross(a, b, c);
    let ab_d = cross(a, b, d);
    let cd_a = cross(c, d, a);
    let cd_b = cross(c, d, b);
    if ab_c == 0 && on_segment(a, b, c) { return true; }
    if ab_d == 0 && on_segment(a, b, d) { return true; }
    if cd_a == 0 && on_segment(c, d, a) { return true; }
    if cd_b == 0 && on_segment(c, d, b) { return true; }
    (ab_c > 0) != (ab_d > 0) && (cd_a > 0) != (cd_b > 0)
}

pub fn validate(points: &[Point]) -> Result<Validated, GeomError> {
    let vertices = prepare(points)?;
    let n = vertices.len();

    for i in 0..n {
        for j in (i + 1)..n {
            if vertices[i] == vertices[j] {
                return Err(GeomError::Duplicate { first: i, second: j });
            }
        }
    }

    let area = signed_area2(&vertices);
    if area == 0 {
        return Err(GeomError::ZeroArea);
    }

    for i in 0..n {
        let prev = vertices[(i + n - 1) % n];
        let current = vertices[i];
        let next = vertices[(i + 1) % n];
        if cross(prev, current, next) == 0 && dot_from(current, prev, next) > 0 {
            return Err(GeomError::AdjacentOverlap { vertex: i });
        }
    }

    for i in 0..n {
        for j in (i + 1)..n {
            if j == i + 1 || (i == 0 && j == n - 1) {
                continue;
            }
            if intersects(vertices[i], vertices[(i + 1) % n], vertices[j], vertices[(j + 1) % n]) {
                return Err(GeomError::Intersection { edge_a: i, edge_b: j });
            }
        }
    }

    let min_index = (0..n).min_by_key(|&i| vertices[i]).unwrap();
    let mut order = Vec::with_capacity(n);
    if area > 0 {
        for k in 0..n { order.push((min_index + k) % n); }
    } else {
        for k in 0..n { order.push((min_index + n - k) % n); }
    }
    let canonical: Vec<Point> = order.iter().map(|&i| vertices[i]).collect();
    let measures = measures_for(&canonical);
    Ok(Validated {
        vertices: canonical,
        source_indices: order,
        measures,
    })
}
