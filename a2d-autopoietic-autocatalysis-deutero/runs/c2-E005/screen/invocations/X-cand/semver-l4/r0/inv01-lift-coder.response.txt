use std::cmp::Ordering;
use std::collections::HashSet;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_sep_missing(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    if *pos >= bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
        return Err(VersionError::MissingCore { at: start });
    }
    if !bytes[*pos].is_ascii_digit() {
        return Err(VersionError::InvalidChar { at: start });
    }
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    let field = &s[start..*pos];
    if field.len() > 1 && field.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    field.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut at = start;
    loop {
        let id_start = at;
        while at < end && bytes[at] != b'.' {
            let b = bytes[at];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at });
            }
            at += 1;
        }
        if at == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..at];
        if prerelease
            && id.len() > 1
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.starts_with('0')
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_owned());
        if at == end {
            break;
        }
        at += 1;
        if at == end {
            return Err(VersionError::EmptyIdentifier { at });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let bytes = s.as_bytes();
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    if bytes.get(pos) != Some(&b'.') {
        return if is_sep_missing(bytes.get(pos).copied()) {
            Err(VersionError::MissingCore { at: pos })
        } else {
            Err(VersionError::InvalidChar { at: pos })
        };
    }
    pos += 1;
    let minor = parse_core(s, &mut pos)?;
    if bytes.get(pos) != Some(&b'.') {
        return if is_sep_missing(bytes.get(pos).copied()) {
            Err(VersionError::MissingCore { at: pos })
        } else {
            Err(VersionError::InvalidChar { at: pos })
        };
    }
    pos += 1;
    let patch = parse_core(s, &mut pos)?;
    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = bytes[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|i| pre_start + i);
        let pre_end = plus.unwrap_or(bytes.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        pos = pre_end;
    }
    if pos < bytes.len() && bytes[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, bytes.len(), false)?;
        pos = bytes.len();
    }
    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version { major, minor, patch, pre, build })
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_numeric_id(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core(a).cmp(&core(b)).then_with(|| {
        match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ord = match (numeric_id(x), numeric_id(y)) {
                        (true, true) => cmp_numeric_id(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        }
    })
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone)]
struct Token<'a> {
    text: &'a str,
    start: usize,
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str) -> Vec<Token<'_>> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !is_ws(bytes[i]) {
            i += 1;
        }
        result.push(Token { text: &s[start..i], start });
    }
    result
}

fn version_error(at: usize, error: VersionError) -> RangeError {
    RangeError::Version { at, error }
}

fn version_full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|e| version_error(at, e))
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    full: Option<Version>,
    numeric_count: usize,
}

fn parse_partial(text: &str, start: usize) -> Result<Partial, RangeError> {
    let fields: Vec<&str> = text.split('.').collect();
    let full_candidate = text.contains('-') || text.contains('+')
        || (fields.len() >= 3 && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X")));
    if full_candidate {
        let v = version_full_operand(text, start)?;
        return Ok(Partial {
            fields: vec![Some(v.major), Some(v.minor), Some(v.patch)],
            full: Some(v),
            numeric_count: 3,
        });
    }

    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at: start });
    }
    let mut parsed = Vec::new();
    let mut wild_seen = false;
    let mut numeric_count = 0;
    for f in fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at: start });
        }
        if matches!(f, "*" | "x" | "X") {
            wild_seen = true;
            parsed.push(None);
        } else {
            if wild_seen
                || !f.bytes().all(|b| b.is_ascii_digit())
                || (f.len() > 1 && f.starts_with('0'))
            {
                return Err(RangeError::Syntax { at: start });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: start })?;
            numeric_count += 1;
            parsed.push(Some(n));
        }
    }
    Ok(Partial { fields: parsed, full: None, numeric_count })
}

fn padded(p: &Partial) -> (u64, u64, u64) {
    let mut vals = [0u64; 3];
    for (i, f) in p.fields.iter().enumerate() {
        vals[i] = f.unwrap_or(0);
    }
    (vals[0], vals[1], vals[2])
}

fn make_version(t: (u64, u64, u64), pre: Vec<String>) -> Version {
    Version { major: t.0, minor: t.1, patch: t.2, pre, build: Vec::new() }
}

fn endpoint(t: (u64, u64, u64), pre_zero: bool) -> Version {
    make_version(t, if pre_zero { vec!["0".to_owned()] } else { Vec::new() })
}

fn bump(t: (u64, u64, u64), field: usize, at: usize) -> Result<(u64, u64, u64), RangeError> {
    let mut vals = [t.0, t.1, t.2];
    vals[field] = vals[field].checked_add(1).ok_or(RangeError::Syntax { at })?;
    for x in vals.iter_mut().skip(field + 1) {
        *x = 0;
    }
    Ok((vals[0], vals[1], vals[2]))
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version) {
    arm.comparators.push(Comparator { op, version });
    if !arm.comparators.last().unwrap().version.pre.is_empty() {
        let t = core(&arm.comparators.last().unwrap().version);
        if !arm.anchors.contains(&t) {
            arm.anchors.push(t);
        }
    }
}

fn parse_bare_or_expanded(tok: &Token<'_>, arm: &mut Arm) -> Result<(), RangeError> {
    let p = parse_partial(tok.text, tok.start)?;
    if let Some(v) = p.full {
        add_cmp(arm, Op::Eq, v);
        return Ok(());
    }
    let k = p.numeric_count;
    let lower = padded(&p);
    if k == 0 {
        return Ok(());
    }
    add_cmp(arm, Op::Ge, make_version(lower, Vec::new()));
    let field = if k == 1 { 0 } else { 1 };
    let upper = bump(lower, field, tok.start)?;
    add_cmp(arm, Op::Lt, endpoint(upper, true));
    Ok(())
}

fn parse_caret_tilde(tok: &Token<'_>, arm: &mut Arm, caret: bool) -> Result<(), RangeError> {
    let operand = &tok.text[1..];
    let start = tok.start + 1;
    if operand.is_empty() {
        return Err(RangeError::Syntax { at: tok.start });
    }
    let p = parse_partial(operand, start)?;
    if p.numeric_count == 0 {
        return Err(RangeError::Syntax { at: tok.start });
    }
    let lower = if let Some(v) = p.full.clone() {
        v
    } else {
        make_version(padded(&p), Vec::new())
    };
    let t = core(&p.full.clone().unwrap_or_else(|| make_version(padded(&p), Vec::new())));
    let k = p.numeric_count;
    let field = if caret {
        if t.0 > 0 || k == 1 { 0 }
        else if t.1 > 0 || k == 2 { 1 }
        else { 2 }
    } else if k == 1 { 0 } else { 1 };
    let upper = bump(t, field, start)?;
    add_cmp(arm, Op::Ge, lower);
    add_cmp(arm, Op::Lt, endpoint(upper, true));
    Ok(())
}

fn parse_explicit_comparator(tok: &Token<'_>, arm: &mut Arm) -> Result<(), RangeError> {
    let (op, n) = if tok.text.starts_with(">=") {
        (Op::Ge, 2)
    } else if tok.text.starts_with("<=") {
        (Op::Le, 2)
    } else {
        match tok.text.as_bytes()[0] {
            b'>' => (Op::Gt, 1),
            b'<' => (Op::Lt, 1),
            b'=' => (Op::Eq, 1),
            _ => unreachable!(),
        }
    };
    let operand = &tok.text[n..];
    let start = tok.start + n;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: tok.start });
    }
    let v = version_full_operand(operand, start)?;
    add_cmp(arm, op, v);
    Ok(())
}

fn parse_hyphen_arm(tokens: &[Token<'_>], arm: &mut Arm) -> Result<(), RangeError> {
    let minus = tokens.iter().position(|t| t.text == "-").unwrap();
    if tokens.len() != 3 || minus != 1 {
        return Err(RangeError::Syntax { at: tokens[minus].start });
    }
    let lo = parse_partial(tokens[0].text, tokens[0].start)?;
    let hi = parse_partial(tokens[2].text, tokens[2].start)?;
    if lo.numeric_count == 0 {
        return Err(RangeError::Syntax { at: tokens[0].start });
    }
    if hi.numeric_count == 0 {
        return Err(RangeError::Syntax { at: tokens[2].start });
    }
    let lower = lo.full.clone().unwrap_or_else(|| make_version(padded(&lo), Vec::new()));
    add_cmp(arm, Op::Ge, lower);
    if let Some(v) = hi.full {
        add_cmp(arm, Op::Le, v);
    } else {
        let t = padded(&hi);
        let upper = bump(t, if hi.numeric_count == 1 { 0 } else { 1 }, tokens[2].start)?;
        add_cmp(arm, Op::Lt, endpoint(upper, true));
    }
    Ok(())
}

fn parse_arm(tokens: &[Token<'_>]) -> Result<Arm, RangeError> {
    let mut arm = Arm { comparators: Vec::new(), anchors: Vec::new() };
    if tokens.iter().any(|t| t.text == "-") {
        parse_hyphen_arm(tokens, &mut arm)?;
        return Ok(arm);
    }
    for tok in tokens {
        let b = tok.text.as_bytes()[0];
        match b {
            b'<' | b'>' | b'=' => parse_explicit_comparator(tok, &mut arm)?,
            b'^' | b'~' => parse_caret_tilde(tok, &mut arm, b == b'^')?,
            b'0'..=b'9' | b'*' | b'x' | b'X' => parse_bare_or_expanded(tok, &mut arm)?,
            _ => return Err(RangeError::Syntax { at: tok.start }),
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut groups: Vec<Vec<Token<'_>>> = vec![Vec::new()];
    let mut separators = Vec::new();
    for token in tokens {
        if token.text == "||" {
            separators.push(token.start);
            groups.push(Vec::new());
        } else {
            groups.last_mut().unwrap().push(token);
        }
    }
    for i in 0..groups.len() {
        if groups[i].is_empty() {
            let at = if i + 1 == groups.len() {
                s.len()
            } else {
                separators[i]
            };
            return Err(RangeError::Empty { at });
        }
    }
    let mut arms = Vec::new();
    for group in &groups {
        arms.push(parse_arm(group)?);
    }
    Ok(Range { arms })
}

fn cmp_satisfies(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !arm.comparators.iter().all(|c| cmp_satisfies(v, c)) {
        return false;
    }
    include_prerelease || v.pre.is_empty() || arm.anchors.contains(&core(v))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates.iter().enumerate().map(|(index, s)| {
        parse_version(s).map_err(|error| CandidateError { index, error })
    }).collect::<Result<_, _>>()?;
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (max && ord == Ordering::Greater) || (!max && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn increment_tuple(t: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if let Some(p) = t.2.checked_add(1) {
        return Some((t.0, t.1, p));
    }
    if let Some(m) = t.1.checked_add(1) {
        return Some((t.0, m, 0));
    }
    t.0.checked_add(1).map(|a| (a, 0, 0))
}

fn tuple_version(t: (u64, u64, u64), pre: &[String]) -> Version {
    make_version(t, pre.to_vec())
}

fn boundary_possible(lower: Option<(&Version, bool)>, upper: Option<(&Version, bool)>, anchors: &[(u64,u64,u64)]) -> bool {
    let cmp_bounds = |a: &Version, b: &Version| compare_versions(a, b);
    if let (Some((l, li)), Some((u, ui))) = (lower, upper) {
        let ord = cmp_bounds(l, u);
        if ord == Ordering::Greater || (ord == Ordering::Equal && (!li || !ui)) {
            return false;
        }
        if ord == Ordering::Equal {
            return true;
        }
    }
    let stable_exists = match (lower, upper) {
        (Some((l, inc)), Some((u, _))) => {
            if !l.pre.is_empty() {
                let t = core(l);
                let rel = make_version(t, Vec::new());
                let lo_ok = compare_versions(&rel, l) == Ordering::Greater
                    || (*inc && compare_versions(&rel, l) == Ordering::Equal);
                lo_ok && compare_versions(&rel, u) != Ordering::Greater
            } else {
                compare_versions(l, u) != Ordering::Greater
            }
        }
        _ => true,
    };
    if stable_exists {
        return true;
    }
    anchors.iter().any(|t| {
        let least = endpoint(*t, true);
        let l_ok = lower.map_or(true, |(l, inc)| {
            let o = compare_versions(&least, l);
            o == Ordering::Greater || (inc && o == Ordering::Equal)
        });
        let u_ok = upper.map_or(true, |(u, inc)| {
            let o = compare_versions(&least, u);
            o == Ordering::Less || (inc && o == Ordering::Equal)
        });
        l_ok && u_ok
    })
}

fn bounds(arm: &Arm) -> (Option<(Version, bool)>, Option<(Version, bool)>) {
    let mut lower: Option<(Version, bool)> = None;
    let mut upper: Option<(Version, bool)> = None;
    for c in &arm.comparators {
        match c.op {
            Op::Eq => {
                lower = Some((c.version.clone(), true));
                upper = Some((c.version.clone(), true));
            }
            Op::Gt | Op::Ge => {
                let inc = c.op == Op::Ge;
                let replace = lower.as_ref().map_or(true, |(v, old_inc)| {
                    compare_versions(&c.version, v) == Ordering::Greater
                        || (compare_versions(&c.version, v) == Ordering::Equal && !inc && *old_inc)
                });
                if replace {
                    lower = Some((c.version.clone(), inc));
                }
            }
            Op::Lt | Op::Le => {
                let inc = c.op == Op::Le;
                let replace = upper.as_ref().map_or(true, |(v, old_inc)| {
                    compare_versions(&c.version, v) == Ordering::Less
                        || (compare_versions(&c.version, v) == Ordering::Equal && !inc && *old_inc)
                });
                if replace {
                    upper = Some((c.version.clone(), inc));
                }
            }
        }
    }
    (lower, upper)
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let mut all = aa.comparators.clone();
            all.extend(bb.comparators.clone());
            let mut anchors = aa.anchors.clone();
            for t in &bb.anchors {
                if !anchors.contains(t) {
                    anchors.push(*t);
                }
            }
            if include_prerelease {
                if interval_has_witness(&all) {
                    return true;
                }
            } else {
                let tmp = Arm { comparators: all, anchors: Vec::new() };
                let (l, u) = bounds(&tmp);
                let lref = l.as_ref().map(|(v, i)| (v, *i));
                let uref = u.as_ref().map(|(v, i)| (v, *i));
                if stable_or_anchor_witness(lref, uref, &aa.anchors, &bb.anchors) {
                    return true;
                }
            }
        }
    }
    false
}

fn stable_or_anchor_witness(
    lower: Option<(&Version, bool)>,
    upper: Option<(&Version, bool)>,
    aa: &[(u64,u64,u64)],
    bb: &[(u64,u64,u64)],
) -> bool {
    if stable_witness(lower, upper) {
        return true;
    }
    aa.iter().any(|t| bb.contains(t) && {
        let least = endpoint(*t, true);
        lower.map_or(true, |(v, inc)| {
            let o = compare_versions(&least, v);
            o == Ordering::Greater || (inc && o == Ordering::Equal)
        }) && upper.map_or(true, |(v, inc)| {
            let o = compare_versions(&least, v);
            o == Ordering::Less || (inc && o == Ordering::Equal)
        })
    })
}

fn stable_witness(lower: Option<(&Version, bool)>, upper: Option<(&Version, bool)>) -> bool {
    match (lower, upper) {
        (Some((l, li)), Some((u, ui))) => {
            let ord = compare_versions(l, u);
            if ord == Ordering::Greater || (ord == Ordering::Equal && (!li || !ui)) {
                return false;
            }
            if l.pre.is_empty() {
                if ord == Ordering::Equal { return li && ui; }
                return true;
            }
            let t = core(l);
            let rel = make_version(t, Vec::new());
            let low_ok = compare_versions(&rel, l) == Ordering::Greater
                || (li && compare_versions(&rel, l) == Ordering::Equal);
            low_ok && (compare_versions(&rel, u) == Ordering::Less
                || (ui && compare_versions(&rel, u) == Ordering::Equal))
        }
        (Some((l, li)), None) => {
            if l.pre.is_empty() { true }
            else {
                let rel = make_version(core(l), Vec::new());
                compare_versions(&rel, l) == Ordering::Greater
                    || (li && compare_versions(&rel, l) == Ordering::Equal)
            }
        }
        (None, Some((u, _))) => u.major > 0 || u.minor > 0 || u.patch > 0 || u.pre.is_empty(),
        (None, None) => true,
    }
}

fn interval_has_witness(comparators: &[Comparator]) -> bool {
    let arm = Arm { comparators: comparators.to_vec(), anchors: Vec::new() };
    let (l, u) = bounds(&arm);
    let lr = l.as_ref().map(|(v, i)| (v, *i));
    let ur = u.as_ref().map(|(v, i)| (v, *i));
    stable_witness(lr, ur) || match (lr, ur) {
        (Some((l, li)), Some((u, ui))) => compare_versions(l, u) != Ordering::Greater
            && (compare_versions(l, u) != Ordering::Equal || (li && ui)),
        _ => true,
    }
}

pub fn canonical(r: &Range) -> String {
    let mut out = Vec::new();
    for arm in &r.arms {
        let (l, u) = bounds(arm);
        let lref = l.as_ref().map(|(v, i)| (v, *i));
        let uref = u.as_ref().map(|(v, i)| (v, *i));
        if !boundary_possible(lref, uref, &arm.anchors) {
            continue;
        }
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let lower = l.as_ref().map_or("*".to_owned(), |(v, inc)| {
            format!("{}{}", if *inc { ">=" } else { ">" }, version_without_build(v))
        });
        let upper = u.as_ref().map_or("*".to_owned(), |(v, inc)| {
            format!("{}{}", if *inc { "<=" } else { "<" }, version_without_build(v))
        });
        let anchor_text = if anchors.is_empty() {
            "-".to_owned()
        } else {
            anchors.iter().map(|t| format!("{}.{}.{}", t.0, t.1, t.2)).collect::<Vec<_>>().join(",")
        };
        out.push(format!("[{};{};pre={}]", lower, upper, anchor_text));
    }
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".to_owned() } else { out.join(" || ") }
}

fn version_without_build(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

#[derive(Clone, Debug)]
pub enum SetExpr {
    Range(String),
    Union(Vec<SetExpr>),
    Intersection(Vec<SetExpr>),
}

#[derive(Clone, Debug)]
pub struct Constraint {
    pub label: String,
    pub expr: SetExpr,
}

#[derive(Clone, Debug)]
pub struct Package {
    pub name: String,
    pub candidates: Vec<String>,
    pub constraints: Vec<Constraint>,
}

#[derive(Clone, Copy, Debug)]
pub enum Preference {
    Highest,
    Lowest,
    StableFirst,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_ranges_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|_| ()).map_err(|error| SelectError::InvalidRange {
            package: package.to_owned(),
            label: label.to_owned(),
            path: path.clone(),
            error,
        }),
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_ranges_expr(child, package, label, path)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn expr_matches(expr: &SetExpr, v: &Version, include: bool) -> bool {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|r| matches_with(v, &r, include)).unwrap_or(false),
        SetExpr::Union(children) => children.iter().any(|e| expr_matches(e, v, include)),
        SetExpr::Intersection(children) => children.iter().all(|e| expr_matches(e, v, include)),
    }
}

fn subset_unsatisfiable(indices: &[usize], constraints: &[Constraint], candidates: &[Version], include: bool) -> bool {
    candidates.iter().all(|v| indices.iter().all(|&i| expr_matches(&constraints[i].expr, v, include)) == false)
}

fn choose_conflict(
    constraints: &[Constraint],
    candidates: &[Version],
    include: bool,
) -> Vec<String> {
    if candidates.is_empty() {
        return Vec::new();
    }
    let n = constraints.len();
    for size in 1..=n {
        let mut found: Vec<Vec<String>> = Vec::new();
        fn combinations(
            start: usize,
            size: usize,
            cur: &mut Vec<usize>,
            n: usize,
            constraints: &[Constraint],
            candidates: &[Version],
            include: bool,
            found: &mut Vec<Vec<String>>,
        ) {
            if cur.len() == size {
                if subset_unsatisfiable(cur, constraints, candidates, include) {
                    let mut labels: Vec<String> = cur.iter().map(|&i| constraints[i].label.clone()).collect();
                    labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
                    found.push(labels);
                }
                return;
            }
            let need = size - cur.len();
            for i in start..=n - need {
                cur.push(i);
                combinations(i + 1, size, cur, n, constraints, candidates, include, found);
                cur.pop();
            }
        }
        combinations(0, size, &mut Vec::new(), n, constraints, candidates, include, &mut found);
        if !found.is_empty() {
            found.sort_by(|a, b| {
                for (x, y) in a.iter().zip(b) {
                    let o = x.as_bytes().cmp(y.as_bytes());
                    if o != Ordering::Equal { return o; }
                }
                a.len().cmp(&b.len())
            });
            return found.remove(0);
        }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }
    for p in packages {
        let mut labels = HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }
    let mut parsed_by_package = Vec::new();
    for p in packages {
        let mut parsed = Vec::new();
        for (index, s) in p.candidates.iter().enumerate() {
            parsed.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index, error,
            })?);
        }
        parsed_by_package.push(parsed);
    }
    for p in packages {
        for c in &p.constraints {
            validate_ranges_expr(&c.expr, &p.name, &c.label, &mut Vec::new())?;
        }
    }

    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let parsed = &parsed_by_package[pi];
        let qualifying: Vec<usize> = parsed.iter().enumerate().filter_map(|(i, v)| {
            if p.constraints.iter().all(|c| expr_matches(&c.expr, v, include_prerelease)) {
                Some(i)
            } else {
                None
            }
        }).collect();

        if qualifying.is_empty() {
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels: choose_conflict(&p.constraints, parsed, include_prerelease),
            });
        }

        let mut best = qualifying[0];
        for &i in qualifying.iter().skip(1) {
            let a = &parsed[i];
            let b = &parsed[best];
            let better = match preference {
                Preference::Highest => compare_versions(a, b) == Ordering::Greater,
                Preference::Lowest => compare_versions(a, b) == Ordering::Less,
                Preference::StableFirst => {
                    match (a.pre.is_empty(), b.pre.is_empty()) {
                        (true, false) => true,
                        (false, true) => false,
                        _ => compare_versions(a, b) == Ordering::Greater,
                    }
                }
            };
            if better {
                best = i;
            }
        }
        selections.push(Selection {
            package: p.name.clone(),
            version: p.candidates[best].clone(),
        });
    }
    Ok(selections)
}
