use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_stop(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    let mut end = start;
    while end < bytes.len() && bytes[end].is_ascii_digit() {
        end += 1;
    }
    if end == start {
        if is_core_stop(bytes.get(start).copied()) {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    let digits = &s[start..end];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    let value = digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })?;
    *pos = end;
    Ok(value)
}

fn valid_ident_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let ident_start = p;
        while p < end && bytes[p] != b'.' {
            if !valid_ident_byte(bytes[p]) {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let ident = &s[ident_start..p];
        if pre
            && ident.len() > 1
            && ident.bytes().all(|b| b.is_ascii_digit())
            && ident.starts_with('0')
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        out.push(ident.to_owned());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    if s.as_bytes().get(p) != Some(&b'.') {
        return if is_core_stop(s.as_bytes().get(p).copied()) {
            Err(VersionError::MissingCore { at: p })
        } else {
            Err(VersionError::InvalidChar { at: p })
        };
    }
    p += 1;
    let minor = parse_core(s, &mut p)?;
    if s.as_bytes().get(p) != Some(&b'.') {
        return if is_core_stop(s.as_bytes().get(p).copied()) {
            Err(VersionError::MissingCore { at: p })
        } else {
            Err(VersionError::InvalidChar { at: p })
        };
    }
    p += 1;
    let patch = parse_core(s, &mut p)?;
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if p < bytes.len() && bytes[p] == b'-' {
        p += 1;
        let pre_start = p;
        let plus = bytes[p..].iter().position(|&b| b == b'+').map(|i| p + i);
        let pre_end = plus.unwrap_or(bytes.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        p = pre_end;
    }

    if p < bytes.len() && bytes[p] == b'+' {
        p += 1;
        build = parse_identifiers(s, p, bytes.len(), false)?;
        p = bytes.len();
    }

    if p != bytes.len() {
        return Err(VersionError::InvalidChar { at: p });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_ident(a: &str, b: &str) -> Ordering {
    let an = numeric_identifier(a);
    let bn = numeric_identifier(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core(a).cmp(&core(b)).then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.pre.iter().zip(&b.pre) {
                let c = compare_ident(x, y);
                if c != Ordering::Equal {
                    return c;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparisons: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn cmp_matches(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens(s: &str, base: usize) -> Vec<(String, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = 0;
    while p < bytes.len() {
        while p < bytes.len() && whitespace(bytes[p]) {
            p += 1;
        }
        if p == bytes.len() {
            break;
        }
        let start = p;
        while p < bytes.len() && !whitespace(bytes[p]) {
            p += 1;
        }
        out.push((s[start..p].to_owned(), base + start));
    }
    out
}

fn full_operand(token: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(token).map_err(|error| RangeError::Version { at, error })
}

#[derive(Clone, Debug)]
enum ParsedOperand {
    Full(Version),
    Partial(Vec<Option<u64>>),
}

fn parse_partial(token: &str, at: usize) -> Result<ParsedOperand, RangeError> {
    let fields: Vec<&str> = token.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut wildcard_seen = false;
    let mut out = Vec::new();
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
            out.push(None);
        } else if f.bytes().all(|b| b.is_ascii_digit()) {
            if wildcard_seen || (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            out.push(Some(n));
        } else {
            return Err(RangeError::Syntax { at });
        }
    }
    if out.iter().any(Option::is_none) {
        if out.iter().skip_while(|x| x.is_some()).any(Option::is_some) {
            return Err(RangeError::Syntax { at });
        }
    }
    Ok(ParsedOperand::Partial(out))
}

fn parse_operand(token: &str, at: usize) -> Result<ParsedOperand, RangeError> {
    let has_suffix = token.contains('-') || token.contains('+');
    let fields: Vec<&str> = token.split('.').collect();
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if has_suffix || (fields.len() >= 3 && !has_wild) {
        return full_operand(token, at).map(ParsedOperand::Full);
    }
    parse_partial(token, at)
}

fn version_tuple(major: u64, minor: u64, patch: u64, pre: Vec<String>) -> Version {
    Version {
        major,
        minor,
        patch,
        pre,
        build: Vec::new(),
    }
}

fn padded(parts: &[Option<u64>]) -> (u64, u64, u64) {
    (
        parts.first().and_then(|x| *x).unwrap_or(0),
        parts.get(1).and_then(|x| *x).unwrap_or(0),
        parts.get(2).and_then(|x| *x).unwrap_or(0),
    )
}

fn increment_tuple(
    tuple: (u64, u64, u64),
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let (mut a, mut b, mut c) = tuple;
    let target = match field {
        0 => &mut a,
        1 => &mut b,
        _ => &mut c,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    if field == 0 {
        b = 0;
        c = 0;
    } else if field == 1 {
        c = 0;
    }
    Ok(version_tuple(a, b, c, vec!["0".to_owned()]))
}

fn anchor(v: &Version, anchors: &mut Vec<(u64, u64, u64)>) {
    if !v.pre.is_empty() {
        anchors.push(core(v));
    }
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version) {
    arm.comparisons.push(Comparator { op, version });
}

fn add_bare(arm: &mut Arm, operand: ParsedOperand, at: usize) -> Result<(), RangeError> {
    match operand {
        ParsedOperand::Full(v) => {
            anchor(&v, &mut arm.anchors);
            add_cmp(arm, Op::Eq, v);
        }
        ParsedOperand::Partial(parts) => {
            let k = parts.iter().take_while(|x| x.is_some()).count();
            if k == 0 {
                return Ok(());
            }
            let tuple = padded(&parts);
            add_cmp(arm, Op::Ge, version_tuple(tuple.0, tuple.1, tuple.2, Vec::new()));
            if k <= 2 {
                let upper = increment_tuple(tuple, k - 1, at)?;
                add_cmp(arm, Op::Lt, upper);
            }
        }
    }
    Ok(())
}

fn add_caret_tilde(
    arm: &mut Arm,
    operand: ParsedOperand,
    at: usize,
    caret: bool,
) -> Result<(), RangeError> {
    match operand {
        ParsedOperand::Full(v) => {
            anchor(&v, &mut arm.anchors);
            let tuple = core(&v);
            add_cmp(arm, Op::Ge, v);
            let field = if caret {
                if tuple.0 > 0 {
                    0
                } else if tuple.1 > 0 {
                    1
                } else {
                    2
                }
            } else if tuple.0 > 0 && false {
                0
            } else {
                1
            };
            let upper = increment_tuple(tuple, field, at)?;
            add_cmp(arm, Op::Lt, upper);
        }
        ParsedOperand::Partial(parts) => {
            let k = parts.iter().take_while(|x| x.is_some()).count();
            if k == 0 {
                return Err(RangeError::Syntax { at });
            }
            let tuple = padded(&parts);
            add_cmp(arm, Op::Ge, version_tuple(tuple.0, tuple.1, tuple.2, Vec::new()));
            let field = if caret {
                if tuple.0 > 0 || k == 1 {
                    0
                } else if tuple.1 > 0 || k == 2 {
                    1
                } else {
                    2
                }
            } else if k == 1 {
                0
            } else {
                1
            };
            let upper = increment_tuple(tuple, field, at)?;
            add_cmp(arm, Op::Lt, upper);
        }
    }
    Ok(())
}

fn parse_hyphen_arm(ts: &[(String, usize)]) -> Result<Arm, RangeError> {
    let minus = ts.iter().position(|(s, _)| s == "-").unwrap();
    if ts.len() != 3 || minus != 1 {
        return Err(RangeError::Syntax { at: ts[minus].1 });
    }
    let mut arm = Arm {
        comparisons: Vec::new(),
        anchors: Vec::new(),
    };
    let lo = parse_operand(&ts[0].0, ts[0].1)?;
    let hi = parse_operand(&ts[2].0, ts[2].1)?;
    let (lparts, lfull) = match lo {
        ParsedOperand::Full(v) => (Some(vec![Some(v.major), Some(v.minor), Some(v.patch)]), Some(v)),
        ParsedOperand::Partial(p) => (Some(p), None),
    };
    let (hparts, hfull) = match hi {
        ParsedOperand::Full(v) => (Some(vec![Some(v.major), Some(v.minor), Some(v.patch)]), Some(v)),
        ParsedOperand::Partial(p) => (Some(p), None),
    };
    let lp = lparts.unwrap();
    let hp = hparts.unwrap();
    let lk = lp.iter().take_while(|x| x.is_some()).count();
    let hk = hp.iter().take_while(|x| x.is_some()).count();
    if lk == 0 {
        return Err(RangeError::Syntax { at: ts[0].1 });
    }
    if hk == 0 {
        return Err(RangeError::Syntax { at: ts[2].1 });
    }
    let lt = padded(&lp);
    add_cmp(&mut arm, Op::Ge, lfull.unwrap_or_else(|| version_tuple(lt.0, lt.1, lt.2, Vec::new())));
    if let Some(v) = hfull {
        anchor(&v, &mut arm.anchors);
        add_cmp(&mut arm, Op::Le, v);
    } else {
        let ht = padded(&hp);
        let upper = if hk <= 2 {
            increment_tuple(ht, hk - 1, ts[2].1)?
        } else {
            version_tuple(ht.0, ht.1, ht.2, Vec::new())
        };
        if hk == 3 {
            add_cmp(&mut arm, Op::Le, upper);
        } else {
            add_cmp(&mut arm, Op::Lt, upper);
        }
    }
    Ok(arm)
}

fn parse_arm(ts: &[(String, usize)]) -> Result<Arm, RangeError> {
    if let Some((_, at)) = ts.iter().find(|(s, _)| s == "-") {
        let _ = at;
        return parse_hyphen_arm(ts);
    }
    let mut arm = Arm {
        comparisons: Vec::new(),
        anchors: Vec::new(),
    };
    for (token, at) in ts {
        if token.starts_with('^') || token.starts_with('~') {
            let caret = token.as_bytes()[0] == b'^';
            let op_start = at + 1;
            let operand = &token[1..];
            if operand.is_empty() {
                return Err(RangeError::Syntax { at: *at });
            }
            let parsed = parse_operand(operand, op_start)?;
            add_caret_tilde(&mut arm, parsed, *at, caret)?;
        } else if matches!(token.as_str(), "*" | "x" | "X")
            || token.as_bytes().first().is_some_and(u8::is_ascii_digit)
        {
            let parsed = parse_operand(token, *at)?;
            add_bare(&mut arm, parsed, *at)?;
        } else if token.starts_with('<')
            || token.starts_with('>')
            || token.starts_with('=')
        {
            let bytes = token.as_bytes();
            let (op, n) = if token.starts_with("<=") {
                (Op::Le, 2)
            } else if token.starts_with(">=") {
                (Op::Ge, 2)
            } else if token.starts_with('<') {
                (Op::Lt, 1)
            } else if token.starts_with('>') {
                (Op::Gt, 1)
            } else {
                (Op::Eq, 1)
            };
            if bytes.len() <= n || !bytes[n].is_ascii_digit() {
                return Err(RangeError::Syntax { at: *at });
            }
            let v = full_operand(&token[n..], at + n)?;
            anchor(&v, &mut arm.anchors);
            add_cmp(&mut arm, op, v);
        } else {
            return Err(RangeError::Syntax { at: *at });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let all = tokens(s, 0);
    if all.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < all.len() {
        if all[i].0 == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: all[i].1 });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
            i += 1;
            if i == all.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            current.push(all[i].clone());
            i += 1;
        }
    }
    if current.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&current)?);
    Ok(Range { arms })
}

fn arm_admits(arm: &Arm, v: &Version) -> bool {
    arm.anchors.iter().any(|&t| t == core(v))
}

fn arm_matches(v: &Version, arm: &Arm, include: bool) -> bool {
    if !arm.comparisons.iter().all(|c| cmp_matches(v, c)) {
        return false;
    }
    v.pre.is_empty() || include || arm_admits(arm, v)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates
        .iter()
        .enumerate()
        .map(|(index, s)| {
            parse_version(s).map_err(|error| CandidateError { index, error })
        })
        .collect::<Result<_, _>>()?;
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let c = compare_versions(v, &parsed[j]);
            if (max && c == Ordering::Greater) || (!max && c == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn next_core(t: (u64, u64, u64)) -> Option<(u64, u64, u64)> {
    if t.2 < u64::MAX {
        Some((t.0, t.1, t.2 + 1))
    } else if t.1 < u64::MAX {
        Some((t.0, t.1 + 1, 0))
    } else if t.0 < u64::MAX {
        Some((t.0 + 1, 0, 0))
    } else {
        None
    }
}

fn least_pre(t: (u64, u64, u64>) -> Version {
    version_tuple(t.0, t.1, t.2, vec!["0".to_owned()])
}
