use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1_000_000_000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;

    fn zero() -> Self {
        Self { negative: false, limbs: vec![0] }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.limbs.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn abs(&self) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    fn mul_small_add(&mut self, multiplier: u32, addend: u32) {
        let mut carry = addend as u64;
        for limb in &mut self.limbs {
            let value = (*limb as u64) * multiplier as u64 + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
        self.normalize();
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let mut start = 0;
        let mut negative = false;
        if bytes[0] == b'+' || bytes[0] == b'-' {
            negative = bytes[0] == b'-';
            start = 1;
            if start == bytes.len() {
                return Err(Error::SignOnly);
            }
        }

        let mut invalid = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !Self::digit_value(byte).is_some_and(|d| d < radix) {
                invalid = Some(index);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || bytes[index - 1] == b'_'
                    || bytes[index + 1] == b'_')
            {
                return Err(Error::Separator { index });
            }
        }

        let digit_indices: Vec<usize> = (start..bytes.len())
            .filter(|&index| bytes[index] != b'_')
            .collect();

        if digit_indices.len() > 1 && Self::digit_value(bytes[digit_indices[0]]) == Some(0) {
            return Err(Error::LeadingZero { index: digit_indices[0] });
        }

        if digit_indices.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_indices[max_digits],
            });
        }

        let mut result = Self::zero();
        for index in digit_indices {
            result.mul_small_add(radix, Self::digit_value(bytes[index]).unwrap());
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut n = self.abs();
        let mut digits = Vec::new();
        if n.is_zero() {
            digits.push(0u8);
        } else {
            while !n.is_zero() {
                let remainder = n.div_small(radix);
                digits.push(remainder as u8);
            }
        }

        while digits.len() < min_digits {
            digits.push(0);
        }

        let mut magnitude = String::new();
        for (i, &digit) in digits.iter().rev().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                magnitude.push('_');
            }
            magnitude.push(if digit < 10 {
                (b'0' + digit) as char
            } else {
                (b'a' + digit - 10) as char
            });
        }

        if self.negative && !self.is_zero() {
            Ok(format!("-{magnitude}"))
        } else {
            Ok(magnitude)
        }
    }

    fn div_small(&mut self, divisor: u32) -> u32 {
        let mut remainder = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let value = remainder * Self::BASE + *limb as u64;
            *limb = (value / divisor as u64) as u32;
            remainder = value % divisor as u64;
        }
        self.normalize();
        remainder as u32
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        self.limbs.len().cmp(&rhs.limbs.len()).then_with(|| {
            self.limbs.iter().rev().cmp(rhs.limbs.iter().rev())
        })
    }

    fn add_abs(a: &Self, b: &Self) -> Self {
        let len = a.limbs.len().max(b.limbs.len());
        let mut result = Vec::with_capacity(len + 1);
        let mut carry = 0u64;
        for i in 0..len {
            let sum = a.limbs.get(i).copied().unwrap_or(0) as u64
                + b.limbs.get(i).copied().unwrap_or(0) as u64
                + carry;
            result.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            result.push(carry as u32);
        }
        let mut value = Self { negative: false, limbs: result };
        value.normalize();
        value
    }

    // Requires |a| >= |b|.
    fn sub_abs(a: &Self, b: &Self) -> Self {
        let mut result = Vec::with_capacity(a.limbs.len());
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let mut value = a.limbs[i] as i64
                - b.limbs.get(i).copied().unwrap_or(0) as i64
                - borrow;
            if value < 0 {
                value += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            result.push(value as u32);
        }
        let mut value = Self { negative: false, limbs: result };
        value.normalize();
        value
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => rhs.cmp_abs(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self::add_abs(self, rhs);
            result.negative = self.negative && !result.is_zero();
            result
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self::sub_abs(self, rhs);
                    result.negative = self.negative;
                    result
                }
                Ordering::Less => {
                    let mut result = Self::sub_abs(rhs, self);
                    result.negative = rhs.negative;
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut result = vec![0u64; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let index = i + j;
                let value = result[index] + a as u64 * b as u64 + carry;
                result[index] = value % Self::BASE;
                carry = value / Self::BASE;
            }
            let mut index = i + rhs.limbs.len();
            while carry != 0 {
                if index == result.len() {
                    result.push(0);
                }
                let value = result[index] + carry;
                result[index] = value % Self::BASE;
                carry = value / Self::BASE;
                index += 1;
            }
        }
        let mut value = Self {
            negative: self.negative != rhs.negative,
            limbs: result.into_iter().map(|x| x as u32).collect(),
        };
        value.normalize();
        value
    }

    fn mul_abs_small(&self, multiplier: u32) -> Self {
        let mut result = self.abs();
        result.mul_small_add(multiplier, 0);
        result
    }

    fn div_rem_abs(dividend: &Self, divisor: &Self) -> (Self, Self) {
        if dividend.cmp_abs(divisor) == Ordering::Less {
            return (Self::zero(), dividend.abs());
        }

        let mut quotient_be = Vec::with_capacity(dividend.limbs.len());
        let mut remainder = Self::zero();

        for &limb in dividend.limbs.iter().rev() {
            remainder.limbs.insert(0, limb);
            remainder.normalize();

            let mut low = 0u32;
            let mut high = (Self::BASE - 1) as u32;
            let mut best = 0u32;
            while low <= high {
                let mid = low + (high - low) / 2;
                let product = divisor.mul_abs_small(mid);
                match product.cmp_abs(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 { break; }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        if mid == u32::MAX { break; }
                        low = mid + 1;
                    }
                }
            }

            let product = divisor.mul_abs_small(best);
            remainder = Self::sub_abs(&remainder, &product);
            quotient_be.push(best);
        }

        quotient_be.reverse();
        let mut quotient = Self { negative: false, limbs: quotient_be };
        quotient.normalize();
        remainder.normalize();
        (quotient, remainder)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = Self::div_rem_abs(&self.abs(), &rhs.abs());
        quotient.negative = (self.negative != rhs.negative) && !quotient.is_zero();
        remainder.negative = self.negative && !remainder.is_zero();
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: LIMIT });
        }
        let mut power = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while power != 0 {
            if power & 1 == 1 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let remainder = a.div_rem(&b).unwrap().1.abs();
            a = b;
            b = remainder;
        }
        a
    }
}

    fn from_digits(negative: bool, mut digits: Vec<u8>) -> Self {
        while digits.len() > 1 && digits.last() == Some(&0) {

            Ordering::Equal => {

                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn abs_add_digits(a: &[u8], b: &[u8]) -> Vec<u8> {

            carry = sum / 10;

        }
        out
    }

        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn abs_mul_digits(a: &[u8], b: &[u8]) -> Vec<u8> {
        if (a.len() == 1 && a[0] == 0) || (b.len() == 1 && b[0] == 0) {
            return vec![0];
        }
        let mut accum = vec![0u32; a.len() + b.len()];
        for (i, &x) in a.iter().enumerate() {
            for (j, &y) in b.iter().enumerate() {
                accum[i + j] += (x as u32) * (y as u32);
            }
        }
        for i in 0..accum.len() - 1 {
            let carry = accum[i] / 10;
            accum[i] %= 10;
            accum[i + 1] += carry;
        }
        let mut out: Vec<u8> = accum.into_iter().map(|x| x as u8).collect();
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn abs_mul_small(digits: &[u8], multiplier: u8) -> Vec<u8> {
        if multiplier == 0 {
            return vec![0];
        }
        let mut out = Vec::with_capacity(digits.len() + 1);
        let mut carry = 0u32;
        for &digit in digits {
            let value = digit as u32 * multiplier as u32 + carry;
            out.push((value % 10) as u8);
            carry = value / 10;
        }
        while carry != 0 {
            out.push((carry % 10) as u8);
            carry /= 10;
        }
        out
    }

    fn abs_add_small(digits: &mut Vec<u8>, value: u8) {
        let mut carry = value as u32;
        let mut i = 0;
        while carry != 0 {
            if i == digits.len() {
                digits.push(0);
            }
            let sum = digits[i] as u32 + carry;
            digits[i] = (sum % 10) as u8;
            carry = sum / 10;
            i += 1;
        }
    }

    fn abs_div_rem_digits(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
        let divisor_big = Self::from_digits(false, divisor.to_vec());
        let mut quotient_be = Vec::with_capacity(dividend.len());
        let mut remainder = vec![0];

        for &digit in dividend.iter().rev() {
            remainder.insert(0, digit);
            while remainder.len() > 1 && remainder.last() == Some(&0) {
                remainder.pop();
            }

            let mut qdigit = 0u8;
            while Self::from_digits(false, remainder.clone()).abs_cmp(&divisor_big)
                != Ordering::Less
            {
                remainder = Self::abs_sub_digits(&remainder, divisor);
                qdigit += 1;
            }
            quotient_be.push(qdigit);
        }

        while quotient_be.len() > 1 && quotient_be.first() == Some(&0) {
            quotient_be.remove(0);
        }
        quotient_be.reverse();
        (quotient_be, remainder)
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut digit_values: Vec<Option<u8>> = vec![None; bytes.len()];
        let mut invalid = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                continue;
            }
            let value = match bytes[i] {
                b'0'..=b'9' => Some(bytes[i] - b'0'),
                b'a'..=b'z' => Some(bytes[i] - b'a' + 10),
                b'A'..=b'Z' => Some(bytes[i] - b'A' + 10),
                _ => None,
            };
            match value {
                Some(v) if (v as u32) < radix => digit_values[i] = Some(v),
                _ => {
                    if invalid.is_none() {
                        invalid = Some(i);
                    }
                }
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut separator_error = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let left_valid = i > start && digit_values[i - 1].is_some();
                let right_valid = i + 1 < bytes.len() && digit_values[i + 1].is_some();
                if !(left_valid && right_valid) && separator_error.is_none() {
                    separator_error = Some(i);
                }
            }
        }
        if let Some(index) = separator_error {
            return Err(Error::Separator { index });
        }

        let mut count = 0usize;
        let mut first_digit_index = None;
        let mut first_digit_value = 0u8;
        for i in start..bytes.len() {
            if let Some(value) = digit_values[i] {
                if first_digit_index.is_none() {
                    first_digit_index = Some(i);
                    first_digit_value = value;
                }
                count += 1;
            }
        }

        if count > 1 && first_digit_value == 0 {
            return Err(Error::LeadingZero {
                index: first_digit_index.unwrap(),
            });
        }

        if count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            for i in start..bytes.len() {
                if digit_values[i].is_some() {
                    seen += 1;
                    if seen == excess_number {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut value = vec![0];
        for i in start..bytes.len() {
            if let Some(digit) = digit_values[i] {
                value = Self::abs_mul_small(&value, radix as u8);
                Self::abs_add_small(&mut value, digit);
            }
        }
        Ok(Self::from_digits(negative, value))
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut value = self.digits.clone();
        let mut encoded = Vec::new();
        if self.is_zero() {
            encoded.push(0u8);
        } else {
            while !(value.len() == 1 && value[0] == 0) {
                let mut quotient_be = Vec::with_capacity(value.len());
                let mut remainder = 0u32;
                for &digit in value.iter().rev() {
                    let current = remainder * 10 + digit as u32;
                    quotient_be.push((current / radix) as u8);
                    remainder = current % radix;
                }
                while quotient_be.len() > 1 && quotient_be.first() == Some(&0) {
                    quotient_be.remove(0);
                }
                quotient_be.reverse();
                encoded.push(remainder as u8);
                value = quotient_be;
            }
        }

        while encoded.len() < min_digits {
            encoded.push(0);
        }

        let digit_char = |d: u8| -> char {
            if d < 10 {
                (b'0' + d) as char
            } else {
                (b'a' + (d - 10)) as char
            }
        };

        let mut output = String::new();
        if self.negative && !self.is_zero() {
            output.push('-');
        }
        for i in (0..encoded.len()).rev() {
            output.push(digit_char(encoded[i]));
            if group > 0 && i > 0 && i % group == 0 {
                output.push('_');
            }
        }
        Ok(output)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        match (self.negative, rhs.negative) {
            (false, false) => Self::from_digits(false, Self::abs_add_digits(&self.digits, &rhs.digits)),
            (true, true) => Self::from_digits(true, Self::abs_add_digits(&self.digits, &rhs.digits)),
            _ => match self.abs_cmp(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => Self::from_digits(
                    self.negative,
                    Self::abs_sub_digits(&self.digits, &rhs.digits),
                ),
                Ordering::Less => Self::from_digits(
                    rhs.negative,
                    Self::abs_sub_digits(&rhs.digits, &self.digits),
                ),
            },
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        Self::from_digits(!self.negative, self.digits.clone())
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        Self::from_digits(
            self.negative != rhs.negative,
            Self::abs_mul_digits(&self.digits, &rhs.digits),
        )
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (quotient, remainder) = Self::abs_div_rem_digits(&self.digits, &rhs.digits);
        Ok((
            Self::from_digits(self.negative != rhs.negative, quotient),
            Self::from_digits(self.negative, remainder),
        ))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: LIMIT });
        }

        let mut exponent = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while exponent != 0 {
            if exponent & 1 == 1 {
                result = result.mul(&base);
            }
            exponent >>= 1;
            if exponent != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = Self::from_digits(false, self.digits.clone());
        let mut b = Self::from_digits(false, rhs.digits.clone());
        while !b.is_zero() {
            let remainder = a.div_rem(&b).expect("nonzero divisor").1;
            a = b;
            b = remainder;
        }
        a
    }

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1,000,000,000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const EXPONENT_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: vec![0],
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.limbs.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(&self, rhs: &Self) -> Self {
        let len = self.limbs.len().max(rhs.limbs.len());
        let mut limbs = Vec::with_capacity(len + 1);
        let mut carry = 0u64;
        for i in 0..len {
            let a = self.limbs.get(i).copied().unwrap_or(0) as u64;
            let b = rhs.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = a + b + carry;
            limbs.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            limbs.push(carry as u32);
        }
        let mut result = Self {
            negative: false,
            limbs,
        };
        result.normalize();
        result
    }

    // Assumes |self| >= |rhs|.
    fn sub_abs(&self, rhs: &Self) -> Self {
        let mut limbs = Vec::with_capacity(self.limbs.len());
        let mut borrow = 0i64;
        for i in 0..self.limbs.len() {
            let a = self.limbs[i] as i64 - borrow;
            let b = rhs.limbs.get(i).copied().unwrap_or(0) as i64;
            if a < b {
                limbs.push((a + Self::BASE as i64 - b) as u32);
                borrow = 1;
            } else {
                limbs.push((a - b) as u32);
                borrow = 0;
            }
        }
        let mut result = Self {
            negative: false,
            limbs,
        };
        result.normalize();
        result
    }

    fn mul_small(&mut self, n: u32) {
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let product = *limb as u64 * n as u64 + carry;
            *limb = (product % Self::BASE) as u32;
            carry = product / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
        self.normalize();
    }

    fn add_small(&mut self, n: u32) {
        let mut carry = n as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[i] as u64 + carry;
            self.limbs[i] = (sum % Self::BASE) as u32;
            carry = sum / Self::BASE;
            i += 1;
        }
        self.normalize();
    }

    fn div_small(&mut self, divisor: u32) -> u32 {
        let mut remainder = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let current = remainder * Self::BASE + *limb as u64;
            *limb = (current / divisor as u64) as u32;
            remainder = current % divisor as u64;
        }
        self.normalize();
        remainder as u32
    }

    fn mul_abs(&self, rhs: &Self) -> Self {
        let mut limbs = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let current = limbs[k] as u64 + a as u64 * b as u64 + carry;
                limbs[k] = (current % Self::BASE) as u32;
                carry = current / Self::BASE;
            }
            let mut k = i + rhs.limbs.len();
            while carry != 0 {
                if k == limbs.len() {
                    limbs.push(0);
                }
                let current = limbs[k] as u64 + carry;
                limbs[k] = (current % Self::BASE) as u32;
                carry = current / Self::BASE;
                k += 1;
            }
        }
        let mut result = Self {
            negative: false,
            limbs,
        };
        result.normalize();
        result
    }

    // Shift the magnitude by one base-1e9 limb and add a limb.
    fn shift_base_add(&mut self, limb: u32) {
        if self.is_zero() {
            self.limbs[0] = limb;
        } else {
            self.limbs.insert(0, limb);
        }
        self.normalize();
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let mut start = 0;
        let mut negative = false;
        if bytes[0] == b'+' || bytes[0] == b'-' {
            negative = bytes[0] == b'-';
            start = 1;
            if start == bytes.len() {
                return Err(Error::SignOnly);
            }
        }

        let digit_value = |byte: u8| -> Option<u32> {
            match byte {
                b'0'..=b'9' => Some((byte - b'0') as u32),
                b'a'..=b'z' => Some((byte - b'a') as u32 + 10),
                b'A'..=b'Z' => Some((byte - b'A') as u32 + 10),
                _ => None,
            }
        };

        let mut first_invalid = None;
        for i in start..bytes.len() {
            if bytes[i] != b'_' && digit_value(bytes[i]).is_none_or(|d| d >= radix) {
                first_invalid = Some(i);
                break;
            }
        }
        if let Some(index) = first_invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut first_bad_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let left_ok = i > start && digit_value(bytes[i - 1]).is_some_and(|d| d < radix);
                let right_ok = i + 1 < bytes.len()
                    && digit_value(bytes[i + 1]).is_some_and(|d| d < radix);
                if !left_ok || !right_ok {
                    first_bad_separator = Some(i);
                    break;
                }
            }
        }
        if let Some(index) = first_bad_separator {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit = digit_value(bytes[start]).unwrap();
        if digit_count > 1 && first_digit == 0 {
            return Err(Error::LeadingZero { index: start });
        }
        if digit_count > max_digits {
            let target = max_digits;
            let mut seen = 0;
            for i in start..bytes.len() {
                if bytes[i] != b'_' {
                    if seen == target {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index: i,
                        });
                    }
                    seen += 1;
                }
            }
        }

        let mut value = Self::zero();
        for &byte in &bytes[start..] {
            if byte != b'_' {
                value.mul_small(radix);
                value.add_small(digit_value(byte).unwrap());
            }
        }
        value.negative = negative && !value.is_zero();
        Ok(value)
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut value = Self {
            negative: false,
            limbs: self.limbs.clone(),
        };
        value.normalize();

        let mut digits = Vec::new();
        if value.is_zero() {
            digits.push(0);
        } else {
            while !value.is_zero() {
                digits.push(value.div_small(radix));
            }
            digits.reverse();
        }
        while digits.len() < min_digits {
            digits.insert(0, 0);
        }

        let mut output = String::new();
        if self.negative && !self.is_zero() {
            output.push('-');
        }
        for (i, digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                output.push('_');
            }
            output.push(match *digit {
                0..=9 => (b'0' + *digit as u8) as char,
                _ => (b'a' + (*digit as u8 - 10)) as char,
            });
        }
        Ok(output)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.signum(), rhs.signum()) {
            (-1, 0 | 1) => Ordering::Less,
            (0 | 1, -1) => Ordering::Greater,
            (0, 0) => Ordering::Equal,
            (1, 1) => self.cmp_abs(rhs),
            (-1, -1) => rhs.cmp_abs(self),
            _ => unreachable!(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = self.add_abs(rhs);
            result.negative = self.negative && !result.is_zero();
            result
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = self.sub_abs(rhs);
                    result.negative = self.negative && !result.is_zero();
                    result
                }
                Ordering::Less => {
                    let mut result = rhs.sub_abs(self);
                    result.negative = rhs.negative && !result.is_zero();
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut result = self.mul_abs(rhs);
        result.negative = self.negative != rhs.negative;
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let mut dividend = Self {
            negative: false,
            limbs: self.limbs.clone(),
        };
        let mut divisor = Self {
            negative: false,
            limbs: rhs.limbs.clone(),
        };
        dividend.normalize();
        divisor.normalize();

        if dividend.cmp_abs(&divisor) == Ordering::Less {
            let mut remainder = self.clone();
            remainder.negative = self.negative && !remainder.is_zero();
            return Ok((Self::zero(), remainder));
        }

        let mut quotient_be = Vec::with_capacity(dividend.limbs.len());
        let mut remainder = Self::zero();
        for &limb in dividend.limbs.iter().rev() {
            remainder.shift_base_add(limb);
            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = low + (high - low) / 2;
                let product = divisor.mul_small_value(mid as u32);
                match product.cmp_abs(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 {
                            break;
                        }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        low = mid + 1;
                    }
                }
            }
            let product = divisor.mul_small_value(best as u32);
            remainder = remainder.sub_abs(&product);
            quotient_be.push(best as u32);
        }
        quotient_be.reverse();
        let mut quotient = Self {
            negative: self.negative != rhs.negative,
            limbs: quotient_be,
        };
        quotient.normalize();
        remainder.negative = self.negative && !remainder.is_zero();
        Ok((quotient, remainder))
    }

    fn mul_small_value(&self, n: u32) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result.mul_small(n);
        result
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::EXPONENT_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::EXPONENT_LIMIT,
            });
        }
        let mut power = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while power != 0 {
            if power & 1 == 1 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = Self {
            negative: false,
            limbs: self.limbs.clone(),
        };
        let mut b = Self {
            negative: false,
            limbs: rhs.limbs.clone(),
        };
        a.normalize();
        b.normalize();
        while !b.is_zero() {
            let (_, remainder) = a.div_rem(&b).expect("nonzero divisor");
            a = b;
            b = Self {
                negative: false,
                limbs: remainder.limbs,
            };
        }
        a.negative = false;
        a.normalize();
        a
    }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    digits: Vec<u8>,
}

impl Big {
    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            digits: vec![1],
        }
    }

    fn normalize(&mut self) {
        let first = self.digits.iter().position(|&d| d != 0);
        match first {
            Some(i) if i > 0 => {
                self.digits.drain(..i);
            }
            None => self.digits = vec![0],
            _ => {}
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn cmp_mag(&self, rhs: &Self) -> Ordering {
        self.digits
            .len()
            .cmp(&rhs.digits.len())
            .then_with(|| self.digits.cmp(&rhs.digits))
    }

    fn add_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let (mut i, mut j, mut carry) = (a.len(), b.len(), 0u16);
        while i > 0 || j > 0 || carry != 0 {
            let av = if i > 0 {
                i -= 1;
                a[i] as u16
            } else {
                0
            };
            let bv = if j > 0 {
                j -= 1;
                b[j] as u16
            } else {
                0
            };
            let sum = av + bv + carry;
            out.push((sum % 10) as u8);
            carry = sum / 10;
        }
        out.reverse();
        out
    }

    // Requires a >= b.
    fn sub_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len());
        let (mut i, mut j, mut borrow) = (a.len(), b.len(), 0i16);
        while i > 0 {
            i -= 1;
            let av = a[i] as i16 - borrow;
            let bv = if j > 0 {
                j -= 1;
                b[j] as i16
            } else {
                0
            };
            if av < bv {
                out.push((av + 10 - bv) as u8);
                borrow = 1;
            } else {
                out.push((av - bv) as u8);
                borrow = 0;
            }
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out.reverse();
        out
    }

    fn mul_mag_digit(a: &[u8], digit: u8) -> Vec<u8> {
        if digit == 0 {
            return vec![0];
        }
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u16;
        for &d in a.iter().rev() {
            let product = d as u16 * digit as u16 + carry;
            out.push((product % 10) as u8);
            carry = product / 10;
        }
        while carry > 0 {
            out.push((carry % 10) as u8);
            carry /= 10;
        }
        out.reverse();
        out
    }

    fn mul_mag(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = vec![0u8; a.len() + b.len()];
        for (i, &x) in a.iter().rev().enumerate() {
            let mut carry = 0u16;
            for (j, &y) in b.iter().rev().enumerate() {
                let k = out.len() - 1 - (i + j);
                let v = out[k] as u16 + x as u16 * y as u16 + carry;
                out[k] = (v % 10) as u8;
                carry = v / 10;
            }
            let mut k = out.len() - 1 - (i + b.len());
            while carry > 0 {
                let v = out[k] as u16 + carry;
                out[k] = (v % 10) as u8;
                carry = v / 10;
                if k == 0 && carry > 0 {
                    out.insert(0, 0);
                    k = 0;
                } else if carry > 0 {
                    k -= 1;
                }
            }
        }
        let first = out.iter().position(|&d| d != 0).unwrap_or(out.len() - 1);
        out.drain(..first);
        out
    }

    fn div_rem_mag(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
        let mut quotient = Vec::with_capacity(dividend.len());
        let mut remainder = vec![0u8];
        for &digit in dividend {
            if remainder.len() == 1 && remainder[0] == 0 {
                remainder[0] = digit;
            } else {
                remainder.push(digit);
            }
            let first = remainder.iter().position(|&d| d != 0).unwrap_or(remainder.len() - 1);
            if first > 0 {
                remainder.drain(..first);
            }

            let mut qdigit = 0u8;
            while Self::cmp_digits(&remainder, divisor) != Ordering::Less {
                remainder = Self::sub_mag(&remainder, divisor);
                qdigit += 1;
            }
            quotient.push(qdigit);
        }
        let first = quotient.iter().position(|&d| d != 0).unwrap_or(quotient.len() - 1);
        if first > 0 {
            quotient.drain(..first);
        }
        (quotient, remainder)
    }

    fn cmp_digits(a: &[u8], b: &[u8]) -> Ordering {
        a.len().cmp(&b.len()).then_with(|| a.cmp(b))
    }

    fn from_digits(digits: Vec<u8>, negative: bool) -> Self {
        let mut result = Self { negative, digits };
        result.normalize();
        result
    }

    fn mul_small_add(&mut self, radix: u32, value: u32) {
        let mut carry = value;
        for d in self.digits.iter_mut().rev() {
            let n = *d as u32 * radix + carry;
            *d = (n % 10) as u8;
            carry = n / 10;
        }
        while carry > 0 {
            self.digits.insert(0, (carry % 10) as u8);
            carry /= 10;
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            let value = match byte {
                b'0'..=b'9' => (byte - b'0') as u32,
                b'a'..=b'z' => (byte - b'a' + 10) as u32,
                b'A'..=b'Z' => (byte - b'A' + 10) as u32,
                _ => {
                    invalid.get_or_insert(index);
                    continue;
                }
            };
            if value >= radix {
                invalid.get_or_insert(index);
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut malformed_separator = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_' {
                let before_valid = index > start && bytes[index - 1] != b'_';
                let after_valid = index + 1 < bytes.len() && bytes[index + 1] != b'_';
                if !before_valid || !after_valid {
                    malformed_separator.get_or_insert(index);
                }
            }
        }
        if let Some(index) = malformed_separator {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_number = max_digits;
            let index = (start..bytes.len())
                .filter(|&i| bytes[i] != b'_')
                .nth(excess_number)
                .unwrap();
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let mut value = Self::zero();
        for &byte in &bytes[start..] {
            if byte == b'_' {
                continue;
            }
            let digit = match byte {
                b'0'..=b'9' => byte - b'0',
                b'a'..=b'z' => byte - b'a' + 10,
                b'A'..=b'Z' => byte - b'A' + 10,
                _ => unreachable!(),
            };
            value.mul_small_add(radix, digit as u32);
        }
        value.negative = negative && !value.is_zero();
        Ok(value)
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut value = self.digits.clone();
        let mut out = Vec::new();
        loop {
            let mut quotient = Vec::with_capacity(value.len());
            let mut rem = 0u32;
            for &digit in &value {
                let n = rem * 10 + digit as u32;
                let q = n / radix;
                rem = n % radix;
                if !quotient.is_empty() || q != 0 {
                    quotient.push(q as u8);
                }
            }
            out.push(rem as u8);
            if quotient.is_empty() {
                break;
            }
            value = quotient;
        }
        out.reverse();

        while out.len() < min_digits.max(1) {
            out.insert(0, 0);
        }

        let mut magnitude = String::new();
        for (i, &digit) in out.iter().enumerate() {
            if group > 0 && i > 0 && (out.len() - i) % group == 0 {
                magnitude.push('_');
            }
            magnitude.push(if digit < 10 {
                (b'0' + digit) as char
            } else {
                (b'a' + digit - 10) as char
            });
        }
        if self.negative {
            magnitude.insert(0, '-');
        }
        Ok(magnitude)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_mag(rhs),
            (true, true) => rhs.cmp_mag(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::from_digits(Self::add_mag(&self.digits, &rhs.digits), self.negative)
        } else {
            match self.cmp_mag(rhs) {
                Ordering::Greater => {
                    Self::from_digits(Self::sub_mag(&self.digits, &rhs.digits), self.negative)
                }
                Ordering::Less => {
                    Self::from_digits(Self::sub_mag(&rhs.digits, &self.digits), rhs.negative)
                }
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        Self::from_digits(
            Self::mul_mag(&self.digits, &rhs.digits),
            self.negative ^ rhs.negative,
        )
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (q, r) = Self::div_rem_mag(&self.digits, &rhs.digits);
        Ok((
            Self::from_digits(q, self.negative ^ rhs.negative),
            Self::from_digits(r, self.negative),
        ))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: LIMIT });
        }
        let mut n = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while n > 0 {
            if n & 1 == 1 {
                result = result.mul(&base);
            }
            n >>= 1;
            if n > 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = Self::from_digits(self.digits.clone(), false);
        let mut b = Self::from_digits(rhs.digits.clone(), false);
        while !b.is_zero() {
            let (_, r) = a.div_rem(&b).expect("nonzero divisor");
            a = b;
            b = r;
        }
        a
    }
}
