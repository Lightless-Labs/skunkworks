#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_scalar(word: &mut Word, c: char, wildcard_active: bool) {
    if wildcard_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_str(word: &mut Word, s: &str, wildcard_active: bool) {
    for c in s.chars() {
        append_scalar(word, c, wildcard_active);
    }
}

fn first_env<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_scalar(&mut word, c, false);
                }
                i = next_i;
            }
            Quote::Double => {
                match c {
                    '"' => {
                        quote = Quote::None;
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let n = input[next_i..].chars().next().unwrap();
                        let after_n = next_i + n.len_utf8();
                        if matches!(n, '"' | '\\' | '$') {
                            append_scalar(&mut word, n, false);
                            i = after_n;
                        } else if n == '\n' {
                            i = after_n;
                        } else {
                            append_scalar(&mut word, '\\', false);
                            i = next_i;
                        }
                    }
                    '`' => return Err(SplitError::CommandSubstitution { pos: i }),
                    '$' => {
                        if input[next_i..].starts_with('(') {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                        if input[next_i..].starts_with('{') {
                            let dollar = i;
                            let content_start = next_i + 1;
                            let mut j = content_start;
                            let mut close = None;
                            while j < input.len() {
                                let sc = input[j..].chars().next().unwrap();
                                let sj = j + sc.len_utf8();
                                if sc == '\\' {
                                    if sj < input.len() {
                                        let escaped = input[sj..].chars().next().unwrap();
                                        j = sj + escaped.len_utf8();
                                    } else {
                                        j = sj;
                                    }
                                } else if sc == '}' {
                                    close = Some(j);
                                    break;
                                } else {
                                    j = sj;
                                }
                            }
                            let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                            let contents = &input[content_start..close];
                            let (name, default) = parse_braced(contents)
                                .ok_or(SplitError::InvalidExpansion { pos: dollar })?;
                            let value = match first_env(env, name) {
                                Some(v) if default.is_none() || !v.is_empty() => v.to_string(),
                                _ => default.map(decode_default).unwrap_or_default(),
                            };
                            active = true;
                            append_str(&mut word, &value, false);
                            i = close + 1;
                        } else if next_i < input.len() {
                            let n = input[next_i..].chars().next().unwrap();
                            if is_name_start(n) {
                                let mut end = next_i + n.len_utf8();
                                while end < input.len() {
                                    let x = input[end..].chars().next().unwrap();
                                    if is_name_continue(x) {
                                        end += x.len_utf8();
                                    } else {
                                        break;
                                    }
                                }
                                let name = &input[next_i..end];
                                active = true;
                                if let Some(value) = first_env(env, name) {
                                    append_str(&mut word, value, false);
                                }
                                i = end;
                            } else {
                                append_scalar(&mut word, '$', false);
                                i = next_i;
                            }
                        } else {
                            append_scalar(&mut word, '$', false);
                            i = next_i;
                        }
                    }
                    _ => {
                        append_scalar(&mut word, c, false);
                        i = next_i;
                    }
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else {
                    match c {
                        '\'' => {
                            active = true;
                            quote = Quote::Single;
                            quote_pos = i;
                            i = next_i;
                        }
                        '"' => {
                            active = true;
                            quote = Quote::Double;
                            quote_pos = i;
                            i = next_i;
                        }
                        '\\' => {
                            if next_i == input.len() {
                                return Err(SplitError::TrailingEscape { pos: i });
                            }
                            let n = input[next_i..].chars().next().unwrap();
                            let after_n = next_i + n.len_utf8();
                            if n == '\n' {
                                i = after_n;
                            } else {
                                active = true;
                                append_scalar(&mut word, n, false);
                                i = after_n;
                            }
                        }
                        '`' => return Err(SplitError::CommandSubstitution { pos: i }),
                        '$' => {
                            if input[next_i..].starts_with('(') {
                                return Err(SplitError::CommandSubstitution { pos: i });
                            }
                            if input[next_i..].starts_with('{') {
                                let dollar = i;
                                let content_start = next_i + 1;
                                let mut j = content_start;
                                let mut close = None;
                                while j < input.len() {
                                    let sc = input[j..].chars().next().unwrap();
                                    let sj = j + sc.len_utf8();
                                    if sc == '\\' {
                                        if sj < input.len() {
                                            let escaped = input[sj..].chars().next().unwrap();
                                            j = sj + escaped.len_utf8();
                                        } else {
                                            j = sj;
                                        }
                                    } else if sc == '}' {
                                        close = Some(j);
                                        break;
                                    } else {
                                        j = sj;
                                    }
                                }
                                let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                                let contents = &input[content_start..close];
                                let (name, default) = parse_braced(contents)
                                    .ok_or(SplitError::InvalidExpansion { pos: dollar })?;
                                let value = match first_env(env, name) {
                                    Some(v) if default.is_none() || !v.is_empty() => v.to_string(),
                                    _ => default.map(decode_default).unwrap_or_default(),
                                };
                                active = true;
                                append_str(&mut word, &value, true);
                                i = close + 1;
                            } else if next_i < input.len() {
                                let n = input[next_i..].chars().next().unwrap();
                                if is_name_start(n) {
                                    let mut end = next_i + n.len_utf8();
                                    while end < input.len() {
                                        let x = input[end..].chars().next().unwrap();
                                        if is_name_continue(x) {
                                            end += x.len_utf8();
                                        } else {
                                            break;
                                        }
                                    }
                                    let name = &input[next_i..end];
                                    active = true;
                                    if let Some(value) = first_env(env, name) {
                                        append_str(&mut word, value, true);
                                    }
                                    i = end;
                                } else {
                                    active = true;
                                    append_scalar(&mut word, '$', true);
                                    i = next_i;
                                }
                            } else {
                                active = true;
                                append_scalar(&mut word, '$', true);
                                i = next_i;
                            }
                        }
                        _ => {
                            active = true;
                            append_scalar(&mut word, c, true);
                            i = next_i;
                        }
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_braced(contents: &str) -> Option<(&str, Option<&str>)> {
    let mut end = 0usize;
    let mut chars = contents.char_indices();
    let (_, first) = chars.next()?;
    if !is_name_start(first) {
        return None;
    }
    end = first.len_utf8();
    for (idx, c) in chars {
        if is_name_continue(c) {
            end = idx + c.len_utf8();
        } else {
            break;
        }
    }
    let name = &contents[..end];
    let rest = &contents[end..];
    if rest.is_empty() {
        Some((name, None))
    } else if let Some(default) = rest.strip_prefix(":-") {
        Some((name, Some(default)))
    } else {
        None
    }
}
