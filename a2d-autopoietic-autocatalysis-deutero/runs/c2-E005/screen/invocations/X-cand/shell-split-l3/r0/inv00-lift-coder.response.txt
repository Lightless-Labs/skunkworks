#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

struct Builder {
    text: String,
    glob: Vec<usize>,
    active: bool,
}

impl Builder {
    fn new() -> Self {
        Self {
            text: String::new(),
            glob: Vec::new(),
            active: false,
        }
    }

    fn append(&mut self, ch: char, wildcard_active: bool) {
        if wildcard_active && matches!(ch, '*' | '?' | '[') {
            self.glob.push(self.text.len());
        }
        self.text.push(ch);
    }

    fn append_str(&mut self, s: &str, wildcard_active: bool) {
        for ch in s.chars() {
            self.append(ch, wildcard_active);
        }
    }

    fn finish(&mut self, out: &mut Vec<Word>) {
        if self.active {
            out.push(Word {
                text: std::mem::take(&mut self.text),
                glob: std::mem::take(&mut self.glob),
            });
            self.active = false;
        }
    }
}

fn is_name_start(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphabetic()
}

fn is_name_continue(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphanumeric()
}

fn decode_default(text: &str) -> String {
    let mut decoded = String::new();
    let mut iter = text.char_indices().peekable();

    while let Some((_, ch)) = iter.next() {
        if ch == '\\' {
            if let Some((_, next)) = iter.next() {
                decoded.push(next);
            } else {
                decoded.push('\\');
            }
        } else {
            decoded.push(ch);
        }
    }

    decoded
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let mut cursor = dollar + 2;
    let mut close = None;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        if ch == '\\' {
            cursor += ch.len_utf8();
            if cursor < input.len() {
                let next = input[cursor..].chars().next().unwrap();
                cursor += next.len_utf8();
            }
        } else if ch == '}' {
            close = Some(cursor);
            break;
        } else {
            cursor += ch.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[dollar + 2..close];

    let name_end = contents
        .char_indices()
        .find_map(|(i, ch)| {
            if (i == 0 && is_name_start(ch)) || (i > 0 && is_name_continue(ch)) {
                None
            } else {
                Some(i)
            }
        })
        .unwrap_or(contents.len());

    let name = &contents[..name_end];
    if name.is_empty()
        || !name.chars().next().is_some_and(is_name_start)
        || !name.chars().all(is_name_continue)
    {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let rest = &contents[name_end..];
    if rest.is_empty() {
        let value = lookup(name, env).unwrap_or("");
        return Ok((close + 1, value.to_string()));
    }

    if let Some(default) = rest.strip_prefix(":-") {
        let value = lookup(name, env).filter(|value| !value.is_empty());
        let selected = value
            .map(str::to_string)
            .unwrap_or_else(|| decode_default(default));
        return Ok((close + 1, selected));
    }

    Err(SplitError::InvalidExpansion { pos: dollar })
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut builder = Builder::new();
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        let next_pos = cursor + ch.len_utf8();

        match quote {
            Quote::Single(_) => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    builder.append(ch, false);
                }
                cursor = next_pos;
            }
            Quote::Double(open_pos) => {
                if ch == '"' {
                    quote = Quote::None;
                    cursor = next_pos;
                } else if ch == '\\' {
                    let slash_pos = cursor;
                    if next_pos == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                    let next = input[next_pos..].chars().next().unwrap();
                    let after_next = next_pos + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next == '$' && input[after_next..].starts_with('(') {
                            // Escaped dollars are literal.
                        }
                        if next != '\n' {
                            builder.active = true;
                            builder.append(next, false);
                        }
                        cursor = after_next;
                    } else {
                        builder.active = true;
                        builder.append('\\', false);
                        cursor = next_pos;
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if ch == '$' {
                    if input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[next_pos..].starts_with('{') {
                        let (end, value) = parse_braced(input, cursor, env)?;
                        builder.active = true;
                        builder.append_str(&value, false);
                        cursor = end;
                    } else if let Some(name_start) = input[next_pos..].chars().next() {
                        if is_name_start(name_start) {
                            let mut end = next_pos;
                            while end < input.len() {
                                let c = input[end..].chars().next().unwrap();
                                if is_name_continue(c) {
                                    end += c.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let value = lookup(&input[next_pos..end], env).unwrap_or("");
                            builder.active = true;
                            builder.append_str(value, false);
                            cursor = end;
                        } else {
                            builder.active = true;
                            builder.append('$', false);
                            cursor = next_pos;
                        }
                    } else {
                        builder.active = true;
                        builder.append('$', false);
                        cursor = next_pos;
                    }
                } else {
                    builder.active = true;
                    builder.append(ch, false);
                    cursor = next_pos;
                }
                let _ = open_pos;
            }
            Quote::None => {
                if matches!(ch, ' ' | '\t' | '\r' | '\n') {
                    builder.finish(&mut words);
                    cursor = next_pos;
                } else if ch == '\'' {
                    builder.active = true;
                    quote = Quote::Single(cursor);
                    cursor = next_pos;
                } else if ch == '"' {
                    builder.active = true;
                    quote = Quote::Double(cursor);
                    cursor = next_pos;
                } else if ch == '\\' {
                    let slash_pos = cursor;
                    if next_pos == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                    let next = input[next_pos..].chars().next().unwrap();
                    let after_next = next_pos + next.len_utf8();
                    if next == '\n' {
                        cursor = after_next;
                    } else {
                        builder.active = true;
                        builder.append(next, false);
                        cursor = after_next;
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if ch == '$' {
                    if input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[next_pos..].starts_with('{') {
                        let (end, value) = parse_braced(input, cursor, env)?;
                        builder.active = true;
                        builder.append_str(&value, true);
                        cursor = end;
                    } else if let Some(name_start) = input[next_pos..].chars().next() {
                        if is_name_start(name_start) {
                            let mut end = next_pos;
                            while end < input.len() {
                                let c = input[end..].chars().next().unwrap();
                                if is_name_continue(c) {
                                    end += c.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let value = lookup(&input[next_pos..end], env).unwrap_or("");
                            builder.active = true;
                            builder.append_str(value, true);
                            cursor = end;
                        } else {
                            builder.active = true;
                            builder.append('$', true);
                            cursor = next_pos;
                        }
                    } else {
                        builder.active = true;
                        builder.append('$', true);
                        cursor = next_pos;
                    }
                } else {
                    builder.active = true;
                    builder.append(ch, true);
                    cursor = next_pos;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    builder.finish(&mut words);
    Ok(words)
}
