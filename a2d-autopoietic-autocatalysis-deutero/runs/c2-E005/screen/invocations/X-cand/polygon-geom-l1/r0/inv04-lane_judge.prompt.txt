pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    if points.len() > 4096 {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, point) in points.iter().enumerate() {
        if !(-1_000_000_000..=1_000_000_000).contains(&point.x)
            || !(-1_000_000_000..=1_000_000_000).contains(&point.y)
        {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = points.len()
        - usize::from(points.len() >= 2 && points.last() == points.first());
    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }
    let polygon = &points[..effective_len];

    let mut signed_area2 = 0_i128;
    for i in 0..effective_len {
        let a = polygon[i];
        let b = polygon[(i + 1) % effective_len];
        signed_area2 += i128::from(a.x) * i128::from(b.y)
            - i128::from(a.y) * i128::from(b.x);
    }

    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };
    let area2 = signed_area2.unsigned_abs();

    let mut edge_milli = Vec::with_capacity(effective_len);
    let mut perimeter_milli = 0_u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..effective_len {
        let a = polygon[i];
        let b = polygon[(i + 1) % effective_len];

        let dx = i128::from(a.x) - i128::from(b.x);
        let dy = i128::from(a.y) - i128::from(b.y);

        let mut root = integer_sqrt(squared_scaled);
        let lower_square = root * root;
        let upper = root + 1;
        let upper_square = upper * upper;
        if squared_scaled - lower_square >= upper_square - squared_scaled {
            root += 1;
        }

        let rounded = root as u64;
        edge_milli.push(rounded);
        perimeter_milli += rounded;

        let replace = match longest_edge {
            None => true,
            Some(current) => {
                if rounded != edge_milli[current] {
                    rounded > edge_milli[current]
                } else {
                    let pair = ordered_endpoints(a, b);
                    let current_pair = ordered_endpoints(
                        polygon[current],
                        polygon[(current + 1) % effective_len],
                    );
                    pair < current_pair || (pair == current_pair && i < current)
                }
            }
        };
        if replace {
            longest_edge = Some(i);
        }
    }

    Ok(Measures {
        signed_area2,
        area2,
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn ordered_endpoints(a: Point, b: Point) -> (Point, Point) {

fn integer_sqrt(n: u128) -> u128 {
    if n < 2 {
        return n;
    }

    let mut x = n;
    let mut y = (x + 1) / 2;
    while y < x {
        x = y;
        y = (x + n / x) / 2;
    }
    x

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    const MAX_VERTICES: usize = 4096;
    const COORD_LIMIT: i64 = 1_000_000_000;

    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    let mut signed_area2 = 0_i128;
    let mut edge_milli = Vec::with_capacity(effective_len);

    for i in 0..effective_len {
        let a = polygon[i];
        let b = polygon[(i + 1) % effective_len];

        signed_area2 += (a.x as i128) * (b.y as i128) - (a.y as i128) * (b.x as i128);

        let dx = (b.x as i128) - (a.x as i128);
        let dy = (b.y as i128) - (a.y as i128);
        let d2 = (dx * dx + dy * dy) as u128;

        let rounded_length = rounded_sqrt(scaled_squared);

        edge_milli.push(rounded_length);
        perimeter_milli += rounded_length;

            Some(current) => {

                    true
                } else if rounded_length < current_length {
                    false
                } else {
                    let current_a = polygon[current];
                    let current_b = polygon[(current + 1) % effective_len];

                    let candidate_pair = ordered_pair(a, b);
                    candidate_pair < current_pair

                }
            }

            longest_edge = Some(i);

fn rounded_sqrt(value: u128) -> u64 {
    let floor = integer_sqrt(value);
    let lower_square = floor * floor;
    let upper = floor + 1;
    let upper_square = upper * upper;

    let rounded = if value - lower_square >= upper_square - value {
        upper
    } else {
        floor
    };

    rounded as u64
}

fn integer_sqrt(value: u128) -> u128 {
    if value < 2 {
        return value;
    }

    let mut low = 1_u128;
    let mut high = value.min(u64::MAX as u128 + 1);

    while low < high {
        let mid = low + (high - low) / 2;
        if mid <= value / mid {
            low = mid + 1;
        } else {
            high = mid;
        }
    }
    low - 1
}

pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = (a.x as i128 - b.x as i128).unsigned_abs();
    let dy = (a.y as i128 - b.y as i128).unsigned_abs();
    let scaled_d2 = (dx * dx + dy * dy) * 1_000_000;

    // Find floor(sqrt(scaled_d2)) using integer arithmetic.
    let mut lo = 0u64;
    let mut hi = 3_000_000_001u64;
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        let square = (mid as u128) * (mid as u128);
        if square <= scaled_d2 {
            lo = mid;
        } else {
            hi = mid;
        }
    }

    let floor = lo as u128;
    let remainder = scaled_d2 - floor * floor;
    if remainder >= floor * 2 + 1 {
        lo + 1
    } else {
        lo
    }
}

    let raw_len = points.len();

        return Err(GeomError::TooManyVertices { len: raw_len });
    }

    for (index, point) in points.iter().enumerate() {

            || point.x > COORD_LIMIT
            || point.y < -COORD_LIMIT
            || point.y > COORD_LIMIT

            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if raw_len >= 2 && points[raw_len - 1] == points[0] {
        raw_len - 1
    } else {
        raw_len
    };

    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });

    let mut signed_area2 = 0i128;

        let a = polygon[i];
        let b = polygon[(i + 1) % effective_len];
        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
        edge_milli.push(rounded_edge_milli(a, b));

    let mut longest_edge = None;

            None => true,
            Some(best) if length > edge_milli[best] => true,
