#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }

    path.split('/')
        .filter(|component| !component.is_empty())
        .last()
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/')
        .filter(|component| !component.is_empty())
        .collect()
}

fn normalize_components<'a>(
    parts: impl IntoIterator<Item = (&'a str, usize)>,
    absolute: bool,
) -> Result<Vec<&'a str>, PathError> {
    let mut retained = Vec::new();

    for (component, at) in parts {
        match component {
            "." => {}
            ".." => {
                if retained.last().is_some_and(|last| *last != "..") {
                    retained.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    retained.push(component);
                }
            }
            _ => retained.push(component),
        }
    }

    Ok(retained)
}

fn render(parts: &[&str], absolute: bool, wants_directory: bool) -> String {
    let mut result = if absolute {
        format!("/{}", parts.join("/"))
    } else if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    };

    if wants_directory && result != "/" && result != "." {
        result.push('/');
    }

    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let wants_directory = directory_intent(path);
    let parts = path
        .match_indices('/')
        .map(|_| ())
        .count();

    let mut indexed = Vec::new();
    let mut offset = 0;
    for component in path.split('/') {
        if !component.is_empty() {
            indexed.push((component, offset));
        }
        offset += component.len() + 1;
    }
    let _ = parts;

    let reduced = normalize_components(indexed, absolute)?;
    Ok(render(&reduced, absolute, wants_directory))
}

fn validated_absolute_context(path: &str) -> Result<Vec<String>, PathError> {
    if path.as_bytes().first() != Some(&b'/') {
        return Err(PathError::InvalidCwd);
    }

    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(components(&normalized)
        .into_iter()
        .map(str::to_string)
        .collect())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validated_absolute_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(home) => Some(
            validated_absolute_context(home).map_err(|_| PathError::InvalidHome)?,
        ),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde_expanded = if bytes.first() == Some(&b'~') {
        match bytes.get(1) {
            None | Some(b'/') => true,
            Some(_) => return Err(PathError::UnsupportedTilde { at: 1 }),
        }
    } else {
        false
    };

    let (mut start, remainder, offset_base, absolute) = if tilde_expanded {
        let start = home_parts.ok_or(PathError::MissingHome)?;
        let skip = if bytes.get(1) == Some(&b'/') { 2 } else { 1 };
        (start, &path[skip..], skip, true)
    } else if bytes.first() == Some(&b'/') {
        (Vec::new(), path, 0, true)
    } else {
        (cwd_parts, path, 0, true)
    };

    let wants_directory = directory_intent(path);
    let mut indexed = Vec::new();
    let mut offset = offset_base;
    for component in remainder.split('/') {
        if !component.is_empty() {
            indexed.push((component, offset));
        }
        offset += component.len() + 1;
    }

    let reduced = normalize_components(indexed, true)?;
    start.extend(reduced);
    Ok(render(&start, absolute, wants_directory))
}
