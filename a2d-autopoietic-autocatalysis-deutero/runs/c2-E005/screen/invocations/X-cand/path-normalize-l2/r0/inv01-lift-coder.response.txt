#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .next_back()
            .is_some_and(|component| component == "." || component == "..")
}

fn reduce_components<'a, I>(
    components: I,
    absolute: bool,
    offset: usize,
    output: &mut Vec<String>,
) -> Result<(), PathError>
where
    I: IntoIterator<Item = (usize, &'a str)>,
{
    for (at, component) in components {
        match component {
            "" | "." => {}
            ".." => {
                if output.last().is_some_and(|last| last != "..") {
                    output.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: offset + at });
                } else {
                    output.push("..".to_owned());
                }
            }
            name => output.push(name.to_owned()),
        }
    }
    Ok(())
}

fn join_path(absolute: bool, components: &[String], directory: bool) -> String {
    let mut result = if absolute {
        format!("/{}", components.join("/"))
    } else if components.is_empty() {
        ".".to_owned()
    } else {
        components.join("/")
    };

    if absolute && components.is_empty() {
        return result;
    }

    if !absolute && components.is_empty() {
        return result;
    }

    if directory {
        result.push('/');
    }

    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.starts_with('/');
    let directory = directory_intent(path);
    let components = path
        .split('/')
        .scan(0usize, |offset, component| {
            let current = *offset;
            *offset += component.len() + 1;
            Some((current, component))
        });

    let mut output = Vec::new();
    reduce_components(components, absolute, 0, &mut output)?;
    Ok(join_path(absolute, &output, directory))
}

fn normalized_absolute_components(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::Escape { at: 0 });
    }

    let normalized = normalize(path)?;
    Ok(normalized
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let mut output = normalized_absolute_components(cwd).map_err(|_| PathError::InvalidCwd)?;

    let normalized_home = match home {
        Some(home) => Some(
            normalized_absolute_components(home).map_err(|_| PathError::InvalidHome)?,
        ),
        None => None,
    };

    validate_raw(path)?;

    let directory = directory_intent(path);

    let (absolute, component_start, base) = if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let home_components = normalized_home.ok_or(PathError::MissingHome)?;
            output = home_components;
            (true, if bytes.len() > 1 { 2 } else { 1 }, true)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        output.clear();
        (true, 0, false)
    } else {
        (true, 0, true)
    };

    let remainder = &path[component_start..];
    let components = remainder
        .split('/')
        .scan(component_start, |offset, component| {
            let current = *offset;
            *offset += component.len() + 1;
            Some((current, component))
        });

    reduce_components(components, absolute, 0, &mut output)?;

    let _ = base;
    Ok(join_path(true, &output, directory))
}
