#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single { start: usize },
    Double { start: usize },
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            }
        } else {
            result.push(c);
        }
    }

    result
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let after_open = dollar + 2;
    let mut closing = None;
    let mut cursor = after_open;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += c.len_utf8();
            if cursor < input.len() {
                let next = input[cursor..].chars().next().unwrap();
                cursor += next.len_utf8();
            }
        } else if c == '}' {
            closing = Some(cursor);
            break;
        } else {
            cursor += c.len_utf8();
        }
    }

    let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[after_open..close];

    let name_end = contents
        .char_indices()
        .find_map(|(i, c)| (!is_name_continue(c)).then_some(i))
        .unwrap_or(contents.len());

    let name = &contents[..name_end];
    if name.is_empty() || !name.chars().next().map(is_name_start).unwrap_or(false) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let remainder = &contents[name_end..];
    let value = if remainder.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default) = remainder.strip_prefix(":-") {
        match lookup(name, env) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let width = c.len_utf8();

        match quote {
            Quote::Single { .. } => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                cursor += width;
            }
            Quote::Double { .. } => {
                if c == '"' {
                    quote = Quote::None;
                    cursor += width;
                } else if c == '\\' {
                    let escape_pos = cursor;
                    cursor += width;
                    if cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    let next = input[cursor..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$') {
                        word.push(next);
                        cursor += next.len_utf8();
                    } else if next == '\n' {
                        cursor += next.len_utf8();
                    } else {
                        word.push('\\');
                    }
                } else if c == '$' {
                    let next_pos = cursor + width;
                    if next_pos < input.len() {
                        let next = input[next_pos..].chars().next().unwrap();
                        if next == '{' {
                            let (value, end) = parse_expansion(input, cursor, env)?;
                            word.push_str(&value);
                            active = true;
                            cursor = end;
                            continue;
                        } else if is_name_start(next) {
                            let mut end = next_pos + next.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let value = lookup(&input[next_pos..end], env).unwrap_or("");
                            word.push_str(value);
                            active = true;
                            cursor = end;
                            continue;
                        }
                    }
                    word.push('$');
                    cursor += width;
                } else {
                    word.push(c);
                    cursor += width;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    cursor += width;
                } else if c == '\'' {
                    quote = Quote::Single { start: cursor };
                    active = true;
                    cursor += width;
                } else if c == '"' {
                    quote = Quote::Double { start: cursor };
                    active = true;
                    cursor += width;
                } else if c == '\\' {
                    let escape_pos = cursor;
                    cursor += width;
                    if cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    let next = input[cursor..].chars().next().unwrap();
                    if next != '\n' {
                        word.push(next);
                        active = true;
                    }
                    cursor += next.len_utf8();
                } else if c == '$' {
                    let next_pos = cursor + width;
                    if next_pos < input.len() {
                        let next = input[next_pos..].chars().next().unwrap();
                        if next == '{' {
                            let (value, end) = parse_expansion(input, cursor, env)?;
                            word.push_str(&value);
                            active = true;
                            cursor = end;
                            continue;
                        } else if is_name_start(next) {
                            let mut end = next_pos + next.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let value = lookup(&input[next_pos..end], env).unwrap_or("");
                            word.push_str(value);
                            active = true;
                            cursor = end;
                            continue;
                        }
                    }
                    word.push('$');
                    active = true;
                    cursor += width;
                } else {
                    word.push(c);
                    active = true;
                    cursor += width;
                }
            }
        }
    }

    match quote {
        Quote::Single { start } => {
            return Err(SplitError::UnterminatedSingle { pos: start });
        }
        Quote::Double { start } => {
            return Err(SplitError::UnterminatedDouble { pos: start });
        }
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
