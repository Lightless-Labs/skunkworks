#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;
    let bytes = input.as_bytes();

    while i < input.len() {
        match quote {
            Quote::Single(_) => {
                let ch = input[i..].chars().next().unwrap();
                if ch == '\'' {
                    quote = Quote::None;
                    i += ch.len_utf8();
                } else {
                    word.push(ch);
                    i += ch.len_utf8();
                }
            }
            Quote::Double(_) => {
                let ch = input[i..].chars().next().unwrap();
                match ch {
                    '"' => {
                        quote = Quote::None;
                        i += 1;
                    }
                    '\\' => {
                        let pos = i;
                        i += 1;
                        if i == input.len() {
                            return Err(SplitError::TrailingEscape { pos });
                        }
                        let next = input[i..].chars().next().unwrap();
                        match next {
                            '"' | '\\' | '$' => {
                                word.push(next);
                                i += next.len_utf8();
                            }
                            '\n' => i += 1,
                            _ => word.push('\\'),
                        }
                    }
                    '$' => {
                        active = true;
                        let (next_i, value) = expand(input, i, env)?;
                        word.push_str(&value);
                        i = next_i;
                    }
                    _ => {
                        word.push(ch);
                        i += ch.len_utf8();
                    }
                }
            }
            Quote::None => {
                let ch = input[i..].chars().next().unwrap();
                match ch {
                    ' ' | '\t' | '\r' | '\n' => {
                        if active {
                            words.push(std::mem::take(&mut word));
                            active = false;
                        }
                        i += 1;
                    }
                    '\'' => {
                        active = true;
                        quote = Quote::Single(i);
                        i += 1;
                    }
                    '"' => {
                        active = true;
                        quote = Quote::Double(i);
                        i += 1;
                    }
                    '\\' => {
                        let pos = i;
                        i += 1;
                        if i == input.len() {
                            return Err(SplitError::TrailingEscape { pos });
                        }
                        let next = input[i..].chars().next().unwrap();
                        if next != '\n' {
                            active = true;
                            word.push(next);
                        }
                        i += next.len_utf8();
                    }
                    '$' => {
                        active = true;
                        let (next_i, value) = expand(input, i, env)?;
                        word.push_str(&value);
                        i = next_i;
                    }
                    _ => {
                        active = true;
                        word.push(ch);
                        i += ch.len_utf8();
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    let _ = bytes;
    Ok(words)
}

fn is_name_start(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphabetic()
}

fn is_name_char(ch: char) -> bool {
    is_name_start(ch) || ch.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> &'a str {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
        .unwrap_or("")
}

fn expand(input: &str, dollar: usize, env: &[(&str, &str)]) -> Result<(usize, String), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((after_dollar, "$".to_string()));
    }

    let next = input[after_dollar..].chars().next().unwrap();
    if next == '{' {
        return expand_braced(input, dollar, env);
    }
    if !is_name_start(next) {
        return Ok((after_dollar, "$".to_string()));
    }

    let mut end = after_dollar;
    for ch in input[after_dollar..].chars() {
        if !is_name_char(ch) {
            break;
        }
        end += ch.len_utf8();
    }
    let name = &input[after_dollar..end];
    Ok((end, lookup(name, env).to_string()))
}

fn expand_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let content_start = dollar + 2;
    let mut i = content_start;
    let mut close = None;

    while i < input.len() {
        let ch = input[i..].chars().next().unwrap();
        if ch == '\\' {
            i += 1;
            if i < input.len() {
                let escaped = input[i..].chars().next().unwrap();
                i += escaped.len_utf8();
            }
        } else if ch == '}' {
            close = Some(i);
            break;
        } else {
            i += ch.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[content_start..close];

    let mut name_end = 0;
    for ch in contents.chars() {
        if name_end == 0 {
            if !is_name_start(ch) {
                break;
            }
        } else if !is_name_char(ch) {
            break;
        }
        name_end += ch.len_utf8();
    }

    if name_end == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_end];
    let rest = &contents[name_end..];

    let value = if rest.is_empty() {
        lookup(name, env).to_string()
    } else if let Some(default) = rest.strip_prefix(":-") {
        let env_value = lookup(name, env);
        if env_value.is_empty() {
            decode_default(default)
        } else {
            env_value.to_string()
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close + 1, value))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut i = 0;

    while i < text.len() {
        let ch = text[i..].chars().next().unwrap();
        if ch == '\\' {
            i += 1;
            if i < text.len() {
                let escaped = text[i..].chars().next().unwrap();
                result.push(escaped);
                i += escaped.len_utf8();
            }
        } else {
            result.push(ch);
            i += ch.len_utf8();
        }
    }

    result
}
