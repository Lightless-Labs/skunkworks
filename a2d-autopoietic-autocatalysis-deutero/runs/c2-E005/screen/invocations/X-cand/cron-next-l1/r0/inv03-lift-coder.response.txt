#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<u8>; 5],
    day_of_month_wildcard: bool,
    day_of_week_wildcard: bool,
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)].clone()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn error(field: Field, offset: usize, make: fn(Field, usize) -> ParseError) -> ParseError {
    make(field, offset)
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if is_step {
        let (_, maximum) = bounds(field);
        let width = maximum - bounds(field).0 + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else {
        let (minimum, maximum) = bounds(field);
        if value < minimum || value > maximum {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit() || matches!(byte, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = bounds(field);
    let mut selected = vec![false; (maximum + 1) as usize];

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| item_start + relative)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let base_end = slash_positions.first().copied().unwrap_or(item.len());
        let step_range = slash_positions.first().map(|&slash| {
            let start = slash + 1;
            (start, item.len())
        });

        let base = &item[..base_end];
        let (base_kind, first_start, first_end, second_start, second_end) =
            if base == b"*" {
                (0u8, 0usize, 0usize, 0usize, 0usize)
            } else {
                let hyphens: Vec<usize> = base
                    .iter()
                    .enumerate()
                    .filter_map(|(i, &b)| (b == b'-').then_some(i))
                    .collect();

                match hyphens.as_slice() {
                    [] if !base.is_empty() && base.iter().all(u8::is_ascii_digit) => {
                        (1, 0, base.len(), 0, 0)
                    }
                    [hyphen]
                        if *hyphen > 0
                            && *hyphen + 1 < base.len()
                            && base[..*hyphen].iter().all(u8::is_ascii_digit)
                            && base[*hyphen + 1..].iter().all(u8::is_ascii_digit) =>
                    {
                        (2, 0, *hyphen, *hyphen + 1, base.len())
                    }
                    _ => {
                        return Err(ParseError::InvalidSyntax {
                            field,
                            offset: item_start,
                        });
                    }
                }
            };

        if base_kind == 0 && base != b"*" {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        if let Some((step_start, step_end)) = step_range {
            if step_start == step_end || !item[step_start..step_end].iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }

        let (start_value, end_value) = match base_kind {
            0 => (minimum, maximum),
            1 => {
                let start = parse_number(
                    base,
                    first_start,
                    first_end,
                    field,
                    false,
                )?;
                let end = if step_range.is_some() { maximum } else { start };
                (start, end)
            }
            _ => {
                let start = parse_number(base, first_start, first_end, field, false)?;
                let end = parse_number(base, second_start, second_end, field, false)?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + second_start,
                    });
                }
                (start, end)
            }
        };

        let step = if let Some((step_start, step_end)) = step_range {
            parse_number(item, step_start, step_end, field, true)?
        } else {
            1
        };

        let mut value = start_value;
        while value <= end_value {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected
        .into_iter()
        .enumerate()
        .filter_map(|(value, is_selected)| is_selected.then_some(value as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut selected: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    for (index, (&field, text)) in fields.iter().zip(pieces.iter()).enumerate() {
        selected[index] = parse_field(text, field)?;
    }

    Ok(Cron {
        selected,
        day_of_month_wildcard: pieces[2] == "*",
        day_of_week_wildcard: pieces[4] == "*",
    })
}
