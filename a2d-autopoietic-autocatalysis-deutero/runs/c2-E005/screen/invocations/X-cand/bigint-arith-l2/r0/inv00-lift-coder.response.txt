```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits; zero is represented as [0].
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut first_invalid = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                first_invalid = Some(index);
                break;
            }
        }
        if let Some(index) = first_invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut first_separator_error = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                first_separator_error = Some(index);
                break;
            }
        }
        if let Some(index) = first_separator_error {
            return Err(Error::Separator { index });
        }

        let first_digit_index = (start..bytes.len())
            .find(|&index| bytes[index] != b'_')
            .expect("nonempty magnitude contains a digit");
        let digit_count = bytes[start..].iter().filter(|&&byte| byte != b'_').count();

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            for (index, &byte) in bytes.iter().enumerate().skip(start) {
                if byte != b'_' {
                    seen += 1;
                    if seen == excess_number {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index,
                        });
                    }
                }
            }
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes[start..].iter().rev() {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }
        let mut result = Self { negative, digits };
        result.normalize();
        Ok(result)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = self.digits.len().max(min_digits).max(1);
        let padding = width - self.digits.len();
        let mut out = String::with_capacity(width + width / group.max(1) + 1);

        if self.negative {
            out.push('-');
        }

        for pos in 0..width {
            if group > 0 && pos > 0 && (width - pos) % group == 0 {
                out.push('_');
            }
            let digit = if pos < padding {
                0
            } else {
                self.digits[self.digits.len() - 1 - (pos - padding)]
            };
            out.push(char::from(b'0' + digit));
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => Self::cmp_magnitude(self, rhs),
            (true, true) => Self::cmp_magnitude(rhs, self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self {
                negative: self.negative,
                digits: Self::add_magnitudes(&self.digits, &rhs.digits),
            };
            result.normalize();
            result
        } else {
            match Self::cmp_magnitude(self, rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self {
                        negative: self.negative,
                        digits: Self::sub_magnitudes(&self.digits, &rhs.digits),
                    };
                    result.normalize();
                    result
                }
                Ordering::Less => {
                    let mut result = Self {
                        negative: rhs.negative,
                        digits: Self::sub_magnitudes(&rhs.digits, &self.digits),
                    };
                    result.normalize();
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut product = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            for (j, &b) in rhs.digits.iter().enumerate() {
                product[i + j] += u32::from(a) * u32::from(b);
            }
        }

        let mut digits = Vec::with_capacity(product.len() + 1);
        let mut carry = 0u32;
        for value in product {
            let total = value + carry;
            digits.push((total % 10) as u8);
            carry = total / 10;
        }
        while carry > 0 {
            digits.push((carry % 10) as u8);
            carry /= 10;
        }

        let mut result = Self {
            negative: self.negative != rhs.negative,
            digits,
        };
        result.normalize();
        result
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn normalize(&mut self) {
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.is_empty() {
            self.digits.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn cmp_magnitude(lhs: &Self, rhs: &Self) -> Ordering {
        lhs.digits
            .len()
            .cmp(&rhs.digits.len())
            .then_with(|| lhs.digits.iter().rev().cmp(rhs.digits.iter().rev()))
    }

    fn add_magnitudes(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
        let mut result = Vec::with_capacity(lhs.len().max(rhs.len()) + 1);
        let mut carry = 0u8;
        for index in 0..lhs.len().max(rhs.len()) {
            let sum = lhs.get(index).copied().unwrap_or(0)
                + rhs.get(index).copied().unwrap_or(0)
                + carry;
            result.push(sum % 10);
            carry = sum / 10;
        }
        if carry > 0 {
            result.push(carry);
        }
        result
    }

    // Precondition: lhs magnitude is at least rhs magnitude.
    fn sub_magnitudes(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
        let mut result = Vec::with_capacity(lhs.len());
        let mut borrow = 0i8;
        for index in 0..lhs.len() {
            let mut digit = i16::from(lhs[index]) - i16::from(borrow as u8);
            let subtrahend = i16::from(rhs.get(index).copied().unwrap_or(0));
            digit -= subtrahend;
            if digit < 0 {
                digit += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            result.push(digit as u8);
        }
        while result.len() > 1 && result.last() == Some(&0) {
            result.pop();
        }
        result
    }
}
```
