```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: HashMap<String, Entry>,
    capacity: Option<usize>,
    next_age: u128,
}

fn canonical_key(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }
    if start == end {
        return Err(Error::EmptyKey);
    }

    for (offset, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphanumeric() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }

    Ok(input[start..end].to_ascii_lowercase())
}

fn compare_entries(a: &Entry, b: &Entry) -> std::cmp::Ordering {
    a.item
        .priority
        .cmp(&b.item.priority)
        .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
        .then_with(|| a.age.cmp(&b.age))
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: HashMap::new(),
            capacity: None,
            next_age: 0,
        }
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self {
            entries: HashMap::new(),
            capacity: Some(capacity),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<Admission, Error> {
        let key = canonical_key(key)?;
        if self.entries.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let proposed = Entry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };

        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(proposed.item));
        }

        if let Some(capacity) = self.capacity {
            if self.entries.len() >= capacity {
                let worst_key = self
                    .entries
                    .iter()
                    .max_by(|(_, a), (_, b)| compare_entries(a, b))
                    .map(|(key, _)| key.clone())
                    .expect("full nonzero queue has an entry");
                let worst = self.entries.get(&worst_key).expect("key exists");
                if compare_entries(&proposed, worst) != std::cmp::Ordering::Less {
                    return Ok(Admission::Discarded(proposed.item));
                }
                let evicted = self.entries.remove(&worst_key).expect("key exists").item;
                self.entries.insert(key, proposed);
                self.next_age = self.next_age.wrapping_add(1);
                return Ok(Admission::Evicted(evicted));
            }
        }

        self.entries.insert(key, proposed);
        self.next_age = self.next_age.wrapping_add(1);
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.entries
            .values()
            .min_by(|a, b| compare_entries(a, b))
            .map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let key = self
            .entries
            .iter()
            .min_by(|(_, a), (_, b)| compare_entries(a, b))
            .map(|(key, _)| key.clone())?;
        self.entries.remove(&key).map(|entry| entry.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonical_key(key)?;
        Ok(self.entries.get(&key).map(|entry| entry.item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonical_key(key)?;
        let entry = self
            .entries
            .get(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        let current = entry.item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries.get_mut(&key).expect("entry exists").item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonical_key(key)?;
        let entry = self
            .entries
            .get(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        let current = entry.item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries.get_mut(&key).expect("entry exists").item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonical_key(key)?;
        let entry = self
            .entries
            .get_mut(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        Ok(std::mem::replace(&mut entry.item.value, value.to_owned()))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonical_key(key)?;
        self.entries
            .remove(&key)
            .map(|entry| entry.item)
            .ok_or(Error::Missing { key })
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        let mut entries: Vec<&Entry> = self.entries.values().collect();
        entries.sort_by(|a, b| compare_entries(a, b));
        entries
            .into_iter()
            .map(|entry| entry.item.clone())
            .collect::<Vec<_>>()
            .into_iter()
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let Some(limit) = capacity else {
            return Vec::new();
        };

        let excess = self.entries.len().saturating_sub(limit);
        let mut ordered: Vec<&Entry> = self.entries.values().collect();
        ordered.sort_by(|a, b| compare_entries(b, a));
        let keys: Vec<String> = ordered
            .into_iter()
            .take(excess)
            .map(|entry| entry.item.key.clone())
            .collect();

        keys.into_iter()
            .filter_map(|key| self.entries.remove(&key).map(|entry| entry.item))
            .collect()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let mut source: Vec<Entry> = other.entries.values().cloned().collect();
        source.sort_by(compare_entries);

        if policy == CollisionPolicy::Reject {
            for entry in &source {
                if self.entries.contains_key(&entry.item.key) {
                    return Err(Error::Collision {
                        key: entry.item.key.clone(),
                    });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),
        };

        for mut entry in source {
            let key = entry.item.key.clone();
            if let Some(existing) = self.entries.get_mut(&key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        existing.item.priority = entry.item.priority;
                        existing.item.value = entry.item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                entry.age = self.next_age;
                self.next_age = self.next_age.wrapping_add(1);
                self.entries.insert(key, entry);
                report.moved += 1;
            }
        }

        other.entries.clear();
        report.evicted = self.set_capacity(self.capacity);
        Ok(report)
    }
}
```
