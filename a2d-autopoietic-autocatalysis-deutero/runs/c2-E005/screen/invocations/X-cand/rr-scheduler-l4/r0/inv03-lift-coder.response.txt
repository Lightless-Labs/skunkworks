```rust
use std::collections::VecDeque;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Process {
    id: u32,
    arrival: u64,
    bursts: Vec<u32>,
    level: usize,
    burst_pos: usize,
    remaining: u32,
    completion: Option<u64>,
}

#[derive(Clone, Copy, Debug)]
enum EventKind {
    Arrival,
    Wake { job: usize, level: usize },
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
}

fn push_idle(idle: &mut Vec<(u64, u64)>, start: u64, end: u64) {
    if start >= end {
        return;
    }
    if let Some(last) = idle.last_mut() {
        if last.1 == start {
            last.1 = end;
            return;
        }
    }
    idle.push((start, end));
}

pub fn schedule(
    jobs: &[Job],
    quanta: &[u32],
    switch_cost: u32,
) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut processes: Vec<Process> = jobs
        .iter()
        .map(|job| Process {
            id: job.id,
            arrival: u64::from(job.arrival),
            bursts: job.bursts.clone(),
            level: job.level,
            burst_pos: 0,
            remaining: job.bursts[0],
            completion: None,
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: u64::from(job.arrival),
            kind: EventKind::Arrival,
            index,
        })
        .collect();
    events.sort_by_key(|event| {
        let kind_order = match event.kind {
            EventKind::Arrival => 0usize,
            EventKind::Wake { .. } => 1usize,
        };
        (event.time, kind_order, event.index)
    });

    let mut event_cursor = 0usize;
    let mut queues: Vec<VecDeque<usize>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut time = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut completed_count = 0usize;

    while completed_count < jobs.len() {
        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor];
            match event.kind {
                EventKind::Arrival => {
                    queues[processes[event.index].level].push_back(event.index);
                }
                EventKind::Wake { job, level } => {
                    processes[job].level = level;
                    queues[level].push_back(job);
                }
            }
            event_cursor += 1;
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = events[event_cursor].time;
            if next_time > time {
                push_idle(&mut idle, time, next_time);
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let job_index = queues[level].pop_front().unwrap();
        let id = processes[job_index].id;

        if last_dispatch_id.is_some_and(|last_id| last_id != id) {
            time += u64::from(switch_cost);
        }

        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor];
            match event.kind {
                EventKind::Arrival => {
                    queues[processes[event.index].level].push_back(event.index);
                }
                EventKind::Wake { job, level } => {
                    processes[job].level = level;
                    queues[level].push_back(job);
                }
            }
            event_cursor += 1;
        }

        let run_start = time;
        let quantum = quanta[level];
        let consumed = processes[job_index].remaining.min(quantum);
        time += u64::from(consumed);
        let run_end = time;
        processes[job_index].remaining -= consumed;

        let burst_finished = processes[job_index].remaining == 0;
        let job_finished = burst_finished
            && processes[job_index].burst_pos + 1 == processes[job_index].bursts.len();

        runs.push(Run {
            id,
            start: run_start,
            end: run_end,
            finished: job_finished,
            level,
        });

        while event_cursor < events.len() && events[event_cursor].time <= run_end {
            let event = events[event_cursor];
            match event.kind {
                EventKind::Arrival => {
                    queues[processes[event.index].level].push_back(event.index);
                }
                EventKind::Wake { job, level } => {
                    processes[job].level = level;
                    queues[level].push_back(job);
                }
            }
            event_cursor += 1;
        }

        if !burst_finished {
            let new_level = (level + 1).min(quanta.len() - 1);
            processes[job_index].level = new_level;
            queues[new_level].push_back(job_index);
        } else if job_finished {
            processes[job_index].completion = Some(run_end);
            completed_count += 1;
        } else {
            let io_duration = u64::from(processes[job_index].bursts[processes[job_index].burst_pos + 1]);
            let next_cpu_pos = processes[job_index].burst_pos + 2;
            let wake_level = level.saturating_sub(1);
            processes[job_index].burst_pos = next_cpu_pos;
            processes[job_index].remaining = processes[job_index].bursts[next_cpu_pos];

            let wake_event = Event {
                time: run_end + io_duration,
                kind: EventKind::Wake {
                    job: job_index,
                    level: wake_level,
                },
                index: job_index,
            };
            let insertion_index = events[event_cursor..]
                .partition_point(|event| {
                    let event_kind = match event.kind {
                        EventKind::Arrival => 0usize,
                        EventKind::Wake { .. } => 1usize,
                    };
                    let wake_kind = 1usize;
                    (event.time, event_kind, event.index)
                        <= (wake_event.time, wake_kind, wake_event.index)
                })
                + event_cursor;
            events.insert(insertion_index, wake_event);
        }

        last_dispatch_id = Some(id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;
    let mut end = 0u64;

    for (index, process) in processes.iter().enumerate() {
        let completion = process.completion.unwrap_or(time);
        let turnaround = completion - process.arrival;
        let cpu_total: u64 = process
            .bursts
            .iter()
            .step_by(2)
            .map(|&value| u64::from(value))
            .sum();
        let io_total: u64 = process
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&value| u64::from(value))
            .sum();
        let waiting = turnaround - cpu_total - io_total;

        stats.push(Stat {
            id: jobs[index].id,
            completion,
            turnaround,
            waiting,
        });
        total_waiting += waiting;
        total_turnaround += turnaround;
        end = end.max(completion);
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = (total_waiting * 100 * 2 + count) / (2 * count);
    let mean_turnaround_x100 = (total_turnaround * 100 * 2 + count) / (2 * count);

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end,
    })
}
```
