#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct RuntimeJob {
    id: u32,
    arrival: u64,
    bursts: Vec<u32>,
    burst_pos: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    level: usize,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut runtime: Vec<RuntimeJob> = jobs
        .iter()
        .map(|job| RuntimeJob {
            id: job.id,
            arrival: u64::from(job.arrival),
            bursts: job.bursts.clone(),
            burst_pos: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
            level: job.level,
        })
        .collect();

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();

    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut completed_count = 0usize;

    while completed_count < jobs.len() {
        events.sort_by_key(|event| (event.time, event.kind, event.index));

        let mut consumed = 0usize;
        while consumed < events.len() && events[consumed].time <= time {
            let event = events[consumed];
            runtime[event.index].level = event.level;
            queues[event.level].push_back(event.index);
            consumed += 1;
        }
        events.drain(0..consumed);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if events.is_empty() {
                break;
            }
            let next_time = events[0].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let id = runtime[index].id;

        if last_dispatch_id.is_some_and(|last_id| last_id != id) {
            time += u64::from(switch_cost);
        }

        events.sort_by_key(|event| (event.time, event.kind, event.index));
        let mut consumed = 0usize;
        while consumed < events.len() && events[consumed].time <= time {
            let event = events[consumed];
            runtime[event.index].level = event.level;
            queues[event.level].push_back(event.index);
            consumed += 1;
        }
        events.drain(0..consumed);

        let quantum = quanta[level];
        let executed = runtime[index].remaining.min(quantum);
        let start = time;
        time += u64::from(executed);
        runtime[index].remaining -= executed;

        let burst_finished = runtime[index].remaining == 0;
        let final_burst = burst_finished && runtime[index].burst_pos + 1 == runtime[index].bursts.len();

        runs.push(Run {
            id,
            start,
            end: time,
            finished: final_burst,
            level,
        });

        events.sort_by_key(|event| (event.time, event.kind, event.index));
        let mut consumed = 0usize;
        while consumed < events.len() && events[consumed].time <= time {
            let event = events[consumed];
            runtime[event.index].level = event.level;
            queues[event.level].push_back(event.index);
            consumed += 1;
        }
        events.drain(0..consumed);

        if !burst_finished {
            let new_level = (level + 1).min(quanta.len() - 1);
            runtime[index].level = new_level;
            queues[new_level].push_back(index);
        } else if final_burst {
            runtime[index].completion = Some(time);
            completed_count += 1;
        } else {
            let io_delay = u64::from(runtime[index].bursts[runtime[index].burst_pos + 1]);
            let next_cpu_pos = runtime[index].burst_pos + 2;
            let wake_level = level.saturating_sub(1);
            runtime[index].burst_pos = next_cpu_pos;
            runtime[index].remaining = runtime[index].bursts[next_cpu_pos];
            events.push(Event {
                time: time + io_delay,
                kind: 1,
                index,
                level: wake_level,
            });
        }

        last_dispatch_id = Some(id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = runtime[index].completion.unwrap_or(time);
        let turnaround = completion - u64::from(job.arrival);
        let cpu: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&length| u64::from(length))
            .sum();
        let io: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&length| u64::from(length))
            .sum();
        let waiting = turnaround - cpu - io;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = (total_waiting * 100 * 2 + count) / (2 * count);
    let mean_turnaround_x100 = (total_turnaround * 100 * 2 + count) / (2 * count);

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end: time,
    })
}
