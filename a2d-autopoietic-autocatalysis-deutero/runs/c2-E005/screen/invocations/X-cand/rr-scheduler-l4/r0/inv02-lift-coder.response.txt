```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Proc {
    id: u32,
    arrival: u64,
    bursts: Vec<u32>,
    level: usize,
    burst_pos: usize,
    remaining: u32,
    completion: Option<u64>,
    cpu_total: u64,
    io_total: u64,
}

#[derive(Clone, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    level: usize,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|previous| previous.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if !(1..=31).contains(&job.bursts.len()) || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let mut procs: Vec<Proc> = jobs
        .iter()
        .map(|job| Proc {
            id: job.id,
            arrival: u64::from(job.arrival),
            bursts: job.bursts.clone(),
            level: job.level,
            burst_pos: 0,
            remaining: job.bursts[0],
            completion: None,
            cpu_total: job
                .bursts
                .iter()
                .step_by(2)
                .map(|&value| u64::from(value))
                .sum(),
            io_total: job
                .bursts
                .iter()
                .skip(1)
                .step_by(2)
                .map(|&value| u64::from(value))
                .sum(),
        })
        .collect();

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
            level: job.level,
        })
        .collect();

    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut last_id: Option<u32> = None;
    let mut completed = 0usize;

    while completed < jobs.len() {
        admit_events(&mut events, &mut queues, time);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if events.is_empty() {
                break;
            }
            let next_time = events.iter().map(|event| event.time).min().unwrap();
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let id = procs[index].id;

        if last_id.is_some_and(|previous| previous != id) {
            time += u64::from(switch_cost);
        }
        admit_events(&mut events, &mut queues, time);

        let start = time;
        let quantum = quanta[level];
        let consumed = procs[index].remaining.min(quantum);
        time += u64::from(consumed);
        let end = time;

        procs[index].remaining -= consumed;
        let burst_finished = procs[index].remaining == 0;
        let job_finished = burst_finished && procs[index].burst_pos + 1 == procs[index].bursts.len();

        runs.push(Run {
            id,
            start,
            end,
            finished: job_finished,
            level,
        });

        admit_events(&mut events, &mut queues, end);

        if job_finished {
            procs[index].completion = Some(end);
            completed += 1;
        } else if burst_finished {
            let io_pos = procs[index].burst_pos + 1;
            let delay = u64::from(procs[index].bursts[io_pos]);
            procs[index].burst_pos += 2;
            procs[index].remaining = procs[index].bursts[procs[index].burst_pos];
            let wake_level = level.saturating_sub(1);
            events.push(Event {
                time: end + delay,
                kind: 1,
                index,
                level: wake_level,
            });
            admit_events(&mut events, &mut queues, end);
        } else {
            procs[index].level = (level + 1).min(quanta.len() - 1);
            queues[procs[index].level].push_back(index);
        }

        last_id = Some(id);
    }

    let stats: Vec<Stat> = procs
        .iter()
        .map(|proc| {
            let completion = proc.completion.unwrap_or(time);
            let turnaround = completion - proc.arrival;
            Stat {
                id: proc.id,
                completion,
                turnaround,
                waiting: turnaround - proc.cpu_total - proc.io_total,
            }
        })
        .collect();

    let count = stats.len() as u64;
    let mean_waiting_x100 = if count == 0 {
        0
    } else {
        let total: u64 = stats.iter().map(|stat| stat.waiting).sum();
        (total * 100 + count / 2) / count
    };
    let mean_turnaround_x100 = if count == 0 {
        0
    } else {
        let total: u64 = stats.iter().map(|stat| stat.turnaround).sum();
        (total * 100 + count / 2) / count
    };

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end: time,
    })
}

fn admit_events(
    events: &mut Vec<Event>,
    queues: &mut [std::collections::VecDeque<usize>],
    through: u64,
) {
    events.sort_by_key(|event| (event.time, event.kind, event.index));
    let split = events.partition_point(|event| event.time <= through);
    let ready: Vec<Event> = events.drain(..split).collect();
    for event in ready {
        queues[event.level].push_back(event.index);
    }
}
```
