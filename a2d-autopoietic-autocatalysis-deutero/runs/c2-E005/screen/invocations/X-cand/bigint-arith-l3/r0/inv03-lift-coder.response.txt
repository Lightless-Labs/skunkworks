```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented as [0] and is never negative.
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if matches!(bytes[0], b'+' | b'-') { 1 } else { 0 };
        if start == bytes.len() && start == 1 {
            return Err(Error::SignOnly);
        }

        // Validate all non-separator bytes before reporting malformed separators.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let mut separator_error = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                separator_error = Some(index);
                break;
            }
        }
        if let Some(index) = separator_error {
            return Err(Error::Separator { index });
        }

        let first_digit = (start..bytes.len()).find(|&i| bytes[i] != b'_');
        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();

        if digit_count > 1 {
            if let Some(index) = first_digit {
                if bytes[index] == b'0' {
                    return Err(Error::LeadingZero { index });
                }
            }
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            let mut excess_index = bytes.len();
            for (index, &byte) in bytes.iter().enumerate().skip(start) {
                if byte != b'_' {
                    seen += 1;
                    if seen == excess_number {
                        excess_index = index;
                        break;
                    }
                }
            }
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: excess_index,
            });
        }

        let mut digits = Vec::with_capacity(digit_count.max(1));
        for &byte in bytes[start..].iter().rev() {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }
        if digits.is_empty() {
            digits.push(0);
        }

        let negative = bytes[0] == b'-' && !Self::is_zero_digits(&digits);
        Ok(Self { negative, digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude_len = self.digits.len().max(1);
        let width = magnitude_len.max(min_digits);
        let mut magnitude = String::with_capacity(width + width / group.max(1));
        let padding = width - magnitude_len;

        for i in 0..width {
            if group > 0 && i > 0 && (width - i) % group == 0 {
                magnitude.push('_');
            }
            if i < padding {
                magnitude.push('0');
            } else {
                magnitude.push(char::from(b'0' + self.digits[magnitude_len - 1 - (i - padding)]));
            }
        }

        if self.negative {
            let mut result = String::with_capacity(magnitude.len() + 1);
            result.push('-');
            result.push_str(&magnitude);
            result
        } else {
            magnitude
        }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => Self::cmp_magnitude(self, rhs),
            (true, true) => Self::cmp_magnitude(rhs, self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            return Self::from_magnitude(
                self.negative,
                Self::add_magnitudes(&self.digits, &rhs.digits),
            );
        }

        match Self::cmp_magnitude(self, rhs) {
            Ordering::Equal => Self::zero(),
            Ordering::Greater => Self::from_magnitude(
                self.negative,
                Self::sub_magnitudes(&self.digits, &rhs.digits),
            ),
            Ordering::Less => Self::from_magnitude(
                rhs.negative,
                Self::sub_magnitudes(&rhs.digits, &self.digits),
            ),
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if Self::is_zero_digits(&self.digits) {
            Self::zero()
        } else {
            Self {
                negative: !self.negative,
                digits: self.digits.clone(),
            }
        }
    }

    pub fn signum(&self) -> i8 {
        if Self::is_zero_digits(&self.digits) {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if Self::is_zero_digits(&self.digits) || Self::is_zero_digits(&rhs.digits) {
            return Self::zero();
        }

        let mut result = vec![0u8; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            let mut carry = 0u16;
            for (j, &b) in rhs.digits.iter().enumerate() {
                let index = i + j;
                let value = result[index] as u16 + a as u16 * b as u16 + carry;
                result[index] = (value % 10) as u8;
                carry = value / 10;
            }

            let mut index = i + rhs.digits.len();
            while carry > 0 {
                let value = result[index] as u16 + carry;
                result[index] = (value % 10) as u8;
                carry = value / 10;
                index += 1;
            }
        }

        Self::from_magnitude(self.negative ^ rhs.negative, result)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if Self::is_zero_digits(&rhs.digits) {
            return Err(Error::DivisionByZero);
        }

        let (quotient_digits, remainder_digits) =
            Self::div_magnitudes(&self.digits, &rhs.digits);

        let quotient = Self::from_magnitude(self.negative ^ rhs.negative, quotient_digits);
        let remainder = Self::from_magnitude(self.negative, remainder_digits);
        Ok((quotient, remainder))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn from_magnitude(negative: bool, mut digits: Vec<u8>) -> Self {
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        let is_zero = Self::is_zero_digits(&digits);
        Self {
            negative: negative && !is_zero,
            digits,
        }
    }

    fn is_zero_digits(digits: &[u8]) -> bool {
        digits.len() == 1 && digits[0] == 0
    }

    fn cmp_magnitude(lhs: &Self, rhs: &Self) -> Ordering {
        match lhs.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => lhs.digits.iter().rev().cmp(rhs.digits.iter().rev()),
            other => other,
        }
    }

    fn add_magnitudes(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
        let mut result = Vec::with_capacity(lhs.len().max(rhs.len()) + 1);
        let mut carry = 0u8;
        for i in 0..lhs.len().max(rhs.len()) {
            let sum = lhs.get(i).copied().unwrap_or(0)
                + rhs.get(i).copied().unwrap_or(0)
                + carry;
            result.push(sum % 10);
            carry = sum / 10;
        }
        if carry > 0 {
            result.push(carry);
        }
        result
    }

    // Precondition: lhs magnitude is at least rhs magnitude.
    fn sub_magnitudes(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
        let mut result = Vec::with_capacity(lhs.len());
        let mut borrow = 0i16;
        for i in 0..lhs.len() {
            let mut digit = lhs[i] as i16 - borrow - rhs.get(i).copied().unwrap_or(0) as i16;
            if digit < 0 {
                digit += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            result.push(digit as u8);
        }
        while result.len() > 1 && result.last() == Some(&0) {
            result.pop();
        }
        result
    }

    fn mul_magnitude_digit(magnitude: &[u8], digit: u8) -> Vec<u8> {
        if digit == 0 {
            return vec![0];
        }
        let mut result = Vec::with_capacity(magnitude.len() + 1);
        let mut carry = 0u16;
        for &value in magnitude {
            let product = value as u16 * digit as u16 + carry;
            result.push((product % 10) as u8);
            carry = product / 10;
        }
        while carry > 0 {
            result.push((carry % 10) as u8);
            carry /= 10;
        }
        result
    }

    fn div_magnitudes(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
        let mut quotient = vec![0u8; dividend.len()];
        let mut remainder = vec![0u8];

        for i in (0..dividend.len()).rev() {
            remainder.insert(0, dividend[i]);
            while remainder.len() > 1 && remainder.last() == Some(&0) {
                remainder.pop();
            }

            let mut quotient_digit = 0u8;
            for candidate in (1..=9u8).rev() {
                let product = Self::mul_magnitude_digit(divisor, candidate);
                if Self::cmp_digit_vectors(&product, &remainder) != Ordering::Greater {
                    quotient_digit = candidate;
                    remainder = Self::sub_magnitudes(&remainder, &product);
                    break;
                }
            }
            quotient[i] = quotient_digit;
        }

        while quotient.len() > 1 && quotient.last() == Some(&0) {
            quotient.pop();
        }
        while remainder.len() > 1 && remainder.last() == Some(&0) {
            remainder.pop();
        }
        (quotient, remainder)
    }

    fn cmp_digit_vectors(lhs: &[u8], rhs: &[u8]) -> Ordering {
        match lhs.len().cmp(&rhs.len()) {
            Ordering::Equal => lhs.iter().rev().cmp(rhs.iter().rev()),
            other => other,
        }
    }
}
```
