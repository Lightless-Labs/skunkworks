fn add_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u8;
    for index in 0..a.len().max(b.len()) {
        let sum = a.get(index).copied().unwrap_or(0)
            + b.get(index).copied().unwrap_or(0)
            + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }
    if carry > 0 {
        result.push(carry);
    }
    result
}

fn sub_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    // Requires |a| >= |b|.
    let mut result = Vec::with_capacity(a.len());

        let mut value = digit as i8 - borrow - b.get(index).copied().unwrap_or(0) as i8;
        if value < 0 {
            value += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push(value as u8);
    }
    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

fn mul_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut result = vec![0u8; a.len() + b.len()];
    for (i, &left) in a.iter().enumerate() {
        let mut carry = 0u16;
        for (j, &right) in b.iter().enumerate() {
            let index = i + j;
            let value = result[index] as u16 + left as u16 * right as u16 + carry;
            result[index] = (value % 10) as u8;
            carry = value / 10;
        }
        let mut index = i + b.len();
        while carry > 0 {
            if index == result.len() {
                result.push(0);
            }
            let value = result[index] as u16 + carry;
            result[index] = (value % 10) as u8;
            carry = value / 10;
            index += 1;
        }
    }
    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

fn mul_magnitude_digit(value: &[u8], digit: u8) -> Vec<u8> {

            return Err(Error::Empty);
        }

        if start == bytes.len() && start == 1 {
            return Err(Error::SignOnly);

        // Validate all non-separator bytes before reporting malformed separators.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {

                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                separator_error = Some(index);
                break;
            }
        }
        if let Some(index) = separator_error {
            return Err(Error::Separator { index });
        }

        let first_digit = (start..bytes.len()).find(|&i| bytes[i] != b'_');

                        break;

                digits.push(byte - b'0');
            }

        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if Self::is_zero_digits(&self.digits) || Self::is_zero_digits(&rhs.digits) {
            return Self::zero();
        }

        let mut result = vec![0u8; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            let mut carry = 0u16;
            for (j, &b) in rhs.digits.iter().enumerate() {
                let index = i + j;
                let value = result[index] as u16 + a as u16 * b as u16 + carry;
                result[index] = (value % 10) as u8;
                carry = value / 10;
            }

            let mut index = i + rhs.digits.len();
            while carry > 0 {
                let value = result[index] as u16 + carry;
                result[index] = (value % 10) as u8;
                carry = value / 10;
                index += 1;
            }
        }

        Self::from_magnitude(self.negative ^ rhs.negative, result)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if Self::is_zero_digits(&rhs.digits) {
            return Err(Error::DivisionByZero);
        }

        let (quotient_digits, remainder_digits) =
            Self::div_magnitudes(&self.digits, &rhs.digits);

        let quotient = Self::from_magnitude(self.negative ^ rhs.negative, quotient_digits);
        let remainder = Self::from_magnitude(self.negative, remainder_digits);
        Ok((quotient, remainder))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn from_magnitude(negative: bool, mut digits: Vec<u8>) -> Self {
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        let is_zero = Self::is_zero_digits(&digits);
        Self {
            negative: negative && !is_zero,
            digits,
        }
    }

    fn is_zero_digits(digits: &[u8]) -> bool {
        digits.len() == 1 && digits[0] == 0
    }

    fn cmp_magnitude(lhs: &Self, rhs: &Self) -> Ordering {
        match lhs.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => lhs.digits.iter().rev().cmp(rhs.digits.iter().rev()),
            other => other,
        }
    }

    fn add_magnitudes(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
        let mut result = Vec::with_capacity(lhs.len().max(rhs.len()) + 1);
        let mut carry = 0u8;
        for i in 0..lhs.len().max(rhs.len()) {
            let sum = lhs.get(i).copied().unwrap_or(0)
                + rhs.get(i).copied().unwrap_or(0)
                + carry;
            result.push(sum % 10);
            carry = sum / 10;
        }
        if carry > 0 {
            result.push(carry);
        }
        result
    }

    // Precondition: lhs magnitude is at least rhs magnitude.
    fn sub_magnitudes(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
        let mut result = Vec::with_capacity(lhs.len());
        let mut borrow = 0i16;
        for i in 0..lhs.len() {
            let mut digit = lhs[i] as i16 - borrow - rhs.get(i).copied().unwrap_or(0) as i16;
            if digit < 0 {
                digit += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            result.push(digit as u8);
        }
        while result.len() > 1 && result.last() == Some(&0) {
            result.pop();
        }
        result
    }

    fn mul_magnitude_digit(magnitude: &[u8], digit: u8) -> Vec<u8> {
        if digit == 0 {
            return vec![0];
        }
        let mut result = Vec::with_capacity(magnitude.len() + 1);
        let mut carry = 0u16;
        for &value in magnitude {
            let product = value as u16 * digit as u16 + carry;
            result.push((product % 10) as u8);
            carry = product / 10;
        }
        while carry > 0 {
            result.push((carry % 10) as u8);
            carry /= 10;
        }
        result
    }

    fn div_magnitudes(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
        let mut quotient = vec![0u8; dividend.len()];
        let mut remainder = vec![0u8];

        for i in (0..dividend.len()).rev() {
            remainder.insert(0, dividend[i]);
            while remainder.len() > 1 && remainder.last() == Some(&0) {
                remainder.pop();
            }

            let mut quotient_digit = 0u8;
            for candidate in (1..=9u8).rev() {
                let product = Self::mul_magnitude_digit(divisor, candidate);
                if Self::cmp_digit_vectors(&product, &remainder) != Ordering::Greater {
                    quotient_digit = candidate;
                    remainder = Self::sub_magnitudes(&remainder, &product);
                    break;
                }
            }
            quotient[i] = quotient_digit;
        }

        while quotient.len() > 1 && quotient.last() == Some(&0) {
            quotient.pop();
        }
        while remainder.len() > 1 && remainder.last() == Some(&0) {
            remainder.pop();
        }
        (quotient, remainder)
    }

    fn cmp_digit_vectors(lhs: &[u8], rhs: &[u8]) -> Ordering {
        match lhs.len().cmp(&rhs.len()) {
            Ordering::Equal => lhs.iter().rev().cmp(rhs.iter().rev()),
            other => other,
        }
    }
}

    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let sign_len = usize::from(bytes[0] == b'+' || bytes[0] == b'-');
        if sign_len == bytes.len() {
            return Err(Error::SignOnly);
        }

        let negative = sign_len == 1 && bytes[0] == b'-';
        let mut invalid = None;
        for (i, &b) in bytes.iter().enumerate().skip(sign_len) {
            if b != b'_' && !b.is_ascii_digit() {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut separator = None;
        for i in sign_len..bytes.len() {
            if bytes[i] == b'_'
                && (i == sign_len
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
            {
                separator = Some(i);
                break;
            }
        }
        if let Some(index) = separator {
            return Err(Error::Separator { index });
        }

        let first_digit = (sign_len..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        let digit_count = bytes[sign_len..].iter().filter(|&&b| b != b'_').count();

        if digit_count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero { index: first_digit });
        }

        if digit_count > max_digits {
            let target = max_digits + 1;
            let mut seen = 0usize;
            for (i, &b) in bytes.iter().enumerate().skip(sign_len) {
                if b != b'_' {
                    seen += 1;
                    if seen == target {
                        return Err(Error::TooManyDigits { limit: max_digits, index: i });
                    }
                }
            }
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &b in bytes[sign_len..].iter().rev() {
            if b != b'_' {
                digits.push(b - b'0');
            }
        }
        let mut value = Self { negative, digits };
        value.normalize();
        Ok(value)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let width = min_digits.max(self.digits.len()).max(1);
        let pad = width - self.digits.len();
        let mut magnitude = String::with_capacity(width + width.saturating_sub(1) / group.max(1));
        for _ in 0..pad {
            magnitude.push('0');
        }
        for &d in self.digits.iter().rev() {
            magnitude.push(char::from(b'0' + d));
        }

        if group > 0 && magnitude.len() > group {
            let mut grouped = String::with_capacity(magnitude.len() + magnitude.len() / group);
            let first = magnitude.len() % group;
            let first = if first == 0 { group } else { first };
            grouped.push_str(&magnitude[..first]);
            for chunk in magnitude[first..].as_bytes().chunks(group) {
                grouped.push('_');
                grouped.push_str(std::str::from_utf8(chunk).expect("ASCII digits"));
            }
            magnitude = grouped;
        }

        if self.negative {
            let mut out = String::with_capacity(magnitude.len() + 1);
            out.push('-');
            out.push_str(&magnitude);
            out
        } else {
            magnitude
        }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {

            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_magnitude(&self.digits, &rhs.digits),
            (true, true) => cmp_magnitude(&rhs.digits, &self.digits),

    pub fn add(&self, rhs: &Self) -> Self {

            let mut result = Self {
                negative: self.negative,

            result.normalize();
            result

                        negative: self.negative,
                        digits: sub_magnitude(&self.digits, &rhs.digits),

                    result.normalize();

                        digits: sub_magnitude(&rhs.digits, &self.digits),

                    result.normalize();
                    result
                }
                Ordering::Equal => Self::zero(),

        }

    pub fn sub(&self, rhs: &Self) -> Self {

    pub fn neg(&self) -> Self {
        let mut result = self.clone();

            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }

    pub fn mul(&self, rhs: &Self) -> Self {

        }
        let mut result = Self {

            digits: mul_magnitude(&self.digits, &rhs.digits),

        }

        let (quotient, remainder) = div_magnitude(&self.digits, &rhs.digits);
        let mut q = Self {
            negative: self.negative ^ rhs.negative,
            digits: quotient,
        };
        let mut r = Self {
            negative: self.negative,
            digits: remainder,
        };
        q.normalize();
        r.normalize();
        Ok((q, r))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0

        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.is_empty() {
            self.digits.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

                    other => return other,

// Requires a >= b.
fn sub_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len());
    let mut borrow = 0i8;
    for i in 0..a.len() {
        let mut digit = a[i] as i8 - b.get(i).copied().unwrap_or(0) as i8 - borrow;
        if digit < 0 {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        out.push(digit as u8);
    }
    while out.len() > 1 && out.last() == Some(&0) {
        out.pop();
    }
    out
}

fn mul_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = vec![0u8; a.len() + b.len()];
    for (i, &x) in a.iter().enumerate() {
        let mut carry = 0u16;
        for (j, &y) in b.iter().enumerate() {
            let k = i + j;
            let value = out[k] as u16 + x as u16 * y as u16 + carry;
            out[k] = (value % 10) as u8;
            carry = value / 10;
        }
        let mut k = i + b.len();
        while carry != 0 {
            let value = out[k] as u16 + carry;
            out[k] = (value % 10) as u8;
            carry = value / 10;
            k += 1;
            if k == out.len() && carry != 0 {
                out.push(0);
            }
        }
    }
    while out.len() > 1 && out.last() == Some(&0) {
        out.pop();
    }
    out
}

// Long division on little-endian base-10 magnitudes.
fn div_magnitude(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
    if cmp_magnitude(dividend, divisor) == Ordering::Less {
        return (vec![0], dividend.to_vec());
    }

    let mut quotient_be = Vec::with_capacity(dividend.len());
    let mut remainder = vec![0u8];

    for &digit in dividend.iter().rev() {
        remainder.insert(0, digit);
        trim_magnitude(&mut remainder);

        let mut qdigit = 0u8;
        while cmp_magnitude(&remainder, divisor) != Ordering::Less {
            remainder = sub_magnitude(&remainder, divisor);
            qdigit += 1;
        }
        quotient_be.push(qdigit);
    }

    while quotient_be.len() > 1 && quotient_be.first() == Some(&0) {
        quotient_be.remove(0);
    }
    quotient_be.reverse();
    trim_magnitude(&mut remainder);
    (quotient_be, remainder)
}

fn trim_magnitude(value: &mut Vec<u8>) {
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    if value.is_empty() {
        value.push(0);
    }
}

pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented as [0] and is never negative.
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };

        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over separator, leading-zero, and length errors.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        // Check separators in the magnitude.
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index });
            }
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut first_excess = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte.is_ascii_digit() {
                digit_count += 1;
                if first_digit.is_none() {
                    first_digit = Some((index, byte));
                }
                if digit_count == max_digits.saturating_add(1) {
                    first_excess = Some(index);
                }
            }
        }

        if digit_count > 1 {
            if let Some((index, b'0')) = first_digit {
                return Err(Error::LeadingZero { index });
            }
        }

        if digit_count > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: first_excess.unwrap_or_else(|| {
                    first_digit.map(|(index, _)| index).unwrap_or(start)
                }),
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes.iter().skip(start) {
            if byte.is_ascii_digit() {
                digits.push(byte - b'0');
            }
        }
        digits.reverse();

        let mut value = Self { negative, digits };
        value.normalize();
        Ok(value)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude_len = self.digits.len().max(1);
        let width = magnitude_len.max(min_digits);
        let padding = width - magnitude_len;

        let mut out = String::with_capacity(
            width.saturating_add(if group > 0 { width.saturating_sub(1) / group } else { 0 })
                .saturating_add(usize::from(self.negative)),
        );
        if self.negative {
            out.push('-');
        }

        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                out.push('_');
            }
            let digit = if position < padding {
                0
            } else {
                self.digits[magnitude_len - 1 - (position - padding)]
            };
            out.push(char::from(b'0' + digit));
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> std::cmp::Ordering {
        use std::cmp::Ordering;

        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => compare_magnitude(&self.digits, &rhs.digits),
            (true, true) => compare_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self {
                negative: self.negative,
                digits: add_magnitude(&self.digits, &rhs.digits),
            };
            result.normalize();
            result
        } else {
            match compare_magnitude(&self.digits, &rhs.digits) {
                std::cmp::Ordering::Greater => {
                    let mut result = Self {
                        negative: self.negative,
                        digits: subtract_magnitude(&self.digits, &rhs.digits),
                    };
                    result.normalize();
                    result
                }
                std::cmp::Ordering::Less => {
                    let mut result = Self {
                        negative: rhs.negative,
                        digits: subtract_magnitude(&rhs.digits, &self.digits),
                    };
                    result.normalize();
                    result
                }
                std::cmp::Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut result = Self {
            negative: self.negative != rhs.negative,
            digits: multiply_magnitude(&self.digits, &rhs.digits),
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let (quotient_digits, remainder_digits) =
            divide_magnitude(&self.digits, &rhs.digits);

        let mut quotient = Self {
            negative: self.negative != rhs.negative,
            digits: quotient_digits,
        };
        quotient.normalize();

        let mut remainder = Self {
            negative: self.negative,
            digits: remainder_digits,
        };
        remainder.normalize();

        Ok((quotient, remainder))
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn normalize(&mut self) {
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.is_empty() {
            self.digits.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }
}

    let mut borrow = 0i16;

    for (index, &digit) in a.iter().enumerate() {
        let mut difference = i16::from(digit)
            - i16::from(*b.get(index).unwrap_or(&0))
            - borrow;
        if difference < 0 {
            difference += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push(difference as u8);
    }

    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

fn multiply_magnitude(a: &[u8], b: &[u8]) -> Vec<u8> {
    if (a.len() == 1 && a[0] == 0) || (b.len() == 1 && b[0] == 0) {
        return vec![0];
    }

    let mut accum = vec![0u64; a.len().saturating_add(b.len())];
    for (i, &x) in a.iter().enumerate() {
        for (j, &y) in b.iter().enumerate() {
            accum[i + j] += u64::from(x) * u64::from(y);
        }
    }

    let mut result = Vec::with_capacity(accum.len().saturating_add(1));
    let mut carry = 0u64;
    for value in accum {
        let total = value + carry;
        result.push((total % 10) as u8);
        carry = total / 10;
    }
    while carry != 0 {
        result.push((carry % 10) as u8);
        carry /= 10;
    }

    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

fn divide_magnitude(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
    if compare_magnitude(dividend, divisor) == std::cmp::Ordering::Less {
        return (vec![0], dividend.to_vec());
    }

    let mut remainder = vec![0];
    let mut quotient_msd = Vec::with_capacity(dividend.len());

    for &digit in dividend.iter().rev() {
        if remainder.len() == 1 && remainder[0] == 0 {
            remainder[0] = digit;
        } else {
            remainder.insert(0, digit);
        }
        trim_magnitude(&mut remainder);

        let mut quotient_digit = 0u8;
        while compare_magnitude(&remainder, divisor) != std::cmp::Ordering::Less {
            remainder = subtract_magnitude(&remainder, divisor);
            quotient_digit += 1;
        }
        quotient_msd.push(quotient_digit);
    }

    quotient_msd.reverse();
    trim_magnitude(&mut quotient_msd);
    trim_magnitude(&mut remainder);
    (quotient_msd, remainder)
}

fn trim_magnitude(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    if digits.is_empty() {
        digits.push(0);
    }
}
