        let mut slash = None;
        let mut denominator_start = 0;
        let mut denominator_end = 0;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
            slash = Some(i);
            i += 1;
            denominator_start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == denominator_start {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
            denominator_end = i;
            if i < bytes.len() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        if numerator_end - numerator_start >= 2 && bytes[numerator_start] == b'0' {
            return Err(RationalError::Parse {
                at: numerator_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }
        if bytes[0] == b'-' && numerator_end - numerator_start == 1 && bytes[1] == b'0' {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::NonCanonical,
            });
        }
        if slash.is_some()
            && denominator_end - denominator_start >= 2
            && bytes[denominator_start] == b'0'
        {
            return Err(RationalError::Parse {
                at: denominator_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }

        let negative = bytes[0] == b'-';
        let numerator_limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let mut numerator_magnitude = 0u128;
        for &byte in &bytes[numerator_start..numerator_end] {
            numerator_magnitude = numerator_magnitude * 10 + (byte - b'0') as u128;
            if numerator_magnitude > numerator_limit {
                return Err(RationalError::Parse {
                    at: 0,
                    kind: ParseKind::OutOfRange,
                });
            }
        }

        let mut denominator_value = 1u128;
        if slash.is_some() {
            denominator_value = 0;
            for &byte in &bytes[denominator_start..denominator_end] {
                denominator_value = denominator_value * 10 + (byte - b'0') as u128;
                if denominator_value > i64::MAX as u128 {
                    return Err(RationalError::Parse {
                        at: denominator_start,
                        kind: ParseKind::OutOfRange,
                    });
                }
            }

            if denominator_value == 0 {
                return Err(RationalError::Parse {
                    at: denominator_start,
                    kind: ParseKind::ZeroDenominator,
                });
            }

            if denominator_value == 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
            if numerator_magnitude == 0 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
            if gcd(numerator_magnitude, denominator_value) > 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let signed_numerator = if negative {
            -(numerator_magnitude as i128)
        } else {
            numerator_magnitude as i128
        };
        rational_from_parts(signed_numerator, denominator_value as i128)
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_wide(num as i128, den as i128).ok_or(RationalError::Overflow)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_wide(n, d).ok_or(RationalError::Overflow)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_wide(n, d).ok_or(RationalError::Overflow)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.num as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_wide(n, d).ok_or(RationalError::Overflow)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let n = self.num as i128 * rhs.den as i128;
        let d = self.den as i128 * rhs.num as i128;
        Self::from_wide(n, d).ok_or(RationalError::Overflow)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_wide(-(self.num as i128), self.den as i128).ok_or(RationalError::Overflow)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }

        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }
        let num_end = i;

        let mut den_start = None;
        let mut den_end = None;
        let mut slash_at = None;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            slash_at = Some(i);
            i += 1;
            let start = i;
            den_start = Some(start);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        if num_end - num_start >= 2 && bytes[num_start] == b'0' {
            return Err(parse_error(num_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_end - num_start == 1 && bytes[num_start] == b'0' {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }
        if let (Some(start), Some(end)) = (den_start, den_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_error(start + 1, ParseKind::NonCanonical));
            }
        }

        let num_limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let num_magnitude = parse_magnitude(bytes, num_start, num_end);
        if num_magnitude > num_limit {
            return Err(parse_error(0, ParseKind::OutOfRange));
        }

        let mut den_magnitude = 1u128;
        if let (Some(start), Some(end)) = (den_start, den_end) {
            den_magnitude = parse_magnitude(bytes, start, end);
            if den_magnitude > i64::MAX as u128 {
                return Err(parse_error(start, ParseKind::OutOfRange));
            }
            if den_magnitude == 0 {
                return Err(parse_error(start, ParseKind::ZeroDenominator));
            }
        }

        if let (Some(start), Some(_)) = (den_start, den_end) {
            let slash = slash_at.unwrap();
            if den_magnitude == 1 {
                return Err(parse_error(slash, ParseKind::NonCanonical));
            }
            if num_magnitude == 0 {
                return Err(parse_error(slash, ParseKind::NonCanonical));
            }
            if gcd_u128(num_magnitude, den_magnitude) > 1 {
                return Err(parse_error(slash, ParseKind::NonCanonical));
            }
            let _ = start;
        }

        let signed_num = if negative {
            if num_magnitude == (1u128 << 63) {
                i64::MIN as i128
            } else {
                -(num_magnitude as i128)
            }
        } else {
            num_magnitude as i128
        };
        Self::from_wide(signed_num, den_magnitude as i128).ok_or(RationalError::Overflow)
    }

    fn from_wide(mut num: i128, mut den: i128) -> Option<Self> {
        if den == 0 {
            return None;
        }
        if den < 0 {
            num = -num;
            den = -den;
        }
        if num == 0 {
            return Some(Self { num: 0, den: 1 });
        }

        let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
        num /= g;
        den /= g;

        if num < i64::MIN as i128 || num > i64::MAX as i128 {
            return None;
        }
        if den < 1 || den > i64::MAX as i128 {
            return None;
        }
        Some(Self {
            num: num as i64,
            den: den as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

fn parse_error(at: usize, kind: ParseKind) -> RationalError {
    RationalError::Parse { at, kind }
}

fn parse_magnitude(bytes: &[u8], start: usize, end: usize) -> u128 {
    let mut value = 0u128;
    for &b in &bytes[start..end] {
        value = value * 10 + (b - b'0') as u128;
    }
    value
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}
