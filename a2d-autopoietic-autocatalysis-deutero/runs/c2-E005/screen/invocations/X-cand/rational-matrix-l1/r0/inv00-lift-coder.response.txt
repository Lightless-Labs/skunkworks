```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_parts(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let (num, den) = if den < 0 {
        (-num, -den)
    } else {
        (num, den)
    };

    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let divisor = gcd_u128(num.unsigned_abs(), den as u128) as i128;
    let reduced_num = num / divisor;
    let reduced_den = den / divisor;

    if reduced_num < i64::MIN as i128
        || reduced_num > i64::MAX as i128
        || reduced_den < 1
        || reduced_den > i64::MAX as i128
    {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: reduced_num as i64,
        den: reduced_den as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        from_parts(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_parts(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_parts(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_parts(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        from_parts(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        from_parts(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        fn parse_error(at: usize, kind: ParseKind) -> RationalError {
            RationalError::Parse { at, kind }
        }

        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }

        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            i += 1;
            let start = i;
            den_start = Some(start);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        let negative = bytes[0] == b'-';
        let num_digits_start = if negative { 1 } else { 0 };
        let num_digits = num_end - num_digits_start;

        if num_digits >= 2 && bytes[num_digits_start] == b'0' {
            return Err(parse_error(num_digits_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == 1 && bytes[1] == b'0' {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        if let (Some(start), Some(end)) = (den_start, den_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_error(start + 1, ParseKind::NonCanonical));
            }
        }

        let num_magnitude = parse_magnitude(&bytes[num_digits_start..num_end]);
        let num_limit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let num_magnitude = match num_magnitude {
            Some(v) if v <= num_limit => v,
            _ => return Err(parse_error(0, ParseKind::OutOfRange)),
        };

        let den_magnitude = if let (Some(start), Some(end)) = (den_start, den_end) {
            match parse_magnitude(&bytes[start..end]) {
                Some(v) if v <= i64::MAX as u128 => Some(v),
                _ => return Err(parse_error(start, ParseKind::OutOfRange)),
            }
        } else {
            None
        };

        if let (Some(start), Some(_)) = (den_start, den_end) {
            if den_magnitude == Some(0) {
                return Err(parse_error(start, ParseKind::ZeroDenominator));
            }
            if den_magnitude == Some(1) {
                return Err(parse_error(start - 1, ParseKind::NonCanonical));
            }
            if num_magnitude == 0 {
                return Err(parse_error(start - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(num_magnitude, den_magnitude.unwrap()) > 1 {
                return Err(parse_error(start - 1, ParseKind::NonCanonical));
            }
        }

        let signed_num = if negative {
            if num_magnitude == (1u128 << 63) {
                i64::MIN as i128
            } else {
                -(num_magnitude as i128)
            }
        } else {
            num_magnitude as i128
        };

        let den = den_magnitude.unwrap_or(1) as i128;
        Ok(Rational {
            num: signed_num as i64,
            den: den as i64,
        })
    }
}

fn parse_magnitude(bytes: &[u8]) -> Option<u128> {
    let mut value = 0u128;
    for &byte in bytes {
        let digit = (byte - b'0') as u128;
        value = value.checked_mul(10)?.checked_add(digit)?;
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
