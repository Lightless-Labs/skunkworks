```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_parts(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let (num, den) = if den < 0 {
        (-num, -den)
    } else {
        (num, den)
    };

    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let divisor = gcd(num.unsigned_abs(), den as u128);
    let reduced_num = num / divisor as i128;
    let reduced_den = den / divisor as i128;

    if reduced_num < i64::MIN as i128
        || reduced_num > i64::MAX as i128
        || reduced_den < 1
        || reduced_den > i64::MAX as i128
    {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: reduced_num as i64,
        den: reduced_den as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_parts(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let num = self.num as i128 * rhs.den as i128
            + rhs.num as i128 * self.den as i128;
        let den = self.den as i128 * rhs.den as i128;
        rational_from_parts(num, den)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let num = self.num as i128 * rhs.den as i128
            - rhs.num as i128 * self.den as i128;
        let den = self.den as i128 * rhs.den as i128;
        rational_from_parts(num, den)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let num = self.num as i128 * rhs.num as i128;
        let den = self.den as i128 * rhs.den as i128;
        rational_from_parts(num, den)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let num = self.num as i128 * rhs.den as i128;
        let den = self.den as i128 * rhs.num as i128;
        rational_from_parts(num, den)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        rational_from_parts(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }

        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_start {
            return Err(RationalError::Parse {
                at: i,
                kind: ParseKind::InvalidByte,
            });
        }

        let numerator_end = i;
        let mut slash = None;
        let mut denominator_start = 0;
        let mut denominator_end = 0;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
            slash = Some(i);
            i += 1;
            denominator_start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == denominator_start {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
            denominator_end = i;
            if i < bytes.len() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        if numerator_end - numerator_start >= 2 && bytes[numerator_start] == b'0' {
            return Err(RationalError::Parse {
                at: numerator_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }
        if bytes[0] == b'-' && numerator_end - numerator_start == 1 && bytes[1] == b'0' {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::NonCanonical,
            });
        }
        if slash.is_some()
            && denominator_end - denominator_start >= 2
            && bytes[denominator_start] == b'0'
        {
            return Err(RationalError::Parse {
                at: denominator_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }

        let negative = bytes[0] == b'-';
        let numerator_limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let mut numerator_magnitude = 0u128;
        for &byte in &bytes[numerator_start..numerator_end] {
            numerator_magnitude = numerator_magnitude * 10 + (byte - b'0') as u128;
            if numerator_magnitude > numerator_limit {
                return Err(RationalError::Parse {
                    at: 0,
                    kind: ParseKind::OutOfRange,
                });
            }
        }

        let mut denominator_value = 1u128;
        if slash.is_some() {
            denominator_value = 0;
            for &byte in &bytes[denominator_start..denominator_end] {
                denominator_value = denominator_value * 10 + (byte - b'0') as u128;
                if denominator_value > i64::MAX as u128 {
                    return Err(RationalError::Parse {
                        at: denominator_start,
                        kind: ParseKind::OutOfRange,
                    });
                }
            }

            if denominator_value == 0 {
                return Err(RationalError::Parse {
                    at: denominator_start,
                    kind: ParseKind::ZeroDenominator,
                });
            }

            if denominator_value == 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
            if numerator_magnitude == 0 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
            if gcd(numerator_magnitude, denominator_value) > 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let signed_numerator = if negative {
            -(numerator_magnitude as i128)
        } else {
            numerator_magnitude as i128
        };
        rational_from_parts(signed_numerator, denominator_value as i128)
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
