#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|column| column == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|column| column == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }

        let result = parse_with_config(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_records(input, b',', b'"')
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_config(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let mut records = parse_records(input, config.delimiter, config.quote)?;

    let (header, rows) = if config.headers && !records.is_empty() {
        let header = records.remove(0);

        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: Vec<(&str, usize)> = Vec::new();
            for (second, name) in header.iter().enumerate() {
                if let Some((_, first)) = seen.iter().find(|(existing, _)| *existing == name) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: *first,
                        second,
                    });
                }
                seen.push((name.as_str(), second));
            }
        }

        for (index, row) in records.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }

        (Some(header), records)
    } else {
        (None, records)
    };

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum ParseState {
    FieldStart,
    Unquoted,
    Quoted,
    AfterQuote,
}

fn parse_records(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut position = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = ParseState::FieldStart;
    let mut quote_open = 0usize;
    let mut record_started = false;
    let mut record_index = 0usize;

    while position < bytes.len() {
        let byte = bytes[position];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: position,
            });
        }

        if state != ParseState::Quoted && byte == b'\r' {
            if bytes.get(position + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: position,
                });
            }
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            position += 2;
            record_index += 1;
            state = ParseState::FieldStart;
            record_started = false;
            continue;
        }

        if state != ParseState::Quoted && byte == b'\n' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            position += 1;
            record_index += 1;
            state = ParseState::FieldStart;
            record_started = false;
            continue;
        }

        match state {
            ParseState::FieldStart => {
                if byte == delimiter {
                    record.push(String::new());
                    record_started = true;
                    position += 1;
                } else if byte == quote {
                    quote_open = position;
                    state = ParseState::Quoted;
                    record_started = true;
                    position += 1;
                } else {
                    let ch = input[position..].chars().next().unwrap();
                    field.push(ch);
                    state = ParseState::Unquoted;
                    record_started = true;
                    position += ch.len_utf8();
                }
            }
            ParseState::Unquoted => {
                if byte == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = ParseState::FieldStart;
                    record_started = true;
                    position += 1;
                } else if byte == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: position,
                    });
                } else {
                    let ch = input[position..].chars().next().unwrap();
                    field.push(ch);
                    position += ch.len_utf8();
                }
            }
            ParseState::Quoted => {
                if byte == quote {
                    if bytes.get(position + 1) == Some(&quote) {
                        field.push(quote as char);
                        position += 2;
                    } else {
                        state = ParseState::AfterQuote;
                        position += 1;
                    }
                } else {
                    let ch = input[position..].chars().next().unwrap();
                    field.push(ch);
                    position += ch.len_utf8();
                }
            }
            ParseState::AfterQuote => {
                if byte == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = ParseState::FieldStart;
                    record_started = true;
                    position += 1;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: position,
                    });
                }
            }
        }
    }

    if state == ParseState::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_open,
        });
    }

    if record_started || !record.is_empty() || !field.is_empty() || state == ParseState::AfterQuote {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}
