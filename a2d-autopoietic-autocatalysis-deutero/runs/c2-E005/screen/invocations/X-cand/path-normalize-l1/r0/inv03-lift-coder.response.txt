#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    let absolute = path.as_bytes()[0] == b'/';
    let directory_intent = path.ends_with('/');
    let mut components: Vec<(String, usize)> = Vec::new();
    let mut last_nonempty: Option<(&str, usize)> = None;

    let mut start = 0;
    for (index, byte) in path.as_bytes().iter().enumerate() {
        if *byte == b'/' {
            if start < index {
                let component = &path[start..index];
                last_nonempty = Some((component, start));
                process_component(component, start, absolute, &mut components)?;
            }
            start = index + 1;
        }
    }

    if start < path.len() {
        let component = &path[start..];
        last_nonempty = Some((component, start));
        process_component(component, start, absolute, &mut components)?;
    }

    let directory_intent =
        directory_intent || matches!(last_nonempty, Some((".", _) | ("..", _)));

    let mut result = String::new();
    if absolute {
        result.push('/');
    }

    for (index, (component, _)) in components.iter().enumerate() {
        if index > 0 {
            result.push('/');
        }
        result.push_str(component);
    }

    if result.is_empty() {
        result.push('.');
    }

    if directory_intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}

fn process_component(
    component: &str,
    at: usize,
    absolute: bool,
    components: &mut Vec<(String, usize)>,
) -> Result<(), PathError> {
    match component {
        "." => {}
        ".." => {
            if let Some((previous, _)) = components.last() {
                if previous != ".." {
                    components.pop();
                    return Ok(());
                }
            }

            if absolute {
                return Err(PathError::Escape { at });
            }

            components.push(("..".to_owned(), at));
        }
        _ => components.push((component.to_owned(), at)),
    }

    Ok(())
}
