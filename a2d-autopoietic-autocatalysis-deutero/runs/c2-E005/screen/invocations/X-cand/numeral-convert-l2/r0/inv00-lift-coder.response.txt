#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let payload_start = if let Some(hash) = bytes.iter().position(|&byte| byte == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::with_capacity(payload.len());

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

// Little-endian base-2^32 limbs.
type BigUint = Vec<u32>;

fn multiply_add(value: &mut BigUint, multiplier: u32, addend: u32) {
    let mut carry = addend as u64;
    for limb in value.iter_mut() {
        let product = (*limb as u64) * (multiplier as u64) + carry;
        *limb = product as u32;
        carry = product >> 32;
    }
    if carry != 0 {
        value.push(carry as u32);
    }
}

fn parse_value(digits: &[u8], radix: u8) -> BigUint {
    let mut value = Vec::new();
    for &digit in digits {
        multiply_add(&mut value, radix as u32, digit as u32);
    }
    value
}

fn add_values(left: &BigUint, right: &BigUint) -> BigUint {
    let mut sum = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u64;
    for index in 0..left.len().max(right.len()) {
        let a = left.get(index).copied().unwrap_or(0) as u64;
        let b = right.get(index).copied().unwrap_or(0) as u64;
        let total = a + b + carry;
        sum.push(total as u32);
        carry = total >> 32;
    }
    if carry != 0 {
        sum.push(carry as u32);
    }
    sum
}

fn divide_small(value: &mut BigUint, divisor: u32) -> u32 {
    let mut remainder = 0u64;
    for limb in value.iter_mut().rev() {
        let current = (remainder << 32) | (*limb as u64);
        *limb = (current / divisor as u64) as u32;
        remainder = current % divisor as u64;
    }
    while value.last() == Some(&0) {
        value.pop();
    }
    remainder as u32
}

fn render(value: &BigUint, radix: u8, style: BaseStyle) -> String {
    let mut digits = Vec::new();
    let mut remaining = value.clone();

    if remaining.is_empty() {
        digits.push(b'0');
    } else {
        while !remaining.is_empty() {
            let digit = divide_small(&mut remaining, radix as u32) as u8;
            let byte = match digit {
                0..=9 => b'0' + digit,
                _ if style.upper => b'A' + (digit - 10),
                _ => b'a' + (digit - 10),
            };
            digits.push(byte);
        }
        digits.reverse();
    }

    if digits.len() < style.min_width {
        let padding = style.min_width - digits.len();
        let mut padded = Vec::with_capacity(style.min_width);
        padded.resize(padding, b'0');
        padded.extend(digits);
        digits = padded;
    }

    if style.group == 0 || digits.len() <= style.group {
        return String::from_utf8(digits).expect("ASCII digits");
    }

    let separators = (digits.len() - 1) / style.group;
    let mut output = Vec::with_capacity(digits.len() + separators);
    for (index, byte) in digits.into_iter().enumerate() {
        if index != 0 && (output.len() - output.iter().filter(|&&b| b == b'_').count()) % style.group == 0 {
            output.push(b'_');
        }
        output.push(byte);
    }
    String::from_utf8(output).expect("ASCII formatted digits")
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;
    let digits = parse_operand(input, from, 0)?;
    let value = parse_value(&digits, from);
    Ok(render(&value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;
    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_value = parse_value(&left_digits, base);
    let right_value = parse_value(&right_digits, base);
    Ok(render(&add_values(&left_value, &right_value), base, style))
}
