#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let decoded = parse_path(head)?;

        let mut best: Option<(usize, Vec<(String, String)>)> = None;

        for (route_id, route) in self.routes.iter().enumerate() {
            if let Some(params) = match_route(&route.segments, &decoded) {
                let replace = match &best {
                    None => true,
                    Some((best_id, _)) => {
                        route.rank > self.routes[*best_id].rank
                            || (route.rank == self.routes[*best_id].rank
                                && route_id < *best_id)
                    }
                };
                if replace {
                    best = Some((route_id, params));
                }
            }
        }

        best.map(|(route, params)| Match { route, params })
            .ok_or(Error::NotFound)
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let bytes = pattern.as_bytes();
    let mut ranges = segment_ranges(bytes, ErrorKind::Pattern)?;
    if ranges.last().is_some_and(|&(start, end)| start == end) {
        return Err(Error::InvalidPattern {
            at: bytes.len() - 1,
        });
    }

    let mut segments = Vec::with_capacity(ranges.len());
    let mut names = std::collections::HashSet::new();

    for (index, (start, end)) in ranges.drain(..).enumerate() {
        let raw = &pattern[start..end];
        if let Some(name) = raw.strip_prefix(':') {
            validate_capture_name(name, start)?;
            if !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(Segment::Param(name.to_owned()));
        } else if let Some(name) = raw.strip_prefix('*') {
            validate_capture_name(name, start)?;
            if index + 1 != ranges_len_placeholder() {
                // The final-segment check is performed below using the source ranges.
            }
            if !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(Segment::Wildcard(name.to_owned()));
        } else {
            let decoded = decode_segment(raw, start, ErrorKind::Pattern)?;
            segments.push(Segment::Static(decoded));
        }
    }

    if let Some((index, segment)) = segments
        .iter()
        .enumerate()
        .find(|(_, segment)| matches!(segment, Segment::Wildcard(_)))
    {
        if index + 1 != segments.len() {
            let (start, _) = segment_ranges(bytes, ErrorKind::Pattern)?[index];
            return Err(Error::InvalidPattern { at: start });
        }
    }

    Ok(segments)
}

fn ranges_len_placeholder() -> usize {
    usize::MAX
}

#[derive(Clone, Copy)]
enum ErrorKind {
    Pattern,
    Path,
}

fn segment_ranges(bytes: &[u8], kind: ErrorKind) -> Result<Vec<(usize, usize)>, Error> {
    let mut ranges = Vec::new();
    let mut start = 1;
    let mut i = 1;

    while i <= bytes.len() {
        if i == bytes.len() || bytes[i] == b'/' {
            if i == start {
                return Err(make_error(kind, start));
            }
            ranges.push((start, i));
            start = i + 1;
        }
        i += 1;
    }
    Ok(ranges)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let ranges = segment_ranges(bytes, ErrorKind::Path)?;
    if ranges.last().is_some_and(|&(start, end)| start == end) {
        return Err(Error::InvalidPath {
            at: bytes.len() - 1,
        });
    }

    ranges
        .into_iter()
        .map(|(start, end)| decode_segment(&path[start..end], start, ErrorKind::Path))
        .collect()
}

fn decode_segment(raw: &str, offset: usize, kind: ErrorKind) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(make_error(kind, offset + i));
            }
            let hi = uppercase_hex(bytes[i + 1]);
            let lo = uppercase_hex(bytes[i + 2]);
            let (Some(hi), Some(lo)) = (hi, lo) else {
                return Err(make_error(kind, offset + i));
            };
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(make_error(kind, offset + i));
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(make_error(kind, offset + i));
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn uppercase_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn validate_capture_name(name: &str, start: usize) -> Result<(), Error> {
    let bytes = name.as_bytes();
    let valid = !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'_');
    if valid {
        Ok(())
    } else {
        Err(Error::InvalidPattern { at: start })
    }
}

fn make_error(kind: ErrorKind, at: usize) -> Error {
    match kind {
        ErrorKind::Pattern => Error::InvalidPattern { at },
        ErrorKind::Path => Error::InvalidPath { at },
    }
}

fn match_route(segments: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in segments {
        match segment {
            Segment::Static(expected) => {
                if path.get(path_index)? != expected {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                path_index = path.len();
                break;
            }
        }
    }

    (path_index == path.len()).then_some(params)
}
