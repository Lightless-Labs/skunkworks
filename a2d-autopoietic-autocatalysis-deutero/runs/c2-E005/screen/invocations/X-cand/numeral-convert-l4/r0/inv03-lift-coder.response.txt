#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const MAX_MAGNITUDE: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine", "ten",
    "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen",
    "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(u128, &str); 6] = [
    (1_000_000_000_000_000_000, "quintillion"),
    (1_000_000_000_000_000, "quadrillion"),
    (1_000_000_000_000, "trillion"),
    (1_000_000_000, "billion"),
    (1_000_000, "million"),
    (1_000, "thousand"),
];

fn append_under_hundred(n: u16, out: &mut String) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        let tens = (n / 10) as usize - 2;
        out.push_str(TENS[tens]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(SMALL[(n % 10) as usize]);
        }
    }
}

fn append_group(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            out.push_str(" and ");
            append_under_hundred(remainder, out);
        }
    } else {
        append_under_hundred(n, out);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > MAX_MAGNITUDE {
        return Err(WordsError::Range);
    }

    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups: Vec<(u16, Option<&'static str>)> = Vec::new();
    let mut remaining = magnitude;

    for &(scale, name) in &SCALES {
        let group = (remaining / scale) as u16;
        remaining %= scale;
        if group != 0 {
            groups.push((group, Some(name)));
        }
    }
    if remaining != 0 {
        groups.push((remaining as u16, None));
    }

    let units_is_small = groups
        .last()
        .map(|(n, scale)| scale.is_none() && *n < 100)
        .unwrap_or(false);
    let has_higher = groups.len() > 1;

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    for (i, (group, scale)) in groups.iter().enumerate() {
        if i != 0 {
            if units_is_small && i + 1 == groups.len() {
                out.push_str(" and ");
            } else {
                out.push(' ');
            }
        }
        append_group(*group, &mut out);
        if let Some(name) = scale {
            out.push(' ');
            out.push_str(name);
        }
    }

    let _ = has_higher;
    Ok(out)
}

#[derive(Clone, Copy)]
struct Token<'a> {
    text: &'a str,
    index: usize,
}

fn dictionary_value(word: &str) -> Option<(u128, WordKind)> {
    if word == "zero" {
        return Some((0, WordKind::Number));
    }
    if let Some(i) = SMALL.iter().position(|&w| w == word) {
        if i != 0 {
            return Some((i as u128, WordKind::Number));
        }
    }
    if let Some(i) = TENS.iter().position(|&w| w == word) {
        return Some((((i + 2) * 10) as u128, WordKind::Number));
    }
    let found = match word {
        "hundred" => Some((100, WordKind::Hundred)),
        "thousand" => Some((1_000, WordKind::Scale)),
        "million" => Some((1_000_000, WordKind::Scale)),
        "billion" => Some((1_000_000_000, WordKind::Scale)),
        "trillion" => Some((1_000_000_000_000, WordKind::Scale)),
        "quadrillion" => Some((1_000_000_000_000_000, WordKind::Scale)),
        "quintillion" => Some((1_000_000_000_000_000_000, WordKind::Scale)),
        "and" => Some((0, WordKind::And)),
        _ => None,
    };
    found.map(|(v, k)| (v as u128, k))
}

#[derive(Clone, Copy)]
enum WordKind {
    Number,
    Hundred,
    Scale,
    And,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !byte.is_ascii_lowercase() && byte != b' ' && byte != b'-' {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push(Token {
                text: &input[start..index],
                index: start,
            });
            start = index + 1;
        }
    }
    tokens.push(Token {
        text: &input[start..],
        index: start,
    });

    let mut values = Vec::with_capacity(tokens.len());
    for (i, token) in tokens.iter().enumerate() {
        if token.text == "minus" {
            if i != 0 {
                return Err(WordsError::InvalidWord { index: token.index });
            }
            values.push(None);
        } else if let Some(value) = dictionary_value(token.text) {
            values.push(Some(value));
        } else {
            return Err(WordsError::InvalidWord { index: token.index });
        }
    }

    let negative = tokens.first().map(|t| t.text == "minus").unwrap_or(false);
    let first_arithmetic = if negative { 1 } else { 0 };
    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for item in values.iter().skip(first_arithmetic) {
        let Some((value, kind)) = item else {
            continue;
        };
        match kind {
            WordKind::Number => group = group.checked_add(*value).ok_or(WordsError::Range)?,
            WordKind::Hundred => group = group.checked_mul(100).ok_or(WordsError::Range)?,
            WordKind::Scale => {
                let scaled = group.checked_mul(*value).ok_or(WordsError::Range)?;
                total = total.checked_add(scaled).ok_or(WordsError::Range)?;
                group = 0;
            }
            WordKind::And => {}
        }
        if total.checked_add(group).ok_or(WordsError::Range)? > MAX_MAGNITUDE {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total.checked_add(group).ok_or(WordsError::Range)?;
    if magnitude > MAX_MAGNITUDE {
        return Err(WordsError::Range);
    }

    let signed = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };
    let canonical = words_encode(signed)?;
    if canonical.as_bytes() != input.as_bytes() {
        let limit = canonical.len().min(input.len());
        let index = canonical
            .as_bytes()
            .iter()
            .zip(input.as_bytes())
            .position(|(a, b)| a != b)
            .unwrap_or(limit);
        return Err(WordsError::NonCanonical { index });
    }
    Ok(signed)
}
