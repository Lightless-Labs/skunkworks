#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 10] = [
    "", "", "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 7] = [
    "quintillion",
    "quadrillion",
    "trillion",
    "billion",
    "million",
    "thousand",
    "",
];

fn append_under_thousand(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            out.push_str(" and ");
            append_under_hundred(remainder, out);
        }
    } else {
        append_under_hundred(n, out);
    }
}

fn append_under_hundred(n: u16, out: &mut String) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        out.push_str(TENS[(n / 10) as usize]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(SMALL[(n % 10) as usize]);
        }
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut n = magnitude;
    for group in groups.iter_mut().rev() {
        *group = (n % 1000) as u16;
        n /= 1000;
    }

    let units = groups[6];
    let has_higher = groups[..6].iter().any(|&g| g != 0);
    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let mut emitted = false;
    for i in 0..7 {
        let group = groups[i];
        if group == 0 {
            continue;
        }

        if emitted {
            if i == 6 && units < 100 && has_higher {
                out.push_str(" and ");
            } else {
                out.push(' ');
            }
        }

        append_under_thousand(group, &mut out);
        if !SCALES[i].is_empty() {
            out.push(' ');
            out.push_str(SCALES[i]);
        }
        emitted = true;
    }
    Ok(out)
}

#[derive(Clone, Copy)]
enum TokenKind {
    Minus,
    And,
    Hundred,
    Scale(u128),
    Number(u128),
}

fn classify(word: &str) -> Option<TokenKind> {
    if word == "minus" {
        return Some(TokenKind::Minus);
    }
    if word == "and" {
        return Some(TokenKind::And);
    }
    if word == "hundred" {
        return Some(TokenKind::Hundred);
    }

    match word {
        "thousand" => return Some(TokenKind::Scale(1_000)),
        "million" => return Some(TokenKind::Scale(1_000_000)),
        "billion" => return Some(TokenKind::Scale(1_000_000_000)),
        "trillion" => return Some(TokenKind::Scale(1_000_000_000_000)),
        "quadrillion" => return Some(TokenKind::Scale(1_000_000_000_000_000)),
        "quintillion" => return Some(TokenKind::Scale(1_000_000_000_000_000_000)),
        _ => {}
    }

    for (i, candidate) in SMALL.iter().enumerate() {
        if word == *candidate {
            return Some(TokenKind::Number(i as u128));
        }
    }
    for i in 2..TENS.len() {
        if word == TENS[i] {
            return Some(TokenKind::Number((i * 10) as u128));
        }
    }
    None
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_lowercase() && byte != b' ' && byte != b'-' {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            let before_is_letter = index > 0 && bytes[index - 1].is_ascii_lowercase();
            let after_is_letter = index + 1 < bytes.len() && bytes[index + 1].is_ascii_lowercase();
            if !before_is_letter || !after_is_letter {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(usize, &str, TokenKind)> = Vec::new();
    let mut start = 0;
    for index in 0..=bytes.len() {
        if index == bytes.len() || bytes[index] == b' ' || bytes[index] == b'-' {
            let word = &input[start..index];
            let kind = classify(word).ok_or(WordsError::InvalidWord { index: start })?;
            tokens.push((start, word, kind));
            start = index + 1;
        }
    }

    let negative = matches!(tokens.first().map(|t| t.2), Some(TokenKind::Minus));
    for (i, (index, _, kind)) in tokens.iter().enumerate() {
        if matches!(kind, TokenKind::Minus) && i != 0 {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let mut total: u128 = 0;
    let mut group: u128 = 0;
    for (_, _, kind) in tokens.iter().skip(usize::from(negative)) {
        match kind {
            TokenKind::Minus => {}
            TokenKind::And => {}
            TokenKind::Number(value) => {
                group = group.checked_add(*value).ok_or(WordsError::Range)?;
            }
            TokenKind::Hundred => {
                group = group.checked_mul(100).ok_or(WordsError::Range)?;
            }
            TokenKind::Scale(scale) => {
                let scaled = group.checked_mul(*scale).ok_or(WordsError::Range)?;
                total = total.checked_add(scaled).ok_or(WordsError::Range)?;
                group = 0;
            }
        }
        if total.checked_add(group).ok_or(WordsError::Range)? > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total.checked_add(group).ok_or(WordsError::Range)?;
    let signed = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    let canonical_bytes = canonical.as_bytes();
    let common = bytes.len().min(canonical_bytes.len());
    let first_difference = (0..common)
        .find(|&i| bytes[i] != canonical_bytes[i])
        .unwrap_or(common);

    if bytes != canonical_bytes {
        return Err(WordsError::NonCanonical {
            index: first_difference,
        });
    }

    Ok(signed)
}
