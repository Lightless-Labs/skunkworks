#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanCase {
    Upper,
    Lower,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanError {
    Empty,
    InvalidByte { index: usize },
    MixedCase { index: usize },
    NonCanonical { index: usize },
    Range,
}

fn encode_positive(mut value: u32, case: &RomanCase) -> String {
    const TOKENS: &[(u32, &str)] = &[
        (1000, "M"),
        (900, "CM"),
        (500, "D"),
        (400, "CD"),
        (100, "C"),
        (90, "XC"),
        (50, "L"),
        (40, "XL"),
        (10, "X"),
        (9, "IX"),
        (5, "V"),
        (4, "IV"),
        (1, "I"),
    ];

    let mut result = String::new();
    for &(amount, token) in TOKENS {
        while value >= amount {
            result.push_str(token);
            value -= amount;
        }
    }

    if *case == RomanCase::Lower {
        result.make_ascii_lowercase();
    }
    result
}

pub fn roman_encode(value: u32, case: RomanCase) -> Result<String, RomanError> {
    if value > 3999 {
        return Err(RomanError::Range);
    }
    if value == 0 {
        return Ok(match case {
            RomanCase::Upper => "N".to_string(),
            RomanCase::Lower => "n".to_string(),
        });
    }
    Ok(encode_positive(value, &case))
}

pub fn roman_decode(input: &str) -> Result<u32, RomanError> {
    if input.is_empty() {
        return Err(RomanError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !matches!(byte, b'I' | b'V' | b'X' | b'L' | b'C' | b'D' | b'M' | b'N'
            | b'i' | b'v' | b'x' | b'l' | b'c' | b'd' | b'm' | b'n')
        {
            return Err(RomanError::InvalidByte { index });
        }
    }

    let mut case = None;
    for (index, byte) in input.bytes().enumerate() {
        let current = if byte.is_ascii_uppercase() {
            Some(RomanCase::Upper)
        } else {
            Some(RomanCase::Lower)
        };
        if let Some(first_case) = &case {
            if *first_case != current.unwrap() {
                return Err(RomanError::MixedCase { index });
            }
        } else {
            case = current;
        }
    }

    let bytes = input.as_bytes();
    if bytes.len() == 1 && matches!(bytes[0], b'N' | b'n') {
        return Ok(0);
    }
    if bytes.iter().any(|&byte| matches!(byte, b'N' | b'n')) {
        return Err(RomanError::NonCanonical { index: 0 });
    }

    fn symbol_value(byte: u8) -> u32 {
        match byte.to_ascii_uppercase() {
            b'I' => 1,
            b'V' => 5,
            b'X' => 10,
            b'L' => 50,
            b'C' => 100,
            b'D' => 500,
            b'M' => 1000,
            _ => 0,
        }
    }

    let mut value = 0u32;
    let mut index = 0;
    while index < bytes.len() {
        let current = symbol_value(bytes[index]);
        if index + 1 < bytes.len() {
            let next = symbol_value(bytes[index + 1]);
            if next > current {
                value = value.saturating_add(next - current).min(4000);
                index += 2;
                if value > 3999 {
                    return Err(RomanError::Range);
                }
                continue;
            }
        }
        value = value.saturating_add(current).min(4000);
        index += 1;
        if value > 3999 {
            return Err(RomanError::Range);
        }
    }

    let input_case = case.unwrap();
    let canonical = encode_positive(value, &input_case);
    let canonical_bytes = canonical.as_bytes();
    let common = bytes.len().min(canonical_bytes.len());
    for index in 0..common {
        if bytes[index] != canonical_bytes[index] {
            return Err(RomanError::NonCanonical { index });
        }
    }
    if bytes.len() != canonical_bytes.len() {
        return Err(RomanError::NonCanonical { index: common });
    }

    Ok(value)
}
