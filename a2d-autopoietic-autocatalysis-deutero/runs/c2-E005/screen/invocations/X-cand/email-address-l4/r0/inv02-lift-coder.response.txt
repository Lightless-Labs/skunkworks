#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone)]
struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize, usize)>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while input.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

fn clean_cfws(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0usize;
    let mut quote_mode = false;
    let mut quote_open = 0usize;
    let mut quote_closed = false;

    while i < input.len() {
        let b = input[i];

        if quote_mode {
            bytes.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    bytes.push(input[i]);
                    positions.push(i);
                    i += 1;
                }
                continue;
            }
            if b == b'"' {
                quote_mode = false;
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        if !quote_closed && bytes.is_empty() && b == b'"' {
            quote_mode = true;
            quote_open = i;
            bytes.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        let is_cfws_start = is_hws(b) || b == b'(' || b == b'\r' || b == b'\n';
        if !is_cfws_start {
            if b == b')' {
                return Err(EmailError::UnexpectedClose { at: i });
            }
            bytes.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        let run_start = i;
        let retained_before = bytes.len();
        let mut depth = 0usize;
        let mut outer_open = None;

        loop {
            if i >= input.len() {
                break;
            }
            let c = input[i];

            if depth > 0 {
                match c {
                    b'\\' => {
                        let slash = i;
                        let Some(&next) = input.get(i + 1) else {
                            return Err(EmailError::CommentEscape { at: slash });
                        };
                        if !(0x20..=0x7e).contains(&next) {
                            return Err(EmailError::CommentEscape { at: slash });
                        }
                        i += 2;
                    }
                    b'(' => {
                        if depth == 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        depth += 1;
                        i += 1;
                    }
                    b')' => {
                        depth -= 1;
                        i += 1;
                    }
                    b'\r' | b'\n' => {
                        if let Some(end) = fold_end(input, i) {
                            i = end;
                        } else {
                            return Err(EmailError::BadFold { at: i });
                        }
                    }
                    _ => {
                        if !(0x20..=0x7e).contains(&c) && c != b'\t' {
                            return Err(EmailError::CommentChar { at: i });
                        }
                        i += 1;
                    }
                }
                continue;
            }

            if is_hws(c) {
                i += 1;
            } else if c == b'\r' || c == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                } else {
                    return Err(EmailError::BadFold { at: i });
                }
            } else if c == b'(' {
                outer_open = Some(i);
                depth = 1;
                i += 1;
            } else {
                break;
            }
        }

        let run_end = i;
        let _ = retained_before;
        runs.push((run_start, run_end, outer_open.unwrap_or(run_start)));
    }

    let _ = quote_open;
    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn original_pos(clean: &Cleaned, index: usize, input_len: usize) -> usize {
    clean.positions.get(index).copied().unwrap_or(input_len)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() || bytes.len() > 64 {
        return false;
    }
    let mut previous_dot = true;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if previous_dot || i + 1 == bytes.len() {
                return false;
            }
            previous_dot = true;
        } else if is_atext(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn is_atext(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn parse_cleaned(
    input: &str,
    clean: &Cleaned,
    allow_single_label: bool,
) -> Result<(Address, Vec<usize>), EmailError> {
    let b = &clean.bytes;
    let n = b.len();

    for &(start, end, _) in &clean.runs {
        let first = clean.runs.iter().position(|r| r.0 == start && r.1 == end).unwrap_or(0);
        let _ = first;
        let left_boundary = start == 0;
        let right_boundary = end == input.len();
        let retained_left = clean.positions.iter().any(|&p| p < start);
        let retained_right = clean.positions.iter().any(|&p| p >= end);
        let left_at = retained_left
            && clean.positions.iter().rposition(|&p| p < start)
                .and_then(|idx| b.get(idx))
                == Some(&b'@');
        let right_at = retained_right
            && clean.positions.iter().position(|&p| p >= end)
                .and_then(|idx| b.get(idx))
                == Some(&b'@');
        if !(left_boundary || right_boundary || left_at || right_at) {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }

    if n == 0 {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if b[0] == b'"' {
        let mut i = 1;
        while i < n {
            if b[i] == b'\\' {
                i += 2;
            } else if b[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: original_pos(clean, 0, input.len()),
            });
        }
    }

    let mut ats = Vec::new();
    for i in 0..n {
        if b[i] == b'@' && !quote_end.map(|q| i <= q).unwrap_or(false) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: original_pos(clean, ats[1], input.len()),
        });
    }
    let at = ats[0];
    if at == 0 {
        return Err(EmailError::EmptyLocal {
            at: original_pos(clean, at, input.len()),
        });
    }

    let local_raw = &b[..at];
    let mut local = Vec::new();
    let mut origins = Vec::new();
    let quoted;

    if local_raw[0] == b'"' {
        let qend = quote_end.unwrap_or(0);
        if qend + 1 != at {
            return Err(EmailError::TrailingLocal {
                at: original_pos(clean, qend + 1, input.len()),
            });
        }
        quoted = true;
        let mut i = 1;
        while i < qend {
            let c = local_raw[i];
            if c == b'\\' {
                if i + 1 >= qend {
                    return Err(EmailError::BadEscape {
                        at: original_pos(clean, i, input.len()),
                    });
                }
                let escaped = local_raw[i + 1];
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: original_pos(clean, i, input.len()),
                    });
                }
                local.push(escaped);
                origins.push(original_pos(clean, i, input.len()));
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&c) {
                    return Err(EmailError::QuotedChar {
                        at: original_pos(clean, i, input.len()),
                    });
                }
                local.push(c);
                origins.push(original_pos(clean, i, input.len()));
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: original_pos(clean, qend, input.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
    } else {
        quoted = false;
        if local_raw.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: original_pos(clean, 64, input.len()),
            });
        }
        let mut previous_dot = false;
        for (i, &c) in local_raw.iter().enumerate() {
            if c == b'.' {
                if i == 0 || i + 1 == local_raw.len() || previous_dot {
                    return Err(EmailError::LocalDot {
                        at: original_pos(clean, i, input.len()),
                    });
                }
                previous_dot = true;
            } else {
                if !is_atext(c) {
                    return Err(EmailError::LocalChar {
                        at: original_pos(clean, i, input.len()),
                    });
                }
                previous_dot = false;
            }
            local.push(c);
            origins.push(original_pos(clean, i, input.len()));
        }
    }

    let domain_start = at + 1;
    let mut domain_end = n;
    let mut root_pos = None;
    if domain_end > domain_start && b[domain_end - 1] == b'.' {
        root_pos = Some(original_pos(clean, domain_end - 1, input.len()));
        domain_end -= 1;
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: original_pos(clean, domain_start, input.len()),
        });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: original_pos(clean, domain_start + 253, input.len()),
        });
    }

    let domain = &b[domain_start..domain_end];
    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = 0usize;
    for i in 0..=domain.len() {
        if i == domain.len() || domain[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }

    for (li, &(ls, le)) in labels.iter().enumerate() {
        if ls == le {
            let pos = if li + 1 == labels.len() && root_pos.is_some() {
                root_pos.unwrap()
            } else if ls < domain.len() {
                original_pos(clean, domain_start + ls, input.len())
            } else {
                original_pos(clean, domain_start + ls, input.len())
            };
            return Err(EmailError::EmptyLabel { at: pos });
        }
        let len = le - ls;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: original_pos(clean, domain_start + ls + 63, input.len()),
            });
        }
        if domain[ls] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: original_pos(clean, domain_start + ls, input.len()),
            });
        }
        if domain[le - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: original_pos(clean, domain_start + le - 1, input.len()),
            });
        }
        for i in ls..le {
            if !domain[i].is_ascii_alphanumeric() && domain[i] != b'-' {
                return Err(EmailError::DomainChar {
                    at: original_pos(clean, domain_start + i, input.len()),
                });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: original_pos(clean, domain_start, input.len()),
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if domain[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: original_pos(clean, domain_start + last_start, input.len()),
        });
    }

    local.shrink_to_fit();
    let local_string = String::from_utf8(local).unwrap();
    let domain_string = String::from_utf8(domain.iter().map(u8::to_ascii_lowercase).collect()).unwrap();
    Ok((
        Address {
            local: local_string,
            domain: domain_string,
            quoted,
        },
        origins,
    ))
}

fn parse_with_policy(
    input: &str,
    policy: &Policy,
    strict_domain: bool,
) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let clean = clean_cfws(raw)?;
    let (mut address, origins) = parse_cleaned(input, &clean, strict_domain || !policy.allow_single_label)?;

    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }
    let mut transformed_origins = origins;
    if policy.strip_subaddress {
        if let Some(plus) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: transformed_origins.get(plus).copied().unwrap_or(0),
                });
            }
            address.local.truncate(plus);
        }
    }
    let local_bytes = address.local.as_bytes();
    let safe = valid_dot_atom(local_bytes);
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_policy(input, &Policy::default(), true)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, policy, false)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' {
                out.push('\\');
            }
            out.push(c);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    use std::collections::HashMap;

    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => result.errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = keys.get(&key) {
                    result.duplicate_of[index] = Some(first_index);
                } else {
                    keys.insert(key, index);
                    result.source_indices.push(index);
                    result.addresses.push(address);
                }
            }
        }
    }
    result
}
