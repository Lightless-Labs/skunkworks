#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone)]
struct Cleaned {
    bytes: Vec<u8>,
    origins: Vec<usize>,
    runs: Vec<(usize, usize, usize)>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while input.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut out = Cleaned {
        bytes: Vec::with_capacity(input.len()),
        origins: Vec::with_capacity(input.len()),
        runs: Vec::new(),
    };
    let mut i = 0;
    let mut depth = 0usize;
    let mut opens = [0usize; 3];
    let mut quote_protected = false;
    let mut quote_open = 0usize;
    let mut quote_escaped = false;
    let mut in_run = false;
    let mut run_start = 0usize;
    let mut run_end = 0usize;

    while i < input.len() {
        let b = input[i];

        if quote_protected {
            out.bytes.push(b);
            out.origins.push(i);
            if quote_escaped {
                quote_escaped = false;
            } else if b == b'\\' {
                quote_escaped = true;
            } else if b == b'"' {
                quote_protected = false;
            }
            i += 1;
            continue;
        }

        let mut consumed = i + 1;
        let is_cfws_start;

        if depth > 0 {
            if b == b'\\' {
                let Some(&next) = input.get(i + 1) else {
                    return Err(EmailError::CommentEscape { at: i });
                };
                if !is_printable(next) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                consumed = i + 2;
                is_cfws_start = true;
            } else if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                opens[depth] = i;
                depth += 1;
                is_cfws_start = true;
            } else if b == b')' {
                depth -= 1;
                is_cfws_start = true;
            } else if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    consumed = end;
                    is_cfws_start = true;
                } else {
                    return Err(EmailError::BadFold { at: i });
                }
            } else if b == b'\t' || is_printable(b) {
                is_cfws_start = true;
            } else {
                return Err(EmailError::CommentChar { at: i });
            }
        } else if b == b' ' || b == b'\t' {
            while consumed < input.len() && is_hws(input[consumed]) {
                consumed += 1;
            }
            is_cfws_start = true;
        } else if b == b'\r' || b == b'\n' {
            if let Some(end) = fold_end(input, i) {
                consumed = end;
                is_cfws_start = true;
            } else {
                return Err(EmailError::BadFold { at: i });
            }
        } else if b == b'(' {
            opens[0] = i;
            depth = 1;
            is_cfws_start = true;
        } else if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        } else if b == b'"' && out.bytes.is_empty() {
            quote_protected = true;
            quote_open = i;
            quote_escaped = false;
            out.bytes.push(b);
            out.origins.push(i);
            i += 1;
            continue;
        } else {
            is_cfws_start = false;
        }

        if depth > 0 || is_cfws_start {
            if !in_run {
                in_run = true;
                run_start = i;
            }
            run_end = consumed;
            i = consumed;
        } else {
            if in_run {
                out.runs.push((run_start, run_end, out.bytes.len()));
                in_run = false;
            }
            out.bytes.push(b);
            out.origins.push(i);
            i += 1;
        }
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: opens[0] });
    }
    if in_run {
        out.runs.push((run_start, run_end, out.bytes.len()));
    }
    let _ = quote_open;
    Ok(out)
}

fn local_allowed(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn validate_dot_atom(bytes: &[u8], origins: &[usize]) -> Result<(), EmailError> {
    if bytes.len() > 64 {
        return Err(EmailError::LocalTooLong {
            at: origins.get(64).copied().unwrap_or_else(|| origins.last().copied().unwrap_or(0)),
        });
    }
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at: origins[i] });
            }
        } else if !local_allowed(b) {
            return Err(EmailError::LocalChar { at: origins[i] });
        }
    }
    Ok(())
}

fn parse_cleaned(clean: &Cleaned, input_len: usize, allow_single_label: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let bytes = &clean.bytes;
    let origins = &clean.origins;

    for &(start, _end, boundary) in &clean.runs {
        let legal = boundary == 0
            || boundary == bytes.len()
            || bytes.get(boundary) == Some(&b'@')
            || (boundary > 0 && bytes.get(boundary - 1) == Some(&b'@'));
        if !legal {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i = (i + 2).min(bytes.len());
            } else if bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: origins[0] });
        }
    }

    let mut ats = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i] == b'@' && !quote_end.map(|q| i <= q).unwrap_or(false) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt { at: origins[ats[1]] });
    }
    let at_idx = ats[0];
    if at_idx == 0 {
        return Err(EmailError::EmptyLocal {
            at: origins.get(at_idx).copied().unwrap_or(input_len),
        });
    }

    let local_bytes = &bytes[..at_idx];
    let local_origins = &origins[..at_idx];
    let (mut local, mut quoted, local_positions) = if local_bytes[0] == b'"' {
        let qend = quote_end.unwrap();
        if qend + 1 != at_idx {
            return Err(EmailError::TrailingLocal {
                at: origins.get(qend + 1).copied().unwrap_or(input_len),
            });
        }
        let mut decoded = Vec::new();
        let mut pos = Vec::new();
        let mut i = 1;
        while i < qend {
            let b = local_bytes[i];
            if b == b'\\' {
                if i + 1 >= qend {
                    return Err(EmailError::BadEscape { at: local_origins[i] });
                }
                let next = local_bytes[i + 1];
                if !matches!(next, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: local_origins[i] });
                }
                decoded.push(next);
                pos.push(local_origins[i]);
                i += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar { at: local_origins[i] });
                }
                decoded.push(b);
                pos.push(local_origins[i]);
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: origins[qend],
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: pos[64] });
        }
        (decoded, true, pos)
    } else {
        validate_dot_atom(local_bytes, local_origins)?;
        (local_bytes.to_vec(), false, local_origins.to_vec())
    };

    let domain_start_idx = at_idx + 1;
    let domain_start = origins.get(domain_start_idx).copied().unwrap_or(input_len);
    if domain_start_idx == bytes.len() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let mut domain_end = bytes.len();
    let mut removed_dot_pos = None;
    if bytes[domain_end - 1] == b'.' {
        removed_dot_pos = Some(origins[domain_end - 1]);
        domain_end -= 1;
    }
    if domain_start_idx == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    let dom = &bytes[domain_start_idx..domain_end];
    if dom.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: origins[domain_start_idx + 253],
        });
    }

    let mut label_starts = Vec::new();
    let mut label_ranges = Vec::new();
    let mut start = 0;
    for i in 0..=dom.len() {
        if i == dom.len() || dom[i] == b'.' {
            label_starts.push(start);
            label_ranges.push((start, i));
            start = i + 1;
        }
    }

    for (idx, &(s, e)) in label_ranges.iter().enumerate() {
        if s == e {
            let pos = if idx + 1 == label_ranges.len() && removed_dot_pos.is_some() {
                removed_dot_pos.unwrap()
            } else if s < dom.len() {
                origins[domain_start_idx + s]
            } else {
                input_len
            };
            return Err(EmailError::EmptyLabel { at: pos });
        }
        if e - s > 63 {
            return Err(EmailError::LabelTooLong {
                at: origins[domain_start_idx + s + 63],
            });
        }
        if dom[s] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: origins[domain_start_idx + s],
            });
        }
        if dom[e - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: origins[domain_start_idx + e - 1],
            });
        }
        for j in s..e {
            if !dom[j].is_ascii_alphanumeric() && dom[j] != b'-' {
                return Err(EmailError::DomainChar {
                    at: origins[domain_start_idx + j],
                });
            }
        }
    }

    if label_ranges.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    let (last_s, last_e) = *label_ranges.last().unwrap();
    if dom[last_s..last_e].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: origins[domain_start_idx + last_s],
        });
    }

    let mut domain = String::from_utf8(dom.to_vec()).unwrap();
    domain.make_ascii_lowercase();
    let local_string = String::from_utf8(std::mem::take(&mut local)).unwrap();
    let _ = local_positions;
    Ok((
        Address {
            local: local_string,
            domain,
            quoted,
        },
        Vec::new(),
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    normalize_email(input, &Policy::default())
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|b| *b >= 128) {
        return Err(EmailError::NonAscii { at });
    }
    let cleaned = cleanup(raw)?;
    let (mut address, _) = parse_cleaned(&cleaned, raw.len(), policy.allow_single_label)?;

    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    let local_offset_positions;
    {
        let mut calculated = Vec::with_capacity(address.local.len());
        let mut quoted_end = None;
        if cleaned.bytes.first() == Some(&b'"') {
            let mut i = 1;
            while i < cleaned.bytes.len() {
                if cleaned.bytes[i] == b'\\' {
                    i = (i + 2).min(cleaned.bytes.len());
                } else if cleaned.bytes[i] == b'"' {
                    quoted_end = Some(i);
                    break;
                } else {
                    i += 1;
                }
            }
        }
        let at_pos = cleaned.bytes.iter().position(|b| *b == b'@' && !quoted_end.map(|q| {
            let qi = cleaned.bytes.iter().position(|x| *x == b'"').unwrap_or(0);
            let _ = qi;
            false
        }).unwrap_or(false));
        let _ = at_pos;
        calculated.resize(address.local.len(), 0);
        local_offset_positions = calculated;
    }

    if policy.strip_subaddress {
        if let Some(plus) = address.local.find('+') {
            if plus == 0 {
                let at = local_offset_positions.get(plus).copied().unwrap_or(0);
                return Err(EmailError::EmptyNormalizedLocal { at });
            }
            address.local.truncate(plus);
        }
    }

    let valid_atom = validate_dot_atom(
        address.local.as_bytes(),
        &(0..address.local.len()).collect::<Vec<_>>(),
    )
    .is_ok();
    if !valid_atom {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    use std::collections::HashMap;

    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: HashMap<(String, String), usize> = HashMap::new();

    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => result.errors.push((i, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = keys.get(&key) {
                    result.duplicate_of[i] = Some(first_index);
                } else {
                    keys.insert(key, i);
                    result.addresses.push(address);
                    result.source_indices.push(i);
                }
            }
        }
    }
    result
}
