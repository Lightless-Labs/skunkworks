use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<Run>,
}

fn at_pos(positions: &[usize], index: usize, end: usize) -> usize {
    positions.get(index).copied().unwrap_or(end)
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !matches!(input.get(j), Some(b' ' | b'\t')) {
        return None;
    }
    while matches!(input.get(j), Some(b' ' | b'\t')) {
        j += 1;
    }
    Some(j)
}

fn printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();

    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut comment_stack = Vec::new();
    let mut quote_protected = false;
    let mut quote_closed = false;

    while i < input.len() {
        let b = input[i];

        if quote_protected && !quote_closed {
            bytes.push(b);
            positions.push(i);
            if b == b'\\' {
                if i + 1 < input.len() {
                    bytes.push(input[i + 1]);
                    positions.push(i + 1);
                    i += 2;
                    continue;
                }
            } else if b == b'"' {
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        if depth > 0 {
            if b == b'\\' {
                if i + 1 >= input.len() || !printable(input[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                comment_stack.push(i);
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                comment_stack.pop();
                i += 1;
                continue;
            }
            if printable(b) || b == b'\t' {
                i += 1;
                continue;
            }
            return Err(EmailError::CommentChar { at: i });
        }

        if b == b' ' || b == b'\t' || b == b'\r' || b == b'\n' || b == b'(' {
            let run_start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                let c = input[i];
                if c == b' ' || c == b'\t' {
                    i += 1;
                } else if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if c == b'(' {
                    depth = 1;
                    outer_open = i;
                    comment_stack.push(i);
                    i += 1;
                    while i < input.len() && depth > 0 {
                        let d = input[i];
                        if d == b'\\' {
                            if i + 1 >= input.len() || !printable(input[i + 1]) {
                                return Err(EmailError::CommentEscape { at: i });
                            }
                            i += 2;
                        } else if d == b'\r' || d == b'\n' {
                            if let Some(end) = fold_end(input, i) {
                                i = end;
                            } else {
                                return Err(EmailError::BadFold { at: i });
                            }
                        } else if d == b'(' {
                            if depth == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            depth += 1;
                            comment_stack.push(i);
                            i += 1;
                        } else if d == b')' {
                            depth -= 1;
                            comment_stack.pop();
                            i += 1;
                        } else if printable(d) || d == b'\t' {
                            i += 1;
                        } else {
                            return Err(EmailError::CommentChar { at: i });
                        }
                    }
                    if depth > 0 {
                        return Err(EmailError::UnclosedComment { at: outer_open });
                    }
                } else {
                    break;
                }
            }
            runs.push(Run {
                start: run_start,
                end: i,
            });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if b == b'"' && bytes.is_empty() {
            quote_protected = true;
            quote_closed = false;
        }
        bytes.push(b);
        positions.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn is_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn is_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut prev_dot = true;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if prev_dot || i + 1 == bytes.len() {
                return false;
            }
            prev_dot = true;
        } else if is_atom_char(b) {
            prev_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_core(input: &str, allow_single_label: bool) -> Result<(Address, Vec<usize>), EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    let bytes = &cleaned.bytes;
    let positions = &cleaned.positions;

    for run in &cleaned.runs {
        let left = run.start == 0;
        let right = run.end == raw.len();
        let before_at = bytes.get(run.start.wrapping_sub(1)) == Some(&b'@');
        let after_at = bytes.get(run.start) == Some(&b'@');
        if !(left || right || before_at || after_at) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_close = None;
    if bytes.first() == Some(&b'"') {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quote_close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_close.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: at_pos(positions, 0, raw.len()),
            });
        }
    }

    let mut separators = Vec::new();
    let mut i = 0;
    if let Some(q) = quote_close {
        i = q + 1;
    }
    while i < bytes.len() {
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: at_pos(positions, separators[1], raw.len()),
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: at_pos(positions, sep, raw.len()),
        });
    }

    let local_bytes = &bytes[..sep];
    let mut local_origins = Vec::new();
    let (local, quoted) = if local_bytes[0] == b'"' {
        let close = quote_close.unwrap();
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: at_pos(positions, close + 1, raw.len()),
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < close {
            let b = local_bytes[j];
            if b == b'\\' {
                if j + 1 >= close || !matches!(local_bytes[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: at_pos(positions, j, raw.len()),
                    });
                }
                decoded.push(local_bytes[j + 1]);
                origins.push(at_pos(positions, j, raw.len()));
                j += 2;
            } else {
                if !printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: at_pos(positions, j, raw.len()),
                    });
                }
                decoded.push(b);
                origins.push(at_pos(positions, j, raw.len()));
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: at_pos(positions, close, raw.len()),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
        local_origins = origins;
        (decoded, true)
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: at_pos(positions, 64, raw.len()),
            });
        }
        for (j, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || local_bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: at_pos(positions, j, raw.len()),
                    });
                }
            } else if !is_atom_char(b) {
                return Err(EmailError::LocalChar {
                    at: at_pos(positions, j, raw.len()),
                });
            }
        }
        local_origins.extend((0..local_bytes.len()).map(|j| at_pos(positions, j, raw.len())));
        (local_bytes.to_vec(), false)
    };

    let domain_start = sep + 1;
    let domain_raw = &bytes[domain_start..];
    let mut domain_end = bytes.len();
    if domain_raw.last() == Some(&b'.') {
        domain_end -= 1;
    }
    let domain_bytes = &bytes[domain_start..domain_end];
    if domain_bytes.is_empty() {
        return Err(EmailError::EmptyDomain {
            at: at_pos(positions, domain_start, raw.len()),
        });
    }
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: at_pos(positions, domain_start + 253, raw.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = 0;
    for j in 0..=domain_bytes.len() {
        if j == domain_bytes.len() || domain_bytes[j] == b'.' {
            labels.push((start, j));
            start = j + 1;
        }
    }

    let removed_dot_pos = if domain_end < bytes.len() {
        at_pos(positions, domain_end, raw.len())
    } else {
        raw.len()
    };

    for (index, &(a, b)) in labels.iter().enumerate() {
        if a == b {
            let pos = if index + 1 == labels.len() && domain_end < bytes.len() {
                removed_dot_pos
            } else if a < domain_bytes.len() {
                at_pos(positions, domain_start + a, raw.len())
            } else {
                at_pos(positions, domain_end, raw.len())
            };
            return Err(EmailError::EmptyLabel { at: pos });
        }
        let len = b - a;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: at_pos(positions, domain_start + a + 63, raw.len()),
            });
        }
        if domain_bytes[a] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: at_pos(positions, domain_start + a, raw.len()),
            });
        }
        if domain_bytes[b - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: at_pos(positions, domain_start + b - 1, raw.len()),
            });
        }
        for j in a..b {
            let c = domain_bytes[j];
            if !c.is_ascii_alphanumeric() && c != b'-' {
                return Err(EmailError::DomainChar {
                    at: at_pos(positions, domain_start + j, raw.len()),
                });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: at_pos(positions, domain_start, raw.len()),
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if domain_bytes[last_start..last_end]
        .iter()
        .all(u8::is_ascii_digit)
    {
        return Err(EmailError::NumericTld {
            at: at_pos(positions, domain_start + last_start, raw.len()),
        });
    }

    let local_string = String::from_utf8(local).unwrap();
    let domain_string = String::from_utf8(domain_bytes.to_vec())
        .unwrap()
        .to_ascii_lowercase();
    Ok((
        Address {
            local: local_string,
            domain: domain_string,
            quoted,
        },
        local_origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_core(input, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_core(input, policy.allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }
    if policy.strip_subaddress {
        if let Some(index) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if index == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.first().copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(index);
        }
    }
    let safe = is_dot_atom(address.local.as_bytes());
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = seen.get(&key) {
                    duplicate_of[index] = Some(first_index);
                } else {
                    seen.insert(key, index);
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
