#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_close = if bytes[0] == b'"' {
        let mut cursor = 1;
        let mut close = None;

        while cursor < bytes.len() {
            match bytes[cursor] {
                b'\\' => {
                    cursor += 2;
                }
                b'"' => {
                    close = Some(cursor);
                    break;
                }
                _ => cursor += 1,
            }
        }

        match close {
            Some(index) => Some(index),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut cursor = 0;
    while cursor < bytes.len() {
        if let Some(close) = quoted_close {
            if cursor <= close {
                cursor = close + 1;
                continue;
            }
        }

        if bytes[cursor] == b'@' {
            separators.push(cursor);
            if separators.len() == 2 {
                return Err(EmailError::MultipleAt { at: cursor });
            }
        }
        cursor += 1;
    }

    let separator = match separators.first() {
        Some(&index) => index,
        None => return Err(EmailError::MissingAt),
    };

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let mut local = String::new();
    let is_quoted = bytes[0] == b'"';

    if is_quoted {
        let close = quoted_close.expect("quoted prefix must have a closing quote");
        if close + 1 != separator {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }

        let mut origins = Vec::new();
        let mut index = 1;
        while index < close {
            let byte = bytes[index];
            if byte == b'\\' {
                let escaped = bytes[index + 1];
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: index });
                }
                local.push(escaped as char);
                origins.push(index);
                index += 2;
            } else {
                if !(0x20..=0x7e).contains(&byte) {
                    return Err(EmailError::QuotedChar { at: index });
                }
                local.push(byte as char);
                origins.push(index);
                index += 1;
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
    } else {
        let local_bytes = &bytes[..separator];

        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for (index, &byte) in local_bytes.iter().enumerate() {
            if byte == b'.' {
                if index == 0
                    || index + 1 == local_bytes.len()
                    || local_bytes[index - 1] == b'.'
                {
                    return Err(EmailError::LocalDot { at: index });
                }
            } else if !byte.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&byte)
            {
                return Err(EmailError::LocalChar { at: index });
            }
        }

        local = String::from_utf8(local_bytes.to_vec()).expect("ASCII input");
    }

    let domain_start = separator + 1;
    let mut domain_end = bytes.len();
    let mut removed_root_dot = None;

    if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        removed_root_dot = Some(domain_end);
    }

    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut label_count = 0usize;
    let mut label_start = domain_start;

    for position in domain_start..=domain_end {
        if position == domain_end || bytes[position] == b'.' {
            if label_start == position {
                let at = if position == domain_end {
                    removed_root_dot.unwrap_or(position)
                } else {
                    position
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let label = &bytes[label_start..position];
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: label_start + 63,
                });
            }

            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: label_start });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: position - 1,
                });
            }

            for (offset, &byte) in label.iter().enumerate() {
                if !byte.is_ascii_alphanumeric() && byte != b'-' {
                    return Err(EmailError::DomainChar {
                        at: label_start + offset,
                    });
                }
            }

            label_count += 1;
            label_start = position + 1;
        }
    }

    if label_count < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let last_label_start = bytes[domain_start..domain_end]
        .iter()
        .rposition(|&byte| byte == b'.')
        .map_or(domain_start, |index| domain_start + index + 1);
    let last_label = &bytes[last_label_start..domain_end];

    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: last_label_start,
        });
    }

    let domain = bytes[domain_start..domain_end]
        .iter()
        .map(|&byte| byte.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted: is_quoted,
    })
}
