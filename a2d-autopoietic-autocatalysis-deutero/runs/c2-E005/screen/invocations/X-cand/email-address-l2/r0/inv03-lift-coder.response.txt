#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_end = if bytes[0] == b'"' {
        let mut i = 1;
        let mut found = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    found = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match found {
            Some(end) => Some(end),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quoted_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let (local, quoted) = if let Some(end) = quoted_end {
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal { at: end + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut pos = 1;
        while pos < end {
            if bytes[pos] == b'\\' {
                let next = bytes[pos + 1];
                match next {
                    b'"' | b'\\' | b'@' => {
                        decoded.push(next);
                        origins.push(pos);
                        pos += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: pos }),
                }
            } else {
                let b = bytes[pos];
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: pos });
                }
                decoded.push(b);
                origins.push(pos);
                pos += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: end });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }

        (String::from_utf8(decoded).unwrap(), true)
    } else {
        let len = sep;
        if len > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for i in 0..len {
            let b = bytes[i];
            if b == b'.' {
                if i == 0 || i + 1 == len || bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: i });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: i });
            }
        }

        (input[..sep].to_string(), false)
    };

    let domain_start = sep + 1;
    let mut domain_end = bytes.len();
    let root_dot = if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut label_starts = Vec::new();
    let mut label_ends = Vec::new();
    let mut start = domain_start;

    for pos in domain_start..=domain_end {
        if pos == domain_end || bytes[pos] == b'.' {
            if start == pos {
                let at = if pos == domain_end {
                    root_dot.unwrap_or(start)
                } else {
                    pos
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let len = pos - start;
            if len > 63 {
                return Err(EmailError::LabelTooLong { at: start + 63 });
            }
            if bytes[start] == b'-' {
                return Err(EmailError::LabelHyphen { at: start });
            }
            if bytes[pos - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos - 1 });
            }
            for idx in start..pos {
                let b = bytes[idx];
                if !b.is_ascii_alphanumeric() && b != b'-' {
                    return Err(EmailError::DomainChar { at: idx });
                }
            }

            label_starts.push(start);
            label_ends.push(pos);
            start = pos + 1;
        }
    }

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let last_start = *label_starts.last().unwrap();
    let last_end = *label_ends.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let domain = input[domain_start..domain_end].to_ascii_lowercase();
    Ok(Address {
        local,
        domain,
        quoted,
    })
}
