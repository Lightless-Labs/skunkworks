#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<(String, usize)> {
    path.split('/')
        .scan(0usize, |offset, part| {
            let start = *offset;
            *offset += part.len() + 1;
            Some((part, start))
        })
        .filter(|(part, _)| !part.is_empty())
        .map(|(part, at)| (part.to_string(), at))
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|(part, _)| part == "." || part == "..")
}

fn render(parts: &[String], absolute: bool, intent: bool) -> String {
    let mut out = if absolute {
        format!("/{}", parts.join("/"))
    } else if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    };
    if absolute && parts.is_empty() {
        return "/".to_string();
    }
    if !absolute && parts.is_empty() {
        return ".".to_string();
    }
    if intent {
        out.push('/');
    }
    out
}

fn normalize_parts(path: &str) -> Result<(Vec<String>, bool, bool), PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let mut stack: Vec<String> = Vec::new();
    for (part, at) in components(path) {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|p| p != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(part);
                }
            }
            _ => stack.push(part),
        }
    }
    Ok((stack, absolute, intent))
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    let (parts, absolute, intent) = normalize_parts(path)?;
    Ok(render(&parts, absolute, intent))
}

fn context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }
    let (parts, absolute, _) = normalize_parts(path).map_err(|_| error)?;
    if !absolute {
        return Err(error);
    }
    Ok(parts)
}

fn tilde_remainder<'a>(path: &'a str, home: Option<&[String]>) -> Result<(bool, Vec<String>, &'a str, usize), PathError> {
    if !path.starts_with('~') {
        return Ok((false, Vec::new(), path, 0));
    }
    let bytes = path.as_bytes();
    if bytes.len() > 1 && bytes[1] != b'/' {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    let home = home.ok_or(PathError::MissingHome)?.to_vec();
    if bytes.len() == 1 {
        return Ok((true, home, "", 1));
    }
    Ok((true, home, &path[2..], 2))
}

fn resolve_core(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: Option<(&[Link], usize)>,
) -> Result<String, PathError> {
    let cwd_parts = context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(h) => Some(context(h, PathError::InvalidHome)?),
        None => None,
    };
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (expanded, tilde_home, remainder, consumed) =
        tilde_remainder(path, home_parts.as_deref())?;
    let absolute = !expanded && path.starts_with('/');
    let mut stack = if absolute {
        Vec::new()
    } else if expanded {
        tilde_home
    } else {
        cwd_parts
    };

    if let Some((table, limit)) = links {
        let mut active = Vec::new();
        let mut count = 0usize;
        let mut queue: Vec<(String, usize)> = if expanded {
            components(remainder)
                .into_iter()
                .map(|(p, at)| (p, at + consumed))
                .collect()
        } else {
            components(path)
        };
        let mut cursor = 0usize;
        while cursor < queue.len() {
            let (part, at) = queue[cursor].clone();
            cursor += 1;
            match part.as_str() {
                "." => continue,
                ".." => {
                    if stack.pop().is_none() {
                        return Err(PathError::Escape { at });
                    }
                    continue;
                }
                _ => stack.push(part),
            }
            let key = format!("/{}", stack.join("/"));
            if let Some(index) = table.iter().position(|link| link.path == key) {
                if active.contains(&index) {
                    return Err(PathError::Loop { index });
                }
                if count >= limit {
                    return Err(PathError::Limit { index });
                }
                count += 1;
                let link_parent_len = stack.len().saturating_sub(1);
                let target = &table[index].target;
                let target_abs = target.starts_with('/');
                if target_abs {
                    stack.clear();
                } else {
                    stack.truncate(link_parent_len);
                }
                active.push(index);
                let mut target_parts: Vec<(String, usize)> = components(target);
                target_parts.extend(queue[cursor..].iter().cloned());
                queue.truncate(cursor);
                queue.extend(target_parts);
                // Active entries are conservatively released when their inserted target
                // work is exhausted; nested lookups retain their own entries.
                let mut remaining = queue.len() - cursor;
                while remaining > 0 {
                    remaining -= 1;
                    let (tp, ta) = queue[cursor].clone();
                    cursor += 1;
                    match tp.as_str() {
                        "." => {}
                        ".." => {
                            if stack.pop().is_none() {
                                return Err(PathError::Escape { at: ta });
                            }
                        }
                        _ => {
                            stack.push(tp);
                            let k = format!("/{}", stack.join("/"));
                            if let Some(nested) = table.iter().position(|l| l.path == k) {
                                if active.contains(&nested) {
                                    return Err(PathError::Loop { index: nested });
                                }
                                if count >= limit {
                                    return Err(PathError::Limit { index: nested });
                                }
                                count += 1;
                                let nlink = &table[nested];
                                let parent_len = stack.len().saturating_sub(1);
                                if nlink.target.starts_with('/') {
                                    stack.clear();
                                } else {
                                    stack.truncate(parent_len);
                                }
                                active.push(nested);
                                let mut nested_parts = components(&nlink.target);
                                nested_parts.extend(queue[cursor..].iter().cloned());
                                queue.truncate(cursor);
                                queue.extend(nested_parts);
                                remaining = queue.len() - cursor;
                            }
                        }
                    }
                }
                active.clear();
                break;
            }
        }
        // Process any request components left after the first expansion.
        for (part, at) in queue.into_iter().skip(cursor) {
            match part.as_str() {
                "." => {}
                ".." => {
                    if stack.pop().is_none() {
                        return Err(PathError::Escape { at });
                    }
                }
                _ => stack.push(part),
            }
        }
    } else {
        for (part, at) in if expanded {
            components(remainder)
                .into_iter()
                .map(|(p, a)| (p, a + consumed))
                .collect()
        } else {
            components(path)
        } {
            match part.as_str() {
                "." => {}
                ".." => {
                    if stack.pop().is_none() {
                        return Err(PathError::Escape { at });
                    }
                }
                _ => stack.push(part),
            }
        }
    }
    Ok(render(&stack, true, intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    resolve_core(path, cwd, home, None)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    context(cwd, PathError::InvalidCwd)?;
    if let Some(h) = home {
        context(h, PathError::InvalidHome)?;
    }
    for (i, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path).is_ok_and(|p| p == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index: i });
        }
        if let Some(first) = links[..i].iter().position(|prior| prior.path == link.path) {
            return Err(PathError::DuplicateLink { first, second: i });
        }
    }
    resolve_core(path, cwd, home, Some((links, max_expansions)))
}

fn seq(path: &str) -> Vec<String> {
    components(path).into_iter().map(|(p, _)| p).collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base.iter().zip(target).take_while(|(a, b)| a == b).count();
    let mut parts = vec!["..".to_string(); base.len() - common];
    parts.extend_from_slice(&target[common..]);
    if parts.is_empty() { ".".into() } else { parts.join("/") }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let b = seq(&resolved_base);
    let t = seq(&resolved_target);
    let target_dir = resolved_target.ends_with('/');
    let mut candidates = vec![route(&b, &t)];
    for link in links {
        if let Ok(probe) = resolve_links(&link.path, "/", None, links, max_expansions) {
            let a = seq(&probe);
            if a.len() <= t.len() && a.iter().zip(&t).all(|(x, y)| x == y) {
                let suffix = &t[a.len()..];
                let mut c = route(&b, &seq(&link.path));
                if !suffix.is_empty() {
                    if c == "." { c = suffix.join("/"); }
                    else { c.push('/'); c.push_str(&suffix.join("/")); }
                }
                candidates.push(c);
            }
        }
    }
    let mut valid = Vec::new();
    for mut c in candidates {
        if c.starts_with('~') {
            c = format!("./{c}");
        }
        if target_dir && c != "." {
            c.push('/');
        }
        if let Ok(result) = resolve_links(&c, &resolved_base, home, links, max_expansions) {
            if seq(&result) == t {
                valid.push(c);
            }
        }
    }
    valid.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())));
    valid.into_iter().next().ok_or(PathError::Empty)
}
