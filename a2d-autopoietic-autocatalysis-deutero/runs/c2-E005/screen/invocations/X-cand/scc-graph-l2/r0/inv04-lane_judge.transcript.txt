            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: to,

                    stack.push(next);
                }

        components.push(component);

    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }

    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: edge_index,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: edge_index,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse_adjacency = vec![Vec::new(); n];

    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse_adjacency[to].push(from);
    }

    // Iterative Kosaraju pass: compute finishing order without recursion.
    let mut visited = vec![false; n];
    let mut finish_order = Vec::with_capacity(n);

    for start in 0..n {
        if visited[start] {
            continue;
        }

        visited[start] = true;
        let mut stack = vec![(start, 0usize)];

        while let Some((vertex, next_neighbor)) = stack.last_mut() {
            if *next_neighbor < adjacency[*vertex].len() {
                let neighbor = adjacency[*vertex][*next_neighbor];
                *next_neighbor += 1;

                if !visited[neighbor] {
                    visited[neighbor] = true;
                    stack.push((neighbor, 0));
                }
            } else {
                let (finished, _) = stack.pop().unwrap();
                finish_order.push(finished);
            }
        }
    }

    let mut assigned = vec![false; n];
    let mut components = Vec::new();

    for &start in finish_order.iter().rev() {
        if assigned[start] {
            continue;
        }

        assigned[start] = true;
        let mut component = Vec::new();
        let mut stack = vec![start];

        while let Some(vertex) = stack.pop() {
            component.push(vertex);
            for &neighbor in &reverse_adjacency[vertex] {
                if !assigned[neighbor] {
                    assigned[neighbor] = true;
                    stack.push(neighbor);
                }
            }
        }

        component.sort_unstable();
        components.push(component);
    }

    components.sort_by(|a, b| {
        b.len()
            .cmp(&a.len())
            .then_with(|| a[0].cmp(&b[0]))
    });

    let mut component_of = vec![0usize; n];
    for (component_id, component) in components.iter().enumerate() {
        for &vertex in component {
            component_of[vertex] = component_id;
        }
    }

    let component_count = components.len();
    let mut cyclic = components
        .iter()
        .map(|component| component.len() >= 2)
        .collect::<Vec<_>>();
    let mut internal_edges = vec![0usize; component_count];
    let mut arc_stats = std::collections::BTreeMap::<(usize, usize), (usize, usize)>::new();

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let from_component = component_of[from];
        let to_component = component_of[to];

        if from_component == to_component {
            internal_edges[from_component] += 1;
            if from == to {
                cyclic[from_component] = true;
            }
        } else {
            let stats = arc_stats
                .entry((from_component, to_component))
                .or_insert((0, edge_index));
            stats.0 += 1;
            stats.1 = stats.1.min(edge_index);
        }
    }

    let mut arcs = arc_stats
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect::<Vec<_>>();

    arcs.sort_by(|a, b| a.from.cmp(&b.from).then_with(|| b.to.cmp(&a.to)));

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
    })
}
