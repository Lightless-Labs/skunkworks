fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = None;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if let Some(end) = previous_end {
            if span.start < end {
                return Err(SpanError::Overlap { span: index });
            }
        }
        previous_end = Some(span.end);
    }
    Ok(())
}

fn hex_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'f' => Some((byte - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse_hex(bytes: &[u8]) -> u32 {
    bytes
        .iter()
        .fold(0u32, |value, &byte| (value << 4) | hex_value(byte).unwrap())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let (mut a, mut b) = (0u32, 0u32);
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut f = Fletcher32::new();
    for &byte in data {
        f.push(byte);
    }
    f.finish()
}

struct Fletcher32 {
    a: u32,
    b: u32,
    pending: Option<u8>,
}

impl Fletcher32 {
    fn new() -> Self {
        Self {
            a: 0,
            b: 0,
            pending: None,
        }
    }

    fn word(&mut self, word: u16) {
        self.a = (self.a + word as u32) % 65_535;
        self.b = (self.b + self.a) % 65_535;
    }

    fn push(&mut self, byte: u8) {
        if let Some(high) = self.pending.take() {
            self.word(((high as u16) << 8) | byte as u16);
        } else {
            self.pending = Some(byte);
        }
    }

    fn finish(mut self) -> u32 {
        if let Some(high) = self.pending.take() {
            self.word((high as u16) << 8);
        }
        (self.b << 16) | self.a
    }
}

    for span in spans {

            f.push(byte);
        }
    }
    Ok(f.finish())
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut crc = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = crc & 0x8000;
                    crc = crc.wrapping_shl(1);
                    if top != 0 {
                        crc ^= 0x1021;
                    }
                }
            }
            crc
        }
        Crc16::X25 => {
            for &byte in data {
                crc ^= byte as u16;
                for _ in 0..8 {
                    let low = crc & 1;
                    crc >>= 1;
                    if low != 0 {
                        crc ^= 0x8408;
                    }
                }
            }
            crc ^ 0xffff
        }
    }
}

    crc16: u16,
    variant: Crc16,
}

impl MultiState {
    fn new(variant: Crc16) -> Self {
        Self {
            base: State::empty(),
            f16a: 0,
            f16b: 0,
            f32: Fletcher32::new(),
            crc16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
            variant,
        }
    }

    fn update(&mut self, byte: u8) {
        self.base.update(byte);
        self.f16a = (self.f16a + byte as u32) % 255;
        self.f16b = (self.f16b + self.f16a) % 255;
        self.f32.push(byte);

        match self.variant {
            Crc16::Xmodem | Crc16::False => {
                self.crc16 ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = self.crc16 & 0x8000;
                    self.crc16 = self.crc16.wrapping_shl(1);
                    if top != 0 {
                        self.crc16 ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                self.crc16 ^= byte as u16;
                for _ in 0..8 {
                    let low = self.crc16 & 1;
                    self.crc16 >>= 1;
                    if low != 0 {
                        self.crc16 ^= 0x8408;
                    }
                }
            }
        }
    }

    fn finish(self) -> (Hashes, u16, u32, u16) {
        let crc16 = if self.variant == Crc16::X25 {
            self.crc16 ^ 0xffff
        } else {
            self.crc16
        };
        (
            self.base.checksum(),
            ((self.f16b as u16) << 8) | self.f16a as u16,
            self.f32.finish(),
            crc16,
        )
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = MultiState::new(variant);
    for span in spans {
        for &byte in &data[span.start..span.end] {
            state.update(byte);
        }
    }
    let (base, fletcher16, fletcher32, crc16) = state.finish();
    Ok(Multi {
        base,
        fletcher16,
        fletcher32,
        crc16,
    })
}

fn zero_byte_crc(mut register: u32) -> u32 {
    for _ in 0..8 {
        let low = register & 1;
        register >>= 1;
        if low != 0 {
            register ^= 0xedb8_8320;
        }
    }
    register
}

fn matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn matrix_square(matrix: &[u32; 32]) -> [u32; 32] {
    let mut square = [0u32; 32];
    for n in 0..32 {
        square[n] = matrix_times(matrix, matrix[n]);
    }
    square
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut matrix = [0u32; 32];
    for bit in 0..32 {
        matrix[bit] = zero_byte_crc(1u32 << bit);
    }

    let mut len = right_len;
    let mut value = left;
    while len != 0 {
        if len & 1 != 0 {
            value = matrix_times(&matrix, value);
        }
        len >>= 1;
        if len != 0 {
            matrix = matrix_square(&matrix);
        }
    }
    value ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part {
        crc32: crc,
        len: total,
    })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut length = 0u64;
    for span in spans {
        let span_len = (span.end - span.start) as u64;
        length += span_len;
    }

    Ok(format!(

    ))
}

fn parse_hex_u64(bytes: &[u8]) -> u64 {
    bytes.iter().fold(0u64, |value, &byte| {
        (value << 4) | hex_value(byte).unwrap() as u64
    })
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for offset in (4..20).chain(25..33).chain(40..48) {
        if hex_value(bytes[offset]).is_none() {
            return Err(VerifyError::Hex { offset });
        }
    }

    let expected_len = parse_hex_u64(&bytes[4..20]);
    let expected_crc = parse_hex(&bytes[25..33]);
    let expected_adler = parse_hex(&bytes[40..48]);

    let mut length = 0u64;
    for span in spans {
        length += (span.end - span.start) as u64;
    }
    if expected_len != length {
        return Err(VerifyError::Length);
    }

    let hashes = selected_state(data, spans).checksum();
    if expected_crc != hashes.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: hashes.crc32,
        });
    }
    if expected_adler != hashes.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: hashes.adler32,
        });
    }
    Ok(())
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,

                span: index,
                offset: span.end,

        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;

    Ok(())

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut chunks = data.chunks_exact(2);
    for chunk in &mut chunks {
        fletcher32_word(&mut a, &mut b, u16::from_be_bytes([chunk[0], chunk[1]]));
    }
    if let Some(&last) = chunks.remainder().first() {
        fletcher32_word(&mut a, &mut b, (last as u16) << 8);
    }
    (b << 16) | a
}

fn crc16_update(mut reg: u16, data: &[u8], variant: Crc16) -> u16 {

        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                reg ^= (byte as u16) << 8;
                for _ in 0..8 {
                    if reg & 0x8000 != 0 {
                        reg = (reg << 1) ^ 0x1021;
                    } else {
                        reg <<= 1;
                    }
                }
            }
            reg
        }
        Crc16::X25 => {
            for &byte in data {
                reg ^= byte as u16;
                for _ in 0..8 {
                    if reg & 1 != 0 {
                        reg = (reg >> 1) ^ 0x8408;
                    } else {
                        reg >>= 1;
                    }
                }
            }
            reg
        }
    }
}

    }

    fn new(variant: Crc16) -> Self {

            crc: 0xffff_ffff,
            a: 1,
            b: 0,
            f16a: 0,
            f16b: 0,
            f32a: 0,
            f32b: 0,
            f32_pending: None,
            c16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
        }
    }

    fn update(&mut self, data: &[u8], variant: Crc16) {
        append_base_state(&mut self.crc, &mut self.a, &mut self.b, data);
        for &byte in data {
            fletcher16_update(&mut self.f16a, &mut self.f16b, byte);
            if let Some(high) = self.f32_pending.take() {
                fletcher32_word(&mut self.f32a, &mut self.f32b, ((high as u16) << 8) | byte as u16);
            } else {
                self.f32_pending = Some(byte);
            }
        }
        self.c16 = crc16_update(self.c16, data, variant);
    }

    fn finish(self, variant: Crc16) -> Multi {
        let mut f32a = self.f32a;
        let mut f32b = self.f32b;
        if let Some(high) = self.f32_pending {
            fletcher32_word(&mut f32a, &mut f32b, (high as u16) << 8);
        }
        Multi {
            base: Hashes {
                crc32: self.crc ^ 0xffff_ffff,
                adler32: (self.b << 16) | self.a,
            },
            fletcher16: ((self.f16b as u16) << 8) | self.f16a as u16,
            fletcher32: (f32b << 16) | f32a,
            crc16: if variant == Crc16::X25 {
                self.c16 ^ 0xffff
            } else {
                self.c16
            },
        }
    }
}

fn zero_byte_operator() -> [u32; 32] {
    let mut matrix = [0u32; 32];
    for bit in 0..32 {
        let mut value = 1u32 << bit;
        for _ in 0..8 {
            let low = value & 1;
            value >>= 1;
            if low != 0 {
                value ^= 0xedb8_8320;
            }
        }
        matrix[bit] = value;
    }
    matrix
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }
    let mut operator = zero_byte_operator();
    let mut squared = [0u32; 32];
    let mut result = left;
    let mut length = right_len;
    while length != 0 {
        if length & 1 != 0 {
            result = gf2_matrix_times(&operator, result);
        }
        length >>= 1;
        if length != 0 {
            gf2_matrix_square(&mut squared, &operator);
            operator = squared;
        }
    }
    result ^ right
}

#[derive(Clone, Copy)]
struct Accumulator {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
    fletcher16_a: u32,
    fletcher16_b: u32,
    fletcher32_a: u32,
    fletcher32_b: u32,
    fletcher32_high: Option<u8>,
    crc16: u16,
    variant: Crc16,
    len: u64,
}

impl Accumulator {
    fn new(variant: Crc16) -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
            fletcher16_a: 0,
            fletcher16_b: 0,
            fletcher32_a: 0,
            fletcher32_b: 0,
            fletcher32_high: None,
            crc16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
            variant,
            len: 0,
        }
    }

    fn from_stream(stream: &Stream, variant: Crc16) -> Self {
        Self {
            crc: stream.crc,
            adler_a: stream.adler_a,
            adler_b: stream.adler_b,
            fletcher16_a: 0,
            fletcher16_b: 0,
            fletcher32_a: 0,
            fletcher32_b: 0,
            fletcher32_high: None,
            crc16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
            variant,
            len: 0,
        }
    }

    fn push(&mut self, byte: u8) {
        self.crc ^= byte as u32;
        for _ in 0..8 {
            let low = self.crc & 1;
            self.crc >>= 1;
            if low != 0 {
                self.crc ^= 0xedb8_8320;
            }
        }

        const ADLER_MOD: u32 = 65_521;
        self.adler_a = (self.adler_a + byte as u32) % ADLER_MOD;
        self.adler_b = (self.adler_b + self.adler_a) % ADLER_MOD;

        self.fletcher16_a = (self.fletcher16_a + byte as u32) % 255;
        self.fletcher16_b = (self.fletcher16_b + self.fletcher16_a) % 255;

        if let Some(high) = self.fletcher32_high.take() {
            let word = ((high as u32) << 8) | byte as u32;
            self.push_fletcher32_word(word);
        } else {
            self.fletcher32_high = Some(byte);
        }

        match self.variant {
            Crc16::Xmodem | Crc16::False => {
                self.crc16 ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let high = self.crc16 & 0x8000;
                    self.crc16 = self.crc16.wrapping_shl(1);
                    if high != 0 {
                        self.crc16 ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                self.crc16 ^= byte as u16;
                for _ in 0..8 {
                    let low = self.crc16 & 1;
                    self.crc16 >>= 1;
                    if low != 0 {
                        self.crc16 ^= 0x8408;
                    }
                }
            }
        }

        self.len += 1;
    }

    fn push_fletcher32_word(&mut self, word: u32) {
        const MOD: u32 = 65_535;
        self.fletcher32_a = (self.fletcher32_a + word) % MOD;
        self.fletcher32_b = (self.fletcher32_b + self.fletcher32_a) % MOD;
    }

    fn hashes(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    fn fletcher32(&self) -> u32 {
        let (a, b) = if let Some(high) = self.fletcher32_high {
            let word = (high as u32) << 8;
            const MOD: u32 = 65_535;
            let a = (self.fletcher32_a + word) % MOD;
            let b = (self.fletcher32_b + a) % MOD;
            (a, b)
        } else {
            (self.fletcher32_a, self.fletcher32_b)
        };
        (b << 16) | a
    }

    fn crc16(&self) -> u16 {
        match self.variant {
            Crc16::Xmodem | Crc16::False => self.crc16,
            Crc16::X25 => self.crc16 ^ 0xffff,
        }
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

fn accumulate_spans(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Accumulator, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new(variant);
    for span in spans {
        for &byte in &data[span.start..span.end] {
            acc.push(byte);
        }
    }
    Ok(acc)
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut register = 0xffff_ffffu32;
    for &byte in data {
        register ^= byte as u32;
        for _ in 0..8 {
            let low = register & 1;
            register >>= 1;
            if low != 0 {
                register ^= 0xedb8_8320;
            }
        }
    }
    register ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    const MOD: u32 = 65_521;
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % MOD;
        b = (b + a) % MOD;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    Ok(accumulate_spans(data, spans, Crc16::Xmodem)?.hashes())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut iter = data.chunks_exact(2);
    for pair in &mut iter {
        let word = ((pair[0] as u32) << 8) | pair[1] as u32;
        a = (a + word) % 65_535;
        b = (b + a) % 65_535;
    }
    if let Some(&byte) = iter.remainder().first() {
        a = (a + ((byte as u32) << 8)) % 65_535;
        b = (b + a) % 65_535;
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    Ok(accumulate_spans(data, spans, Crc16::Xmodem)?.fletcher32())
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut register = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                register ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let high = register & 0x8000;
                    register = register.wrapping_shl(1);
                    if high != 0 {
                        register ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                register ^= byte as u16;
                for _ in 0..8 {
                    let low = register & 1;
                    register >>= 1;
                    if low != 0 {
                        register ^= 0x8408;
                    }
                }
            }
        }
    }
    match variant {
        Crc16::Xmodem | Crc16::False => register,
        Crc16::X25 => register ^ 0xffff,
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    let acc = accumulate_spans(data, spans, variant)?;
    Ok(Multi {
        base: acc.hashes(),
        fletcher16: ((acc.fletcher16_b << 8) | acc.fletcher16_a) as u16,
        fletcher32: acc.fletcher32(),
        crc16: acc.crc16(),
    })
}

fn matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = matrix_times(matrix, matrix[index]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }

    let mut even = [0u32; 32];
    matrix_square(&mut even, &odd);
    matrix_square(&mut odd, &even);

    let mut len = right_len;
    let mut shifted = left;
    loop {
        matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            shifted = matrix_times(&even, shifted);
        }
        len >>= 1;
        if len == 0 {
            break;
        }

        matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            shifted = matrix_times(&odd, shifted);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }
    shifted ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut combined = 0u32;
    for part in parts {
        combined = crc32_combine(combined, part.crc32, part.len);
    }
    Ok(Part {
        crc32: combined,
        len: total,
    })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    let acc = accumulate_spans(data, spans, Crc16::Xmodem)?;
    let hashes = acc.hashes();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        acc.len, hashes.crc32, hashes.adler32
    ))
}

fn parse_hex(bytes: &[u8], start: usize, end: usize) -> Result<u64, VerifyError> {
    let mut value = 0u64;
    for offset in start..end {
        let digit = match bytes[offset] {
            b'0'..=b'9' => (bytes[offset] - b'0') as u64,
            b'a'..=b'f' => (bytes[offset] - b'a' + 10) as u64,
            _ => return Err(VerifyError::Hex { offset }),
        };
        value = (value << 4) | digit;
    }
    Ok(value)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    let acc = accumulate_spans(data, spans, Crc16::Xmodem).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();

    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let length = parse_hex(bytes, 4, 20)?;
    let expected_crc = parse_hex(bytes, 25, 33)? as u32;
    let expected_adler = parse_hex(bytes, 40, 48)? as u32;

    if length != acc.len {
        return Err(VerifyError::Length);
    }

    let actual = acc.hashes();
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        let mut acc = Accumulator::from_stream(self, Crc16::Xmodem);
        for &byte in data {
            acc.push(byte);
        }
        self.crc = acc.crc;
        self.adler_a = acc.adler_a;
        self.adler_b = acc.adler_b;
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        let mut acc = Accumulator::from_stream(self, Crc16::Xmodem);
        for span in spans {
            for &byte in &data[span.start..span.end] {
                acc.push(byte);
            }
        }
        self.crc = acc.crc;
        self.adler_a = acc.adler_a;
        self.adler_b = acc.adler_b;
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let crc = parse_snapshot_hex(bytes, 1, 9)?;
        let adler = parse_snapshot_hex(bytes, 11, 19)?;
        Self::from_seed(crc as u32, adler as u32)
    }
}

fn parse_snapshot_hex(bytes: &[u8], start: usize, end: usize) -> Result<u64, SnapshotError> {

        let digit = match bytes[offset] {

            b'a'..=b'f' => (bytes[offset] - b'a' + 10) as u64,
            _ => return Err(SnapshotError::Hex { offset }),
        };
        value = (value << 4) | digit;
    }
    Ok(value)
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc ^= byte as u32;
            for _ in 0..8 {
                let low = self.crc & 1;
                self.crc >>= 1;
                if low != 0 {
                    self.crc ^= 0xedb8_8320;
                }
            }
            self.adler_a = (self.adler_a + byte as u32) % 65_521;
            self.adler_b = (self.adler_b + self.adler_a) % 65_521;
        }
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut crc = 0u32;
        for offset in 1..9 {
            let digit = hex_value(bytes[offset]).ok_or(SnapshotError::Hex { offset })?;
            crc = (crc << 4) | digit as u32;
        }

        let mut adler = 0u32;
        for offset in 11..19 {
            let digit = hex_value(bytes[offset]).ok_or(SnapshotError::Hex { offset })?;
            adler = (adler << 4) | digit as u32;
        }

        Self::from_seed(crc, adler)
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

fn fletcher32_update(a: &mut u32, b: &mut u32, word: u32) {
    *a = (*a + word) % 65_535;
    *b = (*b + *a) % 65_535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut index = 0usize;
    while index + 1 < data.len() {
        let word = ((data[index] as u32) << 8) | data[index + 1] as u32;
        fletcher32_update(&mut a, &mut b, word);
        index += 2;
    }
    if index < data.len() {
        fletcher32_update(&mut a, &mut b, (data[index] as u32) << 8);
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u32;
    let mut b = 0u32;
    let mut pending = None;

    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending.take() {
                let word = ((high as u32) << 8) | byte as u32;
                fletcher32_update(&mut a, &mut b, word);
            } else {
                pending = Some(byte);
            }
        }
    }

    if let Some(high) = pending {
        fletcher32_update(&mut a, &mut b, (high as u32) << 8);
    }

    Ok((b << 16) | a)
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut register = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                register ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = register & 0x8000;
                    register = register.wrapping_shl(1);
                    if top != 0 {
                        register ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                register ^= byte as u16;
                for _ in 0..8 {
                    let low = register & 1;
                    register >>= 1;
                    if low != 0 {
                        register ^= 0x8408;
                    }
                }
            }
        }
    }

    if variant == Crc16::X25 {
        register ^ 0xffff
    } else {
        register
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut state = ChecksumState::new();
    state.crc16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    update_spans(&mut state, data, spans, variant);
    state.finish_fletcher32();

    let mut crc16_result = state.crc16;
    if variant == Crc16::X25 {
        crc16_result ^= 0xffff;
    }

    Ok(Multi {
        base: state.hashes(),
        fletcher16: ((state.fletcher16_b << 8) | state.fletcher16_a) as u16,
        fletcher32: (state.fletcher32_b << 16) | state.fletcher32_a,
        crc16: crc16_result,
    })
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    let mut even = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for slot in odd.iter_mut().skip(1) {
        *slot = row;
        row <<= 1;
    }

    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut crc = left;
    let mut length = right_len;

    loop {
        gf2_matrix_square(&mut even, &odd);
        if length & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }

        gf2_matrix_square(&mut odd, &even);
        if length & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }
    }

    crc ^ right
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Part {
    pub crc32: u32,
    pub len: u64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CombineError {
    LengthOverflow { part: usize },
    EmptyCrc { part: usize },
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut combined = 0u32;
    for part in parts {
        combined = crc32_combine(combined, part.crc32, part.len);
    }

    Ok(Part {
        crc32: combined,
        len: total,
    })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut state = ChecksumState::new();
    let mut length = 0u64;
    for span in spans {
        let span_len = span.end - span.start;
        length += span_len as u64;
        for &byte in &data[span.start..span.end] {
            state.update_base(byte);
        }
    }
    let hashes = state.hashes();

    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        length, hashes.crc32, hashes.adler32
    ))
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VerifyError {
    Span(SpanError),
    Shape,
    Hex { offset: usize },
    Length,
    Mismatch {
        field: usize,
        expected: u32,
        actual: u32,
    },
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let mut length = 0u64;
    for offset in 4..20 {
        let digit = hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        length = (length << 4) | digit as u64;
    }

    let mut expected_crc = 0u32;
    for offset in 25..33 {
        let digit = hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        expected_crc = (expected_crc << 4) | digit as u32;
    }

    let mut expected_adler = 0u32;
    for offset in 40..48 {
        let digit = hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        expected_adler = (expected_adler << 4) | digit as u32;
    }

    let mut state = ChecksumState::new();
    let mut actual_length = 0u64;
    for span in spans {
        actual_length += (span.end - span.start) as u64;
        for &byte in &data[span.start..span.end] {
            state.update_base(byte);
        }
    }

    if length != actual_length {
        return Err(VerifyError::Length);
    }

    let actual = state.hashes();
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }

    Ok(())
}
