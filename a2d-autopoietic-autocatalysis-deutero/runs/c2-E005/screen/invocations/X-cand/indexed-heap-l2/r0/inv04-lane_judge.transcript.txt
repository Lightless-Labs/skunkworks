            if !valid {

        a.item.priority < b.item.priority
            || (a.item.priority == b.item.priority
                && (a.item.value.len() < b.item.value.len()
                    || (a.item.value.len() == b.item.value.len() && a.age < b.age)))

                break;

            self.swap_slots(index, parent);
            index = parent;
        }

            let left = index.saturating_mul(2).saturating_add(1);

                break;

            let right = left + 1;
            let best = if right < len && Self::better(&self.entries[right], &self.entries[left]) {

pub struct IndexedHeap {
    items: Vec<HeapEntry>,
    indices: std::collections::HashMap<String, usize>,
    next_age: u128,
}

struct HeapEntry {
    item: Item,
    age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            items: Vec::new(),
            indices: std::collections::HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.items.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = canonicalize_key(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let age = self.next_age;
        self.next_age = self.next_age.wrapping_add(1);
        let entry = HeapEntry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age,
        };
        let index = self.items.len();
        self.items.push(entry);
        self.indices.insert(key, index);
        self.sift_up(index);
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.items.first().map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.items.is_empty() {
            return None;
        }
        Some(self.remove_at(0).item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize_key(key)?;
        Ok(self.indices.get(&key).map(|&index| self.items[index].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize_key(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing {
            key: key.clone(),
        })?;
        let current = self.items[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.items[index].item.priority = priority;
        self.reheapify(index);
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize_key(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing {
            key: key.clone(),
        })?;
        let current = self.items[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.items[index].item.priority = priority;
        self.reheapify(index);
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonicalize_key(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing {
            key: key.clone(),
        })?;
        let old = std::mem::replace(&mut self.items[index].item.value, value.to_owned());
        self.reheapify(index);
        Ok(old)
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonicalize_key(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing {
            key: key.clone(),
        })?;
        Ok(self.remove_at(index).item)
    }

    fn less(&self, a: usize, b: usize) -> bool {
        let left = &self.items[a];
        let right = &self.items[b];
        (
            left.item.priority,
            left.item.value.len(),
            left.age,
        ) < (
            right.item.priority,
            right.item.value.len(),
            right.age,
        )
    }

    fn swap(&mut self, a: usize, b: usize) {
        self.items.swap(a, b);
        self.indices.insert(self.items[a].item.key.clone(), a);
        self.indices.insert(self.items[b].item.key.clone(), b);
    }

    fn sift_up(&mut self, mut index: usize) {
        while index > 0 {
            let parent = (index - 1) / 2;
            if !self.less(index, parent) {
                break;
            }
            self.swap(index, parent);
            index = parent;
        }
    }

    fn sift_down(&mut self, mut index: usize) {
        loop {
            let left = index * 2 + 1;
            if left >= self.items.len() {
                break;
            }
            let right = left + 1;
            let best = if right < self.items.len() && self.less(right, left) {
                right
            } else {
                left
            };
            if !self.less(best, index) {
                break;
            }
            self.swap(index, best);
            index = best;
        }
    }

    fn reheapify(&mut self, index: usize) {
        if index > 0 && self.less(index, (index - 1) / 2) {
            self.sift_up(index);
        } else {
            self.sift_down(index);
        }
    }

    fn remove_at(&mut self, index: usize) -> HeapEntry {
        let last = self.items.len() - 1;
        self.swap(index, last);
        let removed = self.items.pop().expect("index exists");
        self.indices.remove(&removed.item.key);
        if index < self.items.len() {
            self.reheapify(index);
        }
        removed
    }
}

fn canonicalize_key(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }
    if start == end {
        return Err(Error::EmptyKey);
    }

    for (offset, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }

    Ok(input[start..end].to_ascii_lowercase())
}
