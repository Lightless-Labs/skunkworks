#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_text(word: &mut Word, value: &str, active_glob: bool) {
    if active_glob {
        for (offset, c) in value.char_indices() {
            if matches!(c, '*' | '?' | '[') {
                word.glob.push(word.text.len() + offset);
            }
        }
    }
    word.text.push_str(value);
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value)
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut iter = text.char_indices().peekable();

    while let Some((_, c)) = iter.next() {
        if c == '\\' {
            if let Some((_, next)) = iter.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }

    out
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    after_dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let mut close = None;
    let mut cursor = after_dollar + 1; // after "${"
    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += c.len_utf8();
            if cursor < input.len() {
                let escaped = input[cursor..].chars().next().unwrap();
                cursor += escaped.len_utf8();
            }
        } else if c == '}' {
            close = Some(cursor);
            break;
        } else {
            cursor += c.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[after_dollar + 1..close];

    let mut name_end = 0;
    for (offset, c) in contents.char_indices() {
        if (offset == 0 && is_name_start(c)) || (offset > 0 && is_name_continue(c)) {
            name_end = offset + c.len_utf8();
        } else {
            break;
        }
    }

    if name_end == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_end];
    let rest = &contents[name_end..];

    let value = if rest.is_empty() {
        lookup(env, name).unwrap_or("").to_string()
    } else if let Some(default) = rest.strip_prefix(":-") {
        if rest.as_bytes().get(1) != Some(&b'-') {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        match lookup(env, name) {
            Some(v) if !v.is_empty() => v.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut cursor = 0usize;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let next_pos = cursor + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    current.text.push(c);
                }
                cursor = next_pos;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    cursor = next_pos;
                } else if c == '\\' {
                    if next_pos == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next_pos..].chars().next().unwrap();
                    let after_escaped = next_pos + escaped.len_utf8();
                    if matches!(escaped, '"' | '\\' | '$' | '\n') {
                        if escaped != '\n' {
                            current.text.push(escaped);
                        }
                        cursor = after_escaped;
                    } else {
                        current.text.push('\\');
                        cursor = next_pos;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    if let Some(after) = input.get(next_pos..) {
                        if after.starts_with('(') {
                            return Err(SplitError::CommandSubstitution { pos: cursor });
                        }
                        if after.starts_with('{') {
                            let (value, end) = parse_expansion(input, cursor, next_pos, env)?;
                            active = true;
                            append_text(&mut current, &value, false);
                            cursor = end;
                        } else if let Some(first) = after.chars().next() {
                            if is_name_start(first) {
                                let mut end = next_pos;
                                for ch in after.chars() {
                                    if is_name_continue(ch) {
                                        end += ch.len_utf8();
                                    } else {
                                        break;
                                    }
                                }
                                let value = lookup(env, &input[next_pos..end]).unwrap_or("");
                                active = true;
                                append_text(&mut current, value, false);
                                cursor = end;
                            } else {
                                current.text.push('$');
                                cursor = next_pos;
                            }
                        } else {
                            current.text.push('$');
                            cursor = next_pos;
                        }
                    } else {
                        current.text.push('$');
                        cursor = next_pos;
                    }
                } else {
                    current.text.push(c);
                    cursor = next_pos;
                }
            }
            Quote::None => {
                if separator(c) {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    cursor = next_pos;
                } else if c == '\'' || c == '"' {
                    quote = if c == '\'' { Quote::Single } else { Quote::Double };
                    quote_start = cursor;
                    active = true;
                    cursor = next_pos;
                } else if c == '\\' {
                    if next_pos == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next_pos..].chars().next().unwrap();
                    let after_escaped = next_pos + escaped.len_utf8();
                    if escaped != '\n' {
                        active = true;
                        current.text.push(escaped);
                    }
                    cursor = after_escaped;
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    if input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    } else if input[next_pos..].starts_with('{') {
                        let (value, end) = parse_expansion(input, cursor, next_pos, env)?;
                        active = true;
                        append_text(&mut current, &value, true);
                        cursor = end;
                    } else if let Some(first) = input[next_pos..].chars().next() {
                        if is_name_start(first) {
                            let mut end = next_pos;
                            for ch in input[next_pos..].chars() {
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let value = lookup(env, &input[next_pos..end]).unwrap_or("");
                            active = true;
                            append_text(&mut current, value, true);
                            cursor = end;
                        } else {
                            active = true;
                            current.text.push('$');
                            cursor = next_pos;
                        }
                    } else {
                        active = true;
                        current.text.push('$');
                        cursor = next_pos;
                    }
                } else {
                    active = true;
                    let start = current.text.len();
                    current.text.push(c);
                    if matches!(c, '*' | '?' | '[') {
                        current.glob.push(start);
                    }
                    cursor = next_pos;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }

    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || matches!(c, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%') {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::new();
    out.push('"');
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (index, word) in words.iter().enumerate() {
        if let Some(pos) = word.as_bytes().iter().position(|&b| b == 0) {
            return Err(JoinError::Nul { word: index, pos });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let candidates = [
            bare_candidate(word),
            single_candidate(word),
            double_candidate(word),
        ];

        let selected = candidates
            .into_iter()
            .flatten()
            .min_by_key(|candidate| candidate.len());

        let encoded_word = match selected {
            Some(value) => value,
            None => {
                let mut out = String::new();
                out.push('\'');
                for c in word.chars() {
                    if c == '\'' {
                        out.push('\'');
                        out.push('\\');
                        out.push('\'');
                        out.push('\'');
                    } else {
                        out.push(c);
                    }
                }
                out.push('\'');
                out
            }
        };
        encoded.push(encoded_word);
    }

    Ok(encoded.join(" "))
}
