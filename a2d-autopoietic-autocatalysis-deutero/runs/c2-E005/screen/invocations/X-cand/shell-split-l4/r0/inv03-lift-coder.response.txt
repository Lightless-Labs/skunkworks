#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn append_char(word: &mut Word, c: char, active: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if active && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn append_text(word: &mut Word, text: &str, active: bool) {
    for c in text.chars() {
        append_char(word, c, active);
    }
}

fn byte_at(input: &str, index: usize) -> Option<char> {
    input.get(index..)?.chars().next()
}

fn find_brace_end(input: &str, mut i: usize) -> Option<usize> {
    while i < input.len() {
        let c = byte_at(input, i)?;
        if c == '}' {
            return Some(i);
        }
        if c == '\\' {
            i += c.len_utf8();
            if i >= input.len() {
                continue;
            }
            if let Some(next) = byte_at(input, i) {
                i += next.len_utf8();
            }
        } else {
            i += c.len_utf8();
        }
    }
    None
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = byte_at(text, i).unwrap();
        if c == '\\' {
            i += 1;
            if i < text.len() {
                let next = byte_at(text, i).unwrap();
                out.push(next);
                i += next.len_utf8();
            }
        } else {
            out.push(c);
            i += c.len_utf8();
        }
    }
    out
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let after_dollar = dollar + 1;
    let first = byte_at(input, after_dollar).unwrap();

    if first != '{' {
        let mut end = after_dollar;
        while end < input.len() {
            let c = byte_at(input, end).unwrap();
            if !is_name_continue(c) {
                break;
            }
            end += c.len_utf8();
        }
        let name = &input[after_dollar..end];
        return Ok((lookup(name, env).unwrap_or("").to_string(), end));
    }

    let content_start = after_dollar + 1;
    let close = find_brace_end(input, content_start)
        .ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[content_start..close];

    let name_end = contents
        .char_indices()
        .find_map(|(idx, c)| (!is_name_continue(c)).then_some(idx))
        .unwrap_or(contents.len());

    let name = &contents[..name_end];
    if name.is_empty() || !name.chars().next().is_some_and(is_name_start) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    if !name.chars().all(is_name_continue) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let value = if name_end == contents.len() {
        lookup(name, env).unwrap_or("").to_string()
    } else if contents[name_end..].starts_with(":-") {
        let default = &contents[name_end + 2..];
        match lookup(name, env) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current: Option<Word> = None;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = byte_at(input, i).unwrap();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_char(current.as_mut().unwrap(), c, false);
                }
                i += c.len_utf8();
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    i += 1;
                } else if c == '\\' {
                    let slash = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = byte_at(input, i).unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_char(current.as_mut().unwrap(), next, false);
                        }
                        i += next.len_utf8();
                    } else {
                        append_char(current.as_mut().unwrap(), '\\', false);
                    }
                } else if c == '$' {
                    let next_pos = i + 1;
                    if next_pos < input.len() && byte_at(input, next_pos) == Some('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if next_pos < input.len()
                        && byte_at(input, next_pos).is_some_and(is_name_start)
                    {
                        current.get_or_insert_with(|| Word {
                            text: String::new(),
                            glob: Vec::new(),
                        });
                        let (value, end) = parse_expansion(input, i, env)?;
                        append_text(current.as_mut().unwrap(), &value, false);
                        i = end;
                    } else if next_pos < input.len()
                        && byte_at(input, next_pos) == Some('{')
                    {
                        current.get_or_insert_with(|| Word {
                            text: String::new(),
                            glob: Vec::new(),
                        });
                        let (value, end) = parse_expansion(input, i, env)?;
                        append_text(current.as_mut().unwrap(), &value, false);
                        i = end;
                    } else {
                        append_char(current.as_mut().unwrap(), '$', false);
                        i += 1;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else {
                    append_char(current.as_mut().unwrap(), c, false);
                    i += c.len_utf8();
                }
                let _ = open;
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if let Some(word) = current.take() {
                        words.push(word);
                    }
                    i += c.len_utf8();
                } else if c == '\'' {
                    current.get_or_insert_with(|| Word {
                        text: String::new(),
                        glob: Vec::new(),
                    });
                    quote = Quote::Single(i);
                    i += 1;
                } else if c == '"' {
                    current.get_or_insert_with(|| Word {
                        text: String::new(),
                        glob: Vec::new(),
                    });
                    quote = Quote::Double(i);
                    i += 1;
                } else if c == '\\' {
                    let slash = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = byte_at(input, i).unwrap();
                    if next != '\n' {
                        current.get_or_insert_with(|| Word {
                            text: String::new(),
                            glob: Vec::new(),
                        });
                        append_char(current.as_mut().unwrap(), next, false);
                    }
                    i += next.len_utf8();
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let next_pos = i + 1;
                    if next_pos < input.len() && byte_at(input, next_pos) == Some('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if next_pos < input.len()
                        && (byte_at(input, next_pos).is_some_and(is_name_start)
                            || byte_at(input, next_pos) == Some('{'))
                    {
                        current.get_or_insert_with(|| Word {
                            text: String::new(),
                            glob: Vec::new(),
                        });
                        let (value, end) = parse_expansion(input, i, env)?;
                        append_text(current.as_mut().unwrap(), &value, true);
                        i = end;
                    } else {
                        current.get_or_insert_with(|| Word {
                            text: String::new(),
                            glob: Vec::new(),
                        });
                        append_char(current.as_mut().unwrap(), '$', true);
                        i += 1;
                    }
                } else {
                    current.get_or_insert_with(|| Word {
                        text: String::new(),
                        glob: Vec::new(),
                    });
                    append_char(current.as_mut().unwrap(), c, true);
                    i += c.len_utf8();
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }
    if let Some(word) = current {
        words.push(word);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        None
    } else {
        let mut out = String::from("\"");
        for c in word.chars() {
            if matches!(c, '\\' | '"' | '$') {
                out.push('\\');
            }
            out.push(c);
        }
        out.push('"');
        Some(out)
    }
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates: Vec<(usize, usize, String)> = Vec::new();
        if let Some(candidate) = bare_candidate(word) {
            candidates.push((candidate.len(), 0, candidate));
        }
        if let Some(candidate) = single_candidate(word) {
            candidates.push((candidate.len(), 1, candidate));
        }
        if let Some(candidate) = double_candidate(word) {
            candidates.push((candidate.len(), 2, candidate));
        }

        let selected = if candidates.is_empty() {
            let mut fallback = String::from("'");
            for c in word.chars() {
                if c == '\'' {
                    fallback.push('\'');
                    fallback.push('\\');
                    fallback.push('\'');
                    fallback.push('\'');
                } else {
                    fallback.push(c);
                }
            }
            fallback.push('\'');
            fallback
        } else {
            candidates.sort_by_key(|(len, priority, _)| (*len, *priority));
            candidates.remove(0).2
        };
        encoded.push(selected);
    }

    Ok(encoded.join(" "))
}
