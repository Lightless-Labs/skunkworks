#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut event_order: Vec<usize> = (0..jobs.len()).collect();
    event_order.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut ready: VecDeque<usize> = VecDeque::new();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut event_cursor = 0usize;
    let mut last_dispatched: Option<u32> = None;

    fn admit_through(
        jobs: &[Job],
        event_order: &[usize],
        event_cursor: &mut usize,
        time: u64,
        ready: &mut VecDeque<usize>,
        completion: &mut [Option<u64>],
    ) {
        while *event_cursor < event_order.len() {
            let index = event_order[*event_cursor];
            let job = &jobs[index];
            if u64::from(job.arrival) > time {
                break;
            }

            if job.ticks == 0 {
                completion[index] = Some(u64::from(job.arrival));
            } else {
                ready.push_back(index);
            }
            *event_cursor += 1;
        }
    }

    while completion.iter().any(Option::is_none) {
        admit_through(
            jobs,
            &event_order,
            &mut event_cursor,
            time,
            &mut ready,
            &mut completion,
        );

        let Some(index) = ready.pop_front() else {
            if event_cursor >= event_order.len() {
                break;
            }

            let next_time = u64::from(jobs[event_order[event_cursor]].arrival);
            if next_time > time {
                if let Some(last_idle) = idle.last_mut() {
                    if last_idle.1 == time {
                        last_idle.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched = None;
            }
            continue;
        };

        let id = jobs[index].id;
        if last_dispatched.is_some_and(|last_id| last_id != id) {
            time += u64::from(switch_cost);
            admit_through(
                jobs,
                &event_order,
                &mut event_cursor,
                time,
                &mut ready,
                &mut completion,
            );
        }

        let start = time;
        let consumed = remaining[index].min(quantum);
        time += u64::from(consumed);
        remaining[index] -= consumed;
        let finished = remaining[index] == 0;

        runs.push(Run {
            id,
            start,
            end: time,
            finished,
        });

        admit_through(
            jobs,
            &event_order,
            &mut event_cursor,
            time,
            &mut ready,
            &mut completion,
        );

        if finished {
            completion[index] = Some(time);
        } else {
            ready.push_back(index);
        }
        last_dispatched = Some(id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].unwrap_or(0)))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}

use std::collections::VecDeque;
