#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut separators = bytes.iter().enumerate().filter(|(_, &b)| b == b'@');
    let Some((separator_at, _)) = separators.next() else {
        return Err(EmailError::MissingAt);
    };
    if let Some((at, _)) = separators.next() {
        return Err(EmailError::MultipleAt { at });
    }

    if separator_at == 0 {
        return Err(EmailError::EmptyLocal { at: separator_at });
    }

    let local = &bytes[..separator_at];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (i, &byte) in local.iter().enumerate() {
        let at = i;
        if byte == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at });
            }
        } else if !byte.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&byte)
        {
            return Err(EmailError::LocalChar { at });
        }
    }

    let domain_start = separator_at + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(has_root_dot);

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain = &raw_domain[..domain_len];
    let root_dot_at = if has_root_dot {
        Some(domain_start + domain_len)
    } else {
        None
    };

    let mut labels: Vec<(usize, &[u8])> = Vec::new();
    let mut label_start = 0;

    for (i, &byte) in domain.iter().enumerate() {
        if byte == b'.' {
            if i == label_start {
                return Err(EmailError::EmptyLabel {
                    at: domain_start + i,
                });
            }
            labels.push((label_start, &domain[label_start..i]));
            label_start = i + 1;
        }
    }

    if label_start == domain.len() {
        return Err(EmailError::EmptyLabel {
            at: root_dot_at.unwrap_or(domain_start + domain_len),
        });
    }
    labels.push((label_start, &domain[label_start..]));

    for &(start, label) in &labels {
        if label.len() > 63 {
            return Err(EmailError::LabelTooLong {
                at: domain_start + start + 63,
            });
        }

        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + start,
            });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + start + label.len() - 1,
            });
        }

        for (i, &byte) in label.iter().enumerate() {
            if !byte.is_ascii_alphanumeric() && byte != b'-' {
                return Err(EmailError::DomainChar {
                    at: domain_start + start + i,
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_label) = labels[labels.len() - 1];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let local_string = String::from_utf8(local.to_vec()).expect("ASCII local part");
    let domain_string = domain
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local: local_string,
        domain: domain_string,
        quoted: false,
    })
}
