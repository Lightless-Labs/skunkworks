use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_missing_byte(byte: Option<u8>) -> bool {
    matches!(byte, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;

    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }

    if *pos == start {
        return Err(if is_core_missing_byte(bytes.get(*pos).copied()) {
            VersionError::MissingCore { at: *pos }
        } else {
            VersionError::InvalidChar { at: *pos }
        });
    }

    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let number = s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })?;

    Ok(number)
}

fn is_identifier_char(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'-'
}

fn parse_identifiers(
    s: &str,
    pos: &mut usize,
    stop_at_plus: bool,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut identifiers = Vec::new();

    loop {
        let start = *pos;
        while *pos < bytes.len()
            && is_identifier_char(bytes[*pos])
            && !(stop_at_plus && bytes[*pos] == b'+')
        {
            *pos += 1;
        }

        if *pos == start {
            return Err(VersionError::EmptyIdentifier { at: start });
        }

        let identifier = &s[start..*pos];
        if identifier
            .as_bytes()
            .iter()
            .any(|byte| !is_identifier_char(*byte))
        {
            let invalid = identifier
                .as_bytes()
                .iter()
                .position(|byte| !is_identifier_char(*byte))
                .unwrap();
            return Err(VersionError::InvalidChar { at: start + invalid });
        }

        if is_pre
            && identifier.len() > 1
            && identifier.as_bytes()[0] == b'0'
            && identifier.as_bytes().iter().all(u8::is_ascii_digit)
        {
            return Err(VersionError::LeadingZero { at: start });
        }

        identifiers.push(identifier.to_owned());

        if *pos == bytes.len() || (stop_at_plus && bytes[*pos] == b'+') {
            return Ok(identifiers);
        }

        if bytes[*pos] == b'.' {
            *pos += 1;
            continue;
        }

        return Err(VersionError::InvalidChar { at: *pos });
    }
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;

    for _ in 0..2 {
        if pos >= s.len() || s.as_bytes()[pos] != b'.' {
            return Err(if is_core_missing_byte(s.as_bytes().get(pos).copied()) {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        pos += 1;

        let value = parse_core(s, &mut pos)?;
        if _ == 0 {
            // The second core field is parsed here; the third is parsed below.
        }
        if _ == 0 {
            let minor = value;
            // Keep the parsed field while continuing through the patch.
            let patch = parse_patch_after_minor(s, &mut pos)?;
            return parse_suffix(s, major, minor, patch, pos);
        }
    }

    unreachable!()
}

fn parse_patch_after_minor(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    if *pos >= s.len() || s.as_bytes()[*pos] != b'.' {
        return Err(if is_core_missing_byte(s.as_bytes().get(*pos).copied()) {
            VersionError::MissingCore { at: *pos }
        } else {
            VersionError::InvalidChar { at: *pos }
        });
    }
    *pos += 1;
    parse_core(s, pos)
}

fn parse_suffix(
    s: &str,
    major: u64,
    minor: u64,
    patch: u64,
    mut pos: usize,
) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() {
        match bytes[pos] {
            b'-' => {
                pos += 1;
                pre = parse_identifiers(s, &mut pos, true, true)?;
                if pos < bytes.len() {
                    pos += 1;
                    build = parse_identifiers(s, &mut pos, false, false)?;
                }
            }
            b'+' => {
                pos += 1;
                build = parse_identifiers(s, &mut pos, false, false)?;
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(u8::is_ascii_digit)
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let a_numeric = numeric_identifier(a);
    let b_numeric = numeric_identifier(b);

    match (a_numeric, b_numeric) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a
            .len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (left, right) in a.pre.iter().zip(&b.pre) {
                let ordering = compare_identifiers(left, right);
                if ordering != Ordering::Equal {
                    return ordering;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Operator {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
struct Comparator {
    operator: Operator,
    version: Version,
}

fn is_range_whitespace(byte: u8) -> bool {
    matches!(byte, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
        pos += 1;
    }

    if pos == bytes.len() {
        return Err(RangeError::Empty { at: 0 });
    }

    while pos < bytes.len() {
        let token_start = pos;
        let mut operand_start = pos;
        let operator = match bytes[pos] {
            b'=' => {
                operand_start += 1;
                Operator::Equal
            }
            b'<' => {
                if bytes.get(pos + 1) == Some(&b'=') {
                    operand_start += 2;
                    Operator::LessEqual
                } else {
                    operand_start += 1;
                    Operator::Less
                }
            }
            b'>' => {
                if bytes.get(pos + 1) == Some(&b'=') {
                    operand_start += 2;
                    Operator::GreaterEqual
                } else {
                    operand_start += 1;
                    Operator::Greater
                }
            }
            b'0'..=b'9' => Operator::Equal,
            _ => return Err(RangeError::Syntax { at: token_start }),
        };

        if operand_start >= bytes.len() || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let token_end = {
            let mut end = operand_start;
            while end < bytes.len() && !is_range_whitespace(bytes[end]) {
                end += 1;
            }
            end
        };

        let version = parse_version(&s[operand_start..token_end]).map_err(|error| {
            RangeError::Version {
                at: operand_start,
                error,
            }
        })?;

        comparators.push(Comparator { operator, version });
        pos = token_end;

        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
    }

    Ok(Range { comparators })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    let candidate_core = (v.major, v.minor, v.patch);

    if !v.pre.is_empty()
        && !r.comparators.iter().any(|comparator| {
            !comparator.version.pre.is_empty()
                && (
                    comparator.version.major,
                    comparator.version.minor,
                    comparator.version.patch,
                ) == candidate_core
        })
    {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let ordering = compare_versions(v, &comparator.version);
        match comparator.operator {
            Operator::Equal => ordering == Ordering::Equal,
            Operator::Less => ordering == Ordering::Less,
            Operator::LessEqual => ordering != Ordering::Greater,
            Operator::Greater => ordering == Ordering::Greater,
            Operator::GreaterEqual => ordering != Ordering::Less,
        }
    })
}
