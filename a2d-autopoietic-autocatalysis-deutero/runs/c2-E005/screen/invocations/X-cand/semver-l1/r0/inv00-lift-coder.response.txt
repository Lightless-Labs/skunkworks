use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let mut core = [0u64; 3];

    for (field, slot) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if start == pos {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: start });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut value = 0u64;
        for &digit in &bytes[start..pos] {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        *slot = value;

        if field < 2 {
            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
            } else if pos == bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        } else if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::InvalidChar { at: pos });
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let pre_start = pos;
        let build_pos = scan_identifiers(bytes, pos, true, &mut pre)?;
        if build_pos < bytes.len() && bytes[build_pos] == b'+' {
            pos = build_pos + 1;
            if pos == bytes.len() {
                return Err(VersionError::EmptyIdentifier { at: pos });
            }
            scan_identifiers(bytes, pos, false, &mut build)?;
        } else {
            pos = build_pos;
        }
        if pre.is_empty() {
            return Err(VersionError::EmptyIdentifier { at: pre_start });
        }
    } else if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        if pos == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: pos });
        }
        scan_identifiers(bytes, pos, false, &mut build)?;
    }

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn scan_identifiers(
    bytes: &[u8],
    mut pos: usize,
    is_pre: bool,
    out: &mut Vec<String>,
) -> Result<usize, VersionError> {
    loop {
        let start = pos;
        while pos < bytes.len() && bytes[pos] != b'.' && !(is_pre && bytes[pos] == b'+') {
            if !bytes[pos].is_ascii_alphanumeric() && bytes[pos] != b'-' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }

        if pos == start {
            return Err(VersionError::EmptyIdentifier { at: start });
        }

        if is_pre
            && bytes[start..pos].iter().all(u8::is_ascii_digit)
            && pos - start > 1
            && bytes[start] == b'0'
        {
            return Err(VersionError::LeadingZero { at: start });
        }

        out.push(String::from_utf8(bytes[start..pos].to_vec()).unwrap());

        if pos < bytes.len() && bytes[pos] == b'.' {
            pos += 1;
            if pos == bytes.len() || (is_pre && bytes[pos] == b'+') {
                return Err(VersionError::EmptyIdentifier { at: pos });
            }
            continue;
        }

        return Ok(pos);
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (left, right) in a.iter().zip(b) {
        let left_numeric = left.bytes().all(|c| c.is_ascii_digit());
        let right_numeric = right.bytes().all(|c| c.is_ascii_digit());

        let ordering = match (left_numeric, right_numeric) {
            (true, true) => left
                .len()
                .cmp(&right.len())
                .then_with(|| left.as_bytes().cmp(right.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => left.as_bytes().cmp(right.as_bytes()),
        };

        if ordering != Ordering::Equal {
            return ordering;
        }
    }

    a.len().cmp(&b.len())
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
    prerelease_cores: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Operator {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Comparator {
    operator: Operator,
    version: Version,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();
    let mut prerelease_cores = Vec::new();

    while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
        pos += 1;
    }
    if pos == bytes.len() {
        return Err(RangeError::Empty { at: 0 });
    }

    while pos < bytes.len() {
        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;
        let token = &s[token_start..token_end];
        let token_bytes = token.as_bytes();

        let (operator, operator_len) = if token_bytes.starts_with(b">=") {
            (Operator::Ge, 2)
        } else if token_bytes.starts_with(b"<=") {
            (Operator::Le, 2)
        } else if token_bytes.starts_with(b">") {
            (Operator::Gt, 1)
        } else if token_bytes.starts_with(b"<") {
            (Operator::Lt, 1)
        } else if token_bytes.starts_with(b"=") {
            (Operator::Eq, 1)
        } else {
            (Operator::Eq, 0)
        };

        let operand_start = token_start + operator_len;
        if operand_start == token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let version = parse_version(&s[operand_start..token_end]).map_err(|error| {
            RangeError::Version {
                at: operand_start,
                error,
            }
        })?;

        if !version.pre.is_empty() {
            prerelease_cores.push((version.major, version.minor, version.patch));
        }
        comparators.push(Comparator { operator, version });

        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
    }

    Ok(Range {
        comparators,
        prerelease_cores,
    })
}

fn is_range_whitespace(byte: u8) -> bool {
    matches!(byte, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn matches(v: &Version, r: &Range) -> bool {
    if !v.pre.is_empty()
        && !r
            .prerelease_cores
            .contains(&(v.major, v.minor, v.patch))
    {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let ordering = compare_versions(v, &comparator.version);
        match comparator.operator {
            Operator::Eq => ordering == Ordering::Equal,
            Operator::Lt => ordering == Ordering::Less,
            Operator::Le => ordering != Ordering::Greater,
            Operator::Gt => ordering == Ordering::Greater,
            Operator::Ge => ordering != Ordering::Less,
        }
    })
}
