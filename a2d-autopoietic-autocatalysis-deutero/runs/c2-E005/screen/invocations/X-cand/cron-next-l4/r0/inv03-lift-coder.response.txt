use std::str::FromStr;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<bool>; 5],
    day_of_month_unrestricted: bool,
    day_of_week_unrestricted: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

#[derive(Clone, Copy)]
struct Domain {
    min: u32,
    max: u32,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn number_error(field: Field, offset: usize, kind: NumberProblem) -> ParseError {
    match kind {
        NumberProblem::LeadingZero => ParseError::LeadingZero { field, offset },
        NumberProblem::Overflow => ParseError::NumberOverflow { field, offset },
        NumberProblem::OutOfRange => ParseError::OutOfRange { field, offset },
    }
}

enum NumberProblem {
    LeadingZero,
    Overflow,
    OutOfRange,
}

fn parse_number(bytes: &[u8], offset: usize, field: Field, bounds: Domain) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(number_error(field, offset, NumberProblem::LeadingZero));
    }
    let mut n = 0u32;
    for &b in bytes {
        n = n
            .checked_mul(10)
            .and_then(|x| x.checked_add((b - b'0') as u32))
            .ok_or_else(|| number_error(field, offset, NumberProblem::Overflow))?;
    }
    if n < bounds.min || n > bounds.max {
        return Err(number_error(field, offset, NumberProblem::OutOfRange));
    }
    Ok(n)
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u32> {
    if !bytes.iter().all(u8::is_ascii_alphabetic) {
        return None;
    }
    let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
    let names: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    names
        .iter()
        .position(|name| name.as_bytes() == upper.as_slice())
        .map(|i| if field == Field::Month { i as u32 + 1 } else { i as u32 })
}

fn is_digits(bytes: &[u8]) -> bool {
    !bytes.is_empty() && bytes.iter().all(u8::is_ascii_digit)
}

fn parse_endpoint(
    field: Field,
    bytes: &[u8],
    offset: usize,
) -> Result<u32, ParseError> {
    if is_digits(bytes) {
        parse_number(bytes, offset, field, domain(field))
    } else if let Some(value) = name_value(field, bytes) {
        Ok(value)
    } else {
        Err(ParseError::InvalidSyntax { field, offset })
    }
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let dom = domain(field);
    let mut selected = vec![false; (dom.max + 1) as usize];

    let mut item_start = 0usize;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());
        let item = &bytes[item_start..item_end];

        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let (base, step_part, step_offset) = if let Some(&slash) = slash_positions.first() {
            (&item[..slash], Some(&item[slash + 1..]), item_start + slash + 1)
        } else {
            (item, None, item_end)
        };

        let (start_bytes, end_bytes): (&[u8], Option<&[u8]>) = if base == b"*" {
            (base, None)
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| (b == b'-').then_some(i))
                .collect();
            if hyphens.len() > 1 {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            if let Some(&hyphen) = hyphens.first() {
                let left = &base[..hyphen];
                let right = &base[hyphen + 1..];
                if left.is_empty() || right.is_empty() || left == b"*" || right == b"*" {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                (left, Some(right))
            } else {
                if base.is_empty() || base.contains(&b'*') {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                (base, None)
            }
        };

        if let Some(step_bytes) = step_part {
            if step_bytes.is_empty() || !is_digits(step_bytes) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }

        let (start, end, end_offset) = if start_bytes == b"*" {
            (dom.min, dom.max, item_start)
        } else {
            let start = parse_endpoint(field, start_bytes, item_start)?;
            if let Some(end_bytes) = end_bytes {
                let relative = end_bytes.as_ptr() as usize - item.as_ptr() as usize;
                let end_offset = item_start + relative;
                let end = parse_endpoint(field, end_bytes, end_offset)?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_offset,
                    });
                }
                (start, end, end_offset)
            } else {
                let end = if step_part.is_some() { dom.max } else { start };
                (start, end, item_start)
            }
        };

        let step = if let Some(step_bytes) = step_part {
            let offset = step_offset;
            if step_bytes.len() > 1 && step_bytes[0] == b'0' {
                return Err(ParseError::LeadingZero { field, offset });
            }
            let mut n = 0u32;
            for &b in step_bytes {
                n = n
                    .checked_mul(10)
                    .and_then(|x| x.checked_add((b - b'0') as u32))
                    .ok_or(ParseError::NumberOverflow { field, offset })?;
            }
            let width = dom.max - dom.min + 1;
            if n == 0 || n > width {
                return Err(ParseError::BadStep { field, offset });
            }
            n
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded;
    let source = if input.as_bytes().first() == Some(&b'@') {
        expanded = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        expanded
    } else {
        input
    };

    let pieces: Vec<&str> = source.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut sets: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for i in 0..5 {
        sets[i] = parse_field(pieces[i], FIELDS[i])?;
    }

    Ok(Cron {
        sets,
        day_of_month_unrestricted: pieces[2] == "*",
        day_of_week_unrestricted: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        self.sets[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &is_set)| is_set.then_some(value as u8))
            .filter(|&value| {
                let d = domain(field);
                (value as u32) >= d.min && (value as u32) <= d.max
            })
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        self.sets[field_index(field)]
            .get(value as usize)
            .copied()
            .unwrap_or(false)
    }

    fn date_matches(&self, dt: CivilDateTime) -> bool {
        let dom_match = self.contains(Field::DayOfMonth, dt.day);
        let dow_match = self.contains(Field::DayOfWeek, weekday(dt.year, dt.month, dt.day));
        match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow_match,
            (false, true) => dom_match,
            (false, false) if dt.day == days_in_month(dt.year, dt.month) => {
                dom_match && dow_match
            }
            (false, false) => dom_match || dow_match,
        }
    }

    fn matches_valid(&self, dt: CivilDateTime) -> bool {
        self.contains(Field::Minute, dt.minute)
            && self.contains(Field::Hour, dt.hour)
            && self.contains(Field::Month, dt.month)
            && self.date_matches(dt)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut date = Date::from_datetime(after);
        loop {
            let year = date.year;
            let month = date.month;
            let day = date.day;
            let day_start = CivilDateTime {
                year,
                month,
                day,
                hour: 0,
                minute: 0,
            };

            if self.contains(Field::Month, month)
                && self.date_matches(day_start)
            {
                for hour in self.allowed(Field::Hour) {
                    for minute in self.allowed(Field::Minute) {
                        let candidate = CivilDateTime {
                            year,
                            month,
                            day,
                            hour,
                            minute,
                        };
                        if candidate > after {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }

            if !date.advance_day() {
                return Ok(None);
            }
        }
    }

    pub fn previous_fire(
        &self,
        before: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(before)?;
        let mut date = Date::from_datetime(before);
        loop {
            let (year, month, day) = (date.year, date.month, date.day);
            let day_start = CivilDateTime {
                year,
                month,
                day,
                hour: 0,
                minute: 0,
            };
            if self.contains(Field::Month, month) && self.date_matches(day_start) {
                let hours = self.allowed(Field::Hour);
                let minutes = self.allowed(Field::Minute);
                for &hour in hours.iter().rev() {
                    for &minute in minutes.iter().rev() {
                        let candidate = CivilDateTime {
                            year,
                            month,
                            day,
                            hour,
                            minute,
                        };
                        if candidate < before {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            if !date.retreat_day() {
                return Ok(None);
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate_datetime(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate_datetime(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while cursor < end {
            match self.next_fire(cursor) {
                Ok(Some(next)) if next < end => {
                    if fires.len() == cap {
                        return Ok(FireWindow {
                            fires,
                            truncated: true,
                        });
                    }
                    fires.push(next);
                    cursor = next;
                }
                _ => break,
            }
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let fields = [
            canonical_field(self, Field::Minute, false),
            canonical_field(self, Field::Hour, false),
            canonical_field(self, Field::DayOfMonth, self.day_of_month_unrestricted),
            canonical_field(self, Field::Month, false),
            canonical_field(self, Field::DayOfWeek, self.day_of_week_unrestricted),
        ];
        fields.join(" ")
    }
}

fn canonical_field(cron: &Cron, field: Field, unrestricted: bool) -> String {
    if unrestricted {
        return "*".to_string();
    }
    let d = domain(field);
    let allowed = cron.allowed(field);
    let full = allowed.len() == (d.max - d.min + 1) as usize;
    if full {
        if matches!(field, Field::DayOfMonth | Field::DayOfWeek) {
            return "*/1".to_string();
        }
        return "*".to_string();
    }
    allowed
        .iter()
        .map(u8::to_string)
        .collect::<Vec<_>>()
        .join(",")
}

fn validate_datetime(dt: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&dt.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&dt.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if dt.day == 0 || dt.day > days_in_month(dt.year, dt.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if dt.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if dt.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

#[derive(Clone, Copy)]
struct Date {
    year: u16,
    month: u8,
    day: u8,
}

impl Date {
    fn from_datetime(dt: CivilDateTime) -> Self {
        Self {
            year: dt.year,
            month: dt.month,
            day: dt.day,
        }
    }

    fn advance_day(&mut self) -> bool {
        if self.day < days_in_month(self.year, self.month) {
            self.day += 1;
            true
        } else if self.month < 12 {
            self.month += 1;
            self.day = 1;
            true
        } else if self.year < 2399 {
            self.year += 1;
            self.month = 1;
            self.day = 1;
            true
        } else {
            false
        }
    }

    fn retreat_day(&mut self) -> bool {
        if self.day > 1 {
            self.day -= 1;
            true
        } else if self.month > 1 {
            self.month -= 1;
            self.day = days_in_month(self.year, self.month);
            true
        } else if self.year > 2000 {
            self.year -= 1;
            self.month = 12;
            self.day = 31;
            true
        } else {
            false
        }
    }
}
