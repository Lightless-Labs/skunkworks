use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return Err(if start == bytes.len()
            || matches!(bytes[start], b'.' | b'-' | b'+')
        {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut ident_start = start;
    let mut i = start;
    loop {
        if i == end || bytes[i] == b'.' {
            if i == ident_start {
                return Err(VersionError::EmptyIdentifier { at: ident_start });
            }
            let ident = &s[ident_start..i];
            if is_pre
                && ident.len() > 1
                && ident.bytes().all(|b| b.is_ascii_digit())
                && ident.starts_with('0')
            {
                return Err(VersionError::LeadingZero { at: ident_start });
            }
            result.push(ident.to_string());
            if i == end {
                break;
            }
            i += 1;
            ident_start = i;
            continue;
        }
        if !bytes[i].is_ascii_alphanumeric() && bytes[i] != b'-' {
            return Err(VersionError::InvalidChar { at: i });
        }
        i += 1;
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    for expected in 0..2 {
        if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        if s.as_bytes()[pos] != b'.' {
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        let n = parse_core_number(s, &mut pos)?;
        if expected == 0 {
            let minor = n;
            if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if s.as_bytes()[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
            let patch = parse_core_number(s, &mut pos)?;
            return parse_version_tail(s, pos, major, minor, patch);
        }
    }
    unreachable!()
}

fn parse_version_tail(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }
    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let plus = s[pos..].find('+').map(|i| pos + i).unwrap_or(s.len());
        pre = parse_identifiers(s, pos, plus, true)?;
        pos = plus;
    }
    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(s, pos, s.len(), false)?;
        pos = s.len();
    }
    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric(x);
        let yn = is_numeric(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn cmp_match(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens(s: &str, base: usize) -> Vec<(String, usize)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && ws(b[i]) {
            i += 1;
        }
        if i == b.len() {
            break;
        }
        let start = i;
        while i < b.len() && !ws(b[i]) {
            i += 1;
        }
        out.push((s[start..i].to_string(), base + start));
    }
    out
}

fn parse_full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

#[derive(Clone, Debug)]
enum Partial {
    Full(Version),
    Fields(Vec<Option<u64>>),
}

impl Partial {
    fn numeric_count(&self) -> usize {
        match self {
            Partial::Full(_) => 3,
            Partial::Fields(v) => v.iter().filter(|x| x.is_some()).count(),
        }
    }

    fn padded(&self) -> Version {
        match self {
            Partial::Full(v) => v.clone(),
            Partial::Fields(fields) => Version {
                major: fields.get(0).and_then(|x| *x).unwrap_or(0),
                minor: fields.get(1).and_then(|x| *x).unwrap_or(0),
                patch: fields.get(2).and_then(|x| *x).unwrap_or(0),
                pre: Vec::new(),
                build: Vec::new(),
            },
        }
    }

    fn is_full(&self) -> bool {
        matches!(self, Partial::Full(_))
    }
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix = s.contains('-') || s.contains('+');
    let raw: Vec<&str> = s.split('.').collect();
    let has_three_no_wild = raw.len() >= 3
        && !raw.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if has_suffix || has_three_no_wild {
        return parse_full_operand(s, at).map(Partial::Full);
    }

    if raw.is_empty() || raw.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut fields = Vec::new();
    let mut seen_wild = false;
    for field in raw {
        if field.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(field, "*" | "x" | "X") {
            seen_wild = true;
            fields.push(None);
        } else if field.bytes().all(|b| b.is_ascii_digit()) {
            if seen_wild
                || (field.len() > 1 && field.starts_with('0'))
                || field.parse::<u64>().is_err()
            {
                return Err(RangeError::Syntax { at });
            }
            fields.push(Some(field.parse::<u64>().unwrap()));
        } else {
            return Err(RangeError::Syntax { at });
        }
    }
    Ok(Partial::Fields(fields))
}

fn endpoint_upper(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut out = Version {
        major: v.major,
        minor: v.minor,
        patch: v.patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    };
    let fail = || RangeError::Syntax { at };
    match field {
        1 => {
            out.major = v.major.checked_add(1).ok_or_else(fail)?;
            out.minor = 0;
            out.patch = 0;
        }
        2 => {
            out.minor = v.minor.checked_add(1).ok_or_else(fail)?;
            out.patch = 0;
        }
        3 => out.patch = v.patch.checked_add(1).ok_or_else(fail)?,
        _ => {}
    }
    Ok(out)
}

fn add(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors
            .push((version.major, version.minor, version.patch));
    }
    arm.comparators.push(Comparator { op, version });
}

fn parse_expanded_token(
    arm: &mut Arm,
    token: &str,
    token_at: usize,
    kind: u8,
    operand: &str,
    operand_at: usize,
) -> Result<(), RangeError> {
    let p = parse_partial(operand, operand_at)?;
    let k = p.numeric_count();
    if kind != b'b' && k == 0 {
        return Err(RangeError::Syntax { at: token_at });
    }
    let lower = p.padded();
    if kind == b'b' && k == 0 {
        return Ok(());
    }
    add(arm, Op::Ge, lower.clone(), true);

    match kind {
        b'b' => {
            if k < 3 {
                let upper = endpoint_upper(&lower, k, operand_at)?;
                add(arm, Op::Lt, upper, false);
            } else if p.is_full() {
                add(arm, Op::Eq, lower, true);
            }
        }
        b'c' => {
            let field = if lower.major > 0 || k == 1 {
                1
            } else if lower.minor > 0 || k == 2 {
                2
            } else {
                3
            };
            let upper = endpoint_upper(&lower, field, operand_at)?;
            add(arm, Op::Lt, upper, false);
        }
        b't' => {
            let field = if k == 1 { 1 } else { 2 };
            let upper = endpoint_upper(&lower, field, operand_at)?;
            add(arm, Op::Lt, upper, false);
        }
        _ => unreachable!(),
    }
    Ok(())
}

fn parse_arm(tokens: &[(String, usize)]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };

    let minus_positions: Vec<usize> = tokens
        .iter()
        .enumerate()
        .filter_map(|(i, (t, _))| if t == "-" { Some(i) } else { None })
        .collect();

    if let Some(&idx) = minus_positions.first() {
        if minus_positions.len() != 1 || idx != 1 || tokens.len() != 3 {
            return Err(RangeError::Syntax { at: tokens[idx].1 });
        }
        let (lower_s, lower_at) = (&tokens[0].0, tokens[0].1);
        let (upper_s, upper_at) = (&tokens[2].0, tokens[2].1);
        let lower = parse_partial(lower_s, lower_at)?;
        let upper = parse_partial(upper_s, upper_at)?;
        if lower.numeric_count() == 0 {
            return Err(RangeError::Syntax { at: lower_at });
        }
        if upper.numeric_count() == 0 {
            return Err(RangeError::Syntax { at: upper_at });
        }
        let lv = lower.padded();
        add(&mut arm, Op::Ge, lv, true);
        if upper.is_full() {
            let uv = upper.padded();
            add(&mut arm, Op::Le, uv, true);
        } else {
            let uv = upper.padded();
            let k = upper.numeric_count();
            let bound = if k == 1 {
                endpoint_upper(&uv, 1, upper_at)?
            } else if k == 2 {
                endpoint_upper(&uv, 2, upper_at)?
            } else {
                uv
            };
            if k < 3 {
                add(&mut arm, Op::Lt, bound, false);
            } else {
                add(&mut arm, Op::Le, bound, false);
            }
        }
        return Ok(arm);
    }

    for (token, at) in tokens {
        let bytes = token.as_bytes();
        if token == "-" {
            unreachable!();
        }
        if bytes.is_empty() {
            return Err(RangeError::Syntax { at: *at });
        }
        if matches!(bytes[0], b'<' | b'>' | b'=') {
            let (op, skip) = match bytes {
                [b'<', b'=', ..] => (Op::Le, 2),
                [b'>', b'=', ..] => (Op::Ge, 2),
                [b'<', ..] => (Op::Lt, 1),
                [b'>', ..] => (Op::Gt, 1),
                [b'=', ..] => (Op::Eq, 1),
                _ => unreachable!(),
            };
            let operand = &token[skip..];
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at: *at });
            }
            let v = parse_full_operand(operand, *at + skip)?;
            add(&mut arm, op, v, true);
        } else if bytes[0] == b'^' || bytes[0] == b'~' {
            if bytes.len() == 1 {
                return Err(RangeError::Syntax { at: *at });
            }
            let kind = if bytes[0] == b'^' { b'c' } else { b't' };
            parse_expanded_token(
                &mut arm,
                token,
                *at,
                kind,
                &token[1..],
                *at + 1,
            )?;
        } else {
            if !(bytes[0].is_ascii_digit() || matches!(bytes[0], b'*' | b'x' | b'X')) {
                return Err(RangeError::Syntax { at: *at });
            }
            parse_expanded_token(&mut arm, token, *at, b'b', token, *at)?;
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0;
    let mut i = 0;
    let mut saw_any = false;

    while i < bytes.len() {
        while i < bytes.len() && ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        if i + 1 < bytes.len() && bytes[i] == b'|' && bytes[i + 1] == b'|' {
            if !saw_any {
                return Err(RangeError::Empty { at: i });
            }
            let tokens = split_tokens(&s[arm_start..i], arm_start);
            if tokens.is_empty() {
                return Err(RangeError::Empty { at: i });
            }
            arms.push(parse_arm(&tokens)?);
            i += 2;
            while i < bytes.len() && ws(bytes[i]) {
                i += 1;
            }
            arm_start = i;
            saw_any = false;
            if i == bytes.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            let start = i;
            while i < bytes.len() && !ws(bytes[i]) {
                if i + 1 < bytes.len() && bytes[i] == b'|' && bytes[i + 1] == b'|' {
                    break;
                }
                i += 1;
            }
            if i == start {
                return Err(RangeError::Syntax { at: start });
            }
            saw_any = true;
            if i < bytes.len() && bytes[i] == b'|' && i + 1 < bytes.len() && bytes[i + 1] == b'|' {
                continue;
            }
        }
    }

    let tail = &s[arm_start..];
    let tokens = split_tokens(tail, arm_start);
    if tokens.is_empty() {
        if arms.is_empty() {
            return Err(RangeError::Empty { at: 0 });
        }
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&tokens)?);

    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm
                .anchors
                .iter()
                .any(|&(a, b, c)| (a, b, c) == (v.major, v.minor, v.patch))
        {
            return false;
        }
        arm.comparators.iter().all(|c| cmp_match(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(
    candidates: &[String],
    r: &Range,
    maximum: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        if let Some(current) = best {
            let ord = compare_versions(version, &parsed[current]);
            if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                best = Some(index);
            }
        } else {
            best = Some(index);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}
