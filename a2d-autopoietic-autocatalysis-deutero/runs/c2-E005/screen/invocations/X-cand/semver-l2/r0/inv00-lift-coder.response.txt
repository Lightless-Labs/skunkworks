use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn missing_start(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    if !matches!(bytes.get(*pos), Some(b'0'..=b'9')) {
        return if missing_start(bytes.get(*pos).copied()) {
            Err(VersionError::MissingCore { at: *pos })
        } else {
            Err(VersionError::InvalidChar { at: *pos })
        };
    }
    while matches!(bytes.get(*pos), Some(b'0'..=b'9')) {
        *pos += 1;
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end && bytes[p] != b'.' {
            let b = bytes[p];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if id_start == p {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if prerelease
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.len() > 1
            && id.starts_with('0')
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(id.to_string());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for _ in 0..2 {
        match s.as_bytes().get(pos).copied() {
            Some(b'.') => pos += 1,
            b => {
                return if missing_start(b) {
                    Err(VersionError::MissingCore { at: pos })
                } else {
                    Err(VersionError::InvalidChar { at: pos })
                }
            }
        }
        let n = parse_core(s, &mut pos)?;
        if pos == 0 {
            unreachable!();
        }
        if _ == 0 {
            // The first parsed field after major is minor.
            let minor = n;
            match s.as_bytes().get(pos).copied() {
                Some(b'.') => {
                    pos += 1;
                    let patch = parse_core(s, &mut pos)?;
                    return parse_version_tail(s, pos, major, minor, patch);
                }
                b => {
                    return if missing_start(b) {
                        Err(VersionError::MissingCore { at: pos })
                    } else {
                        Err(VersionError::InvalidChar { at: pos })
                    }
                }
            }
        }
    }
    unreachable!()
}

fn parse_version_tail(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();

    match bytes.get(pos).copied() {
        None => {}
        Some(b'-') => {
            pos += 1;
            let build_mark = bytes[pos..].iter().position(|&b| b == b'+').map(|i| pos + i);
            let pre_end = build_mark.unwrap_or(s.len());
            pre = parse_identifiers(s, pos, pre_end, true)?;
            if let Some(bp) = build_mark {
                build = parse_identifiers(s, bp + 1, s.len(), false)?;
            }
        }
        Some(b'+') => {
            build = parse_identifiers(s, pos + 1, s.len(), false)?;
        }
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_numeric(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let xn = is_numeric(x);
                    let yn = is_numeric(y);
                    let ord = match (xn, yn) {
                        (true, true) => cmp_numeric(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Token<'a> {
    text: &'a str,
    start: usize,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str) -> Vec<Token<'_>> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = 0;
    while p < bytes.len() {
        while p < bytes.len() && ws(bytes[p]) {
            p += 1;
        }
        if p == bytes.len() {
            break;
        }
        let start = p;
        while p < bytes.len() && !ws(bytes[p]) {
            p += 1;
        }
        out.push(Token { text: &s[start..p], start });
    }
    out
}

fn version_at(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn endpoint_partial(
    text: &str,
    start: usize,
) -> Result<(Version, usize, bool), RangeError> {
    let has_suffix = text.contains('-') || text.contains('+');
    let fields: Vec<&str> = text.split('.').collect();
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if has_suffix || (fields.len() >= 3 && !has_wild) {
        let v = version_at(text, start)?;
        return Ok((v, 3, true));
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at: start });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in &fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at: start });
        }
        if matches!(*f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !f.bytes().all(|b| b.is_ascii_digit())
                || (f.len() > 1 && f.starts_with('0'))
            {
                return Err(RangeError::Syntax { at: start });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: start })?;
            nums.push(n);
        }
    }
    let k = nums.len();
    if k == 0 && fields.iter().any(|f| !matches!(*f, "*" | "x" | "X")) {
        return Err(RangeError::Syntax { at: start });
    }
    let mut v = Version {
        major: *nums.get(0).unwrap_or(&0),
        minor: *nums.get(1).unwrap_or(&0),
        patch: *nums.get(2).unwrap_or(&0),
        pre: Vec::new(),
        build: Vec::new(),
    };
    if wildcard_seen && nums.len() != fields.iter().take_while(|f| !matches!(**f, "*" | "x" | "X")).count() {
        return Err(RangeError::Syntax { at: start });
    }
    // Partial values have zero-filled absent fields.
    v.pre.clear();
    Ok((v, k, false))
}

fn bump(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut out = Version {
        major: v.major,
        minor: v.minor,
        patch: v.patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    };
    let target = match field {
        0 => &mut out.major,
        1 => &mut out.minor,
        _ => &mut out.patch,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    if field <= 0 {
        out.minor = 0;
        out.patch = 0;
    } else if field == 1 {
        out.patch = 0;
    }
    Ok(out)
}

fn add(c: &mut Vec<Comparator>, op: Op, version: Version, anchor: bool) {
    c.push(Comparator { op, version, anchor });
}

fn parse_operand_token(
    tok: &Token<'_>,
    comps: &mut Vec<Comparator>,
) -> Result<(), RangeError> {
    let text = tok.text;
    let first = text.as_bytes()[0];
    if matches!(first, b'<' | b'>' | b'=') {
        let (op, n) = if text.starts_with("<=") {
            (Op::Le, 2)
        } else if text.starts_with(">=") {
            (Op::Ge, 2)
        } else if first == b'<' {
            (Op::Lt, 1)
        } else if first == b'>' {
            (Op::Gt, 1)
        } else {
            (Op::Eq, 1)
        };
        let operand = &text[n..];
        if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
            return Err(RangeError::Syntax { at: tok.start });
        }
        let v = version_at(operand, tok.start + n)?;
        add(comps, op, v, false);
        return Ok(());
    }

    let (mode, offset) = match first {
        b'^' => (Some(b'^'), 1),
        b'~' => (Some(b'~'), 1),
        _ => (None, 0),
    };
    let operand = &text[offset..];
    let op_start = tok.start + offset;
    if operand.is_empty() {
        return Err(RangeError::Syntax { at: tok.start });
    }
    let (mut lower, k, full) = endpoint_partial(operand, op_start)?;
    if mode.is_none() {
        if full {
            add(comps, Op::Eq, lower.clone(), !lower.pre.is_empty());
        } else if k == 1 || k == 2 {
            add(comps, Op::Ge, lower.clone(), false);
            let upper = bump(&lower, if k == 1 { 0 } else { 1 }, op_start)?;
            add(comps, Op::Lt, upper, false);
        }
        return Ok(());
    }
    if k == 0 {
        return Err(RangeError::Syntax { at: tok.start });
    }
    let anchor = !lower.pre.is_empty();
    add(comps, Op::Ge, lower.clone(), anchor);
    let field = if mode == Some(b'^') {
        if lower.major > 0 || k == 1 {
            0
        } else if lower.minor > 0 || k == 2 {
            1
        } else {
            2
        }
    } else if k == 1 {
        0
    } else {
        1
    };
    lower.pre.clear();
    lower.build.clear();
    let upper = bump(&lower, field, op_start)?;
    add(comps, Op::Lt, upper, false);
    Ok(())
}

fn parse_hyphen_arm(tokens: &[Token<'_>], comps: &mut Vec<Comparator>) -> Result<(), RangeError> {
    let dash = tokens.iter().position(|t| t.text == "-").unwrap();
    if tokens.len() != 3 || dash != 1 {
        return Err(RangeError::Syntax { at: tokens[dash].start });
    }
    let (lo, lk, lf) = endpoint_partial(tokens[0].text, tokens[0].start)?;
    let (hi, hk, hf) = endpoint_partial(tokens[2].text, tokens[2].start)?;
    if lk == 0 {
        return Err(RangeError::Syntax { at: tokens[0].start });
    }
    if hk == 0 {
        return Err(RangeError::Syntax { at: tokens[2].start });
    }
    add(comps, Op::Ge, lo.clone(), !lo.pre.is_empty());
    if hf {
        add(comps, Op::Le, hi.clone(), !hi.pre.is_empty());
    } else if hk == 1 || hk == 2 {
        let upper = bump(&hi, if hk == 1 { 0 } else { 1 }, tokens[2].start)?;
        add(comps, Op::Lt, upper, false);
    } else {
        add(comps, Op::Le, hi, false);
    }
    let _ = lf;
    Ok(())
}

fn parse_arm(tokens: &[Token<'_>]) -> Result<Arm, RangeError> {
    let mut comps = Vec::new();
    if let Some(dash) = tokens.iter().position(|t| t.text == "-") {
        if tokens.len() != 3 || dash != 1 {
            return Err(RangeError::Syntax { at: tokens[dash].start });
        }
        parse_hyphen_arm(tokens, &mut comps)?;
        return Ok(Arm { comparators: comps });
    }
    for tok in tokens {
        let b = tok.text.as_bytes()[0];
        if !matches!(b, b'0'..=b'9' | b'*' | b'x' | b'X' | b'^' | b'~' | b'<' | b'>' | b'=') {
            return Err(RangeError::Syntax { at: tok.start });
        }
        parse_operand_token(tok, &mut comps)?;
    }
    Ok(Arm { comparators: comps })
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].text == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: tokens[i].start });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
            i += 1;
            if i == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            current.push(tokens[i].clone());
            i += 1;
        }
    }
    if current.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&current)?);
    Ok(Range { arms })
}

fn cmp_ok(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let admission = v.pre.is_empty()
            || arm.comparators.iter().any(|c| {
                c.anchor
                    && !c.version.pre.is_empty()
                    && (v.major, v.minor, v.patch)
                        == (c.version.major, c.version.minor, c.version.patch)
            });
        admission && arm.comparators.iter().all(|c| cmp_ok(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        match parse_version(s) {
            Ok(v) => parsed.push(v),
            Err(error) => return Err(CandidateError { index, error }),
        }
    }
    let mut best: Option<usize> = None;
    for i in 0..candidates.len() {
        if !matches(&parsed[i], r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(&parsed[i], &parsed[j]);
            if (want_max && ord == Ordering::Greater) || (!want_max && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}
