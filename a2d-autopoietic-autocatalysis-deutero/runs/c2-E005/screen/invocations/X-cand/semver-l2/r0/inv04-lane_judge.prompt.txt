use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_terminator(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return Err(if is_core_terminator(bytes.get(*pos).copied()) {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }

    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn valid_ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut ident_start = start;

    loop {
        let mut i = ident_start;
        while i < end && bytes[i] != b'.' {
            if !valid_ident_char(bytes[i]) {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        if prerelease {
            let ident = &s[ident_start..i];
            if ident.bytes().all(|b| b.is_ascii_digit())
                && ident.len() > 1
                && ident.as_bytes()[0] == b'0'
            {
                return Err(VersionError::LeadingZero { at: ident_start });
            }
        }
        out.push(s[ident_start..i].to_string());

        if i == end {
            break;
        }
        ident_start = i + 1;
        if ident_start == end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for expected in 0..2 {
        if pos >= s.len() || s.as_bytes()[pos] != b'.' {
            return Err(if is_core_terminator(s.as_bytes().get(pos).copied()) {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        pos += 1;
        let value = parse_core(s, &mut pos)?;
        if expected == 0 {
            let minor = value;
            if pos >= s.len() || s.as_bytes()[pos] != b'.' {
                return Err(if is_core_terminator(s.as_bytes().get(pos).copied()) {
                    VersionError::MissingCore { at: pos }
                } else {
                    VersionError::InvalidChar { at: pos }
                });
            }
            pos += 1;
            let patch = parse_core(s, &mut pos)?;
            let (pre, build) = parse_version_tail(s, pos)?;
            return Ok(Version {
                major,
                minor,
                patch,
                pre,
                build,
            });
        }
    }

    unreachable!()
}

fn parse_version_tail(s: &str, pos: usize) -> Result<(Vec<String>, Vec<String>), VersionError> {
    let bytes = s.as_bytes();
    if pos == bytes.len() {
        return Ok((Vec::new(), Vec::new()));
    }
    if bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    let mut cursor = pos;

    if bytes[cursor] == b'-' {
        cursor += 1;
        let pre_end = bytes[cursor..]
            .iter()
            .position(|&b| b == b'+')
            .map(|n| cursor + n)
            .unwrap_or(bytes.len());
        pre = parse_identifiers(s, cursor, pre_end, true)?;
        cursor = pre_end;
    }

    if cursor < bytes.len() && bytes[cursor] == b'+' {
        cursor += 1;
        build = parse_identifiers(s, cursor, bytes.len(), false)?;
    }

    Ok((pre, build))
}

fn is_numeric_ident(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_ident(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        (false, false) => {}
    }

    for (x, y) in a.pre.iter().zip(&b.pre) {
        let xn = is_numeric_ident(x);
        let yn = is_numeric_ident(y);
        let ord = match (xn, yn) {
            (true, true) => compare_numeric_ident(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.pre.len().cmp(&b.pre.len())
}

#[derive(Clone, Copy)]
struct Token<'a> {
    text: &'a str,
    start: usize,
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str) -> Vec<Token<'_>> {
    let bytes = s.as_bytes();
    let mut tokens = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !is_ws(bytes[i]) {
            i += 1;
        }
        tokens.push(Token {
            text: &s[start..i],
            start,
        });
    }
    tokens
}

struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
    explicit_pre: bool,
}

impl Partial {
    fn numeric_count(&self) -> usize {
        self.nums.len()
    }

    fn padded(&self) -> (u64, u64, u64) {
        (
            self.nums.first().copied().unwrap_or(0),

            self.nums.get(2).copied().unwrap_or(0),

    }

fn looks_full(s: &str) -> bool {
    let fields = split_fields(s);
    s.contains('-') || s.contains('+') || (fields.len() >= 3 && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X")))
}

fn parse_partial(token: Token<'_>) -> Result<Partial, RangeError> {
    let s = token.text;
    if looks_full(s) {
        let version = parse_version(s).map_err(|error| RangeError::Version {
            at: token.start,
            error,
        })?;
        return Ok(Partial {
            nums: vec![version.major, version.minor, version.patch],
            explicit_pre: !version.pre.is_empty(),
            full: Some(version),
        });
    }

    let fields = split_fields(s);
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at: token.start });
    }

    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for field in fields {
        if matches!(field, "*" | "x" | "X") {
            wildcard_seen = true;
            continue;
        }
        if field.is_empty() || wildcard_seen {
            return Err(RangeError::Syntax { at: token.start });
        }
        if !field.bytes().all(|b| b.is_ascii_digit())
            || (field.len() > 1 && field.as_bytes()[0] == b'0')
        {
            return Err(RangeError::Syntax { at: token.start });
        }
        let n = field
            .parse::<u64>()
            .map_err(|_| RangeError::Syntax { at: token.start })?;
        nums.push(n);
    }

    Ok(Partial {
        nums,
        full: None,
        explicit_pre: false,
    })
}

fn add_anchor(arm: &mut Arm, partial: &Partial) {
    if partial.explicit_pre {
        if let Some(v) = &partial.full {
            arm.anchors.push(tuple(v));
        }
    }
}

fn push_comparator(arm: &mut Arm, op: Op, v: Version) {
    let p = Partial {
        explicit_pre: !v.pre.is_empty(),
        nums: vec![v.major, v.minor, v.patch],
        full: Some(v.clone()),
    };
    add_anchor(arm, &p);
    arm.comparators.push(version_at(v, op));
}

fn increment_upper(
    nums: (u64, u64, u64),
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let (mut major, mut minor, mut patch) = nums;
    match field {
        0 => {
            major = major.checked_add(1).ok_or(RangeError::Syntax { at })?;
            Ok(upper_tuple(major, 0, 0))
        }
        1 => {
            minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
            Ok(upper_tuple(major, minor, 0))
        }
        _ => {
            patch = patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
            Ok(upper_tuple(major, minor, patch))
        }
    }
}

    };
    let high = increment_upper((a, b, c), field, operand_start)?;
    arm.comparators.push(version_at(high, Op::Lt));
    Ok(())
}

fn parse_explicit_comparator(token: Token<'_>, arm: &mut Arm) -> Result<(), RangeError> {
    let s = token.text;
    let (op, n) = if s.starts_with(">=") {
        (Op::Ge, 2)
    } else if s.starts_with("<=") {
        (Op::Le, 2)
    } else if s.starts_with('>') {
        (Op::Gt, 1)
    } else if s.starts_with('<') {
        (Op::Lt, 1)
    } else if s.starts_with('=') {
        (Op::Eq, 1)
    } else {
        unreachable!()
    };
    let operand = &s[n..];
    let operand_start = token.start + n;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: token.start });
    }
    let version = parse_version(operand).map_err(|error| RangeError::Version {
        at: operand_start,
        error,
    })?;
    push_comparator(arm, op, version);
    Ok(())
}

fn parse_hyphen_arm(tokens: &[Token<'_>], arm: &mut Arm) -> Result<(), RangeError> {
    let minuses: Vec<usize> = tokens
        .iter()
        .filter(|t| t.text == "-")
        .map(|t| t.start)
        .collect();
    if minuses.len() != 1 || tokens.len() != 3 || tokens[1].text != "-" {
        return Err(RangeError::Syntax {
            at: minuses[0],
        });
    }

    let lower = parse_partial(tokens[0])?;
    let upper = parse_partial(tokens[2])?;
    if lower.numeric_count() == 0 {
        return Err(RangeError::Syntax { at: tokens[0].start });
    }
    if upper.numeric_count() == 0 {
        return Err(RangeError::Syntax { at: tokens[2].start });
    }

    let low_v = lower
        .full
        .clone()
        .unwrap_or_else(|| {
            let (a, b, c) = lower.padded();
            stable_tuple(a, b, c)
        });
    if lower.explicit_pre {
        arm.anchors.push(tuple(&low_v));
    }
    arm.comparators.push(version_at(low_v, Op::Ge));

    if let Some(full) = upper.full.clone() {
        if upper.explicit_pre {
            arm.anchors.push(tuple(&full));
        }
        arm.comparators.push(version_at(full, Op::Le));
    } else {
        let k = upper.numeric_count();
        let field = if k == 1 { 0 } else { 1 };
        let high = increment_upper(upper.padded(), field, tokens[2].start)?;
        arm.comparators.push(version_at(high, Op::Lt));
    }
    Ok(())
}

fn parse_arm(tokens: &[Token<'_>]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };
    if tokens.iter().any(|t| t.text == "-") {
        parse_hyphen_arm(tokens, &mut arm)?;
        return Ok(arm);
    }

    for token in tokens {
        let first = token.text.as_bytes()[0];
        if matches!(first, b'<' | b'>' | b'=') {
            parse_explicit_comparator(*token, &mut arm)?;
        } else if matches!(first, b'^' | b'~' | b'*' | b'x' | b'X')
            || first.is_ascii_digit()
        {
            parse_bare_or_expanded(*token, &mut arm)?;
        } else {
            return Err(RangeError::Syntax { at: token.start });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].text == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: tokens[i].start });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
            i += 1;
            if i == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            current.push(tokens[i]);
            i += 1;
        }
    }
    if current.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&current)?);
    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &Arm) -> bool {
    if !v.pre.is_empty() && !arm.anchors.contains(&tuple(v)) {
        return false;
    }
    arm.comparators.iter().all(|c| c.test(v))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some(index),
            Some(old) => {
                let ord = compare_versions(version, &parsed[old]);
                if (want_max && ord == Ordering::Greater) || (!want_max && ord == Ordering::Less) {
                    best = Some(index);
                }
            }
        }
    }
    Ok(best.map(|index| candidates[index].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;

    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }

    if *pos == start {
        if *pos == bytes.len()
            || bytes[*pos] == b'.'
            || bytes[*pos] == b'-'
            || bytes[*pos] == b'+'
        {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }

    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn valid_identifier_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    base: usize,
    numeric_leading_zero: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut start = 0;

    loop {
        let end = s[start..]
            .find('.')
            .map(|i| start + i)
            .unwrap_or(s.len());

        if end == start {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }

        for i in start..end {
            if !valid_identifier_char(bytes[i]) {
                return Err(VersionError::InvalidChar { at: base + i });
            }
        }

        if numeric_leading_zero
            && bytes[start..end].iter().all(|b| is_digit(*b))
            && end - start > 1
            && bytes[start] == b'0'
        {
            return Err(VersionError::LeadingZero { at: base + start });
        }

        out.push(s[start..end].to_string());

        if end == s.len() {
            break;
        }
        start = end + 1;
        if start == s.len() {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
    }

    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for expected in 0..1 {
        let _ = expected;
        if pos >= s.len() || s.as_bytes()[pos] != b'.' {
            let at = pos;
            if at == s.len()
                || s.as_bytes()[at] == b'-'
                || s.as_bytes()[at] == b'+'
            {
                return Err(VersionError::MissingCore { at });
            }
            return Err(VersionError::InvalidChar { at });
        }
        pos += 1;
    }

    let minor = parse_core(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        let at = pos;
        if at == s.len() || s.as_bytes()[at] == b'-' || s.as_bytes()[at] == b'+' {
            return Err(VersionError::MissingCore { at });
        }
        return Err(VersionError::InvalidChar { at });
    }
    pos += 1;

    let patch = parse_core(s, &mut pos)?;
    if pos < s.len() && s.as_bytes()[pos] != b'-' && s.as_bytes()[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < s.len() && s.as_bytes()[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = s[pre_start..].find('+').map(|i| pre_start + i);
        let pre_end = plus.unwrap_or(s.len());
        pre = parse_identifiers(&s[pre_start..pre_end], pre_start, true)?;
        pos = pre_end;
    }

    if pos < s.len() && s.as_bytes()[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(&s[build_start..], build_start, false)?;
        pos = s.len();
    }

    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(|b| is_digit(*b))
}

fn compare_numeric_identifiers(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (x, y) in a.pre.iter().zip(&b.pre) {
        let xn = is_numeric_identifier(x);
        let yn = is_numeric_identifier(y);
        let order = match (xn, yn) {
            (true, true) => compare_numeric_identifiers(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if order != Ordering::Equal {
            return order;
        }
    }

    a.pre.len().cmp(&b.pre.len())
}

pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    checks: Vec<Check>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Check {
    op: Op,
    version: Version,
    anchor: bool,
}

impl Check {
    fn matches(&self, v: &Version) -> bool {
        let c = compare_versions(v, &self.version);
        match self.op {
            Op::Eq => c == Ordering::Equal,
            Op::Lt => c == Ordering::Less,
            Op::Le => c != Ordering::Greater,
            Op::Gt => c == Ordering::Greater,
            Op::Ge => c != Ordering::Less,
        }
    }
}

fn tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn endpoint(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn upper_endpoint(
    major: u64,
    minor: u64,
    patch: u64,
    pre_zero: bool,
) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: if pre_zero { vec!["0".to_string()] } else { Vec::new() },
        build: Vec::new(),
    }
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

#[derive(Clone, Debug)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
    fields: usize,
}

fn parse_partial(s: &str, base: usize) -> Result<Partial, RangeError> {
    let has_suffix = s.contains('-') || s.contains('+');
    let fields: Vec<&str> = s.split('.').collect();
    let has_wildcard = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));

    if has_suffix || (fields.len() >= 3 && !has_wildcard) {
        let version = parse_version(s).map_err(|error| RangeError::Version { at: base, error })?;
        return Ok(Partial {
            nums: vec![version.major, version.minor, version.patch],
            full: Some(version),
            fields: 3,
        });
    }

    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at: base });
    }

    let mut nums = Vec::new();
    let mut saw_wildcard = false;
    for field in &fields {
        if matches!(*field, "*" | "x" | "X") {
            saw_wildcard = true;
        } else {
            if field.is_empty() || saw_wildcard || !field.as_bytes().iter().all(|b| is_digit(*b)) {
                return Err(RangeError::Syntax { at: base });
            }
            if field.len() > 1 && field.as_bytes()[0] == b'0' {
                return Err(RangeError::Syntax { at: base });
            }
            let n = field.parse::<u64>().map_err(|_| RangeError::Syntax { at: base })?;
            nums.push(n);
        }
    }

    Ok(Partial {
        nums,
        full: None,
        fields: fields.len(),
    })
}

fn padded(p: &Partial) -> (u64, u64, u64) {
    (
        *p.nums.get(0).unwrap_or(&0),
        *p.nums.get(1).unwrap_or(&0),
        *p.nums.get(2).unwrap_or(&0),
    )
}

fn add_check(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() {
        arm.anchors.push(tuple(&v));
    }
    arm.checks.push(Check {
        op,
        version: v,
        anchor,
    });
}

fn bump_error(base: usize) -> RangeError {
    RangeError::Syntax { at: base }
}

fn expand_bare(p: Partial, base: usize, arm: &mut Arm) -> Result<(), RangeError> {
    if let Some(v) = p.full {
        add_check(arm, Op::Eq, v, true);
        return Ok(());
    }
    let k = p.nums.len();
    if k == 0 {
        return Ok(());
    }
    let (a, b, c) = padded(&p);
    add_check(arm, Op::Ge, endpoint(a, b, c), false);
    if k == 1 {
        let next = a.checked_add(1).ok_or_else(|| bump_error(base))?;
        add_check(arm, Op::Lt, upper_endpoint(next, 0, 0, true), false);
    } else if k == 2 {
        let next = b.checked_add(1).ok_or_else(|| bump_error(base))?;
        add_check(arm, Op::Lt, upper_endpoint(a, next, 0, true), false);
    }
    Ok(())
}

fn expand_caret_or_tilde(
    p: Partial,
    base: usize,
    caret: bool,
    arm: &mut Arm,
) -> Result<(), RangeError> {
    let k = p.nums.len();
    if k == 0 {
        return Err(RangeError::Syntax { at: base });
    }

    let (a, b, c) = if let Some(ref v) = p.full {
        (v.major, v.minor, v.patch)
    } else {
        padded(&p)
    };
    let lower = p.full.clone().unwrap_or_else(|| endpoint(a, b, c));
    add_check(arm, Op::Ge, lower, true);

    let (u, v, w) = if caret {
        if a > 0 || k == 1 {
            (a.checked_add(1).ok_or_else(|| bump_error(base))?, 0, 0)
        } else if b > 0 || k == 2 {
            (a, b.checked_add(1).ok_or_else(|| bump_error(base))?, 0)
        } else {
            (a, b, c.checked_add(1).ok_or_else(|| bump_error(base))?)
        }
    } else if k == 1 {
        (a.checked_add(1).ok_or_else(|| bump_error(base))?, 0, 0)
    } else {
        (a, b.checked_add(1).ok_or_else(|| bump_error(base))?, 0)
    };

    add_check(arm, Op::Lt, upper_endpoint(u, v, w, true), false);
    Ok(())
}

fn parse_full_comparator(
    token: &str,
    token_start: usize,
    op: Op,
    prefix_len: usize,
    arm: &mut Arm,
) -> Result<(), RangeError> {
    let operand_start = token_start + prefix_len;
    if operand_start >= token_start + token.len()
        || !token.as_bytes()[prefix_len].is_ascii_digit()
    {
        return Err(RangeError::Syntax { at: token_start });
    }
    let operand = &token[prefix_len..];
    let version =
        parse_version(operand).map_err(|error| RangeError::Version { at: operand_start, error })?;
    add_check(arm, op, version, true);
    Ok(())
}

fn parse_hyphen_arm(
    tokens: &[(usize, &str)],
    arm: &mut Arm,
) -> Result<(), RangeError> {
    let minus = tokens.iter().find(|(_, t)| *t == "-").unwrap().0;
    if tokens.len() != 3 || tokens[1].1 != "-" {
        return Err(RangeError::Syntax { at: minus });
    }

    let (lo_start, lo_s) = tokens[0];
    let (hi_start, hi_s) = tokens[2];
    let lo = parse_partial(lo_s, lo_start)?;
    let hi = parse_partial(hi_s, hi_start)?;

    if lo.nums.is_empty() {
        return Err(RangeError::Syntax { at: lo_start });
    }
    if hi.nums.is_empty() {
        return Err(RangeError::Syntax { at: hi_start });
    }

    let (a, b, c) = if let Some(ref v) = lo.full {
        (v.major, v.minor, v.patch)
    } else {
        padded(&lo)
    };
    let lower = lo.full.clone().unwrap_or_else(|| endpoint(a, b, c));
    add_check(arm, Op::Ge, lower, true);

    if let Some(v) = hi.full {
        add_check(arm, Op::Le, v, true);
    } else {
        let (a, b, c) = padded(&hi);
        let (u, v, w) = if hi.nums.len() == 1 {
            (a.checked_add(1).ok_or_else(|| bump_error(hi_start))?, 0, 0)
        } else {
            (a, b.checked_add(1).ok_or_else(|| bump_error(hi_start))?, 0)
        };
        let _ = c;
        add_check(arm, Op::Lt, upper_endpoint(u, v, w, true), false);
    }
    Ok(())
}

fn parse_arm(
    s: &str,
    base: usize,
    arm: &mut Arm,
) -> Result<(), RangeError> {
    let bytes = s.as_bytes();
    let mut tokens = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !is_ws(bytes[i]) {
            i += 1;
        }
        tokens.push((base + start, &s[start..i]));
    }

    if tokens.iter().any(|(_, t)| *t == "-") {
        return parse_hyphen_arm(&tokens, arm);
    }

    for (start, token) in tokens {
        let first = token.as_bytes()[0];
        match first {
            b'<' | b'>' | b'=' => {
                let (op, len) = if token.starts_with("<=") {
                    (Op::Le, 2)
                } else if token.starts_with(">=") {
                    (Op::Ge, 2)
                } else if token.starts_with('<') {
                    (Op::Lt, 1)
                } else if token.starts_with('>') {
                    (Op::Gt, 1)
                } else {
                    (Op::Eq, 1)
                };
                parse_full_comparator(token, start, op, len, arm)?;
            }
            b'^' | b'~' => {
                let p = parse_partial(&token[1..], start + 1)?;
                expand_caret_or_tilde(p, start, first == b'^', arm)?;
            }
            b'*' | b'x' | b'X' | b'0'..=b'9' => {
                let p = parse_partial(token, start)?;
                expand_bare(p, start, arm)?;
            }
            _ => return Err(RangeError::Syntax { at: start }),
        }
    }

    Ok(())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0;
    let mut i = 0;
    let mut found_any_non_ws = false;
    let mut previous_separator = None;

    while i < bytes.len() {
        if !is_ws(bytes[i]) {
            found_any_non_ws = true;
        }
        if bytes[i] == b'|' && i + 1 < bytes.len() && bytes[i + 1] == b'|' {
            let before = &s[arm_start..i];
            if before.bytes().all(is_ws) {
                return Err(RangeError::Empty { at: i });
            }
            let mut arm = Arm {
                checks: Vec::new(),
                anchors: Vec::new(),
            };
            parse_arm(before, arm_start, &mut arm)?;
            arms.push(arm);
            previous_separator = Some(i);
            i += 2;
            arm_start = i;
        } else {
            i += 1;
        }
    }

    let trailing = &s[arm_start..];
    if trailing.bytes().all(is_ws) {
        if let Some(_) = previous_separator {
            return Err(RangeError::Empty { at: s.len() });
        }
        if !found_any_non_ws {
            return Err(RangeError::Empty { at: 0 });
        }
    } else {
        let mut arm = Arm {
            checks: Vec::new(),
            anchors: Vec::new(),
        };
        parse_arm(trailing, arm_start, &mut arm)?;
        arms.push(arm);
    }

    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &Arm) -> bool {
    if !v.pre.is_empty() && !arm.anchors.contains(&tuple(v)) {
        return false;
    }
    arm.checks.iter().all(|c| c.matches(v))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (i, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some(i),
            Some(j) => {
                let order = compare_versions(version, &parsed[j]);
                if (want_max && order == Ordering::Greater)
                    || (!want_max && order == Ordering::Less)
                {
                    best = Some(i);
                }
            }
        }
    }

    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return Err(if start == bytes.len()
            || matches!(bytes[start], b'.' | b'-' | b'+')
        {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut ident_start = start;
    let mut i = start;
    loop {
        if i == end || bytes[i] == b'.' {
            if i == ident_start {
                return Err(VersionError::EmptyIdentifier { at: ident_start });
            }
            let ident = &s[ident_start..i];
            if is_pre
                && ident.len() > 1
                && ident.bytes().all(|b| b.is_ascii_digit())
                && ident.starts_with('0')
            {
                return Err(VersionError::LeadingZero { at: ident_start });
            }
            result.push(ident.to_string());
            if i == end {
                break;
            }
            i += 1;
            ident_start = i;
            continue;
        }
        if !bytes[i].is_ascii_alphanumeric() && bytes[i] != b'-' {
            return Err(VersionError::InvalidChar { at: i });
        }
        i += 1;
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    for expected in 0..2 {
        if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        if s.as_bytes()[pos] != b'.' {
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        let n = parse_core_number(s, &mut pos)?;
        if expected == 0 {
            let minor = n;
            if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if s.as_bytes()[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
            let patch = parse_core_number(s, &mut pos)?;
            return parse_version_tail(s, pos, major, minor, patch);
        }
    }
    unreachable!()
}

fn parse_version_tail(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }
    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let plus = s[pos..].find('+').map(|i| pos + i).unwrap_or(s.len());
        pre = parse_identifiers(s, pos, plus, true)?;
        pos = plus;
    }
    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(s, pos, s.len(), false)?;
        pos = s.len();
    }
    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric(x);
        let yn = is_numeric(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn cmp_match(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens(s: &str, base: usize) -> Vec<(String, usize)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && ws(b[i]) {
            i += 1;
        }
        if i == b.len() {
            break;
        }
        let start = i;
        while i < b.len() && !ws(b[i]) {
            i += 1;
        }
        out.push((s[start..i].to_string(), base + start));
    }
    out
}

fn parse_full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

#[derive(Clone, Debug)]
enum Partial {
    Full(Version),
    Fields(Vec<Option<u64>>),
}

impl Partial {
    fn numeric_count(&self) -> usize {
        match self {
            Partial::Full(_) => 3,
            Partial::Fields(v) => v.iter().filter(|x| x.is_some()).count(),
        }
    }

    fn padded(&self) -> Version {
        match self {
            Partial::Full(v) => v.clone(),
            Partial::Fields(fields) => Version {
                major: fields.get(0).and_then(|x| *x).unwrap_or(0),
                minor: fields.get(1).and_then(|x| *x).unwrap_or(0),
                patch: fields.get(2).and_then(|x| *x).unwrap_or(0),
                pre: Vec::new(),
                build: Vec::new(),
            },
        }
    }

    fn is_full(&self) -> bool {
        matches!(self, Partial::Full(_))
    }
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix = s.contains('-') || s.contains('+');
    let raw: Vec<&str> = s.split('.').collect();
    let has_three_no_wild = raw.len() >= 3
        && !raw.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if has_suffix || has_three_no_wild {
        return parse_full_operand(s, at).map(Partial::Full);
    }

    if raw.is_empty() || raw.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut fields = Vec::new();
    let mut seen_wild = false;
    for field in raw {
        if field.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(field, "*" | "x" | "X") {
            seen_wild = true;
            fields.push(None);
        } else if field.bytes().all(|b| b.is_ascii_digit()) {
            if seen_wild
                || (field.len() > 1 && field.starts_with('0'))
                || field.parse::<u64>().is_err()
            {
                return Err(RangeError::Syntax { at });
            }
            fields.push(Some(field.parse::<u64>().unwrap()));
        } else {
            return Err(RangeError::Syntax { at });
        }
    }
    Ok(Partial::Fields(fields))
}

fn endpoint_upper(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut out = Version {
        major: v.major,
        minor: v.minor,
        patch: v.patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    };
    let fail = || RangeError::Syntax { at };
    match field {
        1 => {
            out.major = v.major.checked_add(1).ok_or_else(fail)?;
            out.minor = 0;
            out.patch = 0;
        }
        2 => {
            out.minor = v.minor.checked_add(1).ok_or_else(fail)?;
            out.patch = 0;
        }
        3 => out.patch = v.patch.checked_add(1).ok_or_else(fail)?,
        _ => {}
    }
    Ok(out)
}

fn add(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors
            .push((version.major, version.minor, version.patch));
    }
    arm.comparators.push(Comparator { op, version });
}

fn parse_expanded_token(
    arm: &mut Arm,
    token: &str,
    token_at: usize,
    kind: u8,
    operand: &str,
    operand_at: usize,
) -> Result<(), RangeError> {
    let p = parse_partial(operand, operand_at)?;
    let k = p.numeric_count();
    if kind != b'b' && k == 0 {
        return Err(RangeError::Syntax { at: token_at });
    }
    let lower = p.padded();
    if kind == b'b' && k == 0 {
        return Ok(());
    }
    add(arm, Op::Ge, lower.clone(), true);

    match kind {
        b'b' => {
            if k < 3 {
                let upper = endpoint_upper(&lower, k, operand_at)?;
                add(arm, Op::Lt, upper, false);
            } else if p.is_full() {
                add(arm, Op::Eq, lower, true);
            }
        }
        b'c' => {
            let field = if lower.major > 0 || k == 1 {
                1
            } else if lower.minor > 0 || k == 2 {
                2
            } else {
                3
            };
            let upper = endpoint_upper(&lower, field, operand_at)?;
            add(arm, Op::Lt, upper, false);
        }
        b't' => {
            let field = if k == 1 { 1 } else { 2 };
            let upper = endpoint_upper(&lower, field, operand_at)?;
            add(arm, Op::Lt, upper, false);
        }
        _ => unreachable!(),
    }
    Ok(())
}

fn parse_arm(tokens: &[(String, usize)]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    };

    let minus_positions: Vec<usize> = tokens
        .iter()
        .enumerate()
        .filter_map(|(i, (t, _))| if t == "-" { Some(i) } else { None })
        .collect();

    if let Some(&idx) = minus_positions.first() {
        if minus_positions.len() != 1 || idx != 1 || tokens.len() != 3 {
            return Err(RangeError::Syntax { at: tokens[idx].1 });
        }
        let (lower_s, lower_at) = (&tokens[0].0, tokens[0].1);
        let (upper_s, upper_at) = (&tokens[2].0, tokens[2].1);
        let lower = parse_partial(lower_s, lower_at)?;
        let upper = parse_partial(upper_s, upper_at)?;
        if lower.numeric_count() == 0 {
            return Err(RangeError::Syntax { at: lower_at });
        }
        if upper.numeric_count() == 0 {
            return Err(RangeError::Syntax { at: upper_at });
        }
        let lv = lower.padded();
        add(&mut arm, Op::Ge, lv, true);
        if upper.is_full() {
            let uv = upper.padded();
            add(&mut arm, Op::Le, uv, true);
        } else {
            let uv = upper.padded();
            let k = upper.numeric_count();
            let bound = if k == 1 {
                endpoint_upper(&uv, 1, upper_at)?
            } else if k == 2 {
                endpoint_upper(&uv, 2, upper_at)?
            } else {
                uv
            };
            if k < 3 {
                add(&mut arm, Op::Lt, bound, false);
            } else {
                add(&mut arm, Op::Le, bound, false);
            }
        }
        return Ok(arm);
    }

    for (token, at) in tokens {
        let bytes = token.as_bytes();
        if token == "-" {
            unreachable!();
        }
        if bytes.is_empty() {
            return Err(RangeError::Syntax { at: *at });
        }
        if matches!(bytes[0], b'<' | b'>' | b'=') {
            let (op, skip) = match bytes {
                [b'<', b'=', ..] => (Op::Le, 2),
                [b'>', b'=', ..] => (Op::Ge, 2),
                [b'<', ..] => (Op::Lt, 1),
                [b'>', ..] => (Op::Gt, 1),
                [b'=', ..] => (Op::Eq, 1),
                _ => unreachable!(),
            };
            let operand = &token[skip..];
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at: *at });
            }
            let v = parse_full_operand(operand, *at + skip)?;
            add(&mut arm, op, v, true);
        } else if bytes[0] == b'^' || bytes[0] == b'~' {
            if bytes.len() == 1 {
                return Err(RangeError::Syntax { at: *at });
            }
            let kind = if bytes[0] == b'^' { b'c' } else { b't' };
            parse_expanded_token(
                &mut arm,
                token,
                *at,
                kind,
                &token[1..],
                *at + 1,
            )?;
        } else {
            if !(bytes[0].is_ascii_digit() || matches!(bytes[0], b'*' | b'x' | b'X')) {
                return Err(RangeError::Syntax { at: *at });
            }
            parse_expanded_token(&mut arm, token, *at, b'b', token, *at)?;
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0;
    let mut i = 0;
    let mut saw_any = false;

    while i < bytes.len() {
        while i < bytes.len() && ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        if i + 1 < bytes.len() && bytes[i] == b'|' && bytes[i + 1] == b'|' {
            if !saw_any {
                return Err(RangeError::Empty { at: i });
            }
            let tokens = split_tokens(&s[arm_start..i], arm_start);
            if tokens.is_empty() {
                return Err(RangeError::Empty { at: i });
            }
            arms.push(parse_arm(&tokens)?);
            i += 2;
            while i < bytes.len() && ws(bytes[i]) {
                i += 1;
            }
            arm_start = i;
            saw_any = false;
            if i == bytes.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            let start = i;
            while i < bytes.len() && !ws(bytes[i]) {
                if i + 1 < bytes.len() && bytes[i] == b'|' && bytes[i + 1] == b'|' {
                    break;
                }
                i += 1;
            }
            if i == start {
                return Err(RangeError::Syntax { at: start });
            }
            saw_any = true;
            if i < bytes.len() && bytes[i] == b'|' && i + 1 < bytes.len() && bytes[i + 1] == b'|' {
                continue;
            }
        }
    }

    let tail = &s[arm_start..];
    let tokens = split_tokens(tail, arm_start);
    if tokens.is_empty() {
        if arms.is_empty() {
            return Err(RangeError::Empty { at: 0 });
        }
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&tokens)?);

    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm
                .anchors
                .iter()
                .any(|&(a, b, c)| (a, b, c) == (v.major, v.minor, v.patch))
        {
            return false;
        }
        arm.comparators.iter().all(|c| cmp_match(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]

    pub error: VersionError,
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,

    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(
    candidates: &[String],
    r: &Range,
    maximum: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        if let Some(current) = best {
            let ord = compare_versions(version, &parsed[current]);
            if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                best = Some(index);
            }
        } else {
            best = Some(index);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}
