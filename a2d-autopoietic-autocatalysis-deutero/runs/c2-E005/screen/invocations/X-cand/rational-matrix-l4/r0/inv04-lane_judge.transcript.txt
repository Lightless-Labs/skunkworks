        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero();
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows:self.rows, cols:rhs.cols, data })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self.clone(), self.cols, self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows:self.rows, cols:self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = vec![zero(); len];
        for r in 0..n {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
            }
            data[r * width + n + r] = one();
        }
        let red = eliminate(Self { rows:n, cols:width, data }, n, n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank:red.pivots.len() });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            out.extend_from_slice(&red.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Self { rows:n, cols:n, data:out })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        let m = self.rows;
        let n = self.cols;
        let k = rhs.cols;
        if m != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients:m, rhs:rhs.rows });
        }
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = m.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = vec![zero(); len];
        for r in 0..m {
            data[r * width..r * width + n]
                .copy_from_slice(&self.data[r * n..r * n + n]);
            data[r * width + n..r * width + width]
                .copy_from_slice(&rhs.data[r * k..r * k + k]);
        }
        let red = eliminate(Self { rows:m, cols:width, data }, n, n)
            .map_err(SolveError::Matrix)?;

        for r in 0..m {
            let coeff_zero = red.matrix.data[r * width..r * width + n].iter().all(|v| v.num == 0);
            if coeff_zero {
                if let Some(j) = (0..k).find(|&j| red.matrix.data[r * width + n + j].num != 0) {
                    return Ok(Solution::None {
                        row:r, rhs_col:j,
                        witness:red.matrix.data[r * width..r * width + width].to_vec(),
                    });
                }
            }
        }

        let mut free = Vec::new();
        let mut pivot_mask = vec![false; n];
        for &p in &red.pivots { pivot_mask[p] = true; }
        for c in 0..n {
            if !pivot_mask[c] { free.push(c); }
        }

        let pk = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular = vec![zero(); pk];
        for (i, &p) in red.pivots.iter().enumerate() {
            for j in 0..k {
                particular[p * k + j] = red.matrix.data[i * width + n + j];
            }
        }
        let particular = Self { rows:n, cols:k, data:particular };
        if red.pivots.len() == n {
            return Ok(Solution::Unique { values:particular, pivots:red.pivots });
        }

        let f = free.len();
        let bf = n.checked_mul(f).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![zero(); bf];
        for (j, &free_col) in free.iter().enumerate() {
            basis[free_col * f + j] = one();
            for (i, &p) in red.pivots.iter().enumerate() {
                let coeff = red.matrix.data[i * width + free_col];
                basis[p * f + j] = coeff.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row:p, col:j, step:red.pivots.len(), source,
                    })
                })?;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis:Self { rows:n, cols:f, data:basis },
            pivots:red.pivots,
        })
    }
}

pub enum SolveError {

    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}

fn zero() -> Rational { Rational { num:0, den:1 } }
fn one() -> Rational { Rational { num:1, den:1 } }

fn eliminate(
    mut matrix: Matrix,
    pivot_limit: usize,
    pivot_step_base: usize,
) -> Result<Reduction, MatrixError> {
    let rows = matrix.rows;
    let cols = matrix.cols;
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;

    for candidate in 0..pivot_limit {
        if p == rows { break; }
        let mut best: Option<usize> = None;
        for r in p..rows {
            let v = matrix.data[r * cols + candidate];
            if v.num == 0 { continue; }
            let key = (v.den, v.num.unsigned_abs(), r);
            if best.map_or(true, |br| {
                let b = matrix.data[br * cols + candidate];
                key < (b.den, b.num.unsigned_abs(), br)
            }) {
                best = Some(r);
            }
        }
        let Some(q) = best else { continue; };
        if q != p {
            for c in 0..cols {
                matrix.data.swap(p * cols + c, q * cols + c);
            }
            swaps.push((p,q));
        }
        let pivot = matrix.data[p * cols + candidate];
        for c in 0..cols {
            let current = matrix.data[p * cols + c];
            matrix.data[p * cols + c] = current.div(&pivot).map_err(|source| {
                MatrixError::Arithmetic { row:p, col:c, step:pivot_step_base + p, source }
            })?;
        }
        for r in 0..rows {
            if r == p { continue; }
            let factor = matrix.data[r * cols + candidate];
            if factor.num == 0 { continue; }
            for c in 0..cols {
                let product = factor.mul(&matrix.data[p * cols + c]).map_err(|source| {
                    MatrixError::Arithmetic { row:r, col:c, step:pivot_step_base + p, source }
                })?;
                matrix.data[r * cols + c].sub(&product).map_err(|source| {
                    MatrixError::Arithmetic { row:r, col:c, step:pivot_step_base + p, source }
                }).map(|v| matrix.data[r * cols + c] = v)?;
            }
        }
        pivots.push(candidate);
        p += 1;
    }
    Ok(Reduction { matrix, pivots, swaps })
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a

impl Rational {
    fn from_i128_parts(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        if d < 0 {
            n = -n;
            d = -d;
        }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        if b.is_empty() {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::Empty });
        }
        let mut i = 0;
        if b[0] == b'-' { i = 1; }
        let ns = i;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == ns {
            return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
        }
        let ne = i;
        let mut ds = None;
        let mut de = None;
        if i < b.len() {
            if b[i] != b'/' {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
            i += 1;
            ds = Some(i);
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if i == ds.unwrap() {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
            de = Some(i);
            if i < b.len() {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
        }

        let parse_err = |at, kind| RationalError::Parse { at, kind };
        let neg = b[0] == b'-';
        let nstart = if neg { 1 } else { 0 };
        let ndigits = ne - nstart;
        if ndigits >= 2 && b[nstart] == b'0' {
            return Err(parse_err(nstart + 1, ParseKind::NonCanonical));
        }
        if neg && ndigits == 1 && b[1] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }
        if let (Some(s), Some(e)) = (ds, de) {
            if e - s >= 2 && b[s] == b'0' {
                return Err(parse_err(s + 1, ParseKind::NonCanonical));
            }
        }

        let limit: u128 = if neg { 1u128 << 63 } else { i64::MAX as u128 };
        let mut nm = 0u128;
        for &c in &b[nstart..ne] {
            nm = nm.saturating_mul(10).saturating_add((c - b'0') as u128);
            if nm > limit {
                return Err(parse_err(0, ParseKind::OutOfRange));
            }
        }
        let mut dm = 1u128;
        if let (Some(s), Some(e)) = (ds, de) {
            dm = 0;
            for &c in &b[s..e] {

                return Err(parse_err(s - 1, ParseKind::NonCanonical));
            }
        }
        let n = if neg { -(nm as i128) } else { nm as i128 };
        Self::from_i128_parts(n, dm as i128)
    }

}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols), right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|e| MatrixError::Arithmetic { row: r, col: c, step: k, source: e })?;
                    acc = acc.add(&term).map_err(|e| MatrixError::Arithmetic { row: r, col: c, step: k, source: e })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    fn eliminate(&self, candidate_cols: usize) -> Result<Reduction, MatrixError> {
        let mut m = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;
        for col in 0..candidate_cols {
            if p == m.rows { break; }
            let mut best: Option<usize> = None;
            for row in p..m.rows {
                let x = m.data[row * m.cols + col];
                if x.num == 0 { continue; }
                let key = (x.den, x.num.unsigned_abs(), row);
                if best.map_or(true, |br| {
                    let y = m.data[br * m.cols + col];
                    key < (y.den, y.num.unsigned_abs(), br)
                }) {
                    best = Some(row);
                }
            }
            let Some(q) = best else { continue };
            if q != p {
                for c in 0..m.cols { m.data.swap(p * m.cols + c, q * m.cols + c); }
                swaps.push((p, q));
            }
            let pivot = m.data[p * m.cols + col];
            for c in 0..m.cols {
                let v = m.data[p * m.cols + c].div(&pivot).map_err(|e| MatrixError::Arithmetic { row: p, col: c, step: p, source: e })?;
                m.data[p * m.cols + c] = v;
            }
            for row in 0..m.rows {
                if row == p { continue; }
                let factor = m.data[row * m.cols + col];
                if factor.num == 0 { continue; }
                for c in 0..m.cols {
                    let prod = factor.mul(&m.data[p * m.cols + c]).map_err(|e| MatrixError::Arithmetic { row, col: c, step: p, source: e })?;
                    let v = m.data[row * m.cols + c].sub(&prod).map_err(|e| MatrixError::Arithmetic { row, col: c, step: p, source: e })?;
                    m.data[row * m.cols + c] = v;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction { matrix: m, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> { self.eliminate(self.cols) }

    pub fn rank(&self) -> Result<usize, MatrixError> { Ok(self.rref()?.rank()) }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c { Rational { num: 1, den: 1 } } else { Rational { num: 0, den: 1 } });
            }
        }
        let aug = Self { rows: n, cols: width, data };
        let red = aug.eliminate(n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank: red.pivots.len() });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            out.extend_from_slice(&red.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Self { rows: n, cols: n, data: out })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients: self.rows, rhs: rhs.rows });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            data.extend_from_slice(&self.data[r * n..r * n + n]);
            data.extend_from_slice(&rhs.data[r * k..r * k + k]);
        }
        let aug = Self { rows: self.rows, cols: width, data };
        let red = aug.eliminate(n).map_err(SolveError::Matrix)?;

        for r in 0..self.rows {
            if red.matrix.data[r * width..r * width + n].iter().all(|x| x.num == 0) {
                if let Some(local) = red.matrix.data[r * width + n..r * width + width].iter().position(|x| x.num != 0) {
                    return Ok(Solution::None {
                        row: r,
                        rhs_col: local,
                        witness: red.matrix.data[r * width..r * width + width].to_vec(),
                    });
                }
            }
        }

        let rank = red.pivots.len();
        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular = vec![Rational { num: 0, den: 1 }; particular_len];
        for (j, &pc) in red.pivots.iter().enumerate() {
            for c in 0..k {
                particular[pc * k + c] = red.matrix.data[j * width + n + c];
            }
        }
        let particular = Matrix { rows: n, cols: k, data: particular };
        if rank == n {
            return Ok(Solution::Unique { values: particular, pivots: red.pivots });
        }

        let mut free = Vec::new();
        let mut pi = 0;
        for c in 0..n {
            if pi < red.pivots.len() && red.pivots[pi] == c { pi += 1; }
            else { free.push(c); }
        }
        let f = free.len();
        let basis_len = n.checked_mul(f).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![Rational { num: 0, den: 1 }; basis_len];
        for (j, &fc) in free.iter().enumerate() {
            basis[fc * f + j] = Rational { num: 1, den: 1 };
            for (i, &pc) in red.pivots.iter().enumerate() {
                let v = red.matrix.data[i * width + fc].neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic { row: pc, col: j, step: rank, source })
                })?;
                basis[pc * f + j] = v;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix { rows: n, cols: f, data: basis },
            pivots: red.pivots,
        })
    }
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {

        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })

            return Err(RationalError::Parse { at: 0, kind: ParseKind::Empty });
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
        }
        let slash = if i < bytes.len() && bytes[i] == b'/' {
            let s = i;
            i += 1;
            Some(s)
        } else {
            None
        };
        let den_start = i;
        if slash.is_some() {
            while i < bytes.len() && bytes[i].is_ascii_digit() {

            }

                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
        }
        if i < bytes.len() {
            return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
        }

        let num_digits = &bytes[num_start..slash.unwrap_or(bytes.len())];
        if num_digits.len() > 1 && num_digits[0] == b'0' {
            return Err(RationalError::Parse {
                at: num_start + 1,
                kind: ParseKind::NonCanonical,
            });

            return Err(RationalError::Parse { at: 0, kind: ParseKind::NonCanonical });
        }
        if let Some(s) = slash {
            let den_digits = &bytes[den_start..];

                return Err(RationalError::Parse {
                    at: den_start + 1,
                    kind: ParseKind::NonCanonical,
                });

            let limit: u128 = if bytes[0] == b'-' { 1u128 << 63 } else { i64::MAX as u128 };
            let mut mag = 0u128;
            for &b in num_digits {
                mag = mag.saturating_mul(10).saturating_add((b - b'0') as u128);
                if mag > limit {
                    return Err(RationalError::Parse { at: 0, kind: ParseKind::OutOfRange });
                }

            let mut den = 0u128;
            for &b in den_digits {
                den = den.saturating_mul(10).saturating_add((b - b'0') as u128);

            }

pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut out = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let term = self.data[r * self.cols + k]
                        .mul(&rhs.data[k * rhs.cols + c])
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term)
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                out.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data: out })
    }

    fn eliminate(&mut self, candidates: usize) -> Result<(Vec<usize>, Vec<(usize, usize)>), MatrixError> {
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0;
        for col in 0..candidates {
            if p == self.rows { break; }
            let mut best: Option<usize> = None;
            for r in p..self.rows {
                let v = self.data[r * self.cols + col];
                if v.num == 0 { continue; }
                let key = (v.den, v.num.unsigned_abs(), r);
                if best.map_or(true, |br| {
                    let b = self.data[br * self.cols + col];
                    key < (b.den, b.num.unsigned_abs(), br)
                }) {
                    best = Some(r);
                }
            }
            let Some(q) = best else { continue };
            if q != p {
                for c in 0..self.cols {
                    self.data.swap(p * self.cols + c, q * self.cols + c);
                }
                swaps.push((p, q));
            }
            let pivot = self.data[p * self.cols + col];
            for c in 0..self.cols {
                let original = self.data[p * self.cols + c];
                self.data[p * self.cols + c] = original.div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic { row: p, col: c, step: p, source })?;
            }
            for r in 0..self.rows {
                if r == p { continue; }
                let factor = self.data[r * self.cols + col];
                if factor.num == 0 { continue; }
                for c in 0..self.cols {
                    let product = factor.mul(&self.data[p * self.cols + c])
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
                    self.data[r * self.cols + c] = self.data[r * self.cols + c].sub(&product)
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok((pivots, swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let (pivots, swaps) = matrix.eliminate(self.cols)?;
        Ok(Reduction { matrix, pivots, swaps })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c { Rational { num: 1, den: 1 } } else { Rational { num: 0, den: 1 } });
            }
        }
        let mut aug = Self { rows: n, cols: width, data };
        let (pivots, _) = aug.eliminate(n)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }
        let mut result = Vec::with_capacity(n * n);
        for r in 0..n {
            result.extend_from_slice(&aug.data[r * width + n..r * width + width]);
        }
        Ok(Self { rows: n, cols: n, data: result })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients: self.rows, rhs: rhs.rows });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            data.extend_from_slice(&self.data[r * n..r * n + n]);
            data.extend_from_slice(&rhs.data[r * k..r * k + k]);
        }
        let mut aug = Self { rows: self.rows, cols: width, data };
        let (pivots, _) = aug.eliminate(n).map_err(SolveError::Matrix)?;

        for r in 0..aug.rows {
            if aug.data[r * width..r * width + n].iter().all(|v| v.num == 0) {
                if let Some(c) = (0..k).find(|&c| aug.data[r * width + n + c].num != 0) {
                    return Ok(Solution::None {
                        row: r,
                        rhs_col: c,
                        witness: aug.data[r * width..(r + 1) * width].to_vec(),
                    });
                }
            }
        }

        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![Rational { num: 0, den: 1 }; particular_len];
        for (j, &pc) in pivots.iter().enumerate() {
            for c in 0..k {
                particular_data[pc * k + c] = aug.data[j * width + n + c];
            }
        }
        let particular = Self { rows: n, cols: k, data: particular_data };
        if pivots.len() == n {
            return Ok(Solution::Unique { values: particular, pivots });
        }
        let free: Vec<usize> = (0..n).filter(|c| !pivots.contains(c)).collect();
        let f = free.len();
        let basis_len = n.checked_mul(f).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![Rational { num: 0, den: 1 }; basis_len];
        for (j, &fc) in free.iter().enumerate() {
            basis_data[fc * f + j] = Rational { num: 1, den: 1 };
            for (i, &pc) in pivots.iter().enumerate() {
                let value = aug.data[i * width + fc].neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic { row: pc, col: j, step: pivots.len(), source })
                })?;
                basis_data[pc * f + j] = value;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Self { rows: n, cols: f, data: basis_data },
            pivots,
        })
    }
}

        let g = gcd_u128(da as u128, db as u128) as i128;
        let l = da / g;
        let r = db / g;

        let r = db / g;

        let g1 = gcd_u128(a.unsigned_abs(), db as u128) as i128;

            }

            if i < bytes.len() {

            }
        }

        if num_digits >= 2 && bytes[num_start] == b'0' {
            return Err(err(num_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == 1 && bytes[num_start] == b'0' {

        }
        if let Some(ds) = den_start {
            let de = den_end.unwrap();
            if de - ds >= 2 && bytes[ds] == b'0' {

            let mag = parse_mag(ds, de, i64::MAX as u128)
                .ok_or_else(|| err(ds, ParseKind::OutOfRange))?;
            if mag == 0 {
                return Err(err(ds, ParseKind::ZeroDenominator));
            }
            d = mag as i64;
            let slash = ds - 1;
            if d == 1 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
            if n == 0 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
            if gcd_u128(n.unsigned_abs() as u128, d as u128) > 1 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
        }
        Ok(Self { num: n, den: d })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

fn checked_len(rows: usize, cols: usize) -> Result<usize, MatrixError> {
    rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)
}

fn arithmetic_error(row: usize, col: usize, step: usize, source: RationalError) -> MatrixError {
    MatrixError::Arithmetic {
        row,
        col,
        step,
        source,
    }
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = checked_len(rows, cols)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = checked_len(self.rows, rhs.cols)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left
                        .mul(&right)
                        .map_err(|e| arithmetic_error(r, c, k, e))?;
                    acc = acc
                        .add(&term)
                        .map_err(|e| arithmetic_error(r, c, k, e))?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    fn eliminate(
        mut matrix: Self,
        pivot_cols: usize,
    ) -> Result<Reduction, MatrixError> {
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..pivot_cols {
            if p == matrix.rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for row in p..matrix.rows {
                let value = matrix.data[row * matrix.cols + col];
                if value.num == 0 {
                    continue;
                }
                let better = match chosen {
                    None => true,
                    Some(prev) => {
                        let old = matrix.data[prev * matrix.cols + col];
                        (value.den, value.num.unsigned_abs(), row)
                            < (old.den, old.num.unsigned_abs(), prev)
                    }
                };
                if better {
                    chosen = Some(row);
                }
            }
            let Some(q) = chosen else {
                continue;
            };
            if q != p {
                for c in 0..matrix.cols {
                    matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = matrix.data[p * matrix.cols + col];
            for c in 0..matrix.cols {
                let current = matrix.data[p * matrix.cols + c];
                matrix.data[p * matrix.cols + c] = current
                    .div(&pivot)
                    .map_err(|e| arithmetic_error(p, c, p, e))?;
            }

            for row in 0..matrix.rows {
                if row == p {
                    continue;
                }
                let factor = matrix.data[row * matrix.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..matrix.cols {
                    let pivot_entry = matrix.data[p * matrix.cols + c];
                    let product = factor
                        .mul(&pivot_entry)
                        .map_err(|e| arithmetic_error(row, c, p, e))?;
                    let current = matrix.data[row * matrix.cols + c];
                    matrix.data[row * matrix.cols + c] = current
                        .sub(&product)
                        .map_err(|e| arithmetic_error(row, c, p, e))?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        Self::eliminate(self.clone(), self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = checked_len(n, width)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(if r == c {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }
        let augmented = Self {
            rows: n,
            cols: width,
            data,
        };
        let reduction = Self::eliminate(augmented, n)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: reduction.pivots.len(),
            });
        }
        let mut result = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                result.push(reduction.matrix.data[r * width + n + c]);
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: result,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let width = self
            .cols
            .checked_add(rhs.cols)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..self.cols {
                data.push(self.data[r * self.cols + c]);
            }
            for c in 0..rhs.cols {
                data.push(rhs.data[r * rhs.cols + c]);
            }
        }

        let augmented = Self {
            rows: self.rows,
            cols: width,
            data,
        };
        let reduction = Self::eliminate(augmented, self.cols).map_err(SolveError::Matrix)?;
        let rank = reduction.pivots.len();

        for row in 0..self.rows {
            let coeff_zero = (0..self.cols)
                .all(|c| reduction.matrix.data[row * width + c].num == 0);
            if coeff_zero {
                for rhs_col in 0..rhs.cols {
                    if reduction.matrix.data[row * width + self.cols + rhs_col].num != 0 {
                        let witness =
                            reduction.matrix.data[row * width..(row + 1) * width].to_vec();
                        return Ok(Solution::None {
                            row,
                            rhs_col,
                            witness,
                        });
                    }
                }
            }
        }

        let particular_len = self
            .cols
            .checked_mul(rhs.cols)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data =
            vec![Rational { num: 0, den: 1 }; particular_len];
        for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
            for rhs_col in 0..rhs.cols {
                particular_data[pivot_col * rhs.cols + rhs_col] =
                    reduction.matrix.data[pivot_row * width + self.cols + rhs_col];
            }
        }
        let particular = Self {
            rows: self.cols,
            cols: rhs.cols,
            data: particular_data,
        };

        if rank == self.cols {
            return Ok(Solution::Unique {
                values: particular,
                pivots: reduction.pivots,
            });
        }

        let mut free = Vec::new();
        let mut pivot_present = vec![false; self.cols];
        for &col in &reduction.pivots {
            pivot_present[col] = true;
        }
        for col in 0..self.cols {
            if !pivot_present[col] {
                free.push(col);
            }
        }

        let basis_len = self
            .cols
            .checked_mul(free.len())
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![Rational { num: 0, den: 1 }; basis_len];
        for (basis_col, &free_col) in free.iter().enumerate() {
            basis_data[free_col * free.len() + basis_col] = Rational { num: 1, den: 1 };
            for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
                let coefficient = reduction.matrix.data[pivot_row * width + free_col];
                let negated = coefficient.neg().map_err(|source| {
                    SolveError::Matrix(arithmetic_error(pivot_col, basis_col, rank, source))
                })?;
                basis_data[pivot_col * free.len() + basis_col] = negated;
            }
        }
        let basis = Self {
            rows: self.cols,
            cols: free.len(),
            data: basis_data,
        };
        Ok(Solution::Infinite {
            particular,
            free,
            basis,
            pivots: reduction.pivots,
        })
    }
}
