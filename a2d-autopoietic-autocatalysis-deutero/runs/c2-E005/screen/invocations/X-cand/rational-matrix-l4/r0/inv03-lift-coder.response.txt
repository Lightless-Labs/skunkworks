use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        use ParseKind::*;
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_err(0, Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(parse_err(i, if i == bytes.len() { InvalidByte } else { InvalidByte }));
        }
        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() && bytes[i] == b'/' {
            i += 1;
            let start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(parse_err(i, InvalidByte));
            }
            den_start = Some(start);
            den_end = Some(i);
        }
        if i < bytes.len() {
            return Err(parse_err(i, InvalidByte));
        }

        let num_digits = num_end - num_start;
        if num_digits >= 2 && bytes[num_start] == b'0' {
            return Err(parse_err(num_start + 1, NonCanonical));
        }
        if negative && num_digits == 1 && bytes[0] == b'-' && bytes[1] == b'0' {
            return Err(parse_err(0, NonCanonical));
        }
        if let (Some(start), Some(end)) = (den_start, den_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_err(start + 1, NonCanonical));
            }
        }

        let num_mag = decimal_magnitude(&bytes[num_start..num_end])
            .ok_or_else(|| parse_err(0, OutOfRange))?;
        let max_mag = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if num_mag > max_mag {
            return Err(parse_err(0, OutOfRange));
        }
        let num = if negative {
            if num_mag == 1u128 << 63 {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };

        let den = if let (Some(start), Some(end)) = (den_start, den_end) {
            let mag = decimal_magnitude(&bytes[start..end])
                .ok_or_else(|| parse_err(start, OutOfRange))?;
            if mag > i64::MAX as u128 {
                return Err(parse_err(start, OutOfRange));
            }
            mag as i64
        } else {
            return Ok(Self { num, den: 1 });
        };

        let slash = num_end;
        let den_offset = den_start.unwrap();
        if den == 0 {
            return Err(parse_err(den_offset, ZeroDenominator));
        }
        if den == 1 {
            return Err(parse_err(slash, NonCanonical));
        }
        if num == 0 {
            return Err(parse_err(slash, NonCanonical));
        }
        if gcd_u128(num.unsigned_abs() as u128, den as u128) > 1 {
            return Err(parse_err(slash, NonCanonical));
        }
        Ok(Self { num, den })
    }

    fn from_i128(mut num: i128, mut den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if den < 0 {
            num = -num;
            den = -den;
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
        num /= g;
        den /= g;
        if num < i64::MIN as i128
            || num > i64::MAX as i128
            || den < 1
            || den > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: num as i64,
            den: den as i64,
        })
    }
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn decimal_magnitude(bytes: &[u8]) -> Option<u128> {
    let mut value = 0u128;
    for &b in bytes {
        value = value.checked_mul(10)?.checked_add((b - b'0') as u128)?;
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = vec![zero(); self.data.len()];
        for r in 0..self.rows {
            for c in 0..self.cols {
                data[c * self.rows + r] = self.data[r * self.cols + c];
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero();
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows:self.rows, cols:rhs.cols, data })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self.clone(), self.cols, self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows:self.rows, cols:self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = vec![zero(); len];
        for r in 0..n {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
            }
            data[r * width + n + r] = one();
        }
        let red = eliminate(Self { rows:n, cols:width, data }, n, n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank:red.pivots.len() });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            out.extend_from_slice(&red.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Self { rows:n, cols:n, data:out })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        let m = self.rows;
        let n = self.cols;
        let k = rhs.cols;
        if m != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients:m, rhs:rhs.rows });
        }
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = m.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = vec![zero(); len];
        for r in 0..m {
            data[r * width..r * width + n]
                .copy_from_slice(&self.data[r * n..r * n + n]);
            data[r * width + n..r * width + width]
                .copy_from_slice(&rhs.data[r * k..r * k + k]);
        }
        let red = eliminate(Self { rows:m, cols:width, data }, n, n)
            .map_err(SolveError::Matrix)?;

        for r in 0..m {
            let coeff_zero = red.matrix.data[r * width..r * width + n].iter().all(|v| v.num == 0);
            if coeff_zero {
                if let Some(j) = (0..k).find(|&j| red.matrix.data[r * width + n + j].num != 0) {
                    return Ok(Solution::None {
                        row:r, rhs_col:j,
                        witness:red.matrix.data[r * width..r * width + width].to_vec(),
                    });
                }
            }
        }

        let mut free = Vec::new();
        let mut pivot_mask = vec![false; n];
        for &p in &red.pivots { pivot_mask[p] = true; }
        for c in 0..n {
            if !pivot_mask[c] { free.push(c); }
        }

        let pk = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular = vec![zero(); pk];
        for (i, &p) in red.pivots.iter().enumerate() {
            for j in 0..k {
                particular[p * k + j] = red.matrix.data[i * width + n + j];
            }
        }
        let particular = Self { rows:n, cols:k, data:particular };
        if red.pivots.len() == n {
            return Ok(Solution::Unique { values:particular, pivots:red.pivots });
        }

        let f = free.len();
        let bf = n.checked_mul(f).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![zero(); bf];
        for (j, &free_col) in free.iter().enumerate() {
            basis[free_col * f + j] = one();
            for (i, &p) in red.pivots.iter().enumerate() {
                let coeff = red.matrix.data[i * width + free_col];
                basis[p * f + j] = coeff.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row:p, col:j, step:red.pivots.len(), source,
                    })
                })?;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis:Self { rows:n, cols:f, data:basis },
            pivots:red.pivots,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize { self.pivots.len() }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}

fn zero() -> Rational { Rational { num:0, den:1 } }
fn one() -> Rational { Rational { num:1, den:1 } }

fn eliminate(
    mut matrix: Matrix,
    pivot_limit: usize,
    pivot_step_base: usize,
) -> Result<Reduction, MatrixError> {
    let rows = matrix.rows;
    let cols = matrix.cols;
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;

    for candidate in 0..pivot_limit {
        if p == rows { break; }
        let mut best: Option<usize> = None;
        for r in p..rows {
            let v = matrix.data[r * cols + candidate];
            if v.num == 0 { continue; }
            let key = (v.den, v.num.unsigned_abs(), r);
            if best.map_or(true, |br| {
                let b = matrix.data[br * cols + candidate];
                key < (b.den, b.num.unsigned_abs(), br)
            }) {
                best = Some(r);
            }
        }
        let Some(q) = best else { continue; };
        if q != p {
            for c in 0..cols {
                matrix.data.swap(p * cols + c, q * cols + c);
            }
            swaps.push((p,q));
        }
        let pivot = matrix.data[p * cols + candidate];
        for c in 0..cols {
            let current = matrix.data[p * cols + c];
            matrix.data[p * cols + c] = current.div(&pivot).map_err(|source| {
                MatrixError::Arithmetic { row:p, col:c, step:pivot_step_base + p, source }
            })?;
        }
        for r in 0..rows {
            if r == p { continue; }
            let factor = matrix.data[r * cols + candidate];
            if factor.num == 0 { continue; }
            for c in 0..cols {
                let product = factor.mul(&matrix.data[p * cols + c]).map_err(|source| {
                    MatrixError::Arithmetic { row:r, col:c, step:pivot_step_base + p, source }
                })?;
                matrix.data[r * cols + c].sub(&product).map_err(|source| {
                    MatrixError::Arithmetic { row:r, col:c, step:pivot_step_base + p, source }
                }).map(|v| matrix.data[r * cols + c] = v)?;
            }
        }
        pivots.push(candidate);
        p += 1;
    }
    Ok(Reduction { matrix, pivots, swaps })
}
