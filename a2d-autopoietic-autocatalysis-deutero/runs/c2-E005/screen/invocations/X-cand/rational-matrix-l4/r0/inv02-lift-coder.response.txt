use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let da = self.den as i128;
        let db = rhs.den as i128;
        let g = gcd_u128(da as u128, db as u128) as i128;
        let l = da / g;
        let r = db / g;
        Self::from_i128(a * r + b * l, l * db)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let da = self.den as i128;
        let db = rhs.den as i128;
        let g = gcd_u128(da as u128, db as u128) as i128;
        let l = da / g;
        let r = db / g;
        Self::from_i128(a * r - b * l, l * db)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let da = self.den as i128;
        let db = rhs.den as i128;
        let g1 = gcd_u128(a.unsigned_abs(), db as u128) as i128;
        let g2 = gcd_u128(b.unsigned_abs(), da as u128) as i128;
        Self::from_i128((a / g1) * (b / g2), (da / g2) * (db / g1))
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let a = self.num as i128;
        let b = rhs.num as i128;
        let da = self.den as i128;
        let db = rhs.den as i128;
        let g1 = gcd_u128(a.unsigned_abs(), b.unsigned_abs()) as i128;
        let g2 = gcd_u128(da as u128, db as u128) as i128;
        Self::from_i128((a / g1) * (db / g2), (da / g2) * (b / g1))
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }

        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(err(i, if i == bytes.len() { ParseKind::InvalidByte } else { ParseKind::InvalidByte }));
        }
        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(err(i, ParseKind::InvalidByte));
            }
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(err(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(err(i, ParseKind::InvalidByte));
            }
        }

        let num_digits = num_end - num_start;
        if num_digits >= 2 && bytes[num_start] == b'0' {
            return Err(err(num_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == 1 && bytes[num_start] == b'0' {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if let Some(ds) = den_start {
            let de = den_end.unwrap();
            if de - ds >= 2 && bytes[ds] == b'0' {
                return Err(err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let parse_mag = |start: usize, end: usize, limit: u128| -> Option<u128> {
            let mut value = 0u128;
            for &byte in &bytes[start..end] {
                value = value.checked_mul(10)?.checked_add((byte - b'0') as u128)?;
                if value > limit {
                    return None;
                }
            }
            Some(value)
        };

        let nlimit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let nmag = parse_mag(num_start, num_end, nlimit)
            .ok_or_else(|| err(0, ParseKind::OutOfRange))?;
        let n = if negative {
            if nmag == 1u128 << 63 {
                i64::MIN
            } else {
                -(nmag as i64)
            }
        } else {
            nmag as i64
        };

        let mut d = 1i64;
        if let (Some(ds), Some(de)) = (den_start, den_end) {
            let mag = parse_mag(ds, de, i64::MAX as u128)
                .ok_or_else(|| err(ds, ParseKind::OutOfRange))?;
            if mag == 0 {
                return Err(err(ds, ParseKind::ZeroDenominator));
            }
            d = mag as i64;
            let slash = ds - 1;
            if d == 1 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
            if n == 0 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
            if gcd_u128(n.unsigned_abs() as u128, d as u128) > 1 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
        }
        Ok(Self { num: n, den: d })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

fn checked_len(rows: usize, cols: usize) -> Result<usize, MatrixError> {
    rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)
}

fn arithmetic_error(row: usize, col: usize, step: usize, source: RationalError) -> MatrixError {
    MatrixError::Arithmetic {
        row,
        col,
        step,
        source,
    }
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = checked_len(rows, cols)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = checked_len(self.rows, rhs.cols)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left
                        .mul(&right)
                        .map_err(|e| arithmetic_error(r, c, k, e))?;
                    acc = acc
                        .add(&term)
                        .map_err(|e| arithmetic_error(r, c, k, e))?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    fn eliminate(
        mut matrix: Self,
        pivot_cols: usize,
    ) -> Result<Reduction, MatrixError> {
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..pivot_cols {
            if p == matrix.rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for row in p..matrix.rows {
                let value = matrix.data[row * matrix.cols + col];
                if value.num == 0 {
                    continue;
                }
                let better = match chosen {
                    None => true,
                    Some(prev) => {
                        let old = matrix.data[prev * matrix.cols + col];
                        (value.den, value.num.unsigned_abs(), row)
                            < (old.den, old.num.unsigned_abs(), prev)
                    }
                };
                if better {
                    chosen = Some(row);
                }
            }
            let Some(q) = chosen else {
                continue;
            };
            if q != p {
                for c in 0..matrix.cols {
                    matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = matrix.data[p * matrix.cols + col];
            for c in 0..matrix.cols {
                let current = matrix.data[p * matrix.cols + c];
                matrix.data[p * matrix.cols + c] = current
                    .div(&pivot)
                    .map_err(|e| arithmetic_error(p, c, p, e))?;
            }

            for row in 0..matrix.rows {
                if row == p {
                    continue;
                }
                let factor = matrix.data[row * matrix.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..matrix.cols {
                    let pivot_entry = matrix.data[p * matrix.cols + c];
                    let product = factor
                        .mul(&pivot_entry)
                        .map_err(|e| arithmetic_error(row, c, p, e))?;
                    let current = matrix.data[row * matrix.cols + c];
                    matrix.data[row * matrix.cols + c] = current
                        .sub(&product)
                        .map_err(|e| arithmetic_error(row, c, p, e))?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        Self::eliminate(self.clone(), self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = checked_len(n, width)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(if r == c {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }
        let augmented = Self {
            rows: n,
            cols: width,
            data,
        };
        let reduction = Self::eliminate(augmented, n)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: reduction.pivots.len(),
            });
        }
        let mut result = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                result.push(reduction.matrix.data[r * width + n + c]);
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: result,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let width = self
            .cols
            .checked_add(rhs.cols)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..self.cols {
                data.push(self.data[r * self.cols + c]);
            }
            for c in 0..rhs.cols {
                data.push(rhs.data[r * rhs.cols + c]);
            }
        }

        let augmented = Self {
            rows: self.rows,
            cols: width,
            data,
        };
        let reduction = Self::eliminate(augmented, self.cols).map_err(SolveError::Matrix)?;
        let rank = reduction.pivots.len();

        for row in 0..self.rows {
            let coeff_zero = (0..self.cols)
                .all(|c| reduction.matrix.data[row * width + c].num == 0);
            if coeff_zero {
                for rhs_col in 0..rhs.cols {
                    if reduction.matrix.data[row * width + self.cols + rhs_col].num != 0 {
                        let witness =
                            reduction.matrix.data[row * width..(row + 1) * width].to_vec();
                        return Ok(Solution::None {
                            row,
                            rhs_col,
                            witness,
                        });
                    }
                }
            }
        }

        let particular_len = self
            .cols
            .checked_mul(rhs.cols)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data =
            vec![Rational { num: 0, den: 1 }; particular_len];
        for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
            for rhs_col in 0..rhs.cols {
                particular_data[pivot_col * rhs.cols + rhs_col] =
                    reduction.matrix.data[pivot_row * width + self.cols + rhs_col];
            }
        }
        let particular = Self {
            rows: self.cols,
            cols: rhs.cols,
            data: particular_data,
        };

        if rank == self.cols {
            return Ok(Solution::Unique {
                values: particular,
                pivots: reduction.pivots,
            });
        }

        let mut free = Vec::new();
        let mut pivot_present = vec![false; self.cols];
        for &col in &reduction.pivots {
            pivot_present[col] = true;
        }
        for col in 0..self.cols {
            if !pivot_present[col] {
                free.push(col);
            }
        }

        let basis_len = self
            .cols
            .checked_mul(free.len())
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![Rational { num: 0, den: 1 }; basis_len];
        for (basis_col, &free_col) in free.iter().enumerate() {
            basis_data[free_col * free.len() + basis_col] = Rational { num: 1, den: 1 };
            for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
                let coefficient = reduction.matrix.data[pivot_row * width + free_col];
                let negated = coefficient.neg().map_err(|source| {
                    SolveError::Matrix(arithmetic_error(pivot_col, basis_col, rank, source))
                })?;
                basis_data[pivot_col * free.len() + basis_col] = negated;
            }
        }
        let basis = Self {
            rows: self.cols,
            cols: free.len(),
            data: basis_data,
        };
        Ok(Solution::Infinite {
            particular,
            free,
            basis,
            pivots: reduction.pivots,
        })
    }
}
