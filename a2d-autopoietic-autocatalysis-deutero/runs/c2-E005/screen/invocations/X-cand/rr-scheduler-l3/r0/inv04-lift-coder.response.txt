```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut arrival_order: Vec<usize> = (0..jobs.len()).collect();
    arrival_order.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut queues: Vec<std::collections::VecDeque<usize>> = (0..quanta.len())
        .map(|_| std::collections::VecDeque::new())
        .collect();
    let mut arrival_cursor = 0usize;
    let mut time = 0u64;
    let mut last_dispatched_id: Option<u32> = None;
    let mut runs: Vec<Run> = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut completion: Vec<(u32, u64)> =
        jobs.iter().map(|job| (job.id, 0)).collect();
    let mut completed = 0usize;

    fn admit(
        jobs: &[Job],
        order: &[usize],
        cursor: &mut usize,
        through: u64,
        queues: &mut [std::collections::VecDeque<usize>],
        completion: &mut [(u32, u64)],
        completed: &mut usize,
    ) {
        while *cursor < order.len()
            && u64::from(jobs[order[*cursor]].arrival) <= through
        {
            let index = order[*cursor];
            *cursor += 1;

            if jobs[index].ticks == 0 {
                completion[index] = (jobs[index].id, u64::from(jobs[index].arrival));
                *completed += 1;
            } else {
                queues[jobs[index].level].push_back(index);
            }
        }
    }

    while completed < jobs.len() {
        admit(
            jobs,
            &arrival_order,
            &mut arrival_cursor,
            time,
            &mut queues,
            &mut completion,
            &mut completed,
        );

        if let Some(level) = queues.iter().position(|queue| !queue.is_empty()) {
            let index = queues[level].pop_front().unwrap();
            let job = &jobs[index];

            if last_dispatched_id.is_some_and(|last_id| last_id != job.id) {
                time += u64::from(switch_cost);
            }

            admit(
                jobs,
                &arrival_order,
                &mut arrival_cursor,
                time,
                &mut queues,
                &mut completion,
                &mut completed,
            );

            let start = time;
            let consumed = remaining[index].min(quanta[level]);
            time += u64::from(consumed);
            remaining[index] -= consumed;
            let finished = remaining[index] == 0;

            runs.push(Run {
                id: job.id,
                start,
                end: time,
                finished,
                level,
            });

            admit(
                jobs,
                &arrival_order,
                &mut arrival_cursor,
                time,
                &mut queues,
                &mut completion,
                &mut completed,
            );

            if finished {
                completion[index] = (job.id, time);
                completed += 1;
            } else {
                let next_level = (level + 1).min(quanta.len() - 1);
                queues[next_level].push_back(index);
            }

            last_dispatched_id = Some(job.id);
        } else if arrival_cursor < arrival_order.len() {
            let next_time = u64::from(jobs[arrival_order[arrival_cursor]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched_id = None;
            }
        } else {
            break;
        }
    }

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
```
