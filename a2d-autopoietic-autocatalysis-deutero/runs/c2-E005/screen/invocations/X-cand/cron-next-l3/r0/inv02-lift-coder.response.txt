#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_unrestricted: bool,
    day_of_week_unrestricted: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn ascii_name(field: Field, s: &str) -> Option<u32> {
    if !s.is_ascii() {
        return None;
    }
    let upper = s.to_ascii_uppercase();
    match field {
        Field::Month => match upper.as_str() {
            "JAN" => Some(1),
            "FEB" => Some(2),
            "MAR" => Some(3),
            "APR" => Some(4),
            "MAY" => Some(5),
            "JUN" => Some(6),
            "JUL" => Some(7),
            "AUG" => Some(8),
            "SEP" => Some(9),
            "OCT" => Some(10),
            "NOV" => Some(11),
            "DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_str() {
            "SUN" => Some(0),
            "MON" => Some(1),
            "TUE" => Some(2),
            "WED" => Some(3),
            "THU" => Some(4),
            "FRI" => Some(5),
            "SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

#[derive(Clone, Copy)]
enum Atom {
    Number(u32),
    Name(u32),
}

impl Atom {
    fn value(self) -> u32 {
        match self {
            Atom::Number(v) | Atom::Name(v) => v,
        }
    }

    fn is_number(self) -> bool {
        matches!(self, Atom::Number(_))
    }
}

fn parse_atom_syntax(field: Field, s: &str) -> Option<Atom> {
    if s.is_empty() {
        return None;
    }
    if s.bytes().all(|b| b.is_ascii_digit()) {
        return Some(Atom::Number(0));
    }
    ascii_name(field, s).map(Atom::Name)
}

fn number_value(field: Field, s: &str, offset: usize) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    let (min, max) = bounds(field);
    if value < min || value > max {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(value)
}

fn parse_atom_value(field: Field, s: &str, offset: usize) -> Result<u32, ParseError> {
    if s.bytes().all(|b| b.is_ascii_digit()) {
        number_value(field, s, offset)
    } else {
        ascii_name(field, s).ok_or(ParseError::InvalidSyntax { field, offset })
    }
}

fn parse_step(field: Field, s: &str, offset: usize) -> Result<u32, ParseError> {
    if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if s.len() > 1 && s.as_bytes()[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in s.as_bytes() {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    let (_, max) = bounds(field);
    let width = max - bounds(field).0 + 1;
    if value == 0 || value > width {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(value)
}

fn parse_item(
    field: Field,
    item: &str,
    offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem { field, offset });
    }

    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    let (base, step_text, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), offset + slash + 1)
    } else {
        (item, None, 0)
    };

    let mut range_parts = base.split('-');
    let first = range_parts.next().unwrap_or("");
    let second = range_parts.next();
    if range_parts.next().is_some() || first.is_empty() || second == Some("") {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    let wildcard = base == "*";
    let (start_atom, end_atom) = if wildcard {
        if second.is_some() {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        (None, None)
    } else {
        if first == "*" || second == Some("*") {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        let a = parse_atom_syntax(field, first)
            .ok_or(ParseError::InvalidSyntax { field, offset })?;
        let b = if let Some(end) = second {
            Some(
                parse_atom_syntax(field, end)
                    .ok_or(ParseError::InvalidSyntax { field, offset })?,
            )
        } else {
            None
        };
        (Some(a), b)
    };

    let step = if let Some(s) = step_text {
        if wildcard || start_atom.is_some() {
            Some(parse_step(field, s, step_offset)?)
        } else {
            unreachable!()
        }
    } else {
        None
    };

    let (start, end) = if wildcard {
        let (min, max) = bounds(field);
        (min, max)
    } else {
        let a = start_atom.unwrap();
        let a_value = if a.is_number() {
            number_value(field, first, offset)?
        } else {
            a.value()
        };
        let (min, max) = bounds(field);
        if a_value < min || a_value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }

        if let (Some(b), Some(end_text)) = (end_atom, second) {
            let end_offset = offset + first.len() + 1;
            let b_value = if b.is_number() {
                number_value(field, end_text, end_offset)?
            } else {
                b.value()
            };
            if b_value < min || b_value > max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: end_offset,
                });
            }
            if a_value > b_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_offset,
                });
            }
            (a_value, b_value)
        } else if step.is_some() {
            (a_value, max)
        } else {
            (a_value, a_value)
        }
    };

    let increment = step.unwrap_or(1);
    let mut value = start;
    loop {
        selected[(value - bounds(field).0) as usize] = true;
        if value > end.saturating_sub(increment) {
            break;
        }
        value += increment;
    }
    Ok(())
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    for (i, b) in text.bytes().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (min, max) = bounds(field);
    let mut selected = vec![false; (max - min + 1) as usize];
    let mut offset = 0;
    for item in text.split(',') {
        parse_item(field, item, offset, &mut selected)?;
        offset += item.len() + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut parsed = Vec::with_capacity(5);
    for (field, text) in FIELDS.into_iter().zip(pieces.iter().copied()) {
        parsed.push(parse_field(field, text)?);
    }
    let selected: [Vec<bool>; 5] = parsed.try_into().ok().unwrap();

    Ok(Cron {
        selected,
        day_of_month_unrestricted: pieces[2] == "*",
        day_of_week_unrestricted: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, _) = bounds(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(min as u8 + i as u8))
            .collect()
    }

    fn selected_value(&self, field: Field, value: u8) -> bool {
        let (min, max) = bounds(field);
        value as u32 >= min
            && (value as u32) <= max
            && self.selected[field_index(field)][(value as u32 - min) as usize]
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.selected_value(Field::Minute, at.minute)
            || !self.selected_value(Field::Hour, at.hour)
            || !self.selected_value(Field::Month, at.month)
        {
            return false;
        }

        let dom = self.selected_value(Field::DayOfMonth, at.day);
        let dow = self.selected_value(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        match (self.day_of_month_unrestricted, self.day_of_week_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) if at.day == days_in_month(at.year, at.month) => dom && dow,
            (false, false) => dom || dow,
        }
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if current.minute < 59 {
                current.minute += 1;
            } else {
                current.minute = 0;
                if current.hour < 23 {
                    current.hour += 1;
                } else {
                    current.hour = 0;
                    if current.day < days_in_month(current.year, current.month) {
                        current.day += 1;
                    } else {
                        current.day = 1;
                        if current.month < 12 {
                            current.month += 1;
                        } else if current.year < 2399 {
                            current.month = 1;
                            current.year += 1;
                        } else {
                            return Ok(None);
                        }
                    }
                }
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}
