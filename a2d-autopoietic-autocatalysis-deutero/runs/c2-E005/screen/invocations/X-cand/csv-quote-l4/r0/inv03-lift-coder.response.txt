use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|v| v == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|v| v == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = Vec::new();
    let mut i = start;
    let mut record = 0usize;
    let mut quoted = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !quoted && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if quoted {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote);
                    i += 2;
                } else {
                    quoted = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                field.push(b);
                i += 1;
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                after_quote = false;
                i += 1;
            } else if b == b'\n' {
                row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                records.push(std::mem::take(&mut row));
                record += 1;
                after_quote = false;
                i += 1;
            } else if b == b'\r' {
                row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
                records.push(std::mem::take(&mut row));
                record += 1;
                after_quote = false;
                i += 2;
            } else {
                return Err(CsvError::TrailingByte { record, byte: i });
            }
            continue;
        }

        if b == delimiter {
            row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
            i += 1;
        } else if b == b'\n' {
            row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
            records.push(std::mem::take(&mut row));
            record += 1;
            i += 1;
        } else if b == b'\r' {
            row.push(String::from_utf8(std::mem::take(&mut field)).unwrap());
            records.push(std::mem::take(&mut row));
            record += 1;
            i += 2;
        } else if b == quote {
            if field.is_empty() {
                quoted = true;
                quote_open = i;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record, byte: i });
            }
        } else {
            field.push(b);
            i += 1;
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_open,
        });
    }

    if i > start && (bytes.last() != Some(&b'\n')) {
        // A final CRLF terminator also ends with LF. Any other consumed input
        // leaves an unfinished record, including a trailing delimiter.
        row.push(String::from_utf8(field).unwrap());
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    let mut firsts: HashMap<&str, usize> = HashMap::new();
    for (second, name) in header.iter().enumerate() {
        if let Some(first) = firsts.get(name.as_str()) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first: *first,
                second,
            });
        }
        firsts.insert(name, second);
    }
    None
}

fn table_from_records(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let mut iter = records.into_iter();
    let header = if config.headers { iter.next() } else { None };
    let rows: Vec<Vec<String>> = iter.collect();

    if let Some(h) = header.as_ref() {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(err) = duplicate_header(h) {
                return Err(err);
            }
        }
        for (idx, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    finished: Option<Result<Table, CsvError>>,
    config_error: Option<CsvError>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        if let Err(component) = validate_dialect(config.delimiter, config.quote) {
            return Err(CsvError::InvalidConfig { component });
        }
        Ok(Self {
            config,
            input: String::new(),
            finished: None,
            config_error: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = self.finished.as_ref() {
            return result.clone();
        }
        let result = if let Some(err) = self.config_error.clone() {
            Err(err)
        } else {
            parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
                .and_then(|records| table_from_records(records, self.config))
        };
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]

    pub spec: usize,
    pub column: Option<usize>,

#[derive(Debug, Clone, PartialEq, Eq)]

    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let digits = if b[0] == b'-' { &b[1..] } else { b };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return b[0] != b'-';
    }
    (b'1'..=b'9').contains(&digits[0]) && digits[1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let core = if b.first() == Some(&b'-') {
        &b[1..]
    } else {
        b
    };
    if core.len() < 4 || core[core.len() - 3] != b'.' {
        return false;
    }
    let integral = &core[..core.len() - 3];
    let frac = &core[core.len() - 2..];
    if !frac.iter().all(u8::is_ascii_digit) {
        return false;
    }
    if integral == b"0" {
        true
    } else {
        !integral.is_empty()
            && (b'1'..=b'9').contains(&integral[0])
            && integral[1..].iter().all(u8::is_ascii_digit)
    }
}

fn evaluate_cell(
    cell: &str,
    spec: &ColumnSpec,
) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind == ColumnType::Text {
            return Ok(Value::Text(String::new()));
        }
        return Err(CellErrorKind::Empty);
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
        ColumnType::Integer => {
            if !integer_syntax(cell) {
                return Err(CellErrorKind::InvalidInteger);
            }
            cell.parse::<i64>()
                .map(Value::Integer)
                .map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(cell) {
                return Err(CellErrorKind::InvalidDecimal);
            }
            let negative = cell.starts_with('-');
            let unsigned = if negative { &cell[1..] } else { cell };
            let dot = unsigned.len() - 3;
            let whole: i128 = unsigned[..dot].parse().unwrap();
            let frac: i128 = unsigned[dot + 1..].parse().unwrap();
            let magnitude = whole * 100 + frac;
            let signed = if negative { -magnitude } else { magnitude };
            i64::try_from(signed)
                .map(Value::Decimal)
                .map_err(|_| CellErrorKind::DecimalOverflow)
        }
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        let mut had_error = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            let column = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let evaluated = if column.is_none() {
                Err(CellErrorKind::UnknownColumn)
            } else {
                let index = column.unwrap();
                match row.get(index) {
                    None => Err(CellErrorKind::MissingCell),
                    Some(cell) => evaluate_cell(cell, spec),
                }
            };

            match evaluated {
                Ok(value) => output[spec_index] = Some(value),
                Err(kind) => {
                    had_error = true;
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column,
                        kind,
                    });
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                }
            }
        }

        result.rows.push(output);
        if policy == ErrorPolicy::Stop && had_error {
            break 'rows;
        }
    }
    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]

    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,

}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn record_at(table: &Table, index: usize) -> Option<&[String]> {
    if let Some(header) = table.header.as_ref() {
        if index == 0 {
            return Some(header);
        }
        table.rows.get(index - 1).map(Vec::as_slice)
    } else {
        table.rows.get(index).map(Vec::as_slice)
    }
}

fn compare_vectors(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let order = x.as_bytes().cmp(y.as_bytes());
        if order != std::cmp::Ordering::Equal {
            return order;
        }
    }
    a.len().cmp(&b.len())
}

fn quote_field(
    field: &str,
    delimiter: u8,
    quote: u8,
    first_field_first_record: bool,
    sole_empty: bool,
) -> String {
    let bytes = field.as_bytes();
    let needs_quote = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
        || (first_field_first_record && field.starts_with('\u{feff}'))
        || (sole_empty && field.is_empty());

    if !needs_quote {
        return field.to_string();
    }

    let mut out = String::with_capacity(field.len() + 2);
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
        }
        out.push(ch);
    }
    out.push(quote as char);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    if let Err(component) = validate_dialect(config.delimiter, config.quote) {
        return Err(WriteError::InvalidConfig { component });
    }

    let total_records = table.rows.len() + usize::from(table.header.is_some());
    for record_index in 0..total_records {
        if let Some(record) = record_at(table, record_index) {
            for (column, cell) in record.iter().enumerate() {
                if let Some(byte) = cell.as_bytes().iter().position(|b| *b == 0) {
                    return Err(WriteError::Nul {
                        record: record_index,
                        column,
                        byte,
                    });
                }
            }
        }
    }

    if let Some(header) = table.header.as_ref() {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for record_index in 0..total_records {
        if record_at(table, record_index).is_some_and(|r| r.is_empty()) {
            return Err(WriteError::ZeroColumns {
                record: record_index,
            });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let Some(header) = table.header.as_ref() {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(error) = duplicate_header(header) {
                    if let CsvError::DuplicateHeader {
                        name,
                        first,
                        second,
                    } = error
                    {
                        return Err(WriteError::DuplicateHeader {
                            name,
                            first,
                            second,
                        });
                    }
                }
            }
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();
    if config.mode == WriteMode::Preserve {
        if let Some(header) = &table.header {
            output_records.push(header.clone());
        }
        output_records.extend(table.rows.clone());
    } else if let Some(header) = &table.header {
        let mut retained: HashMap<&str, usize> = HashMap::new();
        for (idx, name) in header.iter().enumerate() {
            match table.duplicates {
                DuplicatePolicy::KeepLast => {
                    retained.insert(name.as_str(), idx);
                }
                DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                    retained.entry(name.as_str()).or_insert(idx);
                }
            }
        }
        let mut columns: Vec<(String, usize)> = retained
            .into_iter()
            .map(|(name, index)| (name.to_string(), index))
            .collect();
        columns.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));

        let new_header: Vec<String> = columns.iter().map(|(name, _)| name.clone()).collect();
        output_records.push(new_header);
        for row in &table.rows {
            output_records.push(
                columns
                    .iter()
                    .map(|(_, index)| row[*index].clone())
                    .collect(),
            );
        }
        output_records[1..].sort_by(|a, b| compare_vectors(a, b));
    } else {
        output_records = table.rows.clone();
        output_records.sort_by(|a, b| compare_vectors(a, b));
    }

    if output_records.is_empty() {
        return Ok(String::new());
    }

    let canonical = config.mode == WriteMode::Canonical;
    let terminator = if canonical || config.line_ending == LineEnding::Lf {
        "\n"
    } else {
        "\r\n"
    };
    let final_terminator = canonical || config.final_terminator;
    let mut out = String::new();

    for (record_index, record) in output_records.iter().enumerate() {
        if record_index > 0 {
            out.push_str(terminator);
        }

        let sole_empty = record.len() == 1 && record[0].is_empty();
        for (column_index, field) in record.iter().enumerate() {
            if column_index > 0 {
                out.push(config.delimiter as char);
            }
            out.push_str(&quote_field(
                field,
                config.delimiter,
                config.quote,
                record_index == 0 && column_index == 0,
                sole_empty,
            ));
        }
    }

    if final_terminator {
        out.push_str(terminator);
    }
    Ok(out)
}
