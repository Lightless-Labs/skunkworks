#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn validate_config(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    if pos == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: after quote
    let mut record_index = 0usize;
    let mut field_open_quote = 0usize;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if state != 2 && b == b'\r' {
            if bytes.get(pos + 1) == Some(&b'\n') {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                pos += 2;
                record_index += 1;
                state = 0;
                continue;
            }
            return Err(CsvError::BareCr {
                record: record_index,
                byte: pos,
            });
        }

        if state != 2 && b == b'\n' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            pos += 1;
            record_index += 1;
            state = 0;
            continue;
        }

        match state {
            0 => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    pos += 1;
                } else if b == quote {
                    field_open_quote = pos;
                    state = 2;
                    pos += 1;
                } else {
                    field.push(b as char);
                    state = 1;
                    pos += 1;
                }
            }
            1 => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = 0;
                    pos += 1;
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: pos,
                    });
                } else {
                    let ch = input[pos..].chars().next().unwrap();
                    field.push(ch);
                    pos += ch.len_utf8();
                }
            }
            2 => {
                if b == quote {
                    if bytes.get(pos + 1) == Some(&quote) {
                        field.push(quote as char);
                        pos += 2;
                    } else {
                        state = 3;
                        pos += 1;
                    }
                } else {
                    let ch = input[pos..].chars().next().unwrap();
                    field.push(ch);
                    pos += ch.len_utf8();
                }
            }
            _ => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = 0;
                    pos += 1;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: pos,
                    });
                }
            }
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: field_open_quote,
        });
    }

    if state != 0 || !record.is_empty() || !field.is_empty() {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn duplicate_header_error(header: &[String]) -> Option<CsvError> {
    let mut seen = std::collections::HashMap::<&str, usize>::new();
    for (second, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first,
                second,
            });
        }
        seen.insert(name, second);
    }
    None
}

fn table_from_records(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut records = records.into_iter();
            (records.next(), records.collect::<Vec<_>>())
        }
    } else {
        (None, records)
    };

    if let Some(ref header) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header_error(header) {
                return Err(error);
            }
        }
        for (index, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    finished: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config.delimiter, config.quote).map_err(|component| {
            CsvError::InvalidConfig { component }
        })?;
        Ok(Self {
            config,
            input: String::new(),
            finished: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.finished {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let start = usize::from(bytes[0] == b'-');
    if start == bytes.len() {
        return false;
    }
    if bytes[start] == b'0' {
        return start == 0 && bytes.len() == 1;
    }
    if !(b'1'..=b'9').contains(&bytes[start]) {
        return false;
    }
    bytes[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.len() < 4 {
        return false;
    }
    let start = usize::from(bytes[0] == b'-');
    if start >= bytes.len() {
        return false;
    }
    let dot = match bytes[start..].iter().position(|&b| b == b'.') {
        Some(i) => start + i,
        None => return false,
    };
    if bytes.len() != dot + 3 {
        return false;
    }
    if !bytes[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    let integral = &bytes[start..dot];
    if integral.is_empty() {
        return false;
    }
    if integral == b"0" {
        return start == 0;
    }
    (b'1'..=b'9').contains(&integral[0])
        && integral[1..].iter().all(u8::is_ascii_digit)
}

fn parse_decimal_hundredths(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) || s == "-0.00" {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let body = if negative { &s[1..] } else { s };
    let (whole, frac) = body.split_once('.').unwrap();
    let whole_value = whole.parse::<i128>().unwrap();
    let fraction = ((frac.as_bytes()[0] - b'0') * 10 + frac.as_bytes()[1] - b'0') as i128;
    let magnitude = whole_value * 100 + fraction;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate_cell(
    cell: &str,
    spec: &ColumnSpec,
) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(cell) {
                return Err(CellErrorKind::InvalidInteger);
            }
            cell.parse::<i64>()
                .map(Value::Integer)
                .map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => parse_decimal_hundredths(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        let mut failed = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            if failed {
                break;
            }
            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (column, value, error) = match resolved {
                None => (
                    None,
                    None,
                    Some(CellErrorKind::UnknownColumn),
                ),
                Some(index) => match row.get(index) {
                    None => (Some(index), None, Some(CellErrorKind::MissingCell)),
                    Some(cell) => match evaluate_cell(cell, spec) {
                        Ok(value) => (Some(index), Some(value), None),
                        Err(kind) => (Some(index), None, Some(kind)),
                    },
                },
            };
            if let Some(kind) = error {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column,
                    kind,
                });
                if policy != ErrorPolicy::Collect {
                    failed = true;
                }
                if policy == ErrorPolicy::Stop {
                    result.rows.push(output);
                    return result;
                }
                if policy == ErrorPolicy::FirstPerRow {
                    break;
                }
            } else {
                output[spec_index] = value;
            }
        }
        result.rows.push(output);
    }

    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns { record: usize },
    DuplicateHeader {
        name: String,
        first: usize,
        second: usize,
    },
}

fn write_records(table: &Table) -> Vec<Vec<String>> {
    let mut records = Vec::with_capacity(table.rows.len() + usize::from(table.header.is_some()));
    if let Some(header) = &table.header {
        records.push(header.clone());
    }
    records.extend(table.rows.iter().cloned());
    records
}

fn bytewise_vec_cmp(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        let order = left.as_bytes().cmp(right.as_bytes());
        if order != std::cmp::Ordering::Equal {
            return order;
        }
    }
    a.len().cmp(&b.len())
}

fn quote_field(value: &str, first_field_first_record: bool, sole_field: bool, config: WriteConfig) -> String {
    let bytes = value.as_bytes();
    let starts_bom = first_field_first_record && value.starts_with('\u{feff}');
    let edge_space = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
    let needs_quote = value.contains(config.delimiter as char)
        || value.contains(config.quote as char)
        || value.contains('\r')
        || value.contains('\n')
        || edge_space
        || starts_bom
        || (sole_field && value.is_empty());

    if !needs_quote {
        return value.to_string();
    }

    let mut out = String::with_capacity(value.len() + 2);
    out.push(config.quote as char);
    for ch in value.chars() {
        if ch == config.quote as char {
            out.push(ch);
        }
        out.push(ch);
    }
    out.push(config.quote as char);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_config(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let original_records = write_records(table);
    for (record_index, record) in original_records.iter().enumerate() {
        for (column_index, cell) in record.iter().enumerate() {
            if let Some(byte) = cell.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record: record_index,
                    column: column_index,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record_index, record) in original_records.iter().enumerate() {
        if record.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: record_index,
            });
        }
    }

    let mut records = original_records;
    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(CsvError::DuplicateHeader {
                    name,
                    first,
                    second,
                }) = duplicate_header_error(header)
                {
                    return Err(WriteError::DuplicateHeader {
                        name,
                        first,
                        second,
                    });
                }
            }

            let mut chosen = std::collections::HashMap::<String, usize>::new();
            for (index, name) in header.iter().enumerate() {
                match table.duplicates {
                    DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                        chosen.entry(name.clone()).or_insert(index);
                    }
                    DuplicatePolicy::KeepLast => {
                        chosen.insert(name.clone(), index);
                    }
                }
            }
            let mut columns: Vec<(String, usize)> =
                chosen.into_iter().map(|(name, index)| (name, index)).collect();
            columns.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));

            let projected_header: Vec<String> =
                columns.iter().map(|(name, _)| name.clone()).collect();
            let mut projected_rows: Vec<Vec<String>> = table
                .rows
                .iter()
                .map(|row| columns.iter().map(|(_, index)| row[*index].clone()).collect())
                .collect();
            projected_rows.sort_by(|a, b| bytewise_vec_cmp(a, b));

            records.clear();
            records.push(projected_header);
            records.extend(projected_rows);
        } else {
            records = table.rows.clone();
            records.sort_by(|a, b| bytewise_vec_cmp(a, b));
        }
    }

    if records.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    let terminator: &str = match config.mode {
        WriteMode::Canonical => "\n",
        WriteMode::Preserve => match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        },
    };

    for (record_index, record) in records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(terminator);
        }
        for (column_index, field) in record.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter as char);
            }
            output.push_str(&quote_field(
                field,
                record_index == 0 && column_index == 0,
                record.len() == 1,
                config,
            ));
        }
    }

    let add_final = match config.mode {
        WriteMode::Canonical => true,
        WriteMode::Preserve => config.final_terminator,
    };
    if add_final {
        output.push_str(terminator);
    }

    Ok(output)
}
