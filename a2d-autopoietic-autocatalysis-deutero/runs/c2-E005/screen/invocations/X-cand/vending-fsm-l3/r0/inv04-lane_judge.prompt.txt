pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    events: Vec<Event>,
}

const DENOMS: [u16; 4] = [1, 3, 4, 10];

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = Vec::<String>::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| id == &product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(product.id.clone());

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for (i, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            events: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let index = DENOMS.iter().position(|&d| d == coin);
        let Some(_) = index else {
            return Err(Error::BadCoin { coin });
        };

        let Some(new_credit) = self.credit.checked_add(coin) else {
            return Err(Error::CreditLimit);
        };
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit = new_credit;
        self.escrow.push(coin);
        self.log(EventKind::Inserted {
            coin,
            credit: new_credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|p| p.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        let id = id.to_owned();
        self.selected = Some(id.clone());
        self.log(EventKind::Selected { id });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };
        let product_index = self.products.iter().position(|p| p.id == id).unwrap();
        let product = &self.products[product_index];

        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let price = product.price;
        let amount = self.credit - price;
        let change = make_change(amount, self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        for &coin in &change {
            let i = DENOMS.iter().position(|&d| d == coin).unwrap();
            proposed[i] -= 1;
        }
        for &coin in &self.escrow {
            let i = DENOMS.iter().position(|&d| d == coin).unwrap();
            proposed[i] += 1;
        }
        for (i, count) in proposed.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        self.tubes = proposed;
        self.products[product_index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        let receipt = Receipt {
            id: id.clone(),
            price,
            change: change.clone(),
        };
        self.log(EventKind::Vended { id, price, change });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let Some(index) = self.products.iter().position(|p| p.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let Some(index) = self.products.iter().position(|p| p.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }

        let Some(stock) = self.products[index].stock.checked_add(count) else {
            return Err(Error::StockLimit { id: id.to_owned() });
        };
        if stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        self.products[index].stock = stock;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    fn log(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }
}

fn make_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut remaining = amount;
    let mut greedy_counts = [0u16; 4];
    for i in (0..4).rev() {
        let coin = DENOMS[i];
        let take = tubes[i].min(remaining / coin);
        greedy_counts[i] = take;
        remaining -= take * coin;
    }

    if remaining == 0 {
        return Some(counts_to_descending(greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    let mut counts = [0u16; 4];
    search_change(3, amount, &tubes, &mut counts, &mut best);

    best.map(counts_to_descending)
}

fn search_change(
    index: isize,
    remaining: u16,
    tubes: &[u16; 4],
    counts: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if index < 0 {
        if remaining != 0 {
            return;
        }
        let total: u32 = counts.iter().map(|&c| c as u32).sum();
        let replace = match best {
            None => true,
            Some(current) => {
                let current_total: u32 = current.iter().map(|&c| c as u32).sum();
                total < current_total
                    || (total == current_total
                        && (0..4).rev().find_map(|i| {
                            (counts[i] != current[i]).then_some(counts[i] > current[i])
                        }) == Some(true))
            }
        };
        if replace {
            *best = Some(*counts);
        }
        return;
    }

    let i = index as usize;
    let coin = DENOMS[i];
    let max_take = tubes[i].min(remaining / coin);
    for take in 0..=max_take {
        counts[i] = take;
        search_change(
            index - 1,
            remaining - take * coin,
            tubes,
            counts,
            best,
        );
    }
    counts[i] = 0;
}

fn counts_to_descending(counts: [u16; 4]) -> Vec<u16> {
    let mut result = Vec::new();
    for i in (0..4).rev() {
        for _ in 0..counts[i] {
            result.push(DENOMS[i]);
        }
    }
    result
}

        self.selected = Some(selected.clone());
        self.push_event(EventKind::Selected { id: selected });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = self.selected.clone().ok_or(Error::NoSelection)?;
        let product_index = self
            .products
            .iter()
            .position(|product| product.id == id)
            .expect("selected product must exist");
        let product = &self.products[product_index];

        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }

        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = choose_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for (index, count) in change_counts(&change).iter().enumerate() {
            proposed_tubes[index] -= count;
        }
        for coin in &self.escrow {
            let index = denomination_index(*coin).expect("escrow contains accepted coins");
            proposed_tubes[index] += 1;
        }
        for (index, count) in proposed_tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        self.products[product_index].stock -= 1;
        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        let receipt = Receipt {
            id: id.clone(),
            price,
            change: change.clone(),
        };
        self.push_event(EventKind::Vended { id, price, change });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.push_event(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.cmp(&b.id));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let index = self
            .products
            .iter()
            .position(|product| product.id == id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;

        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.push_event(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let index = self
            .products
            .iter()
            .position(|product| product.id == id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;

        if count == 0 {
            return Err(Error::ZeroRestock);
        }

        let new_stock = u32::from(self.products[index].stock) + u32::from(count);
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        self.products[index].stock = new_stock as u16;
        self.push_event(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: new_stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    fn push_event(&mut self, kind: EventKind) {
        self.events.push(Event {
            seq: self.events.len(),
            kind,
        });
    }
}

fn valid_id(id: &str) -> bool {
    !id.is_empty() && id.len() <= 8 && id.bytes().all(|byte| byte.is_ascii_lowercase())
}

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMINATIONS.iter().position(|&denomination| denomination == coin)
}

fn change_counts(change: &[u16]) -> [u16; 4] {
    let mut counts = [0u16; 4];
    for &coin in change {
        if let Some(index) = denomination_index(coin) {
            counts[index] += 1;
        }
    }
    counts
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut greedy_counts = [0u16; 4];
    let mut remaining = amount;
    for index in (0..4).rev() {
        let denomination = DENOMINATIONS[index];
        let taken = tubes[index].min(remaining / denomination);
        greedy_counts[index] = taken;
        remaining -= taken * denomination;
    }

    if remaining == 0 {
        return Some(counts_to_descending_vec(&greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    let mut current = [0u16; 4];
    search_change(3, amount, tubes, &mut current, &mut best);
    best.map(|counts| counts_to_descending_vec(&counts))
}

fn search_change(
    index: isize,
    remaining: u16,
    tubes: &[u16; 4],
    current: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if remaining == 0 {
        let is_better = match best {
            None => true,
            Some(previous) => {
                let current_total: u32 = current.iter().map(|&count| u32::from(count)).sum();
                let previous_total: u32 = previous.iter().map(|&count| u32::from(count)).sum();
                current_total < previous_total
                    || (current_total == previous_total
                        && lexicographically_greater_descending(current, previous))
            }
        };
        if is_better {
            *best = Some(*current);
        }
        return;
    }
    if index < 0 {
        return;
    }

    let denomination = DENOMINATIONS[index as usize];
    let max_count = tubes[index as usize].min(remaining / denomination);
    for count in (0..=max_count).rev() {
        current[index as usize] = count;
        let next_remaining = remaining - count * denomination;
        if let Some(previous) = best {
            let used: u32 = current.iter().map(|&value| u32::from(value)).sum();
            let previous_total: u32 =
                previous.iter().map(|&value| u32::from(value)).sum();
            if used > previous_total {
                continue;
            }
        }
        search_change(index - 1, next_remaining, tubes, current, best);
    }
    current[index as usize] = 0;
}

fn lexicographically_greater_descending(left: &[u16; 4], right: &[u16; 4]) -> bool {
    for index in (0..4).rev() {
        if left[index] != right[index] {
            return left[index] > right[index];
        }
    }
    false
}

fn counts_to_descending_vec(counts: &[u16; 4]) -> Vec<u16> {
    let total: usize = counts.iter().map(|&count| usize::from(count)).sum();
    let mut coins = Vec::with_capacity(total);
    for index in (0..4).rev() {
        coins.extend(std::iter::repeat_n(
            DENOMINATIONS[index],
            usize::from(counts[index]),
        ));
    }
    coins
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    if amount == 0 {
        return Some([0; 4]);
    }

    let mut greedy = [0u16; 4];
    let mut remaining = u32::from(amount);
    for i in (0..4).rev() {
        let denomination = u32::from(DENOMINATIONS[i]);
        let count = u32::from(tubes[i]).min(remaining / denomination);
        greedy[i] = count as u16;
        remaining -= count * denomination;
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u32::MAX;
    let mut candidate = [0u16; 4];
    search_change(
        3,
        u32::from(amount),
        tubes,
        &mut candidate,
        &mut best,
        &mut best_count,
    );
    best
}

fn search_change(
    index: i32,
    remaining: u32,
    tubes: &[u16; 4],
    candidate: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
    best_count: &mut u32,
) {
    if index < 0 {
        if remaining != 0 {
            return;
        }
        let count: u32 = candidate.iter().map(|&n| u32::from(n)).sum();
        let replace = match best {
            None => true,
            Some(current) => {
                count < *best_count
                    || (count == *best_count && lex_greater_desc(candidate, current))
            }
        };
        if replace {
            *best = Some(*candidate);
            *best_count = count;
        }
        return;
    }

    let i = index as usize;
    let denomination = u32::from(DENOMINATIONS[i]);
    let max = u32::from(tubes[i]).min(remaining / denomination);

    for count in 0..=max {
        let next_count = candidate
            .iter()
            .map(|&n| u32::from(n))
            .sum::<u32>()
            + count;
        if next_count > *best_count {
            break;
        }
        candidate[i] = count as u16;
        search_change(
            index - 1,
            remaining - count * denomination,
            tubes,
            candidate,
            best,
            best_count,
        );
    }
    candidate[i] = 0;
}

fn lex_greater_desc(a: &[u16; 4], b: &[u16; 4]) -> bool {
    for i in (0..4).rev() {
        if a[i] != b[i] {
            return a[i] > b[i];
        }
    }
    false
}

fn counts_to_descending_coins(counts: &[u16; 4]) -> Vec<u16> {
    let total: usize = counts.iter().map(|&n| usize::from(n)).sum();
    let mut coins = Vec::with_capacity(total);
    for i in (0..4).rev() {
        for _ in 0..counts[i] {
            coins.push(DENOMINATIONS[i]);
        }
    }
    coins
}

            id: id.clone(),
            price,
            change: change.clone(),
        };
        self.push_event(EventKind::Vended { id, price, change });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.push_event(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let index = self.product_index(id).ok_or_else(|| Error::UnknownProduct {
            id: id.to_owned(),
        })?;
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice {
                id: self.products[index].id.clone(),
            });
        }

        let product_id = self.products[index].id.clone();
        let old = self.products[index].price;
        self.products[index].price = price;
        self.push_event(EventKind::PriceChanged {
            id: product_id,
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let index = self.product_index(id).ok_or_else(|| Error::UnknownProduct {
            id: id.to_owned(),
        })?;
        if count == 0 {
            return Err(Error::ZeroRestock);
        }

        let stock = u32::from(self.products[index].stock) + u32::from(count);
        if stock > 100 {
            return Err(Error::StockLimit {
                id: self.products[index].id.clone(),
            });
        }

        let product_id = self.products[index].id.clone();
        let stock = stock as u16;
        self.products[index].stock = stock;
        self.push_event(EventKind::Restocked {
            id: product_id,
            added: count,
            stock,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    fn product_index(&self, id: &str) -> Option<usize> {
        self.products.iter().position(|product| product.id == id)
    }

    fn push_event(&mut self, kind: EventKind) {
        self.events.push(Event {
            seq: self.events.len(),
            kind,
        });
    }
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMINATIONS.iter().position(|&d| d == coin)
}

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut remaining = amount;
    let mut greedy_counts = [0u16; 4];
    for i in (0..4).rev() {
        let denomination = DENOMINATIONS[i];
        let take = tubes[i].min(remaining / denomination);
        greedy_counts[i] = take;
        remaining -= take * denomination;
    }

    if remaining == 0 {
        return Some(counts_to_descending_coins(greedy_counts));
    }

    let mut best: Option<([u16; 4], u32)> = None;
    search_change(
        3,
        amount,
        tubes,
        [0; 4],
        0,
        &mut best,
    );

    best.map(|(counts, _)| counts_to_descending_coins(counts))
}

fn search_change(
    index: isize,
    remaining: u16,
    tubes: [u16; 4],
    counts: [u16; 4],
    total: u32,
    best: &mut Option<([u16; 4], u32)>,
) {
    if remaining == 0 {
        let replace = match best {
            None => true,
            Some((best_counts, best_total)) => {
                total < *best_total
                    || (total == *best_total
                        && descending_lex_greater(&counts, best_counts))
            }
        };
        if replace {
            *best = Some((counts, total));
        }
        return;
    }

    if index < 0 {
        return;
    }

    let i = index as usize;
    let denomination = DENOMINATIONS[i];
    let max_take = tubes[i].min(remaining / denomination);
    for take in (0..=max_take).rev() {
        let next_total = total + u32::from(take);
        if let Some((_, best_total)) = best {
            if next_total > *best_total {
                continue;
            }
        }

        let mut next_counts = counts;
        next_counts[i] = take;
        search_change(
            index - 1,
            remaining - take * denomination,
            tubes,
            next_counts,
            next_total,
            best,
        );
    }
}

fn descending_lex_greater(a: &[u16; 4], b: &[u16; 4]) -> bool {
    for i in (0..4).rev() {
        if a[i] != b[i] {
            return a[i] > b[i];
        }
    }
    false
}

fn counts_to_descending_coins(counts: [u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for i in (0..4).rev() {
        for _ in 0..counts[i] {
            coins.push(DENOMINATIONS[i]);
        }
    }
    coins
}
