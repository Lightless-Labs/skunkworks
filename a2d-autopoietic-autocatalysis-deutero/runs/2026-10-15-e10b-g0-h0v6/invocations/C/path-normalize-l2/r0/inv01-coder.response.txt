#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.bytes().position(|byte| byte == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.bytes().position(|byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .bytes()
        .position(|byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .rev()
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn reduce(
    stack: &mut Vec<String>,
    parts: impl IntoIterator<Item = (String, usize)>,
    absolute: bool,
) -> Result<(), PathError> {
    for (part, at) in parts {
        match part.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push(part);
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, intent: bool) -> String {
    let joined = stack.join("/");
    let mut result = if absolute {
        if joined.is_empty() {
            "/".to_owned()
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_owned()
    } else {
        joined
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn parts_with_offsets(path: &str, base: usize) -> Vec<(String, usize)> {
    let mut parts = Vec::new();
    let mut offset = base;

    for part in path.split('/') {
        if !part.is_empty() {
            parts.push((part.to_owned(), offset));
        }
        offset += part.len() + 1;
    }

    parts
}

fn normalize_validated(path: &str) -> Result<String, PathError> {
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();

    reduce(&mut stack, parts_with_offsets(path, 0), absolute)?;
    Ok(render(&stack, absolute, intent))
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    normalize_validated(path)
}

fn validate_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') || validate_raw(path).is_err() {
        return Err(());
    }

    let normalized = normalize_validated(path).map_err(|_| ())?;
    Ok(normalized
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect())
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(validate_context(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let is_tilde = bytes.first() == Some(&b'~');
    let expanded_home = if is_tilde {
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            Some(home_components.as_ref().ok_or(PathError::MissingHome)?)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else {
        None
    };

    let absolute = !is_tilde && bytes.first() == Some(&b'/');
    let mut stack = if absolute {
        Vec::new()
    } else if let Some(home) = expanded_home {
        home.clone()
    } else {
        cwd_components
    };

    let start = if is_tilde {
        if bytes.get(1) == Some(&b'/') { 2 } else { 1 }
    } else {
        0
    };

    let intent = directory_intent(path);
    reduce(&mut stack, parts_with_offsets(&path[start..], start), true)?;
    Ok(render(&stack, true, intent))
}

fn main() {}
