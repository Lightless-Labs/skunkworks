```rust
use std::collections::VecDeque;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Process {
    id: u32,
    arrival: u64,
    bursts: Vec<u32>,
    burst_index: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
}

#[derive(Clone, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    wake_level: usize,
}

fn admit_through(
    limit: u64,
    events: &[Event],
    cursor: &mut usize,
    queues: &mut [VecDeque<usize>],
    processes: &[Process],
) {
    while *cursor < events.len() && events[*cursor].time <= limit {
        let event = &events[*cursor];
        let level = if event.kind == 0 {
            processes[event.index].level
        } else {
            event.wake_level
        };
        queues[level].push_back(event.index);
        *cursor += 1;
    }
}

fn round_mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        0
    } else {
        (total * 100 + count as u64 / 2) / count as u64
    }
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let mut processes: Vec<Process> = jobs
        .iter()
        .map(|job| Process {
            id: job.id,
            arrival: job.arrival as u64,
            bursts: job.bursts.clone(),
            burst_index: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: job.arrival as u64,
            kind: 0,
            index,
            wake_level: 0,
        })
        .collect();
    events.sort_by_key(|event| (event.time, event.kind, event.index));

    let mut queues: Vec<VecDeque<usize>> =
        (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut event_cursor = 0usize;
    let mut now = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();

    while processes.iter().any(|process| process.completion.is_none()) {
        admit_through(now, &events, &mut event_cursor, &mut queues, &processes);

        let Some(level) = queues.iter().position(|queue| !queue.is_empty()) else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = events[event_cursor].time;
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let process_index = queues[level].pop_front().unwrap();
        let selected_id = processes[process_index].id;

        if last_dispatch_id.is_some_and(|previous_id| previous_id != selected_id) {
            now += switch_cost as u64;
            admit_through(now, &events, &mut event_cursor, &mut queues, &processes);
        }

        let run_start = now;
        let quantum = quanta[level];
        let consumed = processes[process_index].remaining.min(quantum);
        now += consumed as u64;
        let run_end = now;

        processes[process_index].remaining -= consumed;
        let burst_finished = processes[process_index].remaining == 0;
        let job_finished = burst_finished
            && processes[process_index].burst_index + 1 == processes[process_index].bursts.len();

        runs.push(Run {
            id: selected_id,
            start: run_start,
            end: run_end,
            finished: job_finished,
            level,
        });

        if job_finished {
            processes[process_index].completion = Some(run_end);
        } else if burst_finished {
            let io_position = processes[process_index].burst_index + 1;
            let io_delay = processes[process_index].bursts[io_position] as u64;
            let next_cpu_position = processes[process_index].burst_index + 2;
            let wake_level = level.saturating_sub(1);

            processes[process_index].burst_index = next_cpu_position;
            processes[process_index].remaining =
                processes[process_index].bursts[next_cpu_position];

            events.push(Event {
                time: run_end + io_delay,
                kind: 1,
                index: process_index,
                wake_level,
            });
            events[event_cursor..].sort_by_key(|event| (event.time, event.kind, event.index));
        }

        admit_through(run_end, &events, &mut event_cursor, &mut queues, &processes);

        if !burst_finished {
            processes[process_index].level = (level + 1).min(quanta.len() - 1);
            queues[processes[process_index].level].push_back(process_index);
        }

        last_dispatch_id = Some(selected_id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, process) in processes.iter().enumerate() {
        let completion = process.completion.unwrap_or(now);
        let turnaround = completion - process.arrival;
        let cpu_total: u64 = process
            .bursts
            .iter()
            .step_by(2)
            .map(|&length| length as u64)
            .sum();
        let io_total: u64 = process
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&length| length as u64)
            .sum();
        let waiting = turnaround - cpu_total - io_total;

        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: jobs[index].id,
            completion,
            turnaround,
            waiting,
        });
    }

    let end = processes
        .iter()
        .filter_map(|process| process.completion)
        .max()
        .unwrap_or(0);

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: round_mean_x100(total_waiting, jobs.len()),
        mean_turnaround_x100: round_mean_x100(total_turnaround, jobs.len()),
        end,
    })
}

fn main() {}
```
