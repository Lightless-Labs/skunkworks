use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }

    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;

    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

fn parse_error(at: usize, kind: ParseKind) -> RationalError {
    RationalError::Parse { at, kind }
}

fn parse_magnitude(bytes: &[u8], limit: u128, error_at: usize) -> Result<u128, RationalError> {
    let mut value = 0u128;
    for &byte in bytes {
        let digit = (byte - b'0') as u128;
        if value > (limit - digit) / 10 {
            return Err(parse_error(error_at, ParseKind::OutOfRange));
        }
        value = value * 10 + digit;
    }
    Ok(value)
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let left_scale = rhs.den as i128 / g;
        let right_scale = self.den as i128 / g;
        let n = self.num as i128 * left_scale + rhs.num as i128 * right_scale;
        let d = self.den as i128 * left_scale;
        rational_from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let left_scale = rhs.den as i128 / g;
        let right_scale = self.den as i128 / g;
        let n = self.num as i128 * left_scale - rhs.num as i128 * right_scale;
        let d = self.den as i128 * left_scale;
        rational_from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.den as u128) as i128;
        let g2 = gcd_u128(rhs.num.unsigned_abs() as u128, self.den as u128) as i128;
        let n = (self.num as i128 / g1) * (rhs.num as i128 / g2);
        let d = (self.den as i128 / g2) * (rhs.den as i128 / g1);
        rational_from_i128(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.num.unsigned_abs() as u128) as i128;
        let g2 = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let n = (self.num as i128 / g1) * (rhs.den as i128 / g2);
        let d = (self.den as i128 / g2) * (rhs.num as i128 / g1);
        rational_from_i128(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        rational_from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let negative = bytes[0] == b'-';
        let num_start = if negative { 1 } else { 0 };
        let mut i = num_start;

        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }

        if i == num_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let num_end = i;
        let (den_start, den_end) = if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            let start = i + 1;
            i = start;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            if i < bytes.len() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            (Some(start), Some(i))
        } else {
            (None, None)
        };

        if num_end - num_start >= 2 && bytes[num_start] == b'0' {
            return Err(parse_error(num_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_end == 2 && bytes[1] == b'0' {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }
        if let (Some(start), Some(end)) = (den_start, den_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_error(start + 1, ParseKind::NonCanonical));
            }
        }

        let magnitude = parse_magnitude(
            &bytes[num_start..num_end],
            if negative { 1u128 << 63 } else { i64::MAX as u128 },
            0,
        )?;

        let mut den_value = 1u128;
        if let (Some(start), Some(end)) = (den_start, den_end) {
            den_value = parse_magnitude(&bytes[start..end], i64::MAX as u128, start)?;
            if den_value == 0 {
                return Err(parse_error(start, ParseKind::ZeroDenominator));
            }

            let slash = start - 1;
            if den_value == 1 {
                return Err(parse_error(slash, ParseKind::NonCanonical));
            }
            if magnitude == 0 {
                return Err(parse_error(slash, ParseKind::NonCanonical));
            }
            if gcd_u128(magnitude, den_value) > 1 {
                return Err(parse_error(slash, ParseKind::NonCanonical));
            }
        }

        let num = if negative {
            if magnitude == (1u128 << 63) {
                i64::MIN
            } else {
                -(magnitude as i64)
            }
        } else {
            magnitude as i64
        };

        Ok(Rational {
            num,
            den: den_value as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch {
        coefficients: usize,
        rhs: usize,
    },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

fn matrix_shape_len(rows: usize, cols: usize) -> Result<usize, MatrixError> {
    rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)
}

fn arithmetic_error(row: usize, col: usize, step: usize, source: RationalError) -> MatrixError {
    MatrixError::Arithmetic {
        row,
        col,
        step,
        source,
    }
}

fn pivot_key(value: Rational, row: usize) -> (i64, u64, usize) {
    (value.den, value.num.unsigned_abs(), row)
}

fn eliminate(
    matrix: &mut Matrix,
    candidate_cols: usize,
) -> Result<(Vec<usize>, Vec<(usize, usize)>), MatrixError> {
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;
    let rows = matrix.rows;
    let cols = matrix.cols;

    for candidate_col in 0..candidate_cols {
        if p == rows {
            break;
        }

        let mut chosen: Option<usize> = None;
        for row in p..rows {
            let value = matrix.data[row * cols + candidate_col];
            if value.num == 0 {
                continue;
            }
            match chosen {
                None => chosen = Some(row),
                Some(old) => {
                    if pivot_key(value, row)
                        .cmp(&pivot_key(matrix.data[old * cols + candidate_col], old))
                        == Ordering::Less
                    {
                        chosen = Some(row);
                    }
                }
            }
        }

        let q = match chosen {
            Some(q) => q,
            None => continue,
        };

        if q != p {
            for col in 0..cols {
                matrix.data.swap(p * cols + col, q * cols + col);
            }
            swaps.push((p, q));
        }

        let pivot = matrix.data[p * cols + candidate_col];
        for col in 0..cols {
            let original = matrix.data[p * cols + col];
            matrix.data[p * cols + col] = original
                .div(&pivot)
                .map_err(|e| arithmetic_error(p, col, p, e))?;
        }

        for row in 0..rows {
            if row == p {
                continue;
            }
            let factor = matrix.data[row * cols + candidate_col];
            if factor.num == 0 {
                continue;
            }
            for col in 0..cols {
                let product = factor
                    .mul(&matrix.data[p * cols + col])
                    .map_err(|e| arithmetic_error(row, col, p, e))?;
                matrix.data[row * cols + col] = matrix.data[row * cols + col]
                    .sub(&product)
                    .map_err(|e| arithmetic_error(row, col, p, e))?;
            }
        }

        pivots.push(candidate_col);
        p += 1;
    }

    Ok((pivots, swaps))
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = matrix_shape_len(rows, cols)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for row in 0..self.cols {
            for col in 0..self.rows {
                data.push(self.data[col * self.cols + row]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let out_len = matrix_shape_len(self.rows, rhs.cols)?;
        let mut data = Vec::with_capacity(out_len);
        let zero = Rational { num: 0, den: 1 };

        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let left = self.data[row * self.cols + k];
                    let right = rhs.data[k * rhs.cols + col];
                    let term = left
                        .mul(&right)
                        .map_err(|e| arithmetic_error(row, col, k, e))?;
                    acc = acc
                        .add(&term)
                        .map_err(|e| arithmetic_error(row, col, k, e))?;
                }
                data.push(acc);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let (pivots, swaps) = eliminate(&mut matrix, self.cols)?;
        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }

        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };
        let one = Rational { num: 1, den: 1 };

        for row in 0..n {
            for col in 0..n {
                data.push(self.data[row * n + col]);
            }
            for col in 0..n {
                data.push(if row == col { one } else { zero });
            }
        }

        let mut augmented = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let (pivots, _) = eliminate(&mut augmented, n)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }

        let result_len = n.checked_mul(n).ok_or(MatrixError::ShapeOverflow)?;
        let mut result = Vec::with_capacity(result_len);
        for row in 0..n {
            for col in 0..n {
                result.push(augmented.data[row * width + n + col]);
            }
        }
        Ok(Matrix {
            rows: n,
            cols: n,
            data: result,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }

        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut augmented_data = Vec::with_capacity(len);
        for row in 0..self.rows {
            let a_start = row * self.cols;
            let b_start = row * rhs.cols;
            augmented_data.extend_from_slice(&self.data[a_start..a_start + self.cols]);
            augmented_data.extend_from_slice(&rhs.data[b_start..b_start + rhs.cols]);
        }

        let mut augmented = Matrix {
            rows: self.rows,
            cols: width,
            data: augmented_data,
        };

        let (pivots, _) = eliminate(&mut augmented, n).map_err(SolveError::Matrix)?;

        for row in 0..self.rows {
            let row_start = row * width;
            let mut coeff_zero = true;
            for col in 0..n {
                if augmented.data[row_start + col].num != 0 {
                    coeff_zero = false;
                    break;
                }
            }
            if coeff_zero {
                for rhs_col in 0..k {
                    let value = augmented.data[row_start + n + rhs_col];
                    if value.num != 0 {
                        return Ok(Solution::None {
                            row,
                            rhs_col,
                            witness: augmented.data[row_start..row_start + width].to_vec(),
                        });
                    }
                }
            }
        }

        let rank = pivots.len();
        let mut is_pivot = vec![false; n];
        for &col in &pivots {
            is_pivot[col] = true;
        }
        let mut free = Vec::new();
        for col in 0..n {
            if !is_pivot[col] {
                free.push(col);
            }
        }

        let particular_len = n
            .checked_mul(k)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let zero = Rational { num: 0, den: 1 };
        let mut particular_data = vec![zero; particular_len];
        for (pivot_row, &pivot_col) in pivots.iter().enumerate() {
            for rhs_col in 0..k {
                particular_data[pivot_col * k + rhs_col] =
                    augmented.data[pivot_row * width + n + rhs_col];
            }
        }
        let particular = Matrix {
            rows: n,
            cols: k,
            data: particular_data,
        };

        if rank == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots,
            });
        }

        let f = free.len();
        let basis_len = n
            .checked_mul(f)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![zero; basis_len];

        for (basis_col, &free_col) in free.iter().enumerate() {
            basis_data[free_col * f + basis_col] = Rational { num: 1, den: 1 };
            for (pivot_row, &pivot_col) in pivots.iter().enumerate() {
                let coefficient = augmented.data[pivot_row * width + free_col];
                basis_data[pivot_col * f + basis_col] =
                    coefficient.neg().map_err(|source| {
                        SolveError::Matrix(arithmetic_error(
                            pivot_col,
                            basis_col,
                            rank,
                            source,
                        ))
                    })?;
            }
        }

        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix {
                rows: n,
                cols: f,
                data: basis_data,
            },
            pivots,
        })
    }
}

fn main() {}
