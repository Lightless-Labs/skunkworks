#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedSystem {
    BalancedTernary,
    Negabinary,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedError {
    Empty,
    InvalidByte { index: usize },
    InvalidChecksum { index: usize },
    LeadingZero,
    Range { index: usize },
    ChecksumMismatch { expected: u8, found: u8 },
    OutOfRange,
}

const LIMIT: i128 = (1i128 << 120) - 1;

pub fn signed_encode(
    value: i128,
    system: SignedSystem,
    checksum: bool,
) -> Result<String, SignedError> {
    if value < -LIMIT || value > LIMIT {
        return Err(SignedError::OutOfRange);
    }

    let mut n = value;
    let mut digits = Vec::new();

    if n == 0 {
        digits.push(b'0');
    } else {
        match system {
            SignedSystem::BalancedTernary => {
                while n != 0 {
                    let rem = n.rem_euclid(3);
                    let digit = match rem {
                        0 => 0,
                        1 => 1,
                        _ => -1,
                    };
                    digits.push(match digit {
                        -1 => b'T',
                        0 => b'0',
                        _ => b'1',
                    });
                    n = (n - digit) / 3;
                }
            }
            SignedSystem::Negabinary => {
                while n != 0 {
                    let digit = n.rem_euclid(2);
                    digits.push(if digit == 0 { b'0' } else { b'1' });
                    n = (n - digit) / -2;
                }
            }
        }
        digits.reverse();
    }

    let mut result = String::from_utf8(digits).expect("ASCII digits are valid UTF-8");
    if checksum {
        let sum = result
            .bytes()
            .enumerate()
            .map(|(i, b)| {
                let code = match b {
                    b'T' => 2u64,
                    b'0' => 0,
                    b'1' => 1,
                    _ => 0,
                };
                ((i as u64 + 1) * code) % 7
            })
            .sum::<u64>()
            % 7;
        result.push(':');
        result.push(char::from(b'0' + sum as u8));
    }
    Ok(result)
}

pub fn signed_decode(input: &str, system: SignedSystem) -> Result<i128, SignedError> {
    if input.is_empty() {
        return Err(SignedError::Empty);
    }

    let bytes = input.as_bytes();
    let colon = bytes.iter().position(|&b| b == b':');
    let payload_end = colon.unwrap_or(bytes.len());

    if payload_end == 0 {
        return Err(SignedError::Empty);
    }

    for (index, &b) in bytes[..payload_end].iter().enumerate() {
        let valid = match system {
            SignedSystem::BalancedTernary => matches!(b, b'T' | b'0' | b'1'),
            SignedSystem::Negabinary => matches!(b, b'0' | b'1'),
        };
        if !valid {
            return Err(SignedError::InvalidByte { index });
        }
    }

    let supplied_checksum = if let Some(colon_index) = colon {
        let suffix = &bytes[colon_index + 1..];
        if suffix.len() != 1 || !matches!(suffix[0], b'0'..=b'6') {
            return Err(SignedError::InvalidChecksum { index: colon_index });
        }
        Some(suffix[0] - b'0')
    } else {
        None
    };

    let payload = &bytes[..payload_end];
    if payload.len() > 1 && payload[0] == b'0' {
        return Err(SignedError::LeadingZero);
    }

    let (base, low_digit, high_digit) = match system {
        SignedSystem::BalancedTernary => (3i128, -1i128, 1i128),
        SignedSystem::Negabinary => (-2i128, 0i128, 1i128),
    };

    let mut accumulator = 0i128;
    for (index, &b) in payload.iter().enumerate() {
        let digit = match b {
            b'T' => -1,
            b'0' => 0,
            b'1' => 1,
            _ => unreachable!(),
        };
        debug_assert!(digit >= low_digit && digit <= high_digit);

        let next = accumulator
            .checked_mul(base)
            .and_then(|v| v.checked_add(digit))
            .ok_or(SignedError::Range { index })?;
        if !(-LIMIT..=LIMIT).contains(&next) {
            return Err(SignedError::Range { index });
        }
        accumulator = next;
    }

    if let Some(found) = supplied_checksum {
        let expected = payload
            .iter()
            .enumerate()
            .map(|(i, &b)| {
                let code = match b {
                    b'T' => 2usize,
                    b'0' => 0,
                    b'1' => 1,
                    _ => 0,
                };
                (((i + 1) % 7) * code) % 7
            })
            .sum::<usize>()
            % 7;
        if found != expected as u8 {
            return Err(SignedError::ChecksumMismatch {
                expected: expected as u8,
                found,
            });
        }
    }

    Ok(accumulator)
}

pub fn signed_recode(
    input: &str,
    from: SignedSystem,
    to: SignedSystem,
    checksum: bool,
) -> Result<String, SignedError> {
    let value = signed_decode(input, from)?;
    signed_encode(value, to, checksum)
}

fn main() {}
