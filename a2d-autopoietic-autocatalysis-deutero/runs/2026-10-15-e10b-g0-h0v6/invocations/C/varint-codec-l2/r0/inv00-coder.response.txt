#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
}

pub fn encode_u64(mut value: u64) -> Vec<u8> {
    let mut bytes = Vec::new();
    loop {
        let payload = (value & 0x7f) as u8;
        value >>= 7;
        if value == 0 {
            bytes.push(payload);
            break;
        }
        bytes.push(payload | 0x80);
    }
    bytes
}

pub fn decode_u64(input: &[u8], start: usize) -> Result<(u64, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut value = 0u64;
    let mut index = start;

    loop {
        if index >= input.len() {
            return Err(Error::Truncated {
                offset: input.len(),
                start,
            });
        }

        let byte = input[index];
        let payload = (byte & 0x7f) as u64;
        let group = index - start;

        if group == 9 {
            if payload > 1 || byte & 0x80 != 0 {
                return Err(Error::Overflow {
                    offset: index,
                    start,
                });
            }
            value |= payload << 63;
            return Ok((value, index + 1));
        }

        value |= payload << (group * 7);
        index += 1;

        if byte & 0x80 == 0 {
            return Ok((value, index));
        }
    }
}

pub fn decode_many(
    input: &[u8],
    start: usize,
    count: usize,
) -> Result<(Vec<u64>, usize), Error> {
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut values = Vec::with_capacity(count);
    let mut cursor = start;
    for _ in 0..count {
        let (value, next) = decode_u64(input, cursor)?;
        values.push(value);
        cursor = next;
    }
    Ok((values, cursor))
}

pub fn encode_i64(value: i64) -> Vec<u8> {
    let zigzag = if value >= 0 {
        (value as u64) << 1
    } else {
        ((-(value + 1)) as u64) * 2 + 1
    };
    encode_u64(zigzag)
}

pub fn decode_i64(input: &[u8], start: usize) -> Result<(i64, usize), Error> {
    let (value, cursor) = decode_u64(input, start)?;
    let decoded = if value & 1 == 0 {
        (value >> 1) as i64
    } else {
        -((value >> 1) as i64) - 1
    };
    Ok((decoded, cursor))
}

pub struct StreamDecoder {
    position: usize,
    pending: Vec<u8>,
    pending_start: usize,
}

impl StreamDecoder {
    pub fn new() -> Self {
        Self {
            position: 0,
            pending: Vec::new(),
            pending_start: 0,
        }
    }

    pub fn push(&mut self, chunk: &[u8]) -> Result<Vec<u64>, Error> {
        let mut bytes = self.pending.clone();
        bytes.extend_from_slice(chunk);

        let mut values = Vec::new();
        let mut cursor = 0usize;
        let mut next_pending_start = self.pending_start;

        while cursor < bytes.len() {
            let integer_start = cursor;
            let absolute_start = if integer_start < self.pending.len() {
                self.pending_start
            } else {
                self.position + (integer_start - self.pending.len())
            };

            match decode_u64(&bytes, integer_start) {
                Ok((value, next)) => {
                    values.push(value);
                    cursor = next;
                    next_pending_start = self.position + chunk.len();
                }
                Err(Error::Truncated { .. }) => {
                    break;
                }
                Err(Error::Overflow { offset, .. }) => {
                    let absolute_offset = if offset < self.pending.len() {
                        self.pending_start + offset
                    } else {
                        self.position + (offset - self.pending.len())
                    };
                    return Err(Error::Overflow {
                        offset: absolute_offset,
                        start: absolute_start,
                    });
                }
                Err(error) => return Err(error),
            }
        }

        let new_position = self.position + chunk.len();
        let remaining = bytes[cursor..].to_vec();
        if remaining.is_empty() {
            self.pending_start = new_position;
        } else if cursor < self.pending.len() {
            self.pending_start = self.pending_start + cursor;
        } else {
            self.pending_start = self.position + (cursor - self.pending.len());
        }
        self.pending = remaining;
        self.position = new_position;
        let _ = next_pending_start;

        Ok(values)
    }

    pub fn finish(&self) -> Result<(), Error> {
        if self.pending.is_empty() {
            Ok(())
        } else {
            Err(Error::Truncated {
                offset: self.position,
                start: self.position - self.pending.len(),
            })
        }
    }

    pub fn position(&self) -> usize {
        self.position
    }

    pub fn pending_len(&self) -> usize {
        self.pending.len()
    }

    pub fn reset(&mut self) {
        self.position = 0;
        self.pending.clear();
        self.pending_start = 0;
    }
}

impl Default for StreamDecoder {
    fn default() -> Self {
        Self::new()
    }
}

fn main() {}
