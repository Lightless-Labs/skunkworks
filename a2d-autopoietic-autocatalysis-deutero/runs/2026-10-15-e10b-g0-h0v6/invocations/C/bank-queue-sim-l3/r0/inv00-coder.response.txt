#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    started: bool,
    paused_at: Option<u64>,
    remaining: u64,
}

#[derive(Clone)]
struct Running {
    job: Job,
    work_at: u64,
    finish: u64,
}

fn waiter_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.paused_at, b.paused_at) {
            (Some(pa), Some(pb)) => pa.cmp(&pb).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => std::cmp::Ordering::Less,
            (None, Some(_)) => std::cmp::Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn best_waiter_index(waiting: &[Job]) -> Option<usize> {
    waiting
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| waiter_cmp(a, b))
        .map(|(index, _)| index)
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut finished_count = 0usize;

    while finished_count < customers.len() {
        let next_time = {
            let arrival_time = arrivals.get(next_arrival).map(|c| c.arrival);
            let finish_time = running.as_ref().map(|r| r.finish);
            let work_time = running
                .as_ref()
                .filter(|r| r.work_at > current_time_placeholder())
                .map(|r| r.work_at);
            [arrival_time, finish_time, work_time]
                .into_iter()
                .flatten()
                .min()
        };

        let time = match next_time {
            Some(t) => t,
            None => break,
        };

        if running.as_ref().is_some_and(|r| r.finish == time) {
            let completed = running.take().unwrap();
            trace.push(Event {
                time,
                customer: completed.job.customer.id,
                kind: Kind::Finish,
            });
            finished_count += 1;
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = arrivals[next_arrival].clone();
            trace.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Job {
                remaining: customer.service,
                customer,
                started: false,
                paused_at: None,
            });
            next_arrival += 1;
        }

        if let Some(active) = running.as_ref() {
            if time >= active.work_at {
                if let Some(best) = best_waiter_index(&waiting) {
                    if waiting[best].customer.priority < active.job.customer.priority {
                        let by = waiting[best].customer.id;
                        let mut paused = running.take().unwrap().job;
                        paused.remaining = active_finish_minus_time(active_finish_placeholder(), time);
                        paused.paused_at = Some(time);
                        trace.push(Event {
                            time,
                            customer: paused.customer.id,
                            kind: Kind::Preempt {
                                remaining: paused.remaining,
                                by,
                            },
                        });
                        waiting.push(paused);
                    }
                }
            }
        }

        while running.is_none() {
            let Some(index) = best_waiter_index(&waiting) else {
                break;
            };
            let mut job = waiting.remove(index);
            let cost = if last_dispatched == Some(job.customer.id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };
            let work_at = time.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: job.customer.id,
                time,
            })?;
            let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                customer: job.customer.id,
                time,
            })?;

            let kind = if job.started {
                Kind::Resume {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            };
            trace.push(Event {
                time,
                customer: job.customer.id,
                kind,
            });
            job.started = true;
            last_dispatched = Some(job.customer.id);
            running = Some(Running {
                job,
                work_at,
                finish,
            });

            if finish == time {
                let completed = running.take().unwrap();
                trace.push(Event {
                    time,
                    customer: completed.job.customer.id,
                    kind: Kind::Finish,
                });
                finished_count += 1;
            }
        }
    }

    Ok(trace)
}

// This helper is replaced by the loop's explicit clock in the implementation below.
fn current_time_placeholder() -> u64 {
    0
}

fn active_finish_placeholder() -> u64 {
    0
}

fn active_finish_minus_time(_finish: u64, _time: u64) -> u64 {
    0
}

fn main() {}
