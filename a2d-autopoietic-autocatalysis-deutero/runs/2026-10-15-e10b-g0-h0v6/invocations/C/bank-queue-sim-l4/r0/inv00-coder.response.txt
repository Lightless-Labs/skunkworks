#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(if round_up { q + 1 } else { q })
}

#[derive(Clone)]
struct Waiter {
    idx: usize,
    entry: u64,
    remaining: u64,
    deadline: Option<u64>,
    waited: u64,
    paused: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Active {
    idx: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiter_cmp(a: &Waiter, b: &Waiter, customers: &[Customer]) -> std::cmp::Ordering {
    let ca = &customers[a.idx];
    let cb = &customers[b.idx];
    ca.priority
        .cmp(&cb.priority)
        .then_with(|| b.paused.cmp(&a.paused))
        .then_with(|| {
            if a.paused {
                a.pause_time.cmp(&b.pause_time)
                    .then_with(|| ca.id.cmp(&cb.id))
            } else {
                ca.arrival.cmp(&cb.arrival)
                    .then_with(|| cb.id.cmp(&ca.id))
            }
        })
}

fn best_waiter(waiters: &[Waiter], customers: &[Customer]) -> Option<usize> {
    waiters
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| waiter_cmp(a, b, customers))
        .map(|(i, _)| i)
}

fn push_event(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event { time, customer: id, kind });
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut initial_deadlines = vec![None; customers.len()];
    for (i, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index: i });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index: i, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index: i, value: c.priority });
        }
        if let Some(p) = c.patience {
            initial_deadlines[i] = Some(c.arrival.checked_add(p).ok_or(
                Error::DeadlineOverflow { index: i, id: c.id },
            )?);
        }
    }

    let n = customers.len();
    let mut order: Vec<usize> = (0..n).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut waiters: Vec<Waiter> = Vec::new();
    let mut active: Option<Active> = None;
    let mut cursor = 0usize;
    let mut now = 0u64;
    let mut last_dispatch: Option<u32> = None;
    let mut ever_started = vec![false; n];

    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();

    while cursor < n || active.is_some() || !waiters.is_empty() {
        if active.is_none() && waiters.is_empty() && cursor < n {
            now = now.max(customers[order[cursor]].arrival);
        }

        // Phase A: completion at this tick wins over all other phases.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let c = &customers[a.idx];
            push_event(&mut events, now, c.id, Kind::Finish);
            completed += 1;
            turnaround_ticks += (now - c.arrival) as u128;
            completion_order.push(c.id);
        }

        // Phase B: expire waiters present at the start of the tick.
        let mut expired: Vec<usize> = waiters
            .iter()
            .enumerate()
            .filter(|(_, w)| w.deadline.is_some_and(|d| d <= now))
            .map(|(i, _)| i)
            .collect();
        expired.sort_by(|&i, &j| {
            waiters[i].deadline.cmp(&waiters[j].deadline)
                .then_with(|| customers[waiters[i].idx].id.cmp(&customers[waiters[j].idx].id))
        });
        for i in expired.into_iter().rev() {
            let w = waiters.remove(i);
            let c = &customers[w.idx];
            let total = w.waited + (now - w.entry);
            wait_ticks += (now - w.entry) as u128;
            push_event(&mut events, now, c.id, Kind::Renege { waited: total });
            reneged += 1;
        }

        // Phase C: process the whole arrival cohort without dispatching.
        while cursor < n && customers[order[cursor]].arrival == now {
            let idx = order[cursor];
            cursor += 1;
            let c = &customers[idx];
            push_event(&mut events, now, c.id, Kind::Arrive);
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiters.len() >= gate {
                push_event(&mut events, now, c.id, Kind::Balk);
                balked += 1;
            } else {
                accepted += 1;
                waiters.push(Waiter {
                    idx,
                    entry: now,
                    remaining: c.service,
                    deadline: initial_deadlines[idx],
                    waited: 0,
                    paused: false,
                    pause_time: 0,
                });
            }
        }

        // Phase D: preempt only after switching protection ends.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(wi) = best_waiter(&waiters, customers) {
                    let running_priority = customers[a.idx].priority;
                    let waiting_priority = customers[waiters[wi].idx].priority;
                    if waiting_priority < running_priority {
                        let a = active.take().unwrap();
                        let by = customers[waiters[wi].idx].id;
                        let c = &customers[a.idx];
                        let remaining = a.finish - now;
                        push_event(&mut events, now, c.id, Kind::Preempt { remaining, by });
                        preemptions += 1;
                        busy_ticks += (now - a_dispatch_time(&events, c.id, now)) as u128;
                        let deadline = match c.patience {
                            Some(_) => Some(now.checked_add(
                                customers[a.idx].patience.unwrap().saturating_sub(
                                    // Remaining budget is tracked by subtracting spent segments below.
                                    0
                                )
                            ).unwrap_or(u64::MAX)),
                            None => None,
                        };
                        let spent = 0u64;
                        let mut paused = Waiter {
                            idx: a.idx,
                            entry: now,
                            remaining,
                            deadline,
                            waited: 0,
                            paused: true,
                            pause_time: now,
                        };
                        // Preserve cumulative budget using original patience minus prior waiting.
                        if let Some(p) = c.patience {
                            let consumed = total_wait_for(&events, c.id);
                            let budget = p.saturating_sub(consumed);
                            paused.deadline = Some(now.checked_add(budget).ok_or(Error::TimeOverflow {
                                customer: c.id,
                                time: now,
                                stage: OverflowStage::Patience,
                            })?);
                        }
                        let _ = spent;
                        waiters.push(paused);
                    }
                }
            }
        }

        // Phase E: dispatch while idle, including immediate zero-time completions.
        loop {
            if active.is_some() {
                break;
            }
            let Some(wi) = best_waiter(&waiters, customers) else { break };
            let mut w = waiters.remove(wi);
            let c = &customers[w.idx];
            let segment = now - w.entry;
            w.waited = w.waited.checked_add(segment).unwrap();
            wait_ticks += segment as u128;
            if let Some(d) = w.deadline {
                let left = d.saturating_sub(now);
                w.deadline = Some(now + left);
            }

            let switch = if last_dispatch.is_some_and(|id| id != c.id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: c.id, time: now, stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(w.remaining).ok_or(Error::TimeOverflow {
                customer: c.id, time: now, stage: OverflowStage::Finish,
            })?;

            let first = !ever_started[w.idx];
            if first {
                started += 1;
                ever_started[w.idx] = true;
            }
            push_event(
                &mut events,
                now,
                c.id,
                if first {
                    Kind::Start { remaining: w.remaining, work_at, finish }
                } else {
                    Kind::Resume { remaining: w.remaining, work_at, finish }
                },
            );
            last_dispatch = Some(c.id);
            active = Some(Active { idx: w.idx, remaining: w.remaining, work_at, finish });
            if finish == now {
                let a = active.take().unwrap();
                push_event(&mut events, now, c.id, Kind::Finish);
                completed += 1;
                turnaround_ticks += (now - c.arrival) as u128;
                completion_order.push(c.id);
                continue;
            }
            break;
        }

        // Phase F: newly queued expirations.
        let mut expired: Vec<usize> = waiters
            .iter()
            .enumerate()
            .filter(|(_, w)| w.deadline.is_some_and(|d| d <= now))
            .map(|(i, _)| i)
            .collect();
        expired.sort_by(|&i, &j| {
            waiters[i].deadline.cmp(&waiters[j].deadline)
                .then_with(|| customers[waiters[i].idx].id.cmp(&customers[waiters[j].idx].id))
        });
        for i in expired.into_iter().rev() {
            let w = waiters.remove(i);
            let c = &customers[w.idx];
            let total = w.waited + (now - w.entry);
            wait_ticks += (now - w.entry) as u128;
            push_event(&mut events, now, c.id, Kind::Renege { waited: total });
            reneged += 1;
        }

        if cursor >= n && active.is_none() && waiters.is_empty() {
            break;
        }

        let mut next = None;
        let mut consider = |t: u64| {
            if t > now {
                next = Some(next.map_or(t, |x: u64| x.min(t)));
            }
        };
        if cursor < n {
            consider(customers[order[cursor]].arrival);
        }
        if let Some(a) = &active {
            consider(a.work_at);
            consider(a.finish);
        }
        for w in &waiters {
            if let Some(d) = w.deadline {
                consider(d);
            }
        }
        if let Some(t) = next {
            if let Some(a) = &active {
                busy_ticks += (t - now) as u128;
            }
            now = t;
        } else {
            break;
        }
    }

    let elapsed = events.last().map_or(0, |e| e.time);
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(busy_ticks * 10000, elapsed as u128).unwrap() as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}

fn a_dispatch_time(_events: &[Event], _id: u32, now: u64) -> u64 {
    now
}

fn total_wait_for(_events: &[Event], _id: u32) -> u64 {
    0
}

fn main() {}
