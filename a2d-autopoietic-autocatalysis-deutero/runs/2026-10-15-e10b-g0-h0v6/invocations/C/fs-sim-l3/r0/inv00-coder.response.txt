```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components: Vec<&str> = Vec::new();
        if !path.starts_with('/') {
            components.extend(self.cwd.split('/').filter(|s| !s.is_empty()));
        }

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                other => components.push(other),
            }
        }

        if components.is_empty() {
            Ok("/".to_string())
        } else {
            Ok(format!("/{}", components.join("/")))
        }
    }

    fn parent_and_name(path: &str) -> Option<(&str, &str)> {
        if path == "/" {
            return None;
        }
        match path.rfind('/') {
            Some(0) => Some(("/", &path[1..])),
            Some(i) => Some((&path[..i], &path[i + 1..])),
            None => None,
        }
    }

    fn check_directory(&self, path: &str) -> Result<(), Error> {
        let mut current = String::from("/");
        if path == "/" {
            return match self.entries.get("/") {
                Some(Entry::Dir) => Ok(()),
                Some(_) => Err(Error::NotDir("/".to_string())),
                None => Err(Error::Missing("/".to_string())),
            };
        }

        for component in path[1..].split('/') {
            current = if current == "/" {
                format!("/{component}")
            } else {
                format!("{current}/{component}")
            };
            match self.entries.get(&current) {
                None => return Err(Error::Missing(current)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(current)),
            }
        }
        Ok(())
    }

    fn check_parent(&self, path: &str) -> Result<(), Error> {
        let (parent, _) = Self::parent_and_name(path).expect("non-root path");
        self.check_directory(parent)
    }

    fn is_within(path: &str, directory: &str) -> bool {
        if directory == "/" {
            return path.starts_with('/');
        }
        path == directory
            || path
                .strip_prefix(directory)
                .is_some_and(|rest| rest.starts_with('/'))
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries
            .keys()
            .any(|candidate| candidate != path && Self::is_within(candidate, path))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Exists(path));
        }
        self.check_parent(&path)?;
        if self.entries.contains_key(&path) {
            return Err(Error::Exists(path));
        }
        self.entries.insert(path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::IsDir(path));
        }
        self.check_parent(&path)?;
        match self.entries.get_mut(&path) {
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            Some(Entry::File(bytes)) => {
                bytes.clear();
                bytes.extend_from_slice(data);
                Ok(())
            }
            None => {
                self.entries
                    .insert(path, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::IsDir(path));
        }
        self.check_parent(&path)?;
        match self.entries.get(&path) {
            None => Err(Error::Missing(path)),
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        self.check_directory(&path)?;
        let mut result = Vec::new();

        for (entry_path, entry) in &self.entries {
            if entry_path == "/" {
                continue;
            }
            if let Some((parent, name)) = Self::parent_and_name(entry_path) {
                if parent == path {
                    result.push((name.to_string(), entry.kind()));
                }
            }
        }

        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        match self.entries.get(&path) {
            None => {
                self.check_directory_parent_for_final(&path)?;
                Err(Error::Missing(path))
            }
            Some(Entry::File(_)) => {
                self.check_parent(&path)?;
                Err(Error::NotDir(path))
            }
            Some(Entry::Dir) => {
                self.check_directory(&path)?;
                self.cwd = path;
                Ok(())
            }
        }
    }

    fn check_directory_parent_for_final(&self, path: &str) -> Result<(), Error> {
        if path == "/" {
            return Err(Error::Missing("/".to_string()));
        }
        self.check_parent(path)
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        self.check_parent(&source)?;
        let source_entry = match self.entries.get(&source) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(source)),
        };

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        self.check_parent(&destination)?;

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(ref existing) = destination_entry {
            match (&source_entry, existing) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(destination)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(destination)),
                (Entry::Dir, Entry::Dir) if self.has_children(&destination) => {
                    return Err(Error::NotEmpty(destination));
                }
                _ => {}
            }
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(candidate, _)| {
                **candidate == source
                    || (matches!(source_entry, Entry::Dir)
                        && Self::is_within(candidate, &source))
            })
            .map(|(candidate, entry)| {
                let suffix = &candidate[source.len()..];
                (format!("{destination}{suffix}"), entry.clone())
            })
            .collect();

        if let Some(existing) = destination_entry {
            if matches!(existing, Entry::Dir) {
                self.entries.retain(|candidate, _| candidate != &destination);
            } else {
                self.entries.remove(&destination);
            }
        }

        self.entries.retain(|candidate, _| {
            *candidate != source
                && !(matches!(source_entry, Entry::Dir)
                    && Self::is_within(candidate, &source))
        });

        for (new_path, entry) in moved {
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }

        self.check_parent(&path)?;
        let entry = match self.entries.get(&path) {
            Some(entry) => entry,
            None => return Err(Error::Missing(path)),
        };

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &path) {
            return Err(Error::Busy(path));
        }

        let nonempty = matches!(entry, Entry::Dir) && self.has_children(&path);
        if nonempty && !recursive {
            return Err(Error::NotEmpty(path));
        }

        self.entries.retain(|candidate, _| {
            *candidate != path
                && !(recursive && nonempty && Self::is_within(candidate, &path))
        });
        Ok(())
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

fn main() {}
```
