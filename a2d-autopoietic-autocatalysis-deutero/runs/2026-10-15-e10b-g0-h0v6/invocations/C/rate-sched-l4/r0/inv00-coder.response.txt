#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
struct RuntimeJob {
    job: Job,
    dependencies: Vec<usize>,
    state: State,
    suspended_remaining: Option<u64>,
    started_at: u64,
    ready_at: u64,
}

pub struct Pool {
    jobs: Vec<RuntimeJob>,
    by_id: std::collections::HashMap<String, usize>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    has_run: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut by_id = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job.tag.as_ref().is_some_and(|tag| !valid_token(tag, 1, 8)) {
                return Err(Error::BadTag(i));
            }
            if by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            by_id.insert(job.id.clone(), i);
        }

        let mut resolved = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut deps = Vec::with_capacity(job.dependencies.len());
            let mut seen = std::collections::HashSet::new();
            for (j, name) in job.dependencies.iter().enumerate() {
                if !seen.insert(name.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match by_id.get(name) {
                    Some(&index) => deps.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            resolved.push(deps);
        }

        fn visit(
            node: usize,
            deps: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[node] == 1 {
                return false;
            }
            if marks[node] == 2 {
                return true;
            }
            marks[node] = 1;
            for &dep in &deps[node] {
                if !visit(dep, deps, marks) {
                    return false;
                }
            }
            marks[node] = 2;
            true
        }

        let mut marks = vec![0u8; jobs.len()];
        for i in 0..jobs.len() {
            if !visit(i, &resolved, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let runtime = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| RuntimeJob {
                ready_at: job.release_at,
                job,
                dependencies: resolved[i].clone(),
                state: State::Waiting {
                    ready_at: 0,
                    attempt: 0,
                },
                suspended_remaining: None,
                started_at: 0,
            })
            .map(|mut r| {
                r.state = State::Waiting {
                    ready_at: r.ready_at,
                    attempt: 0,
                };
                r
            })
            .collect();

        Ok(Self {
            jobs: runtime,
            by_id,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
        })
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if self.has_run && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut lines = Vec::new();
        let mut time = if self.has_run {
            self.horizon.checked_add(1).unwrap_or(u64::MAX)
        } else {
            0
        };

        if !self.has_run {
            self.process_timestamp(0, &mut lines);
        }

        while time <= until {
            let Some(next) = self.next_event_time(time, until) else {
                break;
            };
            time = next;
            self.process_timestamp(time, &mut lines);
            if time == u64::MAX {
                break;
            }
            time += 1;
        }

        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let &i = self.by_id.get(id).ok_or(Error::Unknown)?;
        Ok(self.jobs[i].state.clone())
    }

    fn next_event_time(&self, after: u64, until: u64) -> Option<u64> {
        let mut next = None;
        let mut consider = |t: u64| {
            if t >= after && t <= until && next.map_or(true, |n| t < n) {
                next = Some(t);
            }
        };
        for r in &self.jobs {
            match r.state {
                State::Waiting { ready_at, .. } => consider(ready_at),
                State::Running { finish_at, .. } => consider(finish_at),
                _ => {}
            }
        }
        next
    }

    fn process_timestamp(&mut self, t: u64, lines: &mut Vec<String>) {
        let mut completions: Vec<usize> = self
            .jobs
            .iter()
            .enumerate()
            .filter_map(|(i, r)| match r.state {
                State::Running { finish_at, .. } if finish_at == t => Some(i),
                _ => None,
            })
            .collect();
        completions.sort_by(|&a, &b| self.jobs[a].job.id.cmp(&self.jobs[b].job.id));

        for i in completions {
            let attempt = match self.jobs[i].state {
                State::Running { attempt, .. } => attempt,
                _ => continue,
            };
            let id = self.jobs[i].job.id.clone();
            if attempt > self.jobs[i].job.failures {
                lines.push(format!("{t} finish {id} {attempt} ok"));
                self.jobs[i].state = State::Succeeded;
                self.jobs[i].suspended_remaining = None;
            } else {
                lines.push(format!("{t} finish {id} {attempt} fail"));
                self.jobs[i].suspended_remaining = None;
                if attempt >= self.jobs[i].job.max_attempts {
                    self.jobs[i].state = State::Failed;
                } else {
                    let shift = u32::from(attempt.saturating_sub(1));
                    let delay = self
                        .retry_base
                        .checked_mul(1u64.checked_shl(shift).unwrap_or(u64::MAX))
                        .unwrap_or(u64::MAX)
                        .min(self.retry_cap);
                    let ready = t.saturating_add(delay);
                    self.jobs[i].ready_at = ready;
                    self.jobs[i].state = State::Waiting {
                        ready_at: ready,
                        attempt,
                    };
                    lines.push(format!("{t} retry {id} {attempt} {ready}"));
                }
            }
        }

        loop {
            let mut blocked = Vec::new();
            for i in 0..self.jobs.len() {
                if !matches!(self.jobs[i].state, State::Waiting { .. }) {
                    continue;
                }
                if self.jobs[i].dependencies.iter().any(|&d| {
                    matches!(self.jobs[d].state, State::Failed | State::Blocked)
                }) {
                    blocked.push(i);
                }
            }
            if blocked.is_empty() {
                break;
            }
            for &i in &blocked {
                self.jobs[i].state = State::Blocked;
            }
            blocked.sort_by(|&a, &b| self.jobs[a].job.id.cmp(&self.jobs[b].job.id));
            for i in blocked {
                let dependency = self.jobs[i]
                    .dependencies
                    .iter()
                    .filter(|&&d| matches!(self.jobs[d].state, State::Failed | State::Blocked))
                    .map(|&d| self.jobs[d].job.id.as_str())
                    .min()
                    .unwrap_or("");
                lines.push(format!(
                    "{t} skip {} {dependency}",
                    self.jobs[i].job.id
                ));
            }
        }

        loop {
            let mut candidates: Vec<usize> = self
                .jobs
                .iter()
                .enumerate()
                .filter_map(|(i, r)| match r.state {
                    State::Waiting { ready_at, .. }
                        if ready_at <= t
                            && r.dependencies.iter().all(|&d| {
                                matches!(self.jobs[d].state, State::Succeeded)
                            }) =>
                    {
                        Some(i)
                    }
                    _ => None,
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, t);
                let pb = self.effective_priority(b, t);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut selected = None;
            for candidate in candidates {
                if let Some(victim) = self.select_victim(candidate) {
                    selected = Some((candidate, Some(victim)));
                    break;
                }
                if self.can_start_without_preemption(candidate) {
                    selected = Some((candidate, None));
                    break;
                }
            }

            let Some((candidate, victim)) = selected else {
                break;
            };

            if let Some(victim) = victim {
                let remaining = match self.jobs[victim].state {
                    State::Running { finish_at, .. } => finish_at.saturating_sub(t),
                    _ => 0,
                };
                let victim_id = self.jobs[victim].job.id.clone();
                let candidate_id = self.jobs[candidate].job.id.clone();
                lines.push(format!("{t} preempt {victim_id} by {candidate_id}"));
                let attempt = match self.jobs[victim].state {
                    State::Running { attempt, .. } => attempt,
                    _ => 0,
                };
                self.jobs[victim].ready_at = t;
                self.jobs[victim].suspended_remaining = Some(remaining);
                self.jobs[victim].state = State::Waiting {
                    ready_at: t,
                    attempt,
                };
            }

            let (attempt, remaining) = match self.jobs[candidate].state {
                State::Waiting { attempt, .. } => {
                    if let Some(saved) = self.jobs[candidate].suspended_remaining.take() {
                        (attempt, saved)
                    } else {
                        let next_attempt = attempt.saturating_add(1);
                        (next_attempt, self.jobs[candidate].job.duration)
                    }
                }
                _ => continue,
            };
            let finish = t.saturating_add(remaining);
            let id = self.jobs[candidate].job.id.clone();
            lines.push(format!("{t} start {id} {attempt} {remaining}"));
            self.jobs[candidate].started_at = t;
            self.jobs[candidate].state = State::Running {
                attempt,
                finish_at: finish,
            };
        }
    }

    fn effective_priority(&self, i: usize, t: u64) -> u8 {
        let base = self.jobs[i].job.priority;
        if self.aging_ms == 0 {
            return base;
        }
        let ready = match self.jobs[i].state {
            State::Waiting { ready_at, .. } => ready_at,
            _ => self.jobs[i].ready_at,
        };
        let age = t.saturating_sub(ready) / self.aging_ms;
        (u64::from(base).saturating_add(age).min(9)) as u8
    }

    fn can_start_without_preemption(&self, candidate: usize) -> bool {
        let running = self
            .jobs
            .iter()
            .filter(|r| matches!(r.state, State::Running { .. }))
            .count();
        if running >= self.capacity {
            return false;
        }
        self.tag_is_free(candidate, None)
    }

    fn select_victim(&self, candidate: usize) -> Option<usize> {
        let running: Vec<usize> = self
            .jobs
            .iter()
            .enumerate()
            .filter_map(|(i, r)| {
                if matches!(r.state, State::Running { .. }) {
                    Some(i)
                } else {
                    None
                }
            })
            .collect();

        let candidate_tag = self.jobs[candidate].job.tag.as_deref();
        let slot_needed = running.len() >= self.capacity;
        let tag_needed = candidate_tag.is_some_and(|tag| {
            running
                .iter()
                .any(|&i| self.jobs[i].job.tag.as_deref() == Some(tag))
        });

        if !slot_needed && !tag_needed {
            return None;
        }

        let mut victims: Vec<usize> = running
            .into_iter()
            .filter(|&v| {
                self.jobs[candidate].job.priority
                    >= self.jobs[v].job.priority.saturating_add(2)
            })
            .filter(|&v| {
                let remaining_after = self
                    .jobs
                    .iter()
                    .enumerate()
                    .filter(|(i, r)| {
                        *i != v && matches!(r.state, State::Running { .. })
                    })
                    .count();
                remaining_after < self.capacity
                    && self.tag_is_free(candidate, Some(v))
            })
            .collect();

        victims.sort_by(|&a, &b| {
            self.jobs[a]
                .job
                .priority
                .cmp(&self.jobs[b].job.priority)
                .then_with(|| self.jobs[b].started_at.cmp(&self.jobs[a].started_at))
                .then_with(|| self.jobs[b].job.id.cmp(&self.jobs[a].job.id))
        });
        victims.into_iter().next()
    }

    fn tag_is_free(&self, candidate: usize, excluding: Option<usize>) -> bool {
        let Some(tag) = self.jobs[candidate].job.tag.as_deref() else {
            return true;
        };
        !self.jobs.iter().enumerate().any(|(i, r)| {
            Some(i) != excluding
                && matches!(r.state, State::Running { .. })
                && r.job.tag.as_deref() == Some(tag)
        })
    }
}

fn valid_token(s: &str, min: usize, max: usize) -> bool {
    let b = s.as_bytes();
    (min..=max).contains(&b.len())
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn main() {}
