#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    a: u32,
    b: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Part {
    pub crc32: u32,
    pub len: u64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CombineError {
    LengthOverflow { part: usize },
    EmptyCrc { part: usize },

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VerifyError {
    Span(SpanError),
    Shape,
    Hex { offset: usize },
    Length,
    Mismatch {
        field: usize,
        expected: u32,
        actual: u32,
    },
}

const ADLER_MOD: u32 = 65_521;

fn crc32_raw_update(mut crc: u32, data: &[u8]) -> u32 {
    for &byte in data {
        crc ^= byte as u32;
        for _ in 0..8 {
            let low = crc & 1;
            crc >>= 1;
            if low != 0 {
                crc ^= 0xedb8_8320;
            }
        }
    }
    crc
}

pub fn crc32(data: &[u8]) -> u32 {
    crc32_raw_update(0xffff_ffff, data) ^ 0xffff_ffff
}

fn adler_update(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *a = (*a + byte as u32) % ADLER_MOD;
        *b = (*b + *a) % ADLER_MOD;
    }
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1;
    let mut b = 0;
    adler_update(&mut a, &mut b, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

fn crc32_combine_validated(left: u32, right: u32, mut right_len: u64) -> u32 {
    let mut result = left;
    let mut operator = [0u32; 32];

    for bit in 0..32 {
        let mut value = 1u32 << bit;
        for _ in 0..8 {
            let low = value & 1;
            value >>= 1;
            if low != 0 {
                value ^= 0xedb8_8320;
            }
        }
        operator[bit] = value;
    }

    while right_len != 0 {
        if right_len & 1 != 0 {
            result = gf2_matrix_times(&operator, result);
        }
        right_len >>= 1;
        if right_len != 0 {
            operator = gf2_matrix_square(&operator);
        }
    }
    result ^ right
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(matrix: &[u32; 32]) -> [u32; 32] {
    let mut square = [0; 32];
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
    square
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        left
    } else {
        crc32_combine_validated(left, right, right_len)
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for span in spans {
        let selected = &data[span.start..span.end];
        crc = crc32_raw_update(crc, selected);
        adler_update(&mut a, &mut b, selected);
    }
    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc = crc32_raw_update(self.crc, data);
        adler_update(&mut self.a, &mut self.b, data);
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        let mut crc = self.crc;
        let mut a = self.a;
        let mut b = self.b;
        for span in spans {
            let selected = &data[span.start..span.end];
            crc = crc32_raw_update(crc, selected);
            adler_update(&mut a, &mut b, selected);
        }
        self.crc = crc;
        self.a = a;
        self.b = b;
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19
            || bytes[0] != b'c'
            || bytes[9] != b':'
            || bytes[10] != b'a'
        {
            return Err(SnapshotError::Shape);
        }

        for offset in (1..9).chain(11..19) {
            if hex_value(bytes[offset]).is_none() {
                return Err(SnapshotError::Hex { offset });
            }
        }

        let crc = parse_hex_u32(&bytes[1..9]);
        let adler = parse_hex_u32(&bytes[11..19]);
        Self::from_seed(crc, adler)
    }
}

fn hex_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'f' => Some((byte - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse_hex_u32(bytes: &[u8]) -> u32 {
    bytes.iter().fold(0, |value, &byte| {
        (value << 4) | hex_value(byte).unwrap()
    })
}

fn fletcher16_update(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *a = (*a + byte as u32) % 255;
        *b = (*b + *a) % 255;
    }
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0;
    let mut b = 0;
    fletcher16_update(&mut a, &mut b, data);
    ((b as u16) << 8) | a as u16
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
struct Fletcher32State {
    a: u32,
    b: u32,
    pending: Option<u8>,
}

impl Fletcher32State {
    fn new() -> Self {
        Self {
            a: 0,
            b: 0,
            pending: None,
        }
    }

    fn update(&mut self, data: &[u8]) {
        let mut index = 0;
        if let Some(high) = self.pending.take() {
            if let Some(&low) = data.first() {
                self.word(((high as u16) << 8) | low as u16);
                index = 1;
            } else {
                self.pending = Some(high);
                return;
            }
        }
        while index + 1 < data.len() {
            self.word(((data[index] as u16) << 8) | data[index + 1] as u16);
            index += 2;
        }
        if index < data.len() {
            self.pending = Some(data[index]);
        }
    }

    fn word(&mut self, word: u16) {
        self.a = (self.a + word as u32) % 65_535;
        self.b = (self.b + self.a) % 65_535;
    }

    fn finish(mut self) -> u32 {
        if let Some(high) = self.pending.take() {
            self.word((high as u16) << 8);
        }
        (self.b << 16) | self.a
    }
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut state = Fletcher32State::new();
    state.update(data);
    state.finish()
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = Fletcher32State::new();
    for span in spans {
        state.update(&data[span.start..span.end]);
    }
    Ok(state.finish())
}

fn crc16_update(mut crc: u16, data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = crc & 0x8000;
                    crc = crc.wrapping_shl(1);
                    if top != 0 {
                        crc ^= 0x1021;
                    }
                }
            }
            crc
        }
        Crc16::X25 => {
            for &byte in data {
                crc ^= byte as u16;
                for _ in 0..8 {
                    let low = crc & 1;
                    crc >>= 1;
                    if low != 0 {
                        crc ^= 0x8408;
                    }
                }
            }
            crc
        }
    }
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem => crc16_update(0, data, variant),
        Crc16::False | Crc16::X25 => crc16_update(0xffff, data, variant),
    }
    .pipe_x25_finalize(variant)
}

trait Crc16Finalize {
    fn pipe_x25_finalize(self, variant: Crc16) -> u16;
}

impl Crc16Finalize for u16 {
    fn pipe_x25_finalize(self, variant: Crc16) -> u16 {
        if variant == Crc16::X25 {
            self ^ 0xffff
        } else {
            self
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc = 0xffff_ffff;
    let mut adler_a = 1;
    let mut adler_b = 0;
    let mut f16_a = 0;
    let mut f16_b = 0;
    let mut f32 = Fletcher32State::new();
    let mut crc16_state = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        crc = crc32_raw_update(crc, selected);
        adler_update(&mut adler_a, &mut adler_b, selected);
        fletcher16_update(&mut f16_a, &mut f16_b, selected);
        f32.update(selected);
        crc16_state = crc16_update(crc16_state, selected, variant);
    }

    if variant == Crc16::X25 {
        crc16_state ^= 0xffff;
    }

    Ok(Multi {
        base: Hashes {
            crc32: crc ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: f32.finish(),
        crc16: crc16_state,
    })
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut combined = 0;
    for part in parts {
        combined = crc32_combine(combined, part.crc32, part.len);
    }
    Ok(Part {
        crc32: combined,
        len: total,
    })
}

fn selected_summary(data: &[u8], spans: &[Span]) -> (u64, Hashes) {
    let mut length = 0u64;
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for span in spans {
        let selected = &data[span.start..span.end];
        length += selected.len() as u64;
        crc = crc32_raw_update(crc, selected);
        adler_update(&mut a, &mut b, selected);
    }
    (
        length,
        Hashes {
            crc32: crc ^ 0xffff_ffff,
            adler32: (b << 16) | a,
        },
    )
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let (length, hashes) = selected_summary(data, spans);
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        length, hashes.crc32, hashes.adler32
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for offset in (4..20).chain(25..33).chain(40..48) {
        if hex_value(bytes[offset]).is_none() {
            return Err(VerifyError::Hex { offset });
        }
    }

    let length = parse_hex_u64(&bytes[4..20]);
    let expected_crc = parse_hex_u32(&bytes[25..33]);
    let expected_adler = parse_hex_u32(&bytes[40..48]);
    let (actual_length, actual) = selected_summary(data, spans);
    if length != actual_length {
        return Err(VerifyError::Length);
    }
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

fn parse_hex_u64(bytes: &[u8]) -> u64 {
    bytes.iter().fold(0, |value, &byte| {
        (value << 4) | hex_value(byte).unwrap() as u64
    })
}
