use std::collections::HashSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let separator = spaces[0];
        if separator == 0 || separator + 1 == bytes.len() {
            return Err(CronError::Layout(separator));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];

        parse_field(text, 0, separator, 60, &mut minutes)?;
        parse_field(text, separator + 1, bytes.len(), 24, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    text: &str,
    field_start: usize,
    field_end: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut item_start = field_start;

    loop {
        let item_end = bytes[item_start..field_end]
            .iter()
            .position(|&b| b == b',')
            .map_or(field_end, |relative| item_start + relative);

        if item_start == item_end {
            return Err(CronError::Syntax(item_start));
        }

        parse_item(text, item_start, item_end, limit, selected)?;

        if item_end == field_end {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(())
}

fn parse_item(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();

    let slash = bytes[start..end]
        .iter()
        .position(|&b| b == b'/')
        .map(|relative| start + relative);

    let (base_end, step_start) = if let Some(slash_pos) = slash {
        if let Some(second_relative) = bytes[slash_pos + 1..end]
            .iter()
            .position(|&b| b == b'/')
        {
            return Err(CronError::Syntax(slash_pos + 1 + second_relative));
        }
        (slash_pos, Some(slash_pos + 1))
    } else {
        (end, None)
    };

    let base_start = start;
    let base_is_star = base_end == base_start + 1 && bytes[base_start] == b'*';

    let dash = bytes[base_start..base_end]
        .iter()
        .position(|&b| b == b'-')
        .map(|relative| base_start + relative);

    if let Some(slash_pos) = slash {
        if !base_is_star && dash.is_none() {
            return Err(CronError::Syntax(slash_pos));
        }
    }

    let (range_start, range_end, dash_position) = if base_is_star {
        (0usize, limit - 1, None)
    } else if let Some(dash_pos) = dash {
        let left = parse_number(text, base_start, dash_pos, limit)?;
        let right = parse_number(text, dash_pos + 1, base_end, limit)?;
        if left > right {
            return Err(CronError::Descending(dash_pos));
        }
        (left, right, Some(dash_pos))
    } else {
        let value = parse_number(text, base_start, base_end, limit)?;
        (value, value, None)
    };

    let step = if let Some(step_start) = step_start {
        parse_step(text, step_start, end, limit)?
    } else {
        1
    };

    let _ = dash_position;
    let mut value = range_start;
    while value <= range_end {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }

    Ok(())
}

fn parse_number(text: &str, start: usize, end: usize, limit: usize) -> Result<usize, CronError> {
    let bytes = text.as_bytes();

    if start == end {
        return Err(CronError::Syntax(start));
    }

    for index in start..end {
        if !bytes[index].is_ascii_digit() {
            return Err(CronError::Syntax(index));
        }
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for index in start..end {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((bytes[index] - b'0') as usize))
            .unwrap_or(usize::MAX);
    }

    if value >= limit {
        return Err(CronError::Bounds(start));
    }

    Ok(value)
}

fn parse_step(text: &str, start: usize, end: usize, limit: usize) -> Result<usize, CronError> {
    let bytes = text.as_bytes();

    if start == end {
        return Err(CronError::Syntax(start));
    }

    for index in start..end {
        if !bytes[index].is_ascii_digit() {
            return Err(CronError::Syntax(index));
        }
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for index in start..end {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((bytes[index] - b'0') as usize))
            .unwrap_or(usize::MAX);
    }

    let max_step = if limit == 60 { 60 } else { 24 };
    if value == 0 || value > max_step {
        return Err(CronError::Step(start));
    }

    Ok(value)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    ids: HashSet<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            ids: HashSet::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.contains(id) {
            return Err(Error::Duplicate);
        }

        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let id = id.to_owned();
        let order = self.jobs.len();

        self.ids.insert(id.clone());
        self.jobs.push(Job {
            id,
            cron,
            priority,
            policy,
            order,
        });

        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let first_minute = match self.last_poll {
            Some(last) => last / 60_000 + 1,
            None => 0,
        };
        let last_minute = now / 60_000;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        if first_minute <= last_minute {
            for job in &self.jobs {
                match job.policy {
                    Policy::All => {
                        for minute in first_minute..=last_minute {
                            if job.cron.matches(minute) {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: minute * 60_000,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                        }
                    }
                    Policy::Once => {
                        if let Some(minute) = (first_minute..=last_minute)
                            .rev()
                            .find(|&minute| job.cron.matches(minute))
                        {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                    Policy::Skip => {
                        if now % 60_000 == 0 {
                            let minute = now / 60_000;
                            if minute >= first_minute && job.cron.matches(minute) {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: now,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|item| item.0).collect())
    }
}

fn main() {}
