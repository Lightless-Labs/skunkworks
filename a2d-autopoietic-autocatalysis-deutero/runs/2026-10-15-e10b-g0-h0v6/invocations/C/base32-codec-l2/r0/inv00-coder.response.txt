#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn value_of(byte: u8, alphabet: &[u8; 32]) -> Option<u8> {
    alphabet.iter().position(|&c| c == byte).map(|v| v as u8)
}

pub fn encode(input: &[u8], alphabet: Alphabet, padding: Padding) -> String {
    let chars = alphabet_bytes(alphabet);
    let mut output = String::new();
    let mut accumulator: u32 = 0;
    let mut bit_count = 0;

    for &byte in input {
        accumulator = (accumulator << 8) | u32::from(byte);
        bit_count += 8;

        while bit_count >= 5 {
            bit_count -= 5;
            let value = ((accumulator >> bit_count) & 0x1f) as usize;
            output.push(chars[value] as char);
        }
    }

    if bit_count > 0 {
        let value = ((accumulator << (5 - bit_count)) & 0x1f) as usize;
        output.push(chars[value] as char);
    }

    if padding == Padding::Required {
        while output.len() % 8 != 0 {
            output.push('=');
        }
    }

    output
}

pub fn decode(
    input: &str,
    alphabet: Alphabet,
    padding: Padding,
) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let chars = alphabet_bytes(alphabet);

    // Lexical validation.
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && value_of(byte, chars).is_none() {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let first_padding = bytes.iter().position(|&byte| byte == b'=');
    let p = first_padding.unwrap_or(bytes.len());

    // Padding placement and mode.
    match padding {
        Padding::Omitted if first_padding.is_some() => {
            return Err(DecodeError::Padding { offset: p });
        }
        Padding::Required => {
            if let Some(index) = first_padding {
                if let Some(offset) = bytes[index..].iter().position(|&byte| byte != b'=') {
                    return Err(DecodeError::Padding {
                        offset: index + offset,
                    });
                }
            }
        }
        Padding::Omitted => {}
    }

    let data_len = p;
    let residue = data_len % 8;
    let expected_padding = match residue {
        0 => 0,
        2 => 6,
        4 => 4,
        5 => 3,
        7 => 1,
        _ => return Err(DecodeError::InvalidLength { offset: bytes.len() }),
    };

    // Required padding must have the prescribed shape and total length.
    if padding == Padding::Required {
        let actual_padding = bytes.len() - data_len;
        if actual_padding != expected_padding || bytes.len() % 8 != 0 {
            return Err(DecodeError::Padding { offset: data_len });
        }
    }

    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };

    if unused_bits > 0 {
        let last = value_of(bytes[data_len - 1], chars).unwrap();
        if (last & ((1 << unused_bits) - 1)) != 0 {
            return Err(DecodeError::TrailingBits {
                offset: data_len - 1,
            });
        }
    }

    let mut output = Vec::with_capacity(data_len * 5 / 8);
    let mut accumulator: u32 = 0;
    let mut bit_count = 0;

    for &byte in &bytes[..data_len] {
        let value = u32::from(value_of(byte, chars).unwrap());
        accumulator = (accumulator << 5) | value;
        bit_count += 5;

        while bit_count >= 8 {
            bit_count -= 8;
            output.push(((accumulator >> bit_count) & 0xff) as u8);
        }
    }

    Ok(output)
}

fn main() {}
