#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
    Duplicate { first: usize, second: usize },
    ZeroArea,
    AdjacentOverlap { vertex: usize },
    Intersection { edge_a: usize, edge_b: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Validated {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub measures: Measures,
}

const COORD_LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<(Vec<Point>, Vec<usize>), GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (i, &p) in points.iter().enumerate() {
        if !in_range(p) {
            return Err(GeomError::OutOfRange { index: i });
        }
    }

    let mut len = points.len();
    if len >= 2 && points[len - 1] == points[0] {
        len -= 1;
    }
    if len < 3 {
        return Err(GeomError::TooFewVertices { len });
    }
    Ok((points[..len].to_vec(), (0..len).collect()))
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn signed_area2(points: &[Point]) -> i128 {
    let n = points.len();
    let mut area = 0i128;
    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        area += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    area
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    (dx * dx + dy * dy) as u128
}

fn rounded_sqrt(n: u128) -> u128 {
    let mut lo = 0u128;
    let mut hi = 1u128;
    while hi <= n / hi {
        hi *= 2;
    }
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid <= n / mid {
            lo = mid;
        } else {
            hi = mid;
        }
    }
    let lower_square = lo * lo;
    let upper = lo + 1;
    let upper_square = upper * upper;
    if n - lower_square >= upper_square - n {
        upper
    } else {
        lo
    }
}

fn edge_milli(a: Point, b: Point) -> u64 {
    rounded_sqrt(squared_distance(a, b) * 1_000_000) as u64
}

fn measures_for(points: &[Point]) -> Measures {
    let area = signed_area2(points);
    let n = points.len();
    let mut edge_lengths = Vec::with_capacity(n);
    let mut perimeter = 0u64;
    let mut longest: Option<usize> = None;

    for i in 0..n {
        let length = edge_milli(points[i], points[(i + 1) % n]);
        edge_lengths.push(length);
        perimeter += length;

        match longest {
            None => longest = Some(i),
            Some(old) => {
                let old_len = edge_lengths[old];
                if length > old_len {
                    longest = Some(i);
                } else if length == old_len {
                    let pair = |idx: usize| {
                        let a = points[idx];
                        let b = points[(idx + 1) % n];
                        if a <= b { (a, b) } else { (b, a) }
                    };
                    if pair(i) < pair(old) || (pair(i) == pair(old) && i < old) {
                        longest = Some(i);
                    }
                }
            }
        }
    }

    Measures {
        signed_area2: area,
        area2: area.unsigned_abs(),
        orientation: if area > 0 {
            Orientation::CounterClockwise
        } else if area < 0 {
            Orientation::Clockwise
        } else {
            Orientation::Degenerate
        },
        edge_milli: edge_lengths,
        perimeter_milli: perimeter,
        longest_edge: longest,
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let (vertices, _) = prepare_polygon(points)?;
    Ok(measures_for(&vertices))
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let (vertices, _) = prepare_polygon(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (i, &q) in queries.iter().enumerate() {
        if !in_range(q) {
            return Err(GeomError::QueryOutOfRange { index: i });
        }
    }

    let n = vertices.len();
    let mut result = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        for i in 0..n {
            if on_segment(vertices[i], vertices[(i + 1) % n], q) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            result.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..n {
            let a = vertices[i];
            let b = vertices[(i + 1) % n];
            if (a.y > q.y) != (b.y > q.y) {
                let c = cross(a, b, q);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                    inside = !inside;
                }
            }
        }
        result.push(if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(result)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (i, &p) in points.iter().enumerate() {
        if !in_range(p) {
            return Err(GeomError::OutOfRange { index: i });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (i, &p) in points.iter().enumerate() {
        if !unique.iter().any(|(q, _)| *q == p) {
            unique.push((p, i));
        }
    }
    unique.sort_by_key(|(p, _)| *p);

    let m = unique.len();
    let output: Vec<(Point, usize)> = if m <= 1 {
        unique.clone()
    } else {
        let all_collinear =
            (2..m).all(|i| cross(unique[0].0, unique[m - 1].0, unique[i].0) == 0);
        if all_collinear {
            match mode {
                Collinear::Include => unique.clone(),
                Collinear::Exclude => vec![unique[0], unique[m - 1]],
            }
        } else {
            let mut lower: Vec<(Point, usize)> = Vec::new();
            for item in &unique {
                while lower.len() >= 2 {
                    let n = lower.len();
                    let turn = cross(lower[n - 2].0, lower[n - 1].0, item.0);
                    if turn > 0 || (mode == Collinear::Include && turn == 0) {
                        break;
                    }
                    lower.pop();
                }
                lower.push(*item);
            }
            let mut upper: Vec<(Point, usize)> = Vec::new();
            for item in unique.iter().rev() {
                while upper.len() >= 2 {
                    let n = upper.len();
                    let turn = cross(upper[n - 2].0, upper[n - 1].0, item.0);
                    if turn > 0 || (mode == Collinear::Include && turn == 0) {
                        break;
                    }
                    upper.pop();
                }
                upper.push(*item);
            }
            lower.pop();
            upper.pop();
            lower.extend(upper);
            lower
        }
    };

    let vertices: Vec<Point> = output.iter().map(|x| x.0).collect();
    let source_indices: Vec<usize> = output.iter().map(|x| x.1).collect();
    let area = if vertices.len() >= 3 {
        signed_area2(&vertices).unsigned_abs()
    } else {
        0
    };
    let perimeter = if vertices.is_empty() {
        0
    } else {
        let n = vertices.len();
        (0..n)
            .map(|i| edge_milli(vertices[i], vertices[(i + 1) % n]))
            .sum()
    };

    Ok(Hull {
        vertices,
        source_indices,
        area2: area,
        perimeter_milli: perimeter,
    })
}

fn segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool {
    let ab_c = cross(a, b, c);
    let ab_d = cross(a, b, d);
    let cd_a = cross(c, d, a);
    let cd_b = cross(c, d, b);

    if ab_c == 0 && on_segment(a, b, c) {
        return true;
    }
    if ab_d == 0 && on_segment(a, b, d) {
        return true;
    }
    if cd_a == 0 && on_segment(c, d, a) {
        return true;
    }
    if cd_b == 0 && on_segment(c, d, b) {
        return true;
    }
    (ab_c > 0) != (ab_d > 0) && (cd_a > 0) != (cd_b > 0)
}

pub fn validate(points: &[Point]) -> Result<Validated, GeomError> {
    let (vertices, _) = prepare_polygon(points)?;
    let n = vertices.len();

    for i in 0..n {
        for j in (i + 1)..n {
            if vertices[i] == vertices[j] {
                return Err(GeomError::Duplicate {
                    first: i,
                    second: j,
                });
            }
        }
    }

    let area = signed_area2(&vertices);
    if area == 0 {
        return Err(GeomError::ZeroArea);
    }

    for i in 0..n {
        let prev = vertices[(i + n - 1) % n];
        let curr = vertices[i];
        let next = vertices[(i + 1) % n];
        let dot = (prev.x as i128 - curr.x as i128) * (next.x as i128 - curr.x as i128)
            + (prev.y as i128 - curr.y as i128) * (next.y as i128 - curr.y as i128);
        if cross(prev, curr, next) == 0 && dot > 0 {
            return Err(GeomError::AdjacentOverlap { vertex: i });
        }
    }

    for i in 0..n {
        for j in (i + 1)..n {
            if j == i + 1 || (i == 0 && j == n - 1) {
                continue;
            }
            if segments_intersect(
                vertices[i],
                vertices[(i + 1) % n],
                vertices[j],
                vertices[(j + 1) % n],
            ) {
                return Err(GeomError::Intersection {
                    edge_a: i,
                    edge_b: j,
                });
            }
        }
    }

    let mut min_index = 0;
    for i in 1..n {
        if vertices[i] < vertices[min_index] {
            min_index = i;
        }
    }

    let mut canonical_vertices = Vec::with_capacity(n);
    let mut source_indices = Vec::with_capacity(n);
    if area > 0 {
        for k in 0..n {
            let idx = (min_index + k) % n;
            canonical_vertices.push(vertices[idx]);
            source_indices.push(idx);
        }
    } else {
        for k in 0..n {
            let idx = (min_index + n - k) % n;
            canonical_vertices.push(vertices[idx]);
            source_indices.push(idx);
        }
    }

    let measures = measures_for(&canonical_vertices);
    Ok(Validated {
        vertices: canonical_vertices,
        source_indices,
        measures,
    })
}

fn main() {}
