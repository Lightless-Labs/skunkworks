#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
struct Variable<'a> {
    name: &'a str,
}

struct Expression<'a> {
    operator: Option<char>,
    variables: Vec<Variable<'a>>,
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let content_start = open + 1;
    let content = &template[content_start..close];

    let operator = content
        .chars()
        .next()
        .filter(|ch| matches!(ch, '+' | '#' | '.' | '/' | ';' | '?' | '&'));

    let vars_start = content_start + operator.map_or(0, char::len_utf8);
    if vars_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut part_start = vars_start;

    loop {
        let part_end = template[part_start..close]
            .find(',')
            .map_or(close, |offset| part_start + offset);

        if part_start == part_end {
            return Err(Error::EmptyVariable { pos: part_start });
        }

        let name = &template[part_start..part_end];
        let bytes = name.as_bytes();

        if !bytes[0].is_ascii_alphabetic() && bytes[0] != b'_' {
            return Err(Error::InvalidName { pos: part_start });
        }
        for (offset, byte) in bytes.iter().enumerate().skip(1) {
            if !byte.is_ascii_alphanumeric() && *byte != b'_' {
                return Err(Error::InvalidName {
                    pos: part_start + offset,
                });
            }
        }

        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: part_start });
        }

        seen.push(name);
        variables.push(Variable { name });

        if part_end == close {
            break;
        }
        part_start = part_end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn encode_value(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%'
            && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            result.push('%');
            result.push((bytes[i + 1] as char).to_ascii_uppercase());
            result.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$'
                    | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';'
                    | b'='
            );

        if unreserved || reserved_byte {
            result.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 0x0f) as usize] as char);
        }

        i += 1;
    }

    result
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator.unwrap_or('\0');
    let reserved = matches!(operator, '+' | '#');
    let mut fields = Vec::new();

    for variable in &expression.variables {
        let value = vars
            .iter()
            .rev()
            .find(|(name, _)| *name == variable.name)
            .map(|(_, value)| *value);

        if let Some(value) = value {
            let encoded = encode_value(value, reserved);
            if matches!(operator, ';' | '?' | '&') {
                if operator == ';' && encoded.is_empty() {
                    fields.push(variable.name.to_owned());
                } else {
                    fields.push(format!("{}={}", variable.name, encoded));
                }
            } else {
                fields.push(encoded);
            }
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match operator {
        '#' => ("#", ","),
        '.' => (".", "."),
        '/' => ("/", "/"),
        ';' => (";", ";"),
        '?' => ("?", "&"),
        '&' => ("&", "&"),
        _ => ("", ","),
    };

    let mut result = String::from(prefix);
    result.push_str(&fields.join(separator));
    result
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < template.len() {
        match template.as_bytes()[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let close = match template[open + 1..].find('}') {
                    Some(offset) => open + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = template[open + 1..close].find('{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                expressions.push((open, close, parse_expression(template, open, close)?));
                cursor = close + 1;
            }
            _ => {
                let ch = template[cursor..].chars().next().unwrap();
                cursor += ch.len_utf8();
            }
        }
    }

    let mut result = String::new();
    let mut literal_start = 0;

    for (open, close, expression) in &expressions {
        result.push_str(&template[literal_start..*open]);
        result.push_str(&expand_expression(expression, vars));
        literal_start = close + 1;
    }

    result.push_str(&template[literal_start..]);
    Ok(result)
}

fn main() {}
