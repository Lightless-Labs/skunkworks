#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Message {
    pub fields: Vec<Field>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Field {
    UInt { tag: u32, value: u64 },
    Bytes { tag: u32, data: Vec<u8> },
    Nested { tag: u32, message: Message },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    StartOutOfBounds { offset: usize },
    Truncated { offset: usize, start: usize },
    Overflow { offset: usize, start: usize },
    LengthOutOfBounds { offset: usize, length: u64 },
    InvalidTag { offset: usize, tag: u64 },
    InvalidWire { offset: usize, wire: u8 },
    OutOfOrder { offset: usize, previous: u32, tag: u32 },
    Duplicate { offset: usize, tag: u32 },
    NonCanonical { offset: usize },
    DepthExceeded { offset: usize, depth: usize },
    InvalidDepthLimit { limit: usize },
}

fn put_varint(mut value: u64, out: &mut Vec<u8>) {
    while value >= 0x80 {
        out.push((value as u8) | 0x80);
        value >>= 7;
    }
    out.push(value as u8);
}

fn read_varint(input: &[u8], cursor: &mut usize, end: usize) -> Result<u64, Error> {
    let start = *cursor;
    let mut value = 0u64;
    for index in 0..10 {
        if *cursor >= end {
            return Err(Error::Truncated { offset: end, start });
        }
        let byte = input[*cursor];
        let offset = *cursor;
        *cursor += 1;
        let payload = byte & 0x7f;

        if index == 9 && (payload > 1 || byte & 0x80 != 0) {
            return Err(Error::Overflow { offset, start });
        }

        value |= (payload as u64) << (index * 7);

        if byte & 0x80 == 0 {
            if index > 0 && payload == 0 {
                return Err(Error::NonCanonical { offset: start });
            }
            return Ok(value);
        }
    }

    unreachable!()
}

fn append_body(message: &Message, max_depth: usize, depth: usize) -> Result<Vec<u8>, Error> {
    let mut body = Vec::new();
    let mut previous: Option<(u32, u8)> = None;

    for (field_index, field) in message.fields.iter().enumerate() {
        let (tag, wire) = match field {
            Field::UInt { tag, .. } => (*tag, 0),
            Field::Bytes { tag, .. } => (*tag, 2),
            Field::Nested { tag, .. } => (*tag, 3),
        };

        if !(1..=15).contains(&tag) {
            return Err(Error::InvalidTag {
                offset: field_index,
                tag: tag as u64,
            });
        }

        if let Some((previous_tag, previous_wire)) = previous {
            if tag < previous_tag {
                return Err(Error::OutOfOrder {
                    offset: field_index,
                    previous: previous_tag,
                    tag,
                });
            }
            if tag == previous_tag && !(previous_wire == 2 && wire == 2) {
                return Err(Error::Duplicate {
                    offset: field_index,
                    tag,
                });
            }
        }

        let key = ((tag as u64) << 3) | wire as u64;
        put_varint(key, &mut body);

        match field {
            Field::UInt { value, .. } => put_varint(*value, &mut body),
            Field::Bytes { data, .. } => {
                put_varint(data.len() as u64, &mut body);
                body.extend_from_slice(data);
            }
            Field::Nested { message, .. } => {
                let child_depth = depth + 1;
                if child_depth > max_depth {
                    return Err(Error::DepthExceeded {
                        offset: field_index,
                        depth: child_depth,
                    });
                }
                let child = append_body(message, max_depth, child_depth)?;
                put_varint(child.len() as u64, &mut body);
                body.extend_from_slice(&child);
            }
        }

        previous = Some((tag, wire));
    }

    Ok(body)
}

pub fn encode_message(message: &Message, max_depth: usize) -> Result<Vec<u8>, Error> {
    if max_depth > 32 {
        return Err(Error::InvalidDepthLimit { limit: max_depth });
    }
    let body = append_body(message, max_depth, 0)?;
    let mut output = Vec::new();
    put_varint(body.len() as u64, &mut output);
    output.extend_from_slice(&body);
    Ok(output)
}

fn decode_body(
    input: &[u8],
    mut cursor: usize,
    end: usize,
    max_depth: usize,
    depth: usize,
) -> Result<Message, Error> {
    let mut fields = Vec::new();
    let mut previous: Option<(u32, u8)> = None;

    while cursor < end {
        let key_start = cursor;
        let key = read_varint(input, &mut cursor, end)?;
        let raw_tag = key >> 3;
        let wire = (key & 7) as u8;

        if !(1..=255).contains(&raw_tag) {
            return Err(Error::InvalidTag {
                offset: key_start,
                tag: raw_tag,
            });
        }
        if !matches!(wire, 0 | 1 | 2 | 3 | 5) {
            return Err(Error::InvalidWire {
                offset: key_start,
                wire,
            });
        }

        let tag = raw_tag as u32;
        if let Some((previous_tag, previous_wire)) = previous {
            if tag < previous_tag {
                return Err(Error::OutOfOrder {
                    offset: key_start,
                    previous: previous_tag,
                    tag,
                });
            }
            if tag == previous_tag && !(previous_wire == 2 && wire == 2) {
                return Err(Error::Duplicate {
                    offset: key_start,
                    tag,
                });
            }
        }
        previous = Some((tag, wire));

        match wire {
            0 => {
                let value = read_varint(input, &mut cursor, end)?;
                if tag <= 15 {
                    fields.push(Field::UInt { tag, value });
                }
            }
            1 | 5 => {
                let size = if wire == 1 { 8usize } else { 4usize };
                if end - cursor < size {
                    return Err(Error::LengthOutOfBounds {
                        offset: key_start,
                        length: size as u64,
                    });
                }
                cursor += size;
            }
            2 => {
                let length_start = cursor;
                let length = read_varint(input, &mut cursor, end)?;
                let remaining = end - cursor;
                if length > remaining as u64 {
                    return Err(Error::LengthOutOfBounds {
                        offset: length_start,
                        length,
                    });
                }
                let length = length as usize;
                if tag <= 15 {
                    let data = input[cursor..cursor + length].to_vec();
                    if let Some(Field::Bytes {
                        tag: last_tag,
                        data: last_data,
                    }) = fields.last_mut()
                    {
                        if *last_tag == tag {
                            last_data.extend_from_slice(&data);
                        } else {
                            fields.push(Field::Bytes { tag, data });
                        }
                    } else {
                        fields.push(Field::Bytes { tag, data });
                    }
                }
                cursor += length;
            }
            3 => {
                let length_start = cursor;
                let length = read_varint(input, &mut cursor, end)?;
                let remaining = end - cursor;
                if length > remaining as u64 {
                    return Err(Error::LengthOutOfBounds {
                        offset: length_start,
                        length,
                    });
                }
                let length = length as usize;
                let child_depth = depth + 1;
                if child_depth > max_depth {
                    return Err(Error::DepthExceeded {
                        offset: key_start,
                        depth: child_depth,
                    });
                }
                let child_end = cursor + length;
                let child = decode_body(input, cursor, child_end, max_depth, child_depth)?;
                cursor = child_end;
                if tag <= 15 {
                    fields.push(Field::Nested {
                        tag,
                        message: child,
                    });
                }
            }
            _ => unreachable!(),
        }
    }

    Ok(Message { fields })
}

pub fn decode_message(
    input: &[u8],
    start: usize,
    max_depth: usize,
) -> Result<(Message, usize), Error> {
    if max_depth > 32 {
        return Err(Error::InvalidDepthLimit { limit: max_depth });
    }
    if start > input.len() {
        return Err(Error::StartOutOfBounds { offset: start });
    }

    let mut cursor = start;
    let header_start = cursor;
    let frame_length = read_varint(input, &mut cursor, input.len())?;
    let remaining = input.len() - cursor;
    if frame_length > remaining as u64 {
        return Err(Error::LengthOutOfBounds {
            offset: header_start,
            length: frame_length,
        });
    }

    let body_length = frame_length as usize;
    let end = cursor + body_length;
    let message = decode_body(input, cursor, end, max_depth, 0)?;
    Ok((message, end))
}

fn main() {}
