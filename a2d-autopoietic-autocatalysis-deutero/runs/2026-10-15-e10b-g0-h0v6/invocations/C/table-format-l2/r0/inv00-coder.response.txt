#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
    AlignmentCount { expected: usize, actual: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let columns = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != columns {
            return Err(TableError::Ragged {
                row,
                expected: columns,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if !('\u{20}'..='\u{7e}').contains(&ch) {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }
        }
    }

    let trimmed: Vec<Vec<&str>> = rows
        .iter()
        .map(|cells| {
            cells
                .iter()
                .map(|cell| cell.trim_end_matches(' '))
                .collect()
        })
        .collect();

    let widths: Vec<usize> = (0..columns)
        .map(|column| {
            trimmed
                .iter()
                .map(|row| row[column].len())
                .max()
                .unwrap_or(0)
                .max(1)
        })
        .collect();

    for (column, &width) in widths.iter().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let mut output = String::new();

    let append_border = |output: &mut String| {
        output.push('+');
        for &width in &widths {
            for _ in 0..width + 2 {
                output.push('-');
            }
            output.push('+');
        }
        output.push('\n');
    };

    let append_row = |output: &mut String, row_index: usize| {
        output.push('|');
        for column in 0..columns {
            let content = trimmed[row_index][column];
            let extra = widths[column] - content.len();
            let alignment = if options.alignments.is_empty() {
                &Alignment::Left
            } else {
                &options.alignments[column]
            };
            let left = match alignment {
                Alignment::Left => 0,
                Alignment::Right => extra,
                Alignment::Center => extra / 2,
            };
            let right = extra - left;

            output.push(' ');
            for _ in 0..left {
                output.push(' ');
            }
            output.push_str(content);
            for _ in 0..right {
                output.push(' ');
            }
            output.push_str(" |");
        }
        output.push('\n');
    };

    append_border(&mut output);
    for row_index in 0..rows.len() {
        append_row(&mut output, row_index);
        if options.header && row_index == 0 && rows.len() > 1 {
            append_border(&mut output);
        }
    }
    append_border(&mut output);

    Ok(output)
}

fn main() {}
