#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let id = self.routes.len();
        let mut rank = Vec::with_capacity(segments.len() + 1);
        for segment in &segments {
            match segment {
                Segment::Static(_) => rank.push(2),
                Segment::Param(_) => rank.push(1),
                Segment::Wildcard(_) => rank.push(0),
            }
        }
        if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
            rank.push(3);
        }
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parts = parse_path(head)?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            let mut params = Vec::new();
            let mut pi = 0;
            let mut ri = 0;
            let mut matched = true;

            while ri < route.segments.len() {
                match &route.segments[ri] {
                    Segment::Static(expected) => {
                        if pi >= parts.len() || parts[pi] != *expected {
                            matched = false;
                            break;
                        }
                        pi += 1;
                        ri += 1;
                    }
                    Segment::Param(name) => {
                        if pi >= parts.len() {
                            matched = false;
                            break;
                        }
                        params.push((name.clone(), parts[pi].clone()));
                        pi += 1;
                        ri += 1;
                    }
                    Segment::Wildcard(name) => {
                        params.push((name.clone(), parts[pi..].join("/")));
                        pi = parts.len();
                        ri += 1;
                    }
                }
            }

            if matched && pi == parts.len() {
                let replace = match &best {
                    None => true,
                    Some((prior, _)) => {
                        route.rank > prior.rank
                            || (route.rank == prior.rank && route.id < prior.id)
                    }
                };
                if replace {
                    best = Some((route, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let mut raw_segments = split_segments(input);
    let trailing_slash = input.ends_with('/');
    if trailing_slash {
        raw_segments.pop();
    }

    let mut segments = Vec::with_capacity(raw_segments.len());
    let mut names = std::collections::HashSet::new();

    for (start, end) in raw_segments {
        let raw = &input[start..end];
        if raw.is_empty() {
            return Err(Error::InvalidPattern { at: start });
        }

        if raw.starts_with(':') || raw.starts_with('*') {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if marker == b'*' && end != input.len() - usize::from(trailing_slash) {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            let decoded = decode_segment(raw, start).map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }

    if trailing_slash {
        return Err(Error::InvalidPattern {
            at: input.len() - 1,
        });
    }

    Ok(segments)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let mut raw_segments = split_segments(input);
    let trailing_slash = input.ends_with('/');
    if trailing_slash {
        raw_segments.pop();
    }

    let mut parts = Vec::with_capacity(raw_segments.len());
    for (start, end) in raw_segments {
        let raw = &input[start..end];
        if raw.is_empty() {
            return Err(Error::InvalidPath { at: start });
        }
        let decoded = decode_segment(raw, start).map_err(|at| Error::InvalidPath { at })?;
        parts.push(decoded);
    }

    if trailing_slash {
        return Err(Error::InvalidPath {
            at: input.len() - 1,
        });
    }

    Ok(parts)
}

fn split_segments(input: &str) -> Vec<(usize, usize)> {
    let mut segments = Vec::new();
    let mut start = 1;
    for (i, byte) in input.bytes().enumerate().skip(1) {
        if byte == b'/' {
            segments.push((start, i));
            start = i + 1;
        }
    }
    segments.push((start, input.len()));
    segments
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let high = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let low = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let decoded = (high << 4) | low;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else {
            if !is_unreserved(byte) {
                return Err(base + i);
            }
            out.push(byte);
            i += 1;
        }
    }

    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn hex_upper(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn main() {}
