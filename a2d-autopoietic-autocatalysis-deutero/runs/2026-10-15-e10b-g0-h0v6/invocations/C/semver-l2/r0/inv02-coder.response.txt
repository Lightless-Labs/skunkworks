use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte_at(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while matches!(byte_at(s, *pos), Some(b'0'..=b'9')) {
        *pos += 1;
    }

    if *pos == start {
        return match byte_at(s, start) {
            None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at: start }),
            _ => Err(VersionError::InvalidChar { at: start }),
        };
    }

    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    numeric_pre: bool,
) -> Result<Vec<String>, VersionError> {
    if start == end {
        return Err(VersionError::EmptyIdentifier { at: start });
    }

    let mut out = Vec::new();
    let mut id_start = start;
    let mut i = start;

    while i <= end {
        if i == end || s.as_bytes()[i] == b'.' {
            if i == id_start {
                return Err(VersionError::EmptyIdentifier { at: id_start });
            }
            let ident = &s[id_start..i];
            if numeric_pre
                && ident.len() > 1
                && ident.as_bytes()[0] == b'0'
                && ident.bytes().all(|b| b.is_ascii_digit())
            {
                return Err(VersionError::LeadingZero { at: id_start });
            }
            out.push(ident.to_string());
            id_start = i + 1;
            i += 1;
            continue;
        }

        let b = s.as_bytes()[i];
        if !(b.is_ascii_alphanumeric() || b == b'-') {
            return Err(VersionError::InvalidChar { at: i });
        }
        i += 1;
    }

    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;

    match byte_at(s, pos) {
        Some(b'.') => pos += 1,
        None | Some(b'-' | b'+') => return Err(VersionError::MissingCore { at: pos }),
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }

    let minor = parse_core(s, &mut pos)?;

    match byte_at(s, pos) {
        Some(b'.') => pos += 1,
        None | Some(b'-' | b'+') => return Err(VersionError::MissingCore { at: pos }),
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }

    let patch = parse_core(s, &mut pos)?;
    parse_version_tail(s, pos, major, minor, patch)
}

fn parse_version_tail(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if byte_at(s, pos) == Some(b'-') {
        pos += 1;
        let plus = s[pos..].find('+').map(|n| pos + n);
        let pre_end = plus.unwrap_or(s.len());
        pre = parse_identifiers(s, pos, pre_end, true)?;
        pos = pre_end;
    }

    if byte_at(s, pos) == Some(b'+') {
        pos += 1;
        build = parse_identifiers(s, pos, s.len(), false)?;
        pos = s.len();
    }

    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }

    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric(x);
        let yn = is_numeric(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }

    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    checks: Vec<Check>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Check {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn check_matches(v: &Version, c: &Check) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn add_check(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors
            .push((version.major, version.minor, version.patch));
    }
    arm.checks.push(Check { op, version });
}

#[derive(Clone, Debug)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
}

fn syntax(at: usize) -> RangeError {
    RangeError::Syntax { at }
}

fn parse_partial_operand(operand: &str, start: usize) -> Result<Partial, RangeError> {
    let has_suffix = operand.contains('-') || operand.contains('+');
    let fields: Vec<&str> = operand.split('.').collect();
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));

    if has_suffix || (fields.len() >= 3 && !has_wild) {
        return parse_version(operand)
            .map(|v| Partial {
                nums: vec![v.major, v.minor, v.patch],
                full: Some(v),
            })
            .map_err(|error| RangeError::Version { at: start, error });
    }

    if fields.is_empty() || fields.len() > 3 {
        return Err(syntax(start));
    }

    let mut nums = Vec::new();
    let mut wildcard_seen = false;

    for field in fields {
        if matches!(field, "*" | "x" | "X") {
            wildcard_seen = true;
            continue;
        }
        if field.is_empty()
            || wildcard_seen
            || !field.bytes().all(|b| b.is_ascii_digit())
            || (field.len() > 1 && field.as_bytes()[0] == b'0')
        {
            return Err(syntax(start));
        }
        let n = field.parse::<u64>().map_err(|_| syntax(start))?;
        nums.push(n);
    }

    Ok(Partial { nums, full: None })
}

fn padded(p: &Partial) -> (u64, u64, u64) {
    (
        *p.nums.first().unwrap_or(&0),
        *p.nums.get(1).unwrap_or(&0),
        *p.nums.get(2).unwrap_or(&0),
    )
}

fn endpoint(tuple: (u64, u64, u64), pre: Vec<String>) -> Version {
    Version {
        major: tuple.0,
        minor: tuple.1,
        patch: tuple.2,
        pre,
        build: Vec::new(),
    }
}

fn increment_upper(
    tuple: (u64, u64, u64),
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let mut t = tuple;
    match field {
        0 => t.0 = t.0.checked_add(1).ok_or_else(|| syntax(at))?,
        1 => t.1 = t.1.checked_add(1).ok_or_else(|| syntax(at))?,
        _ => t.2 = t.2.checked_add(1).ok_or_else(|| syntax(at))?,
    }

    if field == 0 {
        t.1 = 0;
        t.2 = 0;
    } else if field == 1 {
        t.2 = 0;
    }

    Ok(endpoint(t, vec!["0".to_string()]))
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens(s: &str, start: usize, end: usize) -> Vec<(usize, &str)> {
    let mut out = Vec::new();
    let mut i = start;

    while i < end {
        while i < end && is_ws(s.as_bytes()[i]) {
            i += 1;
        }
        if i >= end {
            break;
        }
        let st = i;
        while i < end && !is_ws(s.as_bytes()[i]) {
            i += 1;
        }
        out.push((st, &s[st..i]));
    }

    out
}

fn expand_partial_bare(arm: &mut Arm, p: Partial, at: usize) -> Result<(), RangeError> {
    if let Some(v) = p.full {
        add_check(arm, Op::Eq, v, true);
        return Ok(());
    }

    let k = p.nums.len();
    if k == 0 {
        return Ok(());
    }

    let t = padded(&p);
    add_check(arm, Op::Ge, endpoint(t, Vec::new()), false);

    if k == 1 {
        add_check(arm, Op::Lt, increment_upper(t, 0, at)?, false);
    } else if k == 2 {
        add_check(arm, Op::Lt, increment_upper(t, 1, at)?, false);
    }

    Ok(())
}

fn expand_caret_tilde(
    arm: &mut Arm,
    p: Partial,
    at: usize,
    caret: bool,
) -> Result<(), RangeError> {
    let k = p.nums.len();
    if k == 0 {
        return Err(syntax(at));
    }

    let t = padded(&p);
    let lower = p
        .full
        .clone()
        .unwrap_or_else(|| endpoint(t, Vec::new()));
    add_check(arm, Op::Ge, lower, true);

    let field = if caret {
        if t.0 > 0 || k == 1 {
            0
        } else if t.1 > 0 || k == 2 {
            1
        } else {
            2
        }
    } else if k == 1 {
        0
    } else {
        1
    };

    add_check(arm, Op::Lt, increment_upper(t, field, at)?, false);
    Ok(())
}

fn parse_hyphen_arm(toks: &[(usize, &str)], arm: &mut Arm) -> Result<(), RangeError> {
    let minus = toks.iter().find(|(_, t)| *t == "-").unwrap().0;
    if toks.len() != 3 || toks[1].1 != "-" {
        return Err(syntax(minus));
    }

    let lo = parse_partial_operand(toks[0].1, toks[0].0)?;
    let hi = parse_partial_operand(toks[2].1, toks[2].0)?;

    if lo.nums.is_empty() {
        return Err(syntax(toks[0].0));
    }
    if hi.nums.is_empty() {
        return Err(syntax(toks[2].0));
    }

    let lt = padded(&lo);
    let lower = lo.full.clone().unwrap_or_else(|| endpoint(lt, Vec::new()));
    add_check(arm, Op::Ge, lower, true);

    if let Some(full) = hi.full {
        add_check(arm, Op::Le, full, true);
    } else {
        let ht = padded(&hi);
        if hi.nums.len() == 1 {
            add_check(arm, Op::Lt, increment_upper(ht, 0, toks[2].0)?, false);
        } else if hi.nums.len() == 2 {
            add_check(arm, Op::Lt, increment_upper(ht, 1, toks[2].0)?, false);
        } else {
            add_check(arm, Op::Le, endpoint(ht, Vec::new()), false);
        }
    }

    Ok(())
}

fn parse_arm(s: &str, start: usize, end: usize) -> Result<Arm, RangeError> {
    let toks = tokens(s, start, end);
    if toks.is_empty() {
        return Err(RangeError::Empty { at: start });
    }

    let mut arm = Arm {
        checks: Vec::new(),
        anchors: Vec::new(),
    };

    if toks.iter().any(|(_, t)| *t == "-") {
        parse_hyphen_arm(&toks, &mut arm)?;
        return Ok(arm);
    }

    for (at, token) in toks {
        if token.starts_with('^') || token.starts_with('~') {
            let caret = token.as_bytes()[0] == b'^';
            let operand_start = at + 1;
            let token_end = at + token.len();
            if operand_start >= token_end {
                return Err(syntax(at));
            }
            let p = parse_partial_operand(&s[operand_start..token_end], operand_start)?;
            expand_caret_tilde(&mut arm, p, at, caret)?;
        } else if token.starts_with('<') || token.starts_with('>') || token.starts_with('=') {
            let (op, oplen) = if token.starts_with("<=") {
                (Op::Le, 2)
            } else if token.starts_with(">=") {
                (Op::Ge, 2)
            } else if token.starts_with('<') {
                (Op::Lt, 1)
            } else if token.starts_with('>') {
                (Op::Gt, 1)
            } else {
                (Op::Eq, 1)
            };

            let operand_start = at + oplen;
            let token_end = at + token.len();
            if operand_start >= token_end || !s.as_bytes()[operand_start].is_ascii_digit() {
                return Err(syntax(at));
            }

            let v = parse_version(&s[operand_start..token_end]).map_err(|error| {
                RangeError::Version {
                    at: operand_start,
                    error,
                }
            })?;
            add_check(&mut arm, op, v, true);
        } else if token.starts_with('*')
            || token.starts_with('x')
            || token.starts_with('X')
            || token.as_bytes()[0].is_ascii_digit()
        {
            let p = parse_partial_operand(token, at)?;
            expand_partial_bare(&mut arm, p, at)?;
        } else {
            return Err(syntax(at));
        }
    }

    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let chunks = tokens(s, 0, s.len());
    if chunks.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let separators: Vec<usize> = chunks
        .iter()
        .enumerate()
        .filter_map(|(idx, (_, token))| (*token == "||").then_some(idx))
        .collect();

    let mut arms = Vec::new();
    let mut chunk_start = 0;

    for sep_idx in separators {
        let sep_start = chunks[sep_idx].0;
        if sep_idx == chunk_start {
            return Err(RangeError::Empty { at: sep_start });
        }

        let arm_start = chunks[chunk_start].0;
        let arm_end = chunks[sep_idx - 1].0 + chunks[sep_idx - 1].1.len();
        arms.push(parse_arm(s, arm_start, arm_end)?);

        chunk_start = sep_idx + 1;
        if chunk_start >= chunks.len() {
            return Err(RangeError::Empty { at: s.len() });
        }
    }

    if chunk_start < chunks.len() {
        let arm_start = chunks[chunk_start].0;
        let last = chunks[chunks.len() - 1];
        let arm_end = last.0 + last.1.len();
        arms.push(parse_arm(s, arm_start, arm_end)?);
    }

    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty()
            && !arm.anchors.contains(&(v.major, v.minor, v.patch))
        {
            return false;
        }
        arm.checks.iter().all(|c| check_matches(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version =
            parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (i, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(version, &parsed[j]);
            if (want_max && ord == Ordering::Greater)
                || (!want_max && ord == Ordering::Less)
            {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }

    Ok(best.map(|i| candidates[i].clone()))
}
