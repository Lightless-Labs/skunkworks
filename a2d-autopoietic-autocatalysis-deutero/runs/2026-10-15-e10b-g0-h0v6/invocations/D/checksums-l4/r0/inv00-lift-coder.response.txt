#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

const ADLER_MOD: u64 = 65_521;

#[derive(Debug, Clone)]
pub struct Stream {
    crc_raw: u32,
    adler_a: u32,
    adler_b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_raw: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= ADLER_MOD as u32 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= ADLER_MOD as u32 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_raw: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_raw ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        update_crc(&mut self.crc_raw, data);
        update_adler(&mut self.adler_a, &mut self.adler_b, data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let sums = self.checksum();
        format!("c{:08x}:a{:08x}", sums.crc32, sums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let crc = parse_hex_u32(bytes, 1, 9)?;
        let adler = parse_hex_u32(bytes, 11, 19)?;
        Self::from_seed(crc, adler)
    }
}

fn update_crc(raw: &mut u32, data: &[u8]) {
    for &byte in data {
        *raw ^= byte as u32;
        for _ in 0..8 {
            let low = *raw & 1;
            *raw >>= 1;
            if low != 0 {
                *raw ^= 0xedb8_8320;
            }
        }
    }
}

fn update_adler(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *a = ((*a as u64 + byte as u64) % ADLER_MOD) as u32;
        *b = ((*b as u64 + *a as u64) % ADLER_MOD) as u32;
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut raw = 0xffff_ffff;
    update_crc(&mut raw, data);
    raw ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1;
    let mut b = 0;
    update_adler(&mut a, &mut b, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }
    Ok(stream.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

fn update_crc16(state: &mut u16, data: &[u8], variant: Crc16) {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                *state ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = *state & 0x8000;
                    *state = state.wrapping_shl(1);
                    if top != 0 {
                        *state ^= 0x1021;
                    }
                }
            }
        }
        Crc16::X25 => {
            for &byte in data {
                *state ^= byte as u16;
                for _ in 0..8 {
                    let low = *state & 1;
                    *state >>= 1;
                    if low != 0 {
                        *state ^= 0x8408;
                    }
                }
            }
        }
    }
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let initial = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    let mut state = initial;
    update_crc16(&mut state, data, variant);
    if variant == Crc16::X25 {
        state ^ 0xffff
    } else {
        state
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

fn update_fletcher16(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *a = (*a + byte as u32) % 255;
        *b = (*b + *a) % 255;
    }
}

fn update_fletcher32(a: &mut u64, b: &mut u64, word: u16) {
    *a = (*a + word as u64) % 65_535;
    *b = (*b + *a) % 65_535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u64;
    let mut b = 0u64;
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        update_fletcher32(&mut a, &mut b, u16::from_be_bytes([pair[0], pair[1]]));
    }
    if let Some(&last) = chunks.remainder().first() {
        update_fletcher32(&mut a, &mut b, (last as u16) << 8);
    }
    ((b as u32) << 16) | a as u32
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u64;
    let mut b = 0u64;
    let mut pending_high: Option<u8> = None;

    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending_high.take() {
                update_fletcher32(&mut a, &mut b, u16::from_be_bytes([high, byte]));
            } else {
                pending_high = Some(byte);
            }
        }
    }

    if let Some(high) = pending_high {
        update_fletcher32(&mut a, &mut b, (high as u16) << 8);
    }
    Ok(((b as u32) << 16) | a as u32)
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut stream = Stream::new();
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32_a = 0u64;
    let mut f32_b = 0u64;
    let mut f32_pending: Option<u8> = None;
    let mut crc16_state = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        stream.update(selected);
        update_fletcher16(&mut f16_a, &mut f16_b, selected);
        update_crc16(&mut crc16_state, selected, variant);

        for &byte in selected {
            if let Some(high) = f32_pending.take() {
                update_fletcher32(&mut f32_a, &mut f32_b, u16::from_be_bytes([high, byte]));
            } else {
                f32_pending = Some(byte);
            }
        }
    }

    if let Some(high) = f32_pending {
        update_fletcher32(&mut f32_a, &mut f32_b, (high as u16) << 8);
    }

    if variant == Crc16::X25 {
        crc16_state ^= 0xffff;
    }

    Ok(Multi {
        base: stream.checksum(),
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: ((f32_b as u32) << 16) | f32_a as u32,
        crc16: crc16_state,
    })
}

fn matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn matrix_square(matrix: &[u32; 32]) -> [u32; 32] {
    let mut square = [0u32; 32];
    for n in 0..32 {
        square[n] = matrix_times(matrix, matrix[n]);
    }
    square
}

fn crc32_zero_byte_operator() -> [u32; 32] {
    let mut operator = [0u32; 32];
    for bit in 0..32 {
        let mut value = 1u32 << bit;
        for _ in 0..8 {
            let low = value & 1;
            value >>= 1;
            if low != 0 {
                value ^= 0xedb8_8320;
            }
        }
        operator[bit] = value;
    }
    operator
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut length = right_len;
    let mut operator = crc32_zero_byte_operator();
    let mut shifted = left;
    while length != 0 {
        if length & 1 != 0 {
            shifted = matrix_times(&operator, shifted);
        }
        length >>= 1;
        if length != 0 {
            operator = matrix_square(&operator);
        }
    }
    shifted ^ right
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Part {
    pub crc32: u32,
    pub len: u64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CombineError {
    LengthOverflow { part: usize },
    EmptyCrc { part: usize },
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut crc = 0;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    let mut length = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        stream.update(selected);
        length += selected.len() as u64;
    }
    let sums = stream.checksum();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        length, sums.crc32, sums.adler32
    ))
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VerifyError {
    Span(SpanError),
    Shape,
    Hex { offset: usize },
    Length,
    Mismatch { field: usize, expected: u32, actual: u32 },
}

fn hex_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'f' => Some((byte - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse_hex_u32(
    bytes: &[u8],
    start: usize,
    end: usize,
) -> Result<u32, SnapshotError> {
    let mut value = 0u32;
    for offset in start..end {
        let digit = hex_value(bytes[offset])
            .ok_or(SnapshotError::Hex { offset })?;
        value = (value << 4) | digit;
    }
    Ok(value)
}

fn parse_hex_u64_verify(
    bytes: &[u8],
    start: usize,
    end: usize,
) -> Result<u64, VerifyError> {
    let mut value = 0u64;
    for offset in start..end {
        let digit = hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        value = (value << 4) | digit as u64;
    }
    Ok(value)
}

fn parse_hex_u32_verify(
    bytes: &[u8],
    start: usize,
    end: usize,
) -> Result<u32, VerifyError> {
    let mut value = 0u32;
    for offset in start..end {
        let digit = hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        value = (value << 4) | digit;
    }
    Ok(value)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    // Validate all fields in byte-offset order before comparing any values.
    let length = parse_hex_u64_verify(bytes, 4, 20)?;
    let crc = parse_hex_u32_verify(bytes, 25, 33)?;
    let adler = parse_hex_u32_verify(bytes, 40, 48)?;

    let mut stream = Stream::new();
    let mut selected_len = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        stream.update(selected);
        selected_len += selected.len() as u64;
    }

    if length != selected_len {
        return Err(VerifyError::Length);
    }

    let actual = stream.checksum();
    if crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: crc,
            actual: actual.crc32,
        });
    }
    if adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
