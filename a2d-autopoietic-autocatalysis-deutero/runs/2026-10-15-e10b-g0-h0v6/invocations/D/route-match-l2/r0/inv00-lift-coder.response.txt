#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
    id: usize,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let (segments, trailing) = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
            rank.push(3);
        }

        let id = self.routes.len();
        let _ = trailing;
        self.routes.push(Route { segments, rank, id });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let (parts, _) = parse_path(path)?;
        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            if let Some(params) = match_route(route, &parts) {
                let replace = match &best {
                    None => true,
                    Some((old, _)) => route.rank > old.rank,
                };
                if replace {
                    best = Some((route, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    parse_segments(input, true).map(|(segments, trailing)| (segments, trailing))
}

fn parse_path(input: &str) -> Result<(Vec<String>, bool), Error> {
    let head_end = input
        .char_indices()
        .find(|(_, c)| *c == '?' || *c == '#')
        .map(|(i, _)| i)
        .unwrap_or(input.len());
    let head = &input[..head_end];
    if !head.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }

    let (segments, trailing) = parse_segments(head, false)?;
    let values = segments
        .into_iter()
        .map(|s| match s {
            Segment::Static(value) => value,
            _ => unreachable!(),
        })
        .collect();
    Ok((values, trailing))
}

fn parse_segments(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let mut spans = Vec::new();
    let mut start = 1usize;
    let mut i = 1usize;

    while i <= bytes.len() {
        if i == bytes.len() || bytes[i] == b'/' {
            if i == start {
                if i == bytes.len() && i > 0 && bytes[i - 1] == b'/' {
                    break;
                }
                return Err(invalid(start));
            }
            spans.push((start, i));
            if i == bytes.len() {
                break;
            }
            start = i + 1;
        }
        i += 1;
    }

    let trailing = bytes.len() > 1 && bytes[bytes.len() - 1] == b'/';
    if trailing {
        // The final empty span is permitted during segment parsing; reject after
        // all preceding segments have been validated.
    }

    let mut segments = Vec::with_capacity(spans.len());
    let mut names = Vec::<String>::new();

    for (index, &(begin, end)) in spans.iter().enumerate() {
        let raw = &input[begin..end];
        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_name(name) || names.iter().any(|n| n == name) {
                return Err(invalid(begin));
            }
            if marker == b'*' && index + 1 != spans.len() {
                return Err(invalid(begin));
            }
            names.push(name.to_owned());
            segments.push(if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            segments.push(Segment::Static(decode_segment(raw, begin).map_err(invalid)?));
        }
    }

    if trailing {
        return Err(invalid(bytes.len() - 1));
    }
    Ok((segments, trailing))
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    if b.is_empty() || !b[0].is_ascii_lowercase() {
        return false;
    }
    b[1..]
        .iter()
        .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_')
}

fn decode_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(base + i);
            }
            out.push(b);
            i += 1;
        }
    }

    Ok(String::from_utf8(out).unwrap())
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut pi = 0usize;

    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if path.get(pi)? != value {
                    return None;
                }
                pi += 1;
            }
            Segment::Param(name) => {
                let value = path.get(pi)?;
                params.push((name.clone(), value.clone()));
                pi += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[pi..].join("/");
                params.push((name.clone(), value));
                pi = path.len();
                break;
            }
        }
    }

    if pi == path.len() {
        Some(params)
    } else {
        None
    }
}
