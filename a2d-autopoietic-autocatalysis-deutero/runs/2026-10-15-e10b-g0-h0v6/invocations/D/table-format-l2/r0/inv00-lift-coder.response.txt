#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
    AlignmentCount { expected: usize, actual: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if !(' '..='~').contains(&ch) {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }
        }
    }

    let trimmed: Vec<Vec<&str>> = rows
        .iter()
        .map(|cells| {
            cells
                .iter()
                .map(|cell| cell.trim_end_matches(' '))
                .collect()
        })
        .collect();

    let mut widths = vec![1usize; n];
    for cells in &trimmed {
        for (column, cell) in cells.iter().enumerate() {
            widths[column] = widths[column].max(cell.len());
        }
    }

    for (column, &width) in widths.iter().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let alignment = |column: usize| {
        if options.alignments.is_empty() {
            &Alignment::Left
        } else {
            &options.alignments[column]
        }
    };

    let mut output = String::new();
    append_border(&mut output, &widths);

    for (row_index, cells) in trimmed.iter().enumerate() {
        output.push('|');
        for (column, cell) in cells.iter().enumerate() {
            output.push(' ');
            let extra = widths[column] - cell.len();
            match alignment(column) {
                Alignment::Left => {
                    output.push_str(cell);
                    push_spaces(&mut output, extra);
                }
                Alignment::Right => {
                    push_spaces(&mut output, extra);
                    output.push_str(cell);
                }
                Alignment::Center => {
                    let left = extra / 2;
                    push_spaces(&mut output, left);
                    output.push_str(cell);
                    push_spaces(&mut output, extra - left);
                }
            }
            output.push(' ');
            output.push('|');
        }
        output.push('\n');

        if options.header && row_index == 0 && rows.len() > 1 {
            append_border(&mut output, &widths);
        }
    }

    append_border(&mut output, &widths);
    Ok(output)
}

fn append_border(output: &mut String, widths: &[usize]) {
    output.push('+');
    for &width in widths {
        push_repeated(output, '-', width + 2);
        output.push('+');
    }
    output.push('\n');
}

fn push_spaces(output: &mut String, count: usize) {
    push_repeated(output, ' ', count);
}

fn push_repeated(output: &mut String, ch: char, count: usize) {
    for _ in 0..count {
        output.push(ch);
    }
}
