#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<u8>; 5],
    day_unrestricted: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

#[derive(Clone, Copy)]
struct Domain {
    min: u8,
    max: u8,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_error(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn ascii_name_value(bytes: &[u8], field: Field) -> Option<u8> {
    if !bytes.iter().all(u8::is_ascii_alphabetic) {
        return None;
    }
    let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
    match field {
        Field::Month => match upper.as_slice() {
            b"JAN" => Some(1),
            b"FEB" => Some(2),
            b"MAR" => Some(3),
            b"APR" => Some(4),
            b"MAY" => Some(5),
            b"JUN" => Some(6),
            b"JUL" => Some(7),
            b"AUG" => Some(8),
            b"SEP" => Some(9),
            b"OCT" => Some(10),
            b"NOV" => Some(11),
            b"DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_slice() {
            b"SUN" => Some(0),
            b"MON" => Some(1),
            b"TUE" => Some(2),
            b"WED" => Some(3),
            b"THU" => Some(4),
            b"FRI" => Some(5),
            b"SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

fn parse_number(
    bytes: &[u8],
    field: Field,
    offset: usize,
    is_step: bool,
) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(parse_error(field, offset, 0));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(parse_error(field, offset, 1));
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| parse_error(field, offset, 2))?;
    }
    if is_step {
        let d = domain(field);
        let width = (d.max - d.min + 1) as u32;
        if value == 0 || value > width {
            return Err(parse_error(field, offset, 5));
        }
    } else {
        let d = domain(field);
        if value < d.min as u32 || value > d.max as u32 {
            return Err(parse_error(field, offset, 3));
        }
    }
    Ok(value)
}

fn parse_atom(bytes: &[u8], field: Field, offset: usize) -> Result<u8, ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) {
        return Ok(parse_number(bytes, field, offset, false)? as u8);
    }
    ascii_name_value(bytes, field).ok_or_else(|| parse_error(field, offset, 0))
}

fn parse_item(
    item: &[u8],
    item_offset: usize,
    field: Field,
    selected: &mut [bool; 60],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    let (base, step_part) = if slash_positions.is_empty() {
        (item, None)
    } else if slash_positions.len() == 1 {
        let pos = slash_positions[0];
        (&item[..pos], Some((&item[pos + 1..], item_offset + pos + 1)))
    } else {
        return Err(parse_error(field, item_offset, 0));
    };

    if base.is_empty() || step_part.is_some_and(|(s, _)| s.is_empty()) {
        return Err(parse_error(field, item_offset, 0));
    }

    let (start_bytes, end_bytes, is_star): (Option<&[u8]>, Option<&[u8]>, bool) =
        if base == b"*" {
            (None, None, true)
        } else {
            if base.contains(&b'*') {
                return Err(parse_error(field, item_offset, 0));
            }
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| (b == b'-').then_some(i))
                .collect();
            match hyphens.as_slice() {
                [] => (Some(base), None, false),
                [pos] if *pos > 0 && *pos + 1 < base.len() => (
                    Some(&base[..*pos]),
                    Some(&base[*pos + 1..]),
                    false,
                ),
                _ => return Err(parse_error(field, item_offset, 0)),
            }
        };

    let dom = domain(field);
    let (start, end) = if is_star {
        (dom.min, dom.max)
    } else {
        let sb = start_bytes.unwrap();
        let start_value = parse_atom(sb, field, item_offset)?;
        if let Some(eb) = end_bytes {
            let end_offset = item_offset + sb.len() + 1;
            let end_value = parse_atom(eb, field, end_offset)?;
            if start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_offset,
                });
            }
            (start_value, end_value)
        } else if step_part.is_some() {
            (start_value, dom.max)
        } else {
            (start_value, start_value)
        }
    };

    let step = if let Some((bytes, offset)) = step_part {
        Some(parse_number(bytes, field, offset, true)? as u8)
    } else {
        None
    };

    let stride = step.unwrap_or(1);
    let mut value = start as u16;
    while value <= end as u16 {
        selected[value as usize] = true;
        value += stride as u16;
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();
    for (offset, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = [false; 60];
    let mut start = 0;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b',' {
            parse_item(&bytes[start..i], start, field, &mut selected)?;
            start = i + 1;
        }
    }
    parse_item(&bytes[start..], start, field, &mut selected)?;

    let d = domain(field);
    Ok((d.min..=d.max)
        .filter(|&v| selected[v as usize])
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded;
    let input = if input.as_bytes().first() == Some(&b'@') {
        expanded = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        expanded
    } else {
        input
    };

    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }
    let mut sets: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    for i in 0..5 {
        sets[i] = parse_field(pieces[i], FIELDS[i])?;
    }
    Ok(Cron {
        sets,
        day_unrestricted: [pieces[2] == "*", pieces[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.sets[field_index(field)].clone()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        self.sets[field_index(field)].binary_search(&value).is_ok()
    }

    fn day_matches(&self, at: CivilDateTime, weekday: u8) -> bool {
        let dom = self.contains(Field::DayOfMonth, at.day);
        let dow = self.contains(Field::DayOfWeek, weekday);
        match self.day_unrestricted {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] if at.day == month_length(at.year, at.month) => dom && dow,
            [false, false] => dom || dow,
        }
    }

    fn matches_valid(&self, at: CivilDateTime, weekday: u8) -> bool {
        self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.day_matches(at, weekday)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at, weekday(at.year, at.month, at.day)))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut date = CivilDateTime {
            hour: 0,
            minute: 0,
            ..after
        };
        let last = CivilDateTime {
            year: 2399,
            month: 12,
            day: 31,
            hour: 23,
            minute: 59,
        };
        loop {
            let wd = weekday(date.year, date.month, date.day);
            if self.contains(Field::Month, date.month) && self.day_matches(date, wd) {
                for &hour in &self.sets[1] {
                    for &minute in &self.sets[0] {
                        let candidate = CivilDateTime { hour, minute, ..date };
                        if candidate > after && candidate <= last {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            if date.year == 2399 && date.month == 12 && date.day == 31 {
                break;
            }
            date = next_date(date);
        }
        Ok(None)
    }

    pub fn previous_fire(
        &self,
        before: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut date = CivilDateTime {
            hour: 0,
            minute: 0,
            ..before
        };
        loop {
            let wd = weekday(date.year, date.month, date.day);
            if self.contains(Field::Month, date.month) && self.day_matches(date, wd) {
                for &hour in self.sets[1].iter().rev() {
                    for &minute in self.sets[0].iter().rev() {
                        let candidate = CivilDateTime { hour, minute, ..date };
                        if candidate < before {
                            return Ok(Some(candidate));
                        }
                    }
                }
            }
            if date.year == 2000 && date.month == 1 && date.day == 1 {
                break;
            }
            date = previous_date(date);
        }
        Ok(None)
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while let Some(fire) = self.next_fire(cursor).expect("validated cursor") {
            if fire >= end {
                break;
            }
            fires.push(fire);
            if fires.len() == cap {
                let truncated = self
                    .next_fire(fire)
                    .expect("validated fire")
                    .is_some_and(|next| next < end);
                return Ok(FireWindow { fires, truncated });
            }
            cursor = fire;
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        (0..5)
            .map(|i| {
                let field = FIELDS[i];
                let values = &self.sets[i];
                let d = domain(field);
                let full = values.len() == (d.max - d.min + 1) as usize;
                if (i == 2 && self.day_unrestricted[0]) || (i == 4 && self.day_unrestricted[1]) {
                    "*".to_string()
                } else if full && matches!(field, Field::Minute | Field::Hour | Field::Month) {
                    "*".to_string()
                } else if full && matches!(field, Field::DayOfMonth | Field::DayOfWeek) {
                    "*/1".to_string()
                } else {
                    values.iter().map(u8::to_string).collect::<Vec<_>>().join(",")
                }
            })
            .collect::<Vec<_>>()
            .join(" ")
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if !(1..=month_length(at.year, at.month)).contains(&at.day) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 if leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_length(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn next_date(mut date: CivilDateTime) -> CivilDateTime {
    if date.day < month_length(date.year, date.month) {
        date.day += 1;
    } else if date.month < 12 {
        date.month += 1;
        date.day = 1;
    } else {
        date.year += 1;
        date.month = 1;
        date.day = 1;
    }
    date
}

fn previous_date(mut date: CivilDateTime) -> CivilDateTime {
    if date.day > 1 {
        date.day -= 1;
    } else if date.month > 1 {
        date.month -= 1;
        date.day = month_length(date.year, date.month);
    } else {
        date.year -= 1;
        date.month = 12;
        date.day = 31;
    }
    date
}
