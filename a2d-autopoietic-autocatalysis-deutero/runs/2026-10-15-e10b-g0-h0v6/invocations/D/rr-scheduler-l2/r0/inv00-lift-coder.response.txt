#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut ready: std::collections::VecDeque<usize> = std::collections::VecDeque::new();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut now = 0u64;
    let mut event_index = 0usize;
    let mut last_dispatched_id: Option<u32> = None;

    let admit_through = |time: u64,
                         event_index: &mut usize,
                         ready: &mut std::collections::VecDeque<usize>,
                         completion: &mut Vec<Option<u64>>| {
        while *event_index < events.len()
            && u64::from(jobs[events[*event_index]].arrival) <= time
        {
            let index = events[*event_index];
            let job = &jobs[index];
            if job.ticks == 0 {
                completion[index] = Some(u64::from(job.arrival));
            } else {
                ready.push_back(index);
            }
            *event_index += 1;
        }
    };

    loop {
        admit_through(now, &mut event_index, &mut ready, &mut completion);

        let Some(index) = ready.pop_front() else {
            if event_index == events.len() {
                break;
            }

            let next_time = u64::from(jobs[events[event_index]].arrival);
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatched_id = None;
            }
            continue;
        };

        let job = &jobs[index];
        if last_dispatched_id.is_some_and(|last_id| last_id != job.id) {
            now += u64::from(switch_cost);
        }
        admit_through(now, &mut event_index, &mut ready, &mut completion);

        let start = now;
        let remaining = job.ticks as u64
            - runs
                .iter()
                .filter(|run: &&Run| run.id == job.id)
                .map(|run| run.end - run.start)
                .sum::<u64>();
        let consumed = remaining.min(u64::from(quantum));
        now += consumed;

        let finished = consumed == remaining;
        runs.push(Run {
            id: job.id,
            start,
            end: now,
            finished,
        });

        admit_through(now, &mut event_index, &mut ready, &mut completion);

        if finished {
            completion[index] = Some(now);
        } else {
            ready.push_back(index);
        }
        last_dispatched_id = Some(job.id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].unwrap()))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: now,
    })
}
