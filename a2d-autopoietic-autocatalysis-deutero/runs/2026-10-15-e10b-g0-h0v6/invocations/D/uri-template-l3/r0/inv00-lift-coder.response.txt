#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<char>,
    variables: Vec<(String, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    for index in (open + 1)..close {
        if bytes[index] == b'{' {
            return Err(Error::NestedOpen { pos: index });
        }
    }

    let mut content_start = open + 1;
    let operator = if content_start < close
        && matches!(
            bytes[content_start],
            b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'
        )
    {
        let op = bytes[content_start] as char;
        content_start += 1;
        Some(op)
    } else {
        None
    };

    if let Some(op) = operator {
        if !matches!(op, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
            return Err(Error::UnsupportedOperator { pos: open + 1 });
        }
    }

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&byte| byte == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..spec_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_byte(byte) {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let name = template[spec_start..spec_end].to_owned();
        if !seen.insert(name.clone()) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        variables.push((name, spec_start));

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        operator,
        variables,
    })
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            let high = bytes[index + 1];
            let low = bytes[index + 2];
            if high.is_ascii_hexdigit() && low.is_ascii_hexdigit() {
                output.push('%');
                output.push((high as char).to_ascii_uppercase());
                output.push((low as char).to_ascii_uppercase());
                index += 3;
                continue;
            }
        }

        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                    | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || reserved_byte {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0F) as usize] as char);
        }
        index += 1;
    }

    output
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let op = expression.operator;
    let reserved = matches!(op, Some('+' | '#'));
    let named = matches!(op, Some(';' | '?' | '&'));

    let mut fields = Vec::new();
    for (name, _) in &expression.variables {
        if let Some(value) = lookup(name, vars) {
            let encoded = encode(value, reserved);
            let field = if named {
                if op == Some(';') && encoded.is_empty() {
                    name.clone()
                } else {
                    format!("{name}={encoded}")
                }
            } else {
                encoded
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match op {
        Some('#') => ("#", ","),
        Some('.') => (".", "."),
        Some('/') => ("/", "/"),
        Some(';') => (";", ";"),
        Some('?') => ("?", "&"),
        Some('&') => ("&", "&"),
        _ => ("", ","),
    };

    format!("{prefix}{}", fields.join(separator))
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let close = bytes[index + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                    .map(|offset| index + 1 + offset)
                    .ok_or(Error::Unclosed { pos: index })?;
                expressions.push(parse_expression(template, index, close)?);
                index = close + 1;
            }
            _ => index += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;
    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);
        output.push_str(&expand_expression(&expression, vars));
        literal_start = expression.end;
    }
    output.push_str(&template[literal_start..]);
    Ok(output)
}
