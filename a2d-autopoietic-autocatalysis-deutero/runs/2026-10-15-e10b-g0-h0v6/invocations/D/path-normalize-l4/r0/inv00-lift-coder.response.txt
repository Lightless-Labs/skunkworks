use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|part| !part.is_empty())
            .last()
            .is_some_and(|part| part == "." || part == "..")
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect()
}

fn reduce_component_stack(
    stack: &mut Vec<String>,
    part: &str,
    absolute: bool,
    at: usize,
) -> Result<(), PathError> {
    match part {
        "." => {}
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_owned());
            }
        }
        _ => stack.push(part.to_owned()),
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, want_dir: bool) -> String {
    let mut result = if absolute {
        if stack.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", stack.join("/"))
        }
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if want_dir && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn normalize_parts(path: &str) -> Result<Vec<String>, PathError> {
    let absolute = path.starts_with('/');
    let mut stack = Vec::new();

    for (start, part) in path
        .split_inclusive('/')
        .scan(0usize, |offset, piece| {
            let start = *offset;
            *offset += piece.len();
            Some((start, piece.strip_suffix('/').unwrap_or(piece)))
        })
    {
        if !part.is_empty() {
            reduce_component_stack(&mut stack, part, absolute, start)?;
        }
    }

    Ok(stack)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let want_dir = directory_intent(path);
    let stack = normalize_parts(path)?;
    Ok(render(&stack, absolute, want_dir))
}

fn validate_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    let stack = components(&normalized);
    if stack.iter().any(|part| part == "..") {
        return Err(());
    }
    Ok(stack)
}

fn request_start<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(bool, Vec<String>, usize, bool), PathError> {
    let want_dir = directory_intent(path);
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        let remainder_start = if path == "~" {
            1
        } else if bytes.get(1) == Some(&b'/') {
            2
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        };

        let home = home.ok_or(PathError::MissingHome)?;
        Ok((
            true,
            components(home),
            remainder_start,
            want_dir,
        ))
    } else if path.starts_with('/') {
        Ok((true, Vec::new(), 0, want_dir))
    } else {
        unreachable!("relative request start is supplied by caller")
    }
}

fn resolve_core(path: &str, cwd: &[String], home: Option<&[String]>) -> Result<String, PathError> {
    validate_raw(path)?;

    let (absolute, mut stack, start, want_dir) = if path.starts_with('~') {
        let (absolute, stack, start, want_dir) = request_start(path, None)?;
        (absolute, stack, start, want_dir)
    } else if path.starts_with('/') {
        (true, Vec::new(), 0, directory_intent(path))
    } else {
        (true, cwd.to_vec(), 0, directory_intent(path))
    };

    let _ = absolute;
    let request_home = if path.starts_with('~') {
        Some(home.ok_or(PathError::MissingHome)?)
    } else {
        None
    };
    if let Some(home_stack) = request_home {
        stack = home_stack.to_vec();
    }

    let raw_remainder = &path[start..];
    let offset_base = start;
    let mut offset = offset_base;
    for piece in raw_remainder.split_inclusive('/') {
        let part = piece.strip_suffix('/').unwrap_or(piece);
        if !part.is_empty() {
            reduce_component_stack(&mut stack, part, true, offset)?;
        }
        offset += piece.len();
    }

    Ok(render(&stack, true, want_dir))
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_stack = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_stack = match home {
        Some(value) => Some(validate_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };
    resolve_core(path, &cwd_stack, home_stack.as_deref())
}

fn link_table(links: &[Link]) -> Result<HashMap<String, (usize, String)>, PathError> {
    let mut table = HashMap::new();

    for (index, link) in links.iter().enumerate() {
        if !link.path.starts_with('/')
            || link.path == "/"
            || link.path.ends_with('/')
            || normalize(&link.path).as_deref() != Ok(link.path.as_str())
            || validate_raw(&link.target).is_err()
        {
            return Err(PathError::InvalidLink { index });
        }

        if let Some((first, _)) = table.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        table.insert(link.path.clone(), (index, link.target.clone()));
    }

    Ok(table)
}

#[derive(Clone)]
struct Work {
    text: String,
    offset_base: usize,
    request: bool,
    target_link: Option<usize>,
}

fn stack_path(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn traverse_with_links(
    mut stack: Vec<String>,
    initial: Vec<Work>,
    table: &HashMap<String, (usize, String)>,
    max_expansions: usize,
) -> Result<Vec<String>, PathError> {
    let mut work = initial;
    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while let Some(mut current) = work.pop() {
        let mut offset = current.offset_base;
        let mut parts: Vec<(String, usize)> = Vec::new();
        for piece in current.text.split_inclusive('/') {
            let part = piece.strip_suffix('/').unwrap_or(piece);
            if !part.is_empty() {
                parts.push((part.to_owned(), offset));
            }
            offset += piece.len();
        }

        if let Some(link_index) = current.target_link.take() {
            // The target is active through its full recursive traversal. The
            // marker is placed below its components, so it is popped after them.
            work.push(Work {
                text: String::new(),
                offset_base: 0,
                request: false,
                target_link: Some(usize::MAX - link_index),
            });
        }

        for (part, at) in parts.into_iter().rev() {
            work.push(Work {
                text: part,
                offset_base: at,
                request: current.request,
                target_link: None,
            });
        }

        // Process one component at a time by converting it into a tiny raw path.
        let Some(item) = work.pop() else {
            continue;
        };
        if item.target_link.is_some() {
            if let Some(encoded) = item.target_link {
                if encoded >= usize::MAX / 2 {
                    let index = usize::MAX - encoded;
                    if active.last() == Some(&index) {
                        active.pop();
                    }
                    continue;
                }
            }
        }

        let part = item.text;
        if part.is_empty() {
            continue;
        }
        match part.as_str() {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape {
                        at: item.offset_base,
                    });
                }
            }
            _ => {
                stack.push(part);
                let matched = stack
                    .last()
                    .map(|_| stack_path(&stack))
                    .and_then(|key| table.get(&key).map(|(index, target)| (*index, target.clone())));

                if let Some((index, target)) = matched {
                    if active.contains(&index) {
                        return Err(PathError::Loop { index });
                    }
                    if expansions >= max_expansions {
                        return Err(PathError::Limit { index });
                    }
                    expansions += 1;
                    active.push(index);

                    // Relative targets begin at the link parent.
                    stack.pop();
                    let target_absolute = target.starts_with('/');
                    if target_absolute {
                        stack.clear();
                    }

                    let target_work = Work {
                        text: target,
                        offset_base: 0,
                        request: false,
                        target_link: Some(index),
                    };
                    // Push target components ahead of the remaining work.
                    let mut target_parts = Vec::new();
                    let mut target_offset = 0usize;
                    for piece in target_work.text.split_inclusive('/') {
                        let component = piece.strip_suffix('/').unwrap_or(piece);
                        if !component.is_empty() {
                            target_parts.push((component.to_owned(), target_offset));
                        }
                        target_offset += piece.len();
                    }
                    work.push(Work {
                        text: String::new(),
                        offset_base: 0,
                        request: false,
                        target_link: Some(index),
                    });
                    for (component, target_at) in target_parts.into_iter().rev() {
                        work.push(Work {
                            text: component,
                            offset_base: target_at,
                            request: false,
                            target_link: None,
                        });
                    }
                }
            }
        }
    }

    Ok(stack)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_stack = validate_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_stack = match home {
        Some(value) => Some(validate_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };
    let table = link_table(links)?;
    validate_raw(path)?;

    let (mut stack, start, want_dir) = if path.starts_with('~') {
        let bytes = path.as_bytes();
        let start = if path == "~" {
            1
        } else if bytes.get(1) == Some(&b'/') {
            2
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        };
        (
            home_stack.clone().ok_or(PathError::MissingHome)?,
            start,
            directory_intent(path),
        )
    } else if path.starts_with('/') {
        (Vec::new(), 0, directory_intent(path))
    } else {
        (cwd_stack.clone(), 0, directory_intent(path))
    };

    let remainder = &path[start..];
    let mut initial = Vec::new();
    let mut offset = start;
    for piece in remainder.split_inclusive('/') {
        let part = piece.strip_suffix('/').unwrap_or(piece);
        if !part.is_empty() {
            initial.push(Work {
                text: part.to_owned(),
                offset_base: offset,
                request: true,
                target_link: None,
            });
        }
        offset += piece.len();
    }

    // Traverse context components first, so links in cwd/home expand too.
    let context = if path.starts_with('/') {
        Vec::new()
    } else if path.starts_with('~') {
        home_stack.unwrap_or_default()
    } else {
        cwd_stack
    };
    stack.clear();

    let mut combined = Vec::new();
    for component in context {
        combined.push(Work {
            text: component,
            offset_base: 0,
            request: false,
            target_link: None,
        });
    }
    combined.extend(initial);

    let result = traverse_with_links(stack, combined, &table, max_expansions)?;
    Ok(render(&result, true, want_dir))
}

fn split_absolute(path: &str) -> Vec<String> {
    components(path)
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();

    let mut parts = vec!["..".to_owned(); base.len() - common];
    parts.extend_from_slice(&target[common..]);
    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;

    let base_parts = split_absolute(&resolved_base);
    let target_parts = split_absolute(&resolved_target);
    let target_is_dir = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        if let Ok(probe) = resolve_links(&link.path, "/", None, links, max_expansions) {
            let probe_parts = split_absolute(&probe);
            if target_parts.starts_with(&probe_parts) {
                let mut candidate = route(&base_parts, &components(&link.path));
                let suffix = &target_parts[probe_parts.len()..];
                if !suffix.is_empty() {
                    if candidate == "." {
                        candidate = suffix.join("/");
                    } else {
                        candidate.push('/');
                        candidate.push_str(&suffix.join("/"));
                    }
                }
                candidates.push(candidate);
            }
        }
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_is_dir && candidate != "." {
            candidate.push('/');
        }

        if let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if split_absolute(&resolved) == target_parts {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|a, b| {
        a.len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes()))
    });
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}
