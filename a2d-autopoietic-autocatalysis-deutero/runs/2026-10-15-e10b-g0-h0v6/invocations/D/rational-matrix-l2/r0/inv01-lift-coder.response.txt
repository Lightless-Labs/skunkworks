use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        let mut n = num;
        let mut d = den;
        if d < 0 {
            n = n.checked_neg().ok_or(RationalError::Overflow)?;
            d = d.checked_neg().ok_or(RationalError::Overflow)?;
        }

        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;

        let num = i64::try_from(n).map_err(|_| RationalError::Overflow)?;
        let den = i64::try_from(d).map_err(|_| RationalError::Overflow)?;
        Ok(Self { num, den })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let left_scale = rhs.den as i128 / g;
        let right_scale = self.den as i128 / g;
        let n = (self.num as i128)
            .checked_mul(left_scale)
            .and_then(|a| {
                (rhs.num as i128)
                    .checked_mul(right_scale)
                    .and_then(|b| a.checked_add(b))
            })
            .ok_or(RationalError::Overflow)?;
        let d = (self.den as i128)
            .checked_mul(left_scale)
            .ok_or(RationalError::Overflow)?;
        Self::from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let left_scale = rhs.den as i128 / g;
        let right_scale = self.den as i128 / g;
        let n = (self.num as i128)
            .checked_mul(left_scale)
            .and_then(|a| {
                (rhs.num as i128)
                    .checked_mul(right_scale)
                    .and_then(|b| a.checked_sub(b))
            })
            .ok_or(RationalError::Overflow)?;
        let d = (self.den as i128)
            .checked_mul(left_scale)
            .ok_or(RationalError::Overflow)?;
        Self::from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.den as u128) as i128;
        let g2 = gcd_u128(rhs.num.unsigned_abs() as u128, self.den as u128) as i128;
        let n = (self.num as i128 / g1)
            .checked_mul(rhs.num as i128 / g2)
            .ok_or(RationalError::Overflow)?;
        let d = (self.den as i128 / g2)
            .checked_mul(rhs.den as i128 / g1)
            .ok_or(RationalError::Overflow)?;
        Self::from_i128(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }

        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.num.unsigned_abs() as u128)
            as i128;
        let g2 = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let n = (self.num as i128 / g1)
            .checked_mul(rhs.den as i128 / g2)
            .ok_or(RationalError::Overflow)?;
        let d = (self.den as i128 / g2)
            .checked_mul(rhs.num as i128 / g1)
            .ok_or(RationalError::Overflow)?;
        Self::from_i128(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let num = self.num.checked_neg().ok_or(RationalError::Overflow)?;
        Ok(Self { num, den: self.den })
    }

    fn parse_error(at: usize, kind: ParseKind) -> RationalError {
        RationalError::Parse { at, kind }
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Self::parse_error(0, ParseKind::Empty));
        }

        let len = bytes.len();
        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
        }
        if i == len {
            return Err(Self::parse_error(i, ParseKind::InvalidByte));
        }
        if !bytes[i].is_ascii_digit() {
            return Err(Self::parse_error(i, ParseKind::InvalidByte));
        }
        while i < len && bytes[i].is_ascii_digit() {
            i += 1;
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;

        if i < len {
            if bytes[i] != b'/' {
                return Err(Self::parse_error(i, ParseKind::InvalidByte));
            }
            i += 1;
            den_start = Some(i);
            if i == len {
                return Err(Self::parse_error(i, ParseKind::InvalidByte));
            }
            if !bytes[i].is_ascii_digit() {
                return Err(Self::parse_error(i, ParseKind::InvalidByte));
            }
            while i < len && bytes[i].is_ascii_digit() {
                i += 1;
            }
            den_end = Some(i);
            if i < len {
                return Err(Self::parse_error(i, ParseKind::InvalidByte));
            }
        }

        let numerator_digits_start = usize::from(bytes[0] == b'-');
        let numerator_digits = &bytes[numerator_digits_start..num_end];

        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(Self::parse_error(
                numerator_digits_start + 1,
                ParseKind::NonCanonical,
            ));
        }
        if bytes[0] == b'-' && numerator_digits == b"0" {
            return Err(Self::parse_error(0, ParseKind::NonCanonical));
        }

        if let (Some(start), Some(end)) = (den_start, den_end) {
            let digits = &bytes[start..end];
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(Self::parse_error(start + 1, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let num_mag = parse_magnitude(
            numerator_digits,
            if negative {
                1u128 << 63
            } else {
                i64::MAX as u128
            },
        )
        .ok_or_else(|| Self::parse_error(0, ParseKind::OutOfRange))?;

        let den_mag = if let (Some(start), Some(end)) = (den_start, den_end) {
            Some(
                parse_magnitude(&bytes[start..end], i64::MAX as u128)
                    .ok_or_else(|| Self::parse_error(start, ParseKind::OutOfRange))?,
            )
        } else {
            None
        };

        if let Some(start) = den_start {
            if den_mag == Some(0) {
                return Err(Self::parse_error(start, ParseKind::ZeroDenominator));
            }
        }

        if let Some(start) = den_start {
            if den_mag == Some(1) || num_mag == 0 {
                return Err(Self::parse_error(start - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den_mag.unwrap()) > 1 {
                return Err(Self::parse_error(start - 1, ParseKind::NonCanonical));
            }
        }

        let n = if negative {
            if num_mag == 1u128 << 63 {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };

        match den_mag {
            Some(d) => Ok(Self {
                num: n,
                den: d as i64,
            }),
            None => Ok(Self { num: n, den: 1 }),
        }
    }
}

fn parse_magnitude(bytes: &[u8], limit: u128) -> Option<u128> {
    let mut value = 0u128;
    for &b in bytes {
        value = value.checked_mul(10)?.checked_add((b - b'0') as u128)?;
        if value > limit {
            return None;
        }
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|base| base.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data
            .get(index)
            .copied()
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }

        let output_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(output_len);
        let zero = Rational { num: 0, den: 1 };

        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }
}
