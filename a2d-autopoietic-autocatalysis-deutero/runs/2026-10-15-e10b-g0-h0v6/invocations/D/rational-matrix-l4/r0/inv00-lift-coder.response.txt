use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn signed_from_magnitude(mag: u128, negative: bool) -> Result<i64, RationalError> {
    if negative {
        if mag > (1u128 << 63) {
            return Err(RationalError::Overflow);
        }
        if mag == (1u128 << 63) {
            Ok(i64::MIN)
        } else {
            Ok(-(mag as i64))
        }
    } else if mag > i64::MAX as u128 {
        Err(RationalError::Overflow)
    } else {
        Ok(mag as i64)
    }
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let negative = (num < 0) ^ (den < 0);
        let nmag = num.unsigned_abs() as u128;
        let dmag = den.unsigned_abs() as u128;
        let g = gcd_u128(nmag, dmag);
        let nm = nmag / g;
        let dm = dmag / g;

        if dm > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        let n = signed_from_magnitude(nm, negative)?;
        Ok(Self {
            num: n,
            den: dm as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            + (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        rational_from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            - (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        rational_from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.num as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        rational_from_i128(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let n = (self.num as i128) * (rhs.den as i128);
        let d = (self.den as i128) * (rhs.num as i128);
        rational_from_i128(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        if b.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if b[0] == b'-' {
            i = 1;
        }
        let nstart = i;
        while i < b.len() && b[i].is_ascii_digit() {
            i += 1;
        }
        if i == nstart {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let nend = i;
        let mut dstart = None;
        let mut dend = None;
        if i < b.len() {
            if b[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            i += 1;
            dstart = Some(i);
            while i < b.len() && b[i].is_ascii_digit() {
                i += 1;
            }
            if i == dstart.unwrap() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            dend = Some(i);
        }
        if i < b.len() {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let neg = b[0] == b'-';
        let n_digits = nend - nstart;
        if n_digits >= 2 && b[nstart] == b'0' {
            return Err(parse_err(nstart + 1, ParseKind::NonCanonical));
        }
        if neg && n_digits == 1 && b[nstart] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }
        if let Some(ds) = dstart {
            if dend.unwrap() - ds >= 2 && b[ds] == b'0' {
                return Err(parse_err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let nmag = parse_magnitude(&b[nstart..nend]);
        let nlimit = if neg { 1u128 << 63 } else { i64::MAX as u128 };
        if nmag > nlimit {
            return Err(parse_err(0, ParseKind::OutOfRange));
        }

        if let Some(ds) = dstart {
            let de = dend.unwrap();
            let dmag = parse_magnitude(&b[ds..de]);
            if dmag > i64::MAX as u128 {
                return Err(parse_err(ds, ParseKind::OutOfRange));
            }
            if dmag == 0 {
                return Err(parse_err(ds, ParseKind::ZeroDenominator));
            }
            if dmag == 1 {
                return Err(parse_err(ds - 1, ParseKind::NonCanonical));
            }
            if nmag == 0 {
                return Err(parse_err(ds - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(nmag, dmag) > 1 {
                return Err(parse_err(ds - 1, ParseKind::NonCanonical));
            }
            let n = signed_from_magnitude(nmag, neg)
                .map_err(|_| parse_err(0, ParseKind::OutOfRange))?;
            return Ok(Self {
                num: n,
                den: dmag as i64,
            });
        }

        let n = signed_from_magnitude(nmag, neg)
            .map_err(|_| parse_err(0, ParseKind::OutOfRange))?;
        Ok(Self { num: n, den: 1 })
    }
}

fn parse_err(at: usize, kind: ParseKind) -> RationalError {
    RationalError::Parse { at, kind }
}

fn parse_magnitude(bytes: &[u8]) -> u128 {
    bytes.iter().fold(0u128, |acc, b| {
        acc.saturating_mul(10).saturating_add((b - b'0') as u128)
    })
}

fn rational_from_i128(n: i128, d: i128) -> Result<Rational, RationalError> {
    if d == 0 {
        return Err(RationalError::DivideByZero);
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let negative = (n < 0) ^ (d < 0);
    let nm = n.unsigned_abs();
    let dm = d.unsigned_abs();
    let g = gcd_u128(nm, dm);
    let nm = nm / g;
    let dm = dm / g;
    if dm > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }
    let num = signed_from_magnitude(nm, negative)?;
    Ok(Rational {
        num,
        den: dm as i64,
    })
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch {
        coefficients: usize,
        rhs: usize,
    },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let idx = row
            .checked_mul(self.cols)
            .and_then(|v| v.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data.get(idx).copied().ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let len = self.data.len();
        let mut data = vec![Rational { num: 0, den: 1 }; len];
        for r in 0..self.rows {
            for c in 0..self.cols {
                data[c * self.rows + r] = self.data[r * self.cols + c];
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let out_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(out_len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self.clone(), self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(if r == c {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }
        let augmented = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let red = eliminate(augmented, n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: red.pivots.len(),
            });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                out.push(red.matrix.data[r * width + n + c]);
            }
        }
        Ok(Matrix {
            rows: n,
            cols: n,
            data: out,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let total = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut data = Vec::with_capacity(total);
        for r in 0..self.rows {
            data.extend_from_slice(&self.data[r * n..r * n + n]);
            data.extend_from_slice(&rhs.data[r * k..r * k + k]);
        }
        let augmented = Matrix {
            rows: self.rows,
            cols: width,
            data,
        };
        let red = eliminate(augmented, n).map_err(SolveError::Matrix)?;

        for r in 0..self.rows {
            let mut coeff_zero = true;
            for c in 0..n {
                if red.matrix.data[r * width + c].num != 0 {
                    coeff_zero = false;
                    break;
                }
            }
            if coeff_zero {
                for c in 0..k {
                    let v = red.matrix.data[r * width + n + c];
                    if v.num != 0 {
                        return Ok(Solution::None {
                            row: r,
                            rhs_col: c,
                            witness: red.matrix.data[r * width..r * width + width].to_vec(),
                        });
                    }
                }
            }
        }

        let rank = red.pivots.len();
        let particular_len = n
            .checked_mul(k)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![Rational { num: 0, den: 1 }; particular_len];
        for (i, &pivot_col) in red.pivots.iter().enumerate() {
            for c in 0..k {
                particular_data[pivot_col * k + c] =
                    red.matrix.data[i * width + n + c];
            }
        }
        let particular = Matrix {
            rows: n,
            cols: k,
            data: particular_data,
        };

        let mut free = Vec::with_capacity(n.saturating_sub(rank));
        let mut pivot_seen = vec![false; n];
        for &p in &red.pivots {
            pivot_seen[p] = true;
        }
        for c in 0..n {
            if !pivot_seen[c] {
                free.push(c);
            }
        }

        if rank == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots: red.pivots,
            });
        }

        let f = free.len();
        let basis_len = n
            .checked_mul(f)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![Rational { num: 0, den: 1 }; basis_len];
        for (j, &free_col) in free.iter().enumerate() {
            basis_data[free_col * f + j] = Rational { num: 1, den: 1 };
            for (i, &pivot_col) in red.pivots.iter().enumerate() {
                let coeff = red.matrix.data[i * width + free_col];
                let value = coeff.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col,
                        col: j,
                        step: rank,
                        source,
                    })
                })?;
                basis_data[pivot_col * f + j] = value;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix {
                rows: n,
                cols: f,
                data: basis_data,
            },
            pivots: red.pivots,
        })
    }
}

fn eliminate(mut matrix: Matrix, pivot_limit: usize) -> Result<Reduction, MatrixError> {
    let rows = matrix.rows;
    let cols = matrix.cols;
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;

    for col in 0..pivot_limit {
        if p == rows {
            break;
        }
        let mut chosen: Option<usize> = None;
        for r in p..rows {
            let value = matrix.data[r * cols + col];
            if value.num == 0 {
                continue;
            }
            match chosen {
                None => chosen = Some(r),
                Some(q) => {
                    let current = matrix.data[q * cols + col];
                    let key = (value.den, value.num.unsigned_abs(), r);
                    let old = (current.den, current.num.unsigned_abs(), q);
                    if key < old {
                        chosen = Some(r);
                    }
                }
            }
        }
        let Some(q) = chosen else {
            continue;
        };

        if q != p {
            for c in 0..cols {
                matrix.data.swap(p * cols + c, q * cols + c);
            }
            swaps.push((p, q));
        }

        let pivot = matrix.data[p * cols + col];
        for c in 0..cols {
            let current = matrix.data[p * cols + c];
            matrix.data[p * cols + c] =
                current.div(&pivot).map_err(|source| MatrixError::Arithmetic {
                    row: p,
                    col: c,
                    step: p,
                    source,
                })?;
        }

        for r in 0..rows {
            if r == p {
                continue;
            }
            let factor = matrix.data[r * cols + col];
            if factor.num == 0 {
                continue;
            }
            for c in 0..cols {
                let product = factor.mul(&matrix.data[p * cols + c]).map_err(|source| {
                    MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: p,
                        source,
                    }
                })?;
                matrix.data[r * cols + c] =
                    matrix.data[r * cols + c]
                        .sub(&product)
                        .map_err(|source| MatrixError::Arithmetic {
                            row: r,
                            col: c,
                            step: p,
                            source,
                        })?;
            }
        }

        pivots.push(col);
        p += 1;
    }

    Ok(Reduction {
        matrix,
        pivots,
        swaps,
    })
}
