#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.split('/')
        .filter(|part| !part.is_empty())
        .last()
        .is_some_and(|part| part == "." || part == "..")
}

fn build_absolute(parts: &[String], intent: bool) -> String {
    if parts.is_empty() {
        return "/".to_string();
    }
    let mut result = format!("/{}", parts.join("/"));
    if intent {
        result.push('/');
    }
    result
}

fn reduce_parts(
    parts: &mut Vec<String>,
    raw: &str,
    base_offset: usize,
    absolute: bool,
) -> Result<(), PathError> {
    let mut offset = 0;
    for component in raw.split('/') {
        let component_offset = offset;
        offset += component.len() + 1;

        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if parts.last().is_some_and(|last| last != "..") {
                parts.pop();
            } else if absolute {
                return Err(PathError::Escape {
                    at: base_offset + component_offset,
                });
            } else {
                parts.push("..".to_string());
            }
        } else {
            parts.push(component.to_string());
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut parts = Vec::new();
    reduce_parts(&mut parts, path, 0, absolute)?;

    if absolute {
        Ok(build_absolute(&parts, intent))
    } else {
        let mut result = if parts.is_empty() {
            ".".to_string()
        } else {
            parts.join("/")
        };
        if intent && result != "." {
            result.push('/');
        }
        Ok(result)
    }
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(normalized
        .split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_string)
        .collect())
}

fn prepare_request<'a>(
    path: &'a str,
    home: Option<&'a [String]>,
) -> Result<(Vec<String>, &'a str, usize, bool), PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);

    let (base, request, offset) = if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            let base = home.ok_or(PathError::MissingHome)?.to_vec();
            let consumed = if bytes.len() > 1 { 2 } else { 1 };
            (base, &path[consumed..], consumed)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else {
        (Vec::new(), path, 0)
    };

    Ok((base, request, offset, intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = match home {
        Some(value) => Some(validate_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let intent = directory_intent(path);

    let (mut parts, request, offset) = if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            let base = home_parts.clone().ok_or(PathError::MissingHome)?;
            let consumed = if bytes.len() > 1 { 2 } else { 1 };
            (base, &path[consumed..], consumed)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        (Vec::new(), path, 0)
    } else {
        (cwd_parts, path, 0)
    };

    reduce_parts(&mut parts, request, offset, true)?;
    Ok(build_absolute(&parts, intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path)
            && link.path != "/";
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((_, first)) = seen.iter().find(|(path, _)| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }
    Ok(())
}

#[derive(Clone)]
struct Work {
    raw: String,
    offset: usize,
    pos: usize,
    origin: Origin,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Origin {
    Context,
    Request,
    Target,
}

fn next_component(work: &mut Work) -> Option<(String, usize)> {
    let bytes = work.raw.as_bytes();
    while work.pos < bytes.len() && bytes[work.pos] == b'/' {
        work.pos += 1;
    }
    if work.pos >= bytes.len() {
        return None;
    }
    let start = work.pos;
    while work.pos < bytes.len() && bytes[work.pos] != b'/' {
        work.pos += 1;
    }
    Some((
        work.raw[start..work.pos].to_string(),
        work.offset + start,
    ))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = match home {
        Some(value) => Some(validate_context(value).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };
    validate_links(links)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (mut stack, request, request_offset, request_absolute) = if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes.get(1) == Some(&b'/') {
            let base = home_parts.clone().ok_or(PathError::MissingHome)?;
            let consumed = if bytes.len() > 1 { 2 } else { 1 };
            (base, &path[consumed..], consumed, false)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        (Vec::new(), path, 0, true)
    } else {
        (cwd_parts, path, 0, false)
    };

    let mut work: Vec<Work> = Vec::new();
    if !request_absolute && !path.starts_with('~') {
        work.push(Work {
            raw: String::new(),
            offset: 0,
            pos: 0,
            origin: Origin::Context,
        });
        // Context is already normalized and is traversed component by component below.
        for part in stack.drain(..) {
            stack.push(part);
        }
    }

    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    // Context components need link lookup too. Replay the normalized cwd/home spelling.
    let context_raw = if !request_absolute && !path.starts_with('~') {
        Some(if path.starts_with('/') {
            String::new()
        } else {
            normalize(cwd)?
        })
    } else if path.starts_with('~') {
        Some(normalize(home.ok_or(PathError::MissingHome)?).map_err(|_| PathError::InvalidHome)?)
    } else {
        None
    };

    let mut pending = Vec::<Work>::new();
    if let Some(raw) = context_raw {
        let parts = raw
            .split('/')
            .filter(|s| !s.is_empty())
            .collect::<Vec<_>>();
        stack.clear();
        for part in parts {
            let candidate = {
                let mut p = stack.clone();
                p.push(part.to_string());
                format!("/{}", p.join("/"))
            };
            stack.push(part.to_string());
            if let Some((index, link)) = links.iter().enumerate().find(|(_, l)| l.path == candidate) {
                if active.contains(&index) {
                    return Err(PathError::Loop { index });
                }
                if expansions >= max_expansions {
                    return Err(PathError::Limit { index });
                }
                expansions += 1;
                active.push(index);
                pending.push(Work {
                    raw: link.target.clone(),
                    offset: 0,
                    pos: 0,
                    origin: Origin::Target,
                });
            }
        }
    }

    pending.push(Work {
        raw: request.to_string(),
        offset: request_offset,
        pos: 0,
        origin: Origin::Request,
    });

    while let Some(mut current) = pending.pop() {
        let mut resume = false;
        while let Some((component, at)) = next_component(&mut current) {
            if component == "." {
                continue;
            }
            if component == ".." {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
                continue;
            }

            stack.push(component);
            let candidate = format!("/{}", stack.join("/"));
            if let Some((index, link)) = links.iter().enumerate().find(|(_, l)| l.path == candidate) {
                if active.contains(&index) {
                    return Err(PathError::Loop { index });
                }
                if expansions >= max_expansions {
                    return Err(PathError::Limit { index });
                }
                expansions += 1;
                active.push(index);

                // The target is processed before the remainder of this work.
                let remainder = current.raw[current.pos..].to_string();
                let remainder_offset = current.offset + current.pos;
                let target = Work {
                    raw: link.target.clone(),
                    offset: 0,
                    pos: 0,
                    origin: Origin::Target,
                };
                pending.push(Work {
                    raw: remainder,
                    offset: remainder_offset,
                    pos: 0,
                    origin: current.origin,
                });
                pending.push(target);
                resume = true;
                break;
            }
        }
        if !resume {
            if current.origin == Origin::Target {
                if let Some(index) = active.pop() {
                    let _ = index;
                }
            }
        }
    }

    Ok(build_absolute(&stack, intent))
}
