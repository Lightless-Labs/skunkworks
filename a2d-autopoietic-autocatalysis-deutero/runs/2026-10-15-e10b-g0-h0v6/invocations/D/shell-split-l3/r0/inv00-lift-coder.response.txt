#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

struct Builder {
    text: String,
    glob: Vec<usize>,
    active: bool,
}

impl Builder {
    fn new() -> Self {
        Self {
            text: String::new(),
            glob: Vec::new(),
            active: false,
        }
    }

    fn push(&mut self, ch: char, wildcard_active: bool) {
        if wildcard_active && matches!(ch, '*' | '?' | '[') {
            self.glob.push(self.text.len());
        }
        self.text.push(ch);
        self.active = true;
    }

    fn push_str(&mut self, text: &str, wildcard_active: bool) {
        if wildcard_active {
            for (offset, ch) in text.char_indices() {
                if matches!(ch, '*' | '?' | '[') {
                    self.glob.push(self.text.len() + offset);
                }
            }
        }
        self.text.push_str(text);
        self.active = true;
    }

    fn finish(&mut self, words: &mut Vec<Word>) {
        if self.active {
            words.push(Word {
                text: std::mem::take(&mut self.text),
                glob: std::mem::take(&mut self.glob),
            });
            self.active = false;
        }
    }
}

fn is_name_start(ch: char) -> bool {
    ch.is_ascii_alphabetic() || ch == '_'
}

fn is_name_continue(ch: char) -> bool {
    ch.is_ascii_alphanumeric() || ch == '_'
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, ch)) = chars.next() {
        if ch == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(ch);
        }
    }

    result
}

fn parse_expansion(
    input: &str,
    dollar_pos: usize,
    after_dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let mut end = after_dollar;
    let mut closing = None;

    while end < input.len() {
        let ch = input[end..].chars().next().unwrap();
        if ch == '\\' {
            end += ch.len_utf8();
            if end < input.len() {
                let next = input[end..].chars().next().unwrap();
                end += next.len_utf8();
            }
        } else if ch == '}' {
            closing = Some(end);
            break;
        } else {
            end += ch.len_utf8();
        }
    }

    let close_pos = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar_pos })?;
    let contents = &input[after_dollar..close_pos];

    let name_end = contents
        .char_indices()
        .find_map(|(i, ch)| (!is_name_continue(ch)).then_some(i))
        .unwrap_or(contents.len());

    let name = &contents[..name_end];
    let valid_name = !name.is_empty()
        && name.chars().next().is_some_and(is_name_start)
        && name.chars().all(is_name_continue);

    if !valid_name {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let remainder = &contents[name_end..];
    let value = if remainder.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default) = remainder.strip_prefix(":-") {
        match lookup(name, env) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    Ok((close_pos + 1, value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Builder::new();
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let ch = input[i..].chars().next().unwrap();
        let next_i = i + ch.len_utf8();

        match quote {
            Quote::Single(_) => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch, false);
                }
                i = next_i;
            }
            Quote::Double(open_pos) => {
                if ch == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if ch == '\\' {
                    if next_i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        word.push(next, false);
                        i = after_next;
                    } else if next == '\n' {
                        i = after_next;
                    } else {
                        word.push('\\', false);
                        i = next_i;
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if ch == '$' {
                    if next_i < input.len() {
                        let next = input[next_i..].chars().next().unwrap();
                        if next == '(' {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                        if next == '{' {
                            let (after, value) =
                                parse_expansion(input, i, next_i + next.len_utf8(), env)?;
                            word.push_str(&value, false);
                            i = after;
                            continue;
                        }
                        if is_name_start(next) {
                            let mut end = next_i;
                            while end < input.len() {
                                let scalar = input[end..].chars().next().unwrap();
                                if !is_name_continue(scalar) {
                                    break;
                                }
                                end += scalar.len_utf8();
                            }
                            let value = lookup(&input[next_i..end], env).unwrap_or("");
                            word.push_str(value, false);
                            i = end;
                            continue;
                        }
                    }
                    word.push('$', false);
                    i = next_i;
                } else {
                    word.push(ch, false);
                    i = next_i;
                }

                if i == input.len() && quote == Quote::Double(open_pos) {
                    break;
                }
            }
            Quote::None => {
                if matches!(ch, ' ' | '\t' | '\r' | '\n') {
                    word.finish(&mut words);
                    i = next_i;
                } else if ch == '\'' {
                    word.active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if ch == '"' {
                    word.active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if ch == '\\' {
                    if next_i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        word.push(next, false);
                        i = after_next;
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if ch == '$' {
                    if next_i < input.len() {
                        let next = input[next_i..].chars().next().unwrap();
                        if next == '(' {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                        if next == '{' {
                            let (after, value) =
                                parse_expansion(input, i, next_i + next.len_utf8(), env)?;
                            word.push_str(&value, true);
                            i = after;
                            continue;
                        }
                        if is_name_start(next) {
                            let mut end = next_i;
                            while end < input.len() {
                                let scalar = input[end..].chars().next().unwrap();
                                if !is_name_continue(scalar) {
                                    break;
                                }
                                end += scalar.len_utf8();
                            }
                            let value = lookup(&input[next_i..end], env).unwrap_or("");
                            word.push_str(value, true);
                            i = end;
                            continue;
                        }
                    }
                    word.push('$', true);
                    i = next_i;
                } else {
                    word.push(ch, true);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    word.finish(&mut words);
    Ok(words)
}
