fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort_by_key(|c| constraint_rank(*c));
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    constraints
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for &b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn sorted_evidence(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut result = evidence.to_vec();
    result.sort_by(|a, b| {
        constraint_rank(a.constraint)
            .cmp(&constraint_rank(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    result
}

fn evidence_lines(evidence: &[Evidence]) -> String {
    let mut out = String::new();
    for item in sorted_evidence(evidence) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(item.constraint));
        out.push_str(" via ");
        out.push_str(&item.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            format!("missing|package={}{}", encode(package), evidence_lines(evidence))
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let constraints = canonical_constraints(requirements.clone());
            format!(
                "no-match|package={}|constraints={}{}",
                encode(package),
                constraints_text(&constraints),
                evidence_lines(evidence)
            )
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let constraints = canonical_constraints(requirements.clone());
            format!(
                "conflict|package={}|selected={}|constraints={}{}",
                encode(package),
                version_text(*selected),
                constraints_text(&constraints),
                evidence_lines(evidence)
            )
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn permits(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(x) => version == x,
        Constraint::AtLeast(x) => version >= x,
        Constraint::Before(x) => version < x,
        Constraint::Caret(x) => {
            version >= x
                && if x.major > 0 {
                    version.major == x.major
                } else if x.minor > 0 {
                    version.major == 0 && version.minor == x.minor
                } else {
                    version == x
                }
        }
    }
}

fn platform_matches(platform: Platform, target: Target) -> bool {
    platform == Platform::Any
        || matches!(
            (platform, target),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn platform_text(platform: Platform) -> &'static str {
    match platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn sorted_features(features: &[String]) -> Vec<String> {
    let mut result = features.to_vec();
    result.sort();
    result.dedup();
    result
}

#[derive(Clone, Debug, PartialEq, Eq, Default)]
struct Active {
    origins: Vec<Evidence>,
    features: BTreeSet<String>,
}

struct Search<'a> {
    catalog: &'a [Package],
    package_indices: BTreeMap<String, usize>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>)>,
    first_error: Option<ResolveError>,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn root_token(index: usize, root: &Requirement) -> String {
        format!(
            "root[{}]:{}@{}+{{{}}}",
            index,
            root.package,
            constraint_text(root.constraint),
            sorted_features(&root.features).join(",")
        )
    }

    fn closure(&self, assigned: &BTreeMap<String, Version>) -> BTreeMap<String, Active> {
        let mut active = BTreeMap::<String, Active>::new();

        for (i, root) in self.roots.iter().enumerate() {
            active
                .entry(root.package.clone())
                .or_default()
                .origins
                .push(Evidence {
                    constraint: root.constraint,
                    path: vec![Self::root_token(i, root)],
                });
            active
                .entry(root.package.clone())
                .or_default()
                .features
                .extend(root.features.iter().cloned());
        }

        loop {
            let before = active.clone();
            let names: Vec<String> = active.keys().cloned().collect();

            for name in names {
                let Some(version) = assigned.get(&name).copied() else {
                    continue;
                };
                let Some(&package_index) = self.package_indices.get(&name) else {
                    continue;
                };
                let Some(release) = self.catalog[package_index]
                    .releases
                    .iter()
                    .find(|release| release.version == version)
                else {
                    continue;
                };

                let mut activated = active[&name].features.clone();
                loop {
                    let old_len = activated.len();
                    let current: Vec<String> = activated.iter().cloned().collect();
                    for feature_name in current {
                        if let Some(declaration) = release
                            .features
                            .iter()
                            .find(|declaration| declaration.name == feature_name)
                        {
                            activated.extend(declaration.enables.iter().cloned());
                        }
                    }
                    if activated.len() == old_len {
                        break;
                    }
                }
                active
                    .entry(name.clone())
                    .or_default()
                    .features
                    .extend(activated.iter().cloned());

                let paths = canonical_paths(&active, self.roots);
                for (dependency_index, dependency) in release.deps.iter().enumerate() {
                    if !platform_matches(dependency.platform, self.target)
                        || dependency
                            .gate
                            .as_ref()
                            .is_some_and(|gate| !activated.contains(gate))
                    {
                        continue;
                    }

                    let token = format!(
                        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        name,
                        version_text(version),
                        dependency_index,
                        dependency.package,
                        constraint_text(dependency.constraint),
                        sorted_features(&dependency.features).join(","),
                        dependency.gate.as_deref().unwrap_or("*"),
                        platform_text(dependency.platform)
                    );
                    let mut path = paths.get(&name).cloned().unwrap_or_default();
                    path.push(token);

                    let destination = active.entry(dependency.package.clone()).or_default();
                    destination.origins.push(Evidence {
                        constraint: dependency.constraint,
                        path,
                    });
                    destination
                        .features
                        .extend(dependency.features.iter().cloned());
                }
            }

            if active == before {
                break;
            }
        }

        active
    }

    fn inspect(
        &mut self,
        assigned: &BTreeMap<String, Version>,
    ) -> Result<(), ResolveError> {
        let active = self.closure(assigned);

        for name in active.keys() {
            if !self.package_indices.contains_key(name) {
                return Err(ResolveError::Missing {
                    package: name.clone(),
                    evidence: active[name].origins.clone(),
                });
            }
        }

        let paths = canonical_paths(&active, self.roots);
        for (name, version) in assigned {
            let Some(package_active) = active.get(name) else {
                continue;
            };
            let Some(&package_index) = self.package_indices.get(name) else {
                continue;
            };
            let release = self.catalog[package_index]
                .releases
                .iter()
                .find(|release| release.version == *version)
                .expect("assigned versions come from the catalog");

            for feature in &package_active.features {
                if !release
                    .features
                    .iter()
                    .any(|declaration| declaration.name == *feature)
                {
                    return Err(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: paths.get(name).cloned().unwrap_or_default(),
                    });
                }
            }
        }

        for (name, version) in assigned {
            let Some(package_active) = active.get(name) else {
                continue;
            };
            let requirements = canonical_constraints(
                package_active
                    .origins
                    .iter()
                    .map(|origin| origin.constraint)
                    .collect(),
            );
            if !requirements
                .iter()
                .all(|constraint| permits(*constraint, *version))
            {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements,
                    evidence: package_active.origins.clone(),
                });
            }
        }

        for (name, package_active) in &active {
            if assigned.contains_key(name) {
                continue;
            }
            let requirements = canonical_constraints(
                package_active
                    .origins
                    .iter()
                    .map(|origin| origin.constraint)
                    .collect(),
            );
            let package_index = self.package_indices[name];
            if !self.catalog[package_index].releases.iter().any(|release| {
                requirements
                    .iter()
                    .all(|constraint| permits(*constraint, release.version))
            }) {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements,
                    evidence: package_active.origins.clone(),
                });
            }
        }

        let Some(name) = active.keys().find(|name| !assigned.contains_key(*name)).cloned()
        else {
            let selected = assigned
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            self.consider(selected);
            return Ok(());
        };

        let package_active = &active[&name];
        let requirements = canonical_constraints(
            package_active
                .origins
                .iter()
                .map(|origin| origin.constraint)
                .collect(),
        );
        let package_index = self.package_indices[&name];
        let mut candidates: Vec<Version> = self.catalog[package_index]
            .releases
            .iter()
            .map(|release| release.version)
            .filter(|version| {
                requirements
                    .iter()
                    .all(|constraint| permits(*constraint, *version))
            })
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.get(&name) {
            if let Some(index) = candidates.iter().position(|version| version == lock) {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;

            let mut child = assigned.clone();
            child.insert(name.clone(), version);
            match self.inspect(&child) {
                Ok(()) => {}
                Err(error @ ResolveError::Limit { .. }) => return Err(error),
                Err(error) => {
                    if self.first_error.is_none() {
                        self.first_error = Some(error);
                    }
                }
            }
        }

        Ok(())
    }

    fn consider(&mut self, solution: Vec<Selection>) {
        let versions: BTreeMap<String, Version> = solution
            .iter()
            .map(|selection| (selection.package.clone(), selection.version))
            .collect();
        let cost = self
            .locks
            .iter()
            .filter(|(name, version)| versions.get(*name) != Some(*version))
            .count();

        let replace = match &self.best {
            None => true,
            Some((best_cost, best_solution)) if cost < *best_cost => true,
            Some((best_cost, best_solution)) if cost > *best_cost => false,
            Some((_, best_solution)) => {
                let best_versions: BTreeMap<String, Version> = best_solution
                    .iter()
                    .map(|selection| (selection.package.clone(), selection.version))
                    .collect();
                let mut ordering = std::cmp::Ordering::Equal;
                for name in &self.catalog_names {
                    let current = versions.get(name);
                    let best = best_versions.get(name);
                    ordering = match (current, best) {
                        (Some(a), Some(b)) => a.cmp(b),
                        (Some(_), None) => std::cmp::Ordering::Greater,
                        (None, Some(_)) => std::cmp::Ordering::Less,
                        (None, None) => std::cmp::Ordering::Equal,
                    };
                    if ordering != std::cmp::Ordering::Equal {
                        break;
                    }
                }
                ordering == std::cmp::Ordering::Greater
            }
        };

        if replace {
            self.best = Some((cost, solution));
        }
    }
}

fn path_better(candidate: &[String], current: &[String]) -> bool {
    candidate.len() < current.len()
        || (candidate.len() == current.len() && candidate < current)
}

fn canonical_paths(
    active: &BTreeMap<String, Active>,
    roots: &[Requirement],
) -> BTreeMap<String, Vec<String>> {
    let mut best = BTreeMap::<String, Vec<String>>::new();

    for (index, root) in roots.iter().enumerate() {
        let path = vec![Search::root_token(index, root)];
        match best.get(&root.package) {
            Some(old) if !path_better(&path, old) => {}
            _ => {
                best.insert(root.package.clone(), path);
            }
        }
    }

    loop {
        let mut changed = false;
        for (destination, package_active) in active {
            for origin in &package_active.origins {
                if origin.path.len() <= 1 {
                    continue;
                }
                match best.get(destination) {
                    Some(old) if !path_better(&origin.path, old) => {}
                    _ => {
                        best.insert(destination.clone(), origin.path.clone());
                        changed = true;
                    }
                }
            }
        }
        if !changed {
            break;
        }
    }

    best
}

fn invalid_name(site: String, name: &str) -> ResolveError {
    ResolveError::InvalidName {
        site,
        name: name.to_string(),
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(invalid_name(format!("root[{}]", i), &root.package));
        }
        for (feature_index, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(invalid_name(
                    format!("root[{}].feature[{}]", i, feature_index),
                    feature,
                ));
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(invalid_name(format!("catalog[{}]", i), &package.name));
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dependency) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dependency.package) {
                    return Err(invalid_name(base.clone(), &dependency.package));
                }
                if let Some(gate) = &dependency.gate {
                    if !valid_name(gate) {
                        return Err(invalid_name(format!("{}.gate", base), gate));
                    }
                }
                for (feature_index, feature) in dependency.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(invalid_name(
                            format!("{}.feature[{}]", base, feature_index),
                            feature,
                        ));
                    }
                }
            }
            for (feature_index, feature) in release.features.iter().enumerate() {
                let base = format!(
                    "catalog[{}].release[{}].feature[{}]",
                    i, j, feature_index
                );
                if !valid_name(&feature.name) {
                    return Err(invalid_name(base.clone(), &feature.name));
                }
                for (enable_index, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(invalid_name(
                            format!("{}.enable[{}]", base, enable_index),
                            enabled,
                        ));
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen_features = BTreeSet::new();
            for feature in &release.features {
                if !seen_features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(invalid_name(format!("lock[{}]", i), &lock.package));
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let package_indices: BTreeMap<String, usize> = catalog
        .iter()
        .enumerate()
        .map(|(index, package)| (package.name.clone(), index))
        .collect();
    let locks_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();
    let catalog_names: Vec<String> = seen_packages.into_iter().collect();

    let mut search = Search {
        catalog,
        package_indices,
        roots,
        locks: locks_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        catalog_names,
    };

    let root_result = search.inspect(&BTreeMap::new());
    if let Err(error @ ResolveError::Limit { .. }) = root_result {
        return Err(error);
    }

    let Some((_, selected)) = search.best.clone() else {
        if let Some(error) = search.first_error {
            return Err(error);
        }
        if let Err(error) = root_result {
            return Err(error);
        }
        return Ok(Resolution {
            selected: Vec::new(),
            changes: Vec::new(),
        });
    };

    let selected_versions: BTreeMap<String, Version> = selected
        .iter()
        .map(|selection| (selection.package.clone(), selection.version))
        .collect();
    let final_active = search.closure(&selected_versions);

    let mut changed_names = BTreeSet::new();
    changed_names.extend(search.locks.keys().cloned());
    changed_names.extend(selected_versions.keys().cloned());

    let mut changes = Vec::new();
    for name in changed_names {
        let before = search.locks.get(&name).copied();
        let after = selected_versions.get(&name).copied();
        if before == after {
            continue;
        }

        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .package_indices
                    .get(&name)
                    .is_some_and(|&index| {
                        search.catalog[index]
                            .releases
                            .iter()
                            .any(|release| release.version == old)
                    });
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let constraints = final_active
                        .get(&name)
                        .map(|active| {
                            active
                                .origins
                                .iter()
                                .map(|origin| origin.constraint)
                                .collect::<Vec<_>>()
                        })
                        .unwrap_or_default();
                    if !constraints.iter().all(|constraint| permits(*constraint, old)) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => continue,
        };

        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }
    changes.sort_by(|a, b| a.package.cmp(&b.package));

    Ok(Resolution { selected, changes })
}
