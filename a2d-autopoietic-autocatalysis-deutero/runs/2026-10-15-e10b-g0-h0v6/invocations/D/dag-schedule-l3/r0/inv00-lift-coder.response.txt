```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventKind {
    Start,
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub task: usize,
    pub worker: usize,
    pub kind: EventKind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub events: Vec<Event>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before checking any other condition.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    // Kahn traversal with the smallest currently available ID first.
    let mut available = vec![false; n];
    for id in 0..n {
        available[id] = indegree[id] == 0;
    }
    let mut order = Vec::with_capacity(n);
    for _ in 0..n {
        let next = match (0..n).find(|&id| available[id]) {
            Some(id) => id,
            None => break,
        };
        available[next] = false;
        order.push(next);
        for &child in &children[next] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available[child] = true;
            }
        }
    }

    if order.len() != n {
        let removed: Vec<bool> = {
            let mut v = vec![false; n];
            for &id in &order {
                v[id] = true;
            }
            v
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Bottom levels, computed in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_successor_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_successor_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_available_at = vec![0u64; workers];
    let mut running: Vec<(usize, usize, u64)> = Vec::new(); // task, worker, finish
    let mut events = Vec::new();
    let mut time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all running tasks at this time in ascending task ID order.
        let mut finishing: Vec<(usize, usize)> = running
            .iter()
            .filter(|(_, _, finish)| *finish == time)
            .map(|&(task, worker, _)| (task, worker))
            .collect();
        finishing.sort_unstable_by_key(|&(task, _)| task);

        for &(task, worker) in &finishing {
            completed[task] = true;
            completed_count += 1;
            makespan = makespan.max(time);
            events.push(Event {
                time,
                task,
                worker,
                kind: EventKind::Finish,
            });
        }
        if !finishing.is_empty() {
            running.retain(|(_, _, finish)| *finish != time);
        }

        loop {
            let worker = (0..workers).find(|&w| worker_available_at[w] <= time);
            let Some(worker) = worker else {
                break;
            };

            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > time {
                    continue;
                }
                if !tasks[id].deps.iter().all(|&dep| completed[dep]) {
                    continue;
                }

                let is_better = match best {
                    None => true,
                    Some(current) => {
                        tails[id] > tails[current]
                            || (tails[id] == tails[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority
                                        && id < current)))
                    }
                };
                if is_better {
                    best = Some(id);
                }
            }

            let Some(task) = best else {
                break;
            };

            let finish = time.checked_add(tasks[task].duration).ok_or(Error::Overflow {
                task,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task,
                stage: Stage::Turnaround,
            })?;

            started[task] = true;
            worker_available_at[worker] = available_at;
            runs[task] = Some(Run {
                start: time,
                finish,
                worker,
            });
            events.push(Event {
                time,
                task,
                worker,
                kind: EventKind::Start,
            });

            if tasks[task].duration == 0 {
                completed[task] = true;
                completed_count += 1;
                makespan = makespan.max(time);
                events.push(Event {
                    time,
                    task,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                running.push((task, worker, finish));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest strictly later event that can change eligibility.
        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > time {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };

        for &(_, _, finish) in &running {
            consider(finish);
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > time {
                consider(tasks[id].release);
            }
        }
        for &available_at in &worker_available_at {
            consider(available_at);
        }

        match next_time {
            Some(next) => time = next,
            None => break,
        }
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("all tasks completed"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}
```
