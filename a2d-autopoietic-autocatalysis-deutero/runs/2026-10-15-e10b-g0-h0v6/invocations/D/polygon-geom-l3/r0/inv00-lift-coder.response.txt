#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

const COORD_LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;
const MAX_QUERIES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn cross(a: Point, b: Point, q: Point) -> i128 {
    let ab_x = b.x as i128 - a.x as i128;
    let ab_y = b.y as i128 - a.y as i128;
    let aq_x = q.x as i128 - a.x as i128;
    let aq_y = q.y as i128 - a.y as i128;
    ab_x * aq_y - ab_y * aq_x
}

fn shoelace(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn rounded_length_milli(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let d2 = (dx * dx + dy * dy) as u128;
    let scaled = d2 * 1_000_000u128;

    // Coordinates are bounded by 1e9, so scaled is at most 8e24.
    // Binary search the integer square root without floating point.
    let mut lo = 0u128;
    let mut hi = 3_000_000_000_000u128;
    while lo < hi {
        let mid = lo + (hi - lo + 1) / 2;
        if mid * mid <= scaled {
            lo = mid;
        } else {
            hi = mid - 1;
        }
    }

    // Round to nearest, with exact half ties upward. Compare
    // scaled against (lo + 1/2)^2 without fractions.
    let twice_scaled = scaled * 4;
    let midpoint_squared = (2 * lo + 1) * (2 * lo + 1);
    if twice_scaled >= midpoint_squared {
        (lo + 1) as u64
    } else {
        lo as u64
    }
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective[effective.len() - 1] == effective[0] {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn measures_for_valid_polygon(points: &[Point]) -> Measures {
    let signed_area2 = shoelace(points);
    let area2 = signed_area2.unsigned_abs();
    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    let mut edge_milli = Vec::with_capacity(points.len());
    let mut perimeter_milli = 0u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..points.len() {
        let j = (i + 1) % points.len();
        let length = rounded_length_milli(points[i], points[j]);
        edge_milli.push(length);
        perimeter_milli += length;

        let replace = match longest_edge {
            None => true,
            Some(best) => {
                if length > edge_milli[best] {
                    true
                } else if length < edge_milli[best] {
                    false
                } else {
                    let endpoints = if points[i] <= points[j] {
                        (points[i], points[j])
                    } else {
                        (points[j], points[i])
                    };
                    let best_next = (best + 1) % points.len();
                    let best_endpoints = if points[best] <= points[best_next] {
                        (points[best], points[best_next])
                    } else {
                        (points[best_next], points[best])
                    };
                    endpoints < best_endpoints
                        || (endpoints == best_endpoints && i < best)
                }
            }
        };
        if replace {
            longest_edge = Some(i);
        }
    }

    Measures {
        signed_area2,
        area2,
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let effective = prepare_polygon(points)?;
    Ok(measures_for_valid_polygon(&effective))
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;

    if queries.len() > MAX_QUERIES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut result = Vec::with_capacity(queries.len());
    for &query in queries {
        let mut boundary = None;
        let mut inside = false;

        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];

            if on_segment(a, b, query) {
                boundary = Some(i);
                break;
            }

            if (a.y > query.y) != (b.y > query.y) {
                let side = cross(a, b, query);
                if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                    inside = !inside;
                }
            }
        }

        result.push(match boundary {
            Some(edge) => Location::Boundary { edge },
            None if inside => Location::Inside,
            None => Location::Outside,
        });
    }

    Ok(result)
}

fn cloud_hull(points: &[Point], mode: Collinear) -> (Vec<Point>, Vec<usize>) {
    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|&(existing, _)| existing == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);

    if unique.len() <= 1 {
        return unique.into_iter().unzip();
    }

    let all_collinear = unique
        .iter()
        .skip(2)
        .all(|&(point, _)| cross(unique[0].0, unique[1].0, point) == 0);

    if all_collinear {
        return match mode {
            Collinear::Include => unique.into_iter().unzip(),
            Collinear::Exclude => {
                let first = unique[0];
                let last = unique[unique.len() - 1];
                vec![first, last].into_iter().unzip()
            }
        };
    }

    let mut lower: Vec<(Point, usize)> = Vec::new();
    for &item in &unique {
        while lower.len() >= 2 {
            let n = lower.len();
            let turn = cross(lower[n - 2].0, lower[n - 1].0, item.0);
            if turn > 0 || (mode == Collinear::Include && turn == 0) {
                break;
            }
            lower.pop();
        }
        lower.push(item);
    }

    let mut upper: Vec<(Point, usize)> = Vec::new();
    for &item in unique.iter().rev() {
        while upper.len() >= 2 {
            let n = upper.len();
            let turn = cross(upper[n - 2].0, upper[n - 1].0, item.0);
            if turn > 0 || (mode == Collinear::Include && turn == 0) {
                break;
            }
            upper.pop();
        }
        upper.push(item);
    }

    lower.pop();
    upper.pop();
    lower.extend(upper);

    let (vertices, indices): (Vec<_>, Vec<_>) = lower.into_iter().unzip();
    (vertices, indices)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let (vertices, source_indices) = cloud_hull(points, mode);
    let signed_area2 = if vertices.len() >= 3 {
        shoelace(&vertices)
    } else {
        0
    };
    let area2 = signed_area2.unsigned_abs();

    let mut perimeter_milli = 0u64;
    if vertices.len() >= 2 {
        for i in 0..vertices.len() {
            perimeter_milli += rounded_length_milli(vertices[i], vertices[(i + 1) % vertices.len()]);
        }
    }

    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}
