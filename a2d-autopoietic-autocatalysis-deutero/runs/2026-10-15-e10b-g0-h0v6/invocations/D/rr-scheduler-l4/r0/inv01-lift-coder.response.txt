#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Process {
    id: u32,
    arrival: u64,
    bursts: Vec<u32>,
    burst_position: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
    cpu_total: u64,
    io_total: u64,
}

#[derive(Clone, Copy, Debug)]
enum EventKind {
    Arrival,
    Wake { job: usize, level: usize },
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind_order: u8,
    input_index: usize,
    kind: EventKind,
}

fn rounded_mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        0
    } else {
        (total * 100 + count as u64 / 2) / count as u64
    }
}

fn admit_events(
    events: &[Event],
    event_cursor: &mut usize,
    time: u64,
    processes: &mut [Process],
    queues: &mut [std::collections::VecDeque<usize>],
) {
    while *event_cursor < events.len() && events[*event_cursor].time <= time {
        let event = events[*event_cursor];
        match event.kind {
            EventKind::Arrival => {
                let job = event.input_index;
                queues[processes[job].level].push_back(job);
            }
            EventKind::Wake { job, level } => {
                processes[job].level = level;
                queues[level].push_back(job);
            }
        }
        *event_cursor += 1;
    }
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|previous| previous.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut processes: Vec<Process> = jobs
        .iter()
        .map(|job| Process {
            id: job.id,
            arrival: job.arrival as u64,
            bursts: job.bursts.clone(),
            burst_position: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
            cpu_total: job
                .bursts
                .iter()
                .step_by(2)
                .map(|&value| value as u64)
                .sum(),
            io_total: job
                .bursts
                .iter()
                .skip(1)
                .step_by(2)
                .map(|&value| value as u64)
                .sum(),
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: job.arrival as u64,
            kind_order: 0,
            input_index: index,
            kind: EventKind::Arrival,
        })
        .collect();
    events.sort_by_key(|event| (event.time, event.kind_order, event.input_index));

    let mut event_cursor = 0usize;
    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut runs: Vec<Run> = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut time = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut completed_count = 0usize;

    while completed_count < jobs.len() {
        admit_events(
            &events,
            &mut event_cursor,
            time,
            &mut processes,
            &mut queues,
        );

        let Some(level) = queues.iter().position(|queue| !queue.is_empty()) else {
            if event_cursor >= events.len() {
                break;
            }

            let next_time = events[event_cursor].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let job_index = queues[level].pop_front().unwrap();
        let selected_id = processes[job_index].id;

        if last_dispatch_id.is_some_and(|last_id| last_id != selected_id) {
            time += switch_cost as u64;
        }

        admit_events(
            &events,
            &mut event_cursor,
            time,
            &mut processes,
            &mut queues,
        );

        let run_level = processes[job_index].level;
        let consumed = processes[job_index]
            .remaining
            .min(quanta[run_level]);
        let start = time;
        time += consumed as u64;
        let end = time;
        processes[job_index].remaining -= consumed;

        let burst_finished = processes[job_index].remaining == 0;
        let job_finished = burst_finished
            && processes[job_index].burst_position + 1
                == processes[job_index].bursts.len() - 1;

        runs.push(Run {
            id: selected_id,
            start,
            end,
            finished: job_finished,
            level: run_level,
        });

        admit_events(
            &events,
            &mut event_cursor,
            end,
            &mut processes,
            &mut queues,
        );

        if job_finished {
            processes[job_index].completion = Some(end);
            completed_count += 1;
        } else if burst_finished {
            let io_position = processes[job_index].burst_position + 1;
            let io_delay = processes[job_index].bursts[io_position] as u64;
            processes[job_index].burst_position += 2;
            processes[job_index].remaining =
                processes[job_index].bursts[processes[job_index].burst_position];

            let promoted_level = run_level.saturating_sub(1);
            events.push(Event {
                time: end + io_delay,
                kind_order: 1,
                input_index: job_index,
                kind: EventKind::Wake {
                    job: job_index,
                    level: promoted_level,
                },
            });
            events[event_cursor..]
                .sort_by_key(|event| (event.time, event.kind_order, event.input_index));
        } else {
            let demoted_level = (run_level + 1).min(quanta.len() - 1);
            processes[job_index].level = demoted_level;
            queues[demoted_level].push_back(job_index);
        }

        last_dispatch_id = Some(selected_id);
    }

    let stats: Vec<Stat> = processes
        .iter()
        .map(|process| {
            let completion = process.completion.unwrap();
            let turnaround = completion - process.arrival;
            Stat {
                id: process.id,
                completion,
                turnaround,
                waiting: turnaround - process.cpu_total - process.io_total,
            }
        })
        .collect();

    let total_waiting: u64 = stats.iter().map(|stat| stat.waiting).sum();
    let total_turnaround: u64 = stats.iter().map(|stat| stat.turnaround).sum();

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: rounded_mean_x100(total_waiting, jobs.len()),
        mean_turnaround_x100: rounded_mean_x100(total_turnaround, jobs.len()),
        end: time,
    })
}
