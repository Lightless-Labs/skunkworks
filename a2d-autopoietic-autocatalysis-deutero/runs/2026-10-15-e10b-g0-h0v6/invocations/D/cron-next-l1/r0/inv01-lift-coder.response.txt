#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_star: [bool; 2],
}

#[derive(Clone, Copy)]
struct FieldSpec {
    field: Field,
    min: u32,
    max: u32,
}

const SPECS: [FieldSpec; 5] = [
    FieldSpec {
        field: Field::Minute,
        min: 0,
        max: 59,
    },
    FieldSpec {
        field: Field::Hour,
        min: 0,
        max: 23,
    },
    FieldSpec {
        field: Field::DayOfMonth,
        min: 1,
        max: 31,
    },
    FieldSpec {
        field: Field::Month,
        min: 1,
        max: 12,
    },
    FieldSpec {
        field: Field::DayOfWeek,
        min: 0,
        max: 6,
    },
];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(value, &yes)| yes.then_some(value as u8))
            .collect()
    }
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] =
        std::array::from_fn(|i| vec![false; (SPECS[i].max + 1) as usize]);
    let mut day_star = [false; 2];

    for i in 0..5 {
        let spec = SPECS[i];
        let text = parts[i];
        if i == 2 || i == 4 {
            day_star[if i == 2 { 0 } else { 1 }] = text.as_bytes() == b"*";
        }
        parse_field(text, spec, &mut selected[i])?;
    }

    Ok(Cron { selected, day_star })
}

fn parse_field(text: &str, spec: FieldSpec, selected: &mut [bool]) -> Result<(), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit() || matches!(byte, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter {
                field: spec.field,
                offset,
            });
        }
    }

    let mut item_start = 0;
    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field: spec.field,
                offset: item_start,
            });
        }
        parse_item(item, item_start, spec, selected)?;
        item_start += item.len() + 1;
    }

    Ok(())
}

fn parse_item(
    item: &str,
    offset: usize,
    spec: FieldSpec,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field: spec.field,
            offset,
        });
    }

    let (base_end, step_start) = if let Some(&slash) = slash_positions.first() {
        (slash, Some(slash + 1))
    } else {
        (bytes.len(), None)
    };

    let base = &bytes[..base_end];
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    let syntax_ok = if base == b"*" {
        true
    } else if hyphens.is_empty() {
        !base.is_empty() && base.iter().all(u8::is_ascii_digit)
    } else if hyphens.len() == 1 {
        let h = hyphens[0];
        h > 0
            && h + 1 < base.len()
            && base[..h].iter().all(u8::is_ascii_digit)
            && base[h + 1..].iter().all(u8::is_ascii_digit)
    } else {
        false
    };

    if !syntax_ok || (step_start.is_some() && step_start == Some(bytes.len())) {
        return Err(ParseError::InvalidSyntax {
            field: spec.field,
            offset,
        });
    }

    let mut range_start = spec.min;
    let mut range_end = spec.max;

    if base != b"*" {
        if hyphens.is_empty() {
            let (value, _) = parse_number(base, offset, spec)?;
            range_start = value;
            // A stepped single number extends to the domain maximum. An
            // unstepped single number selects only itself.
            if step_start.is_none() {
                range_end = value;
            }
        } else {
            let h = hyphens[0];
            let (start, _) = parse_number(&base[..h], offset, spec)?;
            let end_offset = offset + h + 1;
            let (end, _) = parse_number(&base[h + 1..], end_offset, spec)?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field: spec.field,
                    offset: end_offset,
                });
            }
            range_start = start;
            range_end = end;
        }
    }

    let step = if let Some(start) = step_start {
        let step_bytes = &bytes[start..];
        if !step_bytes.iter().all(u8::is_ascii_digit) {
            return Err(ParseError::InvalidSyntax {
                field: spec.field,
                offset,
            });
        }
        Some(parse_step(step_bytes, offset + start, spec)?)
    } else {
        None
    };

    if let Some(step) = step {
        let mut value = range_start;
        while value <= range_end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }
    } else {
        for value in range_start..=range_end {
            selected[value as usize] = true;
        }
    }

    Ok(())
}

fn parse_number(bytes: &[u8], offset: usize, spec: FieldSpec) -> Result<(u32, usize), ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero {
            field: spec.field,
            offset,
        });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field: spec.field,
                offset,
            })?;
    }

    if value < spec.min || value > spec.max {
        return Err(ParseError::OutOfRange {
            field: spec.field,
            offset,
        });
    }
    Ok((value, offset))
}

fn parse_step(bytes: &[u8], offset: usize, spec: FieldSpec) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero {
            field: spec.field,
            offset,
        });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field: spec.field,
                offset,
            })?;
    }

    let width = spec.max - spec.min + 1;
    if value == 0 || value > width {
        return Err(ParseError::BadStep {
            field: spec.field,
            offset,
        });
    }
    Ok(value)
}
