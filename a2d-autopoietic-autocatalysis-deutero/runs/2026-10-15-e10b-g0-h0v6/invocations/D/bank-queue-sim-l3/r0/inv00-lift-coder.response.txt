#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    started: bool,
    remaining: u64,
    pause_time: u64,
}

#[derive(Clone)]
struct Running {
    job: Job,
    work_at: u64,
    finish: u64,
}

fn better_waiter(a: &Job, b: &Job) -> bool {
    let ap = a.customer.priority;
    let bp = b.customer.priority;
    if ap != bp {
        return ap < bp;
    }
    if a.started != b.started {
        return a.started;
    }
    if a.started {
        if a.pause_time != b.pause_time {
            return a.pause_time < b.pause_time;
        }
        a.customer.id < b.customer.id
    } else if a.customer.arrival != b.customer.arrival {
        a.customer.arrival < b.customer.arrival
    } else {
        a.customer.id > b.customer.id
    }
}

fn best_index(waiting: &[Job]) -> Option<usize> {
    let mut best = None;
    for i in 0..waiting.len() {
        if best.is_none_or(|j| better_waiter(&waiting[i], &waiting[j])) {
            best = Some(i);
        }
    }
    best
}

fn take_best(waiting: &mut Vec<Job>) -> Option<Job> {
    best_index(waiting).map(|i| waiting.remove(i))
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut seen = std::collections::HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut finished_count = 0usize;

    while finished_count < customers.len() {
        // A scheduled finish is always the first event at this tick.
        if running.as_ref().is_some_and(|r| r.finish == now) {
            let done = running.take().unwrap();
            trace.push(Event {
                time: now,
                customer: done.job.customer.id,
                kind: Kind::Finish,
            });
            finished_count += 1;
        }

        // Process the complete arrival cohort before deciding whether to preempt.
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let c = arrivals[next_arrival].clone();
            trace.push(Event {
                time: now,
                customer: c.id,
                kind: Kind::Arrive,
            });
            waiting.push(Job {
                remaining: c.service,
                customer: c,
                started: false,
                pause_time: 0,
            });
            next_arrival += 1;
        }

        // Reconsider a running job only once its switching protection has ended.
        if let Some(active) = running.as_ref() {
            if now >= active.work_at {
                if let Some(index) = best_index(&waiting) {
                    if waiting[index].customer.priority < active.job.customer.priority {
                        let by = waiting[index].customer.id;
                        let mut paused = running.take().unwrap().job;
                        paused.remaining = active.finish - now;
                        paused.pause_time = now;
                        trace.push(Event {
                            time: now,
                            customer: paused.customer.id,
                            kind: Kind::Preempt {
                                remaining: paused.remaining,
                                by,
                            },
                        });
                        waiting.push(paused);
                    }
                }
            }
        }

        // Dispatch repeatedly at this tick, including inline zero-duration finishes.
        while running.is_none() {
            let Some(mut job) = take_best(&mut waiting) else {
                break;
            };
            let cost = if last_dispatched == Some(job.customer.id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: job.customer.id,
                time: now,
            })?;
            let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                customer: job.customer.id,
                time: now,
            })?;

            let kind = if job.started {
                Kind::Resume {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            } else {
                job.started = true;
                Kind::Start {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            };
            trace.push(Event {
                time: now,
                customer: job.customer.id,
                kind,
            });
            last_dispatched = Some(job.customer.id);

            if finish == now {
                trace.push(Event {
                    time: now,
                    customer: job.customer.id,
                    kind: Kind::Finish,
                });
                finished_count += 1;
            } else {
                running = Some(Running {
                    job,
                    work_at,
                    finish,
                });
            }
        }

        if finished_count == customers.len() {
            break;
        }

        let mut next_time = None;
        if next_arrival < arrivals.len() {
            next_time = Some(arrivals[next_arrival].arrival);
        }
        if let Some(active) = running.as_ref() {
            if active.finish > now {
                next_time = Some(next_time.map_or(active.finish, |t| t.min(active.finish)));
            }
            if active.work_at > now {
                next_time = Some(next_time.map_or(active.work_at, |t| t.min(active.work_at)));
            }
        }

        // Any remaining waiters imply the server should have been dispatched above.
        if let Some(t) = next_time {
            now = t;
        } else {
            break;
        }
    }

    Ok(trace)
}
