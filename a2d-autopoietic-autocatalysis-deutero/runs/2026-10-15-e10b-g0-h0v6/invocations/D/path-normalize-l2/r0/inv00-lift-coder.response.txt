#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> impl Iterator<Item = (usize, &str)> {
    path.split('/')
        .scan(0usize, |offset, component| {
            let at = *offset;
            *offset += component.len() + 1;
            Some((at, component))
        })
        .filter(|(_, component)| !component.is_empty())
}

fn reduce(
    stack: &mut Vec<String>,
    path: &str,
    absolute: bool,
    offset_adjustment: usize,
) -> Result<(), PathError> {
    for (at, component) in components(path) {
        match component {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape {
                        at: at + offset_adjustment,
                    });
                } else {
                    stack.push("..".to_owned());
                }
            }
            name => stack.push(name.to_owned()),
        }
    }
    Ok(())
}

fn assemble(stack: &[String], absolute: bool, want_directory: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if want_directory && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate(path)?;

    let absolute = path.starts_with('/');
    let want_directory = directory_intent(path);
    let mut stack = Vec::new();
    reduce(&mut stack, path, absolute, 0)?;

    Ok(assemble(&stack, absolute, want_directory))
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_normalized = if cwd.starts_with('/') {
        normalize(cwd).map_err(|_| PathError::InvalidCwd)?
    } else {
        return Err(PathError::InvalidCwd);
    };

    let home_normalized = match home {
        Some(home) => {
            if !home.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(normalize(home).map_err(|_| PathError::InvalidHome)?)
        }
        None => None,
    };

    validate(path)?;

    let (base, remainder, offset_adjustment, absolute) = if path.starts_with('~') {
        let bytes = path.as_bytes();
        if path == "~" || bytes.get(1) == Some(&b'/') {
            let base = home_normalized.ok_or(PathError::MissingHome)?;
            let remainder = if path == "~" { "" } else { &path[2..] };
            (base, remainder, 2, true)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        ("/".to_owned(), path, 0, true)
    } else {
        (cwd_normalized, path, 0, true)
    };

    let mut stack: Vec<String> = base
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect();

    let want_directory = directory_intent(path);
    reduce(&mut stack, remainder, absolute, offset_adjustment)?;

    Ok(assemble(&stack, true, want_directory))
}
