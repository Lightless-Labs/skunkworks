#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Debug)]
struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<usize>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_fold(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while input.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

fn printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::new();
    let mut positions = Vec::new();
    let mut runs = Vec::new();
    let mut i = 0;
    let mut comment_depth = 0usize;
    let mut outer_comment = 0usize;
    let mut quote_protected = false;
    let mut quote_closed = false;
    let mut run_open = false;

    while i < input.len() {
        let b = input[i];

        if comment_depth > 0 {
            if b == b'\\' {
                let Some(&next) = input.get(i + 1) else {
                    return Err(EmailError::CommentEscape { at: i });
                };
                if !printable(next) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = is_fold(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if !printable(b) && !is_hws(b) && b != b'(' && b != b')' {
                return Err(EmailError::CommentChar { at: i });
            }
            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_depth += 1;
            } else if b == b')' {
                comment_depth -= 1;
            }
            i += 1;
            continue;
        }

        if quote_protected && !quote_closed {
            bytes.push(b);
            positions.push(i);
            if b == b'\\' {
                if i + 1 < input.len() {
                    i += 1;
                    bytes.push(input[i]);
                    positions.push(i);
                }
            } else if b == b'"' {
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'\r' || b == b'\n' || b == b'(' {
            if !run_open {
                runs.push(i);
                run_open = true;
            }
            if b == b'(' {
                comment_depth = 1;
                outer_comment = i;
                i += 1;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = is_fold(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            i += 1;
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if bytes.is_empty() && b == b'"' {
            quote_protected = true;
            bytes.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        run_open = false;
        bytes.push(b);
        positions.push(i);
        i += 1;
    }

    if comment_depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }

    Ok(Cleaned { bytes, positions, runs })
}

fn pos(c: &Cleaned, idx: usize, input_len: usize) -> usize {
    c.positions.get(idx).copied().unwrap_or(input_len)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let cleaned = cleanup(raw)?;
    let b = &cleaned.bytes;

    for &run_at in &cleaned.runs {
        let retained_before = cleaned.positions.iter().position(|&p| p > run_at);
        let before_idx = retained_before.and_then(|i| i.checked_sub(1));
        let after_idx = retained_before;
        let at_adjacent = before_idx.map(|i| b[i] == b'@').unwrap_or(false)
            || after_idx.map(|i| b[i] == b'@').unwrap_or(false);
        let at_edge = before_idx.is_none() || after_idx.is_none();
        if !at_edge && !at_adjacent {
            return Err(EmailError::CfwsBoundary { at: run_at });
        }
    }

    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if b[0] == b'"' {
        let mut i = 1;
        while i < b.len() {
            if b[i] == b'\\' {
                i += 2;
            } else if b[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: pos(&cleaned, 0, raw.len()) });
        }
    }

    let protected_end = quote_end.map(|i| i + 1).unwrap_or(0);
    let ats: Vec<usize> = (protected_end..b.len()).filter(|&i| b[i] == b'@').collect();
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt { at: pos(&cleaned, ats[1], raw.len()) });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: pos(&cleaned, sep, raw.len()) });
    }

    let local_bytes = &b[..sep];
    let (local, quoted) = if b[0] == b'"' {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal { at: pos(&cleaned, end + 1, raw.len()) });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < end {
            if b[i] == b'\\' {
                let next = b[i + 1];
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape { at: pos(&cleaned, i, raw.len()) });
                }
                decoded.push(next);
                origins.push(pos(&cleaned, i, raw.len()));
                i += 2;
            } else {
                if !printable(b[i]) {
                    return Err(EmailError::QuotedChar { at: pos(&cleaned, i, raw.len()) });
                }
                decoded.push(b[i]);
                origins.push(pos(&cleaned, i, raw.len()));
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: pos(&cleaned, end, raw.len()) });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: pos(&cleaned, 64, raw.len()) });
        }
        let mut previous_dot = false;
        for (i, &ch) in local_bytes.iter().enumerate() {
            if ch == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || previous_dot {
                    return Err(EmailError::LocalDot { at: pos(&cleaned, i, raw.len()) });
                }
                previous_dot = true;
            } else {
                previous_dot = false;
                if !ch.is_ascii_alphanumeric()
                    && !b"!#$%&'*+-/=?^_`{|}~".contains(&ch)
                {
                    return Err(EmailError::LocalChar { at: pos(&cleaned, i, raw.len()) });
                }
            }
        }
        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start = sep + 1;
    let mut domain_end = b.len();
    let root_dot = if domain_end > domain_start && b[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: pos(&cleaned, domain_start, raw.len()) });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong { at: pos(&cleaned, domain_start + 253, raw.len()) });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || b[i] == b'.' {
            if start == i {
                let at = if i == domain_end {
                    root_dot.map(|r| pos(&cleaned, r, raw.len())).unwrap_or(pos(&cleaned, i, raw.len()))
                } else {
                    pos(&cleaned, i, raw.len())
                };
                return Err(EmailError::EmptyLabel { at });
            }
            let len = i - start;
            if len > 63 {
                return Err(EmailError::LabelTooLong { at: pos(&cleaned, start + 63, raw.len()) });
            }
            if b[start] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos(&cleaned, start, raw.len()) });
            }
            if b[i - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos(&cleaned, i - 1, raw.len()) });
            }
            for j in start..i {
                if !b[j].is_ascii_alphanumeric() && b[j] != b'-' {
                    return Err(EmailError::DomainChar { at: pos(&cleaned, j, raw.len()) });
                }
            }
            labels.push((start, i));
            start = i + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: pos(&cleaned, domain_start, raw.len()) });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if b[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos(&cleaned, last_start, raw.len()) });
    }

    let domain = String::from_utf8(b[domain_start..domain_end].to_vec())
        .unwrap()
        .to_ascii_lowercase();
    Ok(Address { local, domain, quoted })
}
