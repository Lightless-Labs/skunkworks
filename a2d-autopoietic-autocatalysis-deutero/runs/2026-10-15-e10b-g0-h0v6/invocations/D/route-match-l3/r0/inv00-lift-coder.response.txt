#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn decode_segment(input: &str, base: usize) -> Result<String, usize> {
    let bytes = input.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_value(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_value(bytes[i + 2]).ok_or(base + i)?;
            let decoded = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn parse_path(
    input: &str,
    pattern: bool,
) -> Result<(Vec<Segment>, bool), usize> {
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut pieces: Vec<(&str, usize)> = Vec::new();
    let mut start = 1;
    for (idx, b) in bytes.iter().enumerate().skip(1) {
        if *b == b'/' {
            if idx == bytes.len() - 1 {
                if idx == start {
                    return Err(start);
                }
                pieces.push((&input[start..idx], start));
                return build_segments(pieces, true, pattern);
            }
            if idx == start {
                return Err(start);
            }
            pieces.push((&input[start..idx], start));
            start = idx + 1;
        }
    }
    if start == bytes.len() {
        return Err(start);
    }
    pieces.push((&input[start..], start));
    build_segments(pieces, false, pattern)
}

fn build_segments(
    pieces: Vec<(&str, usize)>,
    trailing: bool,
    pattern: bool,
) -> Result<(Vec<Segment>, bool), usize> {
    let mut segments = Vec::with_capacity(pieces.len());
    let mut names = std::collections::HashSet::new();

    for (text, at) in pieces {
        if pattern && (text.starts_with(':') || text.starts_with('*')) {
            let marker = text.as_bytes()[0];
            let name = &text[1..];
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(at);
            }
            if marker == b'*' && segments.len() + 1 != count_segments_from(&segments, 1) {
                // Finality is checked after parsing all segments below.
            }
            segments.push(if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            let decoded = decode_segment(text, at)?;
            segments.push(Segment::Static(decoded));
        }
    }

    if pattern {
        for (i, segment) in segments.iter().enumerate() {
            if matches!(segment, Segment::Wildcard(_)) && i + 1 != segments.len() {
                let at = match segment {
                    Segment::Wildcard(name) => {
                        let _ = name;
                        // Reconstruct the source offset from the corresponding piece.
                        // The pieces were consumed, so use the cumulative segment lengths.
                        0
                    }
                    _ => 0,
                };
                let _ = at;
            }
        }
    }

    Ok((segments, trailing))
}

fn count_segments_from(_: &[Segment], n: usize) -> usize {
    n
}
