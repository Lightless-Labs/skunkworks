pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut arrival_order: Vec<usize> = (0..jobs.len()).collect();
    arrival_order.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut completion_times = vec![0u64; jobs.len()];
    let mut completed = vec![false; jobs.len()];
    let mut ready = std::collections::VecDeque::new();
    let mut next_arrival = 0usize;
    let mut time = 0u64;
    let mut last_dispatched_id: Option<u32> = None;
    let mut completed_count = 0usize;

    fn admit(
        jobs: &[Job],
        arrival_order: &[usize],
        next_arrival: &mut usize,
        time: u64,
        ready: &mut std::collections::VecDeque<usize>,
        completed: &mut [bool],
        completion_times: &mut [u64],
        completed_count: &mut usize,
    ) {
        while *next_arrival < arrival_order.len() {
            let index = arrival_order[*next_arrival];
            let job = &jobs[index];
            if u64::from(job.arrival) > time {
                break;
            }
            *next_arrival += 1;
            if job.ticks == 0 {
                completed[index] = true;
                completion_times[index] = u64::from(job.arrival);
                *completed_count += 1;
            } else {
                ready.push_back(index);
            }
        }
    }

    while completed_count < jobs.len() {
        admit(
            jobs,
            &arrival_order,
            &mut next_arrival,
            time,
            &mut ready,
            &mut completed,
            &mut completion_times,
            &mut completed_count,
        );

        if ready.is_empty() {
            if next_arrival >= arrival_order.len() {
                break;
            }
            let next_time = u64::from(jobs[arrival_order[next_arrival]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched_id = None;
            }
            continue;
        }

        let index = ready.pop_front().expect("ready queue is non-empty");
        let job = &jobs[index];

        if last_dispatched_id.is_some_and(|id| id != job.id) {
            time += u64::from(switch_cost);
        }

        admit(
            jobs,
            &arrival_order,
            &mut next_arrival,
            time,
            &mut ready,
            &mut completed,
            &mut completion_times,
            &mut completed_count,
        );

        let remaining = u64::from(job.ticks) - (time - completion_times[index]);
        let consumed = remaining.min(u64::from(quantum));
        let start = time;
        time += consumed;
        let finished = consumed == remaining;

        runs.push(Run {
            id: job.id,
            start,
            end: time,
            finished,
        });

        admit(
            jobs,
            &arrival_order,
            &mut next_arrival,
            time,
            &mut ready,
            &mut completed,
            &mut completion_times,
            &mut completed_count,
        );

        if finished {
            completed[index] = true;
            completion_times[index] = time;
            completed_count += 1;
        } else {
            // Store remaining work as a negative offset from the original ticks.
            completion_times[index] = time - (u64::from(job.ticks) - (remaining - consumed));
            ready.push_back(index);
        }

        last_dispatched_id = Some(job.id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion_times[index]))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}

fn main() {}
