#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_char(word: &mut Word, c: char, active_glob: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn append_str(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append_char(word, c, active_glob);
    }
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.chars();

    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }

    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_char(&mut word, c, false);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    if matches!(escaped, '"' | '\\' | '$' | '\n') {
                        if escaped != '\n' {
                            append_char(&mut word, escaped, false);
                        }
                        i = next_i + escaped.len_utf8();
                    } else {
                        append_char(&mut word, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    let (value, end) = parse_expansion(input, i, env)?;
                    if let Some(value) = value {
                        append_str(&mut word, &value, false);
                        active = true;
                        i = end;
                    } else {
                        append_char(&mut word, '$', false);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_char(&mut word, c, false);
                    active = true;
                    i = next_i;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single;
                    quote_pos = i;
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double;
                    quote_pos = i;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    if escaped != '\n' {
                        append_char(&mut word, escaped, false);
                        active = true;
                    }
                    i = next_i + escaped.len_utf8();
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    let (value, end) = parse_expansion(input, i, env)?;
                    if let Some(value) = value {
                        append_str(&mut word, &value, true);
                        active = true;
                        i = end;
                    } else {
                        append_char(&mut word, '$', true);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_char(&mut word, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_expansion<'a>(
    input: &'a str,
    dollar: usize,
    env: &'a [(&str, &str)],
) -> Result<(Option<String>, usize), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((None, after_dollar));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut cursor = content_start;
        let mut close = None;

        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            let next = cursor + c.len_utf8();
            if c == '\\' && next < input.len() {
                let escaped = input[next..].chars().next().unwrap();
                cursor = next + escaped.len_utf8();
            } else if c == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor = next;
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let (name, default) = if let Some(op_pos) = contents.find(":-") {
            (&contents[..op_pos], Some(&contents[op_pos + 2..]))
        } else {
            (contents, None)
        };

        let mut name_chars = name.chars();
        let valid_name = name_chars.next().is_some_and(is_name_start)
            && name_chars.all(is_name_char);

        if !valid_name || (default.is_none() && name.contains(':')) {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        if let Some(default) = default {
            let selected = lookup(name, env).filter(|value| !value.is_empty());
            let value = selected
                .map(str::to_owned)
                .unwrap_or_else(|| decode_default(default));
            return Ok((Some(value), close + 1));
        }

        return Ok((
            Some(lookup(name, env).unwrap_or("").to_owned()),
            close + 1,
        ));
    }

    if is_name_start(first) {
        let mut end = after_dollar;
        while end < input.len() {
            let c = input[end..].chars().next().unwrap();
            if is_name_char(c) {
                end += c.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[after_dollar..end];
        return Ok((Some(lookup(name, env).unwrap_or("").to_owned()), end));
    }

    Ok((None, after_dollar))
}

fn encode_bare(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }

    let mut result = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric()
            || matches!(c, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%')
        {
            result.push(c);
        } else {
            result.push('\\');
            result.push(c);
        }
    }
    Some(result)
}

fn encode_single(word: &str) -> Option<String> {
    if word.contains('\'') {
        return None;
    }
    let mut result = String::with_capacity(word.len() + 2);
    result.push('\'');
    result.push_str(word);
    result.push('\'');
    Some(result)
}

fn encode_double(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut result = String::with_capacity(word.len() + 2);
    result.push('"');
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            result.push('\\');
        }
        result.push(c);
    }
    result.push('"');
    Some(result)
}

fn encode_fallback(word: &str) -> String {
    let mut result = String::new();
    result.push('\'');
    for c in word.chars() {
        if c == '\'' {
            result.push('\'');
            result.push('\\');
            result.push('\'');
            result.push('\'');
        } else {
            result.push(c);
        }
    }
    result.push('\'');
    result
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let candidates = [
            encode_bare(word),
            encode_single(word),
            encode_double(word),
        ];

        let best = candidates
            .into_iter()
            .enumerate()
            .filter_map(|(priority, candidate)| {
                candidate.map(|text| (text.len(), priority, text))
            })
            .min_by_key(|(len, priority, _)| (*len, *priority))
            .map(|(_, _, text)| text)
            .unwrap_or_else(|| encode_fallback(word));

        encoded.push(best);
    }

    Ok(encoded.join(" "))
}
