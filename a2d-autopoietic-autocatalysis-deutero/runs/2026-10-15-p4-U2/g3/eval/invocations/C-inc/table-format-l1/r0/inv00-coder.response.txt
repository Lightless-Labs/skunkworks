#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
}

pub fn render(rows: &[Vec<String>]) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let expected = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != expected {
            return Err(TableError::Ragged {
                row,
                expected,
                actual: cells.len(),
            });
        }
    }

    let mut trimmed = Vec::with_capacity(rows.len());
    let mut widths = vec![1usize; expected];

    for (row, cells) in rows.iter().enumerate() {
        let mut trimmed_row = Vec::with_capacity(expected);
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if !(' '..='~').contains(&ch) {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }

            let content = cell.trim_end_matches(' ');
            widths[column] = widths[column].max(content.len());
            trimmed_row.push(content);
        }
        trimmed.push(trimmed_row);
    }

    for (column, &width) in widths.iter().enumerate() {
        if width > 32 {
            return Err(TableError::TooWide { column, width });
        }
    }

    let mut output = String::new();

    let append_border = |output: &mut String| {
        output.push('+');
        for &width in &widths {
            for _ in 0..(width + 2) {
                output.push('-');
            }
            output.push('+');
        }
        output.push('\n');
    };

    append_border(&mut output);

    for row in &trimmed {
        output.push('|');
        for (column, content) in row.iter().enumerate() {
            output.push(' ');
            output.push_str(content);
            for _ in content.len()..widths[column] {
                output.push(' ');
            }
            output.push(' ');
            output.push('|');
        }
        output.push('\n');
    }

    append_border(&mut output);
    Ok(output)
}

fn main() {}
