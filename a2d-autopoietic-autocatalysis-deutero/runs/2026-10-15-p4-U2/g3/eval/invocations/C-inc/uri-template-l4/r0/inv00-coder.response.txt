#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semicolon,
    Query,
    Ampersand,
}

impl Operator {
    fn separator(self) -> char {
        match self {
            Self::Simple | Self::Plus | Self::Hash => ',',
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon => ';',
            Self::Query | Self::Ampersand => '&',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Hash => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Query => Some('?'),
            Self::Ampersand => Some('&'),
            Self::Simple | Self::Plus => None,
        }
    }

    fn reserved(self) -> bool {
        matches!(self, Self::Plus | Self::Hash)
    }

    fn named_fields(self) -> bool {
        matches!(self, Self::Semicolon | Self::Query | Self::Ampersand)
    }
}

#[derive(Clone)]
struct Specifier {
    name: String,
    start: usize,
    modifier: Modifier,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Expression {
    operator: Operator,
    specs: Vec<Specifier>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn parse_expression(content: &str, content_pos: usize, close_pos: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let (operator, offset) = if let Some(&first) = bytes.first() {
        match first {
            b'+' => (Operator::Plus, 1),
            b'#' => (Operator::Hash, 1),
            b'.' => (Operator::Dot, 1),
            b'/' => (Operator::Slash, 1),
            b';' => (Operator::Semicolon, 1),
            b'?' => (Operator::Query, 1),
            b'&' => (Operator::Ampersand, 1),
            b'=' | b'!' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator { pos: content_pos });
            }
            _ => (Operator::Simple, 0),
        }
    } else {
        (Operator::Simple, 0)
    };

    let body = &content[offset..];
    let body_pos = content_pos + offset;
    if body.is_empty() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut start = 0;

    loop {
        let end = body[start..]
            .find(',')
            .map(|relative| start + relative)
            .unwrap_or(body.len());

        if start == end {
            return Err(Error::EmptyVariable {
                pos: body_pos + start,
            });
        }

        let part = &body[start..end];
        let part_bytes = part.as_bytes();
        let modifier_at = part_bytes
            .iter()
            .position(|&byte| byte == b'*' || byte == b':')
            .unwrap_or(part_bytes.len());
        let name_part = &part[..modifier_at];

        if name_part.is_empty() {
            return Err(Error::InvalidName {
                pos: body_pos + start,
            });
        }

        let name_bytes = name_part.as_bytes();
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName {
                pos: body_pos + start,
            });
        }
        if let Some(index) = name_bytes
            .iter()
            .enumerate()
            .skip(1)
            .find_map(|(index, &byte)| (!is_name_continue(byte)).then_some(index))
        {
            return Err(Error::InvalidName {
                pos: body_pos + start + index,
            });
        }

        let modifier = if modifier_at == part_bytes.len() {
            Modifier::None
        } else {
            let modifier_pos = body_pos + start + modifier_at;
            let suffix = &part[modifier_at..];
            if suffix == "*" {
                Modifier::Explode
            } else if suffix.starts_with(':') {
                let digits = &suffix[1..];
                let valid = (1..=4).contains(&digits.len())
                    && digits.as_bytes()[0].is_ascii_digit()
                    && digits.as_bytes()[0] != b'0'
                    && digits.bytes().all(|byte| byte.is_ascii_digit());
                if !valid {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                let value = digits
                    .parse::<usize>()
                    .map_err(|_| Error::InvalidModifier { pos: modifier_pos })?;
                if !(1..=9999).contains(&value) {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                Modifier::Prefix(value)
            } else {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
        };

        let name = name_part.to_owned();
        if seen.iter().any(|previous| previous == &name) {
            return Err(Error::DuplicateVariable {
                pos: body_pos + start,
            });
        }
        seen.push(name.clone());
        specs.push(Specifier {
            name,
            start: body_pos + start,
            modifier,
        });

        if end == body.len() {
            break;
        }
        start = end + 1;
    }

    Ok(Expression { operator, specs })
}

fn parse_template(template: &str) -> Result<Vec<(usize, usize, Expression)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let close = match bytes[open + 1..].iter().position(|&byte| byte == b'}') {
                    Some(relative) => open + 1 + relative,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(relative) = bytes[open + 1..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative,
                    });
                }

                let content = &template[open + 1..close];
                let expression = parse_expression(content, open + 1, close)?;
                expressions.push((open, close + 1, expression));
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    Ok(expressions)
}

fn selected_value<'a>(vars: &'a [(&str, Value)], name: &str) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn take_prefix(value: &str, count: usize) -> &str {
    let end = value
        .char_indices()
        .nth(count)
        .map(|(index, _)| index)
        .unwrap_or(value.len());
    &value[..end]
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];
        if byte == b'%' && index + 2 < bytes.len()
            && bytes[index + 1].is_ascii_hexdigit()
            && bytes[index + 2].is_ascii_hexdigit()
        {
            output.push('%');
            output.push((bytes[index + 1] as char).to_ascii_uppercase());
            output.push((bytes[index + 2] as char).to_ascii_uppercase());
            index += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let is_reserved = reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                    | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || is_reserved {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    output
}

fn name_value(name: &str, value: String, operator: Operator) -> String {
    if operator == Operator::Semicolon && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expression.operator;
    let mut fields = Vec::<String>::new();

    for spec in &expression.specs {
        let Some(value) = selected_value(vars, &spec.name) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.start });
        }

        match value {
            Value::Scalar(raw) => {
                let raw = match spec.modifier {
                    Modifier::Prefix(count) => take_prefix(raw, count),
                    Modifier::None | Modifier::Explode => raw,
                };
                let encoded = encode(raw, operator.reserved());
                fields.push(if operator.named_fields() {
                    name_value(&spec.name, encoded, operator)
                } else {
                    encoded
                });
            }
            Value::List(items) => {
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if items.is_empty() {
                    continue;
                }
                if exploded {
                    for item in items {
                        let encoded = encode(item, operator.reserved());
                        fields.push(if operator.named_fields() {
                            name_value(&spec.name, encoded, operator)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, operator.reserved()))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if operator.named_fields() {
                        name_value(&spec.name, joined, operator)
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.iter().any(|seen| *seen == key) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                    keys.push(key);
                }

                let exploded = matches!(spec.modifier, Modifier::Explode);
                if entries.is_empty() {
                    continue;
                }

                if exploded {
                    for (key, value) in entries {
                        let encoded_key = encode(key, operator.reserved());
                        let encoded_value = encode(value, operator.reserved());
                        if operator == Operator::Semicolon && encoded_value.is_empty() {
                            fields.push(encoded_key);
                        } else {
                            fields.push(format!("{encoded_key}={encoded_value}"));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, operator.reserved()));
                        components.push(encode(value, operator.reserved()));
                    }
                    let joined = components.join(",");
                    fields.push(if operator.named_fields() {
                        name_value(&spec.name, joined, operator)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    if let Some(prefix) = operator.prefix() {
        output.push(prefix);
    }
    output.push_str(&fields.join(&operator.separator().to_string()));
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[cursor..start]);
        output.push_str(&expand_expression(&expression, vars)?);
        cursor = end;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}
