```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index].iter().any(|prior| prior.id == customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| b.arrival.cmp(&a.arrival).then_with(|| a.id.cmp(&b.id)));

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting: Vec<usize> = Vec::new();
    let mut arrival_index = 0;
    let mut current_time = 0u64;
    let mut completion: Option<(u64, u32)> = None;
    let mut last_finish: Option<u64> = None;
    let mut first_dispatch = true;

    while arrival_index < arrivals.len() || !waiting.is_empty() || completion.is_some() {
        let next_arrival = arrivals.get(arrival_index).map(|c| c.arrival);
        let next_completion = completion.map(|(time, _)| time);

        let time = match (next_arrival, next_completion) {
            (Some(a), Some(f)) => a.min(f),
            (Some(a), None) => a,
            (None, Some(f)) => f,
            (None, None) => break,
        };
        current_time = time;

        if let Some((finish_time, customer_id)) = completion {
            if finish_time == time {
                events.push(Event {
                    time,
                    customer: customer_id,
                    kind: Kind::Finish,
                });
                completion = None;
                last_finish = Some(time);
            }
        }

        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == time {
            let customer = arrivals[arrival_index];
            events.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(customer.id as usize);
            arrival_index += 1;
        }

        if completion.is_none() && !waiting.is_empty() {
            let customer_id = waiting.remove(0) as u32;
            let customer = customers.iter().find(|c| c.id == customer_id).unwrap();
            let wake = first_dispatch || last_finish.is_some_and(|finish| time > finish);
            let finish = time
                .checked_add(customer.service)
                .and_then(|t| t.checked_add(if wake { wake_cost } else { 0 }))
                .ok_or(Error::TimeOverflow {
                    customer: customer_id,
                    time,
                })?;

            events.push(Event {
                time,
                customer: customer_id,
                kind: Kind::Start { finish, wake },
            });
            first_dispatch = false;

            if finish == time {
                events.push(Event {
                    time,
                    customer: customer_id,
                    kind: Kind::Finish,
                });
                last_finish = Some(time);
            } else {
                completion = Some((finish, customer_id));
            }
        }
    }

    let _ = current_time;
    Ok(events)
}

fn main() {}
```
