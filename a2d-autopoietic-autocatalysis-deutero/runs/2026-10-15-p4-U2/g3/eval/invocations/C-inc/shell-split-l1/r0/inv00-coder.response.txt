#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = String::new();
    let mut active = false;
    let mut quote: Option<(char, usize)> = None;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match quote {
            Some(('\'', _)) => {
                if ch == '\'' {
                    quote = None;
                } else {
                    current.push(ch);
                }
            }
            Some(('"', _)) => {
                if ch == '"' {
                    quote = None;
                } else if ch == '\\' {
                    let Some(&(next_pos, next)) = chars.peek() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if next == '"' || next == '\\' || next == '$' || next == '\n' {
                        chars.next();
                        if next != '\n' {
                            current.push(next);
                        }
                    } else {
                        current.push('\\');
                        let _ = next_pos;
                    }
                } else {
                    current.push(ch);
                }
            }
            _ => {
                if ch == ' ' || ch == '\t' || ch == '\r' || ch == '\n' {
                    if active {
                        words.push(std::mem::take(&mut current));
                        active = false;
                    }
                } else if ch == '\'' || ch == '"' {
                    active = true;
                    quote = Some((ch, pos));
                } else if ch == '\\' {
                    let Some(&(_, next)) = chars.peek() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if next == '\n' {
                        chars.next();
                    } else {
                        chars.next();
                        current.push(next);
                        active = true;
                    }
                } else {
                    current.push(ch);
                    active = true;
                }
            }
        }
    }

    if let Some((quote_char, pos)) = quote {
        return Err(if quote_char == '\'' {
            SplitError::UnterminatedSingle { pos }
        } else {
            SplitError::UnterminatedDouble { pos }
        });
    }

    if active {
        words.push(current);
    }

    Ok(words)
}

fn main() {}
