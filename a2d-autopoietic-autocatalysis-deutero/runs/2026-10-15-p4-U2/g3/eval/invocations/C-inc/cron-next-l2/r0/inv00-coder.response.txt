#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    allowed: [Vec<u8>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(bytes: &[u8], start: usize, field: Field) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }
    Ok(value)
}

fn parse_field(input: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = input.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit() || matches!(byte, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = bounds(field);
    let width = maximum - minimum + 1;
    let mut selected = vec![false; (maximum + 1) as usize];

    let mut item_start = 0;
    for item in input.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item_bytes = item.as_bytes();
        let slash_positions: Vec<usize> = item_bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let (base, step_bytes, step_offset) = match slash_positions.as_slice() {
            [] => (item_bytes, None, 0),
            [slash] => (
                &item_bytes[..*slash],
                Some(&item_bytes[*slash + 1..]),
                item_start + *slash + 1,
            ),
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                })
            }
        };

        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_kind = if is_star && hyphens.is_empty() {
            Some((None, None))
        } else if !is_star && hyphens.is_empty() && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            Some((Some((0, base.len())), None))
        } else if !is_star && hyphens.len() == 1 {
            let hyphen = hyphens[0];
            let start_digits = &base[..hyphen];
            let end_digits = &base[hyphen + 1..];
            if !start_digits.is_empty()
                && !end_digits.is_empty()
                && start_digits.iter().all(u8::is_ascii_digit)
                && end_digits.iter().all(u8::is_ascii_digit)
            {
                Some((Some((0, hyphen)), Some((hyphen + 1, base.len()))))
            } else {
                None
            }
        } else {
            None
        };

        let Some((start_span, end_span)) = base_kind else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        };

        let (start_value, end_value) = if let Some((start_index, start_end)) = start_span {
            let start_offset = item_start + start_index;
            let start = parse_number(&base[start_index..start_end], start_offset, field)?;
            if start < minimum || start > maximum {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: start_offset,
                });
            }

            if let Some((end_index, end_end)) = end_span {
                let end_offset = item_start + end_index;
                let end = parse_number(&base[end_index..end_end], end_offset, field)?;
                if end < minimum || end > maximum {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: end_offset,
                    });
                }
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_offset,
                    });
                }
                (start, end)
            } else if is_star {
                (minimum, maximum)
            } else if step_bytes.is_some() {
                (start, maximum)
            } else {
                (start, start)
            }
        } else {
            (minimum, maximum)
        };

        let step = if let Some(step_bytes) = step_bytes {
            if step_bytes.is_empty() || !step_bytes.iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            let value = parse_number(step_bytes, step_offset, field)?;
            if value == 0 || value > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_offset,
                });
            }
            value
        } else {
            1
        };

        let mut value = start_value;
        while value <= end_value {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        item_start += item.len() + 1;
    }

    Ok(selected
        .into_iter()
        .enumerate()
        .filter_map(|(value, yes)| yes.then_some(value as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut allowed: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    for (index, field) in FIELDS.into_iter().enumerate() {
        allowed[index] = parse_field(pieces[index], field)?;
    }

    Ok(Cron {
        allowed,
        day_of_month_is_star: pieces[2] == "*",
        day_of_week_is_star: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.allowed[match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        }]
            .clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(contains(&self.allowed[0], at.minute)
            && contains(&self.allowed[1], at.hour)
            && contains(&self.allowed[2], at.day)
            && contains(&self.allowed[3], at.month)
            && contains(&self.allowed[4], weekday))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if !increment_minute(&mut current) {
                return Ok(None);
            }
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn contains(values: &[u8], value: u8) -> bool {
    values.binary_search(&value).is_ok()
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

fn increment_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;

    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;

    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;

    if at.month < 12 {
        at.month += 1;
        return true;
    }
    at.month = 1;

    if at.year < 2399 {
        at.year += 1;
        return true;
    }

    false
}
