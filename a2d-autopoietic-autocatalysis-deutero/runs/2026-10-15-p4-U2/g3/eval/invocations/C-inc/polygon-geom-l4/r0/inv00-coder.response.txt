```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
    Duplicate { first: usize, second: usize },
    ZeroArea,
    AdjacentOverlap { vertex: usize },
    Intersection { edge_a: usize, edge_b: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Validated {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub measures: Measures,
}

const LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-LIMIT..=LIMIT).contains(&p.x) && (-LIMIT..=LIMIT).contains(&p.y)
}

fn prepare(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }
    let mut vertices = points.to_vec();
    if vertices.len() >= 2 && vertices[0] == *vertices.last().unwrap() {
        vertices.pop();
    }
    if vertices.len() < 3 {
        return Err(GeomError::TooFewVertices { len: vertices.len() });
    }
    Ok(vertices)
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn signed_area2(vertices: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..vertices.len() {
        let a = vertices[i];
        let b = vertices[(i + 1) % vertices.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn edge_length_milli(a: Point, b: Point) -> u64 {
    let dx = (a.x as i128 - b.x as i128) as u128;
    let dy = (a.y as i128 - b.y as i128) as u128;
    let n = (dx * dx + dy * dy) * 1_000_000u128;

    // Integer square root, rounded to nearest with half ties upward.
    let mut lo = 0u128;
    let mut hi = 3_000_000_001u128;
    while lo + 1 < hi {
        let mid = lo + (hi - lo) / 2;
        if mid * mid <= n {
            lo = mid;
        } else {
            hi = mid;
        }
    }
    let lower_diff = n - lo * lo;
    let upper_diff = hi * hi - n;
    if upper_diff <= lower_diff {
        hi as u64
    } else {
        lo as u64
    }
}

fn measures_for(vertices: &[Point]) -> Measures {
    let signed = signed_area2(vertices);
    let mut edge_milli = Vec::with_capacity(vertices.len());
    let mut perimeter = 0u64;
    let mut longest: Option<usize> = None;

    for i in 0..vertices.len() {
        let length = edge_length_milli(vertices[i], vertices[(i + 1) % vertices.len()]);
        edge_milli.push(length);
        perimeter += length;
        match longest {
            None => longest = Some(i),
            Some(old) => {
                if length > edge_milli[old]
                    || (length == edge_milli[old]
                        && {
                            let pair = edge_pair(vertices[i], vertices[(i + 1) % vertices.len()]);
                            let old_pair =
                                edge_pair(vertices[old], vertices[(old + 1) % vertices.len()]);
                            pair < old_pair || (pair == old_pair && i < old)
                        })
                {
                    longest = Some(i);
                }
            }
        }
    }

    Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation: if signed > 0 {
            Orientation::CounterClockwise
        } else if signed < 0 {
            Orientation::Clockwise
        } else {
            Orientation::Degenerate
        },
        edge_milli,
        perimeter_milli: perimeter,
        longest_edge: longest,
    }
}

fn edge_pair(a: Point, b: Point) -> (Point, Point) {
    if a <= b {
        (a, b)
    } else {
        (b, a)
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let vertices = prepare(points)?;
    Ok(measures_for(&vertices))
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let vertices = prepare(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut result = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        for i in 0..vertices.len() {
            if on_segment(vertices[i], vertices[(i + 1) % vertices.len()], q) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            result.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..vertices.len() {
            let a = vertices[i];
            let b = vertices[(i + 1) % vertices.len()];
            if (a.y > q.y) != (b.y > q.y) {
                let c = cross(a, b, q);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                    inside = !inside;
                }
            }
        }
        result.push(if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(result)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|(p, _)| *p == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|(p, _)| *p);

    if unique.len() <= 1 {
        let vertices: Vec<Point> = unique.iter().map(|(p, _)| *p).collect();
        let source_indices: Vec<usize> = unique.iter().map(|(_, i)| *i).collect();
        return Ok(Hull {
            vertices,
            source_indices,
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let all_collinear = unique
        .iter()
        .skip(2)
        .all(|(p, _)| cross(unique[0].0, unique[1].0, *p) == 0);

    let selected: Vec<(Point, usize)> = if all_collinear {
        match mode {
            Collinear::Include => unique,
            Collinear::Exclude => vec![unique[0], *unique.last().unwrap()],
        }
    } else {
        let mut lower: Vec<(Point, usize)> = Vec::new();
        for item in unique.iter().copied() {
            while lower.len() >= 2 {
                let turn = cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, item.0);
                if if mode == Collinear::Include {
                    turn < 0
                } else {
                    turn <= 0
                } {
                    lower.pop();
                } else {
                    break;
                }
            }
            lower.push(item);
        }
        let mut upper: Vec<(Point, usize)> = Vec::new();
        for item in unique.iter().copied().rev() {
            while upper.len() >= 2 {
                let turn = cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, item.0);
                if if mode == Collinear::Include {
                    turn < 0
                } else {
                    turn <= 0
                } {
                    upper.pop();
                } else {
                    break;
                }
            }
            upper.push(item);
        }
        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower
    };

    let vertices: Vec<Point> = selected.iter().map(|(p, _)| *p).collect();
    let source_indices: Vec<usize> = selected.iter().map(|(_, i)| *i).collect();
    let measures = if vertices.len() >= 3 {
        measures_for(&vertices)
    } else if vertices.len() == 2 {
        let length = edge_length_milli(vertices[0], vertices[1]);
        Measures {
            signed_area2: 0,
            area2: 0,
            orientation: Orientation::Degenerate,
            edge_milli: vec![length, length],
            perimeter_milli: length * 2,
            longest_edge: Some(0),
        }
    } else {
        Measures {
            signed_area2: 0,
            area2: 0,
            orientation: Orientation::Degenerate,
            edge_milli: vec![0; vertices.len()],
            perimeter_milli: 0,
            longest_edge: if vertices.is_empty() { None } else { Some(0) },
        }
    };

    Ok(Hull {
        vertices,
        source_indices,
        area2: measures.area2,
        perimeter_milli: measures.perimeter_milli,
    })
}

fn segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool {
    let ab_c = cross(a, b, c);
    let ab_d = cross(a, b, d);
    let cd_a = cross(c, d, a);
    let cd_b = cross(c, d, b);

    if ab_c == 0 && on_segment(a, b, c) {
        return true;
    }
    if ab_d == 0 && on_segment(a, b, d) {
        return true;
    }
    if cd_a == 0 && on_segment(c, d, a) {
        return true;
    }
    if cd_b == 0 && on_segment(c, d, b) {
        return true;
    }
    (ab_c > 0) != (ab_d > 0) && (cd_a > 0) != (cd_b > 0)
}

pub fn validate(points: &[Point]) -> Result<Validated, GeomError> {
    let vertices = prepare(points)?;
    let n = vertices.len();

    let mut duplicate: Option<(usize, usize)> = None;
    for i in 0..n {
        for j in (i + 1)..n {
            if vertices[i] == vertices[j] {
                let pair = (i, j);
                if duplicate.map_or(true, |old| pair < old) {
                    duplicate = Some(pair);
                }
            }
        }
    }
    if let Some((first, second)) = duplicate {
        return Err(GeomError::Duplicate { first, second });
    }

    let area = signed_area2(&vertices);
    if area == 0 {
        return Err(GeomError::ZeroArea);
    }

    for i in 0..n {
        let prev = vertices[(i + n - 1) % n];
        let curr = vertices[i];
        let next = vertices[(i + 1) % n];
        if cross(prev, curr, next) == 0 {
            let dot = (prev.x as i128 - curr.x as i128) * (next.x as i128 - curr.x as i128)
                + (prev.y as i128 - curr.y as i128) * (next.y as i128 - curr.y as i128);
            if dot > 0 {
                return Err(GeomError::AdjacentOverlap { vertex: i });
            }
        }
    }

    for i in 0..n {
        for j in (i + 1)..n {
            if j == i + 1 || (i == 0 && j == n - 1) {
                continue;
            }
            if segments_intersect(
                vertices[i],
                vertices[(i + 1) % n],
                vertices[j],
                vertices[(j + 1) % n],
            ) {
                return Err(GeomError::Intersection {
                    edge_a: i,
                    edge_b: j,
                });
            }
        }
    }

    let minimum = (0..n).min_by_key(|&i| vertices[i]).unwrap();
    let ccw = area > 0;
    let mut indices = Vec::with_capacity(n);
    for step in 0..n {
        let index = if ccw {
            (minimum + step) % n
        } else {
            (minimum + n - step % n) % n
        };
        indices.push(index);
    }
    let canonical: Vec<Point> = indices.iter().map(|&i| vertices[i]).collect();
    let source_indices = indices;
    let measures = measures_for(&canonical);

    Ok(Validated {
        vertices: canonical,
        source_indices,
        measures,
    })
}

fn main() {}
```
