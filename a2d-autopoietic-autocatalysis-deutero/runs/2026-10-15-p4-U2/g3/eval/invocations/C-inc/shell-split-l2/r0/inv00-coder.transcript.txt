    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },

enum Expansion {
    Simple { end: usize, value: String },
    Braced { end: usize, value: String },
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> &'a str {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
        .unwrap_or("")
}

        return Ok(None);

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.chars();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut pos = 0;
    let mut quote: Option<(char, usize)> = None;

    while pos < input.len() {
        let c = input[pos..].chars().next().unwrap();
        let next_pos = pos + c.len_utf8();

        if let Some((q, _)) = quote {
            if q == '\'' {
                if c == '\'' {
                    quote = None;
                } else {
                    word.push(c);
                }
                pos = next_pos;
                continue;
            }

            if c == '"' {
                quote = None;
                pos = next_pos;
                continue;
            }
            if c == '\\' {
                if next_pos == input.len() {
                    return Err(SplitError::TrailingEscape { pos });
                }
                let next = input[next_pos..].chars().next().unwrap();
                let after_next = next_pos + next.len_utf8();
                if matches!(next, '"' | '\\' | '$') {
                    word.push(next);
                    pos = after_next;
                } else if next == '\n' {
                    pos = after_next;
                } else {
                    word.push('\\');
                    pos = next_pos;
                }
                continue;
            }
            if c == '$' {
                if let Some(expansion) = expand(input, pos, env)? {
                    match expansion {
                        Expansion::Simple { end, value } | Expansion::Braced { end, value } => {
                            word.push_str(&value);
                            active = true;
                            pos = end;
                            continue;
                        }
                    }
                }
            }
            word.push(c);
            pos = next_pos;
            continue;
        }

        if c == ' ' || c == '\t' || c == '\r' || c == '\n' {
            if active {
                words.push(std::mem::take(&mut word));
                active = false;
            }
            pos = next_pos;
            continue;
        }

        if c == '\'' || c == '"' {
            quote = Some((c, pos));
            active = true;
            pos = next_pos;
            continue;
        }

        if c == '\\' {
            if next_pos == input.len() {
                return Err(SplitError::TrailingEscape { pos });
            }
            let next = input[next_pos..].chars().next().unwrap();
            let after_next = next_pos + next.len_utf8();
            if next != '\n' {
                word.push(next);
                active = true;
            }
            pos = after_next;
            continue;
        }

        if c == '$' {
            if let Some(expansion) = expand(input, pos, env)? {
                match expansion {
                    Expansion::Simple { end, value } | Expansion::Braced { end, value } => {
                        word.push_str(&value);
                        active = true;
                        pos = end;
                        continue;
                    }
                }
            }
        }

        word.push(c);
        active = true;
        pos = next_pos;
    }

        if q == '\'' {
            return Err(SplitError::UnterminatedSingle { pos: opening });
        }
        return Err(SplitError::UnterminatedDouble { pos: opening });
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
