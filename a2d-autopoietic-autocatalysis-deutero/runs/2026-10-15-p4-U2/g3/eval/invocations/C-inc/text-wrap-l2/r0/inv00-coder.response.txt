#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let mut normalized = String::with_capacity(input.len());
    let mut chars = input.chars().peekable();
    while let Some(ch) = chars.next() {
        if ch == '\r' {
            if chars.peek() == Some(&'\n') {
                chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let scalars: Vec<char> = normalized.chars().collect();
    let mut result = Vec::new();
    let mut segment_start = 0;
    let mut i = 0;

    while i < scalars.len() {
        if scalars[i] == '\n' {
            let mut end = i + 1;
            while end < scalars.len() && scalars[end] == '\n' {
                end += 1;
            }
            if end - i >= 2 {
                wrap_segment(&scalars[segment_start..i], options, &mut result);
                result.extend(std::iter::repeat(String::new()).take(end - i - 1));
                segment_start = end;
            }
            i = end;
        } else {
            i += 1;
        }
    }
    wrap_segment(&scalars[segment_start..], options, &mut result);

    Ok(result)
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn wrap_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let words: Vec<Vec<char>> = segment
        .split(|ch| is_separator(*ch))
        .filter(|part| !part.is_empty())
        .map(|part| part.to_vec())
        .collect();

    if words.is_empty() {
        return;
    }

    let mut first_line = true;
    let mut content = String::new();
    let mut content_len = 0usize;

    for word in words {
        let mut offset = 0usize;

        while offset < word.len() {
            let indent = if first_line {
                options.first_indent
            } else {
                options.rest_indent
            };
            let capacity = options.width - indent;
            let suffix_len = word.len() - offset;

            if suffix_len <= capacity {
                let separator_len = usize::from(content_len > 0);
                if content_len + separator_len + suffix_len <= capacity {
                    if separator_len > 0 {
                        content.push(' ');
                        content_len += 1;
                    }
                    for &ch in &word[offset..] {
                        content.push(ch);
                    }
                    content_len += suffix_len;
                    offset = word.len();
                } else {
                    emit_line(
                        options,
                        first_line,
                        &mut content,
                        output,
                    );
                    content_len = 0;
                    first_line = false;
                }
            } else if content_len == 0 {
                for &ch in &word[offset..offset + capacity] {
                    content.push(ch);
                }
                content_len = capacity;
                offset += capacity;
                emit_line(options, first_line, &mut content, output);
                content_len = 0;
                first_line = false;
            } else {
                let available = capacity - content_len - 1;
                if available >= 2 {
                    content.push(' ');
                    content_len += 1;
                    for &ch in &word[offset..offset + available] {
                        content.push(ch);
                    }
                    content_len += available;
                    offset += available;
                    emit_line(options, first_line, &mut content, output);
                    content_len = 0;
                    first_line = false;
                } else {
                    emit_line(options, first_line, &mut content, output);
                    content_len = 0;
                    first_line = false;
                }
            }
        }
    }

    if content_len > 0 {
        emit_line(options, first_line, &mut content, output);
    }
}

fn emit_line(
    options: &WrapOptions,
    first_line: bool,
    content: &mut String,
    output: &mut Vec<String>,
) {
    let indent = if first_line {
        options.first_indent
    } else {
        options.rest_indent
    };
    let mut line = " ".repeat(indent);
    line.push_str(content);
    output.push(line);
    content.clear();
}

fn main() {}
