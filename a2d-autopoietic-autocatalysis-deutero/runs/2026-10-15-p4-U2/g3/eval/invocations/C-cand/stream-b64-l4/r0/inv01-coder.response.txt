#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
    pub strict: bool,
    pub max_symbols: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
    NonCanonical { offset: usize },
    MixedAlphabet { offset: usize },
    Overlong { offset: usize, limit: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 64] {
    const STANDARD: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    const URL_SAFE: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";
    match alphabet {
        Alphabet::Standard => STANDARD,
        Alphabet::UrlSafe => URL_SAFE,
    }
}

fn lookup(byte: u8, alphabet: Alphabet, strict: bool, offset: usize) -> Result<u8, Error> {
    if let Some(index) = alphabet_bytes(alphabet).iter().position(|&b| b == byte) {
        return Ok(index as u8);
    }
    if strict {
        let foreign = match alphabet {
            Alphabet::Standard => byte == b'-' || byte == b'_',
            Alphabet::UrlSafe => byte == b'+' || byte == b'/',
        };
        if foreign {
            return Err(Error::MixedAlphabet { offset });
        }
    }
    Err(Error::InvalidByte { offset, byte })
}

fn append_wrapped_symbol(out: &mut String, symbol: u8, options: Options, row: &mut usize) {
    if let Whitespace::Wrapped { width, ending } = options.whitespace {
        if *row == width {
            match ending {
                LineEnding::Lf => out.push('\n'),
                LineEnding::CrLf => out.push_str("\r\n"),
            }
            *row = 0;
        }
        out.push(symbol as char);
        *row += 1;
    } else {
        out.push(symbol as char);
    }
}

fn encode_complete_group(input: &[u8], options: Options, out: &mut String, row: &mut usize) {
    let table = alphabet_bytes(options.alphabet);
    let vals = [
        input[0] >> 2,
        ((input[0] & 0x03) << 4) | (input[1] >> 4),
        ((input[1] & 0x0f) << 2) | (input[2] >> 6),
        input[2] & 0x3f,
    ];
    for value in vals {
        append_wrapped_symbol(out, table[value as usize], options, row);
    }
}

fn encode_tail(tail: &[u8], options: Options, out: &mut String, row: &mut usize) {
    if tail.is_empty() {
        return;
    }

    let table = alphabet_bytes(options.alphabet);
    let first = tail[0] >> 2;
    let second = ((tail[0] & 0x03) << 4) | (tail.get(1).copied().unwrap_or(0) >> 4);

    append_wrapped_symbol(out, table[first as usize], options, row);
    append_wrapped_symbol(out, table[second as usize], options, row);

    if tail.len() == 1 {
        if options.padding != Padding::Forbidden {
            append_wrapped_symbol(out, b'=', options, row);
            append_wrapped_symbol(out, b'=', options, row);
        }
    } else {
        let third = (tail[1] & 0x0f) << 2;
        append_wrapped_symbol(out, table[third as usize], options, row);
        if options.padding != Padding::Forbidden {
            append_wrapped_symbol(out, b'=', options, row);
        }
    }
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut result = encoder.push(input)?;
    result.push_str(&encoder.finish()?);
    Ok(result)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut result = decoder.push(input)?;
    result.extend(decoder.finish()?);
    Ok(result)
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    row: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: Vec::new(),
            row: 0,
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        if input.is_empty() {
            return Ok(String::new());
        }

        let mut combined = Vec::with_capacity(self.tail.len() + input.len());
        combined.extend_from_slice(&self.tail);
        combined.extend_from_slice(input);
        self.tail.clear();

        let complete_len = combined.len() / 3 * 3;
        let mut output = String::new();
        for group in combined[..complete_len].chunks_exact(3) {
            encode_complete_group(group, self.options, &mut output, &mut self.row);
        }
        self.tail.extend_from_slice(&combined[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        let mut output = String::new();
        encode_tail(&self.tail, self.options, &mut output, &mut self.row);
        self.tail.clear();
        Ok(output)
    }
}

#[derive(Clone, Copy)]
struct Digit {
    value: u8,
    offset: usize,
}

#[derive(Clone, Copy)]
enum FrameState {
    Row,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    offset: usize,
    symbols: usize,
    group: Vec<Digit>,
    pending_second_pad: Option<usize>,
    terminal_padding: bool,
    frame_state: FrameState,
    row_symbols: usize,
    finished: bool,
    sticky: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            offset: 0,
            symbols: 0,
            group: Vec::with_capacity(4),
            pending_second_pad: None,
            terminal_padding: false,
            frame_state: FrameState::Row,
            row_symbols: 0,
            finished: false,
            sticky: None,
        })
    }

    fn is_ignored(&self, byte: u8) -> bool {
        self.options.whitespace == Whitespace::IgnoreAscii
            && matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
    }

    fn emit_group(&mut self, output: &mut Vec<u8>) -> Result<(), Error> {
        if self.group.len() != 4 {
            return Ok(());
        }

        let a = self.group[0].value;
        let b = self.group[1].value;

        if self.group[2].value == 64 {
            if self.options.strict && b & 0x0f != 0 {
                return Err(Error::NonCanonical {
                    offset: self.group[1].offset,
                });
            }
            output.push((a << 2) | (b >> 4));
            self.terminal_padding = true;
        } else if self.group[3].value == 64 {
            let c = self.group[2].value;
            if self.options.strict && c & 0x03 != 0 {
                return Err(Error::NonCanonical {
                    offset: self.group[2].offset,
                });
            }
            output.push((a << 2) | (b >> 4));
            output.push((b << 4) | (c >> 2));
            self.terminal_padding = true;
        } else {
            let c = self.group[2].value;
            let d = self.group[3].value;
            output.push((a << 2) | (b >> 4));
            output.push((b << 4) | (c >> 2));
            output.push((c << 6) | d);
        }

        self.group.clear();
        self.pending_second_pad = None;
        Ok(())
    }

    fn scan_symbol(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if let Some(limit) = self.options.max_symbols {
            if self.symbols >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.symbols += 1;

        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if let Some(pad_offset) = self.pending_second_pad {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.group.push(Digit { value: 64, offset });
            let _ = pad_offset;
            return self.emit_group(output);
        }

        if byte == b'=' {
            match self.group.len() {
                0 | 1 => Err(Error::BadPadding { offset }),
                2 => {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset });
                    }
                    self.group.push(Digit { value: 64, offset });
                    self.pending_second_pad = Some(offset);
                    Ok(())
                }
                3 => {
                    if self.options.padding == Padding::Forbidden {
                        return Err(Error::BadPadding { offset });
                    }
                    self.group.push(Digit { value: 64, offset });
                    self.emit_group(output)
                }
                _ => Err(Error::BadPadding { offset }),
            }
        } else {
            let value = lookup(byte, self.options.alphabet, self.options.strict, offset)?;
            self.group.push(Digit { value, offset });
            if self.group.len() == 4 {
                self.emit_group(output)?;
            }
            Ok(())
        }
    }

    fn frame_byte(&mut self, byte: u8, offset: usize) -> Result<bool, Error> {
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            match self.frame_state {
                FrameState::NeedLf { .. } => {
                    if byte != b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    let cr_offset = match self.frame_state {
                        FrameState::NeedLf { cr_offset } => cr_offset,
                        _ => unreachable!(),
                    };
                    self.frame_state = FrameState::NeedSymbol {
                        separator_offset: cr_offset,
                    };
                    return Ok(false);
                }
                FrameState::NeedSymbol { .. } => {
                    self.frame_state = FrameState::Row;
                }
                FrameState::Row => {}
            }

            if self.row_symbols == width {
                match ending {
                    LineEnding::Lf => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.frame_state = FrameState::NeedSymbol {
                            separator_offset: offset,
                        };
                    }
                    LineEnding::CrLf => {
                        if byte != b'\r' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.frame_state = FrameState::NeedLf { cr_offset: offset };
                    }
                }
                self.row_symbols = 0;
                return Ok(false);
            }

            if matches!(byte, b'\r' | b'\n' | b' ' | b'\t' | 11 | 12) {
                return Err(Error::BadWrap { offset });
            }
            self.row_symbols += 1;
            Ok(true)
        } else if self.is_ignored(byte) {
            Ok(false)
        } else {
            Ok(true)
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;

            let result = match self.frame_byte(byte, offset) {
                Ok(false) => Ok(()),
                Ok(true) => self.scan_symbol(byte, offset, &mut output),
                Err(error) => Err(error),
            };
            if let Err(error) = result {
                self.sticky = Some(error.clone());
                return Err(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Whitespace::Wrapped { .. } = self.options.whitespace {
            match self.frame_state {
                FrameState::NeedLf { cr_offset } => {
                    let error = Error::BadWrap { offset: cr_offset };
                    self.sticky = Some(error.clone());
                    return Err(error);
                }
                FrameState::NeedSymbol { separator_offset } => {
                    let error = Error::BadWrap {
                        offset: separator_offset,
                    };
                    self.sticky = Some(error.clone());
                    return Err(error);
                }
                FrameState::Row => {}
            }
        }

        if let Some(offset) = self.pending_second_pad {
            let error = Error::BadPadding { offset };
            self.sticky = Some(error.clone());
            return Err(error);
        }

        let mut output = Vec::new();
        match self.group.len() {
            0 => {}
            1 => {
                let error = Error::Truncated {
                    offset: self.group[0].offset,
                };
                self.sticky = Some(error.clone());
                return Err(error);
            }
            2 | 3 => {
                if self.options.padding == Padding::Required {
                    let error = Error::Truncated {
                        offset: self.group[0].offset,
                    };
                    self.sticky = Some(error.clone());
                    return Err(error);
                }

                if self.options.strict {
                    let offending = if self.group.len() == 2 {
                        (self.group[1].value & 0x0f != 0).then_some(self.group[1].offset)
                    } else {
                        (self.group[2].value & 0x03 != 0).then_some(self.group[2].offset)
                    };
                    if let Some(offset) = offending {
                        let error = Error::NonCanonical { offset };
                        self.sticky = Some(error.clone());
                        return Err(error);
                    }
                }

                let a = self.group[0].value;
                let b = self.group[1].value;
                output.push((a << 2) | (b >> 4));
                if self.group.len() == 3 {
                    let c = self.group[2].value;
                    output.push((b << 4) | (c >> 2));
                }
            }
            _ => {}
        }

        self.group.clear();
        self.finished = true;
        Ok(output)
    }
}

fn main() {}
