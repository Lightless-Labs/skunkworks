#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append_literal(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_expansion(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append_literal(word, c, active_glob);
    }
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current: Option<Word> = None;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_literal(current.as_mut().unwrap(), c, false);
                }
                i = next_i;
            }
            Quote::Double(open_pos) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    let Some(next) = input[next_i..].chars().next() else {
                        return Err(SplitError::TrailingEscape { pos: i });
                    };
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_literal(current.as_mut().unwrap(), next, false);
                        }
                        i = after_next;
                    } else {
                        append_literal(current.as_mut().unwrap(), '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (end, value) = parse_braced(input, i, env)?;
                        append_expansion(current.as_mut().unwrap(), &value, false);
                        i = end;
                    } else if let Some(name_start) = input[next_i..].chars().next().filter(|x| is_name_start(*x)) {
                        let _ = name_start;
                        let mut end = next_i;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let value = lookup(&input[next_i..end], env).unwrap_or("");
                        append_expansion(current.as_mut().unwrap(), value, false);
                        i = end;
                    } else {
                        append_literal(current.as_mut().unwrap(), '$', false);
                        i = next_i;
                    }
                } else {
                    append_literal(current.as_mut().unwrap(), c, false);
                    i = next_i;
                }

                let _ = open_pos;
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if let Some(word) = current.take() {
                        words.push(word);
                    }
                    i = next_i;
                } else if c == '\'' {
                    current.get_or_insert_with(|| Word { text: String::new(), glob: Vec::new() });
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    current.get_or_insert_with(|| Word { text: String::new(), glob: Vec::new() });
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    let Some(next) = input[next_i..].chars().next() else {
                        return Err(SplitError::TrailingEscape { pos: i });
                    };
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        current.get_or_insert_with(|| Word { text: String::new(), glob: Vec::new() });
                        append_literal(current.as_mut().unwrap(), next, false);
                        i = after_next;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        current.get_or_insert_with(|| Word { text: String::new(), glob: Vec::new() });
                        let (end, value) = parse_braced(input, i, env)?;
                        append_expansion(current.as_mut().unwrap(), &value, true);
                        i = end;
                    } else if input[next_i..].chars().next().is_some_and(is_name_start) {
                        current.get_or_insert_with(|| Word { text: String::new(), glob: Vec::new() });
                        let mut end = next_i;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_continue(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let value = lookup(&input[next_i..end], env).unwrap_or("");
                        append_expansion(current.as_mut().unwrap(), value, true);
                        i = end;
                    } else {
                        current.get_or_insert_with(|| Word { text: String::new(), glob: Vec::new() });
                        append_literal(current.as_mut().unwrap(), '$', true);
                        i = next_i;
                    }
                } else {
                    current.get_or_insert_with(|| Word { text: String::new(), glob: Vec::new() });
                    append_literal(current.as_mut().unwrap(), c, true);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }
    if let Some(word) = current {
        words.push(word);
    }
    Ok(words)
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let content_start = dollar + 2;
    let mut cursor = content_start;
    let mut close = None;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += 1;
            if cursor < input.len() {
                let escaped = input[cursor..].chars().next().unwrap();
                cursor += escaped.len_utf8();
            }
        } else if c == '}' {
            close = Some(cursor);
            break;
        } else {
            cursor += c.len_utf8();
        }
    }

    let Some(close_pos) = close else {
        return Err(SplitError::UnterminatedExpansion { pos: dollar });
    };

    let contents = &input[content_start..close_pos];
    let mut name_end = 0;
    for (offset, c) in contents.char_indices() {
        if (offset == 0 && is_name_start(c)) || (offset > 0 && is_name_continue(c)) {
            name_end = offset + c.len_utf8();
        } else {
            break;
        }
    }

    if name_end == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_end];
    let rest = &contents[name_end..];
    let value = if rest.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default) = rest.strip_prefix(":-") {
        match lookup(name, env) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close_pos + 1, value))
}

fn main() {}
