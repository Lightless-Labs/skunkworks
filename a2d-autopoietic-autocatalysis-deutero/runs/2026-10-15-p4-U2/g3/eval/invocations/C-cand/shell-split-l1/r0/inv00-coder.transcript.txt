pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single { opening: usize },
    Double { opening: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().expect("cursor is in bounds");
        let next_cursor = cursor + ch.len_utf8();

        match quote {
            Quote::None => match ch {
                '\'' => {
                    quote = Quote::Single { opening: cursor };
                    active = true;
                    cursor = next_cursor;
                }
                '"' => {
                    quote = Quote::Double { opening: cursor };
                    active = true;
                    cursor = next_cursor;
                }
                '\\' => {
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next_cursor..]
                        .chars()
                        .next()
                        .expect("next cursor is in bounds");
                    let after_escaped = next_cursor + escaped.len_utf8();
                    if escaped != '\n' {
                        word.push(escaped);
                        active = true;
                    }
                    cursor = after_escaped;
                }
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    cursor = next_cursor;
                }
                _ => {
                    word.push(ch);
                    active = true;
                    cursor = next_cursor;
                }
            },
            Quote::Single { opening } => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
                let _ = opening;
                cursor = next_cursor;
            }
            Quote::Double { opening } => match ch {
                '"' => {
                    quote = Quote::None;
                    cursor = next_cursor;
                }
                '\\' => {
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next_cursor..]
                        .chars()
                        .next()
                        .expect("next cursor is in bounds");
                    let after_escaped = next_cursor + escaped.len_utf8();
                    match escaped {
                        '"' | '\\' | '$' => word.push(escaped),
                        '\n' => {}
                        _ => {
                            word.push('\\');
                            word.push(escaped);
                        }
                    }
                    cursor = after_escaped;
                }
                _ => {
                    word.push(ch);
                    cursor = next_cursor;
                }
            },

            return Err(SplitError::UnterminatedSingle { pos: opening });
        }
        Quote::Double { opening } => {
            return Err(SplitError::UnterminatedDouble { pos: opening });
        }
        Quote::None => {}
    }

    }
