#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

pub fn encode(input: &[u8]) -> String {
    let mut output = String::with_capacity(input.len().div_ceil(3) * 4);
    let mut chunks = input.chunks_exact(3);

    for chunk in &mut chunks {
        let n = ((chunk[0] as u32) << 16)
            | ((chunk[1] as u32) << 8)
            | (chunk[2] as u32);
        output.push(ALPHABET[((n >> 18) & 0x3f) as usize] as char);
        output.push(ALPHABET[((n >> 12) & 0x3f) as usize] as char);
        output.push(ALPHABET[((n >> 6) & 0x3f) as usize] as char);
        output.push(ALPHABET[(n & 0x3f) as usize] as char);
    }

    match chunks.remainder() {
        [a] => {
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
            output.push('=');
            output.push('=');
        }
        [a, b] => {
            output.push(ALPHABET[(a >> 2) as usize] as char);
            output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
            output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
            output.push('=');
        }
        _ => {}
    }

    output
}

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut output = Vec::with_capacity(input.len() / 4 * 3);
    let mut values = [0u8; 4];
    let mut offsets = [0usize; 4];
    let mut position = 0usize;
    let mut group_start = 0usize;
    let mut pending_padding: Option<usize> = None;
    let mut finished = false;

    for (offset, &byte) in input.iter().enumerate() {
        if finished {
            return Err(Error::BadPadding { offset });
        }

        if let Some(padding_offset) = pending_padding {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            offsets[3] = offset;
            output.push((values[0] << 2) | (values[1] >> 4));
            finished = true;
            pending_padding = None;
            let _ = padding_offset;
            continue;
        }

        if position < 2 && byte == b'=' {
            return Err(Error::BadPadding { offset });
        }

        if position == 2 && byte == b'=' {
            pending_padding = Some(offset);
            continue;
        }

        if position == 3 && byte == b'=' {
            output.push((values[0] << 2) | (values[1] >> 4));
            output.push((values[1] << 4) | (values[2] >> 2));
            finished = true;
            continue;
        }

        let value = digit(byte).ok_or(Error::InvalidByte { offset, byte })?;
        if position == 0 {
            group_start = offset;
        }
        values[position] = value;
        offsets[position] = offset;
        position += 1;

        if position == 4 {
            output.push((values[0] << 2) | (values[1] >> 4));
            output.push((values[1] << 4) | (values[2] >> 2));
            output.push((values[2] << 6) | values[3]);
            position = 0;
        }
    }

    if let Some(offset) = pending_padding {
        return Err(Error::BadPadding { offset });
    }
    if position != 0 {
        return Err(Error::Truncated { offset: group_start });
    }

    Ok(output)
}

fn main() {}
