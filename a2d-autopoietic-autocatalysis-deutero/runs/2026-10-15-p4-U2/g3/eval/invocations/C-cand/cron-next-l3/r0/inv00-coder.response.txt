#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    allowed: [Vec<u8>; 5],
    unrestricted_days: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_error(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn parse_number(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(parse_error(field, offset, 0));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(parse_error(field, offset, 1));
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| parse_error(field, offset, 2))?;
    }
    Ok(value)
}

fn name_value(s: &str, field: Field) -> Option<u32> {
    let upper = s.to_ascii_uppercase();
    match field {
        Field::Month => match upper.as_str() {
            "JAN" => Some(1),
            "FEB" => Some(2),
            "MAR" => Some(3),
            "APR" => Some(4),
            "MAY" => Some(5),
            "JUN" => Some(6),
            "JUL" => Some(7),
            "AUG" => Some(8),
            "SEP" => Some(9),
            "OCT" => Some(10),
            "NOV" => Some(11),
            "DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_str() {
            "SUN" => Some(0),
            "MON" => Some(1),
            "TUE" => Some(2),
            "WED" => Some(3),
            "THU" => Some(4),
            "FRI" => Some(5),
            "SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

fn parse_atom(s: &str, field: Field, offset: usize) -> Result<(u32, bool), ParseError> {
    if s.as_bytes().iter().all(u8::is_ascii_digit) && !s.is_empty() {
        Ok((parse_number(s, field, offset)?, true))
    } else if let Some(value) = name_value(s, field) {
        Ok((value, false))
    } else {
        Err(parse_error(field, offset, 0))
    }
}

fn parse_item(
    item: &str,
    field: Field,
    offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let (min, max) = bounds(field);
    let width = max - min + 1;
    let bytes = item.as_bytes();

    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(parse_error(field, offset, 0));
    }

    let (base, step_text, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), offset + slash + 1)
    } else {
        (item, None, offset)
    };

    let is_star = base == "*";
    let hyphens: Vec<usize> = base
        .as_bytes()
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();
    if hyphens.len() > 1 || (is_star && !hyphens.is_empty()) {
        return Err(parse_error(field, offset, 0));
    }

    let (left, right, right_offset) = if let Some(&hyphen) = hyphens.first() {
        let left = &base[..hyphen];
        let right = &base[hyphen + 1..];
        if left.is_empty() || right.is_empty() || left == "*" || right == "*" {
            return Err(parse_error(field, offset, 0));
        }
        (left, Some(right), offset + hyphen + 1)
    } else if is_star {
        ("*", None, offset)
    } else {
        (base, None, offset)
    };

    if step_text.is_some_and(str::is_empty) {
        return Err(parse_error(field, offset, 0));
    }

    let mut start = min;
    let mut end = max;
    let mut start_numeric = false;
    let mut end_numeric = false;
    let mut start_offset = offset;
    let mut end_offset = right_offset;

    if !is_star {
        let (value, numeric) = parse_atom(left, field, offset)?;
        start = value;
        start_numeric = numeric;
        if let Some(right) = right {
            let (value, numeric) = parse_atom(right, field, right_offset)?;
            end = value;
            end_numeric = numeric;
        } else {
            end = start;
        }
    }

    for (value, numeric, number_offset) in [
        (start, start_numeric, start_offset),
        (end, end_numeric, end_offset),
    ] {
        if numeric {
            if value < min || value > max {
                return Err(parse_error(field, number_offset, 3));
            }
        }
    }
    if start > end {
        return Err(parse_error(field, end_offset, 4));
    }

    let step = if let Some(text) = step_text {
        let value = parse_number(text, field, step_offset)?;
        if value == 0 || value > width {
            return Err(parse_error(field, step_offset, 5));
        }
        value
    } else {
        1
    };

    if step_text.is_none() && !is_star && right.is_none() {
        selected[start as usize] = true;
        return Ok(());
    }

    let mut value = start;
    while value <= end {
        selected[value as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }

    let mut selected: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut unrestricted_days = [false; 2];

    for (index, (&field, text)) in FIELDS.iter().zip(pieces.iter()).enumerate() {
        for (offset, &byte) in text.as_bytes().iter().enumerate() {
            if !(byte.is_ascii_digit()
                || byte.is_ascii_alphabetic()
                || matches!(byte, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        if index == 2 {
            unrestricted_days[0] = *text == "*";
        } else if index == 4 {
            unrestricted_days[1] = *text == "*";
        }

        let mut values = vec![false; 60];
        let mut item_offset = 0usize;
        for item in text.split(',') {
            if item.is_empty() {
                return Err(ParseError::EmptyItem {
                    field,
                    offset: item_offset,
                });
            }
            parse_item(item, field, item_offset, &mut values)?;
            item_offset += item.len() + 1;
        }

        let (min, max) = bounds(field);
        selected[index] = (min..=max)
            .filter(|&v| values[v as usize])
            .map(|v| v as u8)
            .collect();
    }

    Ok(Cron {
        allowed: selected,
        unrestricted_days,
    })
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(at: CivilDateTime) -> u8 {
    let mut days = 0u32;
    for year in 2000..at.year {
        days += if is_leap(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        days += days_in_month(at.year, month) as u32;
    }
    days += (at.day - 1) as u32;
    ((6 + days) % 7) as u8
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.allowed[field_index(field)].clone()
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.allowed[0].contains(&at.minute)
            || !self.allowed[1].contains(&at.hour)
            || !self.allowed[3].contains(&at.month)
        {
            return false;
        }

        let dom = self.allowed[2].contains(&at.day);
        let dow = self.allowed[4].contains(&weekday(at));
        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut candidate = after;
        loop {
            if candidate.year == 2399
                && candidate.month == 12
                && candidate.day == 31
                && candidate.hour == 23
                && candidate.minute == 59
            {
                return Ok(None);
            }

            if candidate.minute < 59 {
                candidate.minute += 1;
            } else {
                candidate.minute = 0;
                if candidate.hour < 23 {
                    candidate.hour += 1;
                } else {
                    candidate.hour = 0;
                    if candidate.day < days_in_month(candidate.year, candidate.month) {
                        candidate.day += 1;
                    } else {
                        candidate.day = 1;
                        if candidate.month < 12 {
                            candidate.month += 1;
                        } else {
                            candidate.month = 1;
                            candidate.year += 1;
                        }
                    }
                }
            }

            if self.matches_valid(candidate) {
                return Ok(Some(candidate));
            }
        }
    }
}

fn main() {}
