#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    text: String,
    consumed: usize,
    pieces: usize,
    gaps: usize,
}

#[derive(Clone)]
struct Layout {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn lex_consumed_greater(a: &[usize], b: &[usize]) -> bool {
    for (x, y) in a.iter().zip(b.iter()) {
        if x != y {
            return x > y;
        }
    }
    a.len() < b.len()
}

fn layout_better(a: &Layout, b: &Layout) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }

    let av: Vec<usize> = a.lines.iter().map(|line| line.consumed).collect();
    let bv: Vec<usize> = b.lines.iter().map(|line| line.consumed).collect();
    if av != bv {
        return lex_consumed_greater(&av, &bv);
    }

    let at: Vec<&str> = a.lines.iter().map(|line| line.text.as_str()).collect();
    let bt: Vec<&str> = b.lines.iter().map(|line| line.text.as_str()).collect();
    at < bt
}

fn append_chars(text: &mut String, chars: &[char]) {
    for &c in chars {
        text.push(c);
    }
}

fn make_line(
    words: &[Word],
    wi: usize,
    off: usize,
    nwi: usize,
    noff: usize,
    marker_end: bool,
    indent: usize,
    pieces: usize,
    consumed: usize,
) -> Line {
    let mut text = " ".repeat(indent);
    let mut gaps = 0;

    if marker_end {
        append_chars(&mut text, &words[wi].chars[off..noff]);
        text.push('-');
    } else if nwi == wi {
        append_chars(&mut text, &words[wi].chars[off..noff]);
    } else {
        let mut first_piece = true;
        for w in wi..nwi {
            if !first_piece {
                text.push(' ');
                gaps += 1;
            }
            let start = if w == wi { off } else { 0 };
            let end = if w + 1 == nwi {
                noff
            } else {
                words[w].chars.len()
            };
            append_chars(&mut text, &words[w].chars[start..end]);
            first_piece = false;
        }
    }

    Line {
        text,
        consumed,
        pieces,
        gaps,
    }
}

fn optimize_paragraph(words: &[Word], options: &WrapOptions) -> Layout {
    use std::collections::HashMap;

    fn solve(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut HashMap<(usize, usize, bool), Layout>,
    ) -> Layout {
        if wi == words.len() {
            return Layout {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
            };
        }

        if let Some(cached) = memo.get(&(wi, off, first)) {
            return cached.clone();
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let capacity = options.width - indent;
        let mut candidates: Vec<(usize, usize, bool, usize, usize, usize)> = Vec::new();

        let mut used = 0usize;
        let mut pieces = 0usize;
        let mut cursor_word = wi;
        let mut cursor_off = off;

        loop {
            if cursor_word >= words.len() {
                if used > 0 {
                    candidates.push((cursor_word, 0, false, used, pieces, used));
                }
                break;
            }

            let word = &words[cursor_word];
            let remaining = word.chars.len() - cursor_off;
            let prefix_space = usize::from(pieces > 0);
            if used + prefix_space >= capacity {
                break;
            }
            let available = capacity - used - prefix_space;

            for &marker in &word.markers {
                if marker > cursor_off && marker - cursor_off + 1 <= available {
                    let part = marker - cursor_off;
                    candidates.push((
                        cursor_word,
                        marker,
                        true,
                        used + prefix_space + part + 1,
                        pieces + 1,
                        used + part,
                    ));
                }
            }

            if remaining <= available {
                used += prefix_space + remaining;
                pieces += 1;
                cursor_word += 1;
                cursor_off = 0;
                candidates.push((cursor_word, 0, false, used, pieces, used));
                continue;
            }

            if pieces == 0 {
                let fits_marker = word.markers.iter().any(|&marker| {
                    marker > cursor_off && marker - cursor_off + 1 <= available
                });
                if !fits_marker && available > 0 {
                    candidates.push((
                        cursor_word,
                        cursor_off + available,
                        false,
                        available,
                        1,
                        available,
                    ));
                }
            }
            break;
        }

        let mut best: Option<Layout> = None;

        for (next_wi, next_off, marker_end, displayed, line_pieces, consumed) in candidates {
            if displayed == 0 || (next_wi == wi && next_off == off) {
                continue;
            }

            let next = solve(next_wi, next_off, false, words, options, memo);
            let final_line = next.lines.is_empty();
            let gap = options.width - (indent + displayed);
            let mut line_cost = (gap as u64) * (gap as u64);

            if final_line {
                line_cost /= 4;
            }
            if marker_end {
                line_cost += options.hyphen_cost as u64;
            }
            if line_pieces == 1 && !final_line {
                line_cost += 3;
            }

            let line = make_line(
                words,
                wi,
                off,
                next_wi,
                next_off,
                marker_end,
                indent,
                line_pieces,
                consumed,
            );
            let mut lines = Vec::with_capacity(next.lines.len() + 1);
            lines.push(line);
            lines.extend(next.lines);

            let candidate = Layout {
                cost: line_cost + next.cost,
                marker_lines: usize::from(marker_end) + next.marker_lines,
                lines,
            };

            if best
                .as_ref()
                .is_none_or(|current| layout_better(&candidate, current))
            {
                best = Some(candidate);
            }
        }

        let result = best.expect("raw splitting guarantees a legal continuation");
        memo.insert((wi, off, first), result.clone());
        result
    }

    solve(0, 0, true, words, options, &mut HashMap::new())
}

fn layout_segment(segment: &[char], options: &WrapOptions) -> Vec<String> {
    let mut words = Vec::<Word>::new();
    let mut current = Word {
        chars: Vec::new(),
        markers: Vec::new(),
    };

    for &c in segment {
        if is_separator(c) {
            if !current.chars.is_empty() {
                words.push(current);
                current = Word {
                    chars: Vec::new(),
                    markers: Vec::new(),
                };
            }
        } else if c == '~' {
            current.markers.push(current.chars.len());
        } else {
            current.chars.push(c);
        }
    }
    if !current.chars.is_empty() {
        words.push(current);
    }
    if words.is_empty() {
        return Vec::new();
    }

    let layout = optimize_paragraph(&words, options);
    let line_count = layout.lines.len();
    let mut result = Vec::with_capacity(line_count);

    for (line_index, line) in layout.lines.iter().enumerate() {
        let mut text = line.text.clone();

        if options.justify
            && line_index + 1 < line_count
            && line.pieces >= 2
            && line.gaps > 0
        {
            let extra = options.width - text.chars().count();
            let each = extra / line.gaps;
            let remainder = extra % line.gaps;
            let chars: Vec<char> = text.chars().collect();
            let indent = if line_index == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            let mut rebuilt = String::new();
            let mut gap_no = 0usize;

            for &c in &chars[indent..] {
                rebuilt.push(c);
                if c == ' ' {
                    let offset = (gap_no + line_index % line.gaps) % line.gaps;
                    let extra_for_gap = each + usize::from(offset < remainder);
                    for _ in 0..extra_for_gap {
                        rebuilt.push(' ');
                    }
                    gap_no += 1;
                }
            }
            text = rebuilt;
        }

        result.push(text);
    }

    result
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let bad_left = i == 0 || is_separator(chars[i - 1].1) || chars[i - 1].1 == '~';
            let bad_right =
                i + 1 == chars.len() || is_separator(chars[i + 1].1) || chars[i + 1].1 == '~';
            if bad_left || bad_right {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::with_capacity(chars.len());
    let mut i = 0usize;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 1;
            }
        } else {
            normalized.push(c);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut j = 0usize;

    while j < normalized.len() {
        if normalized[j] == '\n' {
            let start = j;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            let run = j - start;
            if run >= 2 {
                output.extend(layout_segment(&segment, options));
                segment.clear();
                for _ in 1..run {
                    output.push(String::new());
                }
            } else {
                segment.push('\n');
            }
        } else {
            segment.push(normalized[j]);
            j += 1;
        }
    }

    output.extend(layout_segment(&segment, options));
    Ok(output)
}

fn main() {}
