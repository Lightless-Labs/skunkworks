fn lex_consumed_greater(a: &[usize], b: &[usize]) -> bool {
    for (x, y) in a.iter().zip(b.iter()) {
        if x != y {
            return x > y;
        }
    }
    a.len() < b.len()

fn layout_better(a: &Layout, b: &Layout) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }

    let a_consumed: Vec<usize> = a.lines.iter().map(|line| line.consumed).collect();
    let b_consumed: Vec<usize> = b.lines.iter().map(|line| line.consumed).collect();
    if a_consumed != b_consumed {
        return lex_consumed_greater(&a_consumed, &b_consumed);

    let b_text: Vec<&str> = b.lines.iter().map(|line| line.text.as_str()).collect();

    start_word: usize,

fn optimize_paragraph(words: &[Word], options: &WrapOptions) -> Layout {
    use std::collections::HashMap;

    fn solve(
        word_index: usize,
        offset: usize,
        first_line: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut HashMap<(usize, usize, bool), Layout>,
    ) -> Layout {
        if word_index == words.len() {
            return Layout {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
            };
        }

        if let Some(result) = memo.get(&(word_index, offset, first_line)) {
            return result.clone();
        }

        let indent = if first_line {
            options.first_indent
        } else {
            options.rest_indent
        };
        let capacity = options.width - indent;

        // Candidate tuple: next word, next offset, marker-ended, displayed columns,
        // number of pieces, spelling scalars consumed.
        let mut candidates: Vec<(usize, usize, bool, usize, usize, usize)> = Vec::new();
        let mut used = 0usize;
        let mut pieces = 0usize;
        let mut cursor_word = word_index;
        let mut cursor_offset = offset;

        loop {
            if cursor_word >= words.len() {
                if used > 0 {
                    candidates.push((cursor_word, 0, false, used, pieces, used));
                }
                break;
            }

            let word = &words[cursor_word];
            let remaining = word.chars.len() - cursor_offset;
            let separator_width = usize::from(pieces > 0);
            if used + separator_width >= capacity {
                break;
            }
            let available = capacity - used - separator_width;

            for &marker in &word.markers {
                if marker > cursor_offset && marker < word.chars.len() {
                    let prefix_len = marker - cursor_offset;
                    if prefix_len + 1 <= available {
                        candidates.push((
                            cursor_word,
                            marker,
                            true,
                            used + separator_width + prefix_len + 1,
                            pieces + 1,
                            used + prefix_len,
                        ));
                    }
                }
            }

            if remaining <= available {
                used += separator_width + remaining;
                pieces += 1;
                cursor_word += 1;
                cursor_offset = 0;
                candidates.push((cursor_word, 0, false, used, pieces, used));
                continue;
            }

            if pieces == 0 && available > 0 {
                let marker_fits = word.markers.iter().any(|&marker| {
                    marker > cursor_offset
                        && marker < word.chars.len()
                        && marker - cursor_offset + 1 <= available
                });
                if !marker_fits {
                    candidates.push((
                        cursor_word,
                        cursor_offset + available,
                        false,
                        available,
                        1,
                        available,
                    ));
                }
            }
            break;
        }

        let mut best: Option<Layout> = None;

        for (next_word, next_offset, marker_end, displayed, line_pieces, consumed) in candidates {
            if displayed == 0 || (next_word == word_index && next_offset == offset) {
                continue;
            }

            let suffix = solve(next_word, next_offset, false, words, options, memo);
            let final_line = suffix.lines.is_empty();
            let gap = options.width - (indent + displayed);
            let squared_gap = (gap as u64) * (gap as u64);
            let mut line_cost = if final_line {
                squared_gap / 4
            } else {
                squared_gap
            };

            if marker_end {
                line_cost += options.hyphen_cost as u64;
            }
            if !final_line && line_pieces == 1 {
                line_cost += 3;
            }

            let line = make_line(
                words,
                word_index,
                offset,
                next_word,
                next_offset,
                marker_end,
                indent,
                line_pieces,
                consumed,
            );

            let mut lines = Vec::with_capacity(suffix.lines.len() + 1);
            lines.push(line);
            lines.extend(suffix.lines);

            let candidate = Layout {
                cost: line_cost + suffix.cost,
                marker_lines: usize::from(marker_end) + suffix.marker_lines,
                lines,
            };

            if best
                .as_ref()
                .is_none_or(|current| layout_better(&candidate, current))
            {
                best = Some(candidate);
            }
        }

        let result = best.expect("raw splitting guarantees a legal continuation");
        memo.insert((word_index, offset, first_line), result.clone());
        result
    }

    solve(0, 0, true, words, options, &mut HashMap::new())
}

fn layout_segment(segment: &[char], options: &WrapOptions) -> Vec<String> {
    let mut words = Vec::<Word>::new();
    let mut current = Word {
        chars: Vec::new(),
        markers: Vec::new(),
    };

    for &c in segment {
        if is_separator(c) {
            if !current.chars.is_empty() {
                words.push(current);
                current = Word {
                    chars: Vec::new(),
                    markers: Vec::new(),
                };
            }
        } else if c == '~' {
            current.markers.push(current.chars.len());
        } else {
            current.chars.push(c);
        }
    }
    if !current.chars.is_empty() {
        words.push(current);
    }
    if words.is_empty() {
        return Vec::new();
    }

    let layout = optimize_paragraph(&words, options);
    let line_count = layout.lines.len();
    let mut result = Vec::with_capacity(line_count);

    for (line_index, line) in layout.lines.iter().enumerate() {
        let mut text = line.text.clone();

        if options.justify
            && line_index + 1 < line_count
            && line.pieces >= 2
            && line.gaps > 0
        {
            let extra = options.width - text.chars().count();
            let each = extra / line.gaps;
            let remainder = extra % line.gaps;
            let chars: Vec<char> = text.chars().collect();
            let indent = if line_index == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };

            let mut rebuilt = String::new();
            let mut gap_number = 0usize;
            for &c in &chars[indent..] {
                rebuilt.push(c);
                if c == ' ' {
                    let relative_gap = (gap_number + line_index % line.gaps) % line.gaps;
                    let added = each + usize::from(relative_gap < remainder);
                    for _ in 0..added {
                        rebuilt.push(' ');
                    }
                    gap_number += 1;
                }
            }
            text = rebuilt;
        }

        result.push(text);
    }

    result
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let invalid_left =
                i == 0 || is_separator(chars[i - 1].1) || chars[i - 1].1 == '~';
            let invalid_right = i + 1 == chars.len()
                || is_separator(chars[i + 1].1)
                || chars[i + 1].1 == '~';
            if invalid_left || invalid_right {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::with_capacity(chars.len());
    let mut i = 0usize;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 1;
            }
        } else {
            normalized.push(c);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut index = 0usize;

    while index < normalized.len() {
        if normalized[index] == '\n' {
            let start = index;
            while index < normalized.len() && normalized[index] == '\n' {
                index += 1;
            }
            let run_len = index - start;
            if run_len >= 2 {
                output.extend(layout_segment(&segment, options));
                segment.clear();
                for _ in 1..run_len {
                    output.push(String::new());
                }
            } else {
                segment.push('\n');
            }
        } else {
            segment.push(normalized[index]);
            index += 1;
        }
    }

    output.extend(layout_segment(&segment, options));
    Ok(output)
}

    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    text: String,
    consumed: usize,
    pieces: usize,
    gaps: usize,
}

#[derive(Clone)]
struct Layout {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn lex_consumed_greater(a: &[usize], b: &[usize]) -> bool {
    for (x, y) in a.iter().zip(b.iter()) {
        if x != y {
            return x > y;
        }
    }
    a.len() < b.len()
}

fn layout_better(a: &Layout, b: &Layout) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }

    let a_consumed: Vec<usize> = a.lines.iter().map(|line| line.consumed).collect();
    let b_consumed: Vec<usize> = b.lines.iter().map(|line| line.consumed).collect();
    if a_consumed != b_consumed {
        return lex_consumed_greater(&a_consumed, &b_consumed);
    }

    let a_text: Vec<&str> = a.lines.iter().map(|line| line.text.as_str()).collect();
    let b_text: Vec<&str> = b.lines.iter().map(|line| line.text.as_str()).collect();
    a_text < b_text
}

fn make_line(
    words: &[Word],
    start_word: usize,
    start_offset: usize,
    next_word: usize,
    next_offset: usize,
    marker_end: bool,
    indent: usize,
    pieces: usize,
    consumed: usize,
) -> Line {
    let mut text = " ".repeat(indent);
    let mut gaps = 0;

    if marker_end {
        text.extend(words[start_word].chars[start_offset..next_offset].iter());
        text.push('-');
    } else if next_word == start_word {
        text.extend(words[start_word].chars[start_offset..next_offset].iter());
    } else {
        let mut first_piece = true;
        for word_index in start_word..next_word {
            if !first_piece {
                text.push(' ');
                gaps += 1;
            }
            let begin = if word_index == start_word {
                start_offset
            } else {
                0
            };
            let end = if word_index + 1 == next_word {
                next_offset
            } else {
                words[word_index].chars.len()
            };
            text.extend(words[word_index].chars[begin..end].iter());
            first_piece = false;
        }
    }

    Line {
        text,
        consumed,
        pieces,
        gaps,
    }
}

fn optimize_paragraph(words: &[Word], options: &WrapOptions) -> Layout {
    use std::collections::HashMap;

    fn solve(
        word_index: usize,
        offset: usize,
        first_line: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut HashMap<(usize, usize, bool), Layout>,
    ) -> Layout {
        if word_index == words.len() {
            return Layout {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
            };
        }

        if let Some(result) = memo.get(&(word_index, offset, first_line)) {
            return result.clone();
        }

        let indent = if first_line {
            options.first_indent
        } else {
            options.rest_indent
        };
        let capacity = options.width - indent;

        let mut candidates: Vec<(usize, usize, bool, usize, usize, usize)> = Vec::new();
        let mut used = 0usize;
        let mut pieces = 0usize;
        let mut cursor_word = word_index;
        let mut cursor_offset = offset;

        loop {
            if cursor_word >= words.len() {
                if used > 0 {
                    candidates.push((cursor_word, 0, false, used, pieces, used));
                }
                break;
            }

            let word = &words[cursor_word];
            let remaining = word.chars.len() - cursor_offset;
            let separator_width = usize::from(pieces > 0);
            if used + separator_width >= capacity {
                break;
            }
            let available = capacity - used - separator_width;

            for &marker in &word.markers {
                if marker > cursor_offset && marker < word.chars.len() {
                    let prefix_len = marker - cursor_offset;
                    if prefix_len + 1 <= available {
                        candidates.push((
                            cursor_word,
                            marker,
                            true,
                            used + separator_width + prefix_len + 1,
                            pieces + 1,
                            used + prefix_len,
                        ));
                    }
                }
            }

            if remaining <= available {
                used += separator_width + remaining;
                pieces += 1;
                cursor_word += 1;
                cursor_offset = 0;
                candidates.push((cursor_word, 0, false, used, pieces, used));
                continue;
            }

            if pieces == 0 && available > 0 {
                let marker_fits = word.markers.iter().any(|&marker| {
                    marker > cursor_offset
                        && marker < word.chars.len()
                        && marker - cursor_offset + 1 <= available
                });
                if !marker_fits {
                    candidates.push((
                        cursor_word,
                        cursor_offset + available,
                        false,
                        available,
                        1,
                        available,
                    ));
                }
            }
            break;
        }

        let mut best: Option<Layout> = None;

        for (next_word, next_offset, marker_end, displayed, line_pieces, consumed) in candidates {
            if displayed == 0 || (next_word == word_index && next_offset == offset) {
                continue;
            }

            let suffix = solve(next_word, next_offset, false, words, options, memo);
            let final_line = suffix.lines.is_empty();
            let gap = options.width - (indent + displayed);
            let squared_gap = (gap as u64) * (gap as u64);
            let mut line_cost = if final_line {
                squared_gap / 4
            } else {
                squared_gap
            };

            if marker_end {
                line_cost += options.hyphen_cost as u64;
            }
            if !final_line && line_pieces == 1 {
                line_cost += 3;
            }

            let line = make_line(
                words,
                word_index,
                offset,
                next_word,
                next_offset,
                marker_end,
                indent,
                line_pieces,
                consumed,
            );

            let mut lines = Vec::with_capacity(suffix.lines.len() + 1);
            lines.push(line);
            lines.extend(suffix.lines);

            let candidate = Layout {
                cost: line_cost + suffix.cost,
                marker_lines: usize::from(marker_end) + suffix.marker_lines,
                lines,
            };

            if best
                .as_ref()
                .is_none_or(|current| layout_better(&candidate, current))
            {
                best = Some(candidate);
            }
        }

        let result = best.expect("raw splitting guarantees a legal continuation");
        memo.insert((word_index, offset, first_line), result.clone());
        result
    }

    solve(0, 0, true, words, options, &mut HashMap::new())
}

fn layout_segment(segment: &[char], options: &WrapOptions) -> Vec<String> {
    let mut words = Vec::<Word>::new();
    let mut current = Word {
        chars: Vec::new(),
        markers: Vec::new(),
    };

    for &c in segment {
        if is_separator(c) {
            if !current.chars.is_empty() {
                words.push(current);
                current = Word {
                    chars: Vec::new(),
                    markers: Vec::new(),
                };
            }
        } else if c == '~' {
            current.markers.push(current.chars.len());
        } else {
            current.chars.push(c);
        }
    }
    if !current.chars.is_empty() {
        words.push(current);
    }
    if words.is_empty() {
        return Vec::new();
    }

    let layout = optimize_paragraph(&words, options);
    let line_count = layout.lines.len();
    let mut result = Vec::with_capacity(line_count);

    for (line_index, line) in layout.lines.iter().enumerate() {
        let mut text = line.text.clone();

        if options.justify
            && line_index + 1 < line_count
            && line.pieces >= 2
            && line.gaps > 0
        {
            let extra = options.width - text.chars().count();
            let each = extra / line.gaps;
            let remainder = extra % line.gaps;
            let chars: Vec<char> = text.chars().collect();
            let indent = if line_index == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };

            let mut rebuilt = String::new();
            let mut gap_number = 0usize;
            for &c in &chars[indent..] {
                rebuilt.push(c);
                if c == ' ' {
                    let relative_gap = (gap_number + line_index % line.gaps) % line.gaps;
                    let added = each + usize::from(relative_gap < remainder);
                    for _ in 0..added {
                        rebuilt.push(' ');
                    }
                    gap_number += 1;
                }
            }
            text = rebuilt;
        }

        result.push(text);
    }

    result
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let invalid_left =
                i == 0 || is_separator(chars[i - 1].1) || chars[i - 1].1 == '~';
            let invalid_right = i + 1 == chars.len()
                || is_separator(chars[i + 1].1)
                || chars[i + 1].1 == '~';
            if invalid_left || invalid_right {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::with_capacity(chars.len());
    let mut i = 0usize;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 1;
            }
        } else {
            normalized.push(c);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut index = 0usize;

    while index < normalized.len() {
        if normalized[index] == '\n' {
            let start = index;
            while index < normalized.len() && normalized[index] == '\n' {
                index += 1;
            }
            let run_len = index - start;
            if run_len >= 2 {
                output.extend(layout_segment(&segment, options));
                segment.clear();
                for _ in 1..run_len {
                    output.push(String::new());
                }
            } else {
                segment.push('\n');
            }
        } else {
            segment.push(normalized[index]);
            index += 1;
        }
    }

    output.extend(layout_segment(&segment, options));
    Ok(output)
}
