#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    text: String,
    consumed: usize,
    marker: bool,
    pieces: usize,
    gaps: usize,
}

#[derive(Clone)]
struct Layout {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn lex_better(a: &[usize], b: &[usize]) -> bool {
    for (x, y) in a.iter().zip(b.iter()) {
        if x != y {
            return x > y;
        }
    }
    a.len() < b.len()
}

fn layout_better(a: &Layout, b: &Layout) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }

    let av: Vec<usize> = a.lines.iter().map(|l| l.consumed).collect();
    let bv: Vec<usize> = b.lines.iter().map(|l| l.consumed).collect();
    if av != bv {
        return lex_better(&av, &bv);
    }

    let at: Vec<&str> = a.lines.iter().map(|l| l.text.as_str()).collect();
    let bt: Vec<&str> = b.lines.iter().map(|l| l.text.as_str()).collect();
    at < bt
}

fn append_chars(s: &mut String, chars: &[char]) {
    for &c in chars {
        s.push(c);
    }
}

fn optimize(words: &[Word], options: &WrapOptions) -> Layout {
    let mut states: Vec<Vec<Option<Layout>>> = words
        .iter()
        .map(|w| vec![None; w.chars.len() + 1])
        .collect();
    let end = (words.len(), 0usize);

    fn solve(
        i: usize,
        off: usize,
        words: &[Word],
        options: &WrapOptions,
        states: &mut [Vec<Option<Layout>>],
    ) -> Layout {
        if i == words.len() {
            return Layout {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
            };
        }
        if let Some(cached) = &states[i][off] {
            return cached.clone();
        }

        let first = i == 0 && off == 0 && states.iter().all(|_| true);
        // First-line status is encoded by the caller's line state through an
        // additional distinction in the dynamic-programming table below.
        let _ = first;
        unreachable!()
    }

    let _ = (&mut states, end, solve as fn(usize, usize, &[Word], &WrapOptions, &mut [Vec<Option<Layout>>]) -> Layout);
    Layout { cost: 0, marker_lines: 0, lines: Vec::new() }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let bad_left = i == 0 || is_separator(chars[i - 1].1) || chars[i - 1].1 == '~';
        let bad_right =
            i + 1 == chars.len() || is_separator(chars[i + 1].1) || chars[i + 1].1 == '~';
        if bad_left || bad_right {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 1;
            }
        } else {
            normalized.push(c);
        }
        i += 1;
    }

    let mut out = Vec::new();
    let mut segment = Vec::new();
    let mut j = 0;
    while j < normalized.len() {
        if normalized[j] == '\n' {
            let start = j;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            let run = j - start;
            if run >= 2 {
                let mut lines = layout_segment(&segment, options);
                out.append(&mut lines);
                segment.clear();
                for _ in 1..run {
                    out.push(String::new());
                }
            } else {
                segment.push('\n');
            }
        } else {
            segment.push(normalized[j]);
            j += 1;
        }
    }
    let mut lines = layout_segment(&segment, options);
    out.append(&mut lines);
    Ok(out)
}

fn layout_segment(segment: &[char], options: &WrapOptions) -> Vec<String> {
    let mut words = Vec::<Word>::new();
    let mut current = Word {
        chars: Vec::new(),
        markers: Vec::new(),
    };
    for &c in segment {
        if is_separator(c) {
            if !current.chars.is_empty() {
                words.push(current);
                current = Word {
                    chars: Vec::new(),
                    markers: Vec::new(),
                };
            }
        } else if c == '~' {
            current.markers.push(current.chars.len());
        } else {
            current.chars.push(c);
        }
    }
    if !current.chars.is_empty() {
        words.push(current);
    }
    if words.is_empty() {
        return Vec::new();
    }

    let layout = optimize_paragraph(&words, options);
    let mut result = Vec::with_capacity(layout.lines.len());
    let count = layout.lines.len();
    for (line_index, line) in layout.lines.iter().enumerate() {
        let mut text = line.text.clone();
        if options.justify && line_index + 1 < count && line.pieces >= 2 && line.gaps > 0 {
            let indent = if line_index == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            let visible = text.chars().count();
            let extra = options.width.saturating_sub(visible);
            let each = extra / line.gaps;
            let rem = extra % line.gaps;
            let start = indent;
            let chars: Vec<char> = text.chars().collect();
            let mut rebuilt = String::new();
            let mut gap_no = 0usize;
            let mut pos = start;
            while pos < chars.len() {
                rebuilt.push(chars[pos]);
                if chars[pos] == ' ' {
                    let add = each + usize::from((gap_no + line_index % line.gaps) % line.gaps < rem);
                    for _ in 0..add {
                        rebuilt.push(' ');
                    }
                    gap_no += 1;
                }
                pos += 1;
            }
            text = rebuilt;
        }
        result.push(text);
    }
    result
}

fn optimize_paragraph(words: &[Word], options: &WrapOptions) -> Layout {
    use std::collections::HashMap;

    fn solve(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut HashMap<(usize, usize, bool), Layout>,
    ) -> Layout {
        if wi == words.len() {
            return Layout {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
            };
        }
        if let Some(v) = memo.get(&(wi, off, first)) {
            return v.clone();
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let cap = options.width - indent;
        let mut best: Option<Layout> = None;
        let mut candidates: Vec<(usize, usize, bool, usize, bool, usize)> = Vec::new();

        // (next word, next offset, marker ending, displayed width, piece count, consumed)
        let mut used = 0usize;
        let mut pieces = 0usize;
        let mut cursor_word = wi;
        let mut cursor_off = off;
        loop {
            if cursor_word >= words.len() {
                if used > 0 {
                    candidates.push((cursor_word, 0, false, used, pieces, used));
                }
                break;
            }
            let word = &words[cursor_word];
            let remaining = word.chars.len() - cursor_off;
            let prefix_space = if pieces == 0 { 0 } else { 1 };
            if used + prefix_space >= cap {
                break;
            }
            let available = cap - used - prefix_space;

            for &marker in &word.markers {
                if marker > cursor_off && marker - cursor_off + 1 <= available {
                    let part = marker - cursor_off;
                    candidates.push((
                        cursor_word,
                        marker,
                        true,
                        used + prefix_space + part + 1,
                        pieces + 1,
                        used + part,
                    ));
                }
            }

            if remaining <= available {
                used += prefix_space + remaining;
                pieces += 1;
                cursor_word += 1;
                cursor_off = 0;
                candidates.push((cursor_word, 0, false, used, pieces, used));
                continue;
            }

            if pieces == 0 {
                let fits_marker = word.markers.iter().any(|&m| {
                    m > cursor_off && m - cursor_off + 1 <= available
                });
                if !fits_marker && available > 0 {
                    candidates.push((
                        cursor_word,
                        cursor_off + available,
                        false,
                        available,
                        1,
                        available,
                    ));
                }
            }
            break;
        }

        for (nwi, noff, marker_end, displayed, npieces, consumed) in candidates {
            if displayed == 0 {
                continue;
            }
            let next = solve(nwi, noff, false, words, options, memo);
            let final_line = next.lines.is_empty();
            let gap = options.width - (indent + displayed);
            let mut line_cost = (gap as u64) * (gap as u64);
            if final_line {
                line_cost /= 4;
            }
            if marker_end {
                line_cost += options.hyphen_cost as u64;
            }
            if nwi == wi && noff == off {
                continue;
            }
            let line = make_line(words, wi, off, nwi, noff, marker_end, indent, npieces, consumed);
            let mut full = Layout {
                cost: line_cost + next.cost,
                marker_lines: usize::from(marker_end) + next.marker_lines,
                lines: {
                    let mut v = vec![line];
                    v.extend(next.lines);
                    v
                },
            };
            if npieces == 1 && !final_line {
                full.cost += 3;
            }
            if best.as_ref().is_none_or(|b| layout_better(&full, b)) {
                best = Some(full);
            }
        }

        let result = best.unwrap_or(Layout {
            cost: 0,
            marker_lines: 0,
            lines: Vec::new(),
        });
        memo.insert((wi, off, first), result.clone());
        result
    }

    let mut memo = HashMap::new();
    solve(0, 0, true, words, options, &mut memo)
}

fn make_line(
    words: &[Word],
    wi: usize,
    off: usize,
    nwi: usize,
    noff: usize,
    marker_end: bool,
    indent: usize,
    pieces: usize,
    consumed: usize,
) -> Line {
    let mut text = " ".repeat(indent);
    let mut gaps = 0usize;
    if marker_end {
        append_chars(&mut text, &words[wi].chars[off..noff]);
        text.push('-');
        Line {
            text,
            consumed,
            marker: true,
            pieces,
            gaps,
        }
    } else if nwi == wi {
        append_chars(&mut text, &words[wi].chars[off..noff]);
        Line {
            text,
            consumed,
            marker: false,
            pieces,
            gaps,
        }
    } else {
        let mut first_piece = true;
        let mut w = wi;
        while w < nwi {
            if !first_piece {
                text.push(' ');
                gaps += 1;
            }
            let start = if w == wi { off } else { 0 };
            let end = if w + 1 == nwi { noff } else { words[w].chars.len() };
            append_chars(&mut text, &words[w].chars[start..end]);
            first_piece = false;
            w += 1;
        }
        Line {
            text,
            consumed,
            marker: false,
            pieces,
            gaps,
        }
    }
}

fn main() {}
