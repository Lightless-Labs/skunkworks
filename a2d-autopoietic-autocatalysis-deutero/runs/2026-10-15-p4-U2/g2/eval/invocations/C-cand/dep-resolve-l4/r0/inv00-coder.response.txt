fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    (1..=24).contains(&b.len())
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn constraint_key(c: &Constraint) -> (u8, Version) {
    match *c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(mut v: Vec<Constraint>) -> Vec<Constraint> {
    v.sort_by_key(|c| constraint_key(c));
    v.dedup();
    v
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(mut e: Vec<Evidence>) -> Vec<Evidence> {
    e.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    e
}

fn constraints_text(v: &[Constraint]) -> String {
    canonical_constraints(v.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn append_evidence(out: &mut String, evidence: &[Evidence]) {
    for e in evidence_sorted(evidence.to_vec()) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn validate_names(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, name) in root.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: name.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, name) in dep.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: name.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, name) in feature.enables.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: name.clone(),
                        });
                    }
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    Ok(())
}

#[derive(Clone, Default)]
struct OriginData {
    constraints: Vec<Constraint>,
    evidence: Vec<Evidence>,
    requests: BTreeSet<String>,
    roots: Vec<(usize, Requirement)>,
}

use std::collections::{BTreeMap, BTreeSet};

#[derive(Clone, Default)]
struct Closure {
    origins: BTreeMap<String, OriginData>,
    features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn target_matches(p: Platform, t: Target) -> bool {
    matches!(
        (p, t),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn platform_text(p: Platform) -> &'static str {
    match p {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn selected_release<'a>(
    catalog: &'a [Package],
    selected: &BTreeMap<String, Version>,
    name: &str,
) -> Option<&'a Release> {
    let version = selected.get(name)?;
    let package = catalog.iter().find(|p| p.name == name)?;
    package.releases.iter().find(|r| r.version == *version)
}

fn root_token(i: usize, r: &Requirement) -> String {
    let fs = r.features.iter().cloned().collect::<BTreeSet<_>>();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        fs.into_iter().collect::<Vec<_>>().join(",")
    )
}

fn dependency_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        dep.features.iter().cloned().collect::<BTreeSet<_>>().into_iter().collect::<Vec<_>>().join(","),
        dep.gate.as_deref().unwrap_or("*"),
        platform_text(dep.platform)
    )
}

fn compute_closure(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut c = Closure::default();
    for (i, root) in roots.iter().enumerate() {
        let o = c.origins.entry(root.package.clone()).or_default();
        o.constraints.push(root.constraint);
        o.evidence.push(Evidence {
            constraint: root.constraint,
            path: vec![root_token(i, root)],
        });
        o.requests.extend(root.features.iter().cloned());
        o.roots.push((i, root.clone()));
    }

    loop {
        let mut changed = false;
        let names = selected.keys().cloned().collect::<Vec<_>>();
        for name in &names {
            let Some(release) = selected_release(catalog, selected, name) else {
                continue;
            };
            let requested = c
                .origins
                .get(name)
                .map(|o| o.requests.clone())
                .unwrap_or_default();
            let active = c.features.entry(name.clone()).or_default();
            for feature in requested {
                if active.insert(feature) {
                    changed = true;
                }
            }

            let mut local_queue = active.clone();
            let mut local_seen = BTreeSet::new();
            while let Some(feature_name) = local_queue.iter().next().cloned() {
                local_queue.remove(&feature_name);
                if !local_seen.insert(feature_name.clone()) {
                    continue;
                }
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        if active.insert(enabled.clone()) {
                            changed = true;
                            local_queue.insert(enabled.clone());
                        }
                    }
                }
            }

            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target) {
                    continue;
                }
                if dep
                    .gate
                    .as_ref()
                    .is_some_and(|gate| !active.contains(gate))
                {
                    continue;
                }

                let token = dependency_token(name, release.version, j, dep);
                let source_path = c.paths.get(name).cloned().unwrap_or_default();
                let mut path = source_path;
                path.push(token);
                let origin = c.origins.entry(dep.package.clone()).or_default();
                if !origin.evidence.contains(&Evidence {
                    constraint: dep.constraint,
                    path: path.clone(),
                }) {
                    origin.constraints.push(dep.constraint);
                    origin.evidence.push(Evidence {
                        constraint: dep.constraint,
                        path,
                    });
                    changed = true;
                }
                let before = origin.requests.len();
                origin.requests.extend(dep.features.iter().cloned());
                if origin.requests.len() != before {
                    changed = true;
                }
            }
        }

        let mut next_paths = BTreeMap::new();
        for (name, origin) in &c.origins {
            for (i, root) in &origin.roots {
                let candidate = vec![root_token(*i, root)];
                choose_path(&mut next_paths, name, candidate);
            }
        }
        for source in &names {
            let Some(release) = selected_release(catalog, selected, source) else {
                continue;
            };
            let Some(source_path) = c.paths.get(source).cloned() else {
                continue;
            };
            let active = c.features.get(source).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !active.contains(g))
                {
                    continue;
                }
                let mut p = source_path.clone();
                p.push(dependency_token(source, release.version, j, dep));
                choose_path(&mut next_paths, &dep.package, p);
            }
        }
        if next_paths != c.paths {
            c.paths = next_paths;
            changed = true;
        }

        // Canonicalize each dependency's provenance after paths have been updated.
        let mut rebuilt = BTreeMap::<String, OriginData>::new();
        for (i, root) in roots.iter().enumerate() {
            let o = rebuilt.entry(root.package.clone()).or_default();
            o.constraints.push(root.constraint);
            o.evidence.push(Evidence {
                constraint: root.constraint,
                path: vec![root_token(i, root)],
            });
            o.requests.extend(root.features.iter().cloned());
            o.roots.push((i, root.clone()));
        }
        for source in &names {
            let Some(release) = selected_release(catalog, selected, source) else {
                continue;
            };
            let Some(source_path) = c.paths.get(source).cloned() else {
                continue;
            };
            let active = c.features.get(source).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !active.contains(g))
                {
                    continue;
                }
                let mut path = source_path.clone();
                path.push(dependency_token(source, release.version, j, dep));
                let o = rebuilt.entry(dep.package.clone()).or_default();
                o.constraints.push(dep.constraint);
                o.evidence.push(Evidence {
                    constraint: dep.constraint,
                    path,
                });
                o.requests.extend(dep.features.iter().cloned());
            }
        }
        for (name, old) in &c.origins {
            let o = rebuilt.entry(name.clone()).or_default();
            if o.roots.is_empty() {
                o.roots = old.roots.clone();
            }
        }
        if rebuilt != c.origins {
            c.origins = rebuilt;
            changed = true;
        }

        if !changed {
            break;
        }
    }

    for origin in c.origins.values_mut() {
        origin.constraints = canonical_constraints(origin.constraints.clone());
        origin.evidence.sort_by(|a, b| {
            constraint_key(a.constraint)
                .cmp(&constraint_key(b.constraint))
                .then_with(|| a.path.cmp(&b.path))
        });
    }
    c
}

fn choose_path(map: &mut BTreeMap<String, Vec<String>>, name: &str, path: Vec<String>) {
    match map.get(name) {
        None => {
            map.insert(name.to_string(), path);
        }
        Some(old) if path.len() < old.len() || (path.len() == old.len() && path < *old) => {
            map.insert(name.to_string(), path);
        }
        _ => {}
    }
}

fn candidate_releases<'a>(
    package: &'a Package,
    constraints: &[Constraint],
) -> Vec<&'a Release> {
    let mut out = package
        .releases
        .iter()
        .filter(|r| constraints.iter().all(|c| matches(*c, r.version)))
        .collect::<Vec<_>>();
    out.sort_by(|a, b| b.version.cmp(&a.version));
    out
}

struct SearchResult {
    best: Option<BTreeMap<String, Version>>,
    best_cost: usize,
    first_error: Option<ResolveError>,
}

fn solution_cost(selected: &BTreeMap<String, Version>, locks: &[Lock]) -> usize {
    locks
        .iter()
        .filter(|l| selected.get(&l.package) != Some(&l.version))
        .count()
}

fn better_solution(
    candidate: &BTreeMap<String, Version>,
    current: &BTreeMap<String, Version>,
    names: &[String],
) -> bool {
    for name in names {
        match (candidate.get(name), current.get(name)) {
            (None, None) => {}
            (None, Some(_)) => return false,
            (Some(_), None) => return true,
            (Some(a), Some(b)) if a != b => return a > b,
            _ => {}
        }
    }
    false
}

fn inspect_node(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Result<Closure, ResolveError> {
    let closure = compute_closure(catalog, roots, selected, target);

    for name in closure
        .origins
        .keys()
        .filter(|n| !catalog.iter().any(|p| p.name == ***n))
    {
        return Err(ResolveError::Missing {
            package: name.clone(),
            evidence: closure.origins[name].evidence.clone(),
        });
    }

    for (package, features) in &closure.features {
        let Some(version) = selected.get(package) else {
            continue;
        };
        let Some(release) = selected_release(catalog, selected, package) else {
            continue;
        };
        for feature in features {
            if !release.features.iter().any(|f| f.name == *feature) {
                return Err(ResolveError::MissingFeature {
                    package: package.clone(),
                    selected: *version,
                    feature: feature.clone(),
                    path: closure.paths.get(package).cloned().unwrap_or_default(),
                });
            }
        }
    }

    for (package, version) in selected {
        if let Some(origin) = closure.origins.get(package) {
            if !origin.constraints.iter().all(|c| matches(*c, *version)) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: origin.constraints.clone(),
                    evidence: origin.evidence.clone(),
                });
            }
        }
    }

    for (package, origin) in &closure.origins {
        if selected.contains_key(package) {
            continue;
        }
        let entry = catalog.iter().find(|p| p.name == *package).unwrap();
        if candidate_releases(entry, &origin.constraints).is_empty() {
            return Err(ResolveError::NoMatch {
                package: package.clone(),
                requirements: origin.constraints.clone(),
                evidence: origin.evidence.clone(),
            });
        }
    }

    Ok(closure)
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
    attempts: &mut usize,
    selected: &mut BTreeMap<String, Version>,
    package_names: &[String],
    result: &mut SearchResult,
) -> Result<(), ResolveError> {
    let closure = match inspect_node(catalog, roots, selected, target) {
        Ok(c) => c,
        Err(e) => {
            if result.first_error.is_none() {
                result.first_error = Some(e);
            }
            return Ok(());
        }
    };

    let next = closure
        .origins
        .keys()
        .find(|name| !selected.contains_key(*name))
        .cloned();

    let Some(package_name) = next else {
        let cost = solution_cost(selected, locks);
        let should_replace = result.best.as_ref().is_none_or(|current| {
            cost < result.best_cost
                || (cost == result.best_cost && better_solution(selected, current, package_names))
        });
        if should_replace {
            result.best = Some(selected.clone());
            result.best_cost = cost;
        }
        return Ok(());
    };

    let package = catalog.iter().find(|p| p.name == package_name).unwrap();
    let origin = &closure.origins[&package_name];
    let mut candidates = candidate_releases(package, &origin.constraints);
    if let Some(lock) = locks.iter().find(|l| l.package == package_name) {
        if let Some(index) = candidates.iter().position(|r| r.version == lock.version) {
            let locked = candidates.remove(index);
            candidates.insert(0, locked);
        }
    }

    for candidate in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit { attempts: max_attempts });
        }
        *attempts += 1;
        selected.insert(package_name.clone(), candidate.version);
        let child = search(
            catalog,
            roots,
            locks,
            target,
            max_attempts,
            attempts,
            selected,
            package_names,
            result,
        );
        selected.remove(&package_name);
        child?;
    }
    Ok(())
}

fn validate_all(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    validate_names(catalog, roots, locks)?;

    let mut names = BTreeSet::new();
    for p in catalog {
        if !names.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }

    for p in catalog {
        let mut seen = BTreeSet::new();
        for r in &p.releases {
            if !seen.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }

    for p in catalog {
        for r in &p.releases {
            let mut seen = BTreeSet::new();
            for f in &r.features {
                if !seen.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: r.version,
                        feature: f.name.clone(),
                    });
                }
            }
        }
    }

    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_all(catalog, roots, locks)?;

    let mut package_names = catalog.iter().map(|p| p.name.clone()).collect::<Vec<_>>();
    package_names.sort();
    package_names.dedup();

    let mut attempts = 0;
    let mut selected = BTreeMap::new();
    let mut result = SearchResult {
        best: None,
        best_cost: usize::MAX,
        first_error: None,
    };

    search(
        catalog,
        roots,
        locks,
        target,
        max_attempts,
        &mut attempts,
        &mut selected,
        &package_names,
        &mut result,
    )?;

    let Some(best) = result.best else {
        return Err(result.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_closure = compute_closure(catalog, roots, &best, target);
    let selected_out = best
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let mut before = BTreeMap::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }
    let mut all = BTreeSet::new();
    all.extend(before.keys().cloned());
    all.extend(best.keys().cloned());
    let mut changes = Vec::new();

    for package in all {
        let b = before.get(&package).copied();
        let a = best.get(&package).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let satisfies = final_closure
                        .origins
                        .get(&package)
                        .map(|o| o.constraints.iter().all(|c| matches(*c, old)))
                        .unwrap_or(true);
                    if !satisfies {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}

fn main() {}
