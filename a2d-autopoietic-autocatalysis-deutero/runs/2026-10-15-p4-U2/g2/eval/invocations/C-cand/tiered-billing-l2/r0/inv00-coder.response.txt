```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
    Graduated,
    Volume,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Bill {
    pub charge: Charge,
    pub minimum_adjustment: u64,
    pub total: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let last_index = tiers.len() - 1;
    let mut previous_bound = 0;

    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous_bound {
                    return Err(Error::Bound { index });
                }
                previous_bound = bound;
            }
            None => {
                if index != last_index {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers[last_index].up_to.is_some() {
        return Err(Error::MissingFinal);
    }

    Ok(())
}

fn add_band(
    total: &mut u64,
    bands: &mut Vec<Band>,
    index: usize,
    units: u64,
    unit: u64,
    start: u64,
) -> Result<(), Error> {
    let amount = units
        .checked_mul(unit)
        .and_then(|product| product.checked_add(start))
        .ok_or(Error::Overflow { index })?;

    let new_total = total
        .checked_add(amount)
        .ok_or(Error::Overflow { index })?;

    *total = new_total;
    bands.push(Band {
        index,
        units,
        amount,
    });
    Ok(())
}

fn quote_validated(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    let mut total = 0;
    let mut bands = Vec::new();
    let mut previous_bound = 0;
    let mut remaining = quantity;

    for (index, tier) in tiers.iter().enumerate() {
        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous_bound;
                let allocated = remaining.min(width);
                previous_bound = bound;
                allocated
            }
            None => remaining,
        };

        if units > 0 {
            add_band(
                &mut total,
                &mut bands,
                index,
                units,
                tier.unit,
                tier.start,
            )?;
            remaining -= units;
        }

        if remaining == 0 {
            break;
        }
    }

    Ok(Charge {
        amount: total,
        bands,
    })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                return Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                });
            }

            let mut amount = 0;
            let mut bands = Vec::new();
            add_band(&mut amount, &mut bands, 0, quantity, *unit, *start)?;
            Ok(Charge { amount, bands })
        }
        Price::Tiers(tiers) => {
            validate_tiers(tiers)?;
            quote_validated(quantity, tiers)
        }
    }
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = match price {
        Price::Flat { .. } => quote(quantity, price)?,
        Price::Tiers(tiers) => {
            validate_tiers(tiers)?;
            match mode {
                Mode::Graduated => quote_validated(quantity, tiers)?,
                Mode::Volume => {
                    if quantity == 0 {
                        Charge {
                            amount: 0,
                            bands: Vec::new(),
                        }
                    } else {
                        let index = tiers
                            .iter()
                            .position(|tier| {
                                tier.up_to
                                    .map(|bound| bound >= quantity)
                                    .unwrap_or(true)
                            })
                            .expect("validated schedule has a final unbounded tier");
                        let tier = &tiers[index];
                        let mut amount = 0;
                        let mut bands = Vec::new();
                        add_band(
                            &mut amount,
                            &mut bands,
                            index,
                            quantity,
                            tier.unit,
                            tier.start,
                        )?;
                        Charge {
                            amount,
                            bands,
                        }
                    }
                }
            }
        }
    };

    let minimum_adjustment = if quantity > 0 && charge.amount < minimum {
        minimum - charge.amount
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .expect("minimum adjustment yields a total no greater than minimum");

    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}

fn main() {}
```
