#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut separators = bytes
        .iter()
        .enumerate()
        .filter(|&(_, byte)| *byte == b'@');

    let Some((separator, _)) = separators.next() else {
        return Err(EmailError::MissingAt);
    };

    if let Some((at, _)) = separators.next() {
        return Err(EmailError::MultipleAt { at });
    }

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local = &bytes[..separator];
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    let allowed_special = b"!#$%&'*+-/=?^_`{|}~";
    for (i, &byte) in local.iter().enumerate() {
        if byte == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at: i });
            }
        } else if !byte.is_ascii_alphanumeric() && !allowed_special.contains(&byte) {
            return Err(EmailError::LocalChar { at: i });
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(root_dot);

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain = &raw_domain[..domain_len];
    let mut label_start = 0;
    let mut label_count = 0;
    let mut last_label = &domain[0..0];
    let mut last_label_start = domain_start;

    for label_end in (0..=domain_len).filter(|&i| i == domain_len || domain[i] == b'.') {
        let label = &domain[label_start..label_end];
        let absolute_start = domain_start + label_start;

        if label.is_empty() {
            let at = if label_start == domain_len && root_dot {
                input.len() - 1
            } else {
                absolute_start
            };
            return Err(EmailError::EmptyLabel { at });
        }

        if label.len() > 63 {
            return Err(EmailError::LabelTooLong {
                at: absolute_start + 63,
            });
        }

        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen { at: absolute_start });
        }

        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: absolute_start + label.len() - 1,
            });
        }

        for (i, &byte) in label.iter().enumerate() {
            if !byte.is_ascii_alphanumeric() && byte != b'-' {
                return Err(EmailError::DomainChar {
                    at: absolute_start + i,
                });
            }
        }

        label_count += 1;
        last_label = label;
        last_label_start = absolute_start;
        label_start = label_end + 1;
    }

    if label_count < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: last_label_start,
        });
    }

    let local = String::from_utf8(local.to_vec()).expect("ASCII input");
    let domain = String::from_utf8(
        domain.iter().map(u8::to_ascii_lowercase).collect(),
    )
    .expect("ASCII input");

    Ok(Address {
        local,
        domain,
        quoted: false,
    })
}

fn main() {}
