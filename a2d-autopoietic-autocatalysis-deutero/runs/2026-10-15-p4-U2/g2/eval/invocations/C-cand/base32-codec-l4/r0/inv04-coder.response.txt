#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

pub struct Decoder {
    config: Config,
    bits: u8,
    bit_count: u8,
    data_count: usize,
    consumed: usize,
    last_data_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: Option<usize>,
    closed: bool,
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn expected_padding(residue: usize) -> Option<usize> {
    match residue {
        0 => Some(0),
        2 => Some(6),
        4 => Some(4),
        5 => Some(3),
        7 => Some(1),
        _ => None,
    }
}

fn encode_complete_bytes(input: &[u8], alphabet: Alphabet) -> String {
    let chars = alphabet_bytes(alphabet);
    let mut out = String::with_capacity(input.len().saturating_mul(8) / 5);
    let mut acc = 0u32;
    let mut bits = 0u8;

    for &byte in input {
        acc = (acc << 8) | u32::from(byte);
        bits += 8;

        while bits >= 5 {
            bits -= 5;
            out.push(chars[((acc >> bits) & 31) as usize] as char);
        }

        if bits == 0 {
            acc = 0;
        } else {
            acc &= (1u32 << bits) - 1;
        }
    }

    out
}

fn encode_final_bytes(input: &[u8], alphabet: Alphabet) -> String {
    let chars = alphabet_bytes(alphabet);
    let mut out = String::new();
    let mut acc = 0u32;
    let mut bits = 0u8;

    for &byte in input {
        acc = (acc << 8) | u32::from(byte);
        bits += 8;

        while bits >= 5 {
            bits -= 5;
            out.push(chars[((acc >> bits) & 31) as usize] as char);
        }

        if bits == 0 {
            acc = 0;
        } else {
            acc &= (1u32 << bits) - 1;
        }
    }

    if bits != 0 {
        out.push(chars[((acc << (5 - bits)) & 31) as usize] as char);
    }

    out
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut combined = Vec::with_capacity(self.pending.len() + input.len());
        combined.extend_from_slice(&self.pending);
        combined.extend_from_slice(input);

        let complete_len = (combined.len() / 5) * 5;
        let output = encode_complete_bytes(&combined[..complete_len], self.config.alphabet);
        self.pending = combined[complete_len..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = encode_final_bytes(&self.pending, self.config.alphabet);
        if self.config.padding == Padding::Required && !self.pending.is_empty() {
            while output.len() % 8 != 0 {
                output.push('=');
            }
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            bits: 0,
            bit_count: 0,
            data_count: 0,
            consumed: 0,
            last_data_offset: None,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: None,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = Self {
            config: self.config,
            bits: self.bits,
            bit_count: self.bit_count,
            data_count: self.data_count,
            consumed: self.consumed,
            last_data_offset: self.last_data_offset,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
            closed: self.closed,
        };
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.consumed;

            if byte == b'=' {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }

                if next.first_padding_offset.is_none() {
                    let residue = next.data_count % 8;
                    match expected_padding(residue) {
                        Some(count) if count != 0 => {
                            next.first_padding_offset = Some(offset);
                            next.expected_padding = Some(count);
                        }
                        _ => return Err(StreamError::Padding { offset }),
                    }
                }

                let expected = next.expected_padding.unwrap_or(0);
                if next.padding_count >= expected {
                    return Err(StreamError::Padding { offset });
                }
                next.padding_count += 1;
            } else {
                let upper = byte.to_ascii_uppercase();
                let value = alphabet_bytes(next.config.alphabet)
                    .iter()
                    .position(|&candidate| candidate == upper)
                    .map(|value| value as u8);

                let Some(value) = value else {
                    return Err(StreamError::InvalidByte { offset, byte });
                };

                if byte.is_ascii_lowercase() && next.config.canonical {
                    return Err(StreamError::NonCanonical {
                        offset,
                        expected: upper,
                    });
                }

                if next.first_padding_offset.is_some() {
                    return Err(StreamError::Padding { offset });
                }

                next.data_count += 1;
                next.last_data_offset = Some(offset);

                // Keep only unresolved bits. Mask after each symbol so bits
                // from already emitted bytes cannot affect later output.
                next.bits = (next.bits << 5) | value;
                next.bit_count += 5;

                while next.bit_count >= 8 {
                    next.bit_count -= 8;
                    output.push((next.bits >> next.bit_count) & 0xff);
                }

                if next.bit_count == 0 {
                    next.bits = 0;
                } else {
                    next.bits &= (1u8 << next.bit_count) - 1;
                }
            }

            next.consumed += 1;
        }

        self.bits = next.bits;
        self.bit_count = next.bit_count;
        self.data_count = next.data_count;
        self.consumed = next.consumed;
        self.last_data_offset = next.last_data_offset;
        self.first_padding_offset = next.first_padding_offset;
        self.padding_count = next.padding_count;
        self.expected_padding = next.expected_padding;

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let total = self.consumed;
        let residue = self.data_count % 8;
        let Some(expected) = expected_padding(residue) else {
            return Err(StreamError::InvalidLength { offset: total });
        };

        if self.config.padding == Padding::Required && self.padding_count != expected {
            return Err(StreamError::Padding {
                offset: self.first_padding_offset.unwrap_or(total),
            });
        }

        let unused = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };

        if unused != 0 && (self.bits & ((1u8 << unused) - 1)) != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(total),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }
}

fn main() {}
