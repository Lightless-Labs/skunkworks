```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum JobState {
    Queued,
    Cancelled,
    Completed,
}

struct StoredJob {
    job: Job,
    sequence: u64,
    state: JobState,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u64,
    jobs: Vec<StoredJob>,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            jobs: Vec::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }

        if job.id.is_empty()
            || job.id.len() > 16
            || !job.id.bytes().all(|byte| {
                byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-'
            })
        {
            return Err(Error::BadId);
        }

        if job.priority > 9 {
            return Err(Error::BadPriority);
        }

        if self.jobs.iter().any(|stored| stored.job.id == job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        self.next_sequence += 1;
        self.jobs.push(StoredJob {
            job,
            sequence,
            state: JobState::Queued,
        });
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }

        let index = self
            .jobs
            .iter()
            .position(|stored| stored.job.id == id)
            .ok_or(Error::Unknown)?;

        match self.jobs[index].state {
            JobState::Queued => self.jobs[index].state = JobState::Cancelled,
            JobState::Cancelled => return Err(Error::Cancelled),
            JobState::Completed => return Err(Error::Completed),
        }

        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }

        let mut selected: Option<(usize, u8, u64)> = None;

        for (index, stored) in self.jobs.iter().enumerate() {
            if stored.state != JobState::Queued || stored.job.ready_at > now {
                continue;
            }

            let effective_priority = if self.aging_ms == 0 {
                stored.job.priority
            } else {
                let age_steps = (now - stored.job.ready_at) / self.aging_ms;
                let boosted = u64::from(stored.job.priority) + age_steps;
                boosted.min(9) as u8
            };

            let is_better = match selected {
                None => true,
                Some((_, best_priority, best_sequence)) => {
                    effective_priority > best_priority
                        || (effective_priority == best_priority
                            && stored.sequence < best_sequence)
                }
            };

            if is_better {
                selected = Some((index, effective_priority, stored.sequence));
            }
        }

        self.last_time = now;

        let Some((index, effective_priority, _)) = selected else {
            return Ok(None);
        };

        self.jobs[index].state = JobState::Completed;
        Ok(Some(Dispatch {
            id: self.jobs[index].job.id.clone(),
            effective_priority,
        }))
    }
}

fn main() {}
```
