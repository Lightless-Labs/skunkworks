    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();

            stop += 1;
        }

        }

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;

            pos += 1;
        }

            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }

                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        } else if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::InvalidChar { at: pos });
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() && bytes[pos] == b'-' {
        let pstart = pos + 1;

            i += 1;
        }
        result.push((start, i));
    }
    result
}

fn parse_partial_or_full(
    token: &str,
    global_start: usize,
) -> Result<(Option<Version>, Vec<Option<u64>>, usize), RangeError> {
    let has_suffix = token.contains('-') || token.contains('+');
    let fields: Vec<&str> = token.split('.').collect();
    let any_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if has_suffix || (fields.len() >= 3 && !any_wild) {
        let v = parse_version(token).map_err(|error| RangeError::Version {
            at: global_start,
            error,
        })?;
        return Ok((Some(v), vec![Some(0), Some(0), Some(0)], 3));
    }

    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at: global_start });
    }
    let mut vals = Vec::new();
    let mut saw_wild = false;
    let mut numeric_count = 0;
    for field in fields {
        if field.is_empty() {
            return Err(RangeError::Syntax { at: global_start });
        }
        if matches!(field, "*" | "x" | "X") {
            saw_wild = true;
            vals.push(None);
        } else {
            if saw_wild
                || !field.bytes().all(is_digit)
                || (field.len() > 1 && field.starts_with('0'))
            {
                return Err(RangeError::Syntax { at: global_start });
            }
            let n = field
                .parse::<u64>()
                .map_err(|_| RangeError::Syntax { at: global_start })?;
            vals.push(Some(n));
            numeric_count += 1;
        }
    }
    Ok((None, vals, numeric_count))
}

fn padded(vals: &[Option<u64>]) -> (u64, u64, u64) {
    let mut out = [0u64; 3];
    for (i, v) in vals.iter().enumerate() {
        if let Some(n) = v {
            out[i] = *n;
        }
    }
    (out[0], out[1], out[2])
}

fn bump_tuple(
    value: (u64, u64, u64),
    field: usize,
    at: usize,
) -> Result<(u64, u64, u64), RangeError> {
    let (mut major, mut minor, mut patch) = value;
    match field {
        0 => {
            major = major.checked_add(1).ok_or(RangeError::Syntax { at })?;
            minor = 0;
            patch = 0;
        }
        1 => {
            minor = minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
            patch = 0;
        }
        _ => patch = patch.checked_add(1).ok_or(RangeError::Syntax { at })?,
    }
    Ok((major, minor, patch))
}

fn operand_expansion(
    arm: &mut Arm,
    token: &str,
    start: usize,
    kind: char,
) -> Result<(), RangeError> {
    let operand_start = start + usize::from(kind != '\0');
    let operand = if kind == '\0' { token } else { &token[1..] };
    if operand.is_empty() {
        return Err(RangeError::Syntax { at: start });
    }

    let (full, vals, k) = parse_partial_or_full(operand, operand_start)?;

    if kind == '\0' {
        if let Some(v) = full {
            add_cmp(arm, CmpOp::Eq, v, true);
        } else if k != 0 {
            let low = padded(&vals);
            if k == 1 || k == 2 {
                add_cmp(arm, CmpOp::Ge, stable_version(low.0, low.1, low.2), false);
                let upper = bump_tuple(low, k - 1, operand_start)?;
                add_cmp(
                    arm,
                    CmpOp::Lt,
                    prerelease_zero(upper.0, upper.1, upper.2),
                    false,
                );
            } else {
                // A full three-number partial is the stable version at this tuple.
                add_cmp(arm, CmpOp::Eq, stable_version(low.0, low.1, low.2), false);
            }
        }
        return Ok(());
    }

    if k == 0 {
        return Err(RangeError::Syntax { at: start });
    }
    let low = padded(&vals);
    let low_version = full
        .clone()
        .unwrap_or_else(|| stable_version(low.0, low.1, low.2));
    add_cmp(arm, CmpOp::Ge, low_version, true);

    let upper_field = match kind {
        '^' if low.0 > 0 || k == 1 => 0,
        '^' if low.1 > 0 || k == 2 => 1,
        '^' => 2,
        '~' if k == 1 => 0,
        '~' => 1,
        _ => unreachable!(),
    };
    let upper = bump_tuple(low, upper_field, operand_start)?;
    add_cmp(
        arm,
        CmpOp::Lt,
        prerelease_zero(upper.0, upper.1, upper.2),
        false,
    );
    Ok(())
}

fn parse_explicit_comparator(
    arm: &mut Arm,
    token: &str,
    start: usize,
) -> Result<(), RangeError> {
    let (op, n) = if token.starts_with(">=") {
        (CmpOp::Ge, 2)
    } else if token.starts_with("<=") {
        (CmpOp::Le, 2)
    } else if token.starts_with('>') {
        (CmpOp::Gt, 1)
    } else if token.starts_with('<') {
        (CmpOp::Lt, 1)
    } else {
        (CmpOp::Eq, 1)
    };

    let operand_start = start + n;
    if operand_start >= start + token.len()
        || !token.as_bytes()[n].is_ascii_digit()
    {
        return Err(RangeError::Syntax { at: start });
    }
    let version = parse_version(&token[n..]).map_err(|error| RangeError::Version {
        at: operand_start,
        error,
    })?;
    add_cmp(arm, op, version, true);
    Ok(())
}

fn empty_arm() -> Arm {
    Arm {
        comparators: Vec::new(),
        anchors: Vec::new(),
    }
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = token_ranges(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut arm_tokens = Vec::new();
    let mut i = 0;

    while i < tokens.len() {
        let (start, end) = tokens[i];
        if &s[start..end] == "||" {
            if arm_tokens.is_empty() {
                return Err(RangeError::Empty { at: start });
            }
            arms.push(parse_arm(s, &arm_tokens)?);
            arm_tokens.clear();
            i += 1;
            if i == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            arm_tokens.push((start, end));
            i += 1;
        }
    }

    if arm_tokens.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(s, &arm_tokens)?);
    Ok(Range { arms })
}

fn parse_arm(s: &str, tokens: &[(usize, usize)]) -> Result<Arm, RangeError> {
    let mut arm = empty_arm();
    let hyphens: Vec<usize> = tokens
        .iter()
        .enumerate()
        .filter_map(|(i, (start, end))| {
            if &s[*start..*end] == "-" {
                Some(i)
            } else {
                None
            }
        })
        .collect();

    if !hyphens.is_empty() {
        if tokens.len() != 3 || hyphens != vec![1] {
            return Err(RangeError::Syntax {
                at: tokens[hyphens[0]].0,
            });
        }
        let (ls, le) = tokens[0];
        let (us, ue) = tokens[2];

        let (lfull, lvals, lk) = parse_partial_or_full(&s[ls..le], ls)?;
        let (ufull, uvals, uk) = parse_partial_or_full(&s[us..ue], us)?;
        if lk == 0 {
            return Err(RangeError::Syntax { at: ls });
        }
        if uk == 0 {
            return Err(RangeError::Syntax { at: us });
        }

        let lt = padded(&lvals);
        let lower = lfull.unwrap_or_else(|| stable_version(lt.0, lt.1, lt.2));
        add_cmp(&mut arm, CmpOp::Ge, lower, true);

        if let Some(upper) = ufull {
            add_cmp(&mut arm, CmpOp::Le, upper, true);
        } else {
            let ut = padded(&uvals);
            if uk == 3 {
                add_cmp(
                    &mut arm,
                    CmpOp::Le,
                    stable_version(ut.0, ut.1, ut.2),
                    false,
                );
            } else {
                let next = bump_tuple(ut, uk - 1, us)?;
                add_cmp(
                    &mut arm,
                    CmpOp::Lt,
                    prerelease_zero(next.0, next.1, next.2),
                    false,
                );
            }
        }
        return Ok(arm);
    }

    for &(start, end) in tokens {
        let token = &s[start..end];
        match token.as_bytes()[0] {
            b'^' => operand_expansion(&mut arm, token, start, '^')?,
            b'~' => operand_expansion(&mut arm, token, start, '~')?,
            b'<' | b'>' | b'=' => parse_explicit_comparator(&mut arm, token, start)?,
            b'0'..=b'9' | b'*' | b'x' | b'X' => {
                operand_expansion(&mut arm, token, start, '\0')?
            }
            _ => return Err(RangeError::Syntax { at: start }),
        }
    }
    Ok(arm)
}

fn within_comparator(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        CmpOp::Lt => ord == Ordering::Less,
        CmpOp::Le => ord != Ordering::Greater,
        CmpOp::Eq => ord == Ordering::Equal,
        CmpOp::Ge => ord != Ordering::Less,
        CmpOp::Gt => ord == Ordering::Greater,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !arm.comparators.iter().all(|c| within_comparator(v, c)) {
        return false;
    }
    include_prerelease
        || v.pre.is_empty()
        || arm.anchors.iter().any(|t| *t == tuple(v))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms
        .iter()
        .any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates
        .iter()
        .enumerate()
        .map(|(index, s)| {
            parse_version(s).map_err(|error| CandidateError { index, error })
        })
        .collect::<Result<_, _>>()?;

    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

#[derive(Clone, Debug)]
pub enum SetExpr {
    Range(String),
    Union(Vec<SetExpr>),
    Intersection(Vec<SetExpr>),
}

#[derive(Clone, Debug)]
pub struct Constraint {
    pub label: String,
    pub expr: SetExpr,
}

#[derive(Clone, Debug)]
pub struct Package {
    pub name: String,
    pub candidates: Vec<String>,
    pub constraints: Vec<Constraint>,
}

#[derive(Clone, Copy, Debug)]
pub enum Preference {
    Highest,
    Lowest,
    StableFirst,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate {
        package: String,
        index: usize,
        error: VersionError,
    },
    InvalidRange {
        package: String,
        label: String,
        path: Vec<usize>,
        error: RangeError,
    },
    NoSolution { package: String, labels: Vec<String> },
}

#[derive(Clone, Debug)]
enum ParsedExpr {
    Range(Range),
    Union(Vec<ParsedExpr>),
    Intersection(Vec<ParsedExpr>),
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
) -> Result<ParsedExpr, SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let range = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_owned(),
                label: label.to_owned(),
                path: path.clone(),
                error,
            })?;
            Ok(ParsedExpr::Range(range))
        }
        SetExpr::Union(children) => {
            let mut out = Vec::new();
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                let parsed = validate_expr(child, package, label, path);
                path.pop();
                out.push(parsed?);
            }
            Ok(ParsedExpr::Union(out))
        }
        SetExpr::Intersection(children) => {
            let mut out = Vec::new();
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                let parsed = validate_expr(child, package, label, path);
                path.pop();
                out.push(parsed?);
            }
            Ok(ParsedExpr::Intersection(out))
        }
    }
}

fn expr_matches(v: &Version, expr: &ParsedExpr, include: bool) -> bool {
    match expr {
        ParsedExpr::Range(r) => matches_with(v, r, include),
        ParsedExpr::Union(xs) => xs.iter().any(|x| expr_matches(v, x, include)),
        ParsedExpr::Intersection(xs) => xs.iter().all(|x| expr_matches(v, x, include)),
    }
}

fn subset_conflict(
    candidates: &[Version],
    constraints: &[(String, ParsedExpr)],
    chosen: &mut Vec<usize>,
    next: usize,
    include: bool,
    best: &mut Option<Vec<usize>>,
) {
    if next == constraints.len() {
        let has_match = candidates.iter().any(|v| {
            chosen
                .iter()
                .all(|i| expr_matches(v, &constraints[*i].1, include))
        });
        if has_match {
            return;
        }

        let mut current = chosen.clone();
        current.sort_by(|a, b| {
            constraints[*a]
                .0
                .as_bytes()
                .cmp(constraints[*b].0.as_bytes())
        });
        let replace = match best {
            None => true,
            Some(old) if current.len() < old.len() => true,
            Some(old) if current.len() == old.len() => {
                let a: Vec<&[u8]> = current
                    .iter()
                    .map(|i| constraints[*i].0.as_bytes())
                    .collect();
                let b: Vec<&[u8]> = old
                    .iter()
                    .map(|i| constraints[*i].0.as_bytes())
                    .collect();
                a < b
            }
            _ => false,
        };
        if replace {
            *best = Some(current);
        }
        return;
    }

    if best
        .as_ref()
        .is_none_or(|current| chosen.len() < current.len())
    {
        subset_conflict(candidates, constraints, chosen, next + 1, include, best);
        chosen.push(next);
        subset_conflict(candidates, constraints, chosen, next + 1, include, best);
        chosen.pop();
    }
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for package in packages {
        if !names.insert(package.name.clone()) {
            return Err(SelectError::DuplicatePackage {
                name: package.name.clone(),
            });
        }
    }

    for package in packages {
        let mut labels = std::collections::HashSet::new();
        for constraint in &package.constraints {
            if !labels.insert(constraint.label.clone()) {
                return Err(SelectError::DuplicateLabel {
                    package: package.name.clone(),
                    label: constraint.label.clone(),
                });
            }
        }
    }

    let mut parsed_candidates = Vec::new();
    for package in packages {
        let mut versions = Vec::new();
        for (index, candidate) in package.candidates.iter().enumerate() {
            versions.push(parse_version(candidate).map_err(|error| {
                SelectError::InvalidCandidate {
                    package: package.name.clone(),
                    index,
                    error,
                }
            })?);
        }
        parsed_candidates.push(versions);
    }

    let mut parsed_constraints = Vec::new();
    for package in packages {
        let mut constraints = Vec::new();
        for constraint in &package.constraints {
            constraints.push((
                constraint.label.clone(),
                validate_expr(
                    &constraint.expr,
                    &package.name,
                    &constraint.label,
                    &mut Vec::new(),
                )?,
            ));
        }
        parsed_constraints.push(constraints);
    }

    let mut selections = Vec::new();
    for (pi, package) in packages.iter().enumerate() {
        let versions = &parsed_candidates[pi];
        let constraints = &parsed_constraints[pi];
        let qualifies = |i: usize| {
            constraints
                .iter()
                .all(|(_, expr)| expr_matches(&versions[i], expr, include_prerelease))
        };

        let mut best: Option<usize> = None;
        for i in 0..versions.len() {
            if !qualifies(i) {
                continue;
            }
            let replace = match best {
                None => true,
                Some(j) => match preference {
                    Preference::Highest => {
                        compare_versions(&versions[i], &versions[j]) == Ordering::Greater
                    }
                    Preference::Lowest => {
                        compare_versions(&versions[i], &versions[j]) == Ordering::Less
                    }
                    Preference::StableFirst => {
                        let stable_i = versions[i].pre.is_empty();
                        let stable_j = versions[j].pre.is_empty();
                        (stable_i && !stable_j)
                            || (stable_i == stable_j
                                && compare_versions(&versions[i], &versions[j])
                                    == Ordering::Greater)
                    }
                },
            };
            if replace {
                best = Some(i);
            }
        }

        if let Some(i) = best {
            selections.push(Selection {
                package: package.name.clone(),
                version: package.candidates[i].clone(),
            });
        } else {
            let labels = if versions.is_empty() {
                Vec::new()
            } else {
                let mut best_conflict = None;
                subset_conflict(
                    versions,
                    constraints,
                    &mut Vec::new(),
                    0,
                    include_prerelease,
                    &mut best_conflict,
                );
                best_conflict
                    .unwrap_or_default()
                    .into_iter()
                    .map(|i| constraints[i].0.clone())
                    .collect()
            };
            return Err(SelectError::NoSolution {
                package: package.name.clone(),
                labels,
            });
        }
    }
    Ok(selections)
}

fn cmp_lower(a: &Bound, b: &Bound) -> Ordering {
    match (a, b) {
        (Bound::Unbounded, Bound::Unbounded) => Ordering::Equal,
        (Bound::Unbounded, _) => Ordering::Less,
        (_, Bound::Unbounded) => Ordering::Greater,
        (Bound::Finite(av, ai), Bound::Finite(bv, bi)) => {
            compare_versions(av, bv).then_with(|| bi.cmp(ai))
        }
    }
}

fn cmp_upper(a: &Bound, b: &Bound) -> Ordering {
    match (a, b) {
        (Bound::Unbounded, Bound::Unbounded) => Ordering::Equal,
        (Bound::Unbounded, _) => Ordering::Greater,
        (_, Bound::Unbounded) => Ordering::Less,
        (Bound::Finite(av, ai), Bound::Finite(bv, bi)) => {
            compare_versions(av, bv).then_with(|| ai.cmp(bi))
        }
    }
}

fn arm_bounds(arm: &Arm) -> (Bound, Bound) {
    let mut lower = Bound::Unbounded;
    let mut upper = Bound::Unbounded;
    for c in &arm.comparators {
        match c.op {
            CmpOp::Eq => {
                let lo = Bound::Finite(c.version.clone(), true);
                let hi = Bound::Finite(c.version.clone(), true);
                if cmp_lower(&lo, &lower) == Ordering::Greater {
                    lower = lo;
                }
                if cmp_upper(&hi, &upper) == Ordering::Less {
                    upper = hi;
                }
            }
            CmpOp::Ge | CmpOp::Gt => {
                let bound = Bound::Finite(c.version.clone(), matches!(c.op, CmpOp::Ge));
                if cmp_lower(&bound, &lower) == Ordering::Greater {
                    lower = bound;
                }
            }
            CmpOp::Le | CmpOp::Lt => {
                let bound = Bound::Finite(c.version.clone(), matches!(c.op, CmpOp::Le));
                if cmp_upper(&bound, &upper) == Ordering::Less {
                    upper = bound;
                }
            }
        }
    }
    (lower, upper)
}

fn core_successor(v: &Version) -> Option<Version> {
    if v.patch < u64::MAX {
        Some(prerelease_zero(v.major, v.minor, v.patch + 1))
    } else if v.minor < u64::MAX {
        Some(prerelease_zero(v.major, v.minor + 1, 0))
    } else if v.major < u64::MAX {
        Some(prerelease_zero(v.major + 1, 0, 0))
    } else {
        None
    }
}

fn least_after(v: &Version) -> Option<Version> {
    if !v.pre.is_empty() {
        let mut out = v.clone();
        out.build.clear();
        out.pre.push("0".to_owned());
        Some(out)
    } else {
        core_successor(v)
    }
}

fn version_from_core_text(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn bound_text(bound: &Bound, lower: bool) -> String {
    match bound {
        Bound::Unbounded => "*".to_owned(),
        Bound::Finite(v, inclusive) => {
            let op = match (lower, inclusive) {
                (true, true) => ">=",
                (true, false) => ">",
                (false, true) => "<=",
                (false, false) => "<",
            };
            format!("{}{}", op, version_from_core_text(v))
        }
    }
}

fn arm_has_default_witness(arm: &Arm) -> bool {
    let (lower, upper) = arm_bounds(arm);
    let mut candidate = match lower {
        Bound::Unbounded => stable_version(0, 0, 0),
        Bound::Finite(v, inclusive) => {
            if inclusive {
                v
            } else if let Some(next) = least_after(&v) {
                next
            } else {
                return false;
            }
        }
    };

    loop {
        if let Bound::Finite(ref upper_v, inclusive) = upper {
            let ord = compare_versions(&candidate, upper_v);
            if ord == Ordering::Greater || (ord == Ordering::Equal && !inclusive) {
                return false;
            }
        }

        if candidate.pre.is_empty() {
            return true;
        }

        if arm.anchors.iter().any(|a| *a == tuple(&candidate)) {
            return true;
        }

        let Some(next) = least_after(&candidate) else {
            return false;
        };
        candidate = next;
    }
}

pub fn canonical(r: &Range) -> String {
    let mut items = Vec::new();
    for arm in &r.arms {
        if !arm_has_default_witness(arm) {
            continue;
        }

        let (lower, upper) = arm_bounds(arm);
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anchor_text = if anchors.is_empty() {
            "-".to_owned()
        } else {
            anchors
                .iter()
                .map(|(a, b, c)| format!("{}.{}.{}", a, b, c))
                .collect::<Vec<_>>()
                .join(",")
        };
        items.push(format!(
            "[{};{};pre={}]",
            bound_text(&lower, true),
            bound_text(&upper, false),
            anchor_text
        ));
    }

    items.sort();
    items.dedup();
    if items.is_empty() {
        "EMPTY".to_owned()
    } else {
        items.join(" || ")
    }
}

fn bounds_overlap(a: &Arm, b: &Arm) -> bool {
    let (al, au) = arm_bounds(a);
    let (bl, bu) = arm_bounds(b);

    let lower = if cmp_lower(&al, &bl) == Ordering::Greater {
        al
    } else {
        bl
    };
    let upper = if cmp_upper(&au, &bu) == Ordering::Less {
        au
    } else {
        bu
    };

    match (&lower, &upper) {
        (Bound::Finite(lv, li), Bound::Finite(uv, ui)) => match compare_versions(lv, uv) {
            Ordering::Less => true,
            Ordering::Greater => false,
            Ordering::Equal => *li && *ui,
        },
        _ => true,
    }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            if !bounds_overlap(aa, bb) {
                continue;
            }
            if include_prerelease {
                return true;
            }

            let (al, au) = arm_bounds(aa);
            let (bl, bu) = arm_bounds(bb);
            let lower = if cmp_lower(&al, &bl) == Ordering::Greater {
                al
            } else {
                bl
            };
            let upper = if cmp_upper(&au, &bu) == Ordering::Less {
                au
            } else {
                bu
            };

            let mut candidate = match lower {
                Bound::Unbounded => stable_version(0, 0, 0),
                Bound::Finite(v, inclusive) => {
                    if inclusive {
                        v
                    } else if let Some(next) = least_after(&v) {
                        next
                    } else {
                        continue;
                    }
                }
            };

            loop {
                if let Bound::Finite(ref upper_v, inclusive) = upper {
                    let ord = compare_versions(&candidate, upper_v);
                    if ord == Ordering::Greater || (ord == Ordering::Equal && !inclusive) {
                        break;
                    }
                }
                if candidate.pre.is_empty() {
                    return true;
                }
                if aa.anchors.iter().any(|x| *x == tuple(&candidate))
                    && bb.anchors.iter().any(|x| *x == tuple(&candidate))
                {
                    return true;
                }
                let Some(next) = least_after(&candidate) else {
                    break;
                };
                candidate = next;
            }
        }
    }
    false
}
