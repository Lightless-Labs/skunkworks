fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> u8 {
    match c {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn constraint_version(c: Constraint) -> Version {
    match c {
        Constraint::Exact(v)
        | Constraint::AtLeast(v)
        | Constraint::Before(v)
        | Constraint::Caret(v) => v,
    }
}

fn constraint_cmp(a: &Constraint, b: &Constraint) -> std::cmp::Ordering {
    constraint_rank(*a)
        .cmp(&constraint_rank(*b))
        .then_with(|| constraint_version(*a).cmp(&constraint_version(*b)))
}

fn constraint_text(c: Constraint) -> String {
    let prefix = match c {
        Constraint::Exact(_) => "=",
        Constraint::AtLeast(_) => ">=",
        Constraint::Before(_) => "<",
        Constraint::Caret(_) => "^",
    };
    format!("{}{}", prefix, version_text(constraint_version(c)))
}

fn canonical_constraints(values: &[Constraint]) -> Vec<Constraint> {
    let mut out = values.to_vec();
    out.sort_by(constraint_cmp);
    out.dedup();
    out
}

fn constraints_text(values: &[Constraint]) -> String {
    canonical_constraints(values)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_text(evidence: &[Evidence]) -> Vec<String> {
    let mut sorted = evidence.to_vec();
    sorted.sort_by(|a, b| {
        constraint_cmp(&a.constraint, &b.constraint).then_with(|| a.path.cmp(&b.path))
    });
    sorted
        .into_iter()
        .map(|e| {
            format!(
                "  C:{} via {}",
                constraint_text(e.constraint),
                e.path.join(" -> ")
            )
        })
        .collect()
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut text = format!("missing|package={}", encode(package));
            for line in evidence_text(evidence) {
                text.push('\n');
                text.push_str(&line);
            }
            text
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut text = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            for line in evidence_text(evidence) {
                text.push('\n');
                text.push_str(&line);
            }
            text
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut text = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            for line in evidence_text(evidence) {
                text.push('\n');
                text.push_str(&line);
            }
            text
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct NodeData {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn platform_matches(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

fn sorted_features(features: &[String]) -> String {
    let mut features = features.to_vec();
    features.sort();
    features.dedup();
    features.join(",")
}

fn root_token(root: &Requirement, index: usize) -> String {
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        root.package,
        constraint_text(root.constraint),
        sorted_features(&root.features)
    )
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        sorted_features(&dep.features),
        gate,
        platform
    )
}

fn closure(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> NodeData {
    let mut data = NodeData::default();

    for (i, root) in roots.iter().enumerate() {
        let token = root_token(root, i);
        data.constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        data.origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
        data.requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        let path = vec![token];
        let current = data.paths.entry(root.package.clone()).or_insert(path.clone());
        if path_better(&path, current) {
            *current = path;
        }
    }

    loop {
        let previous = data.clone();

        for (name, version) in selected {
            let Some(package) = catalog.get(name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
                continue;
            };

            let mut activated = data.requests.get(name).cloned().unwrap_or_default();
            let mut queue: Vec<String> = activated.iter().cloned().collect();
            while let Some(feature_name) = queue.pop() {
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        if activated.insert(enabled.clone()) {
                            queue.push(enabled.clone());
                        }
                    }
                }
            }
            data.requests.insert(name.clone(), activated.clone());

            for (index, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !activated.contains(gate))
                {
                    continue;
                }

                data.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);

                let Some(source_path) = data.paths.get(name).cloned() else {
                    continue;
                };
                let mut path = source_path;
                path.push(dep_token(name, *version, index, dep));

                data.origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });

                let current = data.paths.entry(dep.package.clone()).or_insert(path.clone());
                if path_better(&path, current) {
                    *current = path;
                }

                data.requests
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());
            }
        }

        if data.constraints == previous.constraints
            && data.origins == previous.origins
            && data.requests == previous.requests
            && data.paths == previous.paths
        {
            break;
        }
    }

    for constraints in data.constraints.values_mut() {
        *constraints = canonical_constraints(constraints);
    }

    data
}

fn active_dependencies(
    catalog: &BTreeMap<String, &Package>,
    selected: &BTreeMap<String, Version>,
    requests: &BTreeMap<String, BTreeSet<String>>,
    target: Target,
) -> BTreeMap<String, Vec<(usize, Dependency)>> {
    let mut result = BTreeMap::new();
    for (name, version) in selected {
        let Some(package) = catalog.get(name) else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
            continue;
        };
        let activated = requests.get(name).cloned().unwrap_or_default();
        let deps: Vec<_> = release
            .deps
            .iter()
            .enumerate()
            .filter(|(_, dep)| {
                platform_matches(dep.platform, target)
                    && dep.gate.as_ref().is_none_or(|gate| activated.contains(gate))
            })
            .map(|(index, dep)| (index, dep.clone()))
            .collect();
        result.insert(name.clone(), deps);
    }
    result
}

fn finalize_closure(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
    data: &mut NodeData,
) {
    data.origins.clear();
    data.paths.clear();

    for (i, root) in roots.iter().enumerate() {
        let token = root_token(root, i);
        data.origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
        let path = vec![token];
        let current = data.paths.entry(root.package.clone()).or_insert(path.clone());
        if path_better(&path, current) {
            *current = path;
        }
    }

    let deps = active_dependencies(catalog, selected, &data.requests, target);
    let mut changed = true;
    while changed {
        changed = false;
        for (source, edges) in &deps {
            let Some(version) = selected.get(source) else {
                continue;
            };
            let Some(source_path) = data.paths.get(source).cloned() else {
                continue;
            };
            for (index, dep) in edges {
                let mut path = source_path.clone();
                path.push(dep_token(source, *version, *index, dep));
                data.origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                let current = data.paths.entry(dep.package.clone()).or_insert(path.clone());
                if path_better(&path, current) {
                    *current = path;
                    changed = true;
                }
            }
        }
    }

    for origins in data.origins.values_mut() {
        origins.sort_by(|a, b| {
            constraint_cmp(&a.constraint, &b.constraint).then_with(|| a.path.cmp(&b.path))
        });
    }
}

struct SearchState {
    attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<BTreeMap<String, Version>>,
    best_cost: usize,
    catalog_names: Vec<String>,
}

fn vector_better(
    a: &BTreeMap<String, Version>,
    b: &BTreeMap<String, Version>,
    names: &[String],
) -> bool {
    for name in names {
        match (a.get(name), b.get(name)) {
            (Some(x), Some(y)) if x != y => return x > y,
            (Some(_), None) => return true,
            (None, Some(_)) => return false,
            _ => {}
        }
    }
    false
}

fn lock_cost(selected: &BTreeMap<String, Version>, locks: &BTreeMap<String, Version>) -> usize {
    locks
        .iter()
        .filter(|(package, version)| selected.get(*package) != Some(*version))
        .count()
}

fn evidence_for(data: &NodeData, package: &str) -> Vec<Evidence> {
    data.origins
        .get(package)
        .cloned()
        .unwrap_or_default()
        .into_iter()
        .map(|origin| Evidence {
            constraint: origin.constraint,
            path: origin.path,
        })
        .collect()
}

fn inspect_node(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
    data: &mut NodeData,
) -> Result<Option<String>, ResolveError> {
    finalize_closure(catalog, roots, selected, target, data);

    if let Some(name) = data
        .constraints
        .keys()
        .find(|name| !catalog.contains_key(*name))
    {
        return Err(ResolveError::Missing {
            package: name.clone(),
            evidence: evidence_for(data, name),
        });
    }

    for (name, version) in selected {
        if let Some(requirements) = data.constraints.get(name) {
            if requirements.iter().any(|c| !matches(*version, *c)) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: requirements.clone(),
                    evidence: evidence_for(data, name),
                });
            }
        }
    }

    for (name, version) in selected {
        let Some(package) = catalog.get(name) else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
            continue;
        };
        if let Some(feature) = data
            .requests
            .get(name)
            .into_iter()
            .flat_map(|features| features.iter())
            .find(|feature| !release.features.iter().any(|f| &f.name == *feature))
        {
            return Err(ResolveError::MissingFeature {
                package: name.clone(),
                selected: *version,
                feature: feature.clone(),
                path: data.paths.get(name).cloned().unwrap_or_default(),
            });
        }
    }

    for name in data.constraints.keys() {
        if selected.contains_key(name) {
            continue;
        }
        let requirements = data.constraints.get(name).cloned().unwrap_or_default();
        let package = catalog[name];
        if !package
            .releases
            .iter()
            .any(|release| requirements.iter().all(|c| matches(release.version, *c)))
        {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements,
                evidence: evidence_for(data, name),
            });
        }
    }

    Ok(data
        .constraints
        .keys()
        .find(|name| !selected.contains_key(*name))
        .cloned())
}

fn search(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    locks: &BTreeMap<String, Version>,
    target: Target,
    selected: &mut BTreeMap<String, Version>,
    state: &mut SearchState,
    budget: usize,
) -> Result<(), ResolveError> {
    let mut data = closure(catalog, roots, selected, target);
    let branch = match inspect_node(catalog, roots, selected, target, &mut data) {
        Ok(branch) => branch,
        Err(error) => {
            if state.first_error.is_none() {
                state.first_error = Some(error);
            }
            return Ok(());
        }
    };

    let Some(name) = branch else {
        let cost = lock_cost(selected, locks);
        if state.best.as_ref().is_none_or(|best| {
            cost < state.best_cost
                || (cost == state.best_cost && vector_better(selected, best, &state.catalog_names))
        }) {
            state.best = Some(selected.clone());
            state.best_cost = cost;
        }
        return Ok(());
    };

    let requirements = data.constraints.get(&name).cloned().unwrap_or_default();
    let mut candidates: Vec<Version> = catalog[&name]
        .releases
        .iter()
        .filter(|release| requirements.iter().all(|c| matches(release.version, *c)))
        .map(|release| release.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    if let Some(lock_version) = locks.get(&name) {
        if let Some(position) = candidates.iter().position(|version| version == lock_version) {
            let locked = candidates.remove(position);
            candidates.insert(0, locked);
        }
    }

    for candidate in candidates {
        if state.attempts == budget {
            return Err(ResolveError::Limit { attempts: budget });
        }
        state.attempts += 1;
        selected.insert(name.clone(), candidate);
        let result = search(catalog, roots, locks, target, selected, state, budget);
        selected.remove(&name);
        result?;
    }

    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let site = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: site.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.gate", site),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", site, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let site = format!("catalog[{}].release[{}].feature[{}]", i, j, f);
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site,
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut package_seen = BTreeSet::new();
    for package in catalog {
        if !package_seen.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_seen = BTreeSet::new();
    for lock in locks {
        if !lock_seen.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|package| (package.name.clone(), package))
        .collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();
    let catalog_names = catalog_map.keys().cloned().collect();

    let mut state = SearchState {
        attempts: 0,
        first_error: None,
        best: None,
        best_cost: usize::MAX,
        catalog_names,
    };
    let mut selected = BTreeMap::new();

    search(
        &catalog_map,
        roots,
        &lock_map,
        target,
        &mut selected,
        &mut state,
        max_attempts,
    )?;

    let Some(best) = state.best else {
        return Err(state.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let mut final_data = closure(&catalog_map, roots, &best, target);
    finalize_closure(&catalog_map, roots, &best, target, &mut final_data);

    let selected_out = best
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut package_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    package_names.extend(best.keys().cloned());
    let mut changes = Vec::new();

    for package in package_names {
        let before = lock_map.get(&package).copied();
        let after = best.get(&package).copied();
        if before == after {
            continue;
        }

        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map.get(&package).is_some_and(|p| {
                    p.releases.iter().any(|release| release.version == old)
                });
                if !exists {
                    ChangeReason::Unavailable
                } else if final_data
                    .constraints
                    .get(&package)
                    .is_some_and(|constraints| constraints.iter().any(|c| !matches(old, *c)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };

        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}

fn main() {}
