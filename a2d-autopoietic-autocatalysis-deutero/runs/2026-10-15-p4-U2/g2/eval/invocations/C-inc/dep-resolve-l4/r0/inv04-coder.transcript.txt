            for line in sorted_evidence_lines(evidence) {
                text.push('\n');
                text.push_str(&line);
            }
            text
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut text = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            for line in sorted_evidence_lines(evidence) {
                text.push('\n');
                text.push_str(&line);
            }
            text
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

#[derive(Clone, Default)]
struct NodeData {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Evidence>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn platform_matches(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

fn sorted_features(features: &[String]) -> String {
    let mut features = features.to_vec();
    features.sort();
    features.dedup();
    features.join(",")
}

fn root_token(root: &Requirement, index: usize) -> String {
    format!(
        "root[{index}]:{}@{}+{{{}}}",
        root.package,
        constraint_text(root.constraint),
        sorted_features(&root.features)
    )
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        sorted_features(&dep.features),
        gate,
        platform
    )
}

fn selected_release<'a>(
    catalog: &'a BTreeMap<String, &'a Package>,
    package: &str,
    version: Version,
) -> Option<&'a Release> {
    catalog
        .get(package)?
        .releases
        .iter()
        .find(|release| release.version == version)
}

fn compute_node(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> NodeData {
    let mut data = NodeData::default();

    for (index, root) in roots.iter().enumerate() {
        let path = vec![root_token(root, index)];
        data.constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        data.origins
            .entry(root.package.clone())
            .or_default()
            .push(Evidence {
                constraint: root.constraint,
                path: path.clone(),
            });
        data.requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        data.paths
            .entry(root.package.clone())
            .and_modify(|old| {
                if path_better(&path, old) {
                    *old = path.clone();
                }
            })
            .or_insert(path);
    }

    // Feature requests and active dependency edges grow monotonically at a node.
    // Build the closure without appending provenance repeatedly: that would
    // multiply identical origins on every fixed-point pass, especially in cycles.
    loop {
        let old_requests = data.requests.clone();
        let old_constraints = data.constraints.clone();

        let selected_packages: Vec<(String, Version)> =
            selected.iter().map(|(name, v)| (name.clone(), *v)).collect();

        for (source, version) in &selected_packages {
            let Some(release) = selected_release(catalog, source, *version) else {
                continue;
            };

            let activated = data.requests.entry(source.clone()).or_default();
            let mut queue: Vec<String> = activated.iter().cloned().collect();
            while let Some(feature_name) = queue.pop() {
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        if activated.insert(enabled.clone()) {
                            queue.push(enabled.clone());
                        }
                    }
                }
            }

            let activated_snapshot = activated.clone();
            for dep in &release.deps {
                if !platform_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !activated_snapshot.contains(gate))
                {
                    continue;
                }
                data.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                data.requests
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());
            }
        }

        if data.requests == old_requests && data.constraints == old_constraints {
            break;
        }
    }

    // Deduplicate the static edge set before provenance. Keep every incoming
    // origin, including exact duplicates from distinct roots/dependencies.
    for constraints in data.constraints.values_mut() {
        *constraints = canonical_constraints(constraints);
    }

    let active_edges = |source: &str, version: Version, requests: &BTreeMap<String, BTreeSet<String>>| {
        let Some(release) = selected_release(catalog, source, version) else {
            return Vec::new();
        };
        let activated = requests.get(source);
        release
            .deps
            .iter()
            .enumerate()
            .filter(|(_, dep)| {
                platform_matches(dep.platform, target)
                    && dep
                        .gate
                        .as_ref()
                        .is_none_or(|gate| activated.is_some_and(|features| features.contains(gate)))
            })
            .map(|(index, dep)| (index, dep))
            .collect::<Vec<_>>()
    };

    // Find canonical shortest paths over the active selected graph.
    // Relaxing only when a path improves guarantees termination even with cycles.
    loop {
        let old_paths = data.paths.clone();
        let sources: Vec<(String, Version)> =
            selected.iter().map(|(name, v)| (name.clone(), *v)).collect();
        for (source, version) in sources {
            let Some(source_path) = data.paths.get(&source).cloned() else {
                continue;
            };
            for (index, dep) in active_edges(&source, version, &data.requests) {
                let mut path = source_path.clone();
                path.push(dep_token(&source, version, index, dep));
                data.paths
                    .entry(dep.package.clone())
                    .and_modify(|old| {
                        if path_better(&path, old) {
                            *old = path.clone();
                        }
                    })
                    .or_insert(path);
            }
        }
        if data.paths == old_paths {
            break;
        }
    }

    // Provenance includes each root and each active edge exactly once, using
    // the canonical path to its source.
    let sources: Vec<(String, Version)> =
        selected.iter().map(|(name, v)| (name.clone(), *v)).collect();
    for (source, version) in sources {
        let Some(source_path) = data.paths.get(&source).cloned() else {
            continue;
        };
        for (index, dep) in active_edges(&source, version, &data.requests) {
            let mut path = source_path.clone();
            path.push(dep_token(&source, version, index, dep));
            data.origins
                .entry(dep.package.clone())
                .or_default()
                .push(Evidence {
                    constraint: dep.constraint,
                    path,
                });
        }
    }

    for origins in data.origins.values_mut() {
        origins.sort_by(|a, b| {
            constraint_cmp(&a.constraint, &b.constraint).then_with(|| a.path.cmp(&b.path))
        });
    }

    data
}

fn evidence_for(data: &NodeData, package: &str) -> Vec<Evidence> {
    data.origins.get(package).cloned().unwrap_or_default()
}

fn vector_better(
    a: &BTreeMap<String, Version>,
    b: &BTreeMap<String, Version>,
    names: &[String],
) -> bool {
    for name in names {
        match (a.get(name), b.get(name)) {
            (Some(left), Some(right)) if left != right => return left > right,
            (Some(_), None) => return true,
            (None, Some(_)) => return false,
            _ => {}
        }
    }
    false
}

fn lock_cost(selected: &BTreeMap<String, Version>, locks: &BTreeMap<String, Version>) -> usize {
    locks
        .iter()
        .filter(|(package, version)| selected.get(*package) != Some(*version))
        .count()
}

fn inspect_node(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Result<(NodeData, Option<String>), ResolveError> {
    let data = compute_node(catalog, roots, selected, target);

    if let Some(package) = data
        .constraints
        .keys()
        .find(|package| !catalog.contains_key(*package))
    {
        return Err(ResolveError::Missing {
            package: package.clone(),
            evidence: evidence_for(&data, package),
        });
    }

    // Feature errors precede numeric conflicts.
    for (package, version) in selected {
        let Some(release) = selected_release(catalog, package, *version) else {
            continue;
        };
        if let Some(feature) = data
            .requests
            .get(package)
            .into_iter()
            .flat_map(|set| set.iter())
            .find(|feature| !release.features.iter().any(|decl| &decl.name == *feature))
        {
            return Err(ResolveError::MissingFeature {
                package: package.clone(),
                selected: *version,
                feature: feature.clone(),
                path: data.paths.get(package).cloned().unwrap_or_default(),
            });
        }
    }

    for (package, version) in selected {
        if let Some(requirements) = data.constraints.get(package) {
            if requirements
                .iter()
                .any(|constraint| !matches(*version, *constraint))
            {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: requirements.clone(),
                    evidence: evidence_for(&data, package),
                });
            }
        }
    }

    for package in data.constraints.keys() {
        if selected.contains_key(package) {
            continue;
        }
        let requirements = data.constraints.get(package).cloned().unwrap_or_default();
        let catalog_package = catalog[package];
        if !catalog_package
            .releases
            .iter()
            .any(|release| requirements.iter().all(|c| matches(release.version, *c)))
        {
            return Err(ResolveError::NoMatch {
                package: package.clone(),
                requirements,
                evidence: evidence_for(&data, package),
            });
        }
    }

    let next = data
        .constraints
        .keys()
        .find(|package| !selected.contains_key(*package))
        .cloned();
    Ok((data, next))
}

struct SearchState {
    attempts: usize,
    budget: usize,
    locks: BTreeMap<String, Version>,
    catalog_names: Vec<String>,
    first_error: Option<ResolveError>,
    best: Option<BTreeMap<String, Version>>,
    best_cost: usize,
}

fn search(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    target: Target,
    selected: &mut BTreeMap<String, Version>,
    state: &mut SearchState,
) -> Result<(), ResolveError> {
    let (data, branch) = match inspect_node(catalog, roots, selected, target) {
        Ok(value) => value,
        Err(error) => {
            if state.first_error.is_none() {
                state.first_error = Some(error);
            }
            return Ok(());
        }
    };

    let Some(package) = branch else {
        let cost = lock_cost(selected, &state.locks);
        if state.best.as_ref().is_none_or(|best| {
            cost < state.best_cost
                || (cost == state.best_cost
                    && vector_better(selected, best, &state.catalog_names))
        }) {
            state.best = Some(selected.clone());
            state.best_cost = cost;
        }
        return Ok(());
    };

    // A safe lower bound: locks already made unreachable or changed cannot be
    // repaired by assigning further packages.
    if state.best.is_some() {
        let mut lower_bound = 0;
        for (locked_package, locked_version) in &state.locks {
            if selected.get(locked_package) == Some(locked_version) {
                continue;
            }
            if selected.contains_key(locked_package) || !data.constraints.contains_key(locked_package)
            {
                lower_bound += 1;
            }
        }
        if lower_bound > state.best_cost {
            return Ok(());
        }
    }

    let requirements = data.constraints.get(&package).cloned().unwrap_or_default();
    let mut candidates: Vec<Version> = catalog[&package]
        .releases
        .iter()
        .filter(|release| {
            requirements
                .iter()
                .all(|constraint| matches(release.version, *constraint))
        })
        .map(|release| release.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    if let Some(locked_version) = state.locks.get(&package) {
        if let Some(index) = candidates.iter().position(|candidate| candidate == locked_version) {
            let locked = candidates.remove(index);
            candidates.insert(0, locked);
        }
    }

    for candidate in candidates {
        if state.attempts == state.budget {
            return Err(ResolveError::Limit {
                attempts: state.budget,
            });
        }
        state.attempts += 1;
        selected.insert(package.clone(), candidate);
        let result = search(catalog, roots, target, selected, state);
        selected.remove(&package);
        result?;
    }

    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{i}].feature[{f}]"),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let site = format!("catalog[{i}].release[{j}].dep[{k}]");
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: site.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{site}.gate"),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{site}.feature[{f}]"),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let site = format!("catalog[{i}].release[{j}].feature[{f}]");
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: site.clone(),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{site}.enable[{e}]"),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen_features = BTreeSet::new();
            for feature in &release.features {
                if !seen_features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|package| (package.name.clone(), package))
        .collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();
    let catalog_names: Vec<String> = catalog_map.keys().cloned().collect();

    let mut state = SearchState {
        attempts: 0,
        budget: max_attempts,
        locks: lock_map.clone(),
        catalog_names,
        first_error: None,
        best: None,
        best_cost: usize::MAX,
    };
    let mut selected = BTreeMap::new();
    search(
        &catalog_map,
        roots,
        target,
        &mut selected,
        &mut state,
    )?;

    let Some(best) = state.best else {
        return Err(state.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_data = compute_node(&catalog_map, roots, &best, target);
    let selected_out = best
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut package_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    package_names.extend(best.keys().cloned());

    let mut changes = Vec::new();
    for package in package_names {
        let before = lock_map.get(&package).copied();
        let after = best.get(&package).copied();
        if before == after {
            continue;
        }

        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let available = catalog_map.get(&package).is_some_and(|p| {
                    p.releases.iter().any(|release| release.version == old)
                });
                if !available {
                    ChangeReason::Unavailable
                } else if final_data
                    .constraints
                    .get(&package)
                    .is_some_and(|constraints| {
                        constraints
                            .iter()
                            .any(|constraint| !matches(old, *constraint))
                    })
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };

        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}

fn main() {}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|x| x.is_ascii_lowercase() || x.is_ascii_digit() || *x == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> u8 {
    match c {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn constraint_version(c: Constraint) -> Version {
    match c {
        Constraint::Exact(v)
        | Constraint::AtLeast(v)
        | Constraint::Before(v)
        | Constraint::Caret(v) => v,
    }
}

fn constraint_cmp(a: &Constraint, b: &Constraint) -> std::cmp::Ordering {
    constraint_rank(*a)
        .cmp(&constraint_rank(*b))
        .then_with(|| constraint_version(*a).cmp(&constraint_version(*b)))
}

fn constraint_text(c: Constraint) -> String {
    let prefix = match c {
        Constraint::Exact(_) => "=",
        Constraint::AtLeast(_) => ">=",
        Constraint::Before(_) => "<",
        Constraint::Caret(_) => "^",
    };
    format!("{prefix}{}", version_text(constraint_version(c)))
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort_by(constraint_cmp);
    out.dedup();
    out
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{b:02X}"));
        }
    }
    out
}

fn sorted_evidence_lines(evidence: &[Evidence]) -> Vec<String> {
    let mut items = evidence.to_vec();
    items.sort_by(|a, b| {
        constraint_cmp(&a.constraint, &b.constraint).then_with(|| a.path.cmp(&b.path))
    });
    items
        .into_iter()
        .map(|e| {
            format!(
                "  C:{} via {}",
                constraint_text(e.constraint),
                e.path.join(" -> ")
            )
        })
        .collect()
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            for line in sorted_evidence_lines(evidence) {
                s.push('\n');
                s.push_str(&line);
            }
            s
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            for line in sorted_evidence_lines(evidence) {
                s.push('\n');
                s.push_str(&line);
            }
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            for line in sorted_evidence_lines(evidence) {
                s.push('\n');
                s.push_str(&line);
            }
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

#[derive(Clone, Default)]
struct NodeData {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Evidence>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn platform_matches(p: Platform, t: Target) -> bool {
    match p {
        Platform::Any => true,
        Platform::Linux => t == Target::Linux,
        Platform::Mac => t == Target::Mac,
        Platform::Windows => t == Target::Windows,
    }
}

fn sorted_features(fs: &[String]) -> String {
    let mut f = fs.to_vec();
    f.sort();
    f.dedup();
    f.join(",")
}

fn root_token(r: &Requirement, i: usize) -> String {
    format!(
        "root[{i}]:{}@{}+{{{}}}",
        r.package,
        constraint_text(r.constraint),
        sorted_features(&r.features)
    )
}

fn dep_token(src: &str, v: Version, i: usize, d: &Dependency) -> String {
    let gate = d.gate.as_deref().unwrap_or("*");
    let platform = match d.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        src,
        version_text(v),
        i,
        d.package,
        constraint_text(d.constraint),
        sorted_features(&d.features),
        gate,
        platform
    )
}

fn selected_release<'a>(
    catalog: &'a BTreeMap<String, &'a Package>,
    package: &str,
    version: Version,
) -> Option<&'a Release> {
    catalog
        .get(package)?
        .releases
        .iter()
        .find(|r| r.version == version)
}

fn compute_node(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> NodeData {
    let mut data = NodeData::default();

    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(root, i)];
        data.constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        data.origins
            .entry(root.package.clone())
            .or_default()
            .push(Evidence {
                constraint: root.constraint,
                path: path.clone(),
            });
        data.requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        data.paths
            .entry(root.package.clone())
            .and_modify(|old| {
                if path_better(&path, old) {
                    *old = path.clone();
                }
            })
            .or_insert(path);
    }

    loop {
        let old_requests = data.requests.clone();
        let old_constraints = data.constraints.clone();
        let sources: Vec<(String, Version)> =
            selected.iter().map(|(p, v)| (p.clone(), *v)).collect();

        for (source, version) in sources {
            let Some(release) = selected_release(catalog, &source, version) else {
                continue;
            };
            let activated = data.requests.entry(source.clone()).or_default();
            let mut queue: Vec<String> = activated.iter().cloned().collect();
            while let Some(name) = queue.pop() {
                if let Some(feature) = release.features.iter().find(|f| f.name == name) {
                    for enabled in &feature.enables {
                        if activated.insert(enabled.clone()) {
                            queue.push(enabled.clone());
                        }
                    }
                }
            }

            let snapshot = activated.clone();
            for dep in &release.deps {
                if !platform_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !snapshot.contains(gate))
                {
                    continue;
                }
                data.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                data.requests
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());
            }
        }
        if data.requests == old_requests && data.constraints == old_constraints {
            break;
        }
    }

    for cs in data.constraints.values_mut() {
        *cs = canonical_constraints(cs);
    }

    let active_edges = |source: &str,
                        version: Version,
                        requests: &BTreeMap<String, BTreeSet<String>>| {
        let Some(release) = selected_release(catalog, source, version) else {
            return Vec::new();
        };
        let activated = requests.get(source);
        release
            .deps
            .iter()
            .enumerate()
            .filter(|(_, dep)| {
                platform_matches(dep.platform, target)
                    && dep
                        .gate
                        .as_ref()
                        .is_none_or(|gate| activated.is_some_and(|fs| fs.contains(gate)))
            })
            .collect::<Vec<_>>()
    };

    loop {
        let old = data.paths.clone();
        let sources: Vec<(String, Version)> =
            selected.iter().map(|(p, v)| (p.clone(), *v)).collect();
        for (source, version) in sources {
            let Some(source_path) = data.paths.get(&source).cloned() else {
                continue;
            };
            for (i, dep) in active_edges(&source, version, &data.requests) {
                let mut path = source_path.clone();
                path.push(dep_token(&source, version, i, dep));
                data.paths
                    .entry(dep.package.clone())
                    .and_modify(|old| {
                        if path_better(&path, old) {
                            *old = path.clone();
                        }
                    })
                    .or_insert(path);
            }
        }
        if data.paths == old {
            break;
        }
    }

    let sources: Vec<(String, Version)> =
        selected.iter().map(|(p, v)| (p.clone(), *v)).collect();
    for (source, version) in sources {
        let Some(source_path) = data.paths.get(&source).cloned() else {
            continue;
        };
        for (i, dep) in active_edges(&source, version, &data.requests) {
            let mut path = source_path.clone();
            path.push(dep_token(&source, version, i, dep));
            data.origins
                .entry(dep.package.clone())
                .or_default()
                .push(Evidence {
                    constraint: dep.constraint,
                    path,
                });
        }
    }
    for origins in data.origins.values_mut() {
        origins.sort_by(|a, b| {
            constraint_cmp(&a.constraint, &b.constraint).then_with(|| a.path.cmp(&b.path))
        });
    }
    data
}

fn evidence_for(data: &NodeData, p: &str) -> Vec<Evidence> {
    data.origins.get(p).cloned().unwrap_or_default()
}

fn vector_better(
    a: &BTreeMap<String, Version>,
    b: &BTreeMap<String, Version>,
    names: &[String],
) -> bool {
    for name in names {
        match (a.get(name), b.get(name)) {
            (Some(x), Some(y)) if x != y => return x > y,
            (Some(_), None) => return true,
            (None, Some(_)) => return false,
            _ => {}
        }
    }
    false
}

fn lock_cost(s: &BTreeMap<String, Version>, locks: &BTreeMap<String, Version>) -> usize {
    locks
        .iter()
        .filter(|(p, v)| s.get(*p) != Some(*v))
        .count()
}

fn inspect_node(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Result<(NodeData, Option<String>), ResolveError> {
    let data = compute_node(catalog, roots, selected, target);

    if let Some(p) = data.constraints.keys().find(|p| !catalog.contains_key(*p)) {
        return Err(ResolveError::Missing {
            package: p.clone(),
            evidence: evidence_for(&data, p),
        });
    }

    for (p, v) in selected {
        let Some(release) = selected_release(catalog, p, *v) else {
            continue;
        };
        if let Some(feature) = data
            .requests
            .get(p)
            .into_iter()
            .flat_map(|fs| fs.iter())
            .find(|f| !release.features.iter().any(|decl| &decl.name == *f))
        {
            return Err(ResolveError::MissingFeature {
                package: p.clone(),
                selected: *v,
                feature: feature.clone(),
                path: data.paths.get(p).cloned().unwrap_or_default(),
            });
        }
    }

    for (p, v) in selected {
        if let Some(reqs) = data.constraints.get(p) {
            if reqs.iter().any(|c| !matches(*v, *c)) {
                return Err(ResolveError::Conflict {
                    package: p.clone(),
                    selected: *v,
                    requirements: reqs.clone(),
                    evidence: evidence_for(&data, p),
                });
            }
        }
    }

    for p in data.constraints.keys() {
        if selected.contains_key(p) {
            continue;
        }
        let reqs = data.constraints.get(p).cloned().unwrap_or_default();
        if !catalog[p]
            .releases
            .iter()
            .any(|r| reqs.iter().all(|c| matches(r.version, *c)))
        {
            return Err(ResolveError::NoMatch {
                package: p.clone(),
                requirements: reqs,
                evidence: evidence_for(&data, p),
            });
        }
    }

    let next = data
        .constraints
        .keys()
        .find(|p| !selected.contains_key(*p))
        .cloned();
    Ok((data, next))
}

struct SearchState {
    attempts: usize,
    budget: usize,
    locks: BTreeMap<String, Version>,
    catalog_names: Vec<String>,
    first_error: Option<ResolveError>,
    best: Option<BTreeMap<String, Version>>,
    best_cost: usize,
}

fn can_improve(
    data: &NodeData,
    selected: &BTreeMap<String, Version>,
    state: &SearchState,
) -> bool {
    let Some(best) = &state.best else {
        return true;
    };

    let mut lower_cost = 0usize;
    for (package, version) in &state.locks {
        if selected.get(package) == Some(version) {
            continue;
        }
        if selected.contains_key(package) || !data.constraints.contains_key(package) {
            lower_cost += 1;
        }
    }

    if lower_cost < state.best_cost {
        return true;
    }
    if lower_cost > state.best_cost {
        return false;
    }

    // An equal-cost branch can still win only if its optimistic version vector
    // beats the current best. Selected entries are fixed; unselected catalog
    // packages are optimistically assigned their largest version, and absence
    // is used only when a package has no releases.
    for name in &state.catalog_names {
        let optimistic = if let Some(v) = selected.get(name) {
            Some(*v)
        } else if let Some(package) = data
            .constraints
            .contains_key(name)
            .then_some(name)
            .and_then(|n| state.catalog_names.iter().find(|x| *x == n))
        {
            let _ = package;
            None
        } else {
            None
        };
        let _ = optimistic;
    }

    // A compact, safe bound: if every catalog name preceding the first
    // unresolved package is already no better than the incumbent, this branch
    // cannot beat it. Names that are still unresolved are treated as +infinity.
    for name in &state.catalog_names {
        if !selected.contains_key(name) {
            return true;
        }
        match (selected.get(name), best.get(name)) {
            (Some(a), Some(b)) if a > b => return true,
            (Some(a), Some(b)) if a < b => return false,
            (Some(_), None) => return true,
            (None, Some(_)) => return false,
            _ => {}
        }
    }
    false
}

fn search(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    target: Target,
    selected: &mut BTreeMap<String, Version>,
    state: &mut SearchState,
) -> Result<(), ResolveError> {
    let (data, branch) = match inspect_node(catalog, roots, selected, target) {
        Ok(x) => x,
        Err(e) => {
            if state.first_error.is_none() {
                state.first_error = Some(e);
            }
            return Ok(());
        }
    };

    let Some(package) = branch else {
        let cost = lock_cost(selected, &state.locks);
        if state.best.as_ref().is_none_or(|best| {
            cost < state.best_cost
                || (cost == state.best_cost
                    && vector_better(selected, best, &state.catalog_names))
        }) {
            state.best = Some(selected.clone());
            state.best_cost = cost;
        }
        return Ok(());
    };

    if !can_improve(&data, selected, state) {
        return Ok(());
    }

    let reqs = data.constraints.get(&package).cloned().unwrap_or_default();
    let mut candidates: Vec<Version> = catalog[&package]
        .releases
        .iter()
        .filter(|r| reqs.iter().all(|c| matches(r.version, *c)))
        .map(|r| r.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    if let Some(locked) = state.locks.get(&package) {
        if let Some(i) = candidates.iter().position(|v| v == locked) {
            candidates.insert(0, candidates.remove(i));
        }
    }

    for candidate in candidates {
        if state.attempts == state.budget {
            return Err(ResolveError::Limit {
                attempts: state.budget,
            });
        }
        state.attempts += 1;
        selected.insert(package.clone(), candidate);
        let result = search(catalog, roots, target, selected, state);
        selected.remove(&package);
        result?;
    }
    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{i}].feature[{f}]"),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let site = format!("catalog[{i}].release[{j}].dep[{k}]");
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: site.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{site}.gate"),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{site}.feature[{f}]"),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let site = format!("catalog[{i}].release[{j}].feature[{f}]");
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: site.clone(),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{site}.enable[{e}]"),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen_features = BTreeSet::new();
            for feature in &release.features {
                if !seen_features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();
    let catalog_names: Vec<String> = catalog_map.keys().cloned().collect();

    let mut state = SearchState {
        attempts: 0,
        budget: max_attempts,
        locks: lock_map.clone(),
        catalog_names,
        first_error: None,
        best: None,
        best_cost: usize::MAX,
    };
    let mut selected = BTreeMap::new();
    search(
        &catalog_map,
        roots,
        target,
        &mut selected,
        &mut state,
    )?;

    let Some(best) = state.best else {
        return Err(state.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_data = compute_node(&catalog_map, roots, &best, target);
    let selected_out = best
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut names: BTreeSet<String> = lock_map.keys().cloned().collect();
    names.extend(best.keys().cloned());
    let mut changes = Vec::new();

    for package in names {
        let before = lock_map.get(&package).copied();
        let after = best.get(&package).copied();
        if before == after {
            continue;
        }

        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let available = catalog_map.get(&package).is_some_and(|p| {
                    p.releases.iter().any(|r| r.version == old)
                });
                if !available {
                    ChangeReason::Unavailable
                } else if final_data
                    .constraints
                    .get(&package)
                    .is_some_and(|cs| cs.iter().any(|c| !matches(old, *c)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };

        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}

fn main() {}
