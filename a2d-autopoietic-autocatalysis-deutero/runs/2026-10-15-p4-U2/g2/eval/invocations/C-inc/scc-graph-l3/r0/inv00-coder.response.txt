use std::cmp::Reverse;
use std::collections::{BTreeMap, BinaryHeap, VecDeque};

pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
    BadQuery { side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Relation {
    Cycle,
    Trivial,
    Forward,
    Backward,
    Separate,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Query {
    pub kind: Relation,
    pub route: Vec<usize>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
    pub topo: Vec<usize>,
    pub depth: Vec<usize>,
}

struct DfsFrame {
    vertex: usize,
    next_neighbor: usize,
}

fn shortest_route(
    start: usize,
    goal: usize,
    adjacency: &[Vec<usize>],
) -> Vec<usize> {
    let mut parent = vec![usize::MAX; adjacency.len()];
    let mut seen = vec![false; adjacency.len()];
    let mut queue = VecDeque::new();
    seen[start] = true;
    queue.push_back(start);

    while let Some(vertex) = queue.pop_front() {
        if vertex == goal {
            break;
        }
        for &next in &adjacency[vertex] {
            if !seen[next] {
                seen[next] = true;
                parent[next] = vertex;
                queue.push_back(next);
            }
        }
    }

    if !seen[goal] {
        return Vec::new();
    }

    let mut route = Vec::new();
    let mut current = goal;
    route.push(current);
    while current != start {
        current = parent[current];
        route.push(current);
    }
    route.reverse();
    route
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }
    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }
    for (index, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge: index,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge: index,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse_adjacency = vec![Vec::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse_adjacency[to].push(from);
    }

    let mut visited = vec![false; n];
    let mut finish_order = Vec::with_capacity(n);
    for start in 0..n {
        if visited[start] {
            continue;
        }
        visited[start] = true;
        let mut stack = vec![DfsFrame {
            vertex: start,
            next_neighbor: 0,
        }];
        while let Some(frame) = stack.last_mut() {
            if frame.next_neighbor < adjacency[frame.vertex].len() {
                let next = adjacency[frame.vertex][frame.next_neighbor];
                frame.next_neighbor += 1;
                if !visited[next] {
                    visited[next] = true;
                    stack.push(DfsFrame {
                        vertex: next,
                        next_neighbor: 0,
                    });
                }
            } else {
                let finished = stack.pop().unwrap().vertex;
                finish_order.push(finished);
            }
        }
    }

    let mut raw_components: Vec<Vec<usize>> = Vec::new();
    let mut raw_component_of = vec![usize::MAX; n];
    for &start in finish_order.iter().rev() {
        if raw_component_of[start] != usize::MAX {
            continue;
        }
        let raw_id = raw_components.len();
        let mut members = Vec::new();
        let mut stack = vec![start];
        raw_component_of[start] = raw_id;
        while let Some(vertex) = stack.pop() {
            members.push(vertex);
            for &next in &reverse_adjacency[vertex] {
                if raw_component_of[next] == usize::MAX {
                    raw_component_of[next] = raw_id;
                    stack.push(next);
                }
            }
        }
        members.sort_unstable();
        raw_components.push(members);
    }

    let mut order: Vec<usize> = (0..raw_components.len()).collect();
    order.sort_by(|&left, &right| {
        raw_components[right]
            .len()
            .cmp(&raw_components[left].len())
            .then_with(|| raw_components[left][0].cmp(&raw_components[right][0]))
    });

    let mut raw_to_id = vec![0; raw_components.len()];
    let mut components = Vec::with_capacity(raw_components.len());
    for (id, raw_id) in order.into_iter().enumerate() {
        raw_to_id[raw_id] = id;
        components.push(raw_components[raw_id].clone());
    }

    let component_of: Vec<usize> = raw_component_of
        .into_iter()
        .map(|raw_id| raw_to_id[raw_id])
        .collect();

    let count = components.len();
    let mut cyclic: Vec<bool> = components.iter().map(|members| members.len() > 1).collect();
    let mut internal_edges = vec![0usize; count];
    let mut arc_data: BTreeMap<(usize, usize), (usize, usize)> = BTreeMap::new();

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let from_component = component_of[from];
        let to_component = component_of[to];
        if from_component == to_component {
            internal_edges[from_component] += 1;
            if from == to {
                cyclic[from_component] = true;
            }
        } else {
            let entry = arc_data
                .entry((from_component, to_component))
                .or_insert((0, edge_index));
            entry.0 += 1;
            entry.1 = entry.1.min(edge_index);
        }
    }

    let mut arcs: Vec<Arc> = arc_data
        .into_iter()
        .map(|((from, to), (multiplicity, first_edge))| Arc {
            from,
            to,
            multiplicity,
            first_edge,
        })
        .collect();
    arcs.sort_by(|left, right| {
        left.from
            .cmp(&right.from)
            .then_with(|| right.to.cmp(&left.to))
    });

    let mut component_adjacency = vec![Vec::new(); count];
    let mut indegree = vec![0usize; count];
    for arc in &arcs {
        component_adjacency[arc.from].push(arc.to);
        indegree[arc.to] += 1;
    }
    for neighbors in &mut component_adjacency {
        neighbors.sort_unstable();
    }

    let mut ready = BinaryHeap::new();
    for id in 0..count {
        if indegree[id] == 0 {
            ready.push((internal_edges[id], components[id][0], Reverse(id)));
        }
    }

    let mut topo = Vec::with_capacity(count);
    let mut depth = vec![0usize; count];
    while let Some((_, _, Reverse(id))) = ready.pop() {
        topo.push(id);
        for &next in &component_adjacency[id] {
            depth[next] = depth[next].max(depth[id] + 1);
            indegree[next] -= 1;
            if indegree[next] == 0 {
                ready.push((internal_edges[next], components[next][0], Reverse(next)));
            }
        }
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
        topo,
        depth,
    })
}

impl Graph {
    pub fn query(&self, from: usize, to: usize) -> Result<Query, GraphError> {
        if from >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 0,
                vertex: from,
            });
        }
        if to >= self.component_of.len() {
            return Err(GraphError::BadQuery {
                side: 1,
                vertex: to,
            });
        }

        let a = self.component_of[from];
        let b = self.component_of[to];

        if a == b {
            if self.cyclic[a] {
                return Ok(Query {
                    kind: Relation::Cycle,
                    route: vec![a, a],
                });
            }
            return Ok(Query {
                kind: Relation::Trivial,
                route: vec![a],
            });
        }

        let mut adjacency = vec![Vec::new(); self.components.len()];
        for arc in &self.arcs {
            adjacency[arc.from].push(arc.to);
        }
        for neighbors in &mut adjacency {
            neighbors.sort_unstable();
            neighbors.dedup();
        }

        let forward = shortest_route(a, b, &adjacency);
        if !forward.is_empty() {
            return Ok(Query {
                kind: Relation::Forward,
                route: forward,
            });
        }

        let mut backward = shortest_route(b, a, &adjacency);
        if !backward.is_empty() {
            backward.reverse();
            return Ok(Query {
                kind: Relation::Backward,
                route: backward,
            });
        }

        Ok(Query {
            kind: Relation::Separate,
            route: Vec::new(),
        })
    }
}

fn main() {}
