#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.bytes().position(|byte| byte == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.bytes().position(|byte| byte == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .bytes()
        .position(|byte| (1..=31).contains(&byte) || byte == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str) -> impl Iterator<Item = (usize, &str)> {
    path.match_indices('/')
        .scan((0usize, path), |state, (slash_at, _)| {
            let (start, remaining) = *state;
            let component = &remaining[..slash_at];
            let component_start = start;
            *state = (start + slash_at + 1, &remaining[slash_at + 1..]);
            Some((component_start, component))
        })
        .chain({
            let start = path.rfind('/').map_or(0, |at| at + 1);
            std::iter::once((start, &path[start..]))
        })
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn normalize_validated(path: &str) -> Result<String, PathError> {
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut stack: Vec<&str> = Vec::new();

    for (at, component) in components(path) {
        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if stack.last().is_some_and(|last| *last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..");
            }
        } else {
            stack.push(component);
        }
    }

    let mut result = String::new();
    if absolute {
        result.push('/');
    }
    result.push_str(&stack.join("/"));

    if result.is_empty() {
        result.push('.');
    }

    if intent && result != "/" && result != "." {
        result.push('/');
    }

    Ok(result)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    normalize_validated(path)
}

fn normalized_context(path: &str) -> Result<Vec<String>, ()> {
    if !path.starts_with('/') {
        return Err(());
    }
    let normalized = normalize(path).map_err(|_| ())?;
    Ok(normalized
        .split('/')
        .filter(|component| !component.is_empty())
        .map(str::to_owned)
        .collect())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = normalized_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let home_components = match home {
        Some(home) => Some(
            normalized_context(home).map_err(|_| PathError::InvalidHome)?,
        ),
        None => None,
    };

    validate_raw(path)?;

    let (absolute, mut stack, remainder, offset) = if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home_components = home_components.ok_or(PathError::MissingHome)?;
        let offset = if bytes.len() > 1 { 2 } else { 1 };
        let remainder = if bytes.len() > 1 { &path[2..] } else { "" };
        (true, home_components, remainder, offset)
    } else if path.starts_with('/') {
        (true, Vec::new(), path, 0)
    } else {
        (true, cwd_components, path, 0)
    };

    let intent = directory_intent(path);
    let mut search_at = 0usize;
    for component in remainder.split('/') {
        let at = offset + search_at;
        search_at += component.len() + 1;

        if component.is_empty() || component == "." {
            continue;
        }
        if component == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at });
            }
        } else {
            stack.push(component.to_owned());
        }
    }

    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    if intent && result != "/" {
        result.push('/');
    }

    if !absolute {
        unreachable!();
    }
    Ok(result)
}

fn main() {}
