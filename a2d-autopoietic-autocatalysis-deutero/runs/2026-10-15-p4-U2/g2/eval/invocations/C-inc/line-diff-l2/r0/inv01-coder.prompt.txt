    Old,
    New,

    },

        old_index: usize,
        line: String,
    },

    },

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    ContextTooLarge { context: usize },
    CarriageReturn { side: Side, byte: usize },
    MissingFinalLf { side: Side },
    TooLarge { old_len: usize, new_len: usize },
}

fn split_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        Vec::new()
    } else {
        text[..text.len() - 1].split('\n').collect()
    }
}

fn validate(text: &str, side: Side) -> Result<(), DiffError> {
    if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn { side, byte });
    }
    if !text.is_empty() && !text.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side });
    }
    Ok(())
}

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }

    validate(old, Side::Old)?;
    validate(new, Side::New)?;

    let old_lines = split_lines(old);
    let new_lines = split_lines(new);

    if old_lines.len() > 128 || new_lines.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old_lines.len(),
            new_len: new_lines.len(),
        });
    }

    let n = old_lines.len();
    let m = new_lines.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old_lines[i] == new_lines[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut edits = Vec::new();
    let (mut i, mut j) = (0, 0);
    while i < n && j < m {
        if old_lines[i] == new_lines[j] {
            edits.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old_lines[i].to_owned(),
            });
            i += 1;
            j += 1;
        } else if lcs[i + 1][j] > lcs[i][j + 1]
            || (lcs[i + 1][j] == lcs[i][j + 1] && (i + j) % 2 == 0)
        {
            edits.push(Edit::Delete {
                old_index: i,
                line: old_lines[i].to_owned(),
            });
            i += 1;
        } else {
            edits.push(Edit::Insert {
                new_index: j,
                line: new_lines[j].to_owned(),
            });
            j += 1;
        }
    }
    while i < n {
        edits.push(Edit::Delete {
            old_index: i,
            line: old_lines[i].to_owned(),
        });
        i += 1;
    }
    while j < m {
        edits.push(Edit::Insert {
            new_index: j,
            line: new_lines[j].to_owned(),
        });
        j += 1;
    }

    let changed: Vec<bool> = edits
        .iter()
        .map(|edit| !matches!(edit, Edit::Keep { .. }))
        .collect();
    if !changed.iter().any(|&value| value) {
        return Ok(String::new());
    }

    let mut groups = Vec::new();
    let mut pos = 0;
    while pos < edits.len() {
        if !changed[pos] {
            pos += 1;
            continue;
        }
        let start = pos;
        while pos < edits.len() && changed[pos] {
            pos += 1;
        }
        groups.push((start, pos));
    }

    let max_gap = 2 * context;
    let mut merged = Vec::new();
    for group in groups {
        if let Some(last) = merged.last_mut() {
            let previous_end: &mut usize = &mut last.1;
            let gap = group.0 - *previous_end;
            if gap <= max_gap {
                *previous_end = group.1;
                continue;
            }
        }
        merged.push(group);
    }

    let mut hunks = Vec::new();
    for (start, end) in merged {
        let mut hunk_start = start;
        let mut preceding = 0;
        while hunk_start > 0 && preceding < context {
            hunk_start -= 1;
            if matches!(edits[hunk_start], Edit::Keep { .. }) {
                preceding += 1;
            } else {
                break;
            }
        }

        let mut hunk_end = end;
        let mut following = 0;
        while hunk_end < edits.len() && following < context {
            if matches!(edits[hunk_end], Edit::Keep { .. }) {
                following += 1;
            } else {
                break;
            }
            hunk_end += 1;
        }
        hunks.push((hunk_start, hunk_end));
    }

    let mut output = String::from("--- before\n+++ after\n");
    for (start, end) in hunks {
        let old_count = edits[start..end]
            .iter()
            .filter(|edit| matches!(edit, Edit::Keep { .. } | Edit::Delete { .. }))
            .count();
        let new_count = edits[start..end]
            .iter()
            .filter(|edit| matches!(edit, Edit::Keep { .. } | Edit::Insert { .. }))
            .count();

        let old_start = edits[..start]
            .iter()
            .filter(|edit| matches!(edit, Edit::Keep { .. } | Edit::Delete { .. }))
            .count();
        let new_start = edits[..start]
            .iter()
            .filter(|edit| matches!(edit, Edit::Keep { .. } | Edit::Insert { .. }))
            .count();
        let old_start = if old_count > 0 {
            old_start + 1
        } else {
            old_start
        };
        let new_start = if new_count > 0 {
            new_start + 1
        } else {
            new_start
        };

        output.push_str(&format!(
            "@@ -{old_start},{old_count} +{new_start},{new_count} @@\n"
        ));

                Edit::Keep { line, .. } => {
                    output.push(' ');
                    output.push_str(line);
                    output.push('\n');
                }
                Edit::Delete { line, .. } => {
                    output.push('-');
                    output.push_str(line);
                    output.push('\n');
                }
                Edit::Insert { line, .. } => {
                    output.push('+');
                    output.push_str(line);
                    output.push('\n');
                }
            }
        }
    }

    Ok(output)
}
