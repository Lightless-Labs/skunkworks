#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut index = if input.starts_with('\u{feff}') { 3 } else { 0 };

    if index == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = State::FieldStart;
    let mut quote_open = 0;
    let mut record_index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: index,
            });
        }

        if state != State::Quoted && byte == b'\r' {
            if index + 1 >= bytes.len() || bytes[index + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: index,
                });
            }
            index += 2;
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_index += 1;
            state = State::FieldStart;
            continue;
        }

        match state {
            State::FieldStart => match byte {
                b'"' => {
                    quote_open = index;
                    state = State::Quoted;
                    index += 1;
                }
                b',' => {
                    record.push(std::mem::take(&mut field));
                    index += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    index += 1;
                }
                _ => {
                    let width = utf8_width(byte);
                    field.push_str(&input[index..index + width]);
                    state = State::Unquoted;
                    index += width;
                }
            },
            State::Unquoted => match byte {
                b'"' => {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: index,
                    });
                }
                b',' => {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    index += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    index += 1;
                }
                _ => {
                    let width = utf8_width(byte);
                    field.push_str(&input[index..index + width]);
                    index += width;
                }
            },
            State::Quoted => {
                if byte == b'"' {
                    if index + 1 < bytes.len() && bytes[index + 1] == b'"' {
                        field.push('"');
                        index += 2;
                    } else {
                        state = State::AfterQuote;
                        index += 1;
                    }
                } else {
                    let width = utf8_width(byte);
                    field.push_str(&input[index..index + width]);
                    index += width;
                }
            }
            State::AfterQuote => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    index += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    index += 1;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: index,
                    });
                }
            },
        }
    }

    if state == State::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_open,
        });
    }

    if state != State::FieldStart || !record.is_empty() {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum State {
    FieldStart,
    Unquoted,
    Quoted,
    AfterQuote,
}

fn utf8_width(first: u8) -> usize {
    match first {
        0x00..=0x7f => 1,
        0xc2..=0xdf => 2,
        0xe0..=0xef => 3,
        _ => 4,
    }
}

fn main() {}
