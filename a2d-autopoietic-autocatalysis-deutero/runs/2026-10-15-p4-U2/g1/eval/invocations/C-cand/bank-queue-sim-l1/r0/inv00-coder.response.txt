#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut arrival_index = 0;
    let mut now = 0u64;
    let mut finish: Option<u64> = None;
    let mut last_finish: Option<u64> = None;

    loop {
        let next_arrival = arrivals.get(arrival_index).map(|c| c.arrival);
        let next_finish = finish;

        let tick = match (next_arrival, next_finish) {
            (None, None) => break,
            (Some(a), None) => a,
            (None, Some(f)) => f,
            (Some(a), Some(f)) => a.min(f),
        };
        now = tick;

        if finish == Some(tick) {
            let finished_customer = match events.iter().rev().find_map(|event| {
                if event.time == tick {
                    match event.kind {
                        Kind::Start { finish: f, .. } if f == tick => Some(event.customer),
                        _ => None,
                    }
                } else {
                    None
                }
            }) {
                Some(id) => id,
                None => {
                    // A positive occupation's Start may be earlier than this tick.
                    let id = events.iter().rev().find_map(|event| match event.kind {
                        Kind::Start { finish: f, .. } if f == tick => Some(event.customer),
                        _ => None,
                    });
                    id.expect("scheduled completion must have a matching start")
                }
            };
            events.push(Event {
                time: tick,
                customer: finished_customer,
                kind: Kind::Finish,
            });
            finish = None;
            last_finish = Some(tick);
        }

        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == tick {
            let customer = arrivals[arrival_index];
            events.push(Event {
                time: tick,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            arrival_index += 1;
        }

        if finish.is_none() {
            if let Some(customer) = waiting.pop_front() {
                let wake = last_finish.map_or(true, |previous| tick > previous);
                let base_finish = tick.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: tick,
                    },
                )?;
                let customer_finish = if wake {
                    base_finish.checked_add(wake_cost).ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time: tick,
                    })?
                } else {
                    base_finish
                };

                events.push(Event {
                    time: tick,
                    customer: customer.id,
                    kind: Kind::Start {
                        finish: customer_finish,
                        wake,
                    },
                });

                if customer_finish == tick {
                    events.push(Event {
                        time: tick,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    last_finish = Some(tick);
                } else {
                    finish = Some(customer_finish);
                }
            }
        }
    }

    Ok(events)
}

fn main() {}
