#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged {
        row: usize,
        expected: usize,
        actual: usize,
    },
    InvalidChar {
        row: usize,
        column: usize,
        byte: usize,
    },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn fixed_overhead(mode: &Mode, columns: usize) -> usize {
    match mode {
        Mode::Grid | Mode::Markdown => 3 * columns + 1,
        Mode::Compact => 2 * columns.saturating_sub(1),
    }
}

fn minimum_column_width(mode: &Mode) -> usize {
    match mode {
        Mode::Markdown => 3,
        Mode::Grid | Mode::Compact => 1,
    }
}

fn validate_characters(
    rows: &[Vec<String>],
) -> Result<(), TableError> {
    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            let mut chars = cell.char_indices().peekable();

            while let Some((byte, ch)) = chars.next() {
                if ch == '\r' {
                    if matches!(chars.peek(), Some((_, '\n'))) {
                        chars.next();
                        continue;
                    }
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }

                if (ch.is_control() && ch != '\n') || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
            }
        }
    }

    Ok(())
}

fn normalized_lines(cell: &str) -> Vec<String> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            line.trim_end_matches(' ')
                .to_owned()
        })
        .collect()
}

fn tokens_for_line(line: &str, markdown: bool) -> Vec<Token> {
    let mut tokens = Vec::new();

    for ch in line.chars() {
        if markdown && ch == '|' {
            tokens.push(Token {
                text: "\\|".to_owned(),
                width: 2,
            });
        } else if markdown && ch == '\\' {
            tokens.push(Token {
                text: "\\\\".to_owned(),
                width: 2,
            });
        } else {
            tokens.push(Token {
                text: ch.to_string(),
                width: 1,
            });
        }
    }

    tokens
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|token| token.width).sum()
}

fn split_fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.iter().map(|token| token.text.as_str()).collect()];
            }

            let available = width.saturating_sub(1);
            let mut used = 0;
            let mut fragment = String::new();

            for token in tokens {
                if used + token.width > available {
                    break;
                }
                used += token.width;
                fragment.push_str(&token.text);
            }
            fragment.push('…');
            vec![fragment]
        }
        Overflow::Wrap => {
            let mut fragments = Vec::new();
            let mut start = 0;

            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;

                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }

                let fragment = tokens[start..end]
                    .iter()
                    .map(|token| token.text.as_str())
                    .collect();
                fragments.push(fragment);
                start = end;
            }

            fragments
        }
    }
}

fn pad(content: &str, content_width: usize, width: usize, alignment: &Alignment) -> String {
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };

    let mut result = String::with_capacity(content.len() + extra);
    result.push_str(&" ".repeat(left));
    result.push_str(content);
    result.push_str(&" ".repeat(right));
    result
}

fn grid_border(widths: &[usize]) -> String {
    let mut result = String::from("+");
    for width in widths {
        result.push_str(&"-".repeat(width + 2));
        result.push('+');
    }
    result.push('\n');
    result
}

fn emit_data_line(
    fragments: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    let mut result = String::new();

    match mode {
        Mode::Grid | Mode::Markdown => {
            result.push('|');
            for (index, fragment) in fragments.iter().enumerate() {
                let content_width = fragment.chars().count();
                result.push(' ');
                result.push_str(&pad(
                    fragment,
                    content_width,
                    widths[index],
                    &alignments[index],
                ));
                result.push_str(" |");
            }
        }
        Mode::Compact => {
            for index in 0..fragments.len() {
                if index > 0 {
                    result.push_str("  ");
                }

                let fragment = &fragments[index];
                let content_width = fragment.chars().count();
                let alignment = &alignments[index];
                let padded = pad(fragment, content_width, widths[index], alignment);

                if index + 1 == fragments.len() {
                    let right_padding = match alignment {
                        Alignment::Left => widths[index].saturating_sub(content_width),
                        Alignment::Right | Alignment::Center => {
                            let extra = widths[index].saturating_sub(content_width);
                            match alignment {
                                Alignment::Right => 0,
                                Alignment::Center => extra - extra / 2,
                                Alignment::Left => 0,
                            }
                        }
                    };

                    let keep = padded.chars().count().saturating_sub(right_padding);
                    result.extend(padded.chars().take(keep));
                } else {
                    result.push_str(&padded);
                }
            }
        }
    }

    result.push('\n');
    result
}

fn emit_markdown_marker(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut result = String::from("|");

    for (index, width) in widths.iter().enumerate() {
        result.push(' ');
        match alignments[index] {
            Alignment::Left => {
                result.push(':');
                result.push_str(&"-".repeat(width - 1));
            }
            Alignment::Right => {
                result.push_str(&"-".repeat(width - 1));
                result.push(':');
            }
            Alignment::Center => {
                result.push(':');
                result.push_str(&"-".repeat(width - 2));
                result.push(':');
            }
        }
        result.push_str(" |");
    }

    result.push('\n');
    result
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let columns = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != columns {
            return Err(TableError::Ragged {
                row,
                expected: columns,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != columns {
        return Err(TableError::OverflowCount {
            expected: columns,
            actual: options.overflow.len(),
        });
    }

    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let min_column = minimum_column_width(&options.mode);
    let overhead = fixed_overhead(&options.mode, columns);
    let minimum = overhead + columns * min_column;

    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    validate_characters(rows)?;

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; columns]
    } else {
        options.alignments.clone()
    };

    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; columns]
    } else {
        options.overflow.clone()
    };

    let markdown = options.mode == Mode::Markdown;
    let logical_lines: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| {
                    normalized_lines(cell)
                        .iter()
                        .map(|line| tokens_for_line(line, markdown))
                        .collect()
                })
                .collect()
        })
        .collect();

    let mut widths = vec![min_column; columns];
    for row in &logical_lines {
        for (column, lines) in row.iter().enumerate() {
            for line in lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let mut total = overhead + widths.iter().sum::<usize>();
        while total > options.max_width {
            let candidate = (0..columns)
                .rev()
                .find(|&column| widths[column] > min_column);
            if let Some(column) = candidate {
                widths[column] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();

    if options.mode == Mode::Grid {
        output.push_str(&grid_border(&widths));
    }

    for (row_index, row) in logical_lines.iter().enumerate() {
        let cell_fragments: Vec<Vec<String>> = row
            .iter()
            .enumerate()
            .map(|(column, lines)| {
                let mut fragments = Vec::new();
                for line in lines {
                    fragments.extend(split_fragments(
                        line,
                        widths[column],
                        &overflows[column],
                    ));
                }
                fragments
            })
            .collect();

        let height = cell_fragments
            .iter()
            .map(Vec::len)
            .max()
            .unwrap_or(1);

        for fragment_index in 0..height {
            let fragments: Vec<String> = cell_fragments
                .iter()
                .map(|cell| cell.get(fragment_index).cloned().unwrap_or_default())
                .collect();

            output.push_str(&emit_data_line(
                &fragments,
                &widths,
                &alignments,
                &options.mode,
            ));
        }

        if options.mode == Mode::Markdown && row_index == 0 {
            output.push_str(&emit_markdown_marker(&widths, &alignments));
        }

        if options.mode == Mode::Grid && options.header && row_index == 0 && rows.len() > 1 {
            output.push_str(&grid_border(&widths));
        }
    }

    if options.mode == Mode::Grid {
        output.push_str(&grid_border(&widths));
    }

    Ok(output)
}

fn main() {}
