#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut result = encoder.push(input).expect("new encoder is open");
    result.push_str(&encoder.finish().expect("new encoder is open"));
    result
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut result = decoder.push(input)?;
    result.extend(decoder.finish()?);
    Ok(result)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                }
                self.tail_len += 1;
                index += 1;
            }
            if self.tail_len == 3 {
                append_triplet(&mut output, self.tail[0], self.tail[1], input[index - 1]);
                self.tail_len = 0;
            }
        }

        while index + 3 <= input.len() {
            append_triplet(
                &mut output,
                input[index],
                input[index + 1],
                input[index + 2],
            );
            index += 3;
        }

        let remaining = input.len() - index;
        self.tail_len = remaining;
        for i in 0..remaining {
            self.tail[i] = input[index + i];
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            1 => {
                let a = self.tail[0];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                output.push_str("==");
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => {}
        }
        self.tail_len = 0;
        Ok(output)
    }
}

fn append_triplet(output: &mut String, a: u8, b: u8, c: u8) {
    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    output.push(ALPHABET[(c & 0x3f) as usize] as char);
}

fn digit(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&b| b == byte).map(|i| i as u8)
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    pending_second_padding: bool,
    padded: bool,
    offset: usize,
    finished: bool,
    error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            len: 0,
            pending_second_padding: false,
            padded: false,
            offset: 0,
            finished: false,
            error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let current_offset = self.offset;
            self.offset += 1;

            if self.padded {
                return self.fail(Error::BadPadding { offset: current_offset });
            }
            if self.pending_second_padding && byte != b'=' {
                return self.fail(Error::BadPadding { offset: current_offset });
            }

            let position = self.len;
            if byte == b'=' {
                if position <= 1 {
                    return self.fail(Error::BadPadding { offset: current_offset });
                }
                if position == 2 {
                    self.pending_second_padding = true;
                    self.group[position] = byte;
                    self.offsets[position] = current_offset;
                    self.len = 3;
                    continue;
                }
                if position == 3 {
                    self.group[position] = byte;
                    self.offsets[position] = current_offset;
                    self.len = 4;
                    output.extend(decode_group(&self.group));
                    self.padded = true;
                    self.len = 0;
                    self.pending_second_padding = false;
                    continue;
                }
            }

            let value = match digit(byte) {
                Some(value) => value,
                None => return self.fail(Error::InvalidByte { offset: current_offset, byte }),
            };

            self.group[position] = value;
            self.offsets[position] = current_offset;
            self.len += 1;

            if self.len == 4 {
                output.extend(decode_group(&self.group));
                self.len = 0;
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if self.pending_second_padding {
            return self.fail(Error::BadPadding {
                offset: self.offsets[2],
            });
        }
        if self.len != 0 {
            return self.fail(Error::Truncated {
                offset: self.offsets[0],
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.error = Some(error.clone());
        Err(error)
    }
}

fn decode_group(group: &[u8; 4]) -> Vec<u8> {
    let a = group[0];
    let b = group[1];

    if group[2] == b'=' {
        return vec![(a << 2) | (b >> 4)];
    }

    let c = group[2];
    let first = (a << 2) | (b >> 4);
    let second = (b << 4) | (c >> 2);

    if group[3] == b'=' {
        vec![first, second]
    } else {
        let d = group[3];
        vec![first, second, (c << 6) | d]
    }
}

fn decode_digit_group(group: &[u8; 4]) -> Vec<u8> {
    let mut values = [0u8; 4];
    for i in 0..4 {
        if group[i] != b'=' {
            values[i] = digit(group[i]).unwrap_or(0);
        }
    }
    decode_group(&values)
}

fn decode_group(group: &[u8; 4]) -> Vec<u8> {
    let a = group[0];
    let b = group[1];
    let first = (a << 2) | (b >> 4);

    if group[2] == b'=' {
        return vec![first];
    }

    let c = group[2];
    let second = (b << 4) | (c >> 2);
    if group[3] == b'=' {
        vec![first, second]
    } else {
        vec![first, second, (c << 6) | group[3]]
    }
}
