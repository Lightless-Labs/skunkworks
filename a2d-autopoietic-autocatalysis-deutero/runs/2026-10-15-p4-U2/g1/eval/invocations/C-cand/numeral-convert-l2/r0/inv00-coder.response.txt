#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let mut payload_start = 0;
    if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = hash + 1;
        if payload_start == bytes.len() {
            return Err(BaseError::Empty { operand });
        }
    }

    let payload = &bytes[payload_start..];

    // Digit errors take precedence over separator and leading-zero errors.
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => {}
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    let mut digits = Vec::new();
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = offset > 0 && payload[offset - 1] != b'_';
            let has_digit_after =
                offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        } else if let Some(value) = digit_value(byte) {
            digits.push(value);
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

// Little-endian arbitrary-precision integer in the requested radix.
fn multiply_add(value: &mut Vec<u8>, radix: u8, digit: u8) {
    let mut carry = digit as u16;
    for limb in value.iter_mut() {
        let next = *limb as u16 * radix as u16 + carry;
        *limb = (next % 10_000) as u8;
        carry = next / 10_000;
    }
    while carry != 0 {
        value.push((carry % 10_000) as u8);
        carry /= 10_000;
    }
}

fn parse_value(digits: &[u8], radix: u8) -> Vec<u8> {
    let mut value = vec![0u8];
    for &digit in digits {
        multiply_add(&mut value, radix, digit);
    }
    value
}

// Convert a base-10000 little-endian integer to the requested radix.
fn render_value(mut value: Vec<u8>, radix: u8, style: &BaseStyle) -> String {
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }

    let mut digits = Vec::new();
    loop {
        let mut remainder = 0u32;
        for limb in value.iter_mut().rev() {
            let current = remainder * 10_000 + *limb as u32;
            *limb = (current / radix as u32) as u8;
            remainder = current % radix as u32;
        }
        digits.push(remainder as u8);

        while value.len() > 1 && value.last() == Some(&0) {
            value.pop();
        }
        if value.len() == 1 && value[0] == 0 {
            break;
        }
    }

    while digits.len() < style.min_width {
        digits.push(0);
    }

    let mut output = String::new();
    for (index, &digit) in digits.iter().rev().enumerate() {
        if style.group != 0 && index != 0 && (digits.len() - index) % style.group == 0 {
            output.push('_');
        }
        let byte = match digit {
            0..=9 => b'0' + digit,
            _ if style.upper => b'A' + (digit - 10),
            _ => b'a' + (digit - 10),
        };
        output.push(byte as char);
    }
    output
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;

    let digits = parse_operand(input, from, 0)?;
    let value = parse_value(&digits, from);
    Ok(render_value(value, to, &style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let mut sum = parse_value(&left_digits, base);
    let right_value = parse_value(&right_digits, base);

    let len = sum.len().max(right_value.len());
    sum.resize(len, 0);

    let mut carry = 0u32;
    for index in 0..len {
        let total = sum[index] as u32
            + right_value.get(index).copied().unwrap_or(0) as u32
            + carry;
        sum[index] = (total % 10_000) as u8;
        carry = total / 10_000;
    }
    if carry != 0 {
        sum.push(carry as u8);
    }

    Ok(render_value(sum, base, &style))
}

fn main() {}
