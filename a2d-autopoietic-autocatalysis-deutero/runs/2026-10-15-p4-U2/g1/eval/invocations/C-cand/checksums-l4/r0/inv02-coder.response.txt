#[derive(Debug, Clone)]
struct Crc32State {
    raw: u32,
}

impl Crc32State {
    fn new() -> Self {
        Self { raw: 0xffff_ffff }
    }

    fn from_finalized(value: u32) -> Self {
        Self {
            raw: value ^ 0xffff_ffff,
        }
    }

    fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.raw ^= byte as u32;
            for _ in 0..8 {
                let low = self.raw & 1;
                self.raw >>= 1;
                if low != 0 {
                    self.raw ^= 0xedb8_8320;
                }
            }
        }
    }

    fn checksum(&self) -> u32 {
        self.raw ^ 0xffff_ffff
    }
}

#[derive(Debug, Clone)]
struct AdlerState {
    a: u32,
    b: u32,
}

impl AdlerState {
    fn new() -> Self {
        Self { a: 1, b: 0 }
    }

    fn from_finalized(value: u32) -> Self {
        Self {
            a: value & 0xffff,
            b: value >> 16,
        }
    }

    fn update(&mut self, data: &[u8]) {
        const MOD: u32 = 65_521;
        for &byte in data {
            self.a = (self.a + byte as u32) % MOD;
            self.b = (self.b + self.a) % MOD;
        }
    }

    fn checksum(&self) -> u32 {
        (self.b << 16) | self.a
    }
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: Crc32State,
    adler: AdlerState,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: Crc32State::new(),
            adler: AdlerState::new(),
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc.checksum(),
            adler32: self.adler.checksum(),
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: Crc32State::from_finalized(crc32),
            adler: AdlerState::from_finalized(adler32),
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc.update(data);
        self.adler.update(data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let sums = self.checksum();
        format!("c{:08x}:a{:08x}", sums.crc32, sums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let crc = parse_snapshot_hex(bytes, 1, 9)?;
        let adler = parse_snapshot_hex(bytes, 11, 19)?;
        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

#[derive(Debug, Clone)]
struct Fletcher32State {
    a: u32,
    b: u32,
    pending_high: Option<u8>,
}

impl Fletcher32State {
    fn new() -> Self {
        Self {
            a: 0,
            b: 0,
            pending_high: None,
        }
    }

    fn word(&mut self, word: u16) {
        const MOD: u32 = 65_535;
        self.a = (self.a + word as u32) % MOD;
        self.b = (self.b + self.a) % MOD;
    }

    fn update(&mut self, data: &[u8]) {
        let mut index = 0usize;
        if let Some(high) = self.pending_high.take() {
            if let Some(&low) = data.first() {
                self.word(u16::from_be_bytes([high, low]));
                index = 1;
            } else {
                self.pending_high = Some(high);
                return;
            }
        }
        while index + 1 < data.len() {
            self.word(u16::from_be_bytes([data[index], data[index + 1]]));
            index += 2;
        }
        if index < data.len() {
            self.pending_high = Some(data[index]);
        }
    }

    fn checksum(mut self) -> u32 {
        if let Some(high) = self.pending_high.take() {
            self.word(u16::from_be_bytes([high, 0]));
        }
        (self.b << 16) | self.a
    }
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut state = Fletcher32State::new();
    state.update(data);
    state.checksum()
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {

    let mut state = Fletcher32State::new();
    for span in spans {
        state.update(&data[span.start..span.end]);

    Ok(state.checksum())
}

#[derive(Debug, Clone)]
struct Crc16State {
    variant: Crc16,
    value: u16,
}

impl Crc16State {
    fn new(variant: Crc16) -> Self {
        let value = match variant {
            Crc16::Xmodem => 0,
            Crc16::False | Crc16::X25 => 0xffff,
        };
        Self { variant, value }
    }

    fn update(&mut self, data: &[u8]) {
        match self.variant {
            Crc16::Xmodem | Crc16::False => {
                for &byte in data {
                    self.value ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        let top = self.value & 0x8000;
                        self.value = self.value.wrapping_shl(1);
                        if top != 0 {
                            self.value ^= 0x1021;
                        }
                    }
                }
            }
            Crc16::X25 => {
                for &byte in data {
                    self.value ^= byte as u16;
                    for _ in 0..8 {
                        let low = self.value & 1;
                        self.value >>= 1;
                        if low != 0 {
                            self.value ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    fn checksum(&self) -> u16 {
        match self.variant {
            Crc16::X25 => self.value ^ 0xffff,
            Crc16::Xmodem | Crc16::False => self.value,
        }
    }
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut state = Crc16State::new(variant);
    state.update(data);
    state.checksum()
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut base = Stream::new();
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32 = Fletcher32State::new();
    let mut crc16_state = Crc16State::new(variant);

    for span in spans {
        let selected = &data[span.start..span.end];
        base.update(selected);
        for &byte in selected {
            f16_a = (f16_a + byte as u32) % 255;
            f16_b = (f16_b + f16_a) % 255;
        }
        f32.update(selected);
        crc16_state.update(selected);
    }

    Ok(Multi {
        base: base.checksum(),
        fletcher16: ((f16_b << 8) | f16_a) as u16,
        fletcher32: f32.checksum(),
        crc16: crc16_state.checksum(),
    })
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
}

fn zero_byte_operator() -> [u32; 32] {
    let mut operator = [0u32; 32];
    let mut one_bit = [0u32; 32];
    one_bit[0] = 0xedb8_8320;
    let mut row = 1u32;
    for entry in one_bit.iter_mut().skip(1) {
        *entry = row;
        row <<= 1;
    }
    let mut two_bits = [0u32; 32];
    gf2_matrix_square(&mut two_bits, &one_bit);
    let mut four_bits = [0u32; 32];
    gf2_matrix_square(&mut four_bits, &two_bits);
    let mut eight_bits = [0u32; 32];
    gf2_matrix_square(&mut eight_bits, &four_bits);

    for (index, slot) in operator.iter_mut().enumerate() {
        *slot = gf2_matrix_times(&eight_bits, 1u32 << index);
    }
    operator
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut operator = zero_byte_operator();
    let mut length = right_len;
    let mut transformed = left;
    while length != 0 {
        if length & 1 != 0 {
            transformed = gf2_matrix_times(&operator, transformed);
        }
        length >>= 1;
        if length != 0 {
            let mut squared = [0u32; 32];
            gf2_matrix_square(&mut squared, &operator);
            operator = squared;
        }
    }
    transformed ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut combined = 0u32;
    for part in parts {
        combined = crc32_combine(combined, part.crc32, part.len);
    }
    Ok(Part {
        crc32: combined,
        len: total,
    })
}

fn parse_snapshot_digit(byte: u8, offset: usize) -> Result<u32, SnapshotError> {
    match byte {
        b'0'..=b'9' => Ok((byte - b'0') as u32),
        b'a'..=b'f' => Ok((byte - b'a' + 10) as u32),
        _ => Err(SnapshotError::Hex { offset }),
    }
}

fn parse_snapshot_hex(
    bytes: &[u8],
    start: usize,
    end: usize,
) -> Result<u32, SnapshotError> {
    let mut value = 0u32;
    for offset in start..end {
        value = (value << 4) | parse_snapshot_digit(bytes[offset], offset)?;
    }
    Ok(value)
}

fn parse_verify_digit(byte: u8, offset: usize) -> Result<u32, VerifyError> {
    match byte {
        b'0'..=b'9' => Ok((byte - b'0') as u32),
        b'a'..=b'f' => Ok((byte - b'a' + 10) as u32),
        _ => Err(VerifyError::Hex { offset }),
    }
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    let mut length = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        length += selected.len() as u64;
        stream.update(selected);
    }
    let sums = stream.checksum();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        length, sums.crc32, sums.adler32
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let mut digits = [0u32; 32];
    let ranges = [(4usize, 20usize), (25, 33), (40, 48)];
    let mut index = 0usize;
    for (start, end) in ranges {
        for offset in start..end {
            digits[index] = parse_verify_digit(bytes[offset], offset)?;
            index += 1;
        }
    }

    let mut expected_len = 0u64;
    for digit in &digits[..16] {
        expected_len = (expected_len << 4) | *digit as u64;
    }
    let mut expected_crc = 0u32;
    for digit in &digits[16..24] {
        expected_crc = (expected_crc << 4) | *digit;
    }
    let mut expected_adler = 0u32;
    for digit in &digits[24..32] {
        expected_adler = (expected_adler << 4) | *digit;
    }

    let mut stream = Stream::new();
    let mut length = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        length += selected.len() as u64;
        stream.update(selected);
    }

    if expected_len != length {
        return Err(VerifyError::Length);
    }

    let actual = stream.checksum();
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
