#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    breaks: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    words: Vec<String>,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn parse_segment(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars_in_word = Vec::new();
    let mut breaks = Vec::new();

    for &c in chars {
        if is_separator(c) {
            if !chars_in_word.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars_in_word),
                    breaks: std::mem::take(&mut breaks),
                });
            }
        } else if c == '~' {
            breaks.push(chars_in_word.len());
        } else {
            chars_in_word.push(c);
        }
    }

    if !chars_in_word.is_empty() {
        words.push(Word {
            chars: chars_in_word,
            breaks,
        });
    }

    words
}

fn word_text(chars: &[char]) -> String {
    chars.iter().collect()
}

fn content_len(line: &Line) -> usize {
    line.words.iter().map(|word| word.chars().count()).sum()
}

fn render_line(
    line: &Line,
    width: usize,
    justify: bool,
    is_final: bool,
    line_index: usize,
) -> String {
    let mut result = " ".repeat(line.indent);
    let pieces_len = content_len(line);
    let gaps = line.words.len().saturating_sub(1);

    let extra = if justify && !is_final && line.words.len() >= 2 {
        width.saturating_sub(line.indent + pieces_len + gaps)
    } else {
        0
    };

    let spaces_each = if gaps > 0 { extra / gaps } else { 0 };
    let remainder = if gaps > 0 { extra % gaps } else { 0 };
    let start_gap = if gaps > 0 { line_index % gaps } else { 0 };

    for (i, word) in line.words.iter().enumerate() {
        if i > 0 {
            let gap_index = i - 1;
            let distance = (gap_index + gaps - start_gap) % gaps;
            let bonus = usize::from(distance < remainder);
            result.push_str(&" ".repeat(1 + spaces_each + bonus));
        }
        result.push_str(word);
    }

    result
}

fn layout_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    let mut lines = Vec::new();
    let mut line = Line {
        indent: options.first_indent,
        words: Vec::new(),
    };
    let mut next_word = 0;
    let mut offset = 0;

    while next_word < words.len() {
        let word = &words[next_word];
        let capacity = options.width - line.indent;
        let current_len = content_len(&line);
        let separator_cost = usize::from(!line.words.is_empty());
        let room = capacity.saturating_sub(current_len + separator_cost);
        let suffix_len = word.chars.len() - offset;

        if suffix_len <= room {
            line.words.push(word_text(&word.chars[offset..]));
            next_word += 1;
            offset = 0;
            continue;
        }

        let available = capacity.saturating_sub(current_len);
        let rightmost = word
            .breaks
            .iter()
            .copied()
            .filter(|&break_at| {
                break_at > offset
                    && break_at < word.chars.len()
                    && break_at - offset + 1 <= available
            })
            .last();

        let accepted = rightmost.filter(|&break_at| {
            if line.words.is_empty() {
                true
            } else {
                let used = break_at - offset + 1;
                let unused = available - used;
                let baseline_unused = capacity - current_len;
                unused * unused + options.hyphen_cost < baseline_unused * baseline_unused
            }
        });

        if let Some(break_at) = accepted {
            let mut piece = word_text(&word.chars[offset..break_at]);
            piece.push('-');
            line.words.push(piece);
            lines.push(line);
            line = Line {
                indent: options.rest_indent,
                words: Vec::new(),
            };
            offset = break_at;
            continue;
        }

        if !line.words.is_empty() {
            lines.push(line);
            line = Line {
                indent: options.rest_indent,
                words: Vec::new(),
            };
            continue;
        }

        if suffix_len > capacity {
            line.words.push(word_text(&word.chars[offset..offset + capacity]));
            lines.push(line);
            line = Line {
                indent: options.rest_indent,
                words: Vec::new(),
            };
            offset += capacity;
        } else {
            line.words.push(word_text(&word.chars[offset..]));
            next_word += 1;
            offset = 0;
        }
    }

    if !line.words.is_empty() {
        lines.push(line);
    }

    let last_line = lines.len().saturating_sub(1);
    lines
        .iter()
        .enumerate()
        .map(|(index, line)| {
            render_line(
                line,
                options.width,
                options.justify,
                index == last_line,
                index,
            )
        })
        .collect()
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();

    for &(index, c) in &original {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..original.len() {
        if original[i].1 == '~' {
            let left_valid =
                i > 0 && original[i - 1].1 != '~' && !is_separator(original[i - 1].1);
            let right_valid = i + 1 < original.len()
                && original[i + 1].1 != '~'
                && !is_separator(original[i + 1].1);

            if !left_valid || !right_valid {
                return Err(WrapError::InvalidMarker {
                    index: original[i].0,
                });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut chars = input.chars().peekable();

    while let Some(c) = chars.next() {
        if c == '\r' {
            if chars.peek() == Some(&'\n') {
                chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut segments = Vec::new();
    let mut boundaries = Vec::new();
    let mut segment = Vec::new();
    let mut index = 0;

    while index < normalized.len() {
        if normalized[index] == '\n' {
            let mut end = index;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }

            if end - index >= 2 {
                segments.push(std::mem::take(&mut segment));
                boundaries.push(end - index - 1);
                index = end;
                continue;
            }
        }

        segment.push(normalized[index]);
        index += 1;
    }
    segments.push(segment);

    let mut output = Vec::new();

    for (paragraph_index, segment) in segments.iter().enumerate() {
        let words = parse_segment(segment);
        if !words.is_empty() {
            output.extend(layout_paragraph(&words, options));
        }

        if paragraph_index < boundaries.len() {
            for _ in 0..boundaries[paragraph_index] {
                output.push(String::new());
            }
        }
    }

    Ok(output)
}

fn main() {}
