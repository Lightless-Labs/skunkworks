use std::collections::BTreeSet;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [BTreeSet<u8>; 5],
    unrestricted_dom: bool,
    unrestricted_dow: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn name_value(field: Field, value: &str) -> Option<u8> {
    let names: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    names
        .iter()
        .position(|name| value.eq_ignore_ascii_case(name))
        .map(|i| if field == Field::Month { (i + 1) as u8 } else { i as u8 })
}

fn parse_number(text: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    let bytes = text.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn parse_atom(text: &str, field: Field, offset: usize) -> Result<u8, ParseError> {
    if text.bytes().all(|b| b.is_ascii_digit()) {
        let n = parse_number(text, field, offset)?;
        let (lo, hi) = bounds(field);
        if n < lo as u32 || n > hi as u32 {
            return Err(ParseError::OutOfRange { field, offset });
        }
        return Ok(n as u8);
    }
    name_value(field, text).ok_or(ParseError::InvalidSyntax { field, offset })
}

fn parse_step(text: &str, field: Field, offset: usize) -> Result<u8, ParseError> {
    if text.is_empty() || !text.bytes().all(|b| b.is_ascii_digit()) {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    let n = parse_number(text, field, offset)?;
    let (lo, hi) = bounds(field);
    let width = hi as u32 - lo as u32 + 1;
    if n == 0 || n > width {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(n as u8)
}

fn parse_field(text: &str, field: Field) -> Result<BTreeSet<u8>, ParseError> {
    for (offset, byte) in text.bytes().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = bounds(field);
    let mut selected = BTreeSet::new();
    let mut item_offset = 0usize;

    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_offset,
            });
        }

        let slashes: Vec<usize> = item.match_indices('/').map(|(i, _)| i).collect();
        if slashes.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }

        let (base, step_text, step_offset) = if let Some(&slash) = slashes.first() {
            (&item[..slash], Some(&item[slash + 1..]), item_offset + slash + 1)
        } else {
            (item, None, item_offset)
        };

        let hyphens: Vec<usize> = base.match_indices('-').map(|(i, _)| i).collect();
        if base.is_empty()
            || step_text.is_some_and(str::is_empty)
            || hyphens.len() > 1
            || (hyphens.len() == 1
                && (base.starts_with('-')
                    || base.ends_with('-')
                    || &base[..hyphens[0]] == "*"
                    || &base[hyphens[0] + 1..] == "*"))
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }

        let step = if let Some(step_text) = step_text {
            Some(parse_step(step_text, field, step_offset)?)
        } else {
            None
        };

        let (start, end) = if base == "*" {
            (minimum, maximum)
        } else if let Some(&hyphen) = hyphens.first() {
            let left = &base[..hyphen];
            let right = &base[hyphen + 1..];
            let valid_atom = |s: &str| {
                (!s.is_empty() && s.bytes().all(|b| b.is_ascii_digit()))
                    || name_value(field, s).is_some()
            };
            if !valid_atom(left) || !valid_atom(right) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_offset,
                });
            }
            let a = parse_atom(left, field, item_offset)?;
            let b = parse_atom(right, field, item_offset + hyphen + 1)?;
            if a > b {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_offset + hyphen + 1,
                });
            }
            (a, b)
        } else {
            let valid_atom = (!base.is_empty() && base.bytes().all(|b| b.is_ascii_digit()))
                || name_value(field, base).is_some();
            if !valid_atom {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_offset,
                });
            }
            let value = parse_atom(base, field, item_offset)?;
            (value, if step.is_some() { maximum } else { value })
        };

        let increment = step.unwrap_or(1);
        let mut value = start;
        while value <= end {
            selected.insert(value);
            match value.checked_add(increment) {
                Some(next) => value = next,
                None => break,
            }
        }

        item_offset += item.len() + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.starts_with('@') {
        match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        input
    };

    let parts: Vec<&str> = expanded.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets: [BTreeSet<u8>; 5] = std::array::from_fn(|_| BTreeSet::new());
    for i in 0..5 {
        sets[i] = parse_field(parts[i], FIELDS[i])?;
    }

    Ok(Cron {
        sets,
        unrestricted_dom: parts[2] == "*",
        unrestricted_dow: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.sets[field_index(field)].iter().copied().collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut date = (after.year, after.month, after.day);
        let mut limit = after.hour as u16 * 60 + after.minute as u16;

        loop {
            if self.date_matches(date.0, date.1, date.2) {
                for hour in &self.sets[1] {
                    for minute in &self.sets[0] {
                        let value = *hour as u16 * 60 + *minute as u16;
                        if value > limit {
                            return Ok(Some(make_datetime(
                                date.0, date.1, date.2, *hour, *minute,
                            )));
                        }
                    }
                }
            }

            if !advance_date(&mut date) {
                return Ok(None);
            }
            limit = 0;
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut date = (before.year, before.month, before.day);
        let mut limit = before.hour as u16 * 60 + before.minute as u16;

        loop {
            if self.date_matches(date.0, date.1, date.2) {
                for hour in self.sets[1].iter().rev() {
                    for minute in self.sets[0].iter().rev() {
                        let value = *hour as u16 * 60 + *minute as u16;
                        if value < limit {
                            return Ok(Some(make_datetime(
                                date.0, date.1, date.2, *hour, *minute,
                            )));
                        }
                    }
                }
            }

            if !retreat_date(&mut date) {
                return Ok(None);
            }
            limit = 24 * 60;
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;

        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut cursor = start;

        while cursor < end {
            let before_cursor = previous_minute(cursor);
            let next = self
                .next_fire(before_cursor)
                .map_err(|_| WindowError::ReversedWindow)?;
            match next {
                Some(fire) if fire < end => {
                    if fires.len() == cap {
                        return Ok(FireWindow {
                            fires,
                            truncated: true,
                        });
                    }
                    fires.push(fire);
                    cursor = fire;
                }
                _ => break,
            }
        }

        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut output = Vec::new();

        for i in 0..5 {
            let field = FIELDS[i];
            let (lo, hi) = bounds(field);
            let full = self.sets[i].len() == (hi - lo + 1) as usize
                && (lo..=hi).all(|n| self.sets[i].contains(&n));

            if (i == 2 && self.unrestricted_dom) || (i == 4 && self.unrestricted_dow) {
                output.push("*".to_string());
            } else if full && matches!(field, Field::Minute | Field::Hour | Field::Month) {
                output.push("*".to_string());
            } else if full && matches!(field, Field::DayOfMonth | Field::DayOfWeek) {
                output.push("*/1".to_string());
            } else {
                output.push(
                    self.sets[i]
                        .iter()
                        .map(u8::to_string)
                        .collect::<Vec<_>>()
                        .join(","),
                );
            }
        }

        output.join(" ")
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        self.sets[0].contains(&at.minute)
            && self.sets[1].contains(&at.hour)
            && self.sets[3].contains(&at.month)
            && self.date_matches(at.year, at.month, at.day)
    }

    fn date_matches(&self, year: u16, month: u8, day: u8) -> bool {
        let dom = self.sets[2].contains(&day);
        let dow = self.sets[4].contains(&weekday(year, month, day));

        match (self.unrestricted_dom, self.unrestricted_dow) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) if day == days_in_month(year, month) => dom && dow,
            (false, false) => dom || dow,
        }
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn make_datetime(year: u16, month: u8, day: u8, hour: u8, minute: u8) -> CivilDateTime {
    CivilDateTime {
        year,
        month,
        day,
        hour,
        minute,
    }
}

fn advance_date(date: &mut (u16, u8, u8)) -> bool {
    if date.2 < days_in_month(date.0, date.1) {
        date.2 += 1;
    } else if date.1 < 12 {
        date.1 += 1;
        date.2 = 1;
    } else if date.0 < 2399 {
        date.0 += 1;
        date.1 = 1;
        date.2 = 1;
    } else {
        return false;
    }
    true
}

fn retreat_date(date: &mut (u16, u8, u8)) -> bool {
    if date.2 > 1 {
        date.2 -= 1;
    } else if date.1 > 1 {
        date.1 -= 1;
        date.2 = days_in_month(date.0, date.1);
    } else if date.0 > 2000 {
        date.0 -= 1;
        date.1 = 12;
        date.2 = 31;
    } else {
        return false;
    }
    true
}

fn previous_minute(at: CivilDateTime) -> CivilDateTime {
    if at.minute > 0 {
        make_datetime(at.year, at.month, at.day, at.hour, at.minute - 1)
    } else if at.hour > 0 {
        make_datetime(at.year, at.month, at.day, at.hour - 1, 59)
    } else {
        let mut date = (at.year, at.month, at.day);
        if retreat_date(&mut date) {
            make_datetime(date.0, date.1, date.2, 23, 59)
        } else {
            at
        }
    }
}

fn main() {}
