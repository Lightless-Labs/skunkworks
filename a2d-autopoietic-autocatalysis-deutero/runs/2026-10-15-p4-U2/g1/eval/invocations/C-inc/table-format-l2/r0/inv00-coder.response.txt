#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    TooWide { column: usize, width: usize },
    AlignmentCount { expected: usize, actual: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    if let Some(row) = rows.iter().position(|row| row.is_empty()) {
        return Err(TableError::EmptyRow { row });
    }

    let columns = rows[0].len();
    if let Some((row, actual)) = rows
        .iter()
        .enumerate()
        .find_map(|(row, cells)| (cells.len() != columns).then_some((row, cells.len())))
    {
        return Err(TableError::Ragged {
            row,
            expected: columns,
            actual,
        });
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }

    let mut trimmed = Vec::with_capacity(rows.len());
    for (row_index, row) in rows.iter().enumerate() {
        let mut trimmed_row = Vec::with_capacity(columns);
        for (column_index, cell) in row.iter().enumerate() {
            if let Some((byte, _)) = cell
                .char_indices()
                .find(|(_, ch)| !(' '..='~').contains(ch))
            {
                return Err(TableError::InvalidChar {
                    row: row_index,
                    column: column_index,
                    byte,
                });
            }
            trimmed_row.push(cell.trim_end_matches(' ').to_owned());
        }
        trimmed.push(trimmed_row);
    }

    let mut widths = vec![1usize; columns];
    for row in &trimmed {
        for (column, cell) in row.iter().enumerate() {
            widths[column] = widths[column].max(cell.len());
        }
    }

    if let Some((column, &width)) = widths.iter().enumerate().find(|(_, width)| **width > 32) {
        return Err(TableError::TooWide { column, width });
    }

    let alignment = |column: usize| {
        if options.alignments.is_empty() {
            &Alignment::Left
        } else {
            &options.alignments[column]
        }
    };

    let mut output = String::new();
    let append_border = |output: &mut String| {
        output.push('+');
        for &width in &widths {
            output.push_str(&"-".repeat(width + 2));
            output.push('+');
        }
        output.push('\n');
    };

    append_border(&mut output);
    for (row_index, row) in trimmed.iter().enumerate() {
        output.push('|');
        for (column, cell) in row.iter().enumerate() {
            let extra = widths[column] - cell.len();
            let (left, right) = match alignment(column) {
                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),
            };
            output.push(' ');
            output.push_str(&" ".repeat(left));
            output.push_str(cell);
            output.push_str(&" ".repeat(right));
            output.push(' ');
            output.push('|');
        }
        output.push('\n');

        if options.header && row_index == 0 && rows.len() > 1 {
            append_border(&mut output);
        }
    }
    append_border(&mut output);

    Ok(output)
}

fn main() {}
