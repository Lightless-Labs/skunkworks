#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing_slash: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

struct ParsedPath {
    segments: Vec<String>,
    trailing_slash: bool,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_static_segment(bytes: &[u8], start: usize, pattern: bool) -> Result<String, Error> {
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            let bad = || {
                if pattern {
                    Error::InvalidPattern { at: start + i }
                } else {
                    Error::InvalidPath { at: start + i }
                }
            };
            if i + 2 >= bytes.len() {
                return Err(bad());
            }
            let hi = hex_value(bytes[i + 1]).ok_or_else(bad)?;
            let lo = hex_value(bytes[i + 2]).ok_or_else(bad)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(bad());
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: start + i }
            } else {
                Error::InvalidPath { at: start + i }
            });
        }
    }
    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn parse_head(input: &str, pattern: bool) -> Result<ParsedPath, Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing_slash: false,
        });
    }

    let trailing_slash = bytes.last() == Some(&b'/');
    let end = if trailing_slash {
        bytes.len() - 1
    } else {
        bytes.len()
    };

    let mut ranges = Vec::new();
    let mut segment_start = 1;
    let mut i = 1;
    while i <= end {
        if i == end || bytes[i] == b'/' {
            if i == segment_start {
                return Err(invalid(segment_start));
            }
            ranges.push((segment_start, i));
            segment_start = i + 1;
        }
        i += 1;
    }

    let mut segments = Vec::with_capacity(ranges.len());
    let mut names = Vec::<String>::new();

    for (index, (start, finish)) in ranges.iter().copied().enumerate() {
        let raw = &bytes[start..finish];
        if pattern && (raw[0] == b':' || raw[0] == b'*') {
            let marker = raw[0];
            let name_bytes = &raw[1..];
            let valid_name = !name_bytes.is_empty()
                && name_bytes[0].is_ascii_lowercase()
                && name_bytes[1..]
                    .iter()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_');
            if !valid_name {
                return Err(invalid(start));
            }
            let name = String::from_utf8(name_bytes.to_vec()).expect("capture names are ASCII");
            if names.iter().any(|n| n == &name) {
                return Err(invalid(start));
            }
            names.push(name.clone());

            if marker == b'*' && index + 1 != ranges.len() {
                return Err(invalid(start));
            }
            segments.push(if marker == b':' {
                Segment::Param(name)
            } else {
                Segment::Wildcard(name)
            });
        } else {
            let value = decode_static_segment(raw, start, pattern)?;
            segments.push(Segment::Static(value));
        }
    }

    Ok(ParsedPath {
        segments: segments
            .into_iter()
            .map(|segment| match segment {
                Segment::Static(value) => value,
                _ => unreachable!(),
            })
            .collect(),
        trailing_slash,
    })
}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool, Vec<u8>, bool), Error> {
    let invalid = |at| Error::InvalidPattern { at };
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false, vec![3], false));
    }

    let trailing_slash = bytes.last() == Some(&b'/');
    let end = if trailing_slash {
        bytes.len() - 1
    } else {
        bytes.len()
    };
    let mut ranges = Vec::new();
    let mut segment_start = 1;
    let mut i = 1;
    while i <= end {
        if i == end || bytes[i] == b'/' {
            if i == segment_start {
                return Err(invalid(segment_start));
            }
            ranges.push((segment_start, i));
            segment_start = i + 1;
        }
        i += 1;
    }

    let mut segments = Vec::with_capacity(ranges.len());
    let mut rank = Vec::with_capacity(ranges.len() + 1);
    let mut names = Vec::<String>::new();
    let mut has_wildcard = false;

    for (index, (start, finish)) in ranges.iter().copied().enumerate() {
        let raw = &bytes[start..finish];
        if raw[0] == b':' || raw[0] == b'*' {
            let marker = raw[0];
            let name_bytes = &raw[1..];
            let valid_name = !name_bytes.is_empty()
                && name_bytes[0].is_ascii_lowercase()
                && name_bytes[1..]
                    .iter()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_');
            if !valid_name {
                return Err(invalid(start));
            }
            let name = String::from_utf8(name_bytes.to_vec()).expect("capture names are ASCII");
            if names.iter().any(|n| n == &name) {
                return Err(invalid(start));
            }
            names.push(name.clone());

            if marker == b'*' {
                if index + 1 != ranges.len() {
                    return Err(invalid(start));
                }
                has_wildcard = true;
                rank.push(0);
                segments.push(Segment::Wildcard(name));
            } else {
                rank.push(1);
                segments.push(Segment::Param(name));
            }
        } else {
            let value = decode_static_segment(raw, start, true)?;
            rank.push(2);
            segments.push(Segment::Static(value));
        }
    }
    if !has_wildcard {
        rank.push(3);
    }
    Ok((segments, trailing_slash, rank, has_wildcard))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing_slash, rank, has_wildcard) = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_owned(),
            segments,
            trailing_slash,
            rank,
            has_wildcard,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .bytes()
            .position(|b| b == b'?' || b == b'#')
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_head(head, false)?;

        let mut decoded_segments = Vec::with_capacity(parsed.segments.len());
        if head == "/" {
            // Root is represented by no segments.
        } else {
            let bytes = head.as_bytes();
            let trailing = parsed.trailing_slash;
            let end = if trailing { bytes.len() - 1 } else { bytes.len() };
            let mut start = 1;
            let mut i = 1;
            while i <= end {
                if i == end || bytes[i] == b'/' {
                    decoded_segments.push(decode_static_segment(&bytes[start..i], start, false)?);
                    start = i + 1;
                }
                i += 1;
            }
        }

        let mut candidates: Vec<(usize, Vec<(String, String)>)> = Vec::new();
        for (id, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict
                && route.trailing_slash != parsed.trailing_slash
            {
                continue;
            }

            let mut params = Vec::new();
            let mut path_index = 0;
            let mut matched = true;
            for segment in &route.segments {
                match segment {
                    Segment::Static(expected) => {
                        if decoded_segments.get(path_index) != Some(expected) {
                            matched = false;
                            break;
                        }
                        path_index += 1;
                    }
                    Segment::Param(name) => {
                        let Some(value) = decoded_segments.get(path_index) else {
                            matched = false;
                            break;
                        };
                        params.push((name.clone(), value.clone()));
                        path_index += 1;
                    }
                    Segment::Wildcard(name) => {
                        let value = decoded_segments[path_index..].join("/");
                        params.push((name.clone(), value));
                        path_index = decoded_segments.len();
                    }
                }
            }
            if matched && path_index == decoded_segments.len() {
                candidates.push((id, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && !candidates
                .iter()
                .any(|(id, _)| self.routes[*id].method == "HEAD")
        {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<(usize, Vec<(String, String)>)> = candidates
            .iter()
            .filter(|(id, _)| self.routes[*id].method == eligible_method)
            .cloned()
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(id, _)| self.routes[*id].method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") && !allowed.iter().any(|m| m == "HEAD") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|(a, _), (b, _)| {
            self.routes[*b]
                .rank
                .cmp(&self.routes[*a].rank)
                .then_with(|| a.cmp(b))
        });
        let (route_id, params) = eligible.remove(0);
        let route = &self.routes[route_id];

        let redirect = if self.policy == SlashPolicy::Redirect
            && head != "/"
            && route.trailing_slash != parsed.trailing_slash
        {
            if route.trailing_slash {
                Some(format!("{head}/"))
            } else {
                Some(head[..head.len() - 1].to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: route_id,
            params,
            redirect,
        })
    }
}

fn main() {}
