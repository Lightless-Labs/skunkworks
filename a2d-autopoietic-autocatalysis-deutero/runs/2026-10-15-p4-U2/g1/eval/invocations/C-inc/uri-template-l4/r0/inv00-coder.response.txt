#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semi,
    Question,
    Amp,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug)]
struct Specifier {
    name: String,
    start: usize,
    modifier: Modifier,
}

#[derive(Debug)]
struct Expression {
    operator: Operator,
    specifiers: Vec<Specifier>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn parse_expression(content: &str, base: usize, close: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let (operator, offset) = if let Some(&first) = bytes.first() {
        let op = match first {
            b'+' => Some(Operator::Plus),
            b'#' => Some(Operator::Hash),
            b'.' => Some(Operator::Dot),
            b'/' => Some(Operator::Slash),
            b';' => Some(Operator::Semi),
            b'?' => Some(Operator::Question),
            b'&' => Some(Operator::Amp),
            _ => None,
        };
        if let Some(op) = op {
            (op, 1)
        } else {
            if matches!(first, b'!' | b'$' | b'\'' | b'(' | b')' | b'*' | b',' | b':' | b'=' | b'[' | b']' | b'@') {
                return Err(Error::InvalidName { pos: base });
            }
            (Operator::Simple, 0)
        }
    } else {
        return Err(Error::EmptyExpression { pos: close });
    };

    if offset == bytes.len() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let rest = &content[offset..];
    let rest_base = base + offset;
    let mut specifiers = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut cursor = 0;

    loop {
        let tail = &rest[cursor..];
        let part_len = tail.find(',').unwrap_or(tail.len());
        let part = &tail[..part_len];
        let start = rest_base + cursor;

        if part.is_empty() {
            return Err(Error::EmptyVariable { pos: start });
        }

        let part_bytes = part.as_bytes();
        let modifier_at = part_bytes.iter().position(|&b| b == b'*' || b == b':');
        let name_len = modifier_at.unwrap_or(part_bytes.len());
        let name = &part[..name_len];

        if name.is_empty() {
            return Err(Error::InvalidName { pos: start });
        }

        let mut name_end = name.len();
        for (i, &b) in name.as_bytes().iter().enumerate() {
            let valid = if i == 0 {
                is_name_start(b)
            } else {
                is_name_continue(b)
            };
            if !valid {
                name_end = i;
                break;
            }
        }
        if name_end != name.len() {
            return Err(Error::InvalidName {
                pos: start + name_end,
            });
        }

        let modifier = if let Some(mpos) = modifier_at {
            let suffix = &part[mpos..];
            let parsed = if suffix == "*" {
                Some(Modifier::Explode)
            } else if let Some(digits) = suffix.strip_prefix(':') {
                if (1..=4).contains(&digits.len())
                    && digits.as_bytes()[0].is_ascii_digit()
                    && digits.as_bytes()[0] != b'0'
                    && digits.bytes().all(|b| b.is_ascii_digit())
                {
                    digits.parse::<usize>().ok().filter(|&n| (1..=9999).contains(&n)).map(Modifier::Prefix)
                } else {
                    None
                }
            } else {
                None
            };
            parsed.ok_or(Error::InvalidModifier {
                pos: start + mpos,
            })?
        } else {
            Modifier::None
        };

        if !seen.insert(name.to_owned()) {
            return Err(Error::DuplicateVariable { pos: start });
        }

        specifiers.push(Specifier {
            name: name.to_owned(),
            start,
            modifier,
        });

        cursor += part_len;
        if cursor == rest.len() {
            break;
        }
        cursor += 1;
        if cursor == rest.len() {
            return Err(Error::EmptyVariable {
                pos: rest_base + cursor,
            });
        }
    }

    Ok(Expression {
        operator,
        specifiers,
    })
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() && bytes[i + 1].is_ascii_hexdigit() && bytes[i + 2].is_ascii_hexdigit() {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let is_reserved = reserved
            && matches!(
                b,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
                    | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || is_reserved {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }

    out
}

fn prefix_chars(value: &str, n: usize) -> String {
    value.chars().take(n).collect()
}

fn field_name_value(name: &str, value: String, operator: Operator) -> String {
    match operator {
        Operator::Semi if value.is_empty() => name.to_owned(),
        Operator::Semi | Operator::Question | Operator::Amp => format!("{name}={value}"),
        _ => value,
    }
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let reserved = matches!(expression.operator, Operator::Plus | Operator::Hash);
    let mut fields = Vec::new();

    for spec in &expression.specifiers {
        let value = match selected_value(&spec.name, vars) {
            Some(value) => value,
            None => continue,
        };

        if matches!(spec.modifier, Modifier::Prefix(_)) && !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch { pos: spec.start });
        }

        match value {
            Value::Scalar(raw) => {
                let selected = match spec.modifier {
                    Modifier::Prefix(n) => prefix_chars(raw, n),
                    _ => raw.clone(),
                };
                let encoded = encode(&selected, reserved);
                fields.push(field_name_value(&spec.name, encoded, expression.operator));
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if spec.modifier == Modifier::Explode {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(field_name_value(&spec.name, encoded, expression.operator));
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(field_name_value(&spec.name, joined, expression.operator));
                }
            }
            Value::Map(entries) => {
                let mut keys = std::collections::HashSet::new();
                for (key, _) in entries {
                    if !keys.insert(key) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                }
                if entries.is_empty() {
                    continue;
                }

                if spec.modifier == Modifier::Explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        let field = if expression.operator == Operator::Semi && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(field_name_value(&spec.name, joined, expression.operator));
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Plus => ("", ","),
        Operator::Hash => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semi => (";", ";"),
        Operator::Question => ("?", "&"),
        Operator::Amp => ("&", "&"),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let mut expressions = Vec::new();
    let mut literal_ranges = Vec::new();
    let mut cursor = 0;

    while cursor < template.len() {
        let tail = &template[cursor..];
        let next_open = tail.find('{');
        let next_close = tail.find('}');

        match (next_open, next_close) {
            (None, None) => {
                literal_ranges.push((cursor, template.len()));
                break;
            }
            (None, Some(close)) => {
                return Err(Error::UnexpectedClose { pos: cursor + close });
            }
            (Some(open), Some(close)) if close < open => {
                return Err(Error::UnexpectedClose { pos: cursor + close });
            }
            (Some(open), _) => {
                let open_pos = cursor + open;
                if open_pos > cursor {
                    literal_ranges.push((cursor, open_pos));
                }
                let content_start = open_pos + 1;
                let close_rel = template[content_start..].find('}');
                let close_pos = match close_rel {
                    Some(rel) => content_start + rel,
                    None => return Err(Error::Unclosed { pos: open_pos }),
                };
                if let Some(inner) = template[content_start..close_pos].find('{') {
                    return Err(Error::NestedOpen {
                        pos: content_start + inner,
                    });
                }
                let expression = parse_expression(
                    &template[content_start..close_pos],
                    content_start,
                    close_pos,
                )?;
                expressions.push((open_pos, close_pos + 1, expression));
                cursor = close_pos + 1;
            }
        }
    }

    let mut out = String::new();
    let mut pos = 0;
    for (start, end, expression) in expressions {
        out.push_str(&template[pos..start]);
        out.push_str(&expand_expression(&expression, vars)?);
        pos = end;
    }
    out.push_str(&template[pos..]);
    Ok(out)
}

fn main() {}
