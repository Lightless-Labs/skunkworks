#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|previous| previous.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut completed: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut ready: std::collections::VecDeque<usize> = std::collections::VecDeque::new();
    let mut runs: Vec<Run> = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut time = 0u64;
    let mut event_pos = 0usize;
    let mut last_id: Option<u32> = None;

    fn admit(
        jobs: &[Job],
        events: &[usize],
        event_pos: &mut usize,
        through: u64,
        remaining: &[u32],
        completed: &mut [Option<u64>],
        ready: &mut std::collections::VecDeque<usize>,
    ) {
        while *event_pos < events.len() {
            let index = events[*event_pos];
            let job = &jobs[index];
            if u64::from(job.arrival) > through {
                break;
            }
            *event_pos += 1;
            if job.ticks == 0 {
                completed[index] = Some(u64::from(job.arrival));
            } else if remaining[index] > 0 {
                ready.push_back(index);
            }
        }
    }

    while completed.iter().any(Option::is_none) {
        admit(
            jobs,
            &events,
            &mut event_pos,
            time,
            &remaining,
            &mut completed,
            &mut ready,
        );

        if ready.is_empty() {
            if event_pos >= events.len() {
                break;
            }
            let next_time = u64::from(jobs[events[event_pos]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_id = None;
            }
            continue;
        }

        let index = ready.pop_front().expect("ready queue is nonempty");
        let id = jobs[index].id;

        if last_id.is_some_and(|previous| previous != id) {
            time += u64::from(switch_cost);
        }

        admit(
            jobs,
            &events,
            &mut event_pos,
            time,
            &remaining,
            &mut completed,
            &mut ready,
        );

        let start = time;
        let consumed = remaining[index].min(quantum);
        time += u64::from(consumed);
        remaining[index] -= consumed;
        let finished = remaining[index] == 0;

        runs.push(Run {
            id,
            start,
            end: time,
            finished,
        });

        admit(
            jobs,
            &events,
            &mut event_pos,
            time,
            &remaining,
            &mut completed,
            &mut ready,
        );

        if finished {
            completed[index] = Some(time);
        } else {
            ready.push_back(index);
        }
        last_id = Some(id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completed[index].unwrap_or(0)))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}

fn main() {}
