#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Status {
    New,
    Open,
    Closed,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Account {
    pub status: Status,
    pub balance: i64,
    pub limit: i64,
    pub version: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { limit: i64 },
    Deposit { amount: i64 },
    Withdraw { amount: i64, floor: i64 },
    Close { expected_version: u64 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    VersionOverflow,
    InvalidLimit,
    AlreadyOpened,
    NotOpened,
    Closed,
    InvalidAmount,
    LimitExceeded,
    InvalidFloor,
    InsufficientFunds,
    FloorViolation,
    CloseVersion { expected: u64, actual: u64 },
    NonZeroBalance,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ReplayError {
    pub index: usize,
    pub kind: ApplyError,
}

pub fn replay(events: &[Event]) -> Result<Account, ReplayError> {
    let mut account = Account {
        status: Status::New,
        balance: 0,
        limit: 0,
        version: 0,
    };

    for (index, event) in events.iter().enumerate() {
        let result = apply_event(&mut account, event);
        if let Err(kind) = result {
            return Err(ReplayError { index, kind });
        }
    }

    Ok(account)
}

fn lifecycle_error(account: &Account) -> Option<ApplyError> {
    match account.status {
        Status::New => Some(ApplyError::NotOpened),
        Status::Closed => Some(ApplyError::Closed),
        Status::Open => None,
    }
}

fn increment_version(account: &mut Account) -> Result<(), ApplyError> {
    account.version = account
        .version
        .checked_add(1)
        .ok_or(ApplyError::VersionOverflow)?;
    Ok(())
}

fn apply_event(account: &mut Account, event: &Event) -> Result<(), ApplyError> {
    match event {
        Event::Open { limit } => {
            if !(1..=1_000_000_000).contains(limit) {
                return Err(ApplyError::InvalidLimit);
            }
            if account.status != Status::New {
                return Err(ApplyError::AlreadyOpened);
            }

            increment_version(account)?;
            account.status = Status::Open;
            account.limit = *limit;
            Ok(())
        }
        Event::Deposit { amount } => {
            if let Some(error) = lifecycle_error(account) {
                return Err(error);
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }

            let capacity = account.limit - account.balance;
            if *amount > capacity {
                return Err(ApplyError::LimitExceeded);
            }

            increment_version(account)?;
            account.balance += *amount;
            Ok(())
        }
        Event::Withdraw { amount, floor } => {
            if let Some(error) = lifecycle_error(account) {
                return Err(error);
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *floor < 0 || *floor > account.limit {
                return Err(ApplyError::InvalidFloor);
            }
            if *amount > account.balance {
                return Err(ApplyError::InsufficientFunds);
            }
            if account.balance - *amount < *floor {
                return Err(ApplyError::FloorViolation);
            }

            increment_version(account)?;
            account.balance -= *amount;
            Ok(())
        }
        Event::Close { expected_version } => {
            if let Some(error) = lifecycle_error(account) {
                return Err(error);
            }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion {
                    expected: *expected_version,
                    actual: account.version,
                });
            }
            if account.balance != 0 {
                return Err(ApplyError::NonZeroBalance);
            }

            increment_version(account)?;
            account.status = Status::Closed;
            Ok(())
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Expected {
    Any,
    Missing,
    Exact(u64),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StoreError {
    InvalidId { byte: usize },
    Conflict {
        expected: Expected,
        actual: Option<u64>,
    },
    Rejected(ReplayError),
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Stream {
    events: Vec<Event>,
    account: Account,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Store {
    streams: std::collections::BTreeMap<String, Stream>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            streams: std::collections::BTreeMap::new(),
        }
    }

    pub fn append(
        &mut self,
        id: &str,
        expected: Expected,
        events: &[Event],
    ) -> Result<Account, StoreError> {
        validate_id(id)?;

        let existing = self.streams.get(id);
        let actual = existing.map(|stream| stream.account.version);
        let matches = match &expected {
            Expected::Any => true,
            Expected::Missing => actual.is_none(),
            Expected::Exact(version) => actual == Some(*version),
        };

        if !matches {
            return Err(StoreError::Conflict { expected, actual });
        }

        let initial = Account {
            status: Status::New,
            balance: 0,
            limit: 0,
            version: 0,
        };
        let mut next_account = existing
            .map(|stream| stream.account.clone())
            .unwrap_or(initial);

        for (index, event) in events.iter().enumerate() {
            if let Err(kind) = apply_event(&mut next_account, event) {
                return Err(StoreError::Rejected(ReplayError { index, kind }));
            }
        }

        if events.is_empty() {
            return Ok(next_account);
        }

        let stream = self
            .streams
            .entry(id.to_owned())
            .or_insert_with(|| Stream {
                events: Vec::new(),
                account: Account {
                    status: Status::New,
                    balance: 0,
                    limit: 0,
                    version: 0,
                },
            });
        stream.events.extend_from_slice(events);
        stream.account = next_account.clone();

        Ok(next_account)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|stream| stream.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|stream| {
            stream
                .events
                .iter()
                .enumerate()
                .filter_map(|(index, event)| {
                    let version = (index as u64) + 1;
                    (version > after).then(|| event.clone())
                })
                .collect()
        }))
    }
}

fn validate_id(id: &str) -> Result<(), StoreError> {
    let bytes = id.as_bytes();

    if bytes.is_empty() {
        return Err(StoreError::InvalidId { byte: 0 });
    }

    for (index, byte) in bytes.iter().enumerate() {
        let valid = if index == 0 {
            byte.is_ascii_lowercase()
        } else {
            byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-'
        };
        if !valid {
            return Err(StoreError::InvalidId { byte: index });
        }
    }

    if bytes.len() > 24 {
        return Err(StoreError::InvalidId { byte: 24 });
    }

    Ok(())
}

fn main() {}
