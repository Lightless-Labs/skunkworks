#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Little-endian base-10 digits. Zero is represented as [0].
    digits: Vec<u8>,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over all separator and numeric errors.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let mut digit_positions = Vec::new();
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                let previous_is_digit = index > start && bytes[index - 1].is_ascii_digit();
                let next_is_digit =
                    index + 1 < bytes.len() && bytes[index + 1].is_ascii_digit();
                if !previous_is_digit || !next_is_digit {
                    return Err(Error::Separator { index });
                }
            } else {
                digit_positions.push(index);
            }
        }

        if digit_positions.len() > 1 && bytes[digit_positions[0]] == b'0' {
            return Err(Error::LeadingZero {
                index: digit_positions[0],
            });
        }

        if digit_positions.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_positions[max_digits],
            });
        }

        let mut digits = Vec::with_capacity(digit_positions.len());
        for &index in digit_positions.iter().rev() {
            digits.push(bytes[index] - b'0');
        }

        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let natural_len = self.digits.len().max(1);
        let width = natural_len.max(min_digits);

        let mut result = String::new();
        for position in (0..width).rev() {
            if group > 0 && position != width - 1 && (position + 1) % group == 0 {
                result.push('_');
            }

            let digit = if position < self.digits.len() {
                self.digits[position]
            } else {
                0
            };
            result.push(char::from(b'0' + digit));
        }
        result
    }

    pub fn cmp_value(&self, rhs: &Self) -> std::cmp::Ordering {
        self.digits
            .len()
            .cmp(&rhs.digits.len())
            .then_with(|| self.digits.iter().rev().cmp(rhs.digits.iter().rev()))
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let max_len = self.digits.len().max(rhs.digits.len());
        let mut digits = Vec::with_capacity(max_len + 1);
        let mut carry = 0u8;

        for index in 0..max_len {
            let left = self.digits.get(index).copied().unwrap_or(0);
            let right = rhs.digits.get(index).copied().unwrap_or(0);
            let sum = left + right + carry;
            digits.push(sum % 10);
            carry = sum / 10;
        }

        if carry != 0 {
            digits.push(carry);
        }

        Self { digits }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == std::cmp::Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut digits = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i16;

        for index in 0..self.digits.len() {
            let left = i16::from(self.digits[index]) - borrow;
            let right = i16::from(rhs.digits.get(index).copied().unwrap_or(0));

            if left < right {
                digits.push((left + 10 - right) as u8);
                borrow = 1;
            } else {
                digits.push((left - right) as u8);
                borrow = 0;
            }
        }

        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }

        Ok(Self { digits })
    }
}

fn main() {}
