#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_char(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn env_value<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value)
}

fn append(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        if active_glob && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let n = c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append(&mut word, &input[i..i + n], false);
                }
                i += n;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i += n;
                } else if c == '\\' {
                    let esc_pos = i;
                    i += 1;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: esc_pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    let next_len = next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append(&mut word, &input[i..i + next_len], false);
                        }
                        i += next_len;
                    } else {
                        append(&mut word, "\\", false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[i + n..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + n..].starts_with('{') {
                        let dollar = i;
                        let content_start = i + 2;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let d = input[j..].chars().next().unwrap();
                            if d == '\\' {
                                j += 1;
                                if j < input.len() {
                                    j += input[j..].chars().next().unwrap().len_utf8();
                                }
                            } else if d == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j += d.len_utf8();
                            }
                        }
                        let Some(close) = close else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar });
                        };
                        let contents = &input[content_start..close];
                        let (name, default) = if let Some(k) = contents.find(":-") {
                            (&contents[..k], Some(&contents[k + 2..]))
                        } else {
                            (contents, None)
                        };
                        if name.is_empty() || !name.chars().next().is_some_and(is_name_start)
                            || !name.chars().all(is_name_char)
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let value = match default {
                            Some(text) => match env_value(name, env) {
                                Some(v) if !v.is_empty() => v.to_string(),
                                _ => {
                                    let mut decoded = String::new();
                                    let mut k = 0;
                                    while k < text.len() {
                                        let d = text[k..].chars().next().unwrap();
                                        if d == '\\' {
                                            k += 1;
                                            if k < text.len() {
                                                let e = text[k..].chars().next().unwrap();
                                                decoded.push(e);
                                                k += e.len_utf8();
                                            } else {
                                                decoded.push('\\');
                                            }
                                        } else {
                                            decoded.push(d);
                                            k += d.len_utf8();
                                        }
                                    }
                                    decoded
                                }
                            },
                            None => env_value(name, env).unwrap_or("").to_string(),
                        };
                        append(&mut word, &value, false);
                        active = true;
                        i = close + 1;
                    } else if input[i + n..].chars().next().is_some_and(is_name_start) {
                        let mut end = i + n;
                        while end < input.len() {
                            let d = input[end..].chars().next().unwrap();
                            if is_name_char(d) {
                                end += d.len_utf8();
                            } else {
                                break;
                            }
                        }
                        let name = &input[i + n..end];
                        append(&mut word, env_value(name, env).unwrap_or(""), false);
                        active = true;
                        i = end;
                    } else {
                        append(&mut word, "$", false);
                        i += n;
                    }
                } else {
                    append(&mut word, &input[i..i + n], false);
                    i += n;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word { text: String::new(), glob: Vec::new() };
                        active = false;
                    }
                    i += n;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i += n;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i += n;
                } else if c == '\\' {
                    let esc_pos = i;
                    i += 1;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: esc_pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    let next_len = next.len_utf8();
                    if next == '\n' {
                        i += next_len;
                    } else {
                        append(&mut word, &input[i..i + next_len], false);
                        active = true;
                        i += next_len;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[i + n..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + n..].starts_with('{') {
                        let dollar = i;
                        let content_start = i + 2;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let d = input[j..].chars().next().unwrap();
                            if d == '\\' {
                                j += 1;
                                if j < input.len() {
                                    j += input[j..].chars().next().unwrap().len_utf8();
                                }
                            } else if d == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j += d.len_utf8();
                            }
                        }
                        let Some(close) = close else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar });
                        };
                        let contents = &input[content_start..close];
                        let (name, default) = if let Some(k) = contents.find(":-") {
                            (&contents[..k], Some(&contents[k + 2..]))
                        } else {
                            (contents, None)
                        };
                        if name.is_empty() || !name.chars().next().is_some_and(is_name_start)
                            || !name.chars().all(is_name_char)
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let value = match default {
                            Some(text) => match env_value(name, env) {
                                Some(v) if !v.is_empty() => v.to_string(),
                                _ => {
                                    let mut decoded = String::new();
                                    let mut k = 0;
                                    while k < text.len() {
                                        let d = text[k..].chars().next().unwrap();
                                        if d == '\\' {
                                            k += 1;
                                            if k < text.len() {
                                                let e = text[k..].chars().next().unwrap();
                                                decoded.push(e);
                                                k += e.len_utf8();
                                            } else {
                                                decoded.push('\\');
                                            }
                                        } else {
                                            decoded.push(d);
                                            k += d.len_utf8();
                                        }
                                    }
                                    decoded
                                }
                            },
                            None => env_value(name, env).unwrap_or("").to_string(),
                        };
                        append(&mut word, &value, true);
                        active = true;
                        i = close + 1;
                    } else if input[i + n..].chars().next().is_some_and(is_name_start) {
                        let mut end = i + n;
                        while end < input.len() {
                            let d = input[end..].chars().next().unwrap();
                            if is_name_char(d) {
                                end += d.len_utf8();
                            } else {
                                break;
                            }
                        }
                        let name = &input[i + n..end];
                        append(&mut word, env_value(name, env).unwrap_or(""), true);
                        active = true;
                        i = end;
                    } else {
                        append(&mut word, "$", false);
                        active = true;
                        i += n;
                    }
                } else {
                    append(&mut word, &input[i..i + n], true);
                    active = true;
                    i += n;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn main() {}
