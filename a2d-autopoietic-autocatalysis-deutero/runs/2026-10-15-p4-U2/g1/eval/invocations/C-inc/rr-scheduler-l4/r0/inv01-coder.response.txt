#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Process {
    id: u32,
    bursts: Vec<u32>,
    burst_pos: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    level: usize,
}

fn add_idle(idle: &mut Vec<(u64, u64)>, start: u64, end: u64) {
    if start >= end {
        return;
    }
    if let Some(last) = idle.last_mut() {
        if last.1 == start {
            last.1 = end;
            return;
        }
    }
    idle.push((start, end));
}

fn admit_through(
    events: &mut Vec<Event>,
    now: u64,
    processes: &mut [Process],
    queues: &mut [std::collections::VecDeque<usize>],
) {
    events.sort_by_key(|event| (event.time, event.kind, event.index));
    let split = events.partition_point(|event| event.time <= now);
    let admitted: Vec<Event> = events.drain(..split).collect();

    for event in admitted {
        processes[event.index].level = event.level;
        queues[event.level].push_back(event.index);
    }
}

fn rounded_mean_x100(total: u64, count: u64) -> u64 {
    if count == 0 {
        0
    } else {
        (total * 100 + count / 2) / count
    }
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut processes: Vec<Process> = jobs
        .iter()
        .map(|job| Process {
            id: job.id,
            bursts: job.bursts.clone(),
            burst_pos: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: job.arrival as u64,
            kind: 0,
            index,
            level: job.level,
        })
        .collect();

    let mut queues: Vec<std::collections::VecDeque<usize>> = (0..quanta.len())
        .map(|_| std::collections::VecDeque::new())
        .collect();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut now = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut completed_count = 0usize;

    while completed_count < jobs.len() {
        admit_through(&mut events, now, &mut processes, &mut queues);

        let Some(level) = queues.iter().position(|queue| !queue.is_empty()) else {
            if events.is_empty() {
                break;
            }
            let next_time = events.iter().map(|event| event.time).min().unwrap();
            if next_time > now {
                add_idle(&mut idle, now, next_time);
                now = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let process_index = queues[level].pop_front().unwrap();
        let process_id = processes[process_index].id;

        if last_dispatch_id.is_some_and(|previous| previous != process_id) {
            now += switch_cost as u64;
        }

        admit_through(&mut events, now, &mut processes, &mut queues);

        let quantum = quanta[level];
        let consumed = processes[process_index].remaining.min(quantum);
        let start = now;
        now += consumed as u64;
        let end = now;
        processes[process_index].remaining -= consumed;

        let cpu_burst_finished = processes[process_index].remaining == 0;
        let final_burst = cpu_burst_finished
            && processes[process_index].burst_pos + 1 == processes[process_index].bursts.len();

        runs.push(Run {
            id: process_id,
            start,
            end,
            finished: final_burst,
            level,
        });

        if cpu_burst_finished && !final_burst {
            let io_position = processes[process_index].burst_pos + 1;
            let wake_time = end + processes[process_index].bursts[io_position] as u64;
            let promoted_level = level.saturating_sub(1);
            processes[process_index].burst_pos += 2;
            processes[process_index].remaining =
                processes[process_index].bursts[processes[process_index].burst_pos];
            events.push(Event {
                time: wake_time,
                kind: 1,
                index: process_index,
                level: promoted_level,
            });
        }

        admit_through(&mut events, end, &mut processes, &mut queues);

        if final_burst {
            processes[process_index].completion = Some(end);
            completed_count += 1;
        } else if !cpu_burst_finished {
            let demoted_level = (level + 1).min(quanta.len() - 1);
            processes[process_index].level = demoted_level;
            queues[demoted_level].push_back(process_index);
        }

        last_dispatch_id = Some(process_id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = processes[index].completion.unwrap_or(now);
        let turnaround = completion - job.arrival as u64;
        let total_cpu: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&burst| burst as u64)
            .sum();
        let total_io: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&burst| burst as u64)
            .sum();
        let waiting = turnaround - total_cpu - total_io;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let end = stats.iter().map(|stat| stat.completion).max().unwrap_or(0);

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: rounded_mean_x100(total_waiting, count),
        mean_turnaround_x100: rounded_mean_x100(total_turnaround, count),
        end,
    })
}

fn main() {}
