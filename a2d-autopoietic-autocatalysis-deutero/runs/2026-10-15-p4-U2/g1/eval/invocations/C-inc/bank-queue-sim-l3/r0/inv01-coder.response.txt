#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    remaining: u64,
    started: bool,
    pause_time: Option<u64>,
}

struct Running {
    job: Job,
    work_at: u64,
    finish: u64,
}

fn waiter_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    use std::cmp::Ordering;

    a.customer.priority.cmp(&b.customer.priority).then_with(|| {
        match (a.pause_time, b.pause_time) {
            (Some(pa), Some(pb)) => pa
                .cmp(&pb)
                .then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        }
    })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;

    loop {
        if next_arrival == arrivals.len() && running.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_tick = arrivals.get(next_arrival).map(|c| c.arrival);
        if let Some(active) = running.as_ref() {
            next_tick = Some(next_tick.map_or(active.finish, |t| t.min(active.finish)));
            if active.work_at > 0 {
                next_tick = Some(next_tick.map_or(active.work_at, |t| t.min(active.work_at)));
            }
        }
        let Some(time) = next_tick else {
            break;
        };

        if running.as_ref().is_some_and(|active| active.finish == time) {
            let active = running.take().unwrap();
            events.push(Event {
                time,
                customer: active.job.customer.id,
                kind: Kind::Finish,
            });
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Job {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: None,
            });
            next_arrival += 1;
        }

        if let Some(active) = running.as_ref() {
            if time >= active.work_at {
                let best = waiting.iter().min_by(|a, b| waiter_cmp(a, b));
                if let Some(best) = best {
                    if best.customer.priority < active.job.customer.priority {
                        let by = best.customer.id;
                        let mut active = running.take().unwrap();
                        let remaining = active.finish - time;
                        active.job.remaining = remaining;
                        active.job.pause_time = Some(time);
                        events.push(Event {
                            time,
                            customer: active.job.customer.id,
                            kind: Kind::Preempt { remaining, by },
                        });
                        waiting.push(active.job);
                    }
                }
            }
        }

        if running.is_none() {
            while !waiting.is_empty() {
                let best_index = (0..waiting.len())
                    .min_by(|&a, &b| waiter_cmp(&waiting[a], &waiting[b]))
                    .unwrap();
                let mut job = waiting.remove(best_index);
                let switching = last_dispatched.is_some_and(|id| id != job.customer.id);
                let overhead = if switching { switch_cost } else { 0 };

                let work_at = time.checked_add(overhead).ok_or(Error::TimeOverflow {
                    customer: job.customer.id,
                    time,
                })?;
                let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                    customer: job.customer.id,
                    time,
                })?;

                let kind = if job.started {
                    Kind::Resume {
                        remaining: job.remaining,
                        work_at,
                        finish,
                    }
                } else {
                    Kind::Start {
                        remaining: job.remaining,
                        work_at,
                        finish,
                    }
                };
                job.started = true;
                job.pause_time = None;
                last_dispatched = Some(job.customer.id);
                events.push(Event {
                    time,
                    customer: job.customer.id,
                    kind,
                });

                if finish == time {
                    events.push(Event {
                        time,
                        customer: job.customer.id,
                        kind: Kind::Finish,
                    });
                } else {
                    running = Some(Running {
                        job,
                        work_at,
                        finish,
                    });
                    break;
                }
            }
        }
    }

    Ok(events)
}

fn main() {}
