#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    displayed: usize,
    consumed: usize,
    marker_end: bool,
    piece_count: usize,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
}

#[derive(Clone, Copy, Hash, PartialEq, Eq)]
struct State {
    word: usize,
    offset: usize,
    first: bool,
}

fn render_prejustified(line: &Line) -> String {
    let mut s = " ".repeat(line.indent);
    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            s.push(' ');
        }
        s.push_str(piece);
    }
    s
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }

    let av: Vec<usize> = a.lines.iter().map(|line| line.consumed).collect();
    let bv: Vec<usize> = b.lines.iter().map(|line| line.consumed).collect();
    if av != bv {
        return av > bv;
    }

    let astr: Vec<String> = a.lines.iter().map(render_prejustified).collect();
    let bstr: Vec<String> = b.lines.iter().map(render_prejustified).collect();
    astr < bstr
}

fn solve(
    state: State,
    words: &[Word],
    options: &WrapOptions,
    memo: &mut std::collections::HashMap<State, Solution>,
) -> Solution {
    let mut word_index = state.word;
    let mut offset = state.offset;
    while word_index < words.len() && offset == words[word_index].chars.len() {
        word_index += 1;
        offset = 0;
    }
    if word_index >= words.len() {
        return Solution {
            cost: 0,
            marker_lines: 0,
            lines: Vec::new(),
        };
    }

    let normalized = State {
        word: word_index,
        offset,
        first: state.first,
    };
    if let Some(solution) = memo.get(&normalized) {
        return solution.clone();
    }

    let indent = if state.first {
        options.first_indent
    } else {
        options.rest_indent
    };
    let capacity = options.width - indent;
    let mut candidates: Vec<(Line, State, u64, usize)> = Vec::new();
    let mut pieces: Vec<String> = Vec::new();
    let mut used = 0usize;
    let mut current_word = word_index;
    let mut current_offset = offset;

    loop {
        if current_word >= words.len() {
            break;
        }

        let word = &words[current_word];
        let remaining = word.chars.len() - current_offset;
        let separator = usize::from(!pieces.is_empty());

        if used + separator >= capacity {
            break;
        }
        let available = capacity - used - separator;

        if remaining <= available {
            let text: String = word.chars[current_offset..].iter().collect();
            let mut next_pieces = pieces.clone();
            next_pieces.push(text.clone());
            let displayed = used + separator + remaining;
            let line = Line {
                indent,
                pieces: next_pieces,
                displayed,
                consumed: remaining,
                marker_end: false,
                piece_count: pieces.len() + 1,
            };
            candidates.push((
                line,
                State {
                    word: current_word + 1,
                    offset: 0,
                    first: false,
                },
                0,
                0,
            ));

            used += separator + remaining;
            pieces.push(text);
            current_word += 1;
            current_offset = 0;
            if current_word >= words.len() {
                break;
            }
            continue;
        }

        for &marker in &word.markers {
            if marker > current_offset
                && marker < word.chars.len()
                && marker - current_offset + 1 <= available
            {
                let mut piece: String = word.chars[current_offset..marker].iter().collect();
                piece.push('-');
                let mut next_pieces = pieces.clone();
                next_pieces.push(piece);
                let line = Line {
                    indent,
                    pieces: next_pieces,
                    displayed: used + separator + marker - current_offset + 1,
                    consumed: marker - current_offset,
                    marker_end: true,
                    piece_count: pieces.len() + 1,
                };
                candidates.push((
                    line,
                    State {
                        word: current_word,
                        offset: marker,
                        first: false,
                    },
                    options.hyphen_cost as u64,
                    1,
                ));
            }
        }

        if pieces.is_empty() && remaining > capacity {
            let marker_fits = word.markers.iter().any(|&marker| {
                marker > current_offset
                    && marker < word.chars.len()
                    && marker - current_offset + 1 <= capacity
            });
            if !marker_fits {
                let text: String = word.chars[current_offset..current_offset + capacity]
                    .iter()
                    .collect();
                let line = Line {
                    indent,
                    pieces: vec![text],
                    displayed: capacity,
                    consumed: capacity,
                    marker_end: false,
                    piece_count: 1,
                };
                candidates.push((
                    line,
                    State {
                        word: current_word,
                        offset: current_offset + capacity,
                        first: false,
                    },
                    (2 * options.width + 1) as u64,
                    0,
                ));
            }
        }
        break;
    }

    let mut best: Option<Solution> = None;
    for (line, next_state, extra_cost, marker_count) in candidates {
        let tail = solve(next_state, words, options, memo);
        let gap = options.width - line.indent - line.displayed;
        let mut line_cost = (gap as u64) * (gap as u64);
        if tail.lines.is_empty() {
            line_cost /= 4;
        }
        line_cost += extra_cost;
        if !tail.lines.is_empty() && line.piece_count == 1 {
            line_cost += 3;
        }

        let mut solution = Solution {
            cost: line_cost + tail.cost,
            marker_lines: marker_count + tail.marker_lines,
            lines: Vec::with_capacity(1 + tail.lines.len()),
        };
        solution.lines.push(line);
        solution.lines.extend(tail.lines);

        if best.as_ref().map_or(true, |current| better(&solution, current)) {
            best = Some(solution);
        }
    }

    let result = best.unwrap_or_else(|| {
        let text: String = words[word_index].chars[offset..]
            .iter()
            .take(capacity)
            .collect();
        let consumed = text.chars().count();
        let line = Line {
            indent,
            pieces: vec![text],
            displayed: consumed,
            consumed,
            marker_end: false,
            piece_count: 1,
        };
        let tail = solve(
            State {
                word: word_index,
                offset: offset + consumed,
                first: false,
            },
            words,
            options,
            memo,
        );
        let gap = options.width - indent - consumed;
        let mut line_cost = (gap as u64) * (gap as u64);
        if tail.lines.is_empty() {
            line_cost /= 4;
        } else {
            line_cost += 3;
        }
        line_cost += (2 * options.width + 1) as u64;

        let mut lines = vec![line];
        lines.extend(tail.lines);
        Solution {
            cost: line_cost + tail.cost,
            marker_lines: tail.marker_lines,
            lines,
        }
    });

    memo.insert(normalized, result.clone());
    result
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn parse_segment(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut i = 0;

    while i < segment.len() {
        if is_separator(segment[i]) {
            i += 1;
            continue;
        }

        let mut chars = Vec::new();
        let mut markers = Vec::new();
        while i < segment.len() && !is_separator(segment[i]) {
            if segment[i] == '~' {
                markers.push(chars.len());
            } else {
                chars.push(segment[i]);
            }
            i += 1;
        }
        words.push(Word { chars, markers });
    }

    words
}

fn render_solution(solution: Solution, options: &WrapOptions) -> Vec<String> {
    let mut result = Vec::new();

    for (line_index, line) in solution.lines.iter().enumerate() {
        let mut rendered = " ".repeat(line.indent);
        let gaps = line.piece_count.saturating_sub(1);
        let justify = options.justify
            && line_index + 1 < solution.lines.len()
            && line.piece_count >= 2;

        if justify && gaps > 0 {
            let extra = options.width - line.indent - line.displayed;
            let each = extra / gaps;
            let remainder = extra % gaps;
            let start = line_index % gaps;

            for (i, piece) in line.pieces.iter().enumerate() {
                if i > 0 {
                    let gap_index = i - 1;
                    let bonus = usize::from((gap_index + gaps - start) % gaps < remainder);
                    rendered.push_str(&" ".repeat(1 + each + bonus));
                }
                rendered.push_str(piece);
            }
        } else {
            for (i, piece) in line.pieces.iter().enumerate() {
                if i > 0 {
                    rendered.push(' ');
                }
                rendered.push_str(piece);
            }
        }

        result.push(rendered);
    }

    result
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let count = input.chars().count();
    if count > 512 {
        return Err(WrapError::InputTooLong { chars: count });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let indexed: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..indexed.len() {
        if indexed[i].1 == '~' {
            let valid_neighbor = |j: isize| -> bool {
                if j < 0 || j as usize >= indexed.len() {
                    return false;
                }
                let c = indexed[j as usize].1;
                c != '~' && !is_separator(c)
            };
            if !valid_neighbor(i as isize - 1) || !valid_neighbor(i as isize + 1) {
                return Err(WrapError::InvalidMarker { index: indexed[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut chars = input.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\r' {
            if chars.peek() == Some(&'\n') {
                chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                let words = parse_segment(&segment);
                if !words.is_empty() {
                    let mut memo = std::collections::HashMap::new();
                    let solution = solve(
                        State {
                            word: 0,
                            offset: 0,
                            first: true,
                        },
                        &words,
                        options,
                        &mut memo,
                    );
                    output.extend(render_solution(solution, options));
                }
                segment.clear();
                for _ in 1..(j - i) {
                    output.push(String::new());
                }
                i = j;
                continue;
            }
        }
        segment.push(normalized[i]);
        i += 1;
    }

    let words = parse_segment(&segment);
    if !words.is_empty() {
        let mut memo = std::collections::HashMap::new();
        let solution = solve(
            State {
                word: 0,
                offset: 0,
                first: true,
            },
            &words,
            options,
            &mut memo,
        );
        output.extend(render_solution(solution, options));
    }

    Ok(output)
}

fn main() {}
