#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    displayed: usize,
    consumed: usize,
    marker_end: bool,
    raw: bool,
    piece_count: usize,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
}

#[derive(Clone, Copy, Hash, PartialEq, Eq)]
struct State {
    word: usize,
    offset: usize,
    first: bool,
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    let av: Vec<usize> = a.lines.iter().map(|l| l.consumed).collect();
    let bv: Vec<usize> = b.lines.iter().map(|l| l.consumed).collect();
    if av != bv {
        return av > bv;
    }
    let astr: Vec<String> = a.lines.iter().map(render_prejustified).collect();
    let bstr: Vec<String> = b.lines.iter().map(render_prejustified).collect();
    astr < bstr
}

fn render_prejustified(line: &Line) -> String {
    let mut s = " ".repeat(line.indent);
    for (i, p) in line.pieces.iter().enumerate() {
        if i > 0 {
            s.push(' ');
        }
        s.push_str(p);
    }
    s
}

fn solve(
    state: State,
    words: &[Word],
    opts: &WrapOptions,
    memo: &mut std::collections::HashMap<State, Solution>,
) -> Solution {
    let mut w = state.word;
    let mut off = state.offset;
    while w < words.len() && off == words[w].chars.len() {
        w += 1;
        off = 0;
    }
    if w >= words.len() {
        return Solution { cost: 0, marker_lines: 0, lines: Vec::new() };
    }

    let normalized = State { word: w, offset: off, first: state.first };
    if let Some(v) = memo.get(&normalized) {
        return v.clone();
    }

    let indent = if state.first { opts.first_indent } else { opts.rest_indent };
    let cap = opts.width - indent;
    let mut candidates: Vec<(Line, State, u64, usize)> = Vec::new();
    let mut pieces: Vec<String> = Vec::new();
    let mut used = 0usize;
    let mut cur_w = w;
    let mut cur_off = off;

    loop {
        if cur_w >= words.len() {
            break;
        }
        let word = &words[cur_w];
        let remaining = word.chars.len() - cur_off;
        let separator = if pieces.is_empty() { 0 } else { 1 };

        if used + separator >= cap {
            break;
        }
        let available = cap - used - separator;

        // Completed-word endings, including endings before another word that could fit.
        if remaining <= available {
            let text: String = word.chars[cur_off..].iter().collect();
            let mut next_pieces = pieces.clone();
            next_pieces.push(text);
            let displayed = used + separator + remaining;
            let line = Line {
                indent,
                pieces: next_pieces,
                displayed,
                consumed: (used.saturating_sub(pieces.iter().map(|p| p.chars().count()).sum::<usize>() + pieces.len().saturating_sub(1)))
                    + remaining,
                marker_end: false,
                raw: false,
                piece_count: pieces.len() + 1,
            };
            candidates.push((
                line,
                State { word: cur_w + 1, offset: 0, first: false },
                0,
                0,
            ));
            used += separator + remaining;
            pieces.push(text);
            cur_w += 1;
            cur_off = 0;
            if cur_w >= words.len() {
                break;
            }
            continue;
        }

        // Marker endings in this word.
        for &m in &word.markers {
            if m > cur_off && m < word.chars.len() {
                let n = m - cur_off;
                if n + 1 <= available {
                    let mut piece: String = word.chars[cur_off..m].iter().collect();
                    piece.push('-');
                    let mut next_pieces = pieces.clone();
                    next_pieces.push(piece);
                    let line = Line {
                        indent,
                        pieces: next_pieces,
                        displayed: used + separator + n + 1,
                        consumed: n,
                        marker_end: true,
                        raw: false,
                        piece_count: pieces.len() + 1,
                    };
                    candidates.push((
                        line,
                        State { word: cur_w, offset: m, first: false },
                        opts.hyphen_cost as u64,
                        1,
                    ));
                }
            }
        }

        // Raw split is possible only at the start of an otherwise empty line,
        // and only when no remaining marker ending fits.
        if pieces.is_empty() && remaining > cap {
            let marker_fits = word.markers.iter().any(|&m| {
                m > cur_off && m < word.chars.len() && m - cur_off + 1 <= cap
            });
            if !marker_fits {
                let text: String = word.chars[cur_off..cur_off + cap].iter().collect();
                let line = Line {
                    indent,
                    pieces: vec![text],
                    displayed: cap,
                    consumed: cap,
                    marker_end: false,
                    raw: true,
                    piece_count: 1,
                };
                candidates.push((
                    line,
                    State { word: cur_w, offset: cur_off + cap, first: false },
                    (2 * opts.width + 1) as u64,
                    0,
                ));
            }
        }
        break;
    }

    // A completed word may also be followed by a partial next word or another completed word;
    // candidates above are generated as the scan advances.

    let mut best: Option<Solution> = None;
    for (line, next_state, extra, marker_count) in candidates {
        let tail = solve(next_state, words, opts, memo);
        let final_line = tail.lines.is_empty();
        let gap = opts.width - line.indent - line.displayed;
        let mut cost = gap as u64 * gap as u64;
        if final_line {
            cost /= 4;
        }
        cost += extra;
        if !final_line && line.piece_count == 1 {
            cost += 3;
        }
        let mut solution = Solution {
            cost: cost + tail.cost,
            marker_lines: marker_count + tail.marker_lines,
            lines: Vec::with_capacity(1 + tail.lines.len()),
        };
        solution.lines.push(line);
        solution.lines.extend(tail.lines);
        if best.as_ref().map_or(true, |b| better(&solution, b)) {
            best = Some(solution);
        }
    }

    let result = best.unwrap_or_else(|| {
        // The specified raw-split rule guarantees a continuation.
        let text: String = words[w].chars[off..].iter().take(cap).collect();
        let n = text.chars().count();
        let line = Line {
            indent,
            pieces: vec![text],
            displayed: n,
            consumed: n,
            marker_end: false,
            raw: true,
            piece_count: 1,
        };
        let tail = solve(
            State { word: w, offset: off + n, first: false },
            words,
            opts,
            memo,
        );
        let gap = opts.width - indent - n;
        let mut cost = gap as u64 * gap as u64;
        if tail.lines.is_empty() {
            cost /= 4;
        } else {
            cost += 3;
        }
        cost += (2 * opts.width + 1) as u64;
        let mut lines = vec![line];
        lines.extend(tail.lines);
        Solution { cost: cost + tail.cost, marker_lines: tail.marker_lines, lines }
    });
    memo.insert(normalized, result.clone());
    result
}

fn parse_segment(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut i = 0;
    while i < segment.len() {
        if is_separator(segment[i]) {
            i += 1;
            continue;
        }
        let mut chars = Vec::new();
        let mut markers = Vec::new();
        while i < segment.len() && !is_separator(segment[i]) {
            if segment[i] == '~' {
                markers.push(chars.len());
            } else {
                chars.push(segment[i]);
            }
            i += 1;
        }
        words.push(Word { chars, markers });
    }
    words
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let count = input.chars().count();
    if count > 512 {
        return Err(WrapError::InputTooLong { chars: count });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let indexed: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..indexed.len() {
        if indexed[i].1 == '~' {
            let valid_neighbor = |j: isize| -> bool {
                if j < 0 || j as usize >= indexed.len() {
                    return false;
                }
                let c = indexed[j as usize].1;
                c != '~' && !is_separator(c)
            };
            if !valid_neighbor(i as isize - 1) || !valid_neighbor(i as isize + 1) {
                return Err(WrapError::InvalidMarker { index: indexed[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut chars = input.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\r' {
            if chars.peek() == Some(&'\n') {
                chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                let words = parse_segment(&segment);
                if !words.is_empty() {
                    let mut memo = std::collections::HashMap::new();
                    let solution = solve(
                        State { word: 0, offset: 0, first: true },
                        &words,
                        options,
                        &mut memo,
                    );
                    output.extend(render_solution(solution, options));
                }
                segment.clear();
                for _ in 1..(j - i) {
                    output.push(String::new());
                }
                i = j;
                continue;
            }
        }
        segment.push(normalized[i]);
        i += 1;
    }

    let words = parse_segment(&segment);
    if !words.is_empty() {
        let mut memo = std::collections::HashMap::new();
        let solution = solve(
            State { word: 0, offset: 0, first: true },
            &words,
            options,
            &mut memo,
        );
        output.extend(render_solution(solution, options));
    }
    Ok(output)
}

fn render_solution(solution: Solution, opts: &WrapOptions) -> Vec<String> {
    let mut result = Vec::new();
    for (line_index, line) in solution.lines.iter().enumerate() {
        let mut s = " ".repeat(line.indent);
        let mut gaps = line.piece_count.saturating_sub(1);
        if opts.justify && line_index + 1 < solution.lines.len() && line.piece_count >= 2 {
            let extra = opts.width.saturating_sub(line.indent + line.displayed);
            if gaps > 0 {
                let each = extra / gaps;
                let rem = extra % gaps;
                let start = line_index % gaps;
                for (i, piece) in line.pieces.iter().enumerate() {
                    if i > 0 {
                        let gap_index = i - 1;
                        let bonus = if (gap_index + gaps - start) % gaps < rem { 1 } else { 0 };
                        s.push_str(&" ".repeat(1 + each + bonus));
                    }
                    s.push_str(piece);
                }
            }
        } else {
            gaps = 0;
            let _ = gaps;
            for (i, piece) in line.pieces.iter().enumerate() {
                if i > 0 {
                    s.push(' ');
                }
                s.push_str(piece);
            }
        }
        result.push(s);
    }
    result
}

fn main() {}
