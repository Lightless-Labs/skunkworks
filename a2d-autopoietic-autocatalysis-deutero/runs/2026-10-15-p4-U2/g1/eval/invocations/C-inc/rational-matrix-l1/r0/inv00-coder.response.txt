```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_wide(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        let (num, den) = if den < 0 { (-num, -den) } else { (num, den) };
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let divisor = gcd(num.unsigned_abs(), den as u128) as i128;
        let num = num / divisor;
        let den = den / divisor;

        if num < i64::MIN as i128 || num > i64::MAX as i128 || den < 1 || den > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }

        Ok(Self {
            num: num as i64,
            den: den as i64,
        })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_wide(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let num = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let den = self.den as i128 * rhs.den as i128;
        Self::from_wide(num, den)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let num = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let den = self.den as i128 * rhs.den as i128;
        Self::from_wide(num, den)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let num = self.num as i128 * rhs.num as i128;
        let den = self.den as i128 * rhs.den as i128;
        Self::from_wide(num, den)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let num = self.num as i128 * rhs.den as i128;
        let den = self.den as i128 * rhs.num as i128;
        Self::from_wide(num, den)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_wide(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }
        let num_start = i;

        if i == bytes.len() {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }
        if !bytes[i].is_ascii_digit() {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }

        let mut den_start = None;
        let mut slash_at = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            slash_at = Some(i);
            i += 1;
            den_start = Some(i);
            if i == bytes.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            if !bytes[i].is_ascii_digit() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i < bytes.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }

        let num_end = slash_at.unwrap_or(bytes.len());
        let num_digits_start = num_start;
        let num_digits = num_end - num_digits_start;
        if num_digits >= 2 && bytes[num_digits_start] == b'0' {
            return Err(parse_err(num_digits_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && num_digits == 1 && bytes[num_digits_start] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        if let Some(ds) = den_start {
            let de = bytes.len();
            if de - ds >= 2 && bytes[ds] == b'0' {
                return Err(parse_err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let num_limit: u128 = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let mut num_mag = 0u128;
        for &b in &bytes[num_digits_start..num_end] {
            num_mag = num_mag.saturating_mul(10).saturating_add((b - b'0') as u128);
            if num_mag > num_limit {
                return Err(parse_err(0, ParseKind::OutOfRange));
            }
        }

        let denominator = if let Some(ds) = den_start {
            let mut value = 0u128;
            for &b in &bytes[ds..] {
                value = value.saturating_mul(10).saturating_add((b - b'0') as u128);
                if value > i64::MAX as u128 {
                    return Err(parse_err(ds, ParseKind::OutOfRange));
                }
            }
            value as u64
        } else {
            1
        };

        if let Some(ds) = den_start {
            if denominator == 0 {
                return Err(parse_err(ds, ParseKind::ZeroDenominator));
            }
            if denominator == 1 {
                return Err(parse_err(slash_at.unwrap(), ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_err(slash_at.unwrap(), ParseKind::NonCanonical));
            }
            if gcd(num_mag, denominator as u128) > 1 {
                return Err(parse_err(slash_at.unwrap(), ParseKind::NonCanonical));
            }
        }

        let signed_num = if negative {
            if num_mag == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };
        Ok(Self {
            num: signed_num,
            den: denominator as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

fn main() {}
```
