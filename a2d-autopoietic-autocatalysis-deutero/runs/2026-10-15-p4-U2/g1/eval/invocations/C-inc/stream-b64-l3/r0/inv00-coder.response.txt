use std::fmt;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

impl fmt::Display for Error {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "{self:?}")
    }
}

impl std::error::Error for Error {}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 64] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/",
        Alphabet::UrlSafe => b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_",
    }
}

fn encode_symbols(input: &[u8], alphabet: Alphabet, padding: Padding) -> String {
    let table = alphabet_bytes(alphabet);
    let mut out = String::new();

    for chunk in input.chunks(3) {
        let a = chunk[0];
        let b = if chunk.len() > 1 { chunk[1] } else { 0 };
        let c = if chunk.len() > 2 { chunk[2] } else { 0 };
        let indices = [
            (a >> 2) as usize,
            (((a & 0x03) << 4) | (b >> 4)) as usize,
            (((b & 0x0f) << 2) | (c >> 6)) as usize,
            (c & 0x3f) as usize,
        ];
        out.push(table[indices[0]] as char);
        out.push(table[indices[1]] as char);
        if chunk.len() > 1 {
            out.push(table[indices[2]] as char);
        } else if padding != Padding::Forbidden {
            out.push('=');
        }
        if chunk.len() > 2 {
            out.push(table[indices[3]] as char);
        } else if padding != Padding::Forbidden {
            out.push('=');
        }
    }
    out
}

fn wrap_symbols(symbols: String, whitespace: Whitespace) -> String {
    let Whitespace::Wrapped { width, ending } = whitespace else {
        return symbols;
    };
    if symbols.len() <= width {
        return symbols;
    }

    let separator = match ending {
        LineEnding::Lf => "\n",
        LineEnding::CrLf => "\r\n",
    };
    let mut out = String::with_capacity(symbols.len() + (symbols.len() / width) * separator.len());
    for (i, chunk) in symbols.as_bytes().chunks(width).enumerate() {
        if i != 0 {
            out.push_str(separator);
        }
        out.push_str(std::str::from_utf8(chunk).expect("base64 symbols are ASCII"));
    }
    out
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let symbols = encode_symbols(input, options.alphabet, options.padding);
    Ok(wrap_symbols(symbols, options.whitespace))
}

fn digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    let table = alphabet_bytes(alphabet);
    table.iter().position(|&b| b == byte).map(|n| n as u8)
}

fn is_ignored(byte: u8, whitespace: Whitespace) -> bool {
    matches!(whitespace, Whitespace::IgnoreAscii)
        && matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
}

#[derive(Clone, Copy)]
enum FrameState {
    Row,
    NeedLf { cr_offset: usize, separator_offset: usize },
    NeedSymbol { separator_offset: usize },
}

struct DecoderCore {
    options: Options,
    group: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    pending_second_pad: Option<usize>,
    terminal: bool,
    offset: usize,
    symbols_in_row: usize,
    frame: FrameState,
    failed: Option<Error>,
    finished: bool,
}

impl DecoderCore {
    fn new(options: Options) -> Self {
        Self {
            options,
            group: [0; 4],
            offsets: [0; 4],
            len: 0,
            pending_second_pad: None,
            terminal: false,
            offset: 0,
            symbols_in_row: 0,
            frame: FrameState::Row,
            failed: None,
            finished: false,
        }
    }

    fn sticky(&mut self, error: Error) -> Error {
        if let Some(existing) = &self.failed {
            return existing.clone();
        }
        self.failed = Some(error.clone());
        error
    }

    fn emit_group(&mut self, out: &mut Vec<u8>) {
        let a = self.group[0];
        let b = self.group[1];
        let c = self.group[2];
        let d = self.group[3];
        out.push((a << 2) | (b >> 4));
        if c != 64 {
            out.push((b << 4) | (c >> 2));
        }
        if d != 64 {
            out.push((c << 6) | d);
        }
        if c == 64 || d == 64 {
            self.terminal = true;
        }
        self.len = 0;
        self.pending_second_pad = None;
    }

    fn scan_symbol(&mut self, byte: u8, at: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if self.terminal {
            return Err(Error::BadPadding { offset: at });
        }

        if let Some(pad_at) = self.pending_second_pad {
            if byte != b'=' {
                return Err(Error::BadPadding { offset: at });
            }
            self.group[2] = 64;
            self.group[3] = 64;
            self.offsets[3] = at;
            self.len = 4;
            self.pending_second_pad = None;
            self.emit_group(out);
            let _ = pad_at;
            return Ok(());
        }

        if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset: at });
            }
            match self.len {
                0 | 1 => return Err(Error::BadPadding { offset: at }),
                2 => {
                    self.group[2] = 64;
                    self.offsets[2] = at;
                    self.pending_second_pad = Some(at);
                    self.len = 3;
                }
                3 => {
                    self.group[3] = 64;
                    self.offsets[3] = at;
                    self.len = 4;
                    self.emit_group(out);
                }
                _ => return Err(Error::BadPadding { offset: at }),
            }
            return Ok(());
        }

        let Some(value) = digit(byte, self.options.alphabet) else {
            return Err(Error::InvalidByte { offset: at, byte });
        };
        if self.len == 3 && self.group[2] == 64 {
            return Err(Error::BadPadding { offset: at });
        }

        self.group[self.len] = value;
        self.offsets[self.len] = at;
        self.len += 1;
        if self.len == 4 {
            self.emit_group(out);
        }
        Ok(())
    }

    fn process_wrapped(&mut self, byte: u8, at: usize) -> Result<bool, Error> {
        let Whitespace::Wrapped { width, ending } = self.options.whitespace else {
            return Ok(false);
        };

        match self.frame {
            FrameState::NeedLf { cr_offset, .. } => {
                if byte != b'\n' {
                    return Err(Error::BadWrap { offset: at });
                }
                self.frame = FrameState::NeedSymbol {
                    separator_offset: cr_offset,
                };
                return Ok(true);
            }
            FrameState::NeedSymbol { .. } => {
                if byte == b'\r' || byte == b'\n' {
                    return Err(Error::BadWrap { offset: at });
                }
                self.frame = FrameState::Row;
                self.symbols_in_row = 0;
            }
            FrameState::Row => {
                if self.symbols_in_row == width {
                    let separator_offset = at;
                    match ending {
                        LineEnding::Lf if byte == b'\n' => {
                            self.frame = FrameState::NeedSymbol { separator_offset };
                            return Ok(true);
                        }
                        LineEnding::CrLf if byte == b'\r' => {
                            self.frame = FrameState::NeedLf {
                                cr_offset: at,
                                separator_offset,
                            };
                            return Ok(true);
                        }
                        _ => return Err(Error::BadWrap { offset: at }),
                    }
                }
                if byte == b'\n' || byte == b'\r' {
                    return Err(Error::BadWrap { offset: at });
                }
            }
        }

        self.symbols_in_row += 1;
        Ok(false)
    }

    fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.failed {
            return Err(error.clone());
        }
        if self.finished {
            return Err(self.sticky(Error::AlreadyFinished));
        }

        let mut out = Vec::new();
        for &byte in input {
            let at = self.offset;
            self.offset += 1;

            if is_ignored(byte, self.options.whitespace) {
                continue;
            }

            match self.process_wrapped(byte, at) {
                Ok(true) => continue,
                Ok(false) => {}
                Err(error) => return Err(self.sticky(error)),
            }

            if let Err(error) = self.scan_symbol(byte, at, &mut out) {
                return Err(self.sticky(error));
            }
        }
        Ok(out)
    }

    fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.failed {
            return Err(error.clone());
        }
        if self.finished {
            return Err(self.sticky(Error::AlreadyFinished));
        }

        match self.frame {
            FrameState::NeedLf { cr_offset, .. } => {
                return Err(self.sticky(Error::BadWrap { offset: cr_offset }));
            }
            FrameState::NeedSymbol { separator_offset } => {
                return Err(self.sticky(Error::BadWrap {
                    offset: separator_offset,
                }));
            }
            FrameState::Row => {}
        }

        if self.pending_second_pad.is_some() {
            let at = self.pending_second_pad.unwrap();
            return Err(self.sticky(Error::BadPadding { offset: at }));
        }

        let mut out = Vec::new();
        match self.len {
            0 => {}
            1 => {
                return Err(self.sticky(Error::Truncated {
                    offset: self.offsets[0],
                }));
            }
            2 if self.options.padding != Padding::Required => {
                let a = self.group[0];
                let b = self.group[1];
                out.push((a << 2) | (b >> 4));
            }
            3 if self.options.padding != Padding::Required && self.group[2] != 64 => {
                let a = self.group[0];
                let b = self.group[1];
                let c = self.group[2];
                out.push((a << 2) | (b >> 4));
                out.push((b << 4) | (c >> 2));
            }
            _ => {
                return Err(self.sticky(Error::Truncated {
                    offset: self.offsets[0],
                }));
            }
        }

        self.finished = true;
        Ok(out)
    }
}

pub struct Decoder {
    core: DecoderCore,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            core: DecoderCore::new(options),
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        self.core.push(input)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        self.core.finish()
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder {
        core: DecoderCore::new(options),
    };
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

pub struct Encoder {
    options: Options,
    pending: [u8; 2],
    pending_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            pending: [0; 2],
            pending_len: 0,
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut combined = Vec::with_capacity(self.pending_len + input.len());
        combined.extend_from_slice(&self.pending[..self.pending_len]);
        combined.extend_from_slice(input);
        let complete_len = combined.len() / 3 * 3;
        let encoded = encode_symbols(
            &combined[..complete_len],
            self.options.alphabet,
            self.options.padding,
        );
        self.pending_len = combined.len() - complete_len;
        self.pending[..self.pending_len]
            .copy_from_slice(&combined[complete_len..]);
        Ok(encoded)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let result = encode_symbols(
            &self.pending[..self.pending_len],
            self.options.alphabet,
            self.options.padding,
        );
        self.pending_len = 0;
        self.finished = true;

        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            // Maintain rows across pushes by applying wrapping to the complete stream
            // would require buffering; encoded output is wrapped below per emitted chunk.
            // Each complete group is width divisible by four, so chunk boundaries align.
            Ok(wrap_symbols(result, Whitespace::Wrapped { width, ending }))
        } else {
            Ok(result)
        }
    }
}

fn main() {}
