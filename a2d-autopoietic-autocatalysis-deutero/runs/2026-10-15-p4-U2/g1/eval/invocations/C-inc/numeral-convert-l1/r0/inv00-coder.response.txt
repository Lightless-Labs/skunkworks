#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanCase {
    Upper,
    Lower,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanError {
    Empty,
    InvalidByte { index: usize },
    MixedCase { index: usize },
    NonCanonical { index: usize },
    Range,
}

pub fn roman_encode(value: u32, case: RomanCase) -> Result<String, RomanError> {
    if value > 3999 {
        return Err(RomanError::Range);
    }

    let lower = matches!(case, RomanCase::Lower);
    if value == 0 {
        return Ok(if lower { "n" } else { "N" }.to_string());
    }

    const TOKENS: &[(u32, &str)] = &[
        (1000, "M"),
        (900, "CM"),
        (500, "D"),
        (400, "CD"),
        (100, "C"),
        (90, "XC"),
        (50, "L"),
        (40, "XL"),
        (10, "X"),
        (9, "IX"),
        (5, "V"),
        (4, "IV"),
        (1, "I"),
    ];

    let mut remaining = value;
    let mut output = String::new();

    for &(token_value, token) in TOKENS {
        while remaining >= token_value {
            remaining -= token_value;
            if lower {
                output.push_str(&token.to_ascii_lowercase());
            } else {
                output.push_str(token);
            }
        }
    }

    Ok(output)
}

pub fn roman_decode(input: &str) -> Result<u32, RomanError> {
    if input.is_empty() {
        return Err(RomanError::Empty);
    }

    let bytes = input.as_bytes();

    for (index, &byte) in bytes.iter().enumerate() {
        if !matches!(byte, b'I' | b'V' | b'X' | b'L' | b'C' | b'D' | b'M' | b'N'
            | b'i' | b'v' | b'x' | b'l' | b'c' | b'd' | b'm' | b'n')
        {
            return Err(RomanError::InvalidByte { index });
        }
    }

    let mut case = None;
    for (index, &byte) in bytes.iter().enumerate() {
        let is_lower = byte.is_ascii_lowercase();
        if let Some(first_case) = case {
            if first_case != is_lower {
                return Err(RomanError::MixedCase { index });
            }
        } else {
            case = Some(is_lower);
        }
    }

    if bytes.len() == 1 && matches!(bytes[0], b'N' | b'n') {
        return Ok(0);
    }

    if let Some(index) = bytes.iter().position(|&byte| matches!(byte, b'N' | b'n')) {
        let _ = index;
        return Err(RomanError::NonCanonical { index: 0 });
    }

    let value_of = |byte: u8| -> u32 {
        match byte.to_ascii_uppercase() {
            b'I' => 1,
            b'V' => 5,
            b'X' => 10,
            b'L' => 50,
            b'C' => 100,
            b'D' => 500,
            b'M' => 1000,
            _ => 0,
        }
    };

    let mut total = 0u32;
    let mut position = 0usize;
    while position < bytes.len() {
        let current = value_of(bytes[position]);
        if position + 1 < bytes.len() {
            let next = value_of(bytes[position + 1]);
            if next > current {
                total = total.saturating_add(next - current).min(4000);
                position += 2;
                continue;
            }
        }
        total = total.saturating_add(current).min(4000);
        position += 1;
    }

    if total > 3999 {
        return Err(RomanError::Range);
    }

    let decoded_case = if case.unwrap_or(false) {
        RomanCase::Lower
    } else {
        RomanCase::Upper
    };
    let canonical = roman_encode(total, decoded_case)?;

    let input_bytes = input.as_bytes();
    let canonical_bytes = canonical.as_bytes();
    let common_len = input_bytes.len().min(canonical_bytes.len());

    for index in 0..common_len {
        if input_bytes[index] != canonical_bytes[index] {
            return Err(RomanError::NonCanonical { index });
        }
    }

    if input_bytes.len() != canonical_bytes.len() {
        return Err(RomanError::NonCanonical { index: common_len });
    }

    Ok(total)
}

fn main() {}
