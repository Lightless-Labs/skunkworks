#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn name_error(site: String, name: &str) -> Option<ResolveError> {
    if valid_name(name) {
        None
    } else {
        Some(ResolveError::InvalidName {
            site,
            name: name.to_owned(),
        })
    }
}

fn canonicalize(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    let (prefix, version) = match constraint {
        Constraint::Exact(v) => ("=", v),
        Constraint::AtLeast(v) => (">=", v),
        Constraint::Before(v) => ("<", v),
        Constraint::Caret(v) => ("^", v),
    };
    format!("{prefix}{}", version_text(version))
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push(char::from_digit((byte >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            out.push(char::from_digit((byte & 0x0f) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => {
            let constraints = canonicalize(requirements.clone())
                .into_iter()
                .map(constraint_text)
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints
            )
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => {
            let constraints = canonicalize(requirements.clone())
                .into_iter()
                .map(constraint_text)
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints
            )
        }
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn validate(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, requirement) in roots.iter().enumerate() {
        if let Some(error) = name_error(format!("root[{i}]"), &requirement.package) {
            return Err(error);
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if let Some(error) = name_error(format!("catalog[{i}]"), &package.name) {
            return Err(error);
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dependency) in release.deps.iter().enumerate() {
                if let Some(error) = name_error(
                    format!("catalog[{i}].release[{j}].dep[{k}]"),
                    &dependency.package,
                ) {
                    return Err(error);
                }
            }
        }
    }

    let mut names = std::collections::BTreeSet::new();
    for package in catalog {
        if !names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    Ok(())
}

fn inspect(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &std::collections::BTreeMap<String, Version>,
) -> Result<
    (
        std::collections::BTreeMap<String, Vec<Constraint>>,
        Option<String>,
    ),
    ResolveError,
> {
    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();

    for requirement in roots {
        active
            .entry(requirement.package.clone())
            .or_default()
            .push(requirement.constraint);
    }

    for (package_name, version) in selected {
        if let Some(package) = catalog.iter().find(|p| p.name == *package_name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dependency in &release.deps {
                    active
                        .entry(dependency.package.clone())
                        .or_default()
                        .push(dependency.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        *constraints = canonicalize(std::mem::take(constraints));
    }

    if let Some(package) = active
        .keys()
        .filter(|name| !catalog.iter().any(|p| p.name == ***name))
        .next()
    {
        return Err(ResolveError::Missing {
            package: package.clone(),
        });
    }

    for (package, version) in selected {
        if let Some(requirements) = active.get(package) {
            if !requirements
                .iter()
                .all(|constraint| matches(*version, *constraint))
            {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: requirements.clone(),
                });
            }
        }
    }

    for (package, requirements) in &active {
        if selected.contains_key(package) {
            continue;
        }
        let entry = catalog.iter().find(|p| p.name == *package).unwrap();
        if !entry.releases.iter().any(|release| {
            requirements
                .iter()
                .all(|constraint| matches(release.version, *constraint))
        }) {
            return Err(ResolveError::NoMatch {
                package: package.clone(),
                requirements: requirements.clone(),
            });
        }
    }

    let next = active.keys().find(|package| !selected.contains_key(*package)).cloned();
    Ok((active, next))
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &mut std::collections::BTreeMap<String, Version>,
    attempts: &mut usize,
    max_attempts: usize,
) -> Result<(), ResolveError> {
    let (_, next) = inspect(catalog, roots, selected)?;
    let Some(package_name) = next else {
        return Ok(());
    };

    let package = catalog.iter().find(|p| p.name == package_name).unwrap();
    let mut candidates: Vec<Version> = package.releases.iter().map(|r| r.version).collect();
    candidates.sort_by(|a, b| b.cmp(a));

    let mut first_error = None;
    for version in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.insert(package_name.clone(), version);

        match search(catalog, roots, selected, attempts, max_attempts) {
            Ok(()) => return Ok(()),
            Err(error @ ResolveError::Limit { .. }) => return Err(error),
            Err(error) => {
                selected.remove(&package_name);
                if first_error.is_none() {
                    first_error = Some(error);
                }
            }
        }
    }

    Err(first_error.unwrap_or_else(|| ResolveError::NoMatch {
        package: package_name,
        requirements: Vec::new(),
    }))
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate(catalog, roots)?;

    let mut selected = std::collections::BTreeMap::new();
    let mut attempts = 0;
    search(catalog, roots, &mut selected, &mut attempts, max_attempts)?;

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}
