```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn validate_names(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn validate_duplicates(catalog: &[Package]) -> Result<(), ResolveError> {
    for i in 0..catalog.len() {
        if catalog[..i].iter().any(|p| p.name == catalog[i].name) {
            return Err(ResolveError::DuplicatePackage {
                package: catalog[i].name.clone(),
            });
        }
    }

    for package in catalog {
        for (i, release) in package.releases.iter().enumerate() {
            if package.releases[..i]
                .iter()
                .any(|r| r.version == release.version)
            {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    Ok(())
}

fn constraint_matches(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn collect_active(
    roots: &[Requirement],
    selected: &[(String, usize)],
    catalog: &[Package],
) -> Vec<(String, Vec<Constraint>)> {
    let mut active: Vec<(String, Vec<Constraint>)> = Vec::new();

    fn add(active: &mut Vec<(String, Vec<Constraint>)>, package: &str, constraint: Constraint) {
        match active.iter_mut().find(|(name, _)| name == package) {
            Some((_, constraints)) => constraints.push(constraint),
            None => active.push((package.to_owned(), vec![constraint])),
        }
    }

    for root in roots {
        add(&mut active, &root.package, root.constraint);
    }

    for (name, release_index) in selected {
        if let Some(package) = catalog.iter().find(|p| p.name == *name) {
            for dep in &package.releases[*release_index].deps {
                add(&mut active, &dep.package, dep.constraint);
            }
        }
    }

    for (_, constraints) in &mut active {
        canonicalize(constraints);
    }
    active.sort_by(|a, b| a.0.cmp(&b.0));
    active
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &mut Vec<(String, usize)>,
    attempts: &mut usize,
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    let active = collect_active(roots, selected, catalog);

    if let Some((name, _)) = active
        .iter()
        .filter(|(name, _)| !catalog.iter().any(|p| p.name == *name))
        .min_by(|a, b| a.0.cmp(&b.0))
    {
        return Err(ResolveError::Missing {
            package: name.clone(),
        });
    }

    for (name, constraints) in &active {
        if let Some((_, release_index)) = selected.iter().find(|(n, _)| n == name) {
            let package = catalog.iter().find(|p| p.name == *name).unwrap();
            let version = package.releases[*release_index].version;
            if !constraints
                .iter()
                .all(|constraint| constraint_matches(*constraint, version))
            {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    let mut candidates_by_name: Vec<(&str, Vec<usize>)> = Vec::new();
    for (name, constraints) in &active {
        if selected.iter().any(|(n, _)| n == name) {
            continue;
        }
        let package = catalog.iter().find(|p| p.name == *name).unwrap();
        let mut candidates: Vec<usize> = package
            .releases
            .iter()
            .enumerate()
            .filter(|(_, release)| {
                constraints
                    .iter()
                    .all(|constraint| constraint_matches(*constraint, release.version))
            })
            .map(|(i, _)| i)
            .collect();
        candidates.sort_by(|a, b| {
            package.releases[*b]
                .version
                .cmp(&package.releases[*a].version)
        });
        if candidates.is_empty() {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
        candidates_by_name.push((name, candidates));
    }

    let Some((name, candidates)) = candidates_by_name.first() else {
        let mut result: Vec<Selection> = selected
            .iter()
            .map(|(name, release_index)| {
                let package = catalog.iter().find(|p| p.name == *name).unwrap();
                Selection {
                    package: name.clone(),
                    version: package.releases[*release_index].version,
                }
            })
            .collect();
        result.sort_by(|a, b| a.package.cmp(&b.package));
        return Ok(result);
    };

    let name = (*name).to_owned();
    let mut first_error = None;

    for release_index in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.push((name.clone(), *release_index));
        let result = search(catalog, roots, selected, attempts, max_attempts);
        selected.pop();

        match result {
            Ok(solution) => return Ok(solution),
            Err(error @ ResolveError::Limit { .. }) => return Err(error),
            Err(error) => {
                if first_error.is_none() {
                    first_error = Some(error);
                }
            }
        }
    }

    Err(first_error.unwrap_or(ResolveError::NoMatch {
        package: name,
        requirements: Vec::new(),
    }))
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;
    validate_duplicates(catalog)?;
    let mut attempts = 0;
    search(catalog, roots, &mut Vec::new(), &mut attempts, max_attempts)
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraint_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut encoded = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            encoded.push(*byte as char);
        } else {
            encoded.push('%');
            encoded.push(char::from_digit((byte >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            encoded.push(char::from_digit((byte & 0x0f) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    encoded
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            canonical_constraint_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            canonical_constraint_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}
```
