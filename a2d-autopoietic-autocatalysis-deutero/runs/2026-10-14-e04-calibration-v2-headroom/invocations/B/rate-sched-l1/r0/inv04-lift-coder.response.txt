```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    sequence: u64,
    jobs: Vec<Entry>,
    reserved: Vec<String>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum State {
    Queued,
    Cancelled,
    Completed,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Entry {
    job: Job,
    sequence: u64,
    state: State,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            sequence: 0,
            jobs: Vec::new(),
            reserved: Vec::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        self.check_time(now)?;

        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved.iter().any(|id| id == &job.id) {
            return Err(Error::Duplicate);
        }

        self.reserved.push(job.id.clone());
        self.jobs.push(Entry {
            job,
            sequence: self.sequence,
            state: State::Queued,
        });
        self.sequence += 1;
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        let index = self.jobs.iter().position(|entry| entry.job.id == id);
        let Some(index) = index else {
            return Err(Error::Unknown);
        };

        match self.jobs[index].state {
            State::Queued => self.jobs[index].state = State::Cancelled,
            State::Cancelled => return Err(Error::Cancelled),
            State::Completed => return Err(Error::Completed),
        }

        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        self.check_time(now)?;

        let mut best: Option<(usize, u8, u64)> = None;
        for (index, entry) in self.jobs.iter().enumerate() {
            if entry.state != State::Queued || entry.job.ready_at > now {
                continue;
            }

            let effective = effective_priority(&entry.job, now, self.aging_ms);
            match best {
                None => best = Some((index, effective, entry.sequence)),
                Some((_, best_priority, best_sequence))
                    if effective > best_priority
                        || (effective == best_priority && entry.sequence < best_sequence) =>
                {
                    best = Some((index, effective, entry.sequence));
                }
                _ => {}
            }
        }

        self.last_time = now;
        let Some((index, effective_priority, _)) = best else {
            return Ok(None);
        };

        self.jobs[index].state = State::Completed;
        Ok(Some(Dispatch {
            id: self.jobs[index].job.id.clone(),
            effective_priority,
        }))
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes
            .iter()
            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-')
}

fn effective_priority(job: &Job, now: u64, aging_ms: u64) -> u8 {
    if aging_ms == 0 {
        return job.priority;
    }
    let gain = (now - job.ready_at) / aging_ms;
    if gain >= 9 - job.priority as u64 {
        9
    } else {
        job.priority + gain as u8
    }
}
```
