#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let minute_field = &text[..split];
        let hour_field = &text[split + 1..];

        let mut cron = Self {
            minutes: [false; 60],
            hours: [false; 24],
        };

        Self::parse_field(minute_field, 0, 60, &mut cron.minutes)?;
        Self::parse_field(hour_field, split + 1, 24, &mut cron.hours)?;
        Ok(cron)
    }

    fn parse_field(
        field: &str,
        offset: usize,
        limit: usize,
        selected: &mut [bool],
    ) -> Result<(), CronError> {
        let bytes = field.as_bytes();
        let mut item_start = 0;

        loop {
            let item_end = bytes[item_start..]
                .iter()
                .position(|&b| b == b',')
                .map(|p| item_start + p)
                .unwrap_or(bytes.len());

            if item_start == item_end {
                return Err(CronError::Syntax(offset + item_start));
            }

            Self::parse_item(
                &field[item_start..item_end],
                offset + item_start,
                limit,
                selected,
            )?;

            if item_end == bytes.len() {
                break;
            }
            item_start = item_end + 1;
        }
        Ok(())
    }

    fn parse_item(
        item: &str,
        offset: usize,
        limit: usize,
        selected: &mut [bool],
    ) -> Result<(), CronError> {
        let bytes = item.as_bytes();
        let slash = bytes.iter().position(|&b| b == b'/');

        if let Some(slash_pos) = slash {
            if bytes[slash_pos + 1..].contains(&b'/') {
                let second = slash_pos + 1
                    + bytes[slash_pos + 1..]
                        .iter()
                        .position(|&b| b == b'/')
                        .unwrap();
                return Err(CronError::Syntax(offset + second));
            }
        }

        let base_end = slash.unwrap_or(bytes.len());
        let base = &item[..base_end];

        let (start, end, has_step) = if base == "*" {
            (0, limit - 1, true)
        } else if let Some(dash) = base.as_bytes().iter().position(|&b| b == b'-') {
            let left = Self::parse_number(&base[..dash], offset, limit)?;
            let right = Self::parse_number(
                &base[dash + 1..],
                offset + dash + 1,
                limit,
            )?;
            if left > right {
                return Err(CronError::Descending(offset + dash));
            }
            (left, right, true)
        } else {
            if let Some(slash_pos) = slash {
                return Err(CronError::Syntax(offset + slash_pos));
            }
            let number = Self::parse_number(base, offset, limit)?;
            (number, number, false)
        };

        let step = if let Some(slash_pos) = slash {
            if !has_step {
                return Err(CronError::Syntax(offset + slash_pos));
            }
            Self::parse_step(
                &item[slash_pos + 1..],
                offset + slash_pos + 1,
                limit,
            )?
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }
        Ok(())
    }

    fn parse_number(fragment: &str, offset: usize, limit: usize) -> Result<usize, CronError> {
        let bytes = fragment.as_bytes();
        if bytes.is_empty() {
            return Err(CronError::Syntax(offset));
        }

        if let Some(pos) = bytes.iter().position(|b| !b.is_ascii_digit()) {
            return Err(CronError::Syntax(offset + pos));
        }

        if bytes.len() > 1 && bytes[0] == b'0' {
            return Err(CronError::Number(offset));
        }

        let mut value = 0usize;
        for &b in bytes {
            value = value
                .saturating_mul(10)
                .saturating_add((b - b'0') as usize);
            if value >= limit {
                return Err(CronError::Bounds(offset));
            }
        }
        Ok(value)
    }

    fn parse_step(fragment: &str, offset: usize, limit: usize) -> Result<usize, CronError> {
        let bytes = fragment.as_bytes();
        if bytes.is_empty() {
            return Err(CronError::Syntax(offset));
        }

        if let Some(pos) = bytes.iter().position(|b| !b.is_ascii_digit()) {
            return Err(CronError::Syntax(offset + pos));
        }

        if bytes.len() > 1 && bytes[0] == b'0' {
            return Err(CronError::Number(offset));
        }

        let max_step = limit;
        let mut value = 0usize;
        for &b in bytes {
            value = value
                .saturating_mul(10)
                .saturating_add((b - b'0') as usize);
            if value > max_step {
                return Err(CronError::Step(offset));
            }
        }

        if value == 0 || value > max_step {
            return Err(CronError::Step(offset));
        }
        Ok(value)
    }

    pub fn matches(&self, minute: u64) -> bool {
        let minute_of_hour = (minute % 60) as usize;
        let hour_of_day = ((minute / 60) % 24) as usize;
        self.minutes[minute_of_hour] && self.hours[hour_of_day]
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    insertion_order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|job| job.id == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;

        let insertion_order = self.jobs.len();
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            insertion_order,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let first_boundary = match self.last_poll {
            Some(last) => last / 60_000 + 1,
            None => 0,
        };
        let last_boundary = now / 60_000;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        if first_boundary <= last_boundary {
            for job in &self.jobs {
                let matching: Vec<u64> = (first_boundary..=last_boundary)
                    .filter(|&minute| job.cron.matches(minute))
                    .collect();

                match job.policy {
                    Policy::All => {
                        for minute in matching {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.insertion_order,
                            ));
                        }
                    }
                    Policy::Once => {
                        if let Some(&minute) = matching.last() {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.insertion_order,
                            ));
                        }
                    }
                    Policy::Skip => {
                        if now % 60_000 == 0 {
                            let minute = now / 60_000;
                            if job.cron.matches(minute) && minute >= first_boundary {
                                emitted.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: now,
                                    },
                                    job.priority,
                                    job.insertion_order,
                                ));
                            }
                        }
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|entry| entry.0).collect())
    }
}
