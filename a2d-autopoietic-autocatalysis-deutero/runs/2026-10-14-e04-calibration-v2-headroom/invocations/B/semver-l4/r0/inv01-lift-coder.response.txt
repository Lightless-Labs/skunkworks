use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_u64_field(s: &str, start: usize, end: usize) -> Result<u64, VersionError> {
    if end - start > 1 && s.as_bytes()[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    s[start..end]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let b = s.as_bytes();
    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for i in 0..3 {
        let start = pos;
        while pos < b.len() && b[pos].is_ascii_digit() {
            pos += 1;
        }
        if pos == start {
            if pos == b.len() || matches!(b[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        core[i] = parse_u64_field(s, start, pos)?;
        if i < 2 {
            if pos >= b.len() || matches!(b[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if b[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < b.len() && b[pos] != b'-' && b[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    if pos < b.len() && b[pos] == b'-' {
        let marker = pos;
        pos += 1;
        let pre_start = pos;
        while pos < b.len() && b[pos] != b'+' {
            pos += 1;
        }
        parse_identifiers(s, pre_start, pos, true, &mut pre)?;
        if pos == marker + 1 {
            return Err(VersionError::EmptyIdentifier { at: pos });
        }
    }

    if pos < b.len() && b[pos] == b'+' {
        pos += 1;
        parse_identifiers(s, pos, b.len(), false, &mut build)?;
    }
    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    if start == end {
        return Err(VersionError::EmptyIdentifier { at: start });
    }
    let b = s.as_bytes();
    let mut part = start;
    let mut i = start;
    loop {
        if i == end || b[i] == b'.' {
            if i == part {
                return Err(VersionError::EmptyIdentifier { at: part });
            }
            let id = &s[part..i];
            if is_pre && id.len() > 1 && id.as_bytes()[0] == b'0'
                && id.bytes().all(|c| c.is_ascii_digit())
            {
                return Err(VersionError::LeadingZero { at: part });
            }
            out.push(id.to_string());
            if i == end {
                break;
            }
            part = i + 1;
            i += 1;
            continue;
        }
        let c = b[i];
        if !(c.is_ascii_alphanumeric() || c == b'-') {
            return Err(VersionError::InvalidChar { at: i });
        }
        i += 1;
    }
    Ok(())
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric(x);
        let yn = is_numeric(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| compare_pre(&a.pre, &b.pre))
}

    full: Option<Version>,

fn parse_full_at(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    if s.contains('-') || s.contains('+') {
        return Ok(Partial {
            nums: Vec::new(),
            full: Some(parse_full_at(s, at)?),
        });
    }
    let fields: Vec<&str> = s.split('.').collect();
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if fields.len() >= 3 && !has_wild {
        return Ok(Partial {
            nums: Vec::new(),
            full: Some(parse_full_at(s, at)?),
        });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !f.bytes().all(|b| b.is_ascii_digit())
                || (f.len() > 1 && f.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }
    Ok(Partial { nums, full: None })
}

fn padded(p: &Partial) -> Version {
    if let Some(v) = &p.full {
        return v.clone();
    }
    Version {
        major: *p.nums.get(0).unwrap_or(&0),
        minor: *p.nums.get(1).unwrap_or(&0),
        patch: *p.nums.get(2).unwrap_or(&0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn upper_increment(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut n = v.clone();
    n.pre = vec!["0".into()];
    n.build.clear();
    let target = match field {
        0 => &mut n.major,
        1 => &mut n.minor,
        _ => &mut n.patch,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    if field == 0 {
        n.minor = 0;
        n.patch = 0;
    } else if field == 1 {
        n.patch = 0;
    }
    Ok(n)
}

fn add_cmp(comps: &mut Vec<Comparator>, anchors: &mut Vec<(u64, u64, u64)>, op: Op, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() {
        anchors.push((v.major, v.minor, v.patch));
    }
    comps.push(Comparator { op, version: v });
}

fn parse_arm(tokens: &[(String, usize)]) -> Result<Arm, RangeError> {
    let mut comps = Vec::new();
    let mut anchors = Vec::new();
    let hyphens: Vec<usize> = tokens
        .iter()
        .filter(|(s, _)| s == "-")
        .map(|(_, at)| *at)
        .collect();

    if !hyphens.is_empty() {
        if tokens.len() != 3 || tokens[1].0 != "-" {
            return Err(RangeError::Syntax { at: hyphens[0] });
        }
        let (lo_s, lo_at) = &tokens[0];
        let (hi_s, hi_at) = &tokens[2];
        let lo = parse_partial(lo_s, *lo_at)?;
        let hi = parse_partial(hi_s, *hi_at)?;
        if lo.nums.is_empty() && lo.full.is_none() {
            return Err(RangeError::Syntax { at: *lo_at });
        }
        if hi.nums.is_empty() && hi.full.is_none() {
            return Err(RangeError::Syntax { at: *hi_at });
        }
        let lower = padded(&lo);
        add_cmp(&mut comps, &mut anchors, Op::Ge, lower, lo.full.as_ref().is_some_and(|v| !v.pre.is_empty()));
        if let Some(v) = hi.full {
            add_cmp(&mut comps, &mut anchors, Op::Le, v, true);
        } else {
            let v = padded(&hi);
            let field = match hi.nums.len() {
                1 => 0,
                _ => 1,
            };
            let upper = upper_increment(&v, field, *hi_at)?;
            add_cmp(&mut comps, &mut anchors, Op::Lt, upper, false);
        }
        return Ok(Arm { comps, anchors });
    }

    for (text, at) in tokens {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RangeError::Syntax { at: *at });
        }
        if matches!(bytes[0], b'<' | b'>' | b'=') {
            let (op, n) = if text.starts_with("<=") {
                (Op::Le, 2)
            } else if text.starts_with(">=") {
                (Op::Ge, 2)
            } else {
                (match bytes[0] {
                    b'<' => Op::Lt,
                    b'>' => Op::Gt,
                    _ => Op::Eq,
                }, 1)
            };
            let operand_at = at + n;
            if operand_at == at + text.len()
                || !text.as_bytes()[n].is_ascii_digit()
            {
                return Err(RangeError::Syntax { at: *at });
            }
            let v = parse_full_at(&text[n..], operand_at)?;
            add_cmp(&mut comps, &mut anchors, op, v, true);
            continue;
        }
        match bytes[0] {
            b'^' | b'~' => {
                let operand_at = at + 1;
                if operand_at >= at + text.len() {
                    return Err(RangeError::Syntax { at: *at });
                }
                let p = parse_partial(&text[1..], operand_at)?;
                if p.nums.is_empty() && p.full.is_none() {
                    return Err(RangeError::Syntax { at: *at });
                }
                let lower = padded(&p);
                let upper_field = if bytes[0] == b'^' {
                    if lower.major > 0 || p.nums.len() == 1 {
                        0
                    } else if lower.minor > 0 || p.nums.len() == 2 {
                        1
                    } else {
                        2
                    }
                } else if p.nums.len() == 1 {
                    0
                } else {
                    1
                };
                let upper = upper_increment(&lower, upper_field, *at)?;
                add_cmp(&mut comps, &mut anchors, Op::Ge, lower.clone(), !lower.pre.is_empty());
                add_cmp(&mut comps, &mut anchors, Op::Lt, upper, false);
            }
            b'*' | b'x' | b'X' | b'0'..=b'9' => {
                let p = parse_partial(text, *at)?;
                if let Some(v) = p.full {
                    add_cmp(&mut comps, &mut anchors, Op::Eq, v, true);
                } else if !p.nums.is_empty() {
                    let lower = padded(&p);
                    let upper_field = match p.nums.len() {
                        1 => 0,
                        2 => 1,
                        _ => 2,
                    };
                    add_cmp(&mut comps, &mut anchors, Op::Ge, lower, false);
                    if p.nums.len() < 3 {
                        let upper = upper_increment(&padded(&p), upper_field, *at)?;
                        add_cmp(&mut comps, &mut anchors, Op::Lt, upper, false);
                    } else {
                        let exact = padded(&p);
                        comps.pop();
                        add_cmp(&mut comps, &mut anchors, Op::Eq, exact, false);
                    }
                }
            }
            _ => return Err(RangeError::Syntax { at: *at }),
        }
    }
    Ok(Arm { comps, anchors })
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b = s.as_bytes();
    let mut tokens_by_arm: Vec<Vec<(String, usize)>> = vec![Vec::new()];
    let mut i = 0;
    let mut saw_any = false;
    while i < b.len() {
        if matches!(b[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
            continue;
        }
        let start = i;
        while i < b.len() && !matches!(b[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        let token = &s[start..i];
        saw_any = true;
        if token == "||" {
            if tokens_by_arm.last().unwrap().is_empty() {
                return Err(RangeError::Empty { at: start });
            }
            tokens_by_arm.push(Vec::new());
        } else {
            tokens_by_arm.last_mut().unwrap().push((token.to_string(), start));
        }
    }
    if !saw_any {
        return Err(RangeError::Empty { at: 0 });
    }
    if tokens_by_arm.last().unwrap().is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    let mut arms = Vec::new();
    for tokens in tokens_by_arm {
        arms.push(parse_arm(&tokens)?);
    }
    Ok(Range { arms })
}

fn cmp_matches(v: &Version, c: &Comparator) -> bool {
    let o = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => o == Ordering::Equal,
        Op::Lt => o == Ordering::Less,
        Op::Le => o != Ordering::Greater,
        Op::Gt => o == Ordering::Greater,
        Op::Ge => o != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !v.pre.is_empty()
        && !include_prerelease
        && !arm.anchors.contains(&(v.major, v.minor, v.patch))
    {
        return false;
    }
    arm.comps.iter().all(|c| cmp_matches(v, c))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (max && ord == Ordering::Greater) || (!max && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    let mut candidates = Vec::new();
    for arm in a.arms.iter().chain(b.arms.iter()) {
        for c in &arm.comps {
            candidates.push(c.version.clone());
        }
    }
    candidates.push(Version { major: 0, minor: 0, patch: 0, pre: Vec::new(), build: Vec::new() });
    candidates.into_iter().any(|v| matches_with(&v, a, include_prerelease) && matches_with(&v, b, include_prerelease))
}

pub fn canonical(r: &Range) -> String {
    let mut out = Vec::new();
    for arm in &r.arms {
        let mut lower: Option<(Version, bool)> = None;
        let mut upper: Option<(Version, bool)> = None;
        for c in &arm.comps {
            match c.op {
                Op::Eq => {
                    update_lower(&mut lower, c.version.clone(), true);
                    update_upper(&mut upper, c.version.clone(), true);
                }
                Op::Gt | Op::Ge => update_lower(&mut lower, c.version.clone(), c.op == Op::Ge),
                Op::Lt | Op::Le => update_upper(&mut upper, c.version.clone(), c.op == Op::Le),
            }
        }
        let low = lower.map_or("*".to_string(), |(v, inclusive)| format!("{}{}", if inclusive { ">=" } else { ">" }, display_version(&v)));
        let high = upper.map_or("*".to_string(), |(v, inclusive)| format!("{}{}", if inclusive { "<=" } else { "<" }, display_version(&v)));
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anchor_text = if anchors.is_empty() { "-".into() } else {
            anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
        };
        out.push(format!("[{low};{high};pre={anchor_text}]"));
    }
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".into() } else { out.join(" || ") }
}

fn update_lower(slot: &mut Option<(Version, bool)>, v: Version, inclusive: bool) {
    let replace = slot.as_ref().map_or(true, |(old, old_inc)| {
        let o = compare_versions(&v, old);
        o == Ordering::Greater || (o == Ordering::Equal && !inclusive && *old_inc)
    });
    if replace { *slot = Some((v, inclusive)); }
}

fn update_upper(slot: &mut Option<(Version, bool)>, v: Version, inclusive: bool) {
    let replace = slot.as_ref().map_or(true, |(old, old_inc)| {
        let o = compare_versions(&v, old);
        o == Ordering::Less || (o == Ordering::Equal && !inclusive && *old_inc)
    });
    if replace { *slot = Some((v, inclusive)); }
}

fn display_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() { s.push('-'); s.push_str(&v.pre.join(".")); }
    s
}

#[derive(Clone, Debug)]
pub enum SetExpr {
    Range(String),
    Union(Vec<SetExpr>),
    Intersection(Vec<SetExpr>),
}

#[derive(Clone, Debug)]
pub struct Constraint {
    pub label: String,
    pub expr: SetExpr,
}

#[derive(Clone, Debug)]
pub struct Package {
    pub name: String,
    pub candidates: Vec<String>,
    pub constraints: Vec<Constraint>,
}

#[derive(Clone, Copy, Debug)]
pub enum Preference {
    Highest,
    Lowest,
    StableFirst,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn eval_expr(expr: &SetExpr, v: &Version, include: bool, parsed: &[(Vec<usize>, Range)]) -> bool {
    match expr {
        SetExpr::Range(_) => parsed.iter().find(|(p, _)| p.is_empty()).is_some_and(|(_, r)| matches_with(v, r, include)),
        SetExpr::Union(xs) => xs.iter().any(|x| eval_expr(x, v, include, parsed)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| eval_expr(x, v, include, parsed)),
    }
}

fn collect_ranges(
    expr: &SetExpr,
    path: &mut Vec<usize>,
    package: &str,
    label: &str,
    parsed: &mut Vec<(Vec<usize>, Range)>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_string(),
                label: label.to_string(),
                path: path.clone(),
                error,
            })?;
            parsed.push((path.clone(), r));
        }
        SetExpr::Union(xs) | SetExpr::Intersection(xs) => {
            for (i, child) in xs.iter().enumerate() {
                path.push(i);
                collect_ranges(child, path, package, label, parsed)?;
                path.pop();
            }
        }
    }
    Ok(())
}

fn constraint_eval(c: &Constraint, v: &Version, include: bool, ranges: &[Range]) -> bool {
    fn run(expr: &SetExpr, v: &Version, include: bool, iter: &mut std::slice::Iter<'_, Range>) -> bool {
        match expr {
            SetExpr::Range(_) => iter.next().is_some_and(|r| matches_with(v, r, include)),
            SetExpr::Union(xs) => xs.iter().any(|x| run(x, v, include, iter)),
            SetExpr::Intersection(xs) => xs.iter().all(|x| run(x, v, include, iter)),
        }
    }
    run(&c.expr, v, include, &mut ranges.iter())
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }

    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }

    let mut parsed_candidates = Vec::new();
    for p in packages {
        let mut vs = Vec::new();
        for (index, s) in p.candidates.iter().enumerate() {
            vs.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index, error
            })?);
        }
        parsed_candidates.push(vs);
    }

    let mut all_ranges: Vec<Vec<Vec<Range>>> = Vec::new();
    for p in packages {
        let mut pc = Vec::new();
        for c in &p.constraints {
            let mut ranges = Vec::new();
            collect_ranges(&c.expr, &mut Vec::new(), &p.name, &c.label, &mut Vec::new())?;
            let mut flat = Vec::new();
            fn flatten(expr: &SetExpr, out: &mut Vec<Range>, p: &Package, label: &str) -> Result<(), SelectError> {
                match expr {
                    SetExpr::Range(s) => out.push(parse_range(s).map_err(|error| SelectError::InvalidRange {
                        package: p.name.clone(), label: label.to_string(), path: Vec::new(), error
                    })?),
                    SetExpr::Union(xs) | SetExpr::Intersection(xs) => for x in xs { flatten(x, out, p, label)?; },
                }
                Ok(())
            }
            flatten(&c.expr, &mut flat, p, &c.label)?;
            pc.push(flat);
        }
        all_ranges.push(pc);
    }

    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let mut qualifying = Vec::new();
        for (i, v) in parsed_candidates[pi].iter().enumerate() {
            if p.constraints.iter().enumerate().all(|(ci, c)| constraint_eval(c, v, include_prerelease, &all_ranges[pi][ci])) {
                qualifying.push(i);
            }
        }
        if qualifying.is_empty() {
            let labels = if p.candidates.is_empty() {
                Vec::new()
            } else {
                minimum_conflict(p, &parsed_candidates[pi], &all_ranges[pi], include_prerelease)
            };
            return Err(SelectError::NoSolution { package: p.name.clone(), labels });
        }
        let chosen = match preference {
            Preference::Highest => choose_extreme(&qualifying, &parsed_candidates[pi], true, false),
            Preference::Lowest => choose_extreme(&qualifying, &parsed_candidates[pi], false, false),
            Preference::StableFirst => {
                let stable: Vec<usize> = qualifying.iter().copied().filter(|&i| parsed_candidates[pi][i].pre.is_empty()).collect();
                if stable.is_empty() { choose_extreme(&qualifying, &parsed_candidates[pi], true, false) }
                else { choose_extreme(&stable, &parsed_candidates[pi], true, false) }
            }
        };
        selections.push(Selection { package: p.name.clone(), version: p.candidates[chosen].clone() });
    }
    Ok(selections)
}

fn choose_extreme(ix: &[usize], vs: &[Version], max: bool, _stable: bool) -> usize {
    let mut best = ix[0];
    for &i in ix.iter().skip(1) {
        let o = compare_versions(&vs[i], &vs[best]);
        if (max && o == Ordering::Greater) || (!max && o == Ordering::Less) {
            best = i;
        }
    }
    best
}

fn minimum_conflict(p: &Package, vs: &[Version], ranges: &[Vec<Range>], include: bool) -> Vec<String> {
    let n = p.constraints.len();
    for k in 1..=n {
        let mut best: Option<Vec<String>> = None;
        let mut chosen = Vec::new();
        fn search(
            start: usize, k: usize, chosen: &mut Vec<usize>, best: &mut Option<Vec<String>>,
            p: &Package, vs: &[Version], ranges: &[Vec<Range>], include: bool,
        ) {
            if chosen.len() == k {
                let satisfiable = vs.iter().enumerate().any(|(vi, v)| {
                    chosen.iter().all(|&ci| constraint_eval(&p.constraints[ci], v, include, &ranges[ci]))
                });
                if !satisfiable {
                    let mut labels: Vec<String> = chosen.iter().map(|&i| p.constraints[i].label.clone()).collect();
                    labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
                    if best.as_ref().is_none_or(|b| labels.iter().map(String::as_str).cmp(b.iter().map(String::as_str)) == Ordering::Less) {
                        *best = Some(labels);
                    }
                }
                return;
            }
            let needed = k - chosen.len();
            if p.constraints.len().saturating_sub(start) < needed { return; }
            for i in start..p.constraints.len() {
                chosen.push(i);
                search(i + 1, k, chosen, best, p, vs, ranges, include);
                chosen.pop();
            }
        }
        search(0, k, &mut chosen, &mut best, p, vs, ranges, include);
        if let Some(labels) = best { return labels; }
    }
    Vec::new()
}
