use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return Err(if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let value = s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })?;
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < bytes.len()
            && bytes[p] != b'.'
            && !(pre && bytes[p] == b'+')
        {
            let b = bytes[p];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if pre && id.len() > 1 && id.as_bytes()[0] == b'0' && id.bytes().all(is_digit) {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_owned());
        if p == bytes.len() || (pre && bytes[p] == b'+') {
            return Ok(out);
        }
        // Here the separator is a dot.
        p += 1;
        if p == bytes.len() || (pre && bytes[p] == b'+') {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        return Err(if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            VersionError::MissingCore { at: pos }
        } else {
            VersionError::InvalidChar { at: pos }
        });
    }
    pos += 1;
    let minor = parse_core(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        return Err(if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            VersionError::MissingCore { at: pos }
        } else {
            VersionError::InvalidChar { at: pos }
        });
    }
    pos += 1;
    let patch = parse_core(s, &mut pos)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < s.len() && s.as_bytes()[pos] == b'-' {
        pos += 1;
        pre = parse_identifiers(s, pos, true)?;
        // Locate the plus delimiter if present.
        while pos < s.len() && s.as_bytes()[pos] != b'+' {
            pos += 1;
        }
    } else if pos < s.len() && s.as_bytes()[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }
    if pos < s.len() {
        // Only plus may remain after the patch or prerelease.
        if s.as_bytes()[pos] != b'+' {
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        build = parse_identifiers(s, pos, false)?;
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_numeric_str(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    if a.is_empty() {
        return if b.is_empty() { Ordering::Equal } else { Ordering::Greater };
    }
    if b.is_empty() {
        return Ordering::Less;
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => cmp_numeric_str(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug)]
pub struct Range {

#[derive(Clone, Debug)]
struct Bound {

    inclusive: bool,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparators: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn split_tokens(s: &str, base: usize) -> Vec<(usize, &str)> {
    let mut out = Vec::new();
    let bytes = s.as_bytes();
    let mut p = 0;
    while p < bytes.len() {
        while p < bytes.len() && is_ws(bytes[p]) {
            p += 1;
        }
        if p == bytes.len() {
            break;
        }
        let start = p;
        while p < bytes.len() && !is_ws(bytes[p]) {
            p += 1;
        }
        out.push((base + start, &s[start..p]));
    }
    out
}

fn version_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn core_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn plain_version(major: u64, minor: u64, patch: u64) -> Version {
    Version { major, minor, patch, pre: Vec::new(), build: Vec::new() }
}

fn endpoint(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: vec!["0".to_owned()],
        build: Vec::new(),
    }
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    numeric_count: usize,
    full: Option<Version>,
}

fn parse_partial(text: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix = text.contains('-') || text.contains('+');
    let fields: Vec<&str> = text.split('.').collect();
    let no_wild = fields.iter().all(|f| !matches!(*f, "*" | "x" | "X"));
    if has_suffix || (fields.len() >= 3 && no_wild) {
        let v = version_operand(text, at)?;
        return Ok(Partial {
            fields: vec![Some(v.major), Some(v.minor), Some(v.patch)],
            numeric_count: 3,
            full: Some(v),
        });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut values = Vec::new();
    let mut wild_seen = false;
    for f in fields {
        if f.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(f, "*" | "x" | "X") {
            wild_seen = true;
            values.push(None);
        } else {
            if wild_seen
                || !f.bytes().all(is_digit)
                || (f.len() > 1 && f.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            values.push(Some(n));
        }
    }
    let numeric_count = values.iter().take_while(|x| x.is_some()).count();
    if values.iter().skip(numeric_count).any(|x| x.is_some()) {
        return Err(RangeError::Syntax { at });
    }
    Ok(Partial { fields: values, numeric_count, full: None })
}

fn padded(p: &Partial) -> (u64, u64, u64) {
    (
        p.fields.first().and_then(|x| *x).unwrap_or(0),
        p.fields.get(1).and_then(|x| *x).unwrap_or(0),
        p.fields.get(2).and_then(|x| *x).unwrap_or(0),
    )
}

fn increment_upper(
    tuple: (u64, u64, u64),
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let (mut a, mut b, mut c) = tuple;
    match field {
        0 => a = a.checked_add(1).ok_or(RangeError::Syntax { at })?,
        1 => b = b.checked_add(1).ok_or(RangeError::Syntax { at })?,
        _ => c = c.checked_add(1).ok_or(RangeError::Syntax { at })?,
    }
    match field {
        0 => Ok(endpoint(a, 0, 0)),
        1 => Ok(endpoint(a, b, 0)),
        _ => Ok(endpoint(a, b, c)),
    }
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version) {
    if matches!(op, Op::Eq | Op::Ge) && !version.pre.is_empty() {
        arm.anchors.push(core_tuple(&version));
    }
    arm.comparators.push(Comparator { op, version });
}

fn parse_operator_token(
    token: &str,
    at: usize,
) -> Result<Option<(Op, &str, usize)>, RangeError> {
    let candidates = [("<=", Op::Le), (">=", Op::Ge), ("=", Op::Eq), ("<", Op::Lt), (">", Op::Gt)];
    for (prefix, op) in candidates {
        if let Some(rest) = token.strip_prefix(prefix) {
            if rest.is_empty() || !rest.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at });
            }
            return Ok(Some((op, rest, at + prefix.len())));
        }
    }
    Ok(None)
}

fn add_partial_expansion(
    arm: &mut Arm,
    token: &str,
    at: usize,
    kind: char,
) -> Result<(), RangeError> {
    let operand_at = at + 1;
    let p = parse_partial(&token[1..], operand_at)?;
    if p.numeric_count == 0 {
        return Err(RangeError::Syntax { at });
    }
    let lower_tuple = padded(&p);
    let mut lower = plain_version(lower_tuple.0, lower_tuple.1, lower_tuple.2);
    if let Some(full) = p.full.clone() {
        lower = full;
    }
    add_cmp(arm, Op::Ge, lower.clone());
    let k = p.numeric_count;
    let field = if kind == '^' {
        let (a, b, _) = lower_tuple;
        if a > 0 || k == 1 { 0 } else if b > 0 || k == 2 { 1 } else { 2 }
    } else if k == 1 {
        0
    } else {
        1
    };
    let upper = increment_upper(lower_tuple, field, at)?;
    arm.comparators.push(Comparator { op: Op::Lt, version: upper });
    Ok(())
}

fn add_bare(
    arm: &mut Arm,
    token: &str,
    at: usize,
) -> Result<(), RangeError> {
    let p = parse_partial(token, at)?;
    if let Some(v) = p.full {
        add_cmp(arm, Op::Eq, v);
    } else if p.numeric_count == 0 {
        // Wildcard operands match all releases.
    } else if p.numeric_count == 3 {
        let tuple = padded(&p);
        add_cmp(arm, Op::Ge, plain_version(tuple.0, tuple.1, tuple.2));
        let upper = increment_upper(tuple, 2, at)?;
        arm.comparators.push(Comparator { op: Op::Lt, version: upper });
    } else {
        let tuple = padded(&p);
        add_cmp(arm, Op::Ge, plain_version(tuple.0, tuple.1, tuple.2));
        let upper = increment_upper(tuple, p.numeric_count - 1, at)?;
        arm.comparators.push(Comparator { op: Op::Lt, version: upper });
    }
    Ok(())
}

fn parse_hyphen_arm(tokens: &[(usize, &str)]) -> Result<Arm, RangeError> {
    let first_minus = tokens.iter().find(|(_, t)| *t == &"-").unwrap().0;
    if tokens.len() != 3 || tokens[1].1 != "-" {
        return Err(RangeError::Syntax { at: first_minus });
    }
    let (la, lt) = tokens[0];
    let (ua, ut) = tokens[2];
    let lower = parse_partial(lt, la)?;
    let upper = parse_partial(ut, ua)?;
    if lower.numeric_count == 0 {
        return Err(RangeError::Syntax { at: la });
    }
    if upper.numeric_count == 0 {
        return Err(RangeError::Syntax { at: ua });
    }
    let mut arm = Arm { comparators: Vec::new(), anchors: Vec::new() };
    let ltup = padded(&lower);
    let lv = lower.full.clone().unwrap_or_else(|| plain_version(ltup.0, ltup.1, ltup.2));
    add_cmp(&mut arm, Op::Ge, lv);
    if let Some(uv) = upper.full {
        add_cmp(&mut arm, Op::Le, uv);
    } else {
        let utup = padded(&upper);
        let field = if upper.numeric_count == 1 { 0 } else { 1 };
        let bound = increment_upper(utup, field, ua)?;
        arm.comparators.push(Comparator { op: Op::Lt, version: bound });
    }
    Ok(arm)
}

fn parse_arm(tokens: &[(usize, &str)]) -> Result<Arm, RangeError> {
    if tokens.iter().any(|(_, t)| *t == "-") {
        return parse_hyphen_arm(tokens);
    }
    let mut arm = Arm { comparators: Vec::new(), anchors: Vec::new() };
    for &(at, token) in tokens {
        if token.starts_with('^') || token.starts_with('~') {
            let kind = token.as_bytes()[0] as char;
            add_partial_expansion(&mut arm, token, at, kind)?;
            continue;
        }
        if let Some((op, operand, operand_at)) = parse_operator_token(token, at)? {
            let version = version_operand(operand, operand_at)?;
            add_cmp(&mut arm, op, version);
            continue;
        }
        let first = token.as_bytes()[0];
        if !(first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X')) {
            return Err(RangeError::Syntax { at });
        }
        add_bare(&mut arm, token, at)?;
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = split_tokens(s, 0);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].1 == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: tokens[i].0 });
            }
            arms.push(parse_arm(&current)?);
            current.clear();
            if i + 1 == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            current.push(tokens[i]);
        }
        i += 1;
    }
    if !current.is_empty() {
        arms.push(parse_arm(&current)?);
    }
    Ok(Range { arms })
}

fn cmp_one(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !arm.comparators.iter().all(|c| cmp_one(v, c)) {
        return false;
    }
    include_prerelease
        || v.pre.is_empty()
        || arm.anchors.contains(&core_tuple(v))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, text) in candidates.iter().enumerate() {
        let version = parse_version(text).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (max && ord == Ordering::Greater) || (!max && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn tuple_cmp(a: (u64, u64, u64), b: (u64, u64, u64)) -> Ordering {
    a.cmp(&b)
}

fn compare_optional_lower(a: &Option<Bound>, b: &Option<Bound>) -> Ordering {
    match (a, b) {
        (None, None) => Ordering::Equal,
        (None, Some(_)) => Ordering::Less,
        (Some(_), None) => Ordering::Greater,
        (Some(x), Some(y)) => compare_versions(&x.version, &y.version)
            .then_with(|| x.inclusive.cmp(&y.inclusive)),
    }
}

fn compare_optional_upper(a: &Option<Bound>, b: &Option<Bound>) -> Ordering {
    match (a, b) {
        (None, None) => Ordering::Equal,
        (None, Some(_)) => Ordering::Greater,
        (Some(_), None) => Ordering::Less,
        (Some(x), Some(y)) => compare_versions(&x.version, &y.version)
            .then_with(|| y.inclusive.cmp(&x.inclusive)),
    }
}

fn arm_bounds(arm: &Arm) -> (Option<Bound>, Option<Bound>, Option<Version>) {
    let mut lower = None;
    let mut upper = None;
    let mut equality = None;
    for c in &arm.comparators {
        match c.op {
            Op::Eq => equality = Some(c.version.clone()),
            Op::Gt | Op::Ge => {
                let b = Bound { version: c.version.clone(), inclusive: matches!(c.op, Op::Ge) };
                if lower.as_ref().map_or(true, |old| compare_optional_lower(&Some(b.clone()), &Some(old.clone())) == Ordering::Greater) {
                    lower = Some(b);
                }
            }
            Op::Lt | Op::Le => {
                let b = Bound { version: c.version.clone(), inclusive: matches!(c.op, Op::Le) };
                if upper.as_ref().map_or(true, |old| compare_optional_upper(&Some(b.clone()), &Some(old.clone())) == Ordering::Less) {
                    upper = Some(b);
                }
            }
        }
    }
    if let Some(v) = equality {
        lower = Some(Bound { version: v.clone(), inclusive: true });
        upper = Some(Bound { version: v.clone(), inclusive: true });
    }
    (lower, upper, None)
}

fn next_pre(v: &Version) -> Version {
    let mut n = v.clone();
    n.build.clear();
    n.pre.push("0".to_owned());
    n
}

fn next_stable(v: &Version) -> Option<Version> {
    if v.patch < u64::MAX {
        Some(endpoint(v.major, v.minor, v.patch + 1))
    } else if v.minor < u64::MAX {
        Some(endpoint(v.major, v.minor + 1, 0))
    } else if v.major < u64::MAX {
        Some(endpoint(v.major + 1, 0, 0))
    } else {
        None
    }
}

fn smallest_at_tuple(tuple: (u64, u64, u64)) -> Version {
    endpoint(tuple.0, tuple.1, tuple.2)
}

fn arm_has_any(arm: &Arm) -> bool {
    let (lower, upper, _) = arm_bounds(arm);
    let mut candidates = Vec::new();
    if let Some(l) = &lower {
        candidates.push(l.version.clone());
        if !l.inclusive {
            if !l.version.pre.is_empty() {
                candidates.push(next_pre(&l.version));
            } else if let Some(n) = next_stable(&l.version) {
                candidates.push(n);
            }
        }
    } else {
        candidates.push(plain_version(0, 0, 0));
    }
    if let Some(u) = &upper {
        candidates.push(u.version.clone());
        if !u.inclusive && !u.version.pre.is_empty() {
            // A strict upper prerelease may still have an earlier witness.
            candidates.push(plain_version(u.version.major, u.version.minor, u.version.patch));
        }
    }
    candidates.into_iter().any(|v| arm_matches(&v, arm, false))
}

fn arm_intersects(a: &Arm, b: &Arm, include_prerelease: bool) -> bool {
    let (al, au, _) = arm_bounds(a);
    let (bl, bu, _) = arm_bounds(b);
    let mut candidates = Vec::new();
    for l in [&al, &bl].into_iter().flatten() {
        candidates.push(l.version.clone());
        if !l.inclusive {
            if !l.version.pre.is_empty() {
                candidates.push(next_pre(&l.version));
            } else if let Some(n) = next_stable(&l.version) {
                candidates.push(n);
            }
        }
    }
    for u in [&au, &bu].into_iter().flatten() {
        candidates.push(u.version.clone());
    }
    if candidates.is_empty() {
        candidates.push(plain_version(0, 0, 0));
    }
    candidates.into_iter().any(|v| {
        arm_matches(&v, a, include_prerelease) && arm_matches(&v, b, include_prerelease)
    })
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    a.arms.iter().any(|x| b.arms.iter().any(|y| arm_intersects(x, y, include_prerelease)))
}

fn version_display(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn display_bound(bound: &Option<Bound>, lower: bool) -> String {
    match bound {
        None => "*".to_owned(),
        Some(b) => {
            let op = match (lower, b.inclusive) {
                (true, true) => ">=",
                (true, false) => ">",
                (false, true) => "<=",
                (false, false) => "<",
            };
            format!("{}{}", op, version_display(&b.version))
        }
    }
}

pub fn canonical(r: &Range) -> String {
    let mut arms = Vec::new();
    for arm in &r.arms {
        if !arm_has_any(arm) {
            continue;
        }
        let (lower, upper, _) = arm_bounds(arm);
        let mut anchors = arm.anchors.clone();
        anchors.sort_unstable();
        anchors.dedup();
        let anchor_text = if anchors.is_empty() {
            "-".to_owned()
        } else {
            anchors.iter().map(|(a, b, c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
        };
        arms.push(format!(
            "[{};{};pre={}]",
            display_bound(&lower, true),
            display_bound(&upper, false),
            anchor_text
        ));
    }
    arms.sort();
    arms.dedup();
    if arms.is_empty() { "EMPTY".to_owned() } else { arms.join(" || ") }
}

#[derive(Clone, Debug)]
pub enum SetExpr {
    Range(String),
    Union(Vec<SetExpr>),
    Intersection(Vec<SetExpr>),
}

#[derive(Clone, Debug)]
pub struct Constraint {
    pub label: String,
    pub expr: SetExpr,
}

#[derive(Clone, Debug)]
pub struct Package {
    pub name: String,
    pub candidates: Vec<String>,
    pub constraints: Vec<Constraint>,
}

#[derive(Clone, Copy, Debug)]
pub enum Preference {
    Highest,
    Lowest,
    StableFirst,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(text) => {
            parse_range(text).map(|_| ()).map_err(|error| SelectError::InvalidRange {
                package: package.to_owned(),
                label: label.to_owned(),
                path: path.clone(),
                error,
            })
        }
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr(child, package, label, path)?;
                path.pop();
            }
            Ok(())
        }
    }
}

fn expr_matches(expr: &SetExpr, v: &Version, include_prerelease: bool) -> bool {
    match expr {
        SetExpr::Range(text) => parse_range(text)
            .map(|r| matches_with(v, &r, include_prerelease))
            .unwrap_or(false),
        SetExpr::Union(children) => children.iter().any(|e| expr_matches(e, v, include_prerelease)),
        SetExpr::Intersection(children) => children.iter().all(|e| expr_matches(e, v, include_prerelease)),
    }
}

fn conflict_labels(
    package: &Package,
    candidates: &[Version],
    include_prerelease: bool,
) -> Vec<String> {
    let n = package.constraints.len();
    if candidates.is_empty() {
        return Vec::new();
    }
    for size in 1..=n {
        let mut best: Option<Vec<String>> = None;
        let mut chosen = Vec::new();
        fn visit(
            start: usize,
            size: usize,
            chosen: &mut Vec<usize>,
            best: &mut Option<Vec<String>>,
            package: &Package,
            candidates: &[Version],
            include_prerelease: bool,
        ) {
            if chosen.len() == size {
                let satisfiable = candidates.iter().any(|v| {
                    chosen.iter().all(|&i| {
                        expr_matches(&package.constraints[i].expr, v, include_prerelease)
                    })
                });
                if !satisfiable {
                    let mut labels = chosen.iter().map(|&i| package.constraints[i].label.clone()).collect::<Vec<_>>();
                    labels.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
                    if best.as_ref().map_or(true, |old| {
                        labels.iter().map(|s| s.as_bytes()).cmp(old.iter().map(|s| s.as_bytes())) == Ordering::Less
                    }) {
                        *best = Some(labels);
                    }
                }
                return;
            }
            let need = size - chosen.len();
            if package.constraints.len().saturating_sub(start) < need {
                return;
            }
            for i in start..=package.constraints.len() - need {
                chosen.push(i);
                visit(i + 1, size, chosen, best, package, candidates, include_prerelease);
                chosen.pop();
            }
        }
        visit(0, size, &mut chosen, &mut best, package, candidates, include_prerelease);
        if let Some(labels) = best {
            return labels;
        }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }
    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }
    let mut parsed_by_package = Vec::new();
    for p in packages {
        let mut parsed = Vec::new();
        for (index, candidate) in p.candidates.iter().enumerate() {
            parsed.push(parse_version(candidate).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(),
                index,
                error,
            })?);
        }
        parsed_by_package.push(parsed);
    }
    for p in packages {
        for c in &p.constraints {
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new())?;
        }
    }

    let mut result = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let parsed = &parsed_by_package[pi];
        let mut qualifying = Vec::new();
        for (i, v) in parsed.iter().enumerate() {
            if p.constraints.iter().all(|c| expr_matches(&c.expr, v, include_prerelease)) {
                qualifying.push(i);
            }
        }
        if qualifying.is_empty() {
            let labels = conflict_labels(p, parsed, include_prerelease);
            return Err(SelectError::NoSolution { package: p.name.clone(), labels });
        }
        let is_better = |i: usize, j: usize| {
            let a = &parsed[i];
            let b = &parsed[j];
            match preference {
                Preference::Highest => compare_versions(a, b) == Ordering::Greater,
                Preference::Lowest => compare_versions(a, b) == Ordering::Less,
                Preference::StableFirst => {
                    match (a.pre.is_empty(), b.pre.is_empty()) {
                        (true, false) => true,
                        (false, true) => false,
                        _ => compare_versions(a, b) == Ordering::Greater,
                    }
                }
            }
        };
        let mut best = qualifying[0];
        for &i in qualifying.iter().skip(1) {
            if is_better(i, best) {
                best = i;
            }
        }
        result.push(Selection { package: p.name.clone(), version: p.candidates[best].clone() });
    }
    Ok(result)
}
