use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_num(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    let mut end = start;
    while end < bytes.len() && is_digit(bytes[end]) {
        end += 1;
    }
    if end == start {
        return Err(match bytes.get(start) {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: start },
            _ => VersionError::InvalidChar { at: start },
        });
    }
    let part = &s[start..end];
    if part.len() > 1 && part.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    let n = part.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })?;
    *pos = end;
    Ok(n)
}

fn parse_identifiers(s: &str, start: usize, is_pre: bool) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut ids = Vec::new();
    let mut pos = start;
    loop {
        let end = if is_pre {
            let mut e = pos;
            while e < bytes.len() && bytes[e] != b'.' && bytes[e] != b'+' {
                e += 1;
            }
            e
        } else {
            let mut e = pos;
            while e < bytes.len() && bytes[e] != b'.' {
                e += 1;
            }
            e
        };
        if end == pos {
            return Err(VersionError::EmptyIdentifier { at: pos });
        }
        for i in pos..end {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
        }
        let value = &s[pos..end];
        if is_pre && value.len() > 1 && value.starts_with('0') && value.bytes().all(is_digit) {
            return Err(VersionError::LeadingZero { at: pos });
        }
        ids.push(value.to_string());
        if end == bytes.len() {
            break;
        }
        if bytes[end] == b'+' && is_pre {
            break;
        }
        pos = end + 1;
        if pos == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: pos });
        }
    }
    Ok(ids)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_num(s, &mut pos)?;
    for expected in 0..2 {
        if pos >= s.len() || s.as_bytes()[pos] != b'.' {
            return Err(match s.as_bytes().get(pos) {
                None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
                _ => VersionError::InvalidChar { at: pos },
            });
        }
        pos += 1;
        if expected == 0 {
            break;
        }
    }
    let minor = parse_core_num(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        return Err(match s.as_bytes().get(pos) {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
            _ => VersionError::InvalidChar { at: pos },
        });
    }
    pos += 1;
    let patch = parse_core_num(s, &mut pos)?;
    if pos < s.len() && s.as_bytes()[pos] != b'-' && s.as_bytes()[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < s.len() && s.as_bytes()[pos] == b'-' {
        pos += 1;
        pre = parse_identifiers(s, pos, true)?;
        while pos < s.len() && s.as_bytes()[pos] != b'+' {
            pos += 1;
        }
    }
    if pos < s.len() && s.as_bytes()[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(s, pos, false)?;
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_id(a: &str, b: &str) -> Ordering {
    match (numeric_id(a), numeric_id(b)) {
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let c = cmp_id(x, y);
                    if c != Ordering::Equal {
                        return c;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

pub struct Range {

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Copy, Debug)]
enum Op { Eq, Lt, Le, Gt, Ge }

#[derive(Clone, Debug)]
struct Arm {
    comparisons: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

fn core(v: &Version) -> (u64, u64, u64) { (v.major, v.minor, v.patch) }

fn vcore(a: u64, b: u64, c: u64, pre: Vec<String>) -> Version {
    Version { major:a, minor:b, patch:c, pre, build:Vec::new() }
}

fn operand_full(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(core(&version));
    }
    arm.comparisons.push(Comparator { op, version, anchor });
}

fn cmp_ok(v: &Version, c: &Comparator) -> bool {
    let o = compare_versions(v, &c.version);
    match c.op { Op::Eq => o == Ordering::Equal, Op::Lt => o == Ordering::Less,
        Op::Le => o != Ordering::Greater, Op::Gt => o == Ordering::Greater,
        Op::Ge => o != Ordering::Less }
}

fn parse_partial(s: &str) -> Result<(Vec<Option<u64>>, Option<Version>), ()> {
    if s.contains('-') || s.contains('+') {
        return parse_version(s).map(|v| (vec![Some(v.major),Some(v.minor),Some(v.patch)],Some(v))).map_err(|_| ());
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() >= 3 && !fields.iter().any(|x| matches!(*x, "*"|"x"|"X")) {
        return parse_version(s).map(|v| (vec![Some(v.major),Some(v.minor),Some(v.patch)],Some(v))).map_err(|_| ());
    }
    if fields.is_empty() || fields.len() > 3 { return Err(()); }
    let mut out = Vec::new();
    let mut wild = false;
    for f in fields {
        if f.is_empty() { return Err(()); }
        if matches!(f, "*"|"x"|"X") {
            wild = true;
            out.push(None);
        } else {
            if wild || !f.bytes().all(is_digit) || (f.len()>1 && f.starts_with('0')) { return Err(()); }
            out.push(Some(f.parse::<u64>().map_err(|_| ())?));
        }
    }
    Ok((out, None))
}

fn padded(fields: &[Option<u64>]) -> (u64,u64,u64,usize) {
    let k = fields.iter().take_while(|x| x.is_some()).count();
    let a = fields.get(0).and_then(|x| *x).unwrap_or(0);
    let b = fields.get(1).and_then(|x| *x).unwrap_or(0);
    let c = fields.get(2).and_then(|x| *x).unwrap_or(0);
    (a,b,c,k)
}

fn bump(a:u64,b:u64,c:u64,which:usize, at:usize) -> Result<Version,RangeError> {
    let mut x=a; let mut y=b; let mut z=c;
    match which {
        0 => { x=x.checked_add(1).ok_or(RangeError::Syntax{at})?; y=0; z=0; }
        1 => { y=y.checked_add(1).ok_or(RangeError::Syntax{at})?; z=0; }
        _ => { z=z.checked_add(1).ok_or(RangeError::Syntax{at})?; }
    }
    Ok(vcore(x,y,z,vec!["0".into()]))
}

fn parse_bound_operand(s:&str, at:usize, arm:&mut Arm, kind:u8) -> Result<(Vec<Option<u64>>,Option<Version>),RangeError> {
    let p=parse_partial(s).map_err(|_| RangeError::Syntax{at})?;
    if kind != 0 && padded(&p.0).3 == 0 { return Err(RangeError::Syntax{at}); }
    Ok(p)
}

fn parse_arm(tokens:&[(usize,&str)], arm_start:usize, arm_end:usize) -> Result<Arm,RangeError> {
    let mut arm=Arm{comparisons:Vec::new(),anchors:Vec::new()};
    if tokens.is_empty() { return Err(RangeError::Empty{at:arm_start}); }
    let hyphens:Vec<usize>=tokens.iter().enumerate().filter(|(_,(_,t))| *t=="-").map(|(i,_)|i).collect();
    if !hyphens.is_empty() {
        let h=hyphens[0];
        if tokens.len()!=3 || h!=1 { return Err(RangeError::Syntax{at:tokens[h].0}); }
        let (la,ls)=tokens[0]; let (ua,us)=tokens[2];
        let lp=parse_bound_operand(ls,la,&mut arm,1)?;
        let up=parse_bound_operand(us,ua,&mut arm,1)?;
        let (a,b,c,k)=padded(&lp.0);
        let lower=lp.1.unwrap_or_else(||vcore(a,b,c,Vec::new()));
        add_cmp(&mut arm,Op::Ge,lower,true);
        if let Some(full)=up.1 { add_cmp(&mut arm,Op::Le,full,true); }
        else {
            let (x,y,z,n)=padded(&up.0);
            let upper=bump(x,y,z,if n==1 {0}else{1},ua)?;
            add_cmp(&mut arm,Op::Lt,upper,false);
        }
        return Ok(arm);
    }
    for &(at,tok) in tokens {
        let bytes=tok.as_bytes();
        if tok.starts_with('^') || tok.starts_with('~') {
            let caret=bytes[0]==b'^'; let os=at+1; let s=&tok[1..];
            let p=parse_bound_operand(s,os,&mut arm,1)?;
            let (a,b,c,k)=padded(&p.0);
            let lower=p.1.unwrap_or_else(||vcore(a,b,c,Vec::new()));
            add_cmp(&mut arm,Op::Ge,lower,true);
            let which=if caret { if a>0 || k==1 {0} else if b>0 || k==2 {1} else {2} } else if k==1 {0} else {1};
            let upper=bump(a,b,c,which,os)?;
            add_cmp(&mut arm,Op::Lt,upper,false);
        } else if matches!(bytes.first(),Some(b'<'|b'>'|b'=')) {
            let (op,n)=if tok.starts_with(">="){(Op::Ge,2)} else if tok.starts_with("<="){(Op::Le,2)}
                else if tok.starts_with(">"){(Op::Gt,1)} else if tok.starts_with("<"){(Op::Lt,1)} else {(Op::Eq,1)};
            if tok.len()==n || !tok.as_bytes()[n].is_ascii_digit() { return Err(RangeError::Syntax{at}); }
            let v=operand_full(&tok[n..],at+n)?;
            add_cmp(&mut arm,op,v,true);
        } else if tok.starts_with('*') || tok.starts_with('x') || tok.starts_with('X') || bytes.first().is_some_and(u8::is_ascii_digit) {
            let p=parse_bound_operand(tok,at,&mut arm,0)?;
            if let Some(v)=p.1 { add_cmp(&mut arm,Op::Eq,v,true); }
            else {
                let (a,b,c,k)=padded(&p.0);
                if k==1 {
                    add_cmp(&mut arm,Op::Ge,vcore(a,0,0,Vec::new()),false);
                    add_cmp(&mut arm,Op::Lt,bump(a,0,0,0,at)?,false);
                } else if k==2 {
                    add_cmp(&mut arm,Op::Ge,vcore(a,b,0,Vec::new()),false);
                    add_cmp(&mut arm,Op::Lt,bump(a,b,0,1,at)?,false);
                }
            }
        } else { return Err(RangeError::Syntax{at}); }
    }
    let _=arm_end;
    Ok(arm)
}

fn ws(b:u8)->bool { matches!(b,b' '|b'\t'|b'\r'|b'\n') }

pub fn parse_range(s:&str)->Result<Range,RangeError>{
    let bytes=s.as_bytes(); let mut arms=Vec::new(); let mut pos=0; let mut pending_empty=None;
    while pos<bytes.len() && ws(bytes[pos]) {pos+=1;}
    if pos==bytes.len(){return Err(RangeError::Empty{at:0});}
    let mut arm_start=pos;
    loop {
        let mut tokens=Vec::new();
        while pos<bytes.len() {
            while pos<bytes.len() && ws(bytes[pos]) {pos+=1;}
            if pos>=bytes.len(){break;}
            if pos+1<bytes.len() && &s[pos..pos+2]=="||" {
                if tokens.is_empty(){return Err(RangeError::Empty{at:pos});}
                let arm=parse_arm(&tokens,arm_start,pos)?;
                arms.push(arm);
                pos+=2;
                while pos<bytes.len() && ws(bytes[pos]) {pos+=1;}
                if pos==bytes.len(){return Err(RangeError::Empty{at:bytes.len()});}
                arm_start=pos;
                break;
            }
            let start=pos;
            while pos<bytes.len() && !ws(bytes[pos]) {pos+=1;}
            tokens.push((start,&s[start..pos]));
        }
        if pos>=bytes.len() {
            let arm=parse_arm(&tokens,arm_start,bytes.len())?;
            arms.push(arm);
            break;
        }
    }
    let _=pending_empty;
    Ok(Range{arms})
}

fn arm_match(v:&Version,a:&Arm,include:bool)->bool{
    a.comparisons.iter().all(|c|cmp_ok(v,c)) &&
        (include || v.pre.is_empty() || a.anchors.contains(&core(v)))
}
pub fn matches_with(v:&Version,r:&Range,include_prerelease:bool)->bool{
    r.arms.iter().any(|a|arm_match(v,a,include_prerelease))
}
pub fn matches(v:&Version,r:&Range)->bool{matches_with(v,r,false)}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index:usize, pub error:VersionError }
pub fn max_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{ extrema(candidates,r,true) }
pub fn min_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{ extrema(candidates,r,false) }
fn extrema(c:&[String],r:&Range,max:bool)->Result<Option<String>,CandidateError>{
    let mut parsed=Vec::new();
    for (i,s) in c.iter().enumerate(){parsed.push(parse_version(s).map_err(|error|CandidateError{index:i,error})?);}
    let mut best:Option<usize>=None;
    for i in 0..c.len(){if matches(&parsed[i],r) && best.map_or(true,|j| {
        let o=compare_versions(&parsed[i],&parsed[j]); if max{o==Ordering::Greater}else{o==Ordering::Less}
    }){best=Some(i);}}
    Ok(best.map(|i|c[i].clone()))
}

fn endpoint_string(v:&Version)->String{
    let mut s=format!("{}.{}.{}",v.major,v.minor,v.patch);
    if !v.pre.is_empty(){s.push('-');s.push_str(&v.pre.join("."));}
    s
}
pub fn canonical(r:&Range)->String{
    let mut out=Vec::new();
    for a in &r.arms{
        let mut lower:Option<(Version,bool)>=None; let mut upper:Option<(Version,bool)>=None;
        for c in &a.comparisons{
            match c.op{
                Op::Eq=>{lower=Some((c.version.clone(),true));upper=Some((c.version.clone(),true));}
                Op::Ge|Op::Gt=>{
                    let inc=c.op==Op::Ge;
                    if lower.as_ref().map_or(true,|(v,i)| compare_versions(&c.version,v)==Ordering::Greater || (compare_versions(&c.version,v)==Ordering::Equal && !inc && *i)){lower=Some((c.version.clone(),inc));}
                }
                Op::Le|Op::Lt=>{
                    let inc=c.op==Op::Le;
                    if upper.as_ref().map_or(true,|(v,i)| compare_versions(&c.version,v)==Ordering::Less || (compare_versions(&c.version,v)==Ordering::Equal && !inc && *i)){upper=Some((c.version.clone(),inc));}
                }
            }
        }
        let lo=lower.map_or("*".to_string(),|(v,i)|format!("{}{}",if i{">="}else{">"},endpoint_string(&v)));
        let hi=upper.map_or("*".to_string(),|(v,i)|format!("{}{}",if i{"<="}else{"<"},endpoint_string(&v)));
        let mut anchors=a.anchors.clone(); anchors.sort();
        anchors.dedup();
        let pre=if anchors.is_empty(){"-".into()}else{anchors.iter().map(|(x,y,z)|format!("{x}.{y}.{z}")).collect::<Vec<_>>().join(",")};
        out.push(format!("[{lo};{hi};pre={pre}]"));
    }
    out.sort();out.dedup();
    if out.is_empty(){"EMPTY".into()}else{out.join(" || ")}
}

pub fn intersects(a:&Range,b:&Range,include_prerelease:bool)->bool{
    for x in &a.arms { for y in &b.arms {
        if arms_intersect(x,y,include_prerelease){return true;}
    }}
    false
}
fn arms_intersect(a:&Arm,b:&Arm,include:bool)->bool{
    let mut cs=a.comparisons.clone();cs.extend(b.comparisons.clone());
    let anchors_a=&a.anchors; let anchors_b=&b.anchors;
    let mut bounds:Vec<Version>=Vec::new();
    for c in &cs {bounds.push(c.version.clone());}
    for (x,y,z) in anchors_a.iter().chain(anchors_b.iter()){bounds.push(vcore(*x,*y,*z,Vec::new()));}
    // Candidate stable witness at any relevant finite boundary or immediately adjacent core.
    let mut cores=vec![(0,0,0),(u64::MAX,u64::MAX,u64::MAX)];
    for v in &bounds{
        let (x,y,z)=core(v); cores.push((x,y,z));
        if z>0 {cores.push((x,y,z-1));}
        if z<u64::MAX {cores.push((x,y,z+1));}
    }
    for (x,y,z) in cores{
        let stable=vcore(x,y,z,Vec::new());
        if cs.iter().all(|c|cmp_ok(&stable,c)) {return true;}
    }
    if include { return false; }
    // Explicit anchor tuples are the only admitted prerelease tuples.
    for t in anchors_a.iter().filter(|t|anchors_b.contains(t)){
        let (x,y,z)=*t;
        let p=vcore(x,y,z,vec!["0".into()]);
        if cs.iter().all(|c|cmp_ok(&p,c)){return true;}
        // Search a few exact endpoint prereleases and their immediate successors.
        for v in &bounds{
            if core(v)==*t && !v.pre.is_empty() {
                if cs.iter().all(|c|cmp_ok(v,c)){return true;}
                let mut next=v.clone();next.pre.push("0".into());
                if cs.iter().all(|c|cmp_ok(&next,c)){return true;}
            }
        }
    }
    false
}

#[derive(Clone, Debug)]
pub enum SetExpr { Range(String), Union(Vec<SetExpr>), Intersection(Vec<SetExpr>) }
#[derive(Clone, Debug)]
pub struct Constraint { pub label:String, pub expr:SetExpr }
#[derive(Clone, Debug)]
pub struct Package { pub name:String, pub candidates:Vec<String>, pub constraints:Vec<Constraint> }
#[derive(Clone, Copy, Debug)]
pub enum Preference { Highest, Lowest, StableFirst }
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection { pub package:String, pub version:String }
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name:String },
    DuplicateLabel { package:String, label:String },
    InvalidCandidate { package:String, index:usize, error:VersionError },
    InvalidRange { package:String, label:String, path:Vec<usize>, error:RangeError },
    NoSolution { package:String, labels:Vec<String> },
}

fn eval_expr(e:&SetExpr,v:&Version,inc:bool)->bool{
    match e{
        SetExpr::Range(s)=>parse_range(s).map(|r|matches_with(v,&r,inc)).unwrap_or(false),
        SetExpr::Union(es)=>es.iter().any(|x|eval_expr(x,v,inc)),
        SetExpr::Intersection(es)=>es.iter().all(|x|eval_expr(x,v,inc)),
    }
}
fn validate_expr(e:&SetExpr,package:&str,label:&str,path:&mut Vec<usize>)->Result<(),SelectError>{
    match e{
        SetExpr::Range(s)=>parse_range(s).map(|_|()).map_err(|error|SelectError::InvalidRange{package:package.into(),label:label.into(),path:path.clone(),error}),
        SetExpr::Union(xs)|SetExpr::Intersection(xs)=>{for (i,x) in xs.iter().enumerate(){path.push(i);validate_expr(x,package,label,path)?;path.pop();}Ok(())}
    }
}
pub fn select_packages(packages:&[Package],preference:Preference,include_prerelease:bool)->Result<Vec<Selection>,SelectError>{
    let mut names=std::collections::HashSet::new();
    for p in packages{if !names.insert(p.name.clone()){return Err(SelectError::DuplicatePackage{name:p.name.clone()});}}
    for p in packages{
        let mut labels=std::collections::HashSet::new();
        for c in &p.constraints{if !labels.insert(c.label.clone()){return Err(SelectError::DuplicateLabel{package:p.name.clone(),label:c.label.clone()});}}
    }
    let mut parsed=Vec::new();
    for p in packages{
        let mut vs=Vec::new();
        for (i,s) in p.candidates.iter().enumerate(){vs.push(parse_version(s).map_err(|error|SelectError::InvalidCandidate{package:p.name.clone(),index:i,error})?);}
        parsed.push(vs);
    }
    for p in packages{for c in &p.constraints{validate_expr(&c.expr,&p.name,&c.label,&mut Vec::new())?;}}
    let mut result=Vec::new();
    for (pi,p) in packages.iter().enumerate(){
        let eligible:Vec<usize>=(0..p.candidates.len()).filter(|&i|p.constraints.iter().all(|c|eval_expr(&c.expr,&parsed[pi][i],include_prerelease))).collect();
        if eligible.is_empty(){
            let labels=if p.candidates.is_empty(){Vec::new()}else{minimal_conflict(p,&parsed[pi],include_prerelease)};
            return Err(SelectError::NoSolution{package:p.name.clone(),labels});
        }
        let stable_exists=preference==Preference::StableFirst && eligible.iter().any(|&i|parsed[pi][i].pre.is_empty());
        let mut best=eligible[0];
        for &i in eligible.iter().skip(1){
            let si=preference==Preference::StableFirst && stable_exists && parsed[pi][i].pre.is_empty();
            let sb=preference==Preference::StableFirst && stable_exists && parsed[pi][best].pre.is_empty();
            if si!=sb {if si{best=i;}continue;}
            let o=compare_versions(&parsed[pi][i],&parsed[pi][best]);
            if (preference==Preference::Lowest && o==Ordering::Less)||(preference!=Preference::Lowest&&o==Ordering::Greater){best=i;}
        }
        result.push(Selection{package:p.name.clone(),version:p.candidates[best].clone()});
    }
    Ok(result)
}
fn minimal_conflict(p:&Package,vs:&[Version],inc:bool)->Vec<String>{
    let n=p.constraints.len();
    for size in 1..=n{
        let mut chosen=Vec::new();
        if let Some(ids)=find_conflict(p,vs,inc,size,0,&mut chosen){let mut labels:Vec<String>=ids.iter().map(|&i|p.constraints[i].label.clone()).collect();labels.sort_by(|a,b|a.as_bytes().cmp(b.as_bytes()));return labels;}
    }
    Vec::new()
}
fn find_conflict(p:&Package,vs:&[Version],inc:bool,left:usize,start:usize,chosen:&mut Vec<usize>)->Option<Vec<usize>>{
    if left==0{
        let any=vs.iter().any(|v|chosen.iter().all(|&i|eval_expr(&p.constraints[i].expr,v,inc)));
        return if any{None}else{Some(chosen.clone())};
    }
    for i in start..=p.constraints.len().saturating_sub(left){
        chosen.push(i);
        if let Some(x)=find_conflict(p,vs,inc,left-1,i+1,chosen){return Some(x);}
        chosen.pop();
    }
    None
}
