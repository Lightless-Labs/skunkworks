fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> u8 {
    match c {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn constraint_cmp(a: &Constraint, b: &Constraint) -> Ordering {
    constraint_rank(*a)
        .cmp(&constraint_rank(*b))
        .then_with(|| match (*a, *b) {
            (Constraint::Exact(x), Constraint::Exact(y))
            | (Constraint::AtLeast(x), Constraint::AtLeast(y))
            | (Constraint::Before(x), Constraint::Before(y))
            | (Constraint::Caret(x), Constraint::Caret(y)) => x.cmp(&y),
            _ => Ordering::Equal,
        })
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort_by(constraint_cmp);
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut list = canonical_constraints(constraints.to_vec());
    list.drain(..)
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn constraint_matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn all_match(constraints: &[Constraint], v: Version) -> bool {
    constraints.iter().all(|c| constraint_matches(*c, v))
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push(char::from_digit((byte >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            out.push(char::from_digit((byte & 15) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    out
}

fn evidence_cmp(a: &Evidence, b: &Evidence) -> Ordering {
    constraint_cmp(&a.constraint, &b.constraint).then_with(|| a.path.cmp(&b.path))
}

fn evidence_lines(evidence: &[Evidence]) -> String {
    let mut sorted = evidence.to_vec();
    sorted.sort_by(evidence_cmp);
    let mut out = String::new();
    for e in sorted {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            encode(package),
            evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct NodeData {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    activated: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn path_cmp(a: &[String], b: &[String]) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.cmp(b))
}

fn platform_matches(platform: Platform, target: Target) -> bool {
    matches!(
        (platform, target),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn root_token(root: &Requirement, index: usize) -> String {
    let mut features = root.features.clone();
    features.sort();
    features.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        root.package,
        constraint_text(root.constraint),
        features.join(",")
    )
}

fn dep_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    let mut features = dep.features.clone();
    features.sort();
    features.dedup();
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        gate,
        platform
    )
    .replace("+{}", &format!("+{{{}}}", features.join(",")))
}

fn compute_node(
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    catalog: &BTreeMap<String, &Package>,
    target: Target,
) -> NodeData {
    let mut data = NodeData::default();
    let mut queue: Vec<(String, String)> = Vec::new();

    for (i, root) in roots.iter().enumerate() {
        let token = root_token(root, i);
        data.constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        data.origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
        data.requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        let candidate = vec![token];
        match data.paths.get(&root.package) {
            Some(old) if path_cmp(old, &candidate) != Ordering::Greater => {}
            _ => {
                data.paths.insert(root.package.clone(), candidate);
            }
        }
        queue.push((root.package.clone(), String::new()));
    }

    let mut processed: BTreeSet<(String, String)> = BTreeSet::new();
    loop {
        let mut progressed = false;

        let active_names: Vec<String> = data
            .constraints
            .keys()
            .chain(data.requests.keys())
            .cloned()
            .collect::<BTreeSet<_>>()
            .into_iter()
            .collect();

        for package_name in active_names {
            let Some(version) = selected.get(&package_name).copied() else {
                continue;
            };
            let Some(package) = catalog.get(&package_name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested: Vec<String> = data
                .requests
                .get(&package_name)
                .cloned()
                .unwrap_or_default()
                .into_iter()
                .collect();

            for feature_name in requested {
                let key = (package_name.clone(), feature_name.clone());
                if !data.activated.entry(package_name.clone()).or_default().insert(feature_name.clone()) {
                    continue;
                }
                progressed = true;
                processed.insert(key);
                if let Some(decl) = release.features.iter().find(|f| f.name == feature_name) {
                    data.requests
                        .entry(package_name.clone())
                        .or_default()
                        .extend(decl.enables.iter().cloned());
                }
            }

            let activated = data
                .activated
                .get(&package_name)
                .cloned()
                .unwrap_or_default();
            let source_path = data.paths.get(&package_name).cloned().unwrap_or_default();

            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target) {
                    continue;
                }
                if dep
                    .gate
                    .as_ref()
                    .is_some_and(|gate| !activated.contains(gate))
                {
                    continue;
                }
                let token = dep_token(&package_name, version, j, dep);
                let next_path = {
                    let mut p = source_path.clone();
                    p.push(token.clone());
                    p
                };
                let was_new = data
                    .constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .last()
                    .is_none();
                data.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                data.origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: next_path.clone(),
                    });
                data.requests
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());

                let improve = match data.paths.get(&dep.package) {
                    Some(old) => path_cmp(&next_path, old) == Ordering::Less,
                    None => true,
                };
                if improve {
                    data.paths.insert(dep.package.clone(), next_path);
                }
                if was_new || improve {
                    progressed = true;
                }
            }
        }

        if !progressed {
            break;
        }

        // Dependency contributions must only be added once per selected source
        // release and dependency. Rebuild the closure from scratch to avoid
        // duplicating constraints and origins on each fixed-point iteration.
        return compute_node_once(roots, selected, catalog, target);
    }

    data
}

fn compute_node_once(
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    catalog: &BTreeMap<String, &Package>,
    target: Target,
) -> NodeData {
    let mut data = NodeData::default();
    for (i, root) in roots.iter().enumerate() {
        let token = root_token(root, i);
        data.constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        data.origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
        data.requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        let p = vec![token];
        if data
            .paths
            .get(&root.package)
            .is_none_or(|old| path_cmp(&p, old) == Ordering::Less)
        {
            data.paths.insert(root.package.clone(), p);
        }
    }

    loop {
        let before_constraints = data.constraints.clone();
        let before_origins = data.origins.clone();
        let before_requests = data.requests.clone();
        let before_paths = data.paths.clone();
        let mut activated = data.activated.clone();

        let names: Vec<String> = data
            .constraints
            .keys()
            .chain(data.requests.keys())
            .cloned()
            .collect::<BTreeSet<_>>()
            .into_iter()
            .collect();

        for name in names {
            let Some(version) = selected.get(&name).copied() else {
                continue;
            };
            let Some(package) = catalog.get(&name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested = data.requests.get(&name).cloned().unwrap_or_default();
            for f in requested {
                if activated.entry(name.clone()).or_default().insert(f.clone()) {
                    if let Some(decl) = release.features.iter().find(|d| d.name == f) {
                        data.requests
                            .entry(name.clone())
                            .or_default()
                            .extend(decl.enables.iter().cloned());
                    }
                }
            }

            let active_features = activated.get(&name).cloned().unwrap_or_default();
            let source_path = data.paths.get(&name).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !active_features.contains(gate))
                {
                    continue;
                }
                let token = dep_token(&name, version, j, dep);
                let mut dep_path = source_path.clone();
                dep_path.push(token);
                data.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                data.origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: dep_path.clone(),
                    });
                data.requests
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());
                if data
                    .paths
                    .get(&dep.package)
                    .is_none_or(|old| path_cmp(&dep_path, old) == Ordering::Less)
                {
                    data.paths.insert(dep.package.clone(), dep_path);
                }
            }
        }

        data.activated = activated;

        // Each dependency edge is a single origin. The iterative closure above
        // can revisit it, so normalize by recomputing via a bounded monotone
        // traversal over distinct edge identities.
        let mut seen_edges: BTreeSet<(String, Version, usize)> = BTreeSet::new();
        let mut clean_constraints = BTreeMap::new();
        let mut clean_origins = BTreeMap::new();
        let mut clean_requests = BTreeMap::new();
        let mut clean_paths = BTreeMap::new();

        for (i, root) in roots.iter().enumerate() {
            let token = root_token(root, i);
            clean_constraints
                .entry(root.package.clone())
                .or_insert_with(Vec::new)
                .push(root.constraint);
            clean_origins
                .entry(root.package.clone())
                .or_insert_with(Vec::new)
                .push(Origin {
                    constraint: root.constraint,
                    path: vec![token.clone()],
                });
            clean_requests
                .entry(root.package.clone())
                .or_insert_with(BTreeSet::new)
                .extend(root.features.iter().cloned());
            let path = vec![token];
            if clean_paths
                .get(&root.package)
                .is_none_or(|old| path_cmp(&path, old) == Ordering::Less)
            {
                clean_paths.insert(root.package.clone(), path);
            }
        }

        // Reconstruct reachable contributions using the computed feature closure.
        let mut changed = true;
        let mut reachable_edges = BTreeSet::new();
        while changed {
            changed = false;
            let names: Vec<String> = clean_constraints
                .keys()
                .chain(clean_requests.keys())
                .cloned()
                .collect::<BTreeSet<_>>()
                .into_iter()
                .collect();
            for name in names {
                let Some(version) = selected.get(&name).copied() else {
                    continue;
                };
                let Some(package) = catalog.get(&name) else {
                    continue;
                };
                let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                    continue;
                };
                let active = data.activated.get(&name).cloned().unwrap_or_default();
                for (j, dep) in release.deps.iter().enumerate() {
                    if !platform_matches(dep.platform, target)
                        || dep.gate.as_ref().is_some_and(|g| !active.contains(g))
                    {
                        continue;
                    }
                    let edge = (name.clone(), version, j);
                    if !reachable_edges.insert(edge.clone()) {
                        continue;
                    }
                    let _ = &mut seen_edges;
                    let dep_path = {
                        let mut path = clean_paths.get(&name).cloned().unwrap_or_default();
                        path.push(dep_token(&name, version, j, dep));
                        path
                    };
                    clean_constraints
                        .entry(dep.package.clone())
                        .or_insert_with(Vec::new)
                        .push(dep.constraint);
                    clean_origins
                        .entry(dep.package.clone())
                        .or_insert_with(Vec::new)
                        .push(Origin {
                            constraint: dep.constraint,
                            path: dep_path.clone(),
                        });
                    clean_requests
                        .entry(dep.package.clone())
                        .or_insert_with(BTreeSet::new)
                        .extend(dep.features.iter().cloned());
                    if clean_paths
                        .get(&dep.package)
                        .is_none_or(|old| path_cmp(&dep_path, old) == Ordering::Less)
                    {
                        clean_paths.insert(dep.package.clone(), dep_path);
                    }
                    changed = true;
                }
            }
        }

        let new_activated = data.activated.clone();
        let mut newly_enabled = false;
        for (name, features) in &clean_requests {
            let Some(version) = selected.get(name).copied() else {
                continue;
            };
            let Some(package) = catalog.get(name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };
            for f in features {
                if !new_activated.get(name).is_some_and(|set| set.contains(f)) {
                    newly_enabled = true;
                }
                if let Some(decl) = release.features.iter().find(|d| d.name == f) {
                    if !clean_requests
                        .get(name)
                        .is_some_and(|set| d.enables.iter().all(|e| set.contains(e)))
                    {
                        newly_enabled = true;
                    }
                }
            }
        }

        data.constraints = clean_constraints;
        data.origins = clean_origins;
        data.requests = clean_requests;
        data.paths = clean_paths;

        if !newly_enabled && data.activated == new_activated {
            break;
        }
        // A finite package/feature/dependency universe guarantees convergence.
        if data.activated == new_activated {
            break;
        }
    }

    data
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(BTreeMap<String, &Package>, BTreeMap<String, Version>), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].gate",
                                i, j, k
                            ),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut catalog_map = BTreeMap::new();
    for package in catalog {
        if catalog_map.contains_key(&package.name) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
        catalog_map.insert(package.name.clone(), package);
    }

    for package in catalog {
        let mut seen = BTreeSet::new();
        for release in &package.releases {
            if !seen.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = BTreeSet::new();
            for feature in &release.features {
                if !seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_map = BTreeMap::new();
    for lock in locks {
        if lock_map.contains_key(&lock.package) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
        lock_map.insert(lock.package.clone(), lock.version);
    }

    Ok((catalog_map, lock_map))
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: &'a BTreeMap<String, &'a Package>,
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>, NodeData)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let data = compute_node(self.roots, selected, self.catalog, self.target);

        let active: BTreeSet<String> = data.constraints.keys().cloned().collect();
        if let Some(name) = active.iter().find(|name| !self.catalog.contains_key(*name)) {
            let evidence = data
                .origins
                .get(name)
                .cloned()
                .unwrap_or_default()
                .into_iter()
                .map(|o| Evidence {
                    constraint: o.constraint,
                    path: o.path,
                })
                .collect();
            self.record_error(ResolveError::Missing {
                package: name.clone(),
                evidence,
            });
            return Ok(());
        }

        for name in selected.keys() {
            let constraints = canonical_constraints(
                data.constraints.get(name).cloned().unwrap_or_default(),
            );
            if !all_match(&constraints, selected[name]) {
                let evidence = data
                    .origins
                    .get(name)
                    .cloned()
                    .unwrap_or_default()
                    .into_iter()
                    .map(|o| Evidence {
                        constraint: o.constraint,
                        path: o.path,
                    })
                    .collect();
                self.record_error(ResolveError::Conflict {
                    package: name.clone(),
                    selected: selected[name],
                    requirements: constraints,
                    evidence,
                });
                return Ok(());
            }
        }

        for name in active.iter().filter(|name| !selected.contains_key(*name)) {
            let package = self.catalog[name];
            let constraints = canonical_constraints(
                data.constraints.get(name).cloned().unwrap_or_default(),
            );
            let any = package
                .releases
                .iter()
                .any(|r| all_match(&constraints, r.version));
            if !any {
                let evidence = data
                    .origins
                    .get(name)
                    .cloned()
                    .unwrap_or_default()
                    .into_iter()
                    .map(|o| Evidence {
                        constraint: o.constraint,
                        path: o.path,
                    })
                    .collect();
                self.record_error(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints,
                    evidence,
                });
                return Ok(());
            }
        }

        for name in selected.keys() {
            let Some(package) = self.catalog.get(name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == selected[name]) else {
                continue;
            };
            let features = data.activated.get(name).cloned().unwrap_or_default();
            for feature in features {
                if !release.features.iter().any(|f| f.name == feature) {
                    let path = data.paths.get(name).cloned().unwrap_or_default();
                    self.record_error(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: selected[name],
                        feature,
                        path,
                    });
                    return Ok(());
                }
            }
        }

        let next = active
            .iter()
            .find(|name| !selected.contains_key(*name))
            .cloned();
        let Some(name) = next else {
            self.record_solution(selected.clone(), data);
            return Ok(());
        };

        let package = self.catalog[&name];
        let constraints = canonical_constraints(
            data.constraints.get(&name).cloned().unwrap_or_default(),
        );
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| all_match(&constraints, *v))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.get(&name) {
            if let Some(index) = candidates.iter().position(|v| v == lock) {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), candidate);
            self.visit(selected)?;
            selected.remove(&name);
        }
        Ok(())
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn record_solution(&mut self, selected: BTreeMap<String, Version>, _data: NodeData) {
        let cost = self
            .locks
            .iter()
            .filter(|(name, version)| selected.get(*name) != Some(*version))
            .count();

        let replace = match &self.best {
            None => true,
            Some((best_cost, best_selected, _)) => {
                if cost < *best_cost {
                    true
                } else if cost > *best_cost {
                    false
                } else {
                    compare_version_vectors(&selected, best_selected, self.catalog) == Ordering::Greater
                }
            }
        };
        if replace {
            self.best = Some((cost, selected, _data));
        }
    }
}

fn compare_version_vectors(
    a: &BTreeMap<String, Version>,
    b: &BTreeMap<String, Version>,
    catalog: &BTreeMap<String, &Package>,
) -> Ordering {
    for name in catalog.keys() {
        let cmp = match (a.get(name), b.get(name)) {
            (None, None) => Ordering::Equal,
            (None, Some(_)) => Ordering::Less,
            (Some(_), None) => Ordering::Greater,
            (Some(x), Some(y)) => x.cmp(y),
        };
        if cmp != Ordering::Equal {
            return cmp;
        }
    }
    Ordering::Equal
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    let (catalog_map, lock_map) = validate_inputs(catalog, roots, locks)?;
    let mut search = Search {
        roots,
        catalog: &catalog_map,
        locks: &lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.visit(&mut BTreeMap::new())?;

    let Some((_, selected, _)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_vec: Vec<Selection> = selected
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut all_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    all_names.extend(selected.keys().cloned());

    let final_data = compute_node(roots, &selected, &catalog_map, target);
    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = selected.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let constraints = final_data
                        .constraints
                        .get(&package)
                        .cloned()
                        .unwrap_or_default();
                    if !all_match(&constraints, old) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
