#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for byte in s.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 15) as usize] as char);
        }
    }
    out
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort();
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical(cs.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn matches_all(v: Version, cs: &[Constraint]) -> bool {
    cs.iter().all(|c| matches(v, *c))
}

fn catalog_constraints(
    roots: &[Requirement],
    catalog: &[Package],
    assigned: &[(String, Version)],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in assigned {
        if let Some(package) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for cs in active.values_mut() {
        *cs = canonical(std::mem::take(cs));
    }
    active
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<(String, Version)>)>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn visit(&mut self, assigned: &mut Vec<(String, Version)>) -> Result<(), ResolveError> {
        let active = catalog_constraints(self.roots, self.catalog, assigned);

        let missing = active
            .keys()
            .filter(|name| !self.catalog.iter().any(|p| p.name == ***name))
            .next()
            .cloned();
        if let Some(package) = missing {
            self.record_error(ResolveError::Missing { package });
            return Ok(());
        }

        let assigned_map: std::collections::BTreeMap<&str, Version> =
            assigned.iter().map(|(n, v)| (n.as_str(), *v)).collect();

        for (package, constraints) in &active {
            if let Some(version) = assigned_map.get(package.as_str()) {
                if !matches_all(*version, constraints) {
                    self.record_error(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                    return Ok(());
                }
            }
        }

        for (package, constraints) in &active {
            if assigned_map.contains_key(package.as_str()) {
                continue;
            }
            let catalog_package = self.catalog.iter().find(|p| p.name == *package).unwrap();
            if !catalog_package
                .releases
                .iter()
                .any(|r| matches_all(r.version, constraints))
            {
                self.record_error(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints.clone(),
                });
                return Ok(());
            }
        }

        let next = active
            .keys()
            .find(|name| !assigned_map.contains_key(name.as_str()))
            .cloned();

        let Some(package) = next else {
            let mut solution = assigned.clone();
            solution.sort_by(|a, b| a.0.cmp(&b.0));
            self.consider(solution);
            return Ok(());
        };

        let constraints = &active[&package];
        let catalog_package = self.catalog.iter().find(|p| p.name == package).unwrap();
        let mut candidates: Vec<Version> = catalog_package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| matches_all(*v, constraints))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.iter().find(|l| l.package == package) {
            if let Some(index) = candidates.iter().position(|v| *v == lock.version) {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        let mut first_branch_error = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assigned.push((package.clone(), candidate));
            let result = self.visit(assigned);
            assigned.pop();

            if let Err(error) = result {
                return Err(error);
            }
            if first_branch_error.is_none() {
                first_branch_error = self.first_error.clone();
            }
        }
        if self.best.is_none() && self.first_error.is_none() {
            if let Some(error) = first_branch_error {
                self.first_error = Some(error);
            }
        }
        Ok(())
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn consider(&mut self, solution: Vec<(String, Version)>) {
        let cost = self
            .locks
            .iter()
            .filter(|lock| {
                !solution
                    .iter()
                    .any(|(name, version)| name == &lock.package && *version == lock.version)
            })
            .count();

        let better = match &self.best {
            None => true,
            Some((best_cost, best_solution)) if cost < *best_cost => true,
            Some((best_cost, best_solution)) if cost > *best_cost => false,
            Some((_, best_solution)) => {
                let mut names: Vec<&String> = self.catalog.iter().map(|p| &p.name).collect();
                names.sort();
                names.dedup();
                for name in names {
                    let current = solution
                        .iter()
                        .find(|(n, _)| n == name)
                        .map(|(_, v)| *v);
                    let prior = best_solution
                        .iter()
                        .find(|(n, _)| n == name)
                        .map(|(_, v)| *v);
                    if current != prior {
                        if current.is_none() {
                            break false;
                        }
                        if prior.is_none() {
                            break true;
                        }
                        if current > prior {
                            break true;
                        }
                        break false;
                    }
                }
                true
            }
        };
        if better {
            self.best = Some((cost, solution));
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut assigned = Vec::new();
    search.visit(&mut assigned)?;

    let Some((_, solution)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::Missing {
            package: String::new(),
        }));
    };

    let selected: Vec<Selection> = solution
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();
    let final_active = catalog_constraints(roots, catalog, &solution);
    let before: std::collections::BTreeMap<&str, Version> = locks
        .iter()
        .map(|l| (l.package.as_str(), l.version))
        .collect();
    let after: std::collections::BTreeMap<&str, Version> = solution
        .iter()
        .map(|(n, v)| (n.as_str(), *v))
        .collect();

    let names: std::collections::BTreeSet<&str> =
        before.keys().chain(after.keys()).copied().collect();
    let mut changes = Vec::new();
    for name in names {
        let b = before.get(name).copied();
        let a = after.get(name).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == name)
                    .map(|p| p.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if !final_active
                    .get(name)
                    .map(|cs| matches_all(old, cs))
                    .unwrap_or(true)
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name.to_string(),
            before: b,
            after: a,
            reason,
        });
    }
    changes.sort_by(|a, b| a.package.cmp(&b.package));

    Ok(Resolution { selected, changes })
}
