#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, req) in roots.iter().enumerate() {
        if !valid_name(&req.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: req.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }

    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    Ok(())
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", byte));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            encode_text(site),
            encode_text(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode_text(package))
        }
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode_text(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode_text(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Option<Version>>, Vec<Selection>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn package(&self, name: &str) -> Option<&'a Package> {
        self.catalog.iter().find(|p| p.name == name)
    }

    fn inspect(
        &mut self,
        assigned: &mut std::collections::BTreeMap<String, Version>,
    ) -> Result<(), ResolveError> {
        let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
            std::collections::BTreeMap::new();

        for root in self.roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }

        for (name, version) in assigned.iter() {
            if let Some(package) = self.package(name) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }

        for constraints in active.values_mut() {
            canonicalize(constraints);
        }

        if let Some(name) = active
            .keys()
            .find(|name| self.package(name).is_none())
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in assigned.iter() {
            if let Some(constraints) = active.get(name) {
                if !constraints.iter().all(|c| matches(*version, *c)) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        let mut candidates_by_name = std::collections::BTreeMap::new();
        for (name, constraints) in &active {
            if assigned.contains_key(name) {
                continue;
            }
            let package = self.package(name).expect("missing packages handled above");
            let mut candidates: Vec<Version> = package
                .releases
                .iter()
                .map(|r| r.version)
                .filter(|v| constraints.iter().all(|c| matches(*v, *c)))
                .collect();
            candidates.sort_by(|a, b| b.cmp(a));
            candidates.dedup();
            if candidates.is_empty() {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
            if let Some(lock) = self.locks.iter().find(|l| l.package == *name) {
                if let Some(index) = candidates.iter().position(|v| *v == lock.version) {
                    let locked = candidates.remove(index);
                    candidates.insert(0, locked);
                }
            }
            candidates_by_name.insert(name.clone(), candidates);
        }

        let Some((name, candidates)) = candidates_by_name.into_iter().next() else {
            self.record_solution(assigned, &active);
            return Ok(());
        };

        let mut first_local_error = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            assigned.insert(name.clone(), candidate);
            match self.inspect(assigned) {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    assigned.remove(&name);
                    return Err(ResolveError::Limit { attempts });
                }
                Err(error) => {
                    if first_local_error.is_none() {
                        first_local_error = Some(error);
                    }
                }
            }
            assigned.remove(&name);
        }

        if let Some(error) = first_local_error {
            Err(error)
        } else {
            Ok(())
        }
    }

    fn record_solution(
        &mut self,
        assigned: &std::collections::BTreeMap<String, Version>,
        active: &std::collections::BTreeMap<String, Vec<Constraint>>,
    ) {
        let locks_by_name: std::collections::BTreeMap<&str, Version> = self
            .locks
            .iter()
            .map(|lock| (lock.package.as_str(), lock.version))
            .collect();

        let cost = self
            .locks
            .iter()
            .filter(|lock| assigned.get(&lock.package) != Some(&lock.version))
            .count();

        let catalog_names: std::collections::BTreeSet<&str> =
            self.catalog.iter().map(|p| p.name.as_str()).collect();
        let vector: Vec<Option<Version>> = catalog_names
            .iter()
            .map(|name| assigned.get(*name).copied())
            .collect();

        let mut selected: Vec<Selection> = assigned
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect();
        selected.sort_by(|a, b| a.package.cmp(&b.package));

        let improves = match &self.best {
            None => true,
            Some((best_cost, best_vector, _)) => {
                cost < *best_cost || (cost == *best_cost && vector > *best_vector)
            }
        };

        let _ = (active, locks_by_name);
        if improves {
            self.best = Some((cost, vector, selected));
        }
    }
}

fn change_reason(
    package: &str,
    before: Version,
    after: Version,
    catalog: &[Package],
    final_constraints: &std::collections::BTreeMap<String, Vec<Constraint>>,
) -> ChangeReason {
    let exists = catalog
        .iter()
        .find(|p| p.name == package)
        .map(|p| p.releases.iter().any(|r| r.version == before))
        .unwrap_or(false);

    if !exists {
        ChangeReason::Unavailable
    } else if final_constraints
        .get(package)
        .map(|cs| !cs.iter().all(|c| matches(before, *c)))
        .unwrap_or(false)
    {
        ChangeReason::Rejected
    } else {
        ChangeReason::Rebalanced
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let mut search = Search {
        catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };

    let mut assigned = std::collections::BTreeMap::new();
    match search.inspect(&mut assigned) {
        Err(ResolveError::Limit { attempts }) => return Err(ResolveError::Limit { attempts }),
        Err(error) => {
            if search.best.is_none() {
                return Err(error);
            }
        }
        Ok(()) => {}
    }

    let Some((_, _, selected)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::Missing {
            package: String::new(),
        }));
    };

    let selected_map: std::collections::BTreeMap<String, Version> = selected
        .iter()
        .map(|s| (s.package.clone(), s.version))
        .collect();

    let mut final_constraints: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        final_constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for selection in &selected {
        if let Some(package) = catalog.iter().find(|p| p.name == selection.package) {
            if let Some(release) = package
                .releases
                .iter()
                .find(|r| r.version == selection.version)
            {
                for dep in &release.deps {
                    final_constraints
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in final_constraints.values_mut() {
        canonicalize(constraints);
    }

    let mut before_map = std::collections::BTreeMap::new();
    for lock in locks {
        before_map.insert(lock.package.clone(), lock.version);
    }

    let all_names: std::collections::BTreeSet<String> = before_map
        .keys()
        .chain(selected_map.keys())
        .cloned()
        .collect();

    let mut changes = Vec::new();
    for package in all_names {
        let before = before_map.get(&package).copied();
        let after = selected_map.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(before_version), Some(after_version)) => change_reason(
                &package,
                before_version,
                after_version,
                catalog,
                &final_constraints,
            ),
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
