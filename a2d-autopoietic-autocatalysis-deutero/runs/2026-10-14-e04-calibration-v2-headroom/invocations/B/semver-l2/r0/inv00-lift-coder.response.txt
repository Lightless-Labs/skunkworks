use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn is_ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return Err(if *pos == bytes.len()
            || matches!(bytes[*pos], b'.' | b'-' | b'+')
        {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut ident_start = start;
    loop {
        let mut i = ident_start;
        while i < end && bytes[i] != b'.' {
            if !is_ident_char(bytes[i]) {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let id = &s[ident_start..i];
        if pre && id.bytes().all(is_digit) && id.len() > 1 && id.as_bytes()[0] == b'0' {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        result.push(id.to_string());
        if i == end {
            break;
        }
        ident_start = i + 1;
        if ident_start == end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for _ in 0..2 {
        if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        if s.as_bytes()[pos] != b'.' {
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        let value = parse_core(s, &mut pos)?;
        if _ == 0 {
            // Kept separate below to preserve left-to-right validation.
            let minor = value;
            if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if s.as_bytes()[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
            let patch = parse_core(s, &mut pos)?;
            return parse_version_tail(s, pos, major, minor, patch);
        }
    }
    unreachable!()
}

fn parse_version_tail(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if pos < s.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < s.len() && bytes[pos] == b'-' {
        let start = pos + 1;
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|n| start + n)
            .unwrap_or(s.len());
        pre = parse_identifiers(s, start, end, true)?;
        pos = end;
    }
    if pos < s.len() && bytes[pos] == b'+' {
        build = parse_identifiers(s, pos + 1, s.len(), false)?;
        pos = s.len();
    }
    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_numeric_id(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.pre.iter().zip(&b.pre) {
        let xn = is_numeric_id(x);
        let yn = is_numeric_id(y);
        let c = match (xn, yn) {
            (true, true) => cmp_numeric_id(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.pre.len().cmp(&b.pre.len())
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    checks: Vec<Check>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Check {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn parse_full_operand(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn add_check(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors
            .push((version.major, version.minor, version.patch));
    }
    arm.checks.push(Check { op, version });
}

fn check_matches(v: &Version, c: &Check) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

fn version_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn padded(fields: &[Option<u64>]) -> Version {
    Version {
        major: fields.first().copied().flatten().unwrap_or(0),
        minor: fields.get(1).copied().flatten().unwrap_or(0),
        patch: fields.get(2).copied().flatten().unwrap_or(0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    numeric: usize,
    full: Option<Version>,
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix = s.bytes().any(|b| b == b'-' || b == b'+');
    let pieces: Vec<&str> = s.split('.').collect();
    let no_wildcard = !pieces.iter().any(|p| matches!(*p, "*" | "x" | "X"));
    if has_suffix || (pieces.len() >= 3 && no_wildcard) {
        return Ok(Partial {
            fields: Vec::new(),
            numeric: 3,
            full: Some(parse_full_operand(s, at)?),
        });
    }
    if pieces.is_empty() || pieces.len() > 3 {
        return Err(RangeError::Syntax { at });
    }
    let mut fields = Vec::new();
    let mut saw_wild = false;
    let mut numeric = 0;
    for p in pieces {
        if p.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(p, "*" | "x" | "X") {
            saw_wild = true;
            fields.push(None);
        } else {
            if saw_wild
                || !p.bytes().all(is_digit)
                || (p.len() > 1 && p.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            let n = p.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            numeric += 1;
            fields.push(Some(n));
        }
    }
    Ok(Partial {
        fields,
        numeric,
        full: None,
    })
}

fn with_lower(partial: &Partial) -> Version {
    if let Some(v) = &partial.full {
        v.clone()
    } else {
        padded(&partial.fields)
    }
}

fn increment_upper(
    low: &Version,
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let mut v = Version {
        major: low.major,
        minor: low.minor,
        patch: low.patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    };
    let target = match field {
        0 => &mut v.major,
        1 => &mut v.minor,
        _ => &mut v.patch,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    match field {
        0 => {
            v.minor = 0;
            v.patch = 0;
        }
        1 => v.patch = 0,
        _ => {}
    }
    Ok(v)
}

fn expand_bare(partial: Partial, at: usize, arm: &mut Arm) -> Result<(), RangeError> {
    if let Some(v) = partial.full {
        add_check(arm, Op::Eq, v, true);
        return Ok(());
    }
    if partial.numeric == 0 {
        return Ok(());
    }
    let lower = padded(&partial.fields);
    add_check(arm, Op::Ge, lower.clone(), true);
    if partial.numeric < 3 {
        let upper = increment_upper(&lower, partial.numeric - 1, at)?;
        add_check(arm, Op::Lt, upper, false);
    }
    Ok(())
}

fn expand_caret_tilde(
    operand: &str,
    at: usize,
    caret: bool,
    arm: &mut Arm,
) -> Result<(), RangeError> {
    let p = parse_partial(operand, at)?;
    if p.numeric == 0 {
        return Err(RangeError::Syntax { at: at.saturating_sub(1) });
    }
    let lower = with_lower(&p);
    add_check(arm, Op::Ge, lower.clone(), true);
    let k = if p.full.is_some() { 3 } else { p.numeric };
    let field = if caret {
        if lower.major > 0 || k == 1 {
            0
        } else if lower.minor > 0 || k == 2 {
            1
        } else {
            2
        }
    } else if k == 1 {
        0
    } else {
        1
    };
    let upper = increment_upper(&lower, field, at)?;
    add_check(arm, Op::Lt, upper, false);
    Ok(())
}

fn expand_hyphen(
    lower_s: &str,
    lower_at: usize,
    upper_s: &str,
    upper_at: usize,
    arm: &mut Arm,
) -> Result<(), RangeError> {
    let lower_p = parse_partial(lower_s, lower_at)?;
    let upper_p = parse_partial(upper_s, upper_at)?;
    if lower_p.numeric == 0 {
        return Err(RangeError::Syntax { at: lower_at });
    }
    if upper_p.numeric == 0 {
        return Err(RangeError::Syntax { at: upper_at });
    }
    let lower = with_lower(&lower_p);
    add_check(arm, Op::Ge, lower, true);
    if let Some(v) = upper_p.full {
        add_check(arm, Op::Le, v, true);
    } else {
        let upper_low = padded(&upper_p.fields);
        let upper_bound = if upper_p.numeric >= 3 {
            upper_low
        } else {
            increment_upper(&upper_low, upper_p.numeric - 1, upper_at)?
        };
        add_check(arm, Op::Lt, upper_bound, false);
    }
    Ok(())
}

fn token_spans(s: &str, start: usize, end: usize) -> Vec<(usize, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && ws(bytes[i]) {
            i += 1;
        }
        if i == end {
            break;
        }
        let st = i;
        while i < end && !ws(bytes[i]) {
            i += 1;
        }
        out.push((st, i));
    }
    out
}

fn parse_arm(s: &str, start: usize, end: usize) -> Result<Arm, RangeError> {
    let spans = token_spans(s, start, end);
    let mut arm = Arm {
        checks: Vec::new(),
        anchors: Vec::new(),
    };
    let minus_positions: Vec<usize> = spans
        .iter()
        .filter_map(|&(a, b)| if &s[a..b] == "-" { Some(a) } else { None })
        .collect();
    if let Some(&minus_at) = minus_positions.first() {
        if spans.len() != 3 || &s[spans[1].0..spans[1].1] != "-" {
            return Err(RangeError::Syntax { at: minus_at });
        }
        let (la, lb) = spans[0];
        let (ua, ub) = spans[2];
        return expand_hyphen(&s[la..lb], la, &s[ua..ub], ua, &mut arm).map(|_| arm);
    }
    for (a, b) in spans {
        let token = &s[a..b];
        let bytes = token.as_bytes();
        if token.is_empty() {
            continue;
        }
        let first = bytes[0];
        if matches!(first, b'<' | b'>' | b'=') {
            let (op, n) = if token.starts_with("<=") {
                (Op::Le, 2)
            } else if token.starts_with(">=") {
                (Op::Ge, 2)
            } else {
                (match first {
                    b'<' => Op::Lt,
                    b'>' => Op::Gt,
                    _ => Op::Eq,
                }, 1)
            };
            let operand_at = a + n;
            if operand_at >= b || !is_digit(s.as_bytes()[operand_at]) {
                return Err(RangeError::Syntax { at: a });
            }
            let v = parse_full_operand(&s[operand_at..b], operand_at)?;
            add_check(&mut arm, op, v, true);
            continue;
        }
        if matches!(first, b'^' | b'~') {
            if b == a + 1 {
                return Err(RangeError::Syntax { at: a });
            }
            expand_caret_tilde(&token[1..], a + 1, first == b'^', &mut arm)?;
            continue;
        }
        if !(is_digit(first) || matches!(first, b'*' | b'x' | b'X')) {
            return Err(RangeError::Syntax { at: a });
        }
        let p = parse_partial(token, a)?;
        expand_bare(p, a, &mut arm)?;
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i..].starts_with(b"||") {
            let left_ws = i == 0 || ws(bytes[i - 1]);
            let right_ws = i + 2 == bytes.len() || ws(bytes[i + 2]);
            if left_ws && right_ws {
                separators.push((i, i + 2));
                i += 2;
                continue;
            }
        }
        i += 1;
    }
    let mut arms = Vec::new();
    let mut start = 0;
    for &(sep_start, sep_end) in &separators {
        let arm = parse_arm(s, start, sep_start)?;
        if arm.checks.is_empty() {
            return Err(RangeError::Empty { at: sep_start });
        }
        arms.push(arm);
        start = sep_end;
    }
    let final_arm = parse_arm(s, start, s.len())?;
    if final_arm.checks.is_empty() {
        if !separators.is_empty() {
            return Err(RangeError::Empty { at: s.len() });
        }
        if s.bytes().all(ws) {
            return Err(RangeError::Empty { at: 0 });
        }
    }
    arms.push(final_arm);
    if arms.len() == 1 && arms[0].checks.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty() && !arm.anchors.contains(&version_tuple(v)) {
            return false;
        }
        arm.checks.iter().all(|c| check_matches(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let v = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        match best {
            None => best = Some(i),
            Some(j) => {
                let ord = compare_versions(v, &parsed[j]);
                if (greatest && ord == Ordering::Greater)
                    || (!greatest && ord == Ordering::Less)
                {
                    best = Some(i);
                }
            }
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}

#[cfg(test)]
mod tests {
    use super::*;

    fn pv(s: &str) -> Version {
        parse_version(s).unwrap()
    }

    fn pr(s: &str) -> Range {
        parse_range(s).unwrap()
    }

    #[test]
    fn public_version() {
        let v = pv("2.1.3-rc.1+build.7");
        assert_eq!((v.major, v.minor, v.patch), (2, 1, 3));
        assert_eq!(v.pre, vec!["rc", "1"]);
        assert_eq!(v.build, vec!["build", "7"]);
        assert_eq!(
            compare_versions(&pv("2.1.3-alpha"), &pv("2.1.3-beta")),
            Ordering::Less
        );
    }

    #[test]
    fn public_comparators() {
        let r = pr(">=2.1.0 <3.0.0");
        assert!(matches(&pv("2.5.0"), &r));
        assert!(!matches(&pv("3.0.0"), &r));
        assert!(matches(
            &pv("2.1.3-rc"),
            &pr(">=2.1.3-beta <2.1.3")
        ));
    }

    #[test]
    fn public_expanded() {
        for s in ["^2.1.0", "~2.5.0", "2.5.x", "2.1 - 2.8"] {
            assert!(matches(&pv("2.5.1"), &pr(s)));
        }
        assert!(matches(&pv("4.0.0"), &pr("2.x || 4.0.0")));
    }

    #[test]
    fn public_extrema() {
        let cs = vec![
            "2.2.0".to_string(),
            "2.8.0".to_string(),
            "3.0.0".to_string(),
        ];
        let r = pr("2.x");
        assert_eq!(max_satisfying(&cs, &r), Ok(Some("2.8.0".into())));
        assert_eq!(min_satisfying(&cs, &r), Ok(Some("2.2.0".into())));
    }
}
