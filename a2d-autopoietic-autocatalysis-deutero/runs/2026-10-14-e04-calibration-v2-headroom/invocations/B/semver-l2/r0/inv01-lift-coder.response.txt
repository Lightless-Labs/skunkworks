use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let b = s.as_bytes();
    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for (field, slot) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < b.len() && b[pos].is_ascii_digit() {
            pos += 1;
        }
        if start == pos {
            if pos == b.len() || matches!(b[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        if pos - start > 1 && b[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let mut n = 0u64;
        for &c in &b[start..pos] {
            n = n
                .checked_mul(10)
                .and_then(|x| x.checked_add((c - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        *slot = n;

        if field < 2 {
            if pos < b.len() && b[pos] == b'.' {
                pos += 1;
            } else if pos == b.len() || matches!(b[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        }
    }

    let (pre, build) = parse_version_tail(s, pos)?;
    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_version_tail(s: &str, mut pos: usize) -> Result<(Vec<String>, Vec<String>), VersionError> {
    let b = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < b.len() && b[pos] != b'-' && b[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    if pos < b.len() && b[pos] == b'-' {
        pos += 1;
        let start = pos;
        let plus = b[pos..].iter().position(|&c| c == b'+').map(|i| pos + i);
        let end = plus.unwrap_or(b.len());
        parse_identifiers(s, start, end, true, &mut pre)?;
        pos = end;
    }
    if pos < b.len() && b[pos] == b'+' {
        pos += 1;
        parse_identifiers(s, pos, b.len(), false, &mut build)?;
        pos = b.len();
    }
    if pos != b.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok((pre, build))
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    let b = s.as_bytes();
    let mut ident_start = start;
    let mut i = start;
    loop {
        if i == end || b[i] == b'.' {
            if i == ident_start {
                return Err(VersionError::EmptyIdentifier { at: ident_start });
            }
            let id = &s[ident_start..i];
            if is_pre && id.len() > 1 && id.as_bytes()[0] == b'0'
                && id.as_bytes().iter().all(u8::is_ascii_digit)
            {
                return Err(VersionError::LeadingZero { at: ident_start });
            }
            out.push(id.to_string());
            if i == end {
                break;
            }
            ident_start = i + 1;
            i += 1;
        } else {
            let c = b[i];
            if !(c.is_ascii_alphanumeric() || c == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
    }
    Ok(())
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_numeric_str(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.pre.iter().zip(&b.pre) {
        let xn = is_numeric(x);
        let yn = is_numeric(y);
        let ord = match (xn, yn) {
            (true, true) => cmp_numeric_str(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.pre.len().cmp(&b.pre.len())
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
    anchor: Option<(u64, u64, u64)>,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn ws(c: u8) -> bool {
    matches!(c, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str, start: usize, end: usize) -> Vec<(usize, usize)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && ws(b[i]) {
            i += 1;
        }
        if i == end {
            break;
        }
        let a = i;
        while i < end && !ws(b[i]) {
            i += 1;
        }
        out.push((a, i));
    }
    out
}

fn parse_operand_full(s: &str, start: usize, end: usize) -> Result<Version, RangeError> {
    parse_version(&s[start..end]).map_err(|error| RangeError::Version { at: start, error })
}

#[derive(Clone, Debug)]
enum ParsedOperand {
    Full(Version),
    Partial { fields: Vec<u64>, k: usize },
}

fn parse_operand(s: &str, start: usize, end: usize) -> Result<ParsedOperand, RangeError> {
    let operand = &s[start..end];
    let b = operand.as_bytes();
    let has_suffix = b.iter().any(|&c| c == b'-' || c == b'+');
    let fields_raw: Vec<&str> = operand.split('.').collect();
    let has_three_no_wild = fields_raw.len() >= 3
        && !fields_raw
            .iter()
            .any(|f| matches!(*f, "*" | "x" | "X"));
    if has_suffix || has_three_no_wild {
        return parse_operand_full(s, start, end).map(ParsedOperand::Full);
    }
    if fields_raw.is_empty() || fields_raw.len() > 3 {
        return Err(RangeError::Syntax { at: start });
    }
    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    for f in fields_raw {
        if f.is_empty() {
            return Err(RangeError::Syntax { at: start });
        }
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !f.bytes().all(|c| c.is_ascii_digit())
                || (f.len() > 1 && f.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at: start });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: start })?;
            fields.push(n);
        }
    }
    Ok(ParsedOperand::Partial {
        k: fields.len(),
        fields,
    })
}

fn padded(fields: &[u64]) -> Version {
    Version {
        major: fields.first().copied().unwrap_or(0),
        minor: fields.get(1).copied().unwrap_or(0),
        patch: fields.get(2).copied().unwrap_or(0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn bump_upper(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut out = Version {
        major: v.major,
        minor: v.minor,
        patch: v.patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    };
    let target = match field {
        0 => &mut out.major,
        1 => &mut out.minor,
        _ => &mut out.patch,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    if field == 0 {
        out.minor = 0;
        out.patch = 0;
    } else if field == 1 {
        out.patch = 0;
    }
    Ok(out)
}

fn add_cmp(out: &mut Vec<Comparator>, op: Op, version: Version, anchor: bool) {
    let tuple = (version.major, version.minor, version.patch);
    out.push(Comparator {
        op,
        anchor: if anchor && !version.pre.is_empty() {
            Some(tuple)
        } else {
            None
        },
        version,
    });
}

fn add_expanded(
    s: &str,
    start: usize,
    end: usize,
    token_start: usize,
    kind: u8,
    out: &mut Vec<Comparator>,
) -> Result<(), RangeError> {
    let parsed = parse_operand(s, start, end)?;
    match parsed {
        ParsedOperand::Full(v) => {
            if kind == b'^' || kind == b'~' {
                let k = 3usize;
                let lower = v.clone();
                let field = if kind == b'^' {
                    if v.major > 0 || k == 1 { 0 } else if v.minor > 0 || k == 2 { 1 } else { 2 }
                } else if k == 1 { 0 } else { 1 };
                let upper = bump_upper(&v, field, start)?;
                add_cmp(out, Op::Ge, lower, true);
                add_cmp(out, Op::Lt, upper, false);
            } else {
                add_cmp(out, Op::Eq, v, true);
            }
        }
        ParsedOperand::Partial { fields, k } => {
            if kind == b'^' || kind == b'~' {
                if k == 0 {
                    return Err(RangeError::Syntax { at: token_start });
                }
                let lower = padded(&fields);
                let field = if kind == b'^' {
                    if lower.major > 0 || k == 1 { 0 } else if lower.minor > 0 || k == 2 { 1 } else { 2 }
                } else if k == 1 { 0 } else { 1 };
                let upper = bump_upper(&lower, field, start)?;
                add_cmp(out, Op::Ge, lower, false);
                add_cmp(out, Op::Lt, upper, false);
            } else if k == 0 {
                // All wildcard bare operand.
            } else if k == 1 || k == 2 {
                let lower = padded(&fields);
                let upper = bump_upper(&lower, k - 1, start)?;
                add_cmp(out, Op::Ge, lower, false);
                add_cmp(out, Op::Lt, upper, false);
            } else {
                add_cmp(out, Op::Eq, padded(&fields), false);
            }
        }
    }
    Ok(())
}

fn parse_hyphen_arm(
    s: &str,
    tokens: &[(usize, usize)],
    out: &mut Vec<Comparator>,
) -> Result<(), RangeError> {
    let minus = tokens.iter().find(|&&(a, b)| &s[a..b] == "-").unwrap().0;
    if tokens.len() != 3 || &s[tokens[1].0..tokens[1].1] != "-" {
        return Err(RangeError::Syntax { at: minus });
    }
    let (la, lb) = tokens[0];
    let (ua, ub) = tokens[2];
    let lower = parse_operand(s, la, lb)?;
    let upper = parse_operand(s, ua, ub)?;
    let lower_v = match lower {
        ParsedOperand::Full(v) => v,
        ParsedOperand::Partial { fields, k } if k > 0 => padded(&fields),
        _ => return Err(RangeError::Syntax { at: la }),
    };
    let upper_v = match upper {
        ParsedOperand::Full(v) => {
            add_cmp(out, Op::Le, v, true);
            None
        }
        ParsedOperand::Partial { fields, k } if k > 0 => Some((padded(&fields), k)),
        _ => return Err(RangeError::Syntax { at: ua }),
    };
    add_cmp(out, Op::Ge, lower_v, true);
    if let Some((v, k)) = upper_v {
        let upper = bump_upper(&v, k - 1, ua)?;
        add_cmp(out, Op::Lt, upper, false);
    }
    Ok(())
}

fn parse_arm(s: &str, start: usize, end: usize) -> Result<Vec<Comparator>, RangeError> {
    let tokens = tokenize(s, start, end);
    let mut out = Vec::new();
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: start });
    }
    if tokens.iter().any(|&(a, b)| &s[a..b] == "-") {
        parse_hyphen_arm(s, &tokens, &mut out)?;
        return Ok(out);
    }

    for &(a, b) in &tokens {
        let token = &s[a..b];
        let tb = token.as_bytes();
        if tb[0] == b'^' || tb[0] == b'~' {
            if tb.len() == 1 {
                return Err(RangeError::Syntax { at: a });
            }
            add_expanded(s, a + 1, b, a, tb[0], &mut out)?;
            continue;
        }

        let (op, operand_start) = if tb.starts_with(b">=") {
            (Some(Op::Ge), a + 2)
        } else if tb.starts_with(b"<=") {
            (Some(Op::Le), a + 2)
        } else if tb[0] == b'>' {
            (Some(Op::Gt), a + 1)
        } else if tb[0] == b'<' {
            (Some(Op::Lt), a + 1)
        } else if tb[0] == b'=' {
            (Some(Op::Eq), a + 1)
        } else {
            (None, a)
        };
        if let Some(op) = op {
            if operand_start == b || !s.as_bytes()[operand_start].is_ascii_digit() {
                return Err(RangeError::Syntax { at: a });
            }
            let v = parse_operand_full(s, operand_start, b)?;
            add_cmp(&mut out, op, v, true);
            continue;
        }

        if !(tb[0].is_ascii_digit() || matches!(tb[0], b'*' | b'x' | b'X')) {
            return Err(RangeError::Syntax { at: a });
        }
        add_expanded(s, a, b, a, 0, &mut out)?;
    }
    Ok(out)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b = s.as_bytes();
    let mut arms = Vec::new();
    let mut i = 0usize;
    let mut arm_start = 0usize;
    let mut saw_any = false;

    while i < b.len() {
        if b[i] == b'|' && i + 1 < b.len() && b[i + 1] == b'|' {
            let sep = i;
            let left = parse_arm(s, arm_start, sep)?;
            arms.push(left);
            saw_any = true;
            i += 2;
            arm_start = i;
        } else {
            i += 1;
        }
    }
    let final_arm = parse_arm(s, arm_start, b.len());
    match final_arm {
        Ok(a) => arms.push(a),
        Err(RangeError::Empty { .. }) if saw_any => {
            return Err(RangeError::Empty { at: b.len() });
        }
        Err(e) => return Err(e),
    }

    if !saw_any && arms.len() == 1 && arms[0].is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    Ok(Range { arms })
}

fn cmp_matches(v: &Version, c: &Comparator) -> bool {
    let ord = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => ord == Ordering::Equal,
        Op::Lt => ord == Ordering::Less,
        Op::Le => ord != Ordering::Greater,
        Op::Gt => ord == Ordering::Greater,
        Op::Ge => ord != Ordering::Less,
    }
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !arm.iter().all(|c| cmp_matches(v, c)) {
            return false;
        }
        if v.pre.is_empty() {
            return true;
        }
        let tuple = (v.major, v.minor, v.patch);
        arm.iter().any(|c| c.anchor == Some(tuple))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        parsed.push(
            parse_version(s).map_err(|error| CandidateError { index, error })?,
        );
    }

    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        match best {
            None => best = Some(i),
            Some(j) => {
                let ord = compare_versions(v, &parsed[j]);
                if (want_max && ord == Ordering::Greater)
                    || (!want_max && ord == Ordering::Less)
                {
                    best = Some(i);
                }
            }
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}
