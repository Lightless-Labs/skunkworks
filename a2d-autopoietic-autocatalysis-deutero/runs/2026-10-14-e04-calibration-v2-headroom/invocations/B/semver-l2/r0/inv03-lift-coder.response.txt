use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn core_missing_start(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.') | Some(b'-') | Some(b'+'))
}

fn parse_core_field(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        if core_missing_start(bytes.get(*pos).copied()) {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn validate_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut id_start = start;

    loop {
        let mut i = id_start;
        while i < end && bytes[i] != b'.' {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        let id = &s[id_start..i];
        if is_pre && id.bytes().all(is_digit) && id.len() > 1 && id.as_bytes()[0] == b'0' {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(id.to_string());

        if i == end {
            break;
        }
        id_start = i + 1;
        if id_start > end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0;
    let major = parse_core_field(s, &mut pos)?;

    if bytes.get(pos) != Some(&b'.') {
        if core_missing_start(bytes.get(pos).copied()) {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let minor = parse_core_field(s, &mut pos)?;
    if bytes.get(pos) != Some(&b'.') {
        if core_missing_start(bytes.get(pos).copied()) {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let patch = parse_core_field(s, &mut pos)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();

    match bytes.get(pos).copied() {
        None => {}
        Some(b'-') => {
            let pre_start = pos + 1;
            let pre_end = bytes[pre_start..]
                .iter()
                .position(|&b| b == b'+')
                .map(|n| pre_start + n)
                .unwrap_or(bytes.len());

            pre = validate_identifiers(s, pre_start, pre_end, true)?;

            if pre_end < bytes.len() {
                let build_start = pre_end + 1;
                build = validate_identifiers(s, build_start, bytes.len(), false)?;
            }
        }
        Some(b'+') => {
            let build_start = pos + 1;
            build = validate_identifiers(s, build_start, bytes.len(), false)?;
        }
        Some(_) => return Err(VersionError::InvalidChar { at: pos }),
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn core_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let an = is_numeric_identifier(a);
    let bn = is_numeric_identifier(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core_tuple(a).cmp(&core_tuple(b)).then_with(|| {
        match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ordering = compare_identifiers(x, y);
                    if ordering != Ordering::Equal {
                        return ordering;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        }
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Arm {
    comparisons: Vec<Comparison>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Comparison {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn comparison_matches(candidate: &Version, comparison: &Comparison) -> bool {
    let ordering = compare_versions(candidate, &comparison.version);
    match comparison.op {
        Op::Eq => ordering == Ordering::Equal,
        Op::Lt => ordering == Ordering::Less,
        Op::Le => ordering != Ordering::Greater,
        Op::Gt => ordering == Ordering::Greater,
        Op::Ge => ordering != Ordering::Less,
    }
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

#[derive(Clone, Debug)]
struct Token {
    text: String,
    start: usize,
}

fn tokenize(s: &str) -> Vec<Token> {
    let bytes = s.as_bytes();
    let mut tokens = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && whitespace(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !whitespace(bytes[i]) {
            i += 1;
        }
        tokens.push(Token {
            text: s[start..i].to_string(),
            start,
        });
    }
    tokens
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    numeric_count: usize,
    full: Option<Version>,
}

fn full_or_partial(text: &str, start: usize) -> Result<Partial, RangeError> {
    let has_sign = text.contains('-') || text.contains('+');
    let fields: Vec<&str> = text.split('.').collect();
    let all_numeric = fields.iter().all(|f| !f.is_empty() && f.bytes().all(is_digit));
    if has_sign || (fields.len() >= 3 && all_numeric) {
        return parse_version(text)
            .map(|v| Partial {
                fields: vec![Some(v.major), Some(v.minor), Some(v.patch)],
                numeric_count: 3,
                full: Some(v),
            })
            .map_err(|error| RangeError::Version { at: start, error });
    }

    if fields.len() > 3 {
        return Err(RangeError::Syntax { at: start });
    }
    let mut parsed = Vec::new();
    let mut numeric_count = 0;
    let mut wildcard_seen = false;
    for field in fields {
        if field.is_empty() {
            return Err(RangeError::Syntax { at: start });
        }
        if field == "*" || field == "x" || field == "X" {
            wildcard_seen = true;
            parsed.push(None);
        } else {
            if wildcard_seen
                || !field.bytes().all(is_digit)
                || (field.len() > 1 && field.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at: start });
            }
            let value = field.parse::<u64>().map_err(|_| RangeError::Syntax { at: start })?;
            parsed.push(Some(value));
            numeric_count += 1;
        }
    }
    Ok(Partial {
        fields: parsed,
        numeric_count,
        full: None,
    })
}

fn padded(partial: &Partial) -> Version {
    let mut nums = [0u64; 3];
    for (i, field) in partial.fields.iter().take(3).enumerate() {
        if let Some(n) = field {
            nums[i] = *n;
        }
    }
    let mut v = Version {
        major: nums[0],
        minor: nums[1],
        patch: nums[2],
        pre: Vec::new(),
        build: Vec::new(),
    };
    if let Some(full) = &partial.full {
        v = full.clone();
    }
    v
}

fn upper_bound(
    lower: &Version,
    index: usize,
    at: usize,
    major: bool,
    minor: bool,
    patch: bool,
) -> Result<Version, RangeError> {
    let mut upper = Version {
        major: lower.major,
        minor: lower.minor,
        patch: lower.patch,
        pre: vec!["0".to_string()],
        build: Vec::new(),
    };
    if major {
        upper.major = upper.major.checked_add(1).ok_or(RangeError::Syntax { at })?;
        upper.minor = 0;
        upper.patch = 0;
    } else if minor {
        upper.minor = upper.minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
        upper.patch = 0;
    } else if patch {
        upper.patch = upper.patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
    }
    let _ = index;
    Ok(upper)
}

fn add(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(core_tuple(&version));
    }
    arm.comparisons.push(Comparison { op, version });
}

fn parse_hyphen_arm(tokens: &[Token], arm: &mut Arm) -> Result<(), RangeError> {
    let minus = tokens.iter().find(|t| t.text == "-").unwrap();
    if tokens.len() != 3 || tokens[1].text != "-" {
        return Err(RangeError::Syntax { at: minus.start });
    }

    let lower = full_or_partial(&tokens[0].text, tokens[0].start)?;
    let upper = full_or_partial(&tokens[2].text, tokens[2].start)?;
    if lower.numeric_count == 0 {
        return Err(RangeError::Syntax { at: tokens[0].start });
    }
    if upper.numeric_count == 0 {
        return Err(RangeError::Syntax { at: tokens[2].start });
    }

    let lower_v = padded(&lower);
    add(arm, Op::Ge, lower_v, lower.full.is_some());

    if let Some(full) = upper.full {
        add(arm, Op::Le, full.clone(), true);
    } else {
        let upper_v = padded(&upper);
        let endpoint = match upper.numeric_count {
            1 => upper_bound(&upper_v, 0, tokens[2].start, true, false, false)?,
            _ => upper_bound(&upper_v, 0, tokens[2].start, false, true, false)?,
        };
        add(arm, Op::Lt, endpoint, false);
    }
    Ok(())
}

fn parse_arm(tokens: &[Token], arm: &mut Arm) -> Result<(), RangeError> {
    if tokens.is_empty() {
        return Ok(());
    }
    if tokens.iter().any(|t| t.text == "-") {
        return parse_hyphen_arm(tokens, arm);
    }

    for token in tokens {
        let text = token.text.as_str();
        let (prefix, operand, operand_start) = if let Some(rest) = text.strip_prefix('^') {
            ("^", rest, token.start + 1)
        } else if let Some(rest) = text.strip_prefix('~') {
            ("~", rest, token.start + 1)
        } else if let Some(rest) = text.strip_prefix(">=") {
            (">=", rest, token.start + 2)
        } else if let Some(rest) = text.strip_prefix("<=") {
            ("<=", rest, token.start + 2)
        } else if let Some(rest) = text.strip_prefix('>') {
            (">", rest, token.start + 1)
        } else if let Some(rest) = text.strip_prefix('<') {
            ("<", rest, token.start + 1)
        } else if let Some(rest) = text.strip_prefix('=') {
            ("=", rest, token.start + 1)
        } else {
            ("", text, token.start)
        };

        if operand.is_empty()
            || !operand
                .as_bytes()
                .first()
                .is_some_and(|b| b.is_ascii_digit() || matches!(b, b'*' | b'x' | b'X'))
        {
            return Err(RangeError::Syntax { at: token.start });
        }

        match prefix {
            ">" | ">=" | "<" | "<=" | "=" => {
                let version = parse_version(operand)
                    .map_err(|error| RangeError::Version { at: operand_start, error })?;
                let op = match prefix {
                    ">" => Op::Gt,
                    ">=" => Op::Ge,
                    "<" => Op::Lt,
                    "<=" => Op::Le,
                    _ => Op::Eq,
                };
                add(arm, op, version, true);
            }
            "^" | "~" => {
                let partial = full_or_partial(operand, operand_start)?;
                if partial.numeric_count == 0 {
                    return Err(RangeError::Syntax { at: token.start });
                }
                let lower = padded(&partial);
                add(arm, Op::Ge, lower.clone(), partial.full.is_some());
                let k = partial.numeric_count;
                let upper = if prefix == "^" {
                    if lower.major > 0 || k == 1 {
                        upper_bound(&lower, 0, operand_start, true, false, false)?
                    } else if lower.minor > 0 || k == 2 {
                        upper_bound(&lower, 0, operand_start, false, true, false)?
                    } else {
                        upper_bound(&lower, 0, operand_start, false, false, true)?
                    }
                } else if k == 1 {
                    upper_bound(&lower, 0, operand_start, true, false, false)?
                } else {
                    upper_bound(&lower, 0, operand_start, false, true, false)?
                };
                add(arm, Op::Lt, upper, false);
            }
            _ => {
                let partial = full_or_partial(operand, operand_start)?;
                if let Some(full) = partial.full {
                    add(arm, Op::Eq, full, true);
                } else if partial.numeric_count == 0 {
                    // Wildcard-only bare operands match all versions.
                } else {
                    let lower = padded(&partial);
                    add(arm, Op::Ge, lower, false);
                    let upper = match partial.numeric_count {
                        1 => upper_bound(&padded(&partial), 0, operand_start, true, false, false)?,
                        2 => upper_bound(&padded(&partial), 0, operand_start, false, true, false)?,
                        _ => continue,
                    };
                    add(arm, Op::Lt, upper, false);
                }
            }
        }
    }
    Ok(())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut arm_tokens = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].text == "||" {
            if arm_tokens.is_empty() {
                return Err(RangeError::Empty { at: tokens[i].start });
            }
            let mut arm = Arm {
                comparisons: Vec::new(),
                anchors: Vec::new(),
            };
            parse_arm(&arm_tokens, &mut arm)?;
            arms.push(arm);
            arm_tokens.clear();
            i += 1;
            if i == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            arm_tokens.push(tokens[i].clone());
            i += 1;
        }
    }
    if arm_tokens.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    let mut arm = Arm {
        comparisons: Vec::new(),
        anchors: Vec::new(),
    };
    parse_arm(&arm_tokens, &mut arm)?;
    arms.push(arm);
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        arm.comparisons.iter().all(|c| comparison_matches(v, c))
            && (v.pre.is_empty() || arm.anchors.iter().any(|&tuple| tuple == core_tuple(v)))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some(index),
            Some(current) => {
                let ordering = compare_versions(version, &parsed[current]);
                if (want_max && ordering == Ordering::Greater)
                    || (!want_max && ordering == Ordering::Less)
                {
                    best = Some(index);
                }
            }
        }
    }
    Ok(best.map(|index| candidates[index].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}
