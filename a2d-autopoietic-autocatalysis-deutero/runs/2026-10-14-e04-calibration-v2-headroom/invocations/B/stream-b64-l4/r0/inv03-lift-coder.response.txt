use std::fmt;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped {
        width: usize,
        ending: LineEnding,
    },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
    pub strict: bool,
    pub max_symbols: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
    NonCanonical { offset: usize },
    MixedAlphabet { offset: usize },
    Overlong { offset: usize, limit: usize },
}

impl fmt::Display for Error {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "{self:?}")
    }
}

impl std::error::Error for Error {}

fn validate_options(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn encode_digit(n: u8, alphabet: Alphabet) -> u8 {
    match n {
        0..=25 => b'A' + n,
        26..=51 => b'a' + n - 26,
        52..=61 => b'0' + n - 52,
        62 => match alphabet {
            Alphabet::Standard => b'+',
            Alphabet::UrlSafe => b'-',
        },
        _ => match alphabet {
            Alphabet::Standard => b'/',
            Alphabet::UrlSafe => b'_',
        },
    }
}

fn digit_value(b: u8, alphabet: Alphabet, strict: bool, offset: usize) -> Result<u8, Error> {
    let value = match b {
        b'A'..=b'Z' => Some(b - b'A'),
        b'a'..=b'z' => Some(b - b'a' + 26),
        b'0'..=b'9' => Some(b - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        b'+' | b'/' | b'-' | b'_' if strict => {
            return Err(Error::MixedAlphabet { offset });
        }
        _ => None,
    };
    value.ok_or(Error::InvalidByte { offset, byte: b })
}

fn wrap_separator(ending: LineEnding) -> &'static str {
    match ending {
        LineEnding::Lf => "\n",
        LineEnding::CrLf => "\r\n",
    }
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    symbols_in_row: usize,
    closed: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            tail: Vec::new(),
            symbols_in_row: 0,
            closed: false,
        })
    }

    fn emit_symbols(&mut self, symbols: &[u8], out: &mut String) {
        for &symbol in symbols {
            if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
                if self.symbols_in_row == width {
                    out.push_str(wrap_separator(ending));
                    self.symbols_in_row = 0;
                }
            }
            out.push(symbol as char);
            self.symbols_in_row += 1;
        }
    }

    fn encode_triple(&mut self, bytes: &[u8], out: &mut String) {
        let a = bytes[0];
        let b = bytes[1];
        let c = bytes[2];
        let symbols = [
            encode_digit(a >> 2, self.options.alphabet),
            encode_digit(((a & 0x03) << 4) | (b >> 4), self.options.alphabet),
            encode_digit(((b & 0x0f) << 2) | (c >> 6), self.options.alphabet),
            encode_digit(c & 0x3f, self.options.alphabet),
        ];
        self.emit_symbols(&symbols, out);
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut out = String::new();
        let mut all = std::mem::take(&mut self.tail);
        all.extend_from_slice(input);

        let full_len = all.len() / 3 * 3;
        for chunk in all[..full_len].chunks_exact(3) {
            self.encode_triple(chunk, &mut out);
        }
        self.tail.extend_from_slice(&all[full_len..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        self.closed = true;
        let mut out = String::new();
        match self.tail.len() {
            0 => {}
            1 => {
                let a = self.tail[0];
                let symbols = [
                    encode_digit(a >> 2, self.options.alphabet),
                    encode_digit((a & 0x03) << 4, self.options.alphabet),
                ];
                self.emit_symbols(&symbols, &mut out);
                if self.options.padding != Padding::Forbidden {
                    self.emit_symbols(b"==", &mut out);
                }
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                let symbols = [
                    encode_digit(a >> 2, self.options.alphabet),
                    encode_digit(((a & 0x03) << 4) | (b >> 4), self.options.alphabet),
                    encode_digit((b & 0x0f) << 2, self.options.alphabet),
                ];
                self.emit_symbols(&symbols, &mut out);
                if self.options.padding != Padding::Forbidden {
                    self.emit_symbols(b"=", &mut out);
                }
            }
            _ => unreachable!(),
        }
        self.tail.clear();
        Ok(out)
    }
}

pub struct Decoder {
    options: Options,
    offset: usize,
    symbols_seen: usize,
    group: Vec<(u8, usize)>,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    wrapped_symbols: usize,
    wrap_state: WrapState,
    finished: bool,
    sticky: Option<Error>,
}

#[derive(Clone, Copy)]
enum WrapState {
    Row,
    NeedLf { cr_offset: usize, first_offset: usize },
    NeedSymbol { first_offset: usize },
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            offset: 0,
            symbols_seen: 0,
            group: Vec::with_capacity(4),
            terminal_padding: false,
            pending_second_padding: None,
            wrapped_symbols: 0,
            wrap_state: WrapState::Row,
            finished: false,
            sticky: None,
        })
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky = Some(error.clone());
        Err(error)
    }

    fn append_group(&mut self, output: &mut Vec<u8>) -> Result<(), Error> {
        let g = &self.group;
        if g.len() != 4 {
            return Ok(());
        }

        let (a, ao) = g[0];
        let (b, bo) = g[1];
        let (c, co) = g[2];
        let (d, _) = g[3];

        if a == b'=' || b == b'=' {
            return Err(Error::BadPadding {
                offset: if a == b'=' { ao } else { bo },
            });
        }

        let av = digit_value(a, self.options.alphabet, self.options.strict, ao)?;
        let bv = digit_value(b, self.options.alphabet, self.options.strict, bo)?;

        if c == b'=' {
            if d != b'=' {
                return Err(Error::BadPadding { offset: g[3].1 });
            }
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset: co });
            }
            if self.options.strict && (bv & 0x0f) != 0 {
                return Err(Error::NonCanonical { offset: bo });
            }
            output.push((av << 2) | (bv >> 4));
            self.terminal_padding = true;
        } else {
            let cv = digit_value(c, self.options.alphabet, self.options.strict, co)?;
            if d == b'=' {
                if self.options.padding == Padding::Forbidden {
                    return Err(Error::BadPadding { offset: g[3].1 });
                }
                if self.options.strict && (cv & 0x03) != 0 {
                    return Err(Error::NonCanonical { offset: co });
                }
                output.push((av << 2) | (bv >> 4));
                output.push((bv << 4) | (cv >> 2));
                self.terminal_padding = true;
            } else {
                let dv = digit_value(d, self.options.alphabet, self.options.strict, g[3].1)?;
                output.push((av << 2) | (bv >> 4));
                output.push((bv << 4) | (cv >> 2));
                output.push((cv << 6) | dv);
            }
        }
        self.group.clear();
        self.pending_second_padding = None;
        Ok(())
    }

    fn is_ignored_whitespace(&self, byte: u8) -> bool {
        self.options.whitespace == Whitespace::IgnoreAscii
            && matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
    }

    fn process_symbol(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if let Some(limit) = self.options.max_symbols {
            if self.symbols_seen >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.symbols_seen += 1;

        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }
        if self.pending_second_padding.is_some() && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }
        if byte == b'=' {
            let position = self.group.len();
            if position < 2 {
                return Err(Error::BadPadding { offset });
            }
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            self.group.push((byte, offset));
            if position == 2 {
                self.pending_second_padding = Some(offset);
            } else {
                self.append_group(output)?;
            }
            return Ok(());
        }

        if self.group.len() >= 2 && self.group[2].0 == b'=' {
            return Err(Error::BadPadding { offset });
        }

        let val = digit_value(byte, self.options.alphabet, self.options.strict, offset)?;
        self.group.push((byte, offset));
        if self.group.len() == 4 {
            let _ = val;
            self.append_group(output)?;
        }
        Ok(())
    }

    fn process_wrapped(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        let (width, ending) = match self.options.whitespace {
            Whitespace::Wrapped { width, ending } => (width, ending),
            _ => unreachable!(),
        };

        match self.wrap_state {
            WrapState::NeedLf { cr_offset, .. } => {
                if byte != b'\n' {
                    return Err(Error::BadWrap { offset });
                }
                self.wrap_state = WrapState::NeedSymbol {
                    first_offset: cr_offset,
                };
                Ok(())
            }
            WrapState::NeedSymbol { first_offset } => {
                if matches!(ending, LineEnding::Lf) && byte == b'\n'
                    || matches!(ending, LineEnding::CrLf) && byte == b'\r'
                {
                    return Err(Error::BadWrap { offset });
                }
                self.wrapped_symbols = 0;
                self.wrap_state = WrapState::Row;
                let _ = first_offset;
                self.process_symbol(byte, offset, output)
            }
            WrapState::Row => {
                if self.wrapped_symbols == width {
                    match ending {
                        LineEnding::Lf if byte == b'\n' => {
                            self.wrap_state = WrapState::NeedSymbol {
                                first_offset: offset,
                            };
                            return Ok(());
                        }
                        LineEnding::CrLf if byte == b'\r' => {
                            self.wrap_state = WrapState::NeedLf {
                                cr_offset: offset,
                                first_offset: offset,
                            };
                            return Ok(());
                        }
                        _ => return Err(Error::BadWrap { offset }),
                    }
                }

                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    return Err(Error::BadWrap { offset });
                }
                self.wrapped_symbols += 1;
                self.process_symbol(byte, offset, output)
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;

            let result = match self.options.whitespace {
                Whitespace::None => self.process_symbol(byte, offset, &mut output),
                Whitespace::IgnoreAscii => {
                    if self.is_ignored_whitespace(byte) {
                        Ok(())
                    } else {
                        self.process_symbol(byte, offset, &mut output)
                    }
                }
                Whitespace::Wrapped { .. } => {
                    self.process_wrapped(byte, offset, &mut output)
                }
            };

            if let Err(error) = result {
                return self.fail(error);
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        if let Whitespace::Wrapped { ending, .. } = self.options.whitespace {
            match self.wrap_state {
                WrapState::NeedLf { cr_offset, .. } => {
                    return self.fail(Error::BadWrap { offset: cr_offset });
                }
                WrapState::NeedSymbol { first_offset } => {
                    return self.fail(Error::BadWrap {
                        offset: first_offset,
                    });
                }
                WrapState::Row => {
                    let _ = ending;
                }
            }
        }

        if let Some(offset) = self.pending_second_padding {
            return self.fail(Error::BadPadding { offset });
        }

        if !self.group.is_empty() {
            match self.options.padding {
                Padding::Required => {
                    return self.fail(Error::Truncated {
                        offset: self.group[0].1,
                    });
                }
                Padding::Forbidden | Padding::Optional => {
                    if self.group.iter().any(|(b, _)| *b == b'=') {
                        return self.fail(Error::BadPadding {
                            offset: self
                                .group
                                .iter()
                                .find(|(b, _)| *b == b'=')
                                .map(|(_, o)| *o)
                                .unwrap(),
                        });
                    }
                    if self.group.len() == 1 {
                        return self.fail(Error::Truncated {
                            offset: self.group[0].1,
                        });
                    }
                }
            }
        }

        let mut output = Vec::new();
        if self.group.len() == 2 {
            let (a, ao) = self.group[0];
            let (b, bo) = self.group[1];
            let av = match digit_value(a, self.options.alphabet, self.options.strict, ao) {
                Ok(v) => v,
                Err(e) => return self.fail(e),
            };
            let bv = match digit_value(b, self.options.alphabet, self.options.strict, bo) {
                Ok(v) => v,
                Err(e) => return self.fail(e),
            };
            if self.options.strict && (bv & 0x0f) != 0 {
                return self.fail(Error::NonCanonical { offset: bo });
            }
            output.push((av << 2) | (bv >> 4));
        } else if self.group.len() == 3 {
            let (a, ao) = self.group[0];
            let (b, bo) = self.group[1];
            let (c, co) = self.group[2];
            let av = match digit_value(a, self.options.alphabet, self.options.strict, ao) {
                Ok(v) => v,
                Err(e) => return self.fail(e),
            };
            let bv = match digit_value(b, self.options.alphabet, self.options.strict, bo) {
                Ok(v) => v,
                Err(e) => return self.fail(e),
            };
            let cv = match digit_value(c, self.options.alphabet, self.options.strict, co) {
                Ok(v) => v,
                Err(e) => return self.fail(e),
            };
            if self.options.strict && (cv & 0x03) != 0 {
                return self.fail(Error::NonCanonical { offset: co });
            }
            output.push((av << 2) | (bv >> 4));
            output.push((bv << 4) | (cv >> 2));
        }

        self.group.clear();
        Ok(output)
    }
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    let mut encoder = Encoder::new(options)?;
    let mut output = encoder.push(input)?;
    output.push_str(&encoder.finish()?);
    Ok(output)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new(options)?;
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}
