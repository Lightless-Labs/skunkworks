#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
    pub strict: bool,
    pub max_symbols: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
    NonCanonical { offset: usize },
    MixedAlphabet { offset: usize },
    Overlong { offset: usize, limit: usize },
}

fn validate_options(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_byte(alphabet: Alphabet, value: u8) -> u8 {
    const BASE: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    match (alphabet, value) {
        (Alphabet::UrlSafe, 62) => b'-',
        (Alphabet::UrlSafe, 63) => b'_',
        _ => BASE[value as usize],
    }
}

fn lookup_digit(alphabet: Alphabet, byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn foreign_alphabet_byte(alphabet: Alphabet, byte: u8) -> bool {
    match alphabet {
        Alphabet::Standard => byte == b'-' || byte == b'_',
        Alphabet::UrlSafe => byte == b'+' || byte == b'/',
    }
}

fn is_ignored_ascii(byte: u8) -> bool {
    matches!(byte, 9 | 10 | 11 | 12 | 13 | 32)
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate_options(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut output = encoder.push(input)?;
    output.push_str(&encoder.finish()?);
    Ok(output)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate_options(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    options: Options,
    pending: Vec<u8>,
    symbols_in_row: usize,
    closed: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            pending: Vec::with_capacity(2),
            symbols_in_row: 0,
            closed: false,
        })
    }

    fn append_symbol(&mut self, output: &mut String, symbol: u8) {
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            if self.symbols_in_row == width {
                match ending {
                    LineEnding::Lf => output.push('\n'),
                    LineEnding::CrLf => output.push_str("\r\n"),
                }
                self.symbols_in_row = 0;
            }
            output.push(symbol as char);
            self.symbols_in_row += 1;
        } else {
            output.push(symbol as char);
        }
    }

    fn append_triplet(&mut self, output: &mut String, bytes: &[u8]) {
        let a = bytes[0];
        let b = bytes[1];
        let c = bytes[2];
        self.append_symbol(output, alphabet_byte(self.options.alphabet, a >> 2));
        self.append_symbol(
            output,
            alphabet_byte(self.options.alphabet, ((a & 0x03) << 4) | (b >> 4)),
        );
        self.append_symbol(
            output,
            alphabet_byte(self.options.alphabet, ((b & 0x0f) << 2) | (c >> 6)),
        );
        self.append_symbol(output, alphabet_byte(self.options.alphabet, c & 0x3f));
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if !self.pending.is_empty() {
            while self.pending.len() < 3 && index < input.len() {
                self.pending.push(input[index]);
                index += 1;
            }
            if self.pending.len() == 3 {
                let bytes = [
                    self.pending[0],
                    self.pending[1],
                    self.pending[2],
                ];
                self.append_triplet(&mut output, &bytes);
                self.pending.clear();
            }
        }

        while index + 3 <= input.len() {
            self.append_triplet(&mut output, &input[index..index + 3]);
            index += 3;
        }

        self.pending.extend_from_slice(&input[index..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        match self.pending.len() {
            0 => {}
            1 => {
                let a = self.pending[0];
                self.append_symbol(&mut output, alphabet_byte(self.options.alphabet, a >> 2));
                self.append_symbol(
                    &mut output,
                    alphabet_byte(self.options.alphabet, (a & 0x03) << 4),
                );
                if self.options.padding != Padding::Forbidden {
                    self.append_symbol(&mut output, b'=');
                    self.append_symbol(&mut output, b'=');
                }
            }
            2 => {
                let a = self.pending[0];
                let b = self.pending[1];
                self.append_symbol(&mut output, alphabet_byte(self.options.alphabet, a >> 2));
                self.append_symbol(
                    &mut output,
                    alphabet_byte(self.options.alphabet, ((a & 0x03) << 4) | (b >> 4)),
                );
                self.append_symbol(
                    &mut output,
                    alphabet_byte(self.options.alphabet, (b & 0x0f) << 2),
                );
                if self.options.padding != Padding::Forbidden {
                    self.append_symbol(&mut output, b'=');
                }
            }
            _ => unreachable!(),
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

#[derive(Clone, Copy)]
enum WrapState {
    Row,
    NeedLf { cr_offset: usize },
    NeedSymbol { separator_offset: usize },
}

pub struct Decoder {
    options: Options,
    offset: usize,
    symbol_count: usize,
    group: [u8; 4],
    group_offsets: [usize; 4],
    group_len: usize,
    pending_second_padding: Option<usize>,
    terminal_padding: bool,
    wrap_symbols: usize,
    wrap_state: WrapState,
    closed: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            offset: 0,
            symbol_count: 0,
            group: [0; 4],
            group_offsets: [0; 4],
            group_len: 0,
            pending_second_padding: None,
            terminal_padding: false,
            wrap_symbols: 0,
            wrap_state: WrapState::Row,
            closed: false,
            sticky_error: None,
        })
    }

    fn emit_completed_group(&mut self, output: &mut Vec<u8>) -> Result<(), Error> {
        let g = self.group;
        let o = self.group_offsets;

        if g[0] == 64 || g[1] == 64 {
            let index = if g[0] == 64 { 0 } else { 1 };
            return Err(Error::BadPadding { offset: o[index] });
        }

        let a = g[0];
        let b = g[1];
        output.push((a << 2) | (b >> 4));

        if g[2] == 64 {
            if g[3] != 64 {
                return Err(Error::BadPadding { offset: o[3] });
            }
            if self.options.strict && (b & 0x0f) != 0 {
                return Err(Error::NonCanonical { offset: o[1] });
            }
            self.terminal_padding = true;
        } else {
            let c = g[2];
            output.push((b << 4) | (c >> 2));

            if g[3] == 64 {
                if self.options.strict && (c & 0x03) != 0 {
                    return Err(Error::NonCanonical { offset: o[2] });
                }
                self.terminal_padding = true;
            } else {
                output.push((c << 6) | g[3]);
            }
        }

        self.group_len = 0;
        Ok(())
    }

    fn process_symbol(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if let Some(limit) = self.options.max_symbols {
            if self.symbol_count >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.symbol_count += 1;

        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.pending_second_padding.is_some() && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        if self.group_len <= 1 && byte == b'=' {
            return Err(Error::BadPadding { offset });
        }

        let digit = if byte == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            if self.group_len == 2 {
                self.pending_second_padding = Some(offset);
            }
            64
        } else if let Some(digit) = lookup_digit(self.options.alphabet, byte) {
            digit
        } else if self.options.strict && foreign_alphabet_byte(self.options.alphabet, byte) {
            return Err(Error::MixedAlphabet { offset });
        } else {
            return Err(Error::InvalidByte { offset, byte });
        };

        self.group[self.group_len] = digit;
        self.group_offsets[self.group_len] = offset;
        self.group_len += 1;

        if self.group_len == 4 {
            if self.group[2] == 64 && self.group[3] != 64 {
                return Err(Error::BadPadding {
                    offset: self.group_offsets[3],
                });
            }
            if self.group[2] == 64 && self.pending_second_padding.is_none() {
                return Err(Error::BadPadding {
                    offset: self.group_offsets[2],
                });
            }
            self.pending_second_padding = None;
            self.emit_completed_group(output)?;
        }

        Ok(())
    }

    fn process_wrapped_byte(
        &mut self,
        byte: u8,
        offset: usize,
    ) -> Result<bool, Error> {
        let (width, ending) = match self.options.whitespace {
            Whitespace::Wrapped { width, ending } => (width, ending),
            _ => unreachable!(),
        };

        match self.wrap_state {
            WrapState::NeedLf { cr_offset } => {
                if byte != b'\n' {
                    return Err(Error::BadWrap { offset });
                }
                self.wrap_state = WrapState::NeedSymbol {
                    separator_offset: cr_offset,
                };
                Ok(true)
            }
            WrapState::NeedSymbol { .. } => {
                self.wrap_state = WrapState::Row;
                self.wrap_symbols = 0;
                self.process_symbol(byte, offset, &mut Vec::new())?;
                // The caller must process symbols into its actual output.
                unreachable!()
            }
            WrapState::Row => {
                if self.wrap_symbols == width {
                    match ending {
                        LineEnding::Lf if byte == b'\n' => {
                            self.wrap_state = WrapState::NeedSymbol {
                                separator_offset: offset,
                            };
                            Ok(true)
                        }
                        LineEnding::CrLf if byte == b'\r' => {
                            self.wrap_state = WrapState::NeedLf { cr_offset: offset };
                            Ok(true)
                        }
                        _ => Err(Error::BadWrap { offset }),
                    }
                } else if matches!(byte, b'\n' | b'\r') {
                    Err(Error::BadWrap { offset })
                } else {
                    Ok(false)
                }
            }
        }
    }

    fn scan_byte(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => self.process_symbol(byte, offset, output),
            Whitespace::IgnoreAscii => {
                if is_ignored_ascii(byte) {
                    Ok(())
                } else {
                    self.process_symbol(byte, offset, output)
                }
            }
            Whitespace::Wrapped { width, ending } => {
                match self.wrap_state {
                    WrapState::NeedLf { cr_offset } => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.wrap_state = WrapState::NeedSymbol {
                            separator_offset: cr_offset,
                        };
                        Ok(())
                    }
                    WrapState::NeedSymbol { separator_offset } => {
                        self.wrap_state = WrapState::Row;
                        self.wrap_symbols = 0;
                        self.process_symbol(byte, offset, output)?;
                        if matches!(self.wrap_state, WrapState::Row) {
                            self.wrap_symbols = 1;
                        }
                        let _ = separator_offset;
                        Ok(())
                    }
                    WrapState::Row => {
                        if self.wrap_symbols == width {
                            match ending {
                                LineEnding::Lf if byte == b'\n' => {
                                    self.wrap_state = WrapState::NeedSymbol {
                                        separator_offset: offset,
                                    };
                                    Ok(())
                                }
                                LineEnding::CrLf if byte == b'\r' => {
                                    self.wrap_state = WrapState::NeedLf { cr_offset: offset };
                                    Ok(())
                                }
                                _ => Err(Error::BadWrap { offset }),
                            }
                        } else if matches!(byte, b'\r' | b'\n') {
                            Err(Error::BadWrap { offset })
                        } else {
                            self.process_symbol(byte, offset, output)?;
                            self.wrap_symbols += 1;
                            Ok(())
                        }
                    }
                }
            }
        }
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;
            if let Err(error) = self.scan_byte(byte, offset, &mut output) {
                return Err(self.fail(error));
            }
        }
        Ok(output)
    }

    fn finish_inner(&mut self) -> Result<Vec<u8>, Error> {
        if let WrapState::NeedLf { cr_offset } = self.wrap_state {
            return Err(Error::BadWrap { offset: cr_offset });
        }
        if let WrapState::NeedSymbol { separator_offset } = self.wrap_state {
            return Err(Error::BadWrap {
                offset: separator_offset,
            });
        }

        if let Some(offset) = self.pending_second_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.group_len == 0 {
            return Ok(Vec::new());
        }

        if self.options.padding == Padding::Required {
            return Err(Error::Truncated {
                offset: self.group_offsets[0],
            });
        }

        match self.group_len {
            1 => Err(Error::Truncated {
                offset: self.group_offsets[0],
            }),
            2 => {
                let a = self.group[0];
                let b = self.group[1];
                if self.options.strict && (b & 0x0f) != 0 {
                    return Err(Error::NonCanonical {
                        offset: self.group_offsets[1],
                    });
                }
                Ok(vec![(a << 2) | (b >> 4)])
            }
            3 => {
                let a = self.group[0];
                let b = self.group[1];
                let c = self.group[2];
                if self.options.strict && (c & 0x03) != 0 {
                    return Err(Error::NonCanonical {
                        offset: self.group_offsets[2],
                    });
                }
                Ok(vec![(a << 2) | (b >> 4), (b << 4) | (c >> 2)])
            }
            _ => unreachable!(),
        }
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        match self.finish_inner() {
            Ok(output) => {
                self.closed = true;
                Ok(output)
            }
            Err(error) => Err(self.fail(error)),
        }
    }
}
