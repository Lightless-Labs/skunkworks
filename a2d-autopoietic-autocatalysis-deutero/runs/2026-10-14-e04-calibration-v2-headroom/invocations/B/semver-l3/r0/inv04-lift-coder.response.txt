use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return Err(if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    let field = &s[start..*pos];
    if field.len() > 1 && field.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    field.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut at = start;

    loop {
        let mut stop = at;
        while stop < end && bytes[stop] != b'.' {
            if !ident_char(bytes[stop]) {
                return Err(VersionError::InvalidChar { at: stop });
            }
            stop += 1;
        }
        if stop == at {
            return Err(VersionError::EmptyIdentifier { at });
        }
        let text = &s[at..stop];
        if pre && text.bytes().all(digit) && text.len() > 1 && text.starts_with('0') {
            return Err(VersionError::LeadingZero { at });
        }
        out.push(text.to_string());
        if stop == end {
            break;
        }
        at = stop + 1;
        if at == end {
            return Err(VersionError::EmptyIdentifier { at });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    for _ in 0..2 {
        if pos >= s.len() || s.as_bytes()[pos] != b'.' {
            return Err(if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        pos += 1;
        if _ == 0 {
            let minor = parse_core_number(s, &mut pos)?;
            if pos >= s.len() || s.as_bytes()[pos] != b'.' {
                return Err(if pos == s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
                    VersionError::MissingCore { at: pos }
                } else {
                    VersionError::InvalidChar { at: pos }
                });
            }
            pos += 1;
            let patch = parse_core_number(s, &mut pos)?;
            let (pre, build) = parse_tail(s, pos)?;
            return Ok(Version { major, minor, patch, pre, build });
        }
    }
    unreachable!()
}

fn parse_tail(s: &str, pos: usize) -> Result<(Vec<String>, Vec<String>), VersionError> {
    let bytes = s.as_bytes();
    if pos == s.len() {
        return Ok((Vec::new(), Vec::new()));
    }
    if bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }
    let mut cursor = pos;
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if bytes[cursor] == b'-' {
        let start = cursor + 1;
        let plus = bytes[start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|i| start + i)
            .unwrap_or(s.len());
        pre = parse_identifiers(s, start, plus, true)?;
        cursor = plus;
    }
    if cursor < s.len() {
        if bytes[cursor] != b'+' {
            return Err(VersionError::InvalidChar { at: cursor });
        }
        build = parse_identifiers(s, cursor + 1, s.len(), false)?;
    }
    Ok((pre, build))
}

fn numeric_ident(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(digit)
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_ident(x);
        let yn = numeric_ident(y);
        let ord = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug, PartialEq, Eq)]
struct Bound {
    version: Version,
    inclusive: bool,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparisons: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

#[derive(Clone)]
struct Token<'a> {
    text: &'a str,
    start: usize,
}

fn tokens(s: &str, base: usize) -> Vec<Token<'_>> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !ws(bytes[i]) {
            i += 1;
        }
        out.push(Token { text: &s[start..i], start: base + start });
    }
    out
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn vcore(c: (u64, u64, u64), pre: Vec<String>) -> Version {
    Version {
        major: c.0,
        minor: c.1,
        patch: c.2,
        pre,
        build: Vec::new(),
    }
}

fn increment(c: (u64, u64, u64), field: usize) -> Option<(u64, u64, u64)> {
    match field {
        0 => Some((c.0.checked_add(1)?, 0, 0)),
        1 => Some((c.0, c.1.checked_add(1)?, 0)),
        _ => Some((c.0, c.1, c.2.checked_add(1)?)),
    }
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(core(&version));
    }
    arm.comparisons.push(Comparator { op, version });
}

fn full_operand(
    text: &str,
    start: usize,
    offset: usize,
) -> Result<Version, RangeError> {
    parse_version(text).map_err(|error| RangeError::Version { at: start + offset, error })
}

#[derive(Clone)]
struct Partial {
    lower: Version,
    k: usize,
    full: bool,
    has_numeric: bool,
}

fn partial_operand(text: &str, start: usize) -> Result<Partial, RangeError> {
    let has_suffix = text.bytes().any(|b| b == b'-' || b == b'+');
    let fields: Vec<&str> = text.split('.').collect();
    let full_candidate = has_suffix || (fields.len() >= 3 && !fields.iter().any(|f| is_wild(f)));
    if full_candidate {
        let v = full_operand(text, start, 0)?;
        return Ok(Partial { lower: v, k: 3, full: true, has_numeric: true });
    }

    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at: start });
    }
    let mut nums = Vec::new();
    let mut wild = false;
    for f in &fields {
        if is_wild(f) {
            wild = true;
            continue;
        }
        if wild
            || !f.bytes().all(digit)
            || (f.len() > 1 && f.starts_with('0'))
        {
            return Err(RangeError::Syntax { at: start });
        }
        let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: start })?;
        nums.push(n);
    }
    let k = nums.len();
    let mut tuple = [0u64; 3];
    for (i, n) in nums.iter().enumerate() {
        tuple[i] = *n;
    }
    Ok(Partial {
        lower: vcore((tuple[0], tuple[1], tuple[2]), Vec::new()),
        k,
        full: false,
        has_numeric: k > 0,
    })
}

fn is_wild(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

fn partial_upper(p: &Partial, kind: &str, start: usize) -> Result<Option<Version>, RangeError> {
    if p.k == 0 {
        return Ok(None);
    }
    let c = core(&p.lower);
    let field = match kind {
        "bare" => {
            if p.k == 1 { 0 } else { 1 }
        }
        "caret" => {
            if c.0 > 0 || p.k == 1 { 0 }
            else if c.1 > 0 || p.k == 2 { 1 }
            else { 2 }
        }
        "tilde" => if p.k == 1 { 0 } else { 1 },
        _ => return Ok(None),
    };
    let bumped = increment(c, field).ok_or(RangeError::Syntax { at: start })?;
    Ok(Some(vcore(bumped, vec!["0".to_string()])))
}

fn expand_partial(
    arm: &mut Arm,
    text: &str,
    start: usize,
    mode: &str,
    prefix_len: usize,
) -> Result<(), RangeError> {
    let operand = &text[prefix_len..];
    let p = partial_operand(operand, start + prefix_len)?;
    if mode != "bare" && !p.has_numeric {
        return Err(RangeError::Syntax { at: start });
    }
    if p.k == 0 {
        return Ok(());
    }
    add_cmp(arm, Op::Ge, p.lower.clone(), true);
    if p.full {
        if mode == "bare" {
            arm.comparisons.pop();
            arm.anchors.pop_if(|_| false);
            add_cmp(arm, Op::Eq, p.lower, true);
        } else if mode == "caret" || mode == "tilde" {
            let upper = partial_upper(&p, mode, start)?;
            if let Some(u) = upper {
                add_cmp(arm, Op::Lt, u, false);
            }
        }
        return Ok(());
    }
    if let Some(upper) = partial_upper(&p, mode, start)? {
        add_cmp(arm, Op::Lt, upper, false);
    }
    Ok(())
}

trait PopIf<T> {
    fn pop_if<F: FnOnce(&T) -> bool>(&mut self, f: F);
}
impl<T> PopIf<T> for Vec<T> {
    fn pop_if<F: FnOnce(&T) -> bool>(&mut self, f: F) {
        if let Some(last) = self.last() {
            if f(last) {
                self.pop();
            }
        }
    }
}

fn parse_comparator_token(token: &Token<'_>, arm: &mut Arm) -> Result<(), RangeError> {
    let b = token.text.as_bytes();
    let (op, n) = if b.starts_with(b">=") { (Op::Ge, 2) }
        else if b.starts_with(b"<=") { (Op::Le, 2) }
        else if b.starts_with(b">") { (Op::Gt, 1) }
        else if b.starts_with(b"<") { (Op::Lt, 1) }
        else if b.starts_with(b"=") { (Op::Eq, 1) }
        else { return Err(RangeError::Syntax { at: token.start }); };
    let operand = &token.text[n..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: token.start });
    }
    let v = full_operand(operand, token.start, n)?;
    add_cmp(arm, op, v, true);
    Ok(())
}

fn parse_hyphen_arm(ts: &[Token<'_>]) -> Result<Arm, RangeError> {
    let mut arm = Arm { comparisons: Vec::new(), anchors: Vec::new() };
    let minus = ts.iter().find(|t| t.text == "-").unwrap();
    if ts.len() != 3 || ts[1].text != "-" {
        return Err(RangeError::Syntax { at: minus.start });
    }
    let lo = partial_operand(ts[0].text, ts[0].start)?;
    let hi = partial_operand(ts[2].text, ts[2].start)?;
    if !lo.has_numeric {
        return Err(RangeError::Syntax { at: ts[0].start });
    }
    if !hi.has_numeric {
        return Err(RangeError::Syntax { at: ts[2].start });
    }
    add_cmp(&mut arm, Op::Ge, lo.lower.clone(), true);
    if hi.full {
        add_cmp(&mut arm, Op::Le, hi.lower.clone(), true);
    } else {
        let upper = if hi.k == 3 {
            hi.lower.clone()
        } else {
            let field = if hi.k == 1 { 0 } else { 1 };
            let c = increment(core(&hi.lower), field)
                .ok_or(RangeError::Syntax { at: ts[2].start })?;
            vcore(c, vec!["0".to_string()])
        };
        if hi.k == 3 {
            add_cmp(&mut arm, Op::Le, upper, false);
        } else {
            add_cmp(&mut arm, Op::Lt, upper, false);
        }
    }
    Ok(arm)
}

fn parse_arm(ts: &[Token<'_>]) -> Result<Arm, RangeError> {
    if let Some(minus) = ts.iter().find(|t| t.text == "-") {
        let _ = minus;
        return parse_hyphen_arm(ts);
    }
    let mut arm = Arm { comparisons: Vec::new(), anchors: Vec::new() };
    for t in ts {
        let b = t.text.as_bytes();
        if b.starts_with(b">") || b.starts_with(b"<") || b.starts_with(b"=") {
            parse_comparator_token(t, &mut arm)?;
        } else if b.starts_with(b"^") {
            expand_partial(&mut arm, t.text, t.start, "caret", 1)?;
        } else if b.starts_with(b"~") {
            expand_partial(&mut arm, t.text, t.start, "tilde", 1)?;
        } else if digit(b[0]) || is_wild(t.text) {
            expand_partial(&mut arm, t.text, t.start, "bare", 0)?;
        } else {
            return Err(RangeError::Syntax { at: t.start });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut spans = Vec::new();
    let mut start = 0;
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i..].starts_with(b"||")
            && (i == 0 || ws(bytes[i - 1]))
            && (i + 2 == bytes.len() || ws(bytes[i + 2]))
        {
            spans.push((start, i));
            i += 2;
            start = i;
        } else {
            i += 1;
        }
    }
    spans.push((start, s.len()));

    if spans.len() == 1 && tokens(s, 0).is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    for (a, b) in spans {
        let ts = tokens(&s[a..b], a);
        if ts.is_empty() {
            return Err(RangeError::Empty { at: if a == 0 { 0 } else { a } });
        }
        arms.push(parse_arm(&ts)?);
    }
    Ok(Range { arms })
}

fn cmp_one(v: &Version, c: &Comparator) -> bool {
    let o = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => o == Ordering::Equal,
        Op::Lt => o == Ordering::Less,
        Op::Le => o != Ordering::Greater,
        Op::Gt => o == Ordering::Greater,
        Op::Ge => o != Ordering::Less,
    }
}

fn admission(v: &Version, arm: &Arm) -> bool {
    arm.anchors.iter().any(|&c| c == core(v))
}

fn arm_matches(v: &Version, arm: &Arm, include: bool) -> bool {
    if !v.pre.is_empty() && !include && !admission(v, arm) {
        return false;
    }
    arm.comparisons.iter().all(|c| cmp_one(v, c))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(&parsed[i], &parsed[j]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn bound_for(arm: &Arm, lower: bool) -> Option<Bound> {
    let mut out: Option<Bound> = None;
    for c in &arm.comparisons {
        let (is_lower, inclusive) = match c.op {
            Op::Eq => (true, true),
            Op::Gt => (true, false),
            Op::Ge => (true, true),
            Op::Lt => (false, false),
            Op::Le => (false, true),
        };
        if c.op == Op::Eq {
            let candidate = Bound { version: c.version.clone(), inclusive: true };
            if let Some(old) = &out {
                let ord = compare_versions(&candidate.version, &old.version);
                if (lower && ord == Ordering::Greater) || (!lower && ord == Ordering::Less) {
                    out = Some(candidate);
                }
            } else {
                out = Some(candidate);
            }
            continue;
        }
        if is_lower != lower {
            continue;
        }
        let candidate = Bound { version: c.version.clone(), inclusive };
        if let Some(old) = &out {
            let ord = compare_versions(&candidate.version, &old.version);
            let stronger = if lower { ord == Ordering::Greater } else { ord == Ordering::Less };
            if stronger || (ord == Ordering::Equal && !candidate.inclusive && old.inclusive) {
                out = Some(candidate);
            }
        } else {
            out = Some(candidate);
        }
    }
    out
}

fn candidate_version_between(
    lower: Option<&Bound>,
    upper: Option<&Bound>,
    anchors: &[ (u64, u64, u64) ],
    include: bool,
) -> bool {
    if let Some(l) = lower {
        if let Some(u) = upper {
            let ord = compare_versions(&l.version, &u.version);
            if ord == Ordering::Greater || (ord == Ordering::Equal && !(l.inclusive && u.inclusive)) {
                return false;
            }
            if ord == Ordering::Equal {
                let v = &l.version;
                return v.pre.is_empty() || include || anchors.iter().any(|&c| c == core(v));
            }
        }
    }
    true
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let lo_a = bound_for(aa, true);
            let lo_b = bound_for(bb, true);
            let hi_a = bound_for(aa, false);
            let hi_b = bound_for(bb, false);
            let lower = match (lo_a, lo_b) {
                (Some(x), Some(y)) => Some(if compare_versions(&x.version, &y.version) == Ordering::Greater { x } else { y }),
                (Some(x), None) | (None, Some(x)) => Some(x),
                _ => None,
            };
            let upper = match (hi_a, hi_b) {
                (Some(x), Some(y)) => Some(if compare_versions(&x.version, &y.version) == Ordering::Less { x } else { y }),
                (Some(x), None) | (None, Some(x)) => Some(x),
                _ => None,
            };
            let mut anchors = aa.anchors.clone();
            anchors.retain(|c| bb.anchors.contains(c));
            if candidate_version_between(lower.as_ref(), upper.as_ref(), &anchors, include_prerelease) {
                return true;
            }
        }
    }
    false
}

fn serialize_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn display_bound(b: Option<Bound>, lower: bool) -> String {
    match b {
        None => "*".to_string(),
        Some(bound) => {
            let op = match (lower, bound.inclusive) {
                (true, true) => ">=",
                (true, false) => ">",
                (false, true) => "<=",
                (false, false) => "<",
            };
            format!("{}{}", op, serialize_version(&bound.version))
        }
    }
}

fn arm_canonical(arm: &Arm) -> String {
    let lower = bound_for(arm, true);
    let upper = bound_for(arm, false);
    let mut anchors = arm.anchors.clone();
    anchors.sort();
    anchors.dedup();
    let anchor_text = if anchors.is_empty() {
        "-".to_string()
    } else {
        anchors.iter().map(|c| format!("{}.{}.{}", c.0, c.1, c.2)).collect::<Vec<_>>().join(",")
    };
    format!(
        "[{};{};pre={}]",
        display_bound(lower, true),
        display_bound(upper, false),
        anchor_text
    )
}

pub fn canonical(r: &Range) -> String {
    let mut out: Vec<String> = r.arms.iter().map(arm_canonical).collect();
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".to_string() } else { out.join(" || ") }
}
