```rust
use std::collections::{HashMap, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

struct Entry {
    job: Job,
    dependencies: Vec<usize>,
    state: State,
    remaining: Option<u64>,
    started_at: u64,
    input_order: usize,
}

pub struct Pool {
    entries: Vec<Entry>,
    by_id: HashMap<String, usize>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    has_run: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut by_id = HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if let Some(tag) = &job.tag {
                if !valid_tag(tag) {
                    return Err(Error::BadTag(i));
                }
            }
            if by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            by_id.insert(job.id.clone(), i);
        }

        let mut resolved = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = HashSet::new();
            let mut deps = Vec::with_capacity(job.dependencies.len());
            for (j, spelling) in job.dependencies.iter().enumerate() {
                if !seen.insert(spelling.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match by_id.get(spelling) {
                    Some(&idx) => deps.push(idx),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            resolved.push(deps);
        }

        if has_cycle(&resolved) {
            return Err(Error::Cycle);
        }

        let entries = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| Entry {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                job,
                dependencies: resolved[i].clone(),
                remaining: None,
                started_at: 0,
                input_order: i,
            })
            .collect();

        Ok(Self {
            entries,
            by_id,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
        })
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if self.has_run && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut trace = Vec::new();
        let start = if self.has_run {
            self.horizon.saturating_add(1)
        } else {
            0
        };

        if start <= until {
            let mut now = start;
            loop {
                self.process_timestamp(now, &mut trace);

                let next = self.next_event_after(now);
                match next {
                    Some(t) if t <= until => now = t,
                    _ => break,
                }
            }
        }

        self.horizon = until;
        self.has_run = true;
        Ok(trace)
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let &idx = self.by_id.get(id).ok_or(Error::Unknown)?;
        Ok(self.entries[idx].state.clone())
    }

    fn process_timestamp(&mut self, now: u64, trace: &mut Vec<String>) {
        let mut completing: Vec<usize> = self
            .entries
            .iter()
            .enumerate()
            .filter_map(|(i, e)| match e.state {
                State::Running { finish_at, .. } if finish_at == now => Some(i),
                _ => None,
            })
            .collect();

        completing.sort_by(|&a, &b| self.entries[a].job.id.as_bytes().cmp(self.entries[b].job.id.as_bytes()));

        for idx in completing {
            let attempt = match self.entries[idx].state {
                State::Running { attempt, .. } => attempt,
                _ => continue,
            };
            let ok = attempt > self.entries[idx].job.failures;
            let id = self.entries[idx].job.id.clone();
            if ok {
                self.entries[idx].state = State::Succeeded;
                self.entries[idx].remaining = None;
                trace.push(format!("{now} finish {id} {attempt} ok"));
            } else {
                trace.push(format!("{now} finish {id} {attempt} fail"));
                self.entries[idx].remaining = None;
                if attempt >= self.entries[idx].job.max_attempts {
                    self.entries[idx].state = State::Failed;
                } else {
                    let delay = retry_delay(self.retry_base, self.retry_cap, attempt);
                    let ready_at = now.saturating_add(delay);
                    self.entries[idx].state = State::Waiting { ready_at, attempt };
                    trace.push(format!("{now} retry {id} {attempt} {ready_at}"));
                }
            }
        }

        let mut newly_blocked = Vec::new();
        loop {
            let mut changed = false;
            for i in 0..self.entries.len() {
                if !matches!(self.entries[i].state, State::Waiting { .. }) {
                    continue;
                }
                let failed_dep = self.entries[i].dependencies.iter().copied().find(|&d| {
                    matches!(self.entries[d].state, State::Failed | State::Blocked)
                });
                if let Some(dep) = failed_dep {
                    self.entries[i].state = State::Blocked;
                    newly_blocked.push((i, dep));
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
        newly_blocked.sort_by(|(a, _), (b, _)| {
            self.entries[*a].job.id.as_bytes().cmp(self.entries[*b].job.id.as_bytes())
        });
        for (idx, dep) in newly_blocked {
            trace.push(format!(
                "{now} skip {} {}",
                self.entries[idx].job.id, self.entries[dep].job.id
            ));
        }

        loop {
            let mut candidates: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, e)| match e.state {
                    State::Waiting { ready_at, .. }
                        if ready_at <= now
                            && e.dependencies.iter().all(|&d| {
                                matches!(self.entries[d].state, State::Succeeded)
                            }) =>
                    {
                        Some(i)
                    }
                    _ => None,
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, now);
                let pb = self.effective_priority(b, now);
                pb.cmp(&pa)
                    .then_with(|| self.entries[a].input_order.cmp(&self.entries[b].input_order))
            });

            let mut selected = None;
            for candidate in candidates {
                if self.can_start_without_preemption(candidate) {
                    selected = Some((candidate, None));
                    break;
                }
                if let Some(victim) = self.choose_victim(candidate) {
                    selected = Some((candidate, Some(victim)));
                    break;
                }
            }

            let Some((candidate, victim)) = selected else {
                break;
            };

            if let Some(victim_idx) = victim {
                let victim_id = self.entries[victim_idx].job.id.clone();
                let candidate_id = self.entries[candidate].job.id.clone();
                if let State::Running { attempt, finish_at } = self.entries[victim_idx].state {
                    self.entries[victim_idx].remaining = Some(finish_at - now);
                    self.entries[victim_idx].state = State::Waiting {
                        ready_at: now,
                        attempt,
                    };
                }
                trace.push(format!("{now} preempt {victim_id} by {candidate_id}"));
            }

            let (attempt, remaining) = match self.entries[candidate].state {
                State::Waiting { attempt, .. } => {
                    if let Some(saved) = self.entries[candidate].remaining.take() {
                        (attempt, saved)
                    } else {
                        let next_attempt = attempt.saturating_add(1);
                        (next_attempt, self.entries[candidate].job.duration)
                    }
                }
                _ => continue,
            };
            let finish_at = now.saturating_add(remaining);
            self.entries[candidate].started_at = now;
            self.entries[candidate].state = State::Running { attempt, finish_at };
            trace.push(format!(
                "{now} start {} {attempt} {remaining}",
                self.entries[candidate].job.id
            ));
        }
    }

    fn can_start_without_preemption(&self, candidate: usize) -> bool {
        let running = self.running_indices();
        if running.len() >= self.capacity {
            return false;
        }
        let tag = self.entries[candidate].job.tag.as_deref();
        !running.iter().any(|&i| {
            tag.is_some() && self.entries[i].job.tag.as_deref() == tag
        })
    }

    fn choose_victim(&self, candidate: usize) -> Option<usize> {
        let tag = self.entries[candidate].job.tag.as_deref();
        let candidate_priority = self.entries[candidate].job.priority;
        let running = self.running_indices();

        let mut eligible: Vec<usize> = running
            .into_iter()
            .filter(|&i| {
                self.entries[i].job.priority.saturating_add(2) <= candidate_priority
            })
            .filter(|&victim| {
                let remaining_after = self
                    .running_indices()
                    .into_iter()
                    .filter(|&i| i != victim)
                    .collect::<Vec<_>>();
                if remaining_after.len() >= self.capacity {
                    return false;
                }
                !remaining_after.iter().any(|&i| {
                    tag.is_some() && self.entries[i].job.tag.as_deref() == tag
                })
            })
            .collect();

        eligible.sort_by(|&a, &b| {
            self.entries[a]
                .job
                .priority
                .cmp(&self.entries[b].job.priority)
                .then_with(|| self.entries[b].started_at.cmp(&self.entries[a].started_at))
                .then_with(|| self.entries[b].job.id.as_bytes().cmp(self.entries[a].job.id.as_bytes()))
        });
        eligible.into_iter().next()
    }

    fn running_indices(&self) -> Vec<usize> {
        self.entries
            .iter()
            .enumerate()
            .filter_map(|(i, e)| matches!(e.state, State::Running { .. }).then_some(i))
            .collect()
    }

    fn effective_priority(&self, idx: usize, now: u64) -> u8 {
        let base = self.entries[idx].job.priority;
        if self.aging_ms == 0 {
            return base;
        }
        let ready_at = match self.entries[idx].state {
            State::Waiting { ready_at, .. } => ready_at,
            _ => now,
        };
        let age = now.saturating_sub(ready_at) / self.aging_ms;
        (base as u64 + age).min(9) as u8
    }

    fn next_event_after(&self, now: u64) -> Option<u64> {
        self.entries
            .iter()
            .filter_map(|e| match e.state {
                State::Waiting { ready_at, .. } if ready_at > now => Some(ready_at),
                State::Running { finish_at, .. } if finish_at > now => Some(finish_at),
                _ => None,
            })
            .min()
    }
}

fn valid_id(s: &str) -> bool {
    (1..=16).contains(&s.len())
        && s.bytes()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
}

fn valid_tag(s: &str) -> bool {
    (1..=8).contains(&s.len())
        && s.bytes()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
}

fn retry_delay(base: u64, cap: u64, attempt: u8) -> u64 {
    let mut delay = base;
    for _ in 1..attempt {
        delay = delay.saturating_mul(2).min(cap);
        if delay == cap {
            break;
        }
    }
    delay.min(cap)
}

fn has_cycle(dependencies: &[Vec<usize>]) -> bool {
    fn visit(node: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
        if marks[node] == 1 {
            return true;
        }
        if marks[node] == 2 {
            return false;
        }
        marks[node] = 1;
        for &dep in &deps[node] {
            if visit(dep, deps, marks) {
                return true;
            }
        }
        marks[node] = 2;
        false
    }

    let mut marks = vec![0; dependencies.len()];
    (0..dependencies.len()).any(|i| visit(i, dependencies, &mut marks))
}
```
