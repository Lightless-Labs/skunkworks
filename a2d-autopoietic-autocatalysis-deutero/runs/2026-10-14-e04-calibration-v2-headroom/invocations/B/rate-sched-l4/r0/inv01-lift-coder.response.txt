```rust
use std::collections::{HashMap, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
enum Phase {
    Waiting {
        ready_at: u64,
        attempt: u8,
        remaining: Option<u64>,
        resume: bool,
    },
    Running {
        attempt: u8,
        finish_at: u64,
        started_at: u64,
    },
    Succeeded,
    Failed,
    Blocked,
}

pub struct Pool {
    jobs: Vec<Job>,
    states: Vec<Phase>,
    dependencies: Vec<Vec<usize>>,
    by_id: HashMap<String, usize>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    started: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut by_id = HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            let id_ok = !job.id.is_empty()
                && job.id.len() <= 16
                && job.id.bytes().all(valid_byte);
            if !id_ok {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if let Some(tag) = &job.tag {
                if tag.is_empty() || tag.len() > 8 || !tag.bytes().all(valid_byte) {
                    return Err(Error::BadTag(i));
                }
            }
            if by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            by_id.insert(job.id.clone(), i);
        }

        let mut dependencies = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = HashSet::new();
            for (j, spelling) in job.dependencies.iter().enumerate() {
                if !seen.insert(spelling.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match by_id.get(spelling) {
                    Some(&dep) => dependencies[i].push(dep),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
        }

        let mut colors = vec![0u8; jobs.len()];
        fn visit(node: usize, dependencies: &[Vec<usize>], colors: &mut [u8]) -> bool {
            if colors[node] == 1 {
                return false;
            }
            if colors[node] == 2 {
                return true;
            }
            colors[node] = 1;
            for &dep in &dependencies[node] {
                if !visit(dep, dependencies, colors) {
                    return false;
                }
            }
            colors[node] = 2;
            true
        }
        for i in 0..jobs.len() {
            if !visit(i, &dependencies, &mut colors) {
                return Err(Error::Cycle);
            }
        }

        let states = jobs
            .iter()
            .map(|job| Phase::Waiting {
                ready_at: job.release_at,
                attempt: 0,
                remaining: None,
                resume: false,
            })
            .collect();

        Ok(Self {
            jobs,
            states,
            dependencies,
            by_id,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            started: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let &index = self.by_id.get(id).ok_or(Error::Unknown)?;
        Ok(match self.states[index] {
            Phase::Waiting {
                ready_at, attempt, ..
            } => State::Waiting { ready_at, attempt },
            Phase::Running {
                attempt, finish_at, ..
            } => State::Running { attempt, finish_at },
            Phase::Succeeded => State::Succeeded,
            Phase::Failed => State::Failed,
            Phase::Blocked => State::Blocked,
        })
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut states = self.states.clone();
        let mut lines = Vec::new();
        let mut now = if self.started { self.horizon } else { 0 };
        let mut process_now = !self.started;

        loop {
            if process_now {
                self.process_timestamp(now, &mut states, &mut lines);
                process_now = false;
            }

            let mut next = None;
            for state in &states {
                let time = match state {
                    Phase::Waiting { ready_at, .. } if *ready_at > now => Some(*ready_at),
                    Phase::Running { finish_at, .. } if *finish_at > now => Some(*finish_at),
                    _ => None,
                };
                if let Some(time) = time {
                    next = Some(next.map_or(time, |old: u64| old.min(time)));
                }
            }

            match next {
                Some(time) if time <= until => {
                    now = time;
                    process_now = true;
                }
                _ => break,
            }
        }

        self.states = states;
        self.horizon = until;
        self.started = true;
        Ok(lines)
    }

    fn process_timestamp(&self, now: u64, states: &mut [Phase], lines: &mut Vec<String>) {
        let mut completed: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| match state {
                Phase::Running { finish_at, .. } if *finish_at == now => Some(i),
                _ => None,
            })
            .collect();
        completed.sort_by(|&a, &b| self.jobs[a].id.as_bytes().cmp(self.jobs[b].id.as_bytes()));

        for index in completed {
            let attempt = match states[index] {
                Phase::Running { attempt, .. } => attempt,
                _ => continue,
            };
            let job = &self.jobs[index];
            if attempt > job.failures {
                lines.push(format!("{now} finish {} {attempt} ok", job.id));
                states[index] = Phase::Succeeded;
            } else {
                lines.push(format!("{now} finish {} {attempt} fail", job.id));
                if attempt >= job.max_attempts {
                    states[index] = Phase::Failed;
                } else {
                    let shift = u32::from(attempt.saturating_sub(1));
                    let delay = self
                        .retry_base
                        .checked_shl(shift)
                        .unwrap_or(u64::MAX)
                        .min(self.retry_cap);
                    let ready_at = now.saturating_add(delay);
                    lines.push(format!("{now} retry {} {attempt} {ready_at}", job.id));
                    states[index] = Phase::Waiting {
                        ready_at,
                        attempt,
                        remaining: None,
                        resume: false,
                    };
                }
            }
        }

        loop {
            let newly_blocked: Vec<usize> = (0..states.len())
                .filter(|&i| {
                    matches!(states[i], Phase::Waiting { .. })
                        && self.dependencies[i].iter().any(|&dep| {
                            matches!(states[dep], Phase::Failed | Phase::Blocked)
                        })
                })
                .collect();
            if newly_blocked.is_empty() {
                break;
            }
            for &i in &newly_blocked {
                states[i] = Phase::Blocked;
            }
            let mut ordered = newly_blocked;
            ordered.sort_by(|&a, &b| self.jobs[a].id.as_bytes().cmp(self.jobs[b].id.as_bytes()));
            for i in ordered {
                if let Some(&dep) = self.dependencies[i].iter().find(|&&dep| {
                    matches!(states[dep], Phase::Failed | Phase::Blocked)
                }) {
                    lines.push(format!(
                        "{now} skip {} {}",
                        self.jobs[i].id, self.jobs[dep].id
                    ));
                }
            }
        }

        loop {
            let mut candidates: Vec<usize> = (0..states.len())
                .filter(|&i| match states[i] {
                    Phase::Waiting { ready_at, .. } => {
                        ready_at <= now
                            && self.dependencies[i]
                                .iter()
                                .all(|&dep| matches!(states[dep], Phase::Succeeded))
                    }
                    _ => false,
                })
                .collect();
            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, now, states);
                let pb = self.effective_priority(b, now, states);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut started_any = false;
            for candidate in candidates {
                if self.can_start(candidate, states) {
                    self.start_job(candidate, now, states, lines);
                    started_any = true;
                    break;
                }
                if let Some(victim) = self.select_victim(candidate, states) {
                    let remaining = match states[victim] {
                        Phase::Running { finish_at, .. } => finish_at - now,
                        _ => continue,
                    };
                    let attempt = match states[victim] {
                        Phase::Running { attempt, .. } => attempt,
                        _ => continue,
                    };
                    lines.push(format!(
                        "{now} preempt {} by {}",
                        self.jobs[victim].id, self.jobs[candidate].id
                    ));
                    states[victim] = Phase::Waiting {
                        ready_at: now,
                        attempt,
                        remaining: Some(remaining),
                        resume: true,
                    };
                    self.start_job(candidate, now, states, lines);
                    started_any = true;
                    break;
                }
            }
            if !started_any {
                break;
            }
        }
    }

    fn effective_priority(&self, index: usize, now: u64, states: &[Phase]) -> u8 {
        let ready_at = match states[index] {
            Phase::Waiting { ready_at, .. } => ready_at,
            _ => now,
        };
        if self.aging_ms == 0 {
            return self.jobs[index].priority;
        }
        let age = now.saturating_sub(ready_at) / self.aging_ms;
        (u64::from(self.jobs[index].priority) + age).min(9) as u8
    }

    fn can_start(&self, index: usize, states: &[Phase]) -> bool {
        let running: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| matches!(s, Phase::Running { .. }).then_some(i))
            .collect();
        if running.len() >= self.capacity {
            return false;
        }
        self.tag_free(index, &running)
    }

    fn tag_free(&self, index: usize, running: &[usize]) -> bool {
        let Some(tag) = &self.jobs[index].tag else {
            return true;
        };
        !running
            .iter()
            .any(|&i| self.jobs[i].tag.as_ref() == Some(tag))
    }

    fn select_victim(&self, candidate: usize, states: &[Phase]) -> Option<usize> {
        let running: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| matches!(s, Phase::Running { .. }).then_some(i))
            .collect();

        let candidate_priority = self.jobs[candidate].priority;
        let mut victims: Vec<usize> = running
            .iter()
            .copied()
            .filter(|&victim| {
                candidate_priority >= self.jobs[victim].priority.saturating_add(2)
            })
            .filter(|&victim| {
                let remaining: Vec<usize> =
                    running.iter().copied().filter(|&i| i != victim).collect();
                remaining.len() < self.capacity && self.tag_free(candidate, &remaining)
            })
            .collect();

        victims.sort_by(|&a, &b| {
            self.jobs[a]
                .priority
                .cmp(&self.jobs[b].priority)
                .then_with(|| {
                    let sa = match states[a] {
                        Phase::Running { started_at, .. } => started_at,
                        _ => 0,
                    };
                    let sb = match states[b] {
                        Phase::Running { started_at, .. } => started_at,
                        _ => 0,
                    };
                    sb.cmp(&sa)
                })
                .then_with(|| self.jobs[b].id.as_bytes().cmp(self.jobs[a].id.as_bytes()))
        });
        victims.first().copied()
    }

    fn start_job(&self, index: usize, now: u64, states: &mut [Phase], lines: &mut Vec<String>) {
        let (attempt, remaining) = match states[index].clone() {
            Phase::Waiting {
                attempt,
                remaining: Some(remaining),
                resume: true,
                ..
            } => (attempt, remaining),
            Phase::Waiting { attempt, .. } => (attempt.saturating_add(1), self.jobs[index].duration),
            _ => return,
        };
        let finish_at = now.saturating_add(remaining);
        lines.push(format!(
            "{now} start {} {attempt} {remaining}",
            self.jobs[index].id
        ));
        states[index] = Phase::Running {
            attempt,
            finish_at,
            started_at: now,
        };
    }
}

fn valid_byte(byte: u8) -> bool {
    byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-'
}
```
