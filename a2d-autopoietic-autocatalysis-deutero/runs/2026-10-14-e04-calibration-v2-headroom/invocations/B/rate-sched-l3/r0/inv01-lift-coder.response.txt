```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobData {
    job: Job,
    dependency_indices: Vec<usize>,
    state: Option<State>,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<JobData>,
    id_indices: std::collections::HashMap<String, usize>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut id_indices = std::collections::HashMap::new();
        for (index, job) in jobs.iter().enumerate() {
            if job.id.is_empty()
                || job.id.len() > 16
                || !job.id.bytes().all(|b| {
                    b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-'
                })
            {
                return Err(Error::BadId(index));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(index));
            }
            if job.max_attempts == 0 || job.max_attempts > 32 {
                return Err(Error::BadAttempts(index));
            }
            if id_indices.contains_key(&job.id) {
                return Err(Error::Duplicate(index));
            }
            id_indices.insert(job.id.clone(), index);
        }

        let mut dependency_indices = Vec::with_capacity(jobs.len());
        for (job_index, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut deps = Vec::with_capacity(job.dependencies.len());
            for (dependency_index, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency) {
                    return Err(Error::DuplicateDependency(job_index, dependency_index));
                }
                let Some(&dep_index) = id_indices.get(dependency) else {
                    return Err(Error::MissingDependency(job_index, dependency_index));
                };
                deps.push(dep_index);
            }
            dependency_indices.push(deps);
        }

        fn visit(
            node: usize,
            dependencies: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[node] == 1 {
                return false;
            }
            if marks[node] == 2 {
                return true;
            }
            marks[node] = 1;
            for &dependency in &dependencies[node] {
                if !visit(dependency, dependencies, marks) {
                    return false;
                }
            }
            marks[node] = 2;
            true
        }

        let mut marks = vec![0; jobs.len()];
        for index in 0..jobs.len() {
            if !visit(index, &dependency_indices, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let jobs = jobs
            .into_iter()
            .zip(dependency_indices)
            .map(|(job, dependency_indices)| JobData {
                job,
                dependency_indices,
                state: None,
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            id_indices,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.jobs.iter().any(|job| {
            matches!(job.state, Some(State::Waiting { .. } | State::Running { .. }))
        }) {
            return Err(Error::Busy);
        }

        for job in &mut self.jobs {
            job.state = Some(State::Waiting { ready_at: now });
            job.attempts = 0;
        }
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut selected: Option<(usize, u8)> = None;
        for (index, job) in self.jobs.iter().enumerate() {
            let Some(State::Waiting { ready_at }) = job.state else {
                continue;
            };
            if ready_at > now
                || !job.dependency_indices.iter().all(|&dependency| {
                    matches!(self.jobs[dependency].state, Some(State::Succeeded))
                })
            {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                job.job.priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                (u64::from(job.job.priority) + age).min(9) as u8
            };
            if selected.is_none_or(|(_, best)| effective > best) {
                selected = Some((index, effective));
            }
        }

        self.last_time = now;
        let Some((index, _)) = selected else {
            return Ok(None);
        };
        let job = &mut self.jobs[index];
        job.attempts += 1;
        let attempt = job.attempts;
        job.state = Some(State::Running { attempt });
        Ok(Some(Ticket {
            id: job.job.id.clone(),
            attempt,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(&index) = self.id_indices.get(id) else {
            return Err(Error::Unknown);
        };
        if !matches!(self.jobs[index].state, Some(State::Running { .. })) {
            return Err(Error::NotRunning);
        }

        self.last_time = now;
        if success {
            self.jobs[index].state = Some(State::Succeeded);
            return Ok(());
        }

        let attempts = self.jobs[index].attempts;
        if attempts == self.jobs[index].job.max_attempts {
            self.jobs[index].state = Some(State::Failed);
            self.propagate_blocked();
        } else {
            let shift = u32::from(attempts.saturating_sub(1));
            let delay = self
                .retry_base
                .checked_mul(1u64.checked_shl(shift).unwrap_or(u64::MAX))
                .unwrap_or(u64::MAX)
                .min(self.retry_cap);
            self.jobs[index].state = Some(State::Waiting {
                ready_at: now.saturating_add(delay),
            });
        }
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(&index) = self.id_indices.get(id) else {
            return Err(Error::Unknown);
        };
        self.jobs[index].state.clone().ok_or(Error::NoCycle)
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut to_block = Vec::new();
            for (index, job) in self.jobs.iter().enumerate() {
                if matches!(job.state, Some(State::Waiting { .. }))
                    && job.dependency_indices.iter().any(|&dependency| {
                        matches!(
                            self.jobs[dependency].state,
                            Some(State::Failed | State::Blocked)
                        )
                    })
                {
                    to_block.push(index);
                }
            }
            if to_block.is_empty() {
                break;
            }
            for index in to_block {
                self.jobs[index].state = Some(State::Blocked);
            }
        }
    }
}
```
