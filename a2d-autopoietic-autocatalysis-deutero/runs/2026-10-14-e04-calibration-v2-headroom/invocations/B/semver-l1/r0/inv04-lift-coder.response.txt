use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    fn parse_core(
        bytes: &[u8],
        pos: &mut usize,
    ) -> Result<u64, VersionError> {
        let start = *pos;
        while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
            *pos += 1;
        }

        if *pos == start {
            return Err(if *pos == bytes.len()
                || matches!(bytes[*pos], b'.' | b'-' | b'+')
            {
                VersionError::MissingCore { at: start }
            } else {
                VersionError::InvalidChar { at: start }
            });
        }

        if *pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut value = 0u64;
        for &digit in &bytes[start..*pos] {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        Ok(value)
    }

    let mut pos = 0;
    let major = parse_core(bytes, &mut pos)?;
    for _ in 0..1 {
        if pos >= bytes.len() || bytes[pos] != b'.' {
            return Err(if pos >= bytes.len()
                || matches!(bytes[pos], b'-' | b'+')
            {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        pos += 1;
    }
    let minor = parse_core(bytes, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        return Err(if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
            VersionError::MissingCore { at: pos }
        } else {
            VersionError::InvalidChar { at: pos }
        });
    }
    pos += 1;
    let patch = parse_core(bytes, &mut pos)?;

    fn parse_identifiers(
        s: &str,
        bytes: &[u8],
        start: usize,
        end: usize,
        prerelease: bool,
    ) -> Result<Vec<String>, VersionError> {
        let mut result = Vec::new();
        let mut ident_start = start;
        let mut i = start;

        while i <= end {
            if i == end || bytes[i] == b'.' {
                if i == ident_start {
                    return Err(VersionError::EmptyIdentifier { at: ident_start });
                }

                for j in ident_start..i {
                    let c = bytes[j];
                    if !(c.is_ascii_alphanumeric() || c == b'-') {
                        return Err(VersionError::InvalidChar { at: j });
                    }
                }

                if prerelease
                    && i - ident_start > 1
                    && bytes[ident_start..i].iter().all(u8::is_ascii_digit)
                    && bytes[ident_start] == b'0'
                {
                    return Err(VersionError::LeadingZero { at: ident_start });
                }

                result.push(s[ident_start..i].to_string());
                ident_start = i + 1;
            }
            i += 1;
        }

        Ok(result)
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let plus = bytes[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|offset| pre_start + offset);
        let pre_end = plus.unwrap_or(bytes.len());
        pre = parse_identifiers(s, bytes, pre_start, pre_end, true)?;
        pos = pre_end;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        let build_start = pos + 1;
        build = parse_identifiers(s, bytes, build_start, bytes.len(), false)?;
        pos = bytes.len();
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let a_numeric = is_numeric_identifier(a);
    let b_numeric = is_numeric_identifier(b);

    match (a_numeric, b_numeric) {
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (left, right) in a.pre.iter().zip(&b.pre) {
                let order = compare_identifiers(left, right);
                if order != Ordering::Equal {
                    return order;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
    admits_prerelease: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

fn is_range_space(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut comparators = Vec::new();
    let mut admissions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() && is_range_space(bytes[pos]) {
        pos += 1;
    }
    if pos == bytes.len() {
        return Err(RangeError::Empty { at: 0 });
    }

    while pos < bytes.len() {
        let token_start = pos;
        while pos < bytes.len() && !is_range_space(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;

        let (op, op_len) = match &bytes[token_start..token_end] {
            [b'<', b'=', ..] => (Op::LessEqual, 2),
            [b'>', b'=', ..] => (Op::GreaterEqual, 2),
            [b'=', ..] => (Op::Equal, 1),
            [b'<', ..] => (Op::Less, 1),
            [b'>', ..] => (Op::Greater, 1),
            _ => (Op::Equal, 0),
        };
        let operand_start = token_start + op_len;

        if operand_start == token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        if !version.pre.is_empty() {
            let core = (version.major, version.minor, version.patch);
            if !admissions.contains(&core) {
                admissions.push(core);
            }
        }

        comparators.push(Comparator { op, version });

        while pos < bytes.len() && is_range_space(bytes[pos]) {
            pos += 1;
        }
    }

    Ok(Range {
        comparators,
        admits_prerelease: admissions,
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    let core = (v.major, v.minor, v.patch);
    if !v.pre.is_empty() && !r.admits_prerelease.contains(&core) {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let order = compare_versions(v, &comparator.version);
        match comparator.op {
            Op::Equal => order == Ordering::Equal,
            Op::Less => order == Ordering::Less,
            Op::LessEqual => order != Ordering::Greater,
            Op::Greater => order == Ordering::Greater,
            Op::GreaterEqual => order != Ordering::Less,
        }
    })
}
