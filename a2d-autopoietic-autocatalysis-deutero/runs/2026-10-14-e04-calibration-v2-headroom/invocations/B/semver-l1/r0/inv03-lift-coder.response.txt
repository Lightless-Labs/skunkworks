use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for (field, value) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if start == pos {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: start });
            }
            return Err(VersionError::InvalidChar { at: start });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut n = 0u64;
        for &digit in &bytes[start..pos] {
            n = match n
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
            {
                Some(n) => n,
                None => return Err(VersionError::Overflow { at: start }),
            };
        }
        *value = n;

        if field < 2 {
            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
            } else if pos == bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        }
    }

    let (pre, build) = if pos == bytes.len() {
        (Vec::new(), Vec::new())
    } else {
        if !matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::InvalidChar { at: pos });
        }

        let mut pre = Vec::new();
        let mut build = Vec::new();

        if bytes[pos] == b'-' {
            pos += 1;
            let pre_start = pos;
            let plus = bytes[pos..].iter().position(|&b| b == b'+').map(|i| pos + i);
            let pre_end = plus.unwrap_or(bytes.len());
            parse_identifiers(s, pre_start, pre_end, true, &mut pre)?;
            pos = pre_end;

            if pos < bytes.len() {
                pos += 1;
                parse_identifiers(s, pos, bytes.len(), false, &mut build)?;
            }
        } else {
            pos += 1;
            parse_identifiers(s, pos, bytes.len(), false, &mut build)?;
        }

        (pre, build)
    };

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    let bytes = s.as_bytes();
    let mut id_start = start;

    loop {
        let id_end = bytes[id_start..end]
            .iter()
            .position(|&b| b == b'.')
            .map(|i| id_start + i)
            .unwrap_or(end);

        if id_start == id_end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        let mut all_digits = true;
        for i in id_start..id_end {
            let b = bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(VersionError::InvalidChar { at: i });
            }
            if !b.is_ascii_digit() {
                all_digits = false;
            }
        }

        if is_pre && all_digits && id_end - id_start > 1 && bytes[id_start] == b'0' {
            return Err(VersionError::LeadingZero { at: id_start });
        }

        out.push(s[id_start..id_end].to_owned());

        if id_end == end {
            break;
        }
        id_start = id_end + 1;
    }

    Ok(())
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_numeric_identifiers(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| {
            match (a.pre.is_empty(), b.pre.is_empty()) {
                (true, true) => Ordering::Equal,
                (true, false) => Ordering::Greater,
                (false, true) => Ordering::Less,
                (false, false) => {
                    for (left, right) in a.pre.iter().zip(&b.pre) {
                        let left_numeric = is_numeric_identifier(left);
                        let right_numeric = is_numeric_identifier(right);
                        let ordering = match (left_numeric, right_numeric) {
                            (true, true) => compare_numeric_identifiers(left, right),
                            (true, false) => Ordering::Less,
                            (false, true) => Ordering::Greater,
                            (false, false) => left.as_bytes().cmp(right.as_bytes()),
                        };
                        if ordering != Ordering::Equal {
                            return ordering;
                        }
                    }
                    a.pre.len().cmp(&b.pre.len())
                }
            }
        })
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: CompareOp,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum CompareOp {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut tokens = Vec::new();
    let mut pos = 0usize;

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }
        let start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        tokens.push((start, pos));
    }

    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut comparators = Vec::with_capacity(tokens.len());
    for (start, end) in tokens {
        let token = &bytes[start..end];
        let (op, operator_len) = if token.starts_with(b">=") {
            (CompareOp::Ge, 2)
        } else if token.starts_with(b"<=") {
            (CompareOp::Le, 2)
        } else if token.starts_with(b">") {
            (CompareOp::Gt, 1)
        } else if token.starts_with(b"<") {
            (CompareOp::Lt, 1)
        } else if token.starts_with(b"=") {
            (CompareOp::Eq, 1)
        } else {
            (CompareOp::Eq, 0)
        };

        let operand_start = start + operator_len;
        if operand_start >= end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: start });
        }

        let operand = &s[operand_start..end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        comparators.push(Comparator { op, version });
    }

    Ok(Range {
        arms: vec![comparators],
    })
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let admitted = !v.pre.is_empty()
            && arm.iter().any(|comparator| {
                !comparator.version.pre.is_empty()
                    && (v.major, v.minor, v.patch)
                        == (
                            comparator.version.major,
                            comparator.version.minor,
                            comparator.version.patch,
                        )
            });

        if !v.pre.is_empty() && !admitted {
            return false;
        }

        arm.iter().all(|comparator| {
            let ordering = compare_versions(v, &comparator.version);
            match comparator.op {
                CompareOp::Eq => ordering == Ordering::Equal,
                CompareOp::Lt => ordering == Ordering::Less,
                CompareOp::Le => ordering != Ordering::Greater,
                CompareOp::Gt => ordering == Ordering::Greater,
                CompareOp::Ge => ordering != Ordering::Less,
            }
        })
    })
}
