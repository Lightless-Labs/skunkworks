#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

pub struct Encoder {
    buffer: [u8; 2],
    buffered: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            buffer: [0; 2],
            buffered: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = String::with_capacity((self.buffered + input.len()) / 3 * 4);
        let mut index = 0;

        if self.buffered != 0 {
            while self.buffered < 3 && index < input.len() {
                if self.buffered < 2 {
                    self.buffer[self.buffered] = input[index];
                }
                self.buffered += 1;
                index += 1;
            }

            if self.buffered == 3 {
                let bytes = [self.buffer[0], self.buffer[1], input[index - 1]];
                encode_group(&bytes, &mut out);
                self.buffered = 0;
            } else {
                return Ok(out);
            }
        }

        while index + 3 <= input.len() {
            encode_group(
                &[input[index], input[index + 1], input[index + 2]],
                &mut out,
            );
            index += 3;
        }

        self.buffered = input.len() - index;
        for i in 0..self.buffered {
            self.buffer[i] = input[index + i];
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut out = String::new();
        match self.buffered {
            1 => {
                let a = self.buffer[0];
                out.push(ALPHABET[(a >> 2) as usize] as char);
                out.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                out.push_str("==");
            }
            2 => {
                let a = self.buffer[0];
                let b = self.buffer[1];
                out.push(ALPHABET[(a >> 2) as usize] as char);
                out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                out.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                out.push('=');
            }
            _ => {}
        }
        self.buffered = 0;
        Ok(out)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    pending_second_padding: Option<usize>,
    terminal_padding: bool,
    next_offset: usize,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            len: 0,
            pending_second_padding: None,
            terminal_padding: false,
            next_offset: 0,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();

        for &byte in input {
            let offset = self.next_offset;
            self.next_offset += 1;

            let result = self.process_byte(byte, offset, &mut out);
            if let Err(error) = result {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }

        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            let error = Error::BadPadding { offset };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        if self.len != 0 {
            let error = Error::Truncated {
                offset: self.offsets[0],
            };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        self.finished = true;
        Ok(Vec::new())
    }

    fn process_byte(
        &mut self,
        byte: u8,
        offset: usize,
        out: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.pending_second_padding.is_some() {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }

            self.group[3] = b'=';
            self.offsets[3] = offset;
            self.pending_second_padding = None;
            self.emit_group(out);
            self.terminal_padding = true;
            return Ok(());
        }

        let position = self.len;
        if position < 2 && byte == b'=' {
            return Err(Error::BadPadding { offset });
        }

        if position == 2 && byte == b'=' {
            self.group[2] = byte;
            self.offsets[2] = offset;
            self.len = 3;
            self.pending_second_padding = Some(offset);
            return Ok(());
        }

        if byte != b'=' && alphabet_value(byte).is_none() {
            return Err(Error::InvalidByte { offset, byte });
        }

        self.group[position] = byte;
        self.offsets[position] = offset;
        self.len += 1;

        if self.len == 4 {
            if self.group[3] == b'=' {
                self.terminal_padding = true;
            }
            self.emit_group(out);
        }

        Ok(())
    }

    fn emit_group(&mut self, out: &mut Vec<u8>) {
        let a = alphabet_value(self.group[0]).unwrap_or(0);
        let b = alphabet_value(self.group[1]).unwrap_or(0);
        out.push((a << 2) | (b >> 4));

        if self.group[2] != b'=' {
            let c = alphabet_value(self.group[2]).unwrap_or(0);
            out.push((b << 4) | (c >> 2));

            if self.group[3] != b'=' {
                let d = alphabet_value(self.group[3]).unwrap_or(0);
                out.push((c << 6) | d);
            }
        }

        self.len = 0;
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("new encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn alphabet_value(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn encode_group(bytes: &[u8; 3], out: &mut String) {
    let a = bytes[0];
    let b = bytes[1];
    let c = bytes[2];

    out.push(ALPHABET[(a >> 2) as usize] as char);
    out.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    out.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    out.push(ALPHABET[(c & 0x3f) as usize] as char);
}
