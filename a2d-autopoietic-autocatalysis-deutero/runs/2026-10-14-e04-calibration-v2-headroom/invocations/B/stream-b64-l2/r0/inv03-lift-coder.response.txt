#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("new encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len != 0 {
            while self.tail_len < 2 && index < input.len() {
                self.tail[self.tail_len] = input[index];
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 2 && index < input.len() {
                encode_triplet(
                    &mut output,
                    self.tail[0],
                    self.tail[1],
                    input[index],
                );
                self.tail_len = 0;
                index += 1;
            }
        }

        while index + 3 <= input.len() {
            encode_triplet(
                &mut output,
                input[index],
                input[index + 1],
                input[index + 2],
            );
            index += 3;
        }

        while index < input.len() {
            self.tail[self.tail_len] = input[index];
            self.tail_len += 1;
            index += 1;
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            1 => {
                let a = self.tail[0];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                output.push_str("==");
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => {}
        }
        self.tail_len = 0;
        Ok(output)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

fn encode_triplet(output: &mut String, a: u8, b: u8, c: u8) {
    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    output.push(ALPHABET[(c & 0x3f) as usize] as char);
}

pub struct Decoder {
    group: [u8; 4],
    group_len: usize,
    group_start: usize,
    pending_second_padding: Option<usize>,
    terminal_padding: bool,
    offset: usize,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            group_len: 0,
            group_start: 0,
            pending_second_padding: None,
            terminal_padding: false,
            offset: 0,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let current_offset = self.offset;
            self.offset += 1;

            let error = if self.terminal_padding {
                Some(Error::BadPadding {
                    offset: current_offset,
                })
            } else if self.pending_second_padding.is_some() {
                if byte == b'=' {
                    self.group[3] = b'=';
                    self.group_len = 4;
                    self.pending_second_padding = None;
                    self.terminal_padding = true;
                    output.push((self.group[0] << 2) | (self.group[1] >> 4));
                    self.group_len = 0;
                    self.group_start = self.offset;
                    None
                } else {
                    Some(Error::BadPadding {
                        offset: current_offset,
                    })
                }
            } else if self.group_len == 0 {
                if byte == b'=' {
                    Some(Error::BadPadding {
                        offset: current_offset,
                    })
                } else if let Some(value) = digit(byte) {
                    self.group_start = current_offset;
                    self.group[0] = value;
                    self.group_len = 1;
                    None
                } else {
                    Some(Error::InvalidByte {
                        offset: current_offset,
                        byte,
                    })
                }
            } else if self.group_len == 1 {
                if byte == b'=' {
                    Some(Error::BadPadding {
                        offset: current_offset,
                    })
                } else if let Some(value) = digit(byte) {
                    self.group[1] = value;
                    self.group_len = 2;
                    None
                } else {
                    Some(Error::InvalidByte {
                        offset: current_offset,
                        byte,
                    })
                }
            } else if self.group_len == 2 {
                if byte == b'=' {
                    self.pending_second_padding = Some(current_offset);
                    self.group[2] = b'=';
                    self.group_len = 3;
                    None
                } else if let Some(value) = digit(byte) {
                    self.group[2] = value;
                    self.group_len = 3;
                    None
                } else {
                    Some(Error::InvalidByte {
                        offset: current_offset,
                        byte,
                    })
                }
            } else if self.group_len == 3 {
                if byte == b'=' {
                    self.group[3] = b'=';
                    self.group_len = 0;
                    self.terminal_padding = true;
                    output.push((self.group[0] << 2) | (self.group[1] >> 4));
                    output.push((self.group[1] << 4) | (self.group[2] >> 2));
                    None
                } else if let Some(value) = digit(byte) {
                    self.group[3] = value;
                    output.push((self.group[0] << 2) | (self.group[1] >> 4));
                    output.push((self.group[1] << 4) | (self.group[2] >> 2));
                    output.push((self.group[2] << 6) | value);
                    self.group_len = 0;
                    self.group_start = self.offset;
                    None
                } else {
                    Some(Error::InvalidByte {
                        offset: current_offset,
                        byte,
                    })
                }
            } else {
                unreachable!()
            };

            if let Some(error) = error {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            let error = Error::BadPadding { offset };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        if self.group_len != 0 {
            let error = Error::Truncated {
                offset: self.group_start,
            };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}
