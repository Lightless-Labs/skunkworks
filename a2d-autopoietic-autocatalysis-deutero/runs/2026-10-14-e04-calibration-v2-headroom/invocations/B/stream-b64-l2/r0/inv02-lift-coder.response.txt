const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::with_capacity((self.tail_len + input.len()) / 3 * 4);
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 3 && index < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[index];
                }
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 3 {
                encode_triplet(
                    self.tail[0],
                    self.tail[1],
                    input[index - 1],
                    &mut output,
                );
                self.tail_len = 0;
            } else {
                return Ok(output);
            }
        }

        while index + 3 <= input.len() {
            encode_triplet(
                input[index],
                input[index + 1],
                input[index + 2],
                &mut output,
            );
            index += 3;
        }

        self.tail_len = input.len() - index;
        for i in 0..self.tail_len {
            self.tail[i] = input[index + i];
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            1 => {
                let a = self.tail[0];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                output.push('=');
                output.push('=');
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => {}
        }
        self.tail_len = 0;
        Ok(output)
    }
}

fn encode_triplet(a: u8, b: u8, c: u8, output: &mut String) {
    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    output.push(ALPHABET[(c & 0x3f) as usize] as char);
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    group_len: usize,
    pending_padding: bool,
    terminal_padding: bool,
    next_offset: usize,
    finished: bool,
    error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            group_len: 0,
            pending_padding: false,
            terminal_padding: false,
            next_offset: 0,
            finished: false,
            error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let offset = self.next_offset;
            self.next_offset += 1;

            let result = self.scan_byte(byte, offset, &mut output);
            if let Err(error) = result {
                self.error = Some(error.clone());
                return Err(error);
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if self.pending_padding {
            let error = Error::BadPadding {
                offset: self.offsets[2],
            };
            self.error = Some(error.clone());
            return Err(error);
        }

        if self.group_len != 0 {
            let error = Error::Truncated {
                offset: self.offsets[0],
            };
            self.error = Some(error.clone());
            return Err(error);
        }

        self.finished = true;
        Ok(Vec::new())
    }

    fn scan_byte(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.pending_padding {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.pending_padding = false;
            self.terminal_padding = true;
            self.group_len = 0;
            return Ok(());
        }

        let position = self.group_len;
        if byte == b'=' {
            match position {
                0 | 1 => return Err(Error::BadPadding { offset }),
                2 => {
                    self.pending_padding = true;
                    self.offsets[2] = offset;
                    return Ok(());
                }
                3 => {
                    let a = self.group[0];
                    let b = self.group[1];
                    let c = self.group[2];
                    output.push((a << 2) | (b >> 4));
                    output.push((b << 4) | (c >> 2));
                    self.group_len = 0;
                    self.terminal_padding = true;
                    return Ok(());
                }
                _ => unreachable!(),
            }
        }

        let value = match alphabet_value(byte) {
            Some(value) => value,
            None => return Err(Error::InvalidByte { offset, byte }),
        };

        self.group[position] = value;
        self.offsets[position] = offset;
        self.group_len += 1;

        if self.group_len == 4 {
            let a = self.group[0];
            let b = self.group[1];
            let c = self.group[2];
            let d = self.group[3];
            output.push((a << 2) | (b >> 4));
            output.push((b << 4) | (c >> 2));
            output.push((c << 6) | d);
            self.group_len = 0;
        }

        Ok(())
    }
}

fn alphabet_value(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("new encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}
