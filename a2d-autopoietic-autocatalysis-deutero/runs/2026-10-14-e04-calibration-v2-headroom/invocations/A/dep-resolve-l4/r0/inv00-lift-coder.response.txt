fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort();
    out.dedup();
    out
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_text(evidence: &[Evidence]) -> Vec<String> {
    let mut e = evidence.to_vec();
    e.sort_by(|a, b| a.constraint.cmp(&b.constraint).then(a.path.cmp(&b.path)));
    e.iter()
        .map(|e| {
            format!(
                "  C:{} via {}",
                constraint_text(e.constraint),
                e.path.join(" -> ")
            )
        })
        .collect()
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            append_evidence(format!("missing|package={}", encode(package)), evidence)
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn append_evidence(first: String, evidence: &[Evidence]) -> String {
    let lines = evidence_text(evidence);
    if lines.is_empty() {
        first
    } else {
        format!("{}\n{}", first, lines.join("\n"))
    }
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn target_matches(p: Platform, t: Target) -> bool {
    matches!(
        (p, t),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, name) in root.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: name.clone(),
                });
            }
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base,
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].gate",
                                i, j, k
                            ),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, name) in dep.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: name.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, name) in feature.enables.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: name.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for p in catalog {
        if !package_names.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }

    for p in catalog {
        let mut versions = BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }

    for p in catalog {
        for r in &p.releases {
            let mut features = BTreeSet::new();
            for f in &r.features {
                if !features.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: r.version,
                        feature: f.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    features: BTreeMap<String, BTreeSet<String>>,
    origins: BTreeMap<String, Vec<Origin>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn closure(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    assigned: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut c = Closure {
        constraints: BTreeMap::new(),
        features: BTreeMap::new(),
        origins: BTreeMap::new(),
        paths: BTreeMap::new(),
    };
    let mut queue: Vec<(String, String)> = Vec::new();

    for (i, root) in roots.iter().enumerate() {
        let fs: BTreeSet<String> = root.features.iter().cloned().collect();
        let token = format!(
            "root[{}]:{}@{}+{{{}}}",
            i,
            root.package,
            constraint_text(root.constraint),
            fs.iter().cloned().collect::<Vec<_>>().join(",")
        );
        c.constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        c.origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
        c.paths
            .entry(root.package.clone())
            .or_insert_with(|| vec![token]);
        let set = c.features.entry(root.package.clone()).or_default();
        for f in fs {
            if set.insert(f.clone()) {
                queue.push((root.package.clone(), f));
            }
        }
    }

    let mut emitted_deps: BTreeSet<(String, Version, usize)> = BTreeSet::new();
    let mut processed_features: BTreeSet<(String, Version, String)> = BTreeSet::new();

    loop {
        while let Some((package, feature_name)) = queue.pop() {
            let Some(version) = assigned.get(&package).copied() else {
                continue;
            };
            let Some(pkg) = catalog.get(&package) else {
                continue;
            };
            let Some(release) = pkg.releases.iter().find(|r| r.version == version) else {
                continue;
            };
            if !processed_features.insert((package.clone(), version, feature_name.clone())) {
                continue;
            }
            if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                for enabled in &feature.enables {
                    if c.features
                        .entry(package.clone())
                        .or_default()
                        .insert(enabled.clone())
                    {
                        queue.push((package.clone(), enabled.clone()));
                    }
                }
            }
        }

        let mut changed = false;
        let assigned_names: Vec<String> = assigned.keys().cloned().collect();
        for source in assigned_names {
            let Some(version) = assigned.get(&source).copied() else {
                continue;
            };
            let Some(pkg) = catalog.get(&source) else {
                continue;
            };
            let Some(release) = pkg.releases.iter().find(|r| r.version == version) else {
                continue;
            };
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !c.features.get(&source).is_some_and(|s| s.contains(g)))
                {
                    continue;
                }
                if !emitted_deps.insert((source.clone(), version, j)) {
                    continue;
                }
                let fs: BTreeSet<String> = dep.features.iter().cloned().collect();
                let feature_text = fs.iter().cloned().collect::<Vec<_>>().join(",");
                let gate = dep.gate.as_deref().unwrap_or("*");
                let platform = match dep.platform {
                    Platform::Any => "any",
                    Platform::Linux => "linux",
                    Platform::Mac => "mac",
                    Platform::Windows => "windows",
                };
                let token = format!(
                    "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                    source,
                    version_text(version),
                    j,
                    dep.package,
                    constraint_text(dep.constraint),
                    feature_text,
                    gate,
                    platform
                );
                let source_path = c.paths.get(&source).cloned().unwrap_or_default();
                let mut path = source_path;
                path.push(token);
                c.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                c.origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                let paths = c.paths.entry(dep.package.clone()).or_insert_with(|| path.clone());
                if path.len() < paths.len() || (path.len() == paths.len() && path < *paths) {
                    *paths = path;
                }
                let dest = c.features.entry(dep.package.clone()).or_default();
                for f in fs {
                    if dest.insert(f.clone()) {
                        queue.push((dep.package.clone(), f));
                    }
                }
                changed = true;
            }
        }

        if !queue.is_empty() {
            continue;
        }
        if !changed {
            break;
        }

        // Newly reachable paths may improve the provenance of already emitted edges.
        // Rebuild the closure on the next outer iteration if any dependency was added.
        // Edges remain unique, so this iteration terminates.
        if emitted_deps.len() > 0 {
            let mut any_unemitted = false;
            for source in assigned.keys() {
                let Some(version) = assigned.get(source) else { continue };
                if let Some(pkg) = catalog.get(source) {
                    if let Some(release) = pkg.releases.iter().find(|r| r.version == *version) {
                        if release.deps.iter().enumerate().any(|(j, dep)| {
                            target_matches(dep.platform, target)
                                && dep.gate.as_ref().is_none_or(|g| c.features.get(source).is_some_and(|s| s.contains(g)))
                                && !emitted_deps.contains(&(source.clone(), *version, j))
                        }) {
                            any_unemitted = true;
                            break;
                        }
                    }
                }
            }
            if !any_unemitted {
                break;
            }
        }
    }

    for cs in c.constraints.values_mut() {
        *cs = canonical_constraints(cs);
    }
    c
}

fn path_order(a: &[String], b: &[String]) -> std::cmp::Ordering {
    a.len().cmp(&b.len()).then(a.cmp(b))
}

struct SearchState {
    attempts: usize,
    max_attempts: usize,
    best: Option<(usize, Vec<Version>, BTreeMap<String, Version>, Closure)>,
    first_error: Option<ResolveError>,
}

fn candidate_versions(
    package: &str,
    constraints: &[Constraint],
    catalog: &BTreeMap<String, &Package>,
    locks: &BTreeMap<String, Version>,
) -> Vec<Version> {
    let mut versions: Vec<Version> = catalog[package]
        .releases
        .iter()
        .map(|r| r.version)
        .filter(|v| constraints.iter().all(|c| matches(*c, *v)))
        .collect();
    versions.sort_by(|a, b| b.cmp(a));
    if let Some(lock) = locks.get(package) {
        if let Some(i) = versions.iter().position(|v| v == lock) {
            let v = versions.remove(i);
            versions.insert(0, v);
        }
    }
    versions
}

fn vector_for(
    all_names: &[String],
    assigned: &BTreeMap<String, Version>,
) -> Vec<Version> {
    all_names
        .iter()
        .map(|n| assigned.get(n).copied().unwrap_or(Version { major: 0, minor: 0, patch: 0 }))
        .collect()
}

fn selected_error(
    assigned: &BTreeMap<String, Version>,
    c: &Closure,
    catalog: &BTreeMap<String, &Package>,
) -> Option<ResolveError> {
    for package in c.constraints.keys() {
        if !catalog.contains_key(package) {
            let mut evidence: Vec<Evidence> = c
                .origins
                .get(package)
                .into_iter()
                .flatten()
                .map(|o| Evidence {
                    constraint: o.constraint,
                    path: o.path.clone(),
                })
                .collect();
            evidence.sort_by(|a, b| {
                a.constraint
                    .cmp(&b.constraint)
                    .then(a.path.cmp(&b.path))
            });
            return Some(ResolveError::Missing {
                package: package.clone(),
                evidence,
            });
        }
    }

    for (package, version) in assigned {
        if let Some(reqs) = c.constraints.get(package) {
            if !reqs.iter().all(|constraint| matches(*constraint, *version)) {
                let evidence = c
                    .origins
                    .get(package)
                    .into_iter()
                    .flatten()
                    .map(|o| Evidence {
                        constraint: o.constraint,
                        path: o.path.clone(),
                    })
                    .collect();
                return Some(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: reqs.clone(),
                    evidence,
                });
            }
        }
    }

    for (package, reqs) in &c.constraints {
        if !assigned.contains_key(package)
            && candidate_versions(package, reqs, catalog, &BTreeMap::new()).is_empty()
        {
            let evidence = c
                .origins
                .get(package)
                .into_iter()
                .flatten()
                .map(|o| Evidence {
                    constraint: o.constraint,
                    path: o.path.clone(),
                })
                .collect();
            return Some(ResolveError::NoMatch {
                package: package.clone(),
                requirements: reqs.clone(),
                evidence,
            });
        }
    }

    for (package, version) in assigned {
        if let Some(names) = c.features.get(package) {
            let release = catalog[package]
                .releases
                .iter()
                .find(|r| r.version == *version)
                .unwrap();
            let declared: BTreeSet<&str> = release.features.iter().map(|f| f.name.as_str()).collect();
            for feature in names {
                if !declared.contains(feature.as_str()) {
                    return Some(ResolveError::MissingFeature {
                        package: package.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: c.paths.get(package).cloned().unwrap_or_default(),
                    });
                }
            }
        }
    }
    None
}

fn lock_cost(assigned: &BTreeMap<String, Version>, locks: &BTreeMap<String, Version>) -> usize {
    locks
        .iter()
        .filter(|(p, v)| assigned.get(*p) != Some(*v))
        .count()
}

fn search(
    assigned: &mut BTreeMap<String, Version>,
    roots: &[Requirement],
    catalog: &BTreeMap<String, &Package>,
    locks: &BTreeMap<String, Version>,
    target: Target,
    all_names: &[String],
    state: &mut SearchState,
) -> Result<(), ResolveError> {
    let c = closure(catalog, roots, assigned, target);
    if let Some(err) = selected_error(assigned, &c, catalog) {
        if matches!(err, ResolveError::Limit { .. }) {
            return Err(err);
        }
        if state.first_error.is_none() {
            state.first_error = Some(err);
        }
        return Ok(());
    }

    let next = c
        .constraints
        .keys()
        .find(|p| !assigned.contains_key(*p))
        .cloned();

    let Some(package) = next else {
        let cost = lock_cost(assigned, locks);
        let vector = vector_for(all_names, assigned);
        let replace = match &state.best {
            None => true,
            Some((best_cost, best_vector, _, _)) => {
                cost < *best_cost || (cost == *best_cost && vector > *best_vector)
            }
        };
        if replace {
            state.best = Some((cost, vector, assigned.clone(), c));
        }
        return Ok(());
    };

    let reqs = c.constraints.get(&package).cloned().unwrap_or_default();
    let candidates = candidate_versions(&package, &reqs, catalog, locks);
    for version in candidates {
        if state.attempts == state.max_attempts {
            return Err(ResolveError::Limit {
                attempts: state.max_attempts,
            });
        }
        state.attempts += 1;
        assigned.insert(package.clone(), version);
        search(
            assigned, roots, catalog, locks, target, all_names, state,
        )?;
        assigned.remove(&package);
    }
    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;
    let mut cat = BTreeMap::new();
    let mut names = Vec::new();
    for p in catalog {
        cat.insert(p.name.clone(), p);
        names.push(p.name.clone());
    }
    names.sort();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut state = SearchState {
        attempts: 0,
        max_attempts,
        best: None,
        first_error: None,
    };
    search(
        &mut BTreeMap::new(),
        roots,
        &cat,
        &lock_map,
        target,
        &names,
        &mut state,
    )?;

    let Some((_, _, chosen, final_closure)) = state.best else {
        return Err(state.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: roots
                .first()
                .map(|r| r.package.clone())
                .unwrap_or_default(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut all_packages: BTreeSet<String> = lock_map.keys().cloned().collect();
    all_packages.extend(chosen.keys().cloned());
    let mut changes = Vec::new();
    for package in all_packages {
        let before = lock_map.get(&package).copied();
        let after = chosen.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = cat
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_closure
                    .constraints
                    .get(&package)
                    .is_some_and(|cs| !cs.iter().all(|c| matches(*c, old)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
