#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression {
    operator: Option<char>,
    names: Vec<String>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[cursor..start]);
        append_expression(&mut output, expression, vars);
        cursor = end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn parse_template(template: &str) -> Result<Vec<(usize, usize, Expression)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let Some(relative_end) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = i + 1 + relative_end;

                if let Some(relative_open) = bytes[start + 1..end].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: start + 1 + relative_open,
                    });
                }

                let expression = parse_expression(template, start, end)?;
                expressions.push((start, end, expression));
                i = end + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn parse_expression(template: &str, start: usize, end: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;
    let operator = if content_start < end
        && matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
    {
        let op = bytes[content_start] as char;
        if op != '+' && op != '#' {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        content_start += 1;
        Some(op)
    } else {
        None
    };

    if content_start == end {
        return Err(Error::EmptyExpression { pos: end });
    }

    let mut names = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..end]
            .iter()
            .position(|&b| b == b',')
            .map_or(end, |offset| spec_start + offset);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..spec_end];
        for (offset, &byte) in name_bytes.iter().enumerate() {
            let valid = if offset == 0 {
                byte.is_ascii_alphabetic() || byte == b'_'
            } else {
                byte.is_ascii_alphanumeric() || byte == b'_'
            };
            if !valid {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let name = template[spec_start..spec_end].to_owned();
        if seen.iter().any(|previous| previous == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        names.push(name);

        if spec_end == end {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression { operator, names })
}

fn append_expression(output: &mut String, expression: Expression, vars: &[(&str, &str)]) {
    let mut fields = Vec::new();

    for name in &expression.names {
        if let Some(value) = vars
            .iter()
            .rev()
            .find_map(|(binding_name, value)| (*binding_name == name).then_some(*value))
        {
            fields.push(encode_value(value, matches!(expression.operator, Some('+' | '#'))));
        }
    }

    if fields.is_empty() {
        return;
    }

    if expression.operator == Some('#') {
        output.push('#');
    }
    output.push_str(&fields.join(","));
}

fn encode_value(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut encoded = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%'
            && i + 2 < bytes.len()
            && is_ascii_hex(bytes[i + 1])
            && is_ascii_hex(bytes[i + 2])
        {
            encoded.push('%');
            encoded.push((bytes[i + 1] as char).to_ascii_uppercase());
            encoded.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            encoded.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            encoded.push('%');
            encoded.push(HEX[(byte >> 4) as usize] as char);
            encoded.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    encoded
}

fn is_ascii_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}
