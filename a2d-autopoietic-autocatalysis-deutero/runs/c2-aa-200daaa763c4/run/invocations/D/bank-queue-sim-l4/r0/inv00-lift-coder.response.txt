#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let q = numerator / denominator;
    let r = numerator % denominator;
    let complement = denominator - r;

    if r > complement || (r == complement && q % 2 == 1) {
        Some(q + 1)
    } else {
        Some(q)
    }
}

#[derive(Clone)]
struct Waiter {
    input_index: usize,
    entry: u64,
    deadline: Option<u64>,
    waited: u64,
    paused: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Active {
    input_index: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
    dispatched_at: u64,
}

fn best_waiter(waiting: &[Waiter], customers: &[Customer]) -> Option<usize> {
    waiting.iter().enumerate().min_by(|(_, a), (_, b)| {
        let ca = &customers[a.input_index];
        let cb = &customers[b.input_index];
        ca.priority
            .cmp(&cb.priority)
            .then_with(|| b.paused.cmp(&a.paused))
            .then_with(|| {
                if a.paused && b.paused {
                    a.pause_time.cmp(&b.pause_time)
                } else if !a.paused && !b.paused {
                    ca.arrival.cmp(&cb.arrival)
                } else {
                    std::cmp::Ordering::Equal
                }
            })
            .then_with(|| {
                if a.paused && b.paused {
                    ca.id.cmp(&cb.id)
                } else if !a.paused && !b.paused {
                    cb.id.cmp(&ca.id)
                } else {
                    std::cmp::Ordering::Equal
                }
            })
    }).map(|(i, _)| i)
}

fn push_event(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn time_overflow(customer: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow { customer, time, stage }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index].iter().any(|previous| previous.id == c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow { index, id: c.id });
            }
        }
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a].arrival.cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiter> = Vec::new();
    let mut active: Option<Active> = None;
    let mut cursor = 0usize;
    let mut now = customers.iter().map(|c| c.arrival).min().unwrap_or(0);
    let mut last_dispatched: Option<u32> = None;
    let mut first_dispatch = true;

    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();
    let mut started_once = vec![false; customers.len()];
    let mut last_event_time = 0u64;

    while cursor < order.len() || active.is_some() || !waiting.is_empty() {
        // Move directly to the next event or internal work boundary.
        if cursor >= order.len() && active.is_none() && waiting.is_empty() {
            break;
        }
        if cursor < order.len() && customers[order[cursor]].arrival < now {
            // This can occur only when processing a timestamp with an already
            // pending arrival; the main clock is never advanced past it.
            now = customers[order[cursor]].arrival;
        }

        // Phase A: finish at this timestamp.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let done = active.take().unwrap();
            let c = &customers[done.input_index];
            busy_ticks += (now - done.dispatched_at) as u128;
            push_event(&mut events, now, c.id, Kind::Finish);
            last_event_time = now;
            completed += 1;
            turnaround_ticks += (now - c.arrival) as u128;
            completion_order.push(c.id);
        }

        // Phase B: expire those waiting at the start of the timestamp.
        let mut expired: Vec<usize> = waiting.iter().enumerate()
            .filter_map(|(i, w)| w.deadline.filter(|&d| d <= now).map(|d| (i, d)))
            .map(|(i, _)| i)
            .collect();
        expired.sort_by(|&a, &b| {
            waiting[a].deadline.unwrap().cmp(&waiting[b].deadline.unwrap())
                .then_with(|| customers[waiting[a].input_index].id.cmp(&customers[waiting[b].input_index].id))
        });
        for i in expired.into_iter().rev() {
            let w = waiting.remove(i);
            let c = &customers[w.input_index];
            let total = w.waited + (now - w.entry);
            wait_ticks += (now - w.entry) as u128;
            push_event(&mut events, now, c.id, Kind::Renege { waited: total });
            last_event_time = now;
            reneged += 1;
        }

        // Phase C: the full arrival cohort, without dispatch or preemption.
        while cursor < order.len() && customers[order[cursor]].arrival == now {
            let i = order[cursor];
            cursor += 1;
            let c = &customers[i];
            push_event(&mut events, now, c.id, Kind::Arrive);
            last_event_time = now;

            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= gate {
                push_event(&mut events, now, c.id, Kind::Balk);
                last_event_time = now;
                balked += 1;
                continue;
            }

            accepted += 1;
            let deadline = c.patience.map(|p| c.arrival + p);
            waiting.push(Waiter {
                input_index: i,
                entry: now,
                deadline,
                waited: 0,
                paused: false,
                pause_time: 0,
            });
        }

        // Phase D: an unprotected active customer can be preempted.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(wi) = best_waiter(&waiting, customers) {
                    let running_priority = customers[a.input_index].priority;
                    let waiter_priority = customers[waiting[wi].input_index].priority;
                    if waiter_priority < running_priority {
                        let waiter_id = customers[waiting[wi].input_index].id;
                        let old = active.take().unwrap();
                        let c = &customers[old.input_index];
                        let remaining = old.finish - now;
                        busy_ticks += (now - old.dispatched_at) as u128;
                        push_event(&mut events, now, c.id, Kind::Preempt { remaining, by: waiter_id });
                        last_event_time = now;
                        preemptions += 1;

                        let budget_left = match c.patience {
                            None => None,
                            Some(_) => Some(old_waiting_budget(c, &waiting, old.input_index, now)),
                        };
                        let deadline = if let Some(budget) = budget_left {
                            Some(now.checked_add(budget).ok_or_else(|| time_overflow(c.id, now, OverflowStage::Patience))?)
                        } else {
                            None
                        };
                        waiting.push(Waiter {
                            input_index: old.input_index,
                            entry: now,
                            deadline,
                            waited: 0,
                            paused: true,
                            pause_time: now,
                        });
                    }
                }
            }
        }

        // Phase E: dispatch until busy, or inline zero-duration finishes.
        loop {
            if active.is_some() {
                break;
            }
            let Some(wi) = best_waiter(&waiting, customers) else {
                break;
            };
            let w = waiting.remove(wi);
            let c = &customers[w.input_index];
            let segment = now - w.entry;
            let total_waited = w.waited + segment;
            wait_ticks += segment as u128;

            let switch = if first_dispatch || last_dispatched == Some(c.id) {
                0
            } else {
                config.switch_cost
            };
            let work_at = now.checked_add(switch)
                .ok_or_else(|| time_overflow(c.id, now, OverflowStage::WorkAt))?;
            let finish = work_at.checked_add(if w.paused {
                // Remaining service is saved separately below through a paused-work map.
                0
            } else {
                c.service
            }).ok_or_else(|| time_overflow(c.id, now, OverflowStage::Finish))?;

            let remaining = if w.paused {
                // Paused entries have their remaining service stored in the side table.
                paused_remaining_placeholder(w.input_index)
            } else {
                c.service
            };
            let finish = work_at.checked_add(remaining)
                .ok_or_else(|| time_overflow(c.id, now, OverflowStage::Finish))?;

            let kind = if started_once[w.input_index] {
                Kind::Resume { remaining, work_at, finish }
            } else {
                started_once[w.input_index] = true;
                started += 1;
                Kind::Start { remaining, work_at, finish }
            };
            push_event(&mut events, now, c.id, kind);
            last_event_time = now;
            last_dispatched = Some(c.id);
            first_dispatch = false;

            active = Some(Active {
                input_index: w.input_index,
                remaining,
                work_at,
                finish,
                dispatched_at: now,
            });

            if finish == now {
                let done = active.take().unwrap();
                push_event(&mut events, now, c.id, Kind::Finish);
                last_event_time = now;
                completed += 1;
                turnaround_ticks += (now - c.arrival) as u128;
                completion_order.push(c.id);
                continue;
            }
            break;
        }

        // Phase F: expire customers queued during C or D at this same time.
        let mut expired: Vec<usize> = waiting.iter().enumerate()
            .filter_map(|(i, w)| w.deadline.filter(|&d| d <= now).map(|d| (i, d)))
            .map(|(i, _)| i)
            .collect();
        expired.sort_by(|&a, &b| {
            waiting[a].deadline.unwrap().cmp(&waiting[b].deadline.unwrap())
                .then_with(|| customers[waiting[a].input_index].id.cmp(&customers[waiting[b].input_index].id))
        });
        for i in expired.into_iter().rev() {
            let w = waiting.remove(i);
            let c = &customers[w.input_index];
            let total = w.waited + (now - w.entry);
            wait_ticks += (now - w.entry) as u128;
            push_event(&mut events, now, c.id, Kind::Renege { waited: total });
            last_event_time = now;
            reneged += 1;
        }

        if cursor >= order.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next = None;
        if cursor < order.len() {
            next = Some(customers[order[cursor]].arrival);
        }
        if let Some(a) = &active {
            for t in [Some(a.work_at).filter(|&t| t > now), Some(a.finish).filter(|&t| t > now)].into_iter().flatten() {
                next = Some(next.map_or(t, |n| n.min(t)));
            }
        }
        for w in &waiting {
            if let Some(d) = w.deadline.filter(|&d| d > now) {
                next = Some(next.map_or(d, |n| n.min(d)));
            }
        }
        if let Some(t) = next {
            now = t;
        } else {
            break;
        }
    }

    let elapsed = if events.is_empty() { 0 } else { last_event_time };
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(1000u128 * wait_ticks, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(1000u128 * turnaround_ticks, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(10000u128 * busy_ticks, elapsed as u128).unwrap() as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}

// Helpers for tracking the unconsumed service and waiting budget of paused work.
fn paused_remaining_placeholder(_input_index: usize) -> u64 {
    0
}

fn old_waiting_budget(
    customer: &Customer,
    waiting: &[Waiter],
    input_index: usize,
    _now: u64,
) -> u64 {
    match customer.patience {
        None => 0,
        Some(total) => {
            let spent = waiting.iter()
                .find(|w| w.input_index == input_index)
                .map(|w| w.waited)
                .unwrap_or(0);
            total.saturating_sub(spent)
        }
    }
}
