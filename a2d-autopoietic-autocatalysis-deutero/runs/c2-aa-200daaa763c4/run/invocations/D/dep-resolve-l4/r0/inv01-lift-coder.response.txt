fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    (1..=24).contains(&bytes.len())
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn canonical_constraints(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort_by(|a, b| {
        constraint_rank(*a)
            .cmp(&constraint_rank(*b))
            .then_with(|| constraint_version(*a).cmp(&constraint_version(*b)))
    });
    constraints.dedup();
    constraints
}

fn constraint_rank(c: Constraint) -> u8 {
    match c {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn constraint_version(c: Constraint) -> Version {
    match c {
        Constraint::Exact(v)
        | Constraint::AtLeast(v)
        | Constraint::Before(v)
        | Constraint::Caret(v) => v,
    }
}

fn constraint_matches(c: Constraint, version: Version) -> bool {
    match c {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn satisfies(constraints: &[Constraint], version: Version) -> bool {
    constraints
        .iter()
        .all(|constraint| constraint_matches(*constraint, version))
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    let prefix = match constraint {
        Constraint::Exact(_) => "=",
        Constraint::AtLeast(_) => ">=",
        Constraint::Before(_) => "<",
        Constraint::Caret(_) => "^",
    };
    format!("{prefix}{}", version_text(constraint_version(constraint)))
}

fn percent_encode(value: &str) -> String {
    let mut output = String::new();
    for byte in value.bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'-') {
            output.push(byte as char);
        } else {
            output.push_str(&format!("%{byte:02X}"));
        }
    }
    output
}

fn requirements_text(requirements: &[Constraint]) -> String {
    canonical_constraints(requirements.to_vec())
        .iter()
        .map(|constraint| constraint_text(*constraint))
        .collect::<Vec<_>>()
        .join(",")
}

fn sorted_evidence(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut result = evidence.to_vec();
    result.sort_by(|a, b| {
        constraint_rank(a.constraint)
            .cmp(&constraint_rank(b.constraint))
            .then_with(|| constraint_version(a.constraint).cmp(&constraint_version(b.constraint)))
            .then_with(|| a.path.cmp(&b.path))
    });
    result
}

fn append_evidence(mut text: String, evidence: &[Evidence]) -> String {
    for entry in sorted_evidence(evidence) {
        text.push_str("\n  C:");
        text.push_str(&constraint_text(entry.constraint));
        text.push_str(" via ");
        text.push_str(&entry.path.join(" -> "));
    }
    text
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            percent_encode(site),
            percent_encode(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", percent_encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            percent_encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            percent_encode(package),
            version_text(*version),
            percent_encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", percent_encode(package))
        }
        ResolveError::Missing { package, evidence } => append_evidence(
            format!("missing|package={}", percent_encode(package)),
            evidence,
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "no-match|package={}|constraints={}",
                percent_encode(package),
                requirements_text(requirements)
            ),
            evidence,
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "conflict|package={}|selected={}|constraints={}",
                percent_encode(package),
                version_text(*selected),
                requirements_text(requirements)
            ),
            evidence,
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            percent_encode(package),
            version_text(*selected),
            percent_encode(feature),
            percent_encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

#[derive(Clone, Debug)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Debug, Default)]
struct Active {
    constraints: Vec<Constraint>,
    origins: Vec<Origin>,
    requested: BTreeSet<String>,
}

#[derive(Clone, Debug)]
struct Closure {
    active: BTreeMap<String, Active>,
    features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
    error: Option<ResolveError>,
}

fn platform_matches(platform: Platform, target: Target) -> bool {
    matches!(
        (platform, target),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn sorted_feature_names(features: &[String]) -> Vec<String> {
    features
        .iter()
        .cloned()
        .collect::<BTreeSet<_>>()
        .into_iter()
        .collect()
}

fn root_token(root: &Requirement, index: usize) -> String {
    format!(
        "root[{index}]:{}@{}+{{{}}}",
        root.package,
        constraint_text(root.constraint),
        sorted_feature_names(&root.features).join(",")
    )
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        sorted_feature_names(&dep.features).join(","),
        gate,
        platform
    )
}

fn path_is_better(candidate: &[String], current: &[String]) -> bool {
    candidate.len() < current.len()
        || (candidate.len() == current.len() && candidate < current)
}

fn build_closure(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let package_map: BTreeMap<&str, &Package> = catalog
        .iter()
        .map(|package| (package.name.as_str(), package))
        .collect();

    let mut active: BTreeMap<String, Active> = BTreeMap::new();
    let mut paths: BTreeMap<String, Vec<String>> = BTreeMap::new();

    for (index, root) in roots.iter().enumerate() {
        let path = vec![root_token(root, index)];
        let entry = active.entry(root.package.clone()).or_default();
        entry.constraints.push(root.constraint);
        entry.origins.push(Origin {
            constraint: root.constraint,
            path: path.clone(),
        });
        entry.requested.extend(root.features.iter().cloned());

        match paths.get(&root.package) {
            None => {
                paths.insert(root.package.clone(), path);
            }
            Some(old) if path_is_better(&path, old) => {
                paths.insert(root.package.clone(), path);
            }
            _ => {}
        }
    }

    let mut activated_features: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();
    let mut processed_dependencies: BTreeSet<(String, usize)> = BTreeSet::new();

    loop {
        let mut changed = false;
        let package_names: Vec<String> = active.keys().cloned().collect();

        for name in package_names {
            let Some(version) = selected.get(&name).copied() else {
                continue;
            };
            let Some(package) = package_map.get(name.as_str()).copied() else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested = active
                .get(&name)
                .map(|entry| entry.requested.clone())
                .unwrap_or_default();
            let feature_set = activated_features.entry(name.clone()).or_default();
            for feature in requested {
                if feature_set.insert(feature) {
                    changed = true;
                }
            }

            let current_features = activated_features
                .get(&name)
                .cloned()
                .unwrap_or_default();
            for feature_name in current_features {
                if let Some(declaration) = release
                    .features
                    .iter()
                    .find(|declaration| declaration.name == feature_name)
                {
                    for enabled in &declaration.enables {
                        if activated_features
                            .entry(name.clone())
                            .or_default()
                            .insert(enabled.clone())
                        {
                            changed = true;
                        }
                    }
                }
            }

            let current_features = activated_features
                .get(&name)
                .cloned()
                .unwrap_or_default();
            for (index, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|gate| !current_features.contains(gate))
                {
                    continue;
                }

                let key = (name.clone(), index);
                if processed_dependencies.insert(key) {
                    let mut path = paths.get(&name).cloned().unwrap_or_default();
                    path.push(dep_token(&name, version, index, dep));

                    let destination = active.entry(dep.package.clone()).or_default();
                    destination.constraints.push(dep.constraint);
                    destination.origins.push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                    destination.requested.extend(dep.features.iter().cloned());

                    match paths.get(&dep.package) {
                        None => {
                            paths.insert(dep.package.clone(), path);
                        }
                        Some(old) if path_is_better(&path, old) => {
                            paths.insert(dep.package.clone(), path);
                        }
                        _ => {}
                    }
                    changed = true;
                }
            }
        }

        if !changed {
            break;
        }
    }

    for entry in active.values_mut() {
        entry.constraints = canonical_constraints(std::mem::take(&mut entry.constraints));
    }

    let missing_package = active
        .keys()
        .find(|name| !package_map.contains_key(name.as_str()))
        .cloned();

    if let Some(package) = missing_package {
        let evidence = active[&package]
            .origins
            .iter()
            .map(|origin| Evidence {
                constraint: origin.constraint,
                path: origin.path.clone(),
            })
            .collect();
        return Closure {
            active,
            features: activated_features,
            paths,
            error: Some(ResolveError::Missing { package, evidence }),
        };
    }

    for (name, version) in selected {
        let Some(entry) = active.get(name) else {
            continue;
        };
        if !satisfies(&entry.constraints, *version) {
            let requirements = entry.constraints.clone();
            let evidence = entry
                .origins
                .iter()
                .map(|origin| Evidence {
                    constraint: origin.constraint,
                    path: origin.path.clone(),
                })
                .collect();
            return Closure {
                active,
                features: activated_features,
                paths,
                error: Some(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements,
                    evidence,
                }),
            };
        }
    }

    for (name, feature_names) in &activated_features {
        let Some(version) = selected.get(name).copied() else {
            continue;
        };
        let Some(package) = package_map.get(name.as_str()).copied() else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == version) else {
            continue;
        };

        if let Some(feature) = feature_names
            .iter()
            .find(|feature| !release.features.iter().any(|declaration| declaration.name == **feature))
        {
            return Closure {
                active,
                features: activated_features,
                paths: paths.clone(),
                error: Some(ResolveError::MissingFeature {
                    package: name.clone(),
                    selected: version,
                    feature: feature.clone(),
                    path: paths.get(name).cloned().unwrap_or_default(),
                }),
            };
        }
    }

    for (name, entry) in &active {
        if selected.contains_key(name) {
            continue;
        }
        let package = package_map[name.as_str()];
        if !package
            .releases
            .iter()
            .any(|release| satisfies(&entry.constraints, release.version))
        {
            let requirements = entry.constraints.clone();
            let evidence = entry
                .origins
                .iter()
                .map(|origin| Evidence {
                    constraint: origin.constraint,
                    path: origin.path.clone(),
                })
                .collect();
            return Closure {
                active,
                features: activated_features,
                paths,
                error: Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements,
                    evidence,
                }),
            };
        }
    }

    Closure {
        active,
        features: activated_features,
        paths,
        error: None,
    }
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>, Closure)>,
    first_error: Option<ResolveError>,
    limit: bool,
    catalog_map: BTreeMap<&'a str, &'a Package>,
    lock_map: BTreeMap<String, Version>,
    catalog_names: Vec<String>,
}

impl Search<'_> {
    fn visit(&mut self, selected: BTreeMap<String, Version>) {
        if self.limit {
            return;
        }

        let closure = build_closure(self.catalog, self.roots, &selected, self.target);
        if let Some(error) = closure.error.clone() {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return;
        }

        let next_package = closure
            .active
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();

        let Some(package_name) = next_package else {
            let cost = self
                .lock_map
                .iter()
                .filter(|(name, version)| selected.get(*name) != Some(*version))
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_selected, _)) if cost < *best_cost => true,
                Some((best_cost, best_selected, _)) if cost == *best_cost => {
                    self.vector_cmp(&selected, best_selected).is_gt()
                }
                _ => false,
            };
            if replace {
                self.best = Some((cost, selected, closure));
            }
            return;
        };

        let active = &closure.active[&package_name];
        let package = self.catalog_map[package_name.as_str()];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .filter(|release| satisfies(&active.constraints, release.version))
            .map(|release| release.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(locked) = self.lock_map.get(&package_name) {
            if let Some(index) = candidates.iter().position(|version| version == locked) {
                let version = candidates.remove(index);
                candidates.insert(0, version);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                self.limit = true;
                return;
            }
            self.attempts += 1;
            let mut child = selected.clone();
            child.insert(package_name.clone(), version);
            self.visit(child);
            if self.limit {
                return;
            }
        }
    }

    fn vector_cmp(
        &self,
        a: &BTreeMap<String, Version>,
        b: &BTreeMap<String, Version>,
    ) -> std::cmp::Ordering {
        for name in &self.catalog_names {
            let ordering = match (a.get(name), b.get(name)) {
                (None, None) => std::cmp::Ordering::Equal,
                (None, Some(_)) => std::cmp::Ordering::Less,
                (Some(_), None) => std::cmp::Ordering::Greater,
                (Some(left), Some(right)) => left.cmp(right),
            };
            if !ordering.is_eq() {
                return ordering;
            }
        }
        std::cmp::Ordering::Equal
    }
}

fn invalid_name(site: String, name: &str) -> ResolveError {
    ResolveError::InvalidName {
        site,
        name: name.to_string(),
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(invalid_name(format!("root[{i}]"), &root.package));
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(invalid_name(
                    format!("root[{i}].feature[{f}]"),
                    feature,
                ));
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(invalid_name(format!("catalog[{i}]"), &package.name));
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let site = format!("catalog[{i}].release[{j}].dep[{k}]");
                if !valid_name(&dep.package) {
                    return Err(invalid_name(site.clone(), &dep.package));
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(invalid_name(format!("{site}.gate"), gate));
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(invalid_name(format!("{site}.feature[{f}]"), feature));
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let site = format!("catalog[{i}].release[{j}].feature[{f}]");
                if !valid_name(&feature.name) {
                    return Err(invalid_name(site.clone(), &feature.name));
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(invalid_name(
                            format!("{site}.enable[{e}]"),
                            enabled,
                        ));
                    }
                }
            }
        }
    }

    let mut package_names = BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut feature_names = BTreeSet::new();
            for feature in &release.features {
                if !feature_names.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(invalid_name(format!("lock[{i}]"), &lock.package));
        }
    }

    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        limit: false,
        catalog_map: catalog
            .iter()
            .map(|package| (package.name.as_str(), package))
            .collect(),
        lock_map: locks
            .iter()
            .map(|lock| (lock.package.clone(), lock.version))
            .collect(),
        catalog_names: catalog.iter().map(|package| package.name.clone()).collect(),
    };
    search.catalog_names.sort();
    search.visit(BTreeMap::new());

    if search.limit {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }

    let Some((_, selected_map, closure)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = selected_map
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let before: BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();
    let after: BTreeMap<String, Version> = selected
        .iter()
        .map(|selection| (selection.package.clone(), selection.version))
        .collect();

    let names: BTreeSet<String> = before.keys().chain(after.keys()).cloned().collect();
    let mut changes = Vec::new();

    for name in names {
        let old = before.get(&name).copied();
        let new = after.get(&name).copied();
        if old == new {
            continue;
        }

        let reason = match (old, new) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old_version), Some(_)) => {
                let exists = search
                    .catalog_map
                    .get(name.as_str())
                    .is_some_and(|package| {
                        package
                            .releases
                            .iter()
                            .any(|release| release.version == old_version)
                    });
                if !exists {
                    ChangeReason::Unavailable
                } else if closure
                    .active
                    .get(&name)
                    .is_some_and(|entry| !satisfies(&entry.constraints, old_version))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };

        changes.push(Change {
            package: name,
            before: old,
            after: new,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
