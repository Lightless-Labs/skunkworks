    DuplicatePackage { package: String },

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if !(1..=24).contains(&bytes.len())
        || !bytes[0].is_ascii_lowercase()
        || bytes[bytes.len() - 1] == b'-'
    {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn canonical_constraints(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort();
    cs.dedup();
    cs
}

fn constraint_matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn satisfies(cs: &[Constraint], v: Version) -> bool {
    cs.iter().all(|c| constraint_matches(*c, v))
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn percent_encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| {
        a.constraint
            .cmp(&b.constraint)
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn requirements_text(requirements: &[Constraint]) -> String {
    canonical_constraints(requirements.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn append_evidence(mut first: String, evidence: &[Evidence]) -> String {
    for e in evidence_sorted(evidence) {
        first.push_str("\n  C:");
        first.push_str(&constraint_text(e.constraint));
        first.push_str(" via ");
        first.push_str(&e.path.join(" -> "));
    }
    first
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            percent_encode(site),
            percent_encode(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", percent_encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            percent_encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            percent_encode(package),
            version_text(*version),
            percent_encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", percent_encode(package))
        }
        ResolveError::Missing { package, evidence } => append_evidence(
            format!("missing|package={}", percent_encode(package)),
            evidence,
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "no-match|package={}|constraints={}",
                percent_encode(package),
                requirements_text(requirements)
            ),
            evidence,
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "conflict|package={}|selected={}|constraints={}",
                percent_encode(package),
                version_text(*selected),
                requirements_text(requirements)
            ),
            evidence,
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            percent_encode(package),
            version_text(*selected),
            percent_encode(feature),
            percent_encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone, Default)]
struct Active {
    constraints: Vec<Constraint>,
    origins: Vec<Origin>,
    requested: BTreeSet<String>,
}

struct Closure {
    active: BTreeMap<String, Active>,
    features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
    error: Option<ResolveError>,
}

use std::collections::{BTreeMap, BTreeSet};

fn target_platform(platform: Platform, target: Target) -> bool {
    matches!(platform, Platform::Any)
        || matches!(
            (platform, target),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn sorted_feature_list(features: &[String]) -> Vec<String> {
    features.iter().cloned().collect::<BTreeSet<_>>().into_iter().collect()
}

fn root_token(r: &Requirement, index: usize) -> String {
    let fs = sorted_feature_list(&r.features).join(",");
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        r.package,
        constraint_text(r.constraint),
        fs
    )
}

fn dep_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    let fs = sorted_feature_list(&dep.features).join(",");
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        fs,
        gate,
        platform
    )
}

fn better_path(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn build_closure(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let package_map: BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut active: BTreeMap<String, Active> = BTreeMap::new();
    let mut paths: BTreeMap<String, Vec<String>> = BTreeMap::new();

    for (i, root) in roots.iter().enumerate() {
        let token = root_token(root, i);
        let path = vec![token];
        active.entry(root.package.clone()).or_default().origins.push(Origin {
            constraint: root.constraint,
            path: path.clone(),
        });
        active
            .entry(root.package.clone())
            .or_default()
            .constraints
            .push(root.constraint);
        active
            .entry(root.package.clone())
            .or_default()
            .requested
            .extend(root.features.iter().cloned());
        match paths.get(&root.package) {
            None => {
                paths.insert(root.package.clone(), path);
            }
            Some(old) if better_path(&path, old) => {
                paths.insert(root.package.clone(), path);
            }
            _ => {}
        }
    }

    let mut features: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();
    let mut queue: Vec<String> = active.keys().cloned().collect();
    let mut processed_deps: BTreeSet<(String, usize)> = BTreeSet::new();

    loop {
        let mut changed = false;

        for name in &queue {
            let Some(version) = selected.get(name).copied() else {
                continue;
            };
            let Some(package) = package_map.get(name.as_str()).copied() else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let requested = active
                .get(name)
                .map(|a| a.requested.clone())
                .unwrap_or_default();
            let fset = features.entry(name.clone()).or_default();
            for feature in requested {
                if fset.insert(feature) {
                    changed = true;
                }
            }

            let current_features = features.get(name).cloned().unwrap_or_default();
            for feature_name in current_features {
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        if features.entry(name.clone()).or_default().insert(enabled.clone()) {
                            changed = true;
                        }
                    }
                }
            }

            let activated = features.get(name).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_platform(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !activated.contains(g))
                {
                    continue;
                }

                let key = (name.clone(), j);
                let source_path = paths.get(name).cloned().unwrap_or_default();
                let token = dep_token(name, version, j, dep);
                let mut dep_path = source_path;
                dep_path.push(token);

                let entry = active.entry(dep.package.clone()).or_default();
                if !processed_deps.contains(&key) {
                    entry.origins.push(Origin {
                        constraint: dep.constraint,
                        path: dep_path.clone(),
                    });
                    entry.constraints.push(dep.constraint);
                    entry.requested.extend(dep.features.iter().cloned());
                    processed_deps.insert(key);
                    changed = true;
                }

                match paths.get(&dep.package) {
                    None => {
                        paths.insert(dep.package.clone(), dep_path);
                        changed = true;
                    }
                    Some(old) if better_path(&dep_path, old) => {
                        paths.insert(dep.package.clone(), dep_path);
                        changed = true;
                    }
                    _ => {}
                }
            }
        }

        // Newly activated dependency requests can enable additional local features.
        let mut next_queue: Vec<String> = active.keys().cloned().collect();
        next_queue.sort();
        if !changed {
            break;
        }
        queue = next_queue;
    }

    for a in active.values_mut() {
        a.constraints = canonical_constraints(std::mem::take(&mut a.constraints));
    }

    let mut missing = active
        .keys()
        .filter(|name| !package_map.contains_key(name.as_str()))
        .cloned()
        .collect::<Vec<_>>();
    missing.sort();

    if let Some(package) = missing.first() {
        let evidence = active[package]
            .origins
            .iter()
            .map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            })
            .collect();
        return Closure {
            active,
            features,
            paths,
            error: Some(ResolveError::Missing { package: package.clone(), evidence }),
        };
    }

    for (name, version) in selected {
        let Some(a) = active.get(name) else {
            continue;
        };
        if !satisfies(&a.constraints, *version) {
            let evidence = a.origins.iter().map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            }).collect();
            return Closure {
                active,
                features,
                paths,
                error: Some(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: a.constraints.clone(),
                    evidence,
                }),
            };
        }
    }

    let mut feature_error: Option<ResolveError> = None;
    for (name, activated) in &features {
        let Some(version) = selected.get(name).copied() else {
            continue;
        };
        let Some(package) = package_map.get(name.as_str()).copied() else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == version) else {
            continue;
        };
        for feature in activated {
            if !release.features.iter().any(|f| f.name == *feature) {
                feature_error = Some(ResolveError::MissingFeature {
                    package: name.clone(),
                    selected: version,
                    feature: feature.clone(),
                    path: paths.get(name).cloned().unwrap_or_default(),
                });
                break;
            }
        }
        if feature_error.is_some() {
            break;
        }
    }
    if feature_error.is_some() {
        return Closure {
            active,
            features,
            paths,
            error: feature_error,
        };
    }

    for (name, a) in &active {
        if selected.contains_key(name) {
            continue;
        }
        let package = package_map[name.as_str()];
        let mut releases = package
            .releases
            .iter()
            .filter(|r| satisfies(&a.constraints, r.version))
            .collect::<Vec<_>>();
        if releases.is_empty() {
            let evidence = a.origins.iter().map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            }).collect();
            return Closure {
                active,
                features,
                paths,
                error: Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: a.constraints.clone(),
                    evidence,
                }),
            };
        }
        releases.clear();
    }

    Closure {
        active,
        features,
        paths,
        error: None,
    }
}

#[derive(Clone)]
struct SearchResult {
    selected: BTreeMap<String, Version>,
    closure: Closure,
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>, Closure)>,
    first_error: Option<ResolveError>,
    limit: bool,
    catalog_map: BTreeMap<&'a str, &'a Package>,
    lock_map: BTreeMap<String, Version>,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn visit(&mut self, selected: BTreeMap<String, Version>) {
        if self.limit {
            return;
        }
        let closure = build_closure(self.catalog, self.roots, &selected, self.target);
        if let Some(error) = closure.error.clone() {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return;
        }

        let next = closure
            .active
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();
        let Some(package_name) = next else {
            let cost = self
                .lock_map
                .iter()
                .filter(|(name, version)| selected.get(*name) != Some(*version))
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_selected, _)) if cost < *best_cost => true,
                Some((best_cost, best_selected, _)) if cost == *best_cost => {
                    self.vector_cmp(&selected, best_selected).is_gt()
                }
                _ => false,
            };
            if replace {
                self.best = Some((cost, selected, closure));
            }
            return;
        };

        let active = &closure.active[&package_name];
        let package = self.catalog_map[package_name.as_str()];
        let mut candidates = package
            .releases
            .iter()
            .filter(|r| satisfies(&active.constraints, r.version))
            .map(|r| r.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(locked) = self.lock_map.get(&package_name) {
            if let Some(pos) = candidates.iter().position(|v| v == locked) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                self.limit = true;
                return;
            }
            self.attempts += 1;
            let mut child = selected.clone();
            child.insert(package_name.clone(), version);
            self.visit(child);
            if self.limit {
                return;
            }
        }
    }

    fn vector_cmp(
        &self,
        a: &BTreeMap<String, Version>,
        b: &BTreeMap<String, Version>,
    ) -> std::cmp::Ordering {
        for name in &self.catalog_names {
            let av = a.get(name);
            let bv = b.get(name);
            let ord = match (av, bv) {
                (None, None) => std::cmp::Ordering::Equal,
                (None, Some(_)) => std::cmp::Ordering::Less,
                (Some(_), None) => std::cmp::Ordering::Greater,
                (Some(x), Some(y)) => x.cmp(y),
            };
            if !ord.is_eq() {
                return ord;
            }
        }
        std::cmp::Ordering::Equal
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let site = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: site.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.gate", site),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", site, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = BTreeSet::new();
            for feature in &release.features {
                if !seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        limit: false,
        catalog_map: catalog.iter().map(|p| (p.name.as_str(), p)).collect(),
        lock_map: locks
            .iter()
            .map(|l| (l.package.clone(), l.version))
            .collect(),
        catalog_names: catalog.iter().map(|p| p.name.clone()).collect(),
    };
    search.catalog_names.sort();
    search.visit(BTreeMap::new());

    if search.limit {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }

    let Some((_, selected_map, closure)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected = selected_map
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let mut before = BTreeMap::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }
    let mut after = BTreeMap::new();
    for selection in &selected {
        after.insert(selection.package.clone(), selection.version);
    }
    let mut names = BTreeSet::new();
    names.extend(before.keys().cloned());
    names.extend(after.keys().cloned());

    let mut changes = Vec::new();
    for name in names {
        let b = before.get(&name).copied();
        let a = after.get(&name).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search.catalog_map.get(name.as_str()).is_some_and(|p| {
                    p.releases.iter().any(|r| r.version == old)
                });
                if !exists {
                    ChangeReason::Unavailable
                } else if closure
                    .active
                    .get(&name)
                    .is_some_and(|a| !satisfies(&a.constraints, old))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
