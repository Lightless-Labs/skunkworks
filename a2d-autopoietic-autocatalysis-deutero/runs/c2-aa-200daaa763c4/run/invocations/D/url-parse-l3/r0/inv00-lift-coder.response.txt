#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn scan_global_bytes(input: &str) -> Result<(), usize> {
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(at);
        }
    }
    Ok(())
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn hex_value(byte: u8) -> u8 {
    match byte {
        b'0'..=b'9' => byte - b'0',
        b'a'..=b'f' => byte - b'a' + 10,
        b'A'..=b'F' => byte - b'A' + 10,
        _ => 0,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut source_offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() || !is_hex(bytes[i + 1]) || !is_hex(bytes[i + 2]) {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]));
            source_offsets.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            source_offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let offset = source_offsets
                .get(error.valid_up_to())
                .copied()
                .unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: offset })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for &byte in text.as_bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };

        if safe {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn component_parse_error(error: CodecError, base: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

fn validate_component(input: &str, base: usize, component: Component) -> Result<(), ParseError> {
    decode_component(component, input).map(|_| ()).map_err(|error| component_parse_error(error, base))
}

fn parse_port(raw: &str, absolute_start: usize) -> Result<u16, ParseError> {
    if raw.is_empty() {
        return Err(ParseError::InvalidPort { at: absolute_start });
    }
    for (offset, byte) in raw.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: absolute_start + offset,
            });
        }
    }
    if raw.len() >= 2 && raw.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: absolute_start });
    }

    let mut value: u32 = 0;
    for byte in raw.bytes() {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((byte - b'0') as u32)) {
            Some(value) => value,
            None => return Err(ParseError::InvalidPort { at: absolute_start }),
        };
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: absolute_start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    if let Err(at) = scan_global_bytes(input) {
        return Err(ParseError::InvalidByte { at });
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(colon) => colon,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(input.len());
    let authority = &input[authority_start..authority_end];

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(index, byte)| (byte == b'@').then_some(index))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, hostport_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(&authority[..at]), at + 1)
    } else {
        (None, 0)
    };

    let hostport = &authority[hostport_start..];
    let hostport_absolute = authority_start + hostport_start;
    let (host, port, ipv6) = if hostport.as_bytes().first() == Some(&b'[') {
        let opening = hostport_absolute;
        let close_rel = match hostport.as_bytes()[1..].iter().position(|&byte| byte == b']') {
            Some(offset) => offset + 1,
            None => return Err(ParseError::InvalidIpv6 { at: opening }),
        };
        let interior = &hostport[1..close_rel];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: opening });
        }
        let suffix = &hostport[close_rel + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.as_bytes()[0] == b':' {
            Some(parse_port(&suffix[1..], hostport_absolute + close_rel + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_absolute + close_rel + 1,
            });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let host_colon = hostport.bytes().position(|byte| byte == b':');
        let (host, port) = if let Some(colon_rel) = host_colon {
            let host = &hostport[..colon_rel];
            let port = parse_port(
                &hostport[colon_rel + 1..],
                hostport_absolute + colon_rel + 1,
            )?;
            (host, Some(port))
        } else {
            (hostport, None)
        };

        if host.is_empty() {
            return Err(ParseError::InvalidHost {
                at: hostport_absolute,
            });
        }
        for (offset, byte) in host.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_absolute + offset,
                });
            }
        }
        (host.to_ascii_lowercase(), port, false)
    };

    let remainder_start = authority_end;
    let fragment_rel = input[remainder_start..].find('#');
    let fragment_start = fragment_rel.map(|offset| remainder_start + offset);

    let pre_fragment_end = fragment_start.unwrap_or(input.len());
    let query_rel = input[remainder_start..pre_fragment_end].find('?');
    let query_start = query_rel.map(|offset| remainder_start + offset);

    let path_end = query_start.unwrap_or(pre_fragment_end);
    let path_raw = &input[remainder_start..path_end];
    let path = if path_raw.is_empty() { "/" } else { path_raw };

    let query = query_start.map(|start| &input[start + 1..pre_fragment_end]);
    let fragment = fragment_start.map(|start| &input[start + 1..]);

    if let Some(userinfo) = userinfo {
        validate_component(userinfo, authority_start, Component::UserInfo)?;
    }
    if !path_raw.is_empty() {
        validate_component(path_raw, remainder_start, Component::Path)?;
    }
    if let (Some(query), Some(start)) = (query, query_start) {
        validate_component(query, start + 1, Component::Query)?;
    }
    if let (Some(fragment), Some(start)) = (fragment, fragment_start) {
        validate_component(fragment, start + 1, Component::Fragment)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo: userinfo.map(str::to_owned),
        host,
        port,
        path: path.to_owned(),
        query: query.map(str::to_owned),
        fragment: fragment.map(str::to_owned),
        ipv6,
    })
}

fn path_error_from_codec(error: CodecError) -> PathError {
    match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(path_error_from_codec)?;

    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_navigation = false;

    for (index, segment) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(path_error_from_codec)?;
        let is_terminal = index + 1 == segments.len();
        match decoded.as_str() {
            "." => {
                terminal_navigation = is_terminal;
            }
            ".." => {
                if !stack.is_empty() {
                    stack.pop();
                }
                terminal_navigation = is_terminal;
            }
            _ => {
                stack.push(segment);
                terminal_navigation = false;
            }
        }
    }

    if terminal_navigation {
        stack.push("");
    }
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn split_reference(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let fragment_start = reference.find('#');
    let pre_fragment_end = fragment_start.unwrap_or(reference.len());
    let query_start = reference[..pre_fragment_end].find('?');
    let path_end = query_start.unwrap_or(pre_fragment_end);

    let query = query_start.map(|start| &reference[start + 1..pre_fragment_end]);
    let fragment = fragment_start.map(|start| &reference[start + 1..]);
    (&reference[..path_end], query, fragment)
}

fn normalise_url_path(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_url = normalise_url_path(base_url)?;
    if let Err(at) = scan_global_bytes(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    let first_boundary = reference
        .bytes()
        .position(|byte| matches!(byte, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_boundary].contains(':') {
        let url = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalise_url_path(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        return normalise_url_path(url);
    }

    let (reference_path, reference_query, reference_fragment) = split_reference(reference);

    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_url.path.clone(),
            if reference.contains('?') {
                reference_query.map(str::to_owned)
            } else {
                base_url.query.clone()
            },
        )
    } else {
        let selected_path = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = base_url.path.rfind('/').unwrap_or(0);
            format!("{}{}", &base_url.path[..=slash], reference_path)
        };
        (selected_path, reference_query.map(str::to_owned))
    };

    let authority = {
        let mut value = String::new();
        if let Some(userinfo) = &base_url.userinfo {
            value.push_str(userinfo);
            value.push('@');
        }
        if base_url.ipv6 {
            value.push('[');
            value.push_str(&base_url.host);
            value.push(']');
        } else {
            value.push_str(&base_url.host);
        }
        if let Some(port) = base_url.port {
            value.push(':');
            value.push_str(&port.to_string());
        }
        value
    };

    let mut absolute = format!("{}://{}{}", base_url.scheme, authority, selected_path);
    if let Some(query) = &selected_query {
        absolute.push('?');
        absolute.push_str(query);
    }
    if let Some(fragment) = reference_fragment {
        absolute.push('#');
        absolute.push_str(fragment);
    }

    let url = parse_url(&absolute).map_err(ResolveError::Reference)?;
    normalise_url_path(url)
}
