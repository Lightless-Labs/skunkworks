```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
struct Runtime {
    state: State,
    resume_remaining: Option<u64>,
    segment_started_at: u64,
}

pub struct Pool {
    jobs: Vec<Job>,
    deps: Vec<Vec<usize>>,
    runtime: Vec<Runtime>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    first_run: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut by_id = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job.tag.as_ref().is_some_and(|t| !valid_token(t, 1, 8)) {
                return Err(Error::BadTag(i));
            }
            if by_id.insert(job.id.as_str(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut deps = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (j, spelling) in job.dependencies.iter().enumerate() {
                if !seen.insert(spelling.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match by_id.get(spelling.as_str()) {
                    Some(&index) => deps[i].push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
        }

        let mut color = vec![0u8; jobs.len()];
        fn visit(i: usize, deps: &[Vec<usize>], color: &mut [u8]) -> bool {
            if color[i] == 1 {
                return false;
            }
            if color[i] == 2 {
                return true;
            }
            color[i] = 1;
            for &d in &deps[i] {
                if !visit(d, deps, color) {
                    return false;
                }
            }
            color[i] = 2;
            true
        }
        for i in 0..jobs.len() {
            if !visit(i, &deps, &mut color) {
                return Err(Error::Cycle);
            }
        }

        let runtime = jobs
            .iter()
            .map(|job| Runtime {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                resume_remaining: None,
                segment_started_at: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            deps,
            runtime,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            first_run: true,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        self.jobs
            .iter()
            .position(|job| job.id == id)
            .map(|i| self.runtime[i].state.clone())
            .ok_or(Error::Unknown)
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut lines = Vec::new();
        let mut now = if self.first_run { 0 } else { self.horizon };
        let mut process_now = self.first_run;

        loop {
            if process_now {
                self.process_timestamp(now, &mut lines);
                process_now = false;
            }

            let next = self.next_event_after(now);
            match next {
                Some(t) if t <= until => {
                    now = t;
                    process_now = true;
                }
                _ => break,
            }
        }

        self.horizon = until;
        self.first_run = false;
        Ok(lines)
    }

    fn process_timestamp(&mut self, now: u64, lines: &mut Vec<String>) {
        let mut completions: Vec<usize> = self
            .runtime
            .iter()
            .enumerate()
            .filter_map(|(i, rt)| match rt.state {
                State::Running { finish_at, .. } if finish_at == now => Some(i),
                _ => None,
            })
            .collect();
        completions.sort_by(|&a, &b| self.jobs[a].id.as_bytes().cmp(self.jobs[b].id.as_bytes()));

        for i in completions {
            let attempt = match self.runtime[i].state {
                State::Running { attempt, .. } => attempt,
                _ => continue,
            };
            let ok = attempt > self.jobs[i].failures;
            lines.push(format!(
                "{} finish {} {} {}",
                now,
                self.jobs[i].id,
                attempt,
                if ok { "ok" } else { "fail" }
            ));
            self.runtime[i].resume_remaining = None;
            if ok {
                self.runtime[i].state = State::Succeeded;
            } else if attempt >= self.jobs[i].max_attempts {
                self.runtime[i].state = State::Failed;
            } else {
                let delay = retry_delay(self.retry_base, self.retry_cap, attempt);
                let ready_at = now.saturating_add(delay);
                self.runtime[i].state = State::Waiting { ready_at, attempt };
                lines.push(format!(
                    "{} retry {} {} {}",
                    now, self.jobs[i].id, attempt, ready_at
                ));
            }
        }

        let mut newly_blocked = Vec::new();
        loop {
            let mut changed = false;
            for i in 0..self.jobs.len() {
                if !matches!(self.runtime[i].state, State::Waiting { .. }) {
                    continue;
                }
                if self.deps[i].iter().any(|&d| {
                    matches!(self.runtime[d].state, State::Failed | State::Blocked)
                }) {
                    self.runtime[i].state = State::Blocked;
                    newly_blocked.push(i);
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
        newly_blocked.sort_by(|&a, &b| self.jobs[a].id.as_bytes().cmp(self.jobs[b].id.as_bytes()));
        for i in newly_blocked {
            let dependency = self.deps[i]
                .iter()
                .filter(|&&d| matches!(self.runtime[d].state, State::Failed | State::Blocked))
                .map(|&d| self.jobs[d].id.as_str())
                .min()
                .unwrap_or("");
            lines.push(format!("{} skip {} {}", now, self.jobs[i].id, dependency));
        }

        loop {
            let mut candidates: Vec<usize> = (0..self.jobs.len())
                .filter(|&i| {
                    matches!(self.runtime[i].state, State::Waiting { ready_at, .. } if ready_at <= now)
                        && self.deps[i].iter().all(|&d| matches!(self.runtime[d].state, State::Succeeded))
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, now);
                let pb = self.effective_priority(b, now);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut selected = None;
            for candidate in candidates {
                let tag = self.jobs[candidate].tag.as_deref();
                let running: Vec<usize> = self
                    .runtime
                    .iter()
                    .enumerate()
                    .filter_map(|(i, rt)| matches!(rt.state, State::Running { .. }).then_some(i))
                    .collect();
                let slot_free = running.len() < self.capacity;
                let tag_free = tag.is_none()
                    || !running.iter().any(|&i| self.jobs[i].tag.as_deref() == tag);

                if slot_free && tag_free {
                    selected = Some((candidate, None));
                    break;
                }

                let mut victims: Vec<usize> = running
                    .iter()
                    .copied()
                    .filter(|&victim| {
                        self.jobs[candidate].priority >= self.jobs[victim].priority.saturating_add(2)
                    })
                    .filter(|&victim| {
                        let remaining_running = running.iter().filter(|&&i| i != victim).count();
                        let after_slot_free = remaining_running < self.capacity;
                        let after_tag_free = tag.is_none()
                            || !running
                                .iter()
                                .filter(|&&i| i != victim)
                                .any(|&i| self.jobs[i].tag.as_deref() == tag);
                        after_slot_free && after_tag_free
                    })
                    .collect();
                victims.sort_by(|&a, &b| {
                    self.jobs[a]
                        .priority
                        .cmp(&self.jobs[b].priority)
                        .then_with(|| {
                            let sa = self.runtime[a].segment_started_at;
                            let sb = self.runtime[b].segment_started_at;
                            sb.cmp(&sa)
                        })
                        .then_with(|| self.jobs[b].id.as_bytes().cmp(self.jobs[a].id.as_bytes()))
                });
                if let Some(victim) = victims.first().copied() {
                    selected = Some((candidate, Some(victim)));
                    break;
                }
            }

            let Some((candidate, victim)) = selected else {
                break;
            };

            if let Some(victim) = victim {
                let (attempt, finish_at) = match self.runtime[victim].state {
                    State::Running { attempt, finish_at } => (attempt, finish_at),
                    _ => continue,
                };
                let remaining = finish_at - now;
                lines.push(format!(
                    "{} preempt {} by {}",
                    now, self.jobs[victim].id, self.jobs[candidate].id
                ));
                self.runtime[victim].state = State::Waiting {
                    ready_at: now,
                    attempt,
                };
                self.runtime[victim].resume_remaining = Some(remaining);
            }

            let (attempt, remaining) = match self.runtime[candidate].state {
                State::Waiting { attempt, .. } => match self.runtime[candidate].resume_remaining {
                    Some(remaining) => (attempt, remaining),
                    None => (attempt.saturating_add(1), self.jobs[candidate].duration),
                },
                _ => continue,
            };
            self.runtime[candidate].resume_remaining = None;
            self.runtime[candidate].segment_started_at = now;
            let finish_at = now.saturating_add(remaining);
            self.runtime[candidate].state = State::Running { attempt, finish_at };
            lines.push(format!(
                "{} start {} {} {}",
                now, self.jobs[candidate].id, attempt, remaining
            ));
        }
    }

    fn effective_priority(&self, i: usize, now: u64) -> u8 {
        let (ready_at, _) = match self.runtime[i].state {
            State::Waiting { ready_at, attempt } => (ready_at, attempt),
            _ => return self.jobs[i].priority,
        };
        if self.aging_ms == 0 {
            return self.jobs[i].priority;
        }
        let base = self.jobs[i].priority as u64;
        let age = now.saturating_sub(ready_at) / self.aging_ms;
        (base + age).min(9) as u8
    }

    fn next_event_after(&self, now: u64) -> Option<u64> {
        self.runtime
            .iter()
            .filter_map(|rt| match rt.state {
                State::Running { finish_at, .. } if finish_at > now => Some(finish_at),
                State::Waiting { ready_at, .. } if ready_at > now => Some(ready_at),
                _ => None,
            })
            .min()
    }
}

fn valid_token(s: &str, min: usize, max: usize) -> bool {
    let b = s.as_bytes();
    (min..=max).contains(&b.len())
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn retry_delay(base: u64, cap: u64, attempt: u8) -> u64 {
    let mut delay = base;
    for _ in 1..attempt {
        delay = delay.saturating_mul(2).min(cap);
        if delay == cap {
            break;
        }
    }
    delay.min(cap)
}
```
