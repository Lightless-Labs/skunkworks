```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Process {
    id: u32,
    arrival: u64,
    bursts: Vec<u32>,
    burst_pos: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    level: usize,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|previous| previous.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut processes: Vec<Process> = jobs
        .iter()
        .map(|job| Process {
            id: job.id,
            arrival: u64::from(job.arrival),
            bursts: job.bursts.clone(),
            burst_pos: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
            level: job.level,
        })
        .collect();

    let mut queues: Vec<VecDeque<usize>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut now = 0u64;
    let mut last_dispatch_id: Option<u32> = None;

    while processes.iter().any(|process| process.completion.is_none()) {
        admit_events(&mut events, &mut queues, &mut processes, now);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            let next_time = events.iter().map(|event| event.time).min().unwrap();
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let id = processes[index].id;

        if last_dispatch_id.is_some() && last_dispatch_id != Some(id) {
            now += u64::from(switch_cost);
        }
        admit_events(&mut events, &mut queues, &mut processes, now);

        let quantum = quanta[level];
        let consumed = processes[index].remaining.min(quantum);
        let start = now;
        now += u64::from(consumed);
        processes[index].remaining -= consumed;

        let burst_finished = processes[index].remaining == 0;
        let job_finished = burst_finished
            && processes[index].burst_pos + 1 == processes[index].bursts.len();
        runs.push(Run {
            id,
            start,
            end: now,
            finished: job_finished,
            level,
        });

        admit_events(&mut events, &mut queues, &mut processes, now);

        if job_finished {
            processes[index].completion = Some(now);
        } else if burst_finished {
            processes[index].burst_pos += 2;
            let io_delay = u64::from(processes[index].bursts[processes[index].burst_pos - 1]);
            let wake_level = level.saturating_sub(1);
            let next_cpu = processes[index].bursts[processes[index].burst_pos];
            processes[index].remaining = next_cpu;
            events.push(Event {
                time: now + io_delay,
                kind: 1,
                index,
                level: wake_level,
            });
            admit_events(&mut events, &mut queues, &mut processes, now);
        } else {
            processes[index].level = (level + 1).min(quanta.len() - 1);
            queues[processes[index].level].push_back(index);
        }

        last_dispatch_id = Some(id);
    }

    let stats: Vec<Stat> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| {
            let completion = processes[index].completion.unwrap();
            let turnaround = completion - u64::from(job.arrival);
            let cpu: u64 = job
                .bursts
                .iter()
                .step_by(2)
                .map(|&length| u64::from(length))
                .sum();
            let io: u64 = job
                .bursts
                .iter()
                .skip(1)
                .step_by(2)
                .map(|&length| u64::from(length))
                .sum();
            Stat {
                id: job.id,
                completion,
                turnaround,
                waiting: turnaround - cpu - io,
            }
        })
        .collect();

    let count = stats.len() as u64;
    let total_waiting: u64 = stats.iter().map(|stat| stat.waiting).sum();
    let total_turnaround: u64 = stats.iter().map(|stat| stat.turnaround).sum();

    Ok(Schedule {
        end: processes
            .iter()
            .filter_map(|process| process.completion)
            .max()
            .unwrap_or(0),
        runs,
        idle,
        stats,
        mean_waiting_x100: (total_waiting * 100 + count / 2) / count,
        mean_turnaround_x100: (total_turnaround * 100 + count / 2) / count,
    })
}

use std::collections::VecDeque;

fn admit_events(
    events: &mut Vec<Event>,
    queues: &mut [VecDeque<usize>],
    processes: &mut [Process],
    through: u64,
) {
    events.sort_by_key(|event| (event.time, event.kind, event.index));
    let admitted = events.partition_point(|event| event.time <= through);
    let ready: Vec<Event> = events.drain(..admitted).collect();

    for event in ready {
        processes[event.index].level = event.level;
        queues[event.level].push_back(event.index);
    }
}
```
