use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte_at(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()
}

fn core_missing_byte(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut nums = [0u64; 3];

    for field in 0..3 {
        let start = pos;
        while matches!(byte_at(s, pos), Some(b'0'..=b'9')) {
            pos += 1;
        }
        if pos == start {
            return Err(if core_missing_byte(byte_at(s, pos)) {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        let mut n = 0u64;
        for &b in &bytes[start..pos] {
            n = match n.checked_mul(10).and_then(|x| x.checked_add((b - b'0') as u64)) {
                Some(v) => v,
                None => return Err(VersionError::Overflow { at: start }),
            };
        }
        nums[field] = n;

        if field < 2 {
            if byte_at(s, pos) != Some(b'.') {
                return Err(if core_missing_byte(byte_at(s, pos)) {
                    VersionError::MissingCore { at: pos }
                } else {
                    VersionError::InvalidChar { at: pos }
                });
            }
            pos += 1;
        }
    }

    match byte_at(s, pos) {
        None | Some(b'-' | b'+') => {}
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if byte_at(s, pos) == Some(b'-') {
        pos += 1;
        let start = pos;
        while pos <= s.len() {
            match byte_at(s, pos) {
                None | Some(b'+') => break,
                Some(b'.') => {
                    if pos == start {
                        return Err(VersionError::EmptyIdentifier { at: start });
                    }
                    pre.push(s[start..pos].to_string());
                    pos += 1;
                    let next = pos;
                    while matches!(byte_at(s, pos), Some(b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-')) {
                        pos += 1;
                    }
                    if pos == next && !matches!(byte_at(s, pos), Some(b'+') | None) {
                        return Err(VersionError::InvalidChar { at: pos });
                    }
                    if pos == next {
                        return Err(VersionError::EmptyIdentifier { at: next });
                    }
                    let ident = &s[next..pos];
                    if ident.bytes().all(|b| b.is_ascii_digit())
                        && ident.len() > 1
                        && ident.as_bytes()[0] == b'0'
                    {
                        return Err(VersionError::LeadingZero { at: next });
                    }
                    pre.push(ident.to_string());
                    // Continue with remaining identifiers from this position.
                    let mut p = pos;
                    while byte_at(s, p) == Some(b'.') {
                        p += 1;
                        let st = p;
                        while matches!(byte_at(s, p), Some(b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-')) {
                            p += 1;
                        }
                        if p == st {
                            return Err(VersionError::EmptyIdentifier { at: st });
                        }
                        let id = &s[st..p];
                        if id.bytes().all(|b| b.is_ascii_digit()) && id.len() > 1 && id.as_bytes()[0] == b'0' {
                            return Err(VersionError::LeadingZero { at: st });
                        }
                        pre.push(id.to_string());
                    }
                    pos = p;
                    break;
                }
                Some(b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-') => pos += 1,
                Some(_) => return Err(VersionError::InvalidChar { at: pos }),
            }
        }
        if pos == start {
            return Err(VersionError::EmptyIdentifier { at: start });
        }
        if pre.is_empty() {
            let ident = &s[start..pos];
            if ident.bytes().all(|b| b.is_ascii_digit()) && ident.len() > 1 && ident.as_bytes()[0] == b'0' {
                return Err(VersionError::LeadingZero { at: start });
            }
            pre.push(ident.to_string());
        }
    }

    if byte_at(s, pos) == Some(b'+') {
        pos += 1;
        let mut start = pos;
        if start == s.len() {
            return Err(VersionError::EmptyIdentifier { at: start });
        }
        while pos <= s.len() {
            match byte_at(s, pos) {
                Some(b'.') | None => {
                    if pos == start {
                        return Err(VersionError::EmptyIdentifier { at: start });
                    }
                    build.push(s[start..pos].to_string());
                    if byte_at(s, pos).is_none() {
                        break;
                    }
                    pos += 1;
                    start = pos;
                }
                Some(b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-') => pos += 1,
                Some(_) => return Err(VersionError::InvalidChar { at: pos }),
            }
        }
    }

    Ok(Version {
        major: nums[0],
        minor: nums[1],
        patch: nums[2],
        pre,
        build,
    })
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_numeric_identifier(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let xn = is_numeric(x);
                    let yn = is_numeric(y);
                    let c = match (xn, yn) {
                        (true, true) => cmp_numeric_identifier(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if c != Ordering::Equal {
                        return c;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Check {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Debug)]
struct Arm {
    checks: Vec<Check>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens_with_offsets(s: &str, start: usize, end: usize) -> Vec<(usize, &str)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && is_ws(bytes[i]) {
            i += 1;
        }
        if i == end {
            break;
        }
        let st = i;
        while i < end && !is_ws(bytes[i]) {
            i += 1;
        }
        out.push((st, &s[st..i]));
    }
    out
}

#[derive(Clone, Debug)]
enum Partial {
    Full(Version),
    Fields(Vec<u64>),
}

fn wildcard(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

fn should_parse_full(s: &str) -> bool {
    s.contains('-')
        || s.contains('+')
        || (s.split('.').count() >= 3 && !s.split('.').any(wildcard))
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    if should_parse_full(s) {
        return parse_version(s)
            .map(Partial::Full)
            .map_err(|error| RangeError::Version { at, error });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|x| x.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wild_seen = false;
    for f in fields {
        if wildcard(f) {
            wild_seen = true;
        } else if f.bytes().all(|b| b.is_ascii_digit()) {
            if wild_seen || (f.len() > 1 && f.as_bytes()[0] == b'0') {
                return Err(RangeError::Syntax { at });
            }
            let mut n = 0u64;
            for b in f.bytes() {
                n = n.checked_mul(10).and_then(|x| x.checked_add((b - b'0') as u64))
                    .ok_or(RangeError::Syntax { at })?;
            }
            nums.push(n);
        } else {
            return Err(RangeError::Syntax { at });
        }
    }
    Ok(Partial::Fields(nums))
}

fn padded(fields: &[u64]) -> Version {
    Version {
        major: *fields.get(0).unwrap_or(&0),
        minor: *fields.get(1).unwrap_or(&0),
        patch: *fields.get(2).unwrap_or(&0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn endpoint(mut v: Version, pre: bool) -> Version {
    if pre {
        v.pre = vec!["0".to_string()];
    }
    v.build.clear();
    v
}

fn bump(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut out = v.clone();
    let target = match field {
        0 => &mut out.major,
        1 => &mut out.minor,
        _ => &mut out.patch,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    if field <= 1 {
        out.minor = if field == 0 { 0 } else { out.minor };
    }
    if field <= 2 {
        out.patch = if field <= 1 { 0 } else { out.patch };
    }
    out.pre = vec!["0".to_string()];
    out.build.clear();
    Ok(out)
}

fn add(checks: &mut Vec<Check>, op: Op, version: Version, anchor: bool) {
    checks.push(Check { op, version, anchor });
}

fn lower_and_upper_for_partial(
    partial: Partial,
    mode: &str,
    at: usize,
) -> Result<(Option<Version>, Option<Version>, bool), RangeError> {
    match partial {
        Partial::Full(v) => {
            let anchored = !v.pre.is_empty();
            if mode == "bare" {
                Ok((Some(v.clone()), Some(v), anchored))
            } else {
                Ok((Some(v), None, anchored))
            }
        }
        Partial::Fields(fields) => {
            let k = fields.len();
            if mode != "bare" && k == 0 {
                return Err(RangeError::Syntax { at });
            }
            let lower = (k > 0).then(|| padded(&fields));
            if mode == "bare" && k < 3 {
                if k == 0 {
                    return Ok((None, None, false));
                }
                let upper = bump(&padded(&fields), k - 1, at)?;
                Ok((lower, Some(upper), false))
            } else if mode == "caret" {
                let base = padded(&fields);
                let field = if base.major > 0 || k == 1 {
                    0
                } else if base.minor > 0 || k == 2 {
                    1
                } else {
                    2
                };
                Ok((lower, Some(bump(&base, field, at)?), false))
            } else if mode == "tilde" {
                let base = padded(&fields);
                let field = if k == 1 { 0 } else { 1 };
                Ok((lower, Some(bump(&base, field, at)?), false))
            } else {
                Ok((lower, None, false))
            }
        }
    }
}

fn parse_expanded_token(
    token: &str,
    token_at: usize,
    checks: &mut Vec<Check>,
) -> Result<(), RangeError> {
    let (mode, operand, op_at) = if let Some(rest) = token.strip_prefix('^') {
        ("caret", rest, token_at + 1)
    } else if let Some(rest) = token.strip_prefix('~') {
        ("tilde", rest, token_at + 1)
    } else {
        ("bare", token, token_at)
    };
    if operand.is_empty() {
        return Err(RangeError::Syntax { at: token_at });
    }
    let p = parse_partial(operand, op_at)?;
    let (lower, upper, anchored) = lower_and_upper_for_partial(p, mode, op_at)?;
    if let Some(lo) = lower {
        if mode == "bare" && upper.as_ref().map(|u| compare_versions(&lo, u) == Ordering::Equal).unwrap_or(false) {
            add(checks, Op::Eq, lo, anchored);
        } else {
            add(checks, Op::Ge, lo, anchored);
        }
    }
    if let Some(hi) = upper {
        add(checks, Op::Lt, hi, false);
    }
    Ok(())
}

fn parse_comparator(token: &str, at: usize, checks: &mut Vec<Check>) -> Result<(), RangeError> {
    let (op, n) = if token.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.starts_with("<=") {
        (Op::Le, 2)
    } else if token.starts_with('>') {
        (Op::Gt, 1)
    } else if token.starts_with('<') {
        (Op::Lt, 1)
    } else if token.starts_with('=') {
        (Op::Eq, 1)
    } else {
        return Err(RangeError::Syntax { at });
    };
    let operand = &token[n..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }
    let v = parse_version(operand).map_err(|error| RangeError::Version { at: at + n, error })?;
    let anchor = !v.pre.is_empty();
    add(checks, op, v, anchor);
    Ok(())
}

fn parse_hyphen(tokens: &[(usize, &str)], checks: &mut Vec<Check>) -> Result<(), RangeError> {
    let minus = tokens.iter().find(|(_, t)| *t == "-").unwrap().0;
    if tokens.len() != 3 || tokens[1].1 != "-" {
        return Err(RangeError::Syntax { at: minus });
    }
    let (la, lt) = tokens[0];
    let (ua, ut) = tokens[2];
    let lp = parse_partial(lt, la)?;
    let up = parse_partial(ut, ua)?;

    let (lower, _, lower_anchor) = lower_and_upper_for_partial(lp.clone(), "hyphen", la)?;
    let lower = match lp {
        Partial::Full(v) => Some(v),
        Partial::Fields(fields) if !fields.is_empty() => Some(padded(&fields)),
        _ => return Err(RangeError::Syntax { at: la }),
    };
    let _ = lower;
    if let Some(lo) = lower {
        add(checks, Op::Ge, lo, lower_anchor);
    }

    match up {
        Partial::Full(v) => {
            let anchor = !v.pre.is_empty();
            add(checks, Op::Le, v, anchor);
        }
        Partial::Fields(fields) if !fields.is_empty() => {
            let k = fields.len();
            let lo = padded(&fields);
            if k == 3 {
                add(checks, Op::Le, lo, false);
            } else {
                let upper = bump(&lo, k - 1, ua)?;
                add(checks, Op::Lt, upper, false);
            }
        }
        _ => return Err(RangeError::Syntax { at: ua }),
    }
    let _ = ut;
    Ok(())
}

fn parse_arm(s: &str, start: usize, end: usize) -> Result<Arm, RangeError> {
    let tokens = tokens_with_offsets(s, start, end);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: start });
    }
    let mut checks = Vec::new();
    if tokens.iter().any(|(_, t)| *t == "-") {
        parse_hyphen(&tokens, &mut checks)?;
        return Ok(Arm { checks });
    }
    for (at, token) in tokens {
        let first = token.as_bytes()[0];
        if matches!(first, b'<' | b'>' | b'=') {
            parse_comparator(token, at, &mut checks)?;
        } else if matches!(first, b'^' | b'~') {
            parse_expanded_token(token, at, &mut checks)?;
        } else if first.is_ascii_digit() || wildcard(token) {
            parse_expanded_token(token, at, &mut checks)?;
        } else {
            return Err(RangeError::Syntax { at });
        }
    }
    Ok(Arm { checks })
}

fn arm_matches(v: &Version, arm: &Arm) -> bool {
    let admitted = !v.pre.is_empty()
        && arm.checks.iter().any(|c| {
            c.anchor
                && !c.version.pre.is_empty()
                && (v.major, v.minor, v.patch)
                    == (c.version.major, c.version.minor, c.version.patch)
        });
    if !v.pre.is_empty() && !admitted {
        return false;
    }
    arm.checks.iter().all(|c| {
        let ord = compare_versions(v, &c.version);
        match c.op {
            Op::Eq => ord == Ordering::Equal,
            Op::Lt => ord == Ordering::Less,
            Op::Le => ord != Ordering::Greater,
            Op::Gt => ord == Ordering::Greater,
            Op::Ge => ord != Ordering::Less,
        }
    })
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0usize;
    let mut i = 0usize;
    let mut saw_non_ws = false;

    while i < bytes.len() {
        if bytes[i..].starts_with(b"||") {
            let before_ws = i == 0 || is_ws(bytes[i - 1]);
            let after = i + 2;
            let after_ws = after == bytes.len() || is_ws(bytes[after]);
            if before_ws && after_ws {
                let arm_end = i;
                let has = bytes[arm_start..arm_end].iter().any(|b| !is_ws(*b));
                if !has {
                    return Err(RangeError::Empty { at: i });
                }
                arms.push(parse_arm(s, arm_start, arm_end)?);
                i += 2;
                while i < bytes.len() && is_ws(bytes[i]) {
                    i += 1;
                }
                arm_start = i;
                if i == bytes.len() {
                    return Err(RangeError::Empty { at: s.len() });
                }
                continue;
            }
        }
        if !is_ws(bytes[i]) {
            saw_non_ws = true;
        }
        i += 1;
    }
    if !saw_non_ws {
        return Err(RangeError::Empty { at: 0 });
    }
    let has = bytes[arm_start..].iter().any(|b| !is_ws(*b));
    if !has {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(s, arm_start, s.len())?);
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        match best {
            None => best = Some(i),
            Some(j) => {
                let ord = compare_versions(v, &parsed[j]);
                if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                    best = Some(i);
                }
            }
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}
