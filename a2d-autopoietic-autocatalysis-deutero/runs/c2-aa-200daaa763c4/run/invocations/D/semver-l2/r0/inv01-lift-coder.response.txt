use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_missing(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut nums = [0u64; 3];

    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if pos == start {
            let b = bytes.get(pos).copied();
            return Err(if is_core_missing(b) {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut n = 0u64;
        for &b in &bytes[start..pos] {
            n = n
                .checked_mul(10)
                .and_then(|x| x.checked_add((b - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        nums[field] = n;

        if field < 2 {
            if bytes.get(pos) != Some(&b'.') {
                let b = bytes.get(pos).copied();
                return Err(if is_core_missing(b) {
                    VersionError::MissingCore { at: pos }
                } else {
                    VersionError::InvalidChar { at: pos }
                });
            }
            pos += 1;
        }
    }

    match bytes.get(pos).copied() {
        None | Some(b'-' | b'+') => {}
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }

    fn parse_identifiers(
        s: &str,
        pos: &mut usize,
        stop_at_plus: bool,
        numeric_leading_zero: bool,
    ) -> Result<Vec<String>, VersionError> {
        let bytes = s.as_bytes();
        let mut ids = Vec::new();

        loop {
            let start = *pos;
            while *pos < bytes.len()
                && (bytes[*pos].is_ascii_alphanumeric() || bytes[*pos] == b'-')
                && !(stop_at_plus && bytes[*pos] == b'+')
            {
                *pos += 1;
            }

            if *pos == start {
                return Err(VersionError::EmptyIdentifier { at: start });
            }

            let id = &s[start..*pos];
            if numeric_leading_zero
                && id.len() > 1
                && id.bytes().all(|b| b.is_ascii_digit())
                && id.as_bytes()[0] == b'0'
            {
                return Err(VersionError::LeadingZero { at: start });
            }
            ids.push(id.to_string());

            match bytes.get(*pos).copied() {
                Some(b'.') => *pos += 1,
                Some(b'+') if stop_at_plus => break,
                None => break,
                Some(_) => return Err(VersionError::InvalidChar { at: *pos }),
            }
        }

        Ok(ids)
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if bytes.get(pos) == Some(&b'-') {
        pos += 1;
        pre = parse_identifiers(s, &mut pos, true, true)?;
    }

    if bytes.get(pos) == Some(&b'+') {
        pos += 1;
        build = parse_identifiers(s, &mut pos, false, false)?;
    }

    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: nums[0],
        minor: nums[1],
        patch: nums[2],
        pre,
        build,
    })
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_numeric_identifier(a: &str, b: &str) -> Ordering {
    a.len()
        .cmp(&b.len())
        .then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let xn = is_numeric(x);
                    let yn = is_numeric(y);
                    let c = match (xn, yn) {
                        (true, true) => cmp_numeric_identifier(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if c != Ordering::Equal {
                        return c;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Check {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Debug)]
struct Arm {
    checks: Vec<Check>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens_with_offsets(s: &str, start: usize, end: usize) -> Vec<(usize, &str)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = start;

    while i < end {
        while i < end && is_ws(bytes[i]) {
            i += 1;
        }
        if i == end {
            break;
        }

        let token_start = i;
        while i < end && !is_ws(bytes[i]) {
            i += 1;
        }
        out.push((token_start, &s[token_start..i]));
    }

    out
}

#[derive(Clone, Debug)]
enum Partial {
    Full(Version),
    Fields(Vec<u64>),
}

fn wildcard(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

fn should_parse_full(s: &str) -> bool {
    s.contains('-')
        || s.contains('+')
        || (s.split('.').count() >= 3 && !s.split('.').any(wildcard))
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    if should_parse_full(s) {
        return parse_version(s)
            .map(Partial::Full)
            .map_err(|error| RangeError::Version { at, error });
    }

    let fields: Vec<&str> = s.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|field| field.is_empty()) {
        return Err(RangeError::Syntax { at });
    }

    let mut nums = Vec::new();
    let mut wildcard_seen = false;

    for field in fields {
        if wildcard(field) {
            wildcard_seen = true;
        } else if field.bytes().all(|b| b.is_ascii_digit()) {
            if wildcard_seen || (field.len() > 1 && field.as_bytes()[0] == b'0') {
                return Err(RangeError::Syntax { at });
            }

            let mut n = 0u64;
            for b in field.bytes() {
                n = n
                    .checked_mul(10)
                    .and_then(|x| x.checked_add((b - b'0') as u64))
                    .ok_or(RangeError::Syntax { at })?;
            }
            nums.push(n);
        } else {
            return Err(RangeError::Syntax { at });
        }
    }

    Ok(Partial::Fields(nums))
}

fn padded(fields: &[u64]) -> Version {
    Version {
        major: *fields.first().unwrap_or(&0),
        minor: *fields.get(1).unwrap_or(&0),
        patch: *fields.get(2).unwrap_or(&0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn bump(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut out = v.clone();
    let target = match field {
        0 => &mut out.major,
        1 => &mut out.minor,
        _ => &mut out.patch,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;

    match field {
        0 => {
            out.minor = 0;
            out.patch = 0;
        }
        1 => out.patch = 0,
        _ => {}
    }

    out.pre = vec!["0".to_string()];
    out.build.clear();
    Ok(out)
}

fn add(checks: &mut Vec<Check>, op: Op, version: Version, anchor: bool) {
    checks.push(Check {
        op,
        version,
        anchor,
    });
}

fn lower_and_upper_for_partial(
    partial: Partial,
    mode: &str,
    at: usize,
) -> Result<(Option<Version>, Option<Version>, bool), RangeError> {
    match partial {
        Partial::Full(v) => {
            let anchored = !v.pre.is_empty();
            if mode == "bare" {
                Ok((Some(v.clone()), Some(v), anchored))
            } else {
                Ok((Some(v), None, anchored))
            }
        }
        Partial::Fields(fields) => {
            let k = fields.len();
            if mode != "bare" && k == 0 {
                return Err(RangeError::Syntax { at });
            }

            let lower = (k > 0).then(|| padded(&fields));
            if mode == "bare" && k < 3 {
                if k == 0 {
                    return Ok((None, None, false));
                }
                let upper = bump(&padded(&fields), k - 1, at)?;
                Ok((lower, Some(upper), false))
            } else if mode == "caret" {
                let base = padded(&fields);
                let field = if base.major > 0 || k == 1 {
                    0
                } else if base.minor > 0 || k == 2 {
                    1
                } else {
                    2
                };
                Ok((lower, Some(bump(&base, field, at)?), false))
            } else if mode == "tilde" {
                let base = padded(&fields);
                let field = if k == 1 { 0 } else { 1 };
                Ok((lower, Some(bump(&base, field, at)?), false))
            } else {
                Ok((lower, None, false))
            }
        }
    }
}

fn parse_expanded_token(
    token: &str,
    token_at: usize,
    checks: &mut Vec<Check>,
) -> Result<(), RangeError> {
    let (mode, operand, operand_at) = if let Some(rest) = token.strip_prefix('^') {
        ("caret", rest, token_at + 1)
    } else if let Some(rest) = token.strip_prefix('~') {
        ("tilde", rest, token_at + 1)
    } else {
        ("bare", token, token_at)
    };

    if operand.is_empty() {
        return Err(RangeError::Syntax { at: token_at });
    }

    let partial = parse_partial(operand, operand_at)?;
    let (lower, upper, anchored) = lower_and_upper_for_partial(partial, mode, operand_at)?;

    if let Some(lo) = lower {
        if mode == "bare"
            && upper
                .as_ref()
                .is_some_and(|hi| compare_versions(&lo, hi) == Ordering::Equal)
        {
            add(checks, Op::Eq, lo, anchored);
        } else {
            add(checks, Op::Ge, lo, anchored);
        }
    }

    if let Some(hi) = upper {
        add(checks, Op::Lt, hi, false);
    }

    Ok(())
}

fn parse_comparator(token: &str, at: usize, checks: &mut Vec<Check>) -> Result<(), RangeError> {
    let (op, n) = if token.starts_with(">=") {
        (Op::Ge, 2)
    } else if token.starts_with("<=") {
        (Op::Le, 2)
    } else if token.starts_with('>') {
        (Op::Gt, 1)
    } else if token.starts_with('<') {
        (Op::Lt, 1)
    } else if token.starts_with('=') {
        (Op::Eq, 1)
    } else {
        return Err(RangeError::Syntax { at });
    };

    let operand = &token[n..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at });
    }

    let version = parse_version(operand)
        .map_err(|error| RangeError::Version { at: at + n, error })?;
    let anchor = !version.pre.is_empty();
    add(checks, op, version, anchor);
    Ok(())
}

fn parse_hyphen(tokens: &[(usize, &str)], checks: &mut Vec<Check>) -> Result<(), RangeError> {
    let minus_at = tokens.iter().find(|(_, token)| *token == "-").unwrap().0;
    if tokens.len() != 3 || tokens[1].1 != "-" {
        return Err(RangeError::Syntax { at: minus_at });
    }

    let (lower_at, lower_text) = tokens[0];
    let (upper_at, upper_text) = tokens[2];
    let lower_partial = parse_partial(lower_text, lower_at)?;
    let upper_partial = parse_partial(upper_text, upper_at)?;

    let lower = match lower_partial {
        Partial::Full(v) => {
            let anchor = !v.pre.is_empty();
            add(checks, Op::Ge, v, anchor);
        }
        Partial::Fields(fields) if !fields.is_empty() => {
            add(checks, Op::Ge, padded(&fields), false);
        }
        _ => return Err(RangeError::Syntax { at: lower_at }),
    };

    let _ = lower;

    match upper_partial {
        Partial::Full(v) => {
            let anchor = !v.pre.is_empty();
            add(checks, Op::Le, v, anchor);
        }
        Partial::Fields(fields) if !fields.is_empty() => {
            let k = fields.len();
            let base = padded(&fields);
            if k == 3 {
                add(checks, Op::Le, base, false);
            } else {
                add(checks, Op::Lt, bump(&base, k - 1, upper_at)?, false);
            }
        }
        _ => return Err(RangeError::Syntax { at: upper_at }),
    }

    Ok(())
}

fn parse_arm(s: &str, start: usize, end: usize) -> Result<Arm, RangeError> {
    let tokens = tokens_with_offsets(s, start, end);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: start });
    }

    let mut checks = Vec::new();
    if tokens.iter().any(|(_, token)| *token == "-") {
        parse_hyphen(&tokens, &mut checks)?;
        return Ok(Arm { checks });
    }

    for (at, token) in tokens {
        let first = token.as_bytes()[0];
        if matches!(first, b'<' | b'>' | b'=') {
            parse_comparator(token, at, &mut checks)?;
        } else if matches!(first, b'^' | b'~') {
            parse_expanded_token(token, at, &mut checks)?;
        } else if first.is_ascii_digit() || wildcard(token) {
            parse_expanded_token(token, at, &mut checks)?;
        } else {
            return Err(RangeError::Syntax { at });
        }
    }

    Ok(Arm { checks })
}

fn arm_matches(v: &Version, arm: &Arm) -> bool {
    let admitted = !v.pre.is_empty()
        && arm.checks.iter().any(|check| {
            check.anchor
                && !check.version.pre.is_empty()
                && (v.major, v.minor, v.patch)
                    == (check.version.major, check.version.minor, check.version.patch)
        });

    if !v.pre.is_empty() && !admitted {
        return false;
    }

    arm.checks.iter().all(|check| {
        let ordering = compare_versions(v, &check.version);
        match check.op {
            Op::Eq => ordering == Ordering::Equal,
            Op::Lt => ordering == Ordering::Less,
            Op::Le => ordering != Ordering::Greater,
            Op::Gt => ordering == Ordering::Greater,
            Op::Ge => ordering != Ordering::Less,
        }
    })
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut arm_start = 0usize;
    let mut i = 0usize;
    let mut saw_non_ws = false;

    while i < bytes.len() {
        if bytes[i..].starts_with(b"||") {
            let before_ws = i == 0 || is_ws(bytes[i - 1]);
            let after = i + 2;
            let after_ws = after == bytes.len() || is_ws(bytes[after]);

            if before_ws && after_ws {
                let arm_has_token = bytes[arm_start..i].iter().any(|b| !is_ws(*b));
                if !arm_has_token {
                    return Err(RangeError::Empty { at: i });
                }

                arms.push(parse_arm(s, arm_start, i)?);
                i += 2;
                while i < bytes.len() && is_ws(bytes[i]) {
                    i += 1;
                }
                arm_start = i;

                if i == bytes.len() {
                    return Err(RangeError::Empty { at: s.len() });
                }
                continue;
            }
        }

        if !is_ws(bytes[i]) {
            saw_non_ws = true;
        }
        i += 1;
    }

    if !saw_non_ws {
        return Err(RangeError::Empty { at: 0 });
    }

    let arm_has_token = bytes[arm_start..].iter().any(|b| !is_ws(*b));
    if !arm_has_token {
        return Err(RangeError::Empty { at: s.len() });
    }

    arms.push(parse_arm(s, arm_start, s.len())?);
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(
    candidates: &[String],
    range: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, text) in candidates.iter().enumerate() {
        let version =
            parse_version(text).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, range) {
            continue;
        }

        match best {
            None => best = Some(index),
            Some(best_index) => {
                let ordering = compare_versions(version, &parsed[best_index]);
                if (greatest && ordering == Ordering::Greater)
                    || (!greatest && ordering == Ordering::Less)
                {
                    best = Some(index);
                }
            }
        }
    }

    Ok(best.map(|index| candidates[index].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}
