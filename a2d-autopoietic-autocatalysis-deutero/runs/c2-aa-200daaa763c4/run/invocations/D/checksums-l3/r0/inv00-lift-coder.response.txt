#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    a: u32,
    b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        update_crc_adler(data, &mut self.crc, &mut self.a, &mut self.b);
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let checksums = self.checksum();
        format!("c{:08x}:a{:08x}", checksums.crc32, checksums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut crc = 0u32;
        let mut adler = 0u32;

        for (offset, byte) in bytes.iter().enumerate() {
            if offset == 0 || offset == 9 || offset == 10 {
                continue;
            }
            let digit = match byte {
                b'0'..=b'9' => (byte - b'0') as u32,
                b'a'..=b'f' => (byte - b'a' + 10) as u32,
                _ => return Err(SnapshotError::Hex { offset }),
            };
            if (1..=8).contains(&offset) {
                crc = (crc << 4) | digit;
            } else {
                adler = (adler << 4) | digit;
            }
        }

        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn update_crc_adler(data: &[u8], crc: &mut u32, a: &mut u32, b: &mut u32) {
    for &byte in data {
        *crc ^= byte as u32;
        for _ in 0..8 {
            let low = *crc & 1;
            *crc >>= 1;
            if low != 0 {
                *crc ^= 0xedb8_8320;
            }
        }

        *a = (*a + byte as u32) % 65521;
        *b = (*b + *a) % 65521;
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut stream = Stream::new();
    stream.update(data);
    stream.checksum().crc32
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 65521;
        b = (b + a) % 65521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    let mut stream = Stream::new();
    stream.update(data);
    stream.checksum()
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }

    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }
    Ok(stream.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

fn fletcher32_update(a: &mut u32, b: &mut u32, word: u16) {
    *a = (*a + word as u32) % 65535;
    *b = (*b + *a) % 65535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        let word = u16::from_be_bytes([pair[0], pair[1]]);
        fletcher32_update(&mut a, &mut b, word);
    }
    if let Some(&last) = chunks.remainder().first() {
        fletcher32_update(&mut a, &mut b, (last as u16) << 8);
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut a = 0u32;
    let mut b = 0u32;
    let mut pending_high: Option<u8> = None;

    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending_high.take() {
                let word = u16::from_be_bytes([high, byte]);
                fletcher32_update(&mut a, &mut b, word);
            } else {
                pending_high = Some(byte);
            }
        }
    }

    if let Some(high) = pending_high {
        fletcher32_update(&mut a, &mut b, (high as u16) << 8);
    }

    Ok((b << 16) | a)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut register = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                register ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = register & 0x8000;
                    register = register.wrapping_shl(1);
                    if top != 0 {
                        register ^= 0x1021;
                    }
                }
            }
            register
        }
        Crc16::X25 => {
            for &byte in data {
                register ^= byte as u16;
                for _ in 0..8 {
                    let low = register & 1;
                    register >>= 1;
                    if low != 0 {
                        register ^= 0x8408;
                    }
                }
            }
            register ^ 0xffff
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut stream = Stream::new();
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32_a = 0u32;
    let mut f32_b = 0u32;
    let mut f32_pending: Option<u8> = None;
    let mut crc16_register = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        stream.update(selected);

        for &byte in selected {
            f16_a = (f16_a + byte as u32) % 255;
            f16_b = (f16_b + f16_a) % 255;

            if let Some(high) = f32_pending.take() {
                fletcher32_update(
                    &mut f32_a,
                    &mut f32_b,
                    u16::from_be_bytes([high, byte]),
                );
            } else {
                f32_pending = Some(byte);
            }

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    crc16_register ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        let top = crc16_register & 0x8000;
                        crc16_register = crc16_register.wrapping_shl(1);
                        if top != 0 {
                            crc16_register ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {
                    crc16_register ^= byte as u16;
                    for _ in 0..8 {
                        let low = crc16_register & 1;
                        crc16_register >>= 1;
                        if low != 0 {
                            crc16_register ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    if let Some(high) = f32_pending {
        fletcher32_update(&mut f32_a, &mut f32_b, (high as u16) << 8);
    }

    let crc16 = match variant {
        Crc16::X25 => crc16_register ^ 0xffff,
        Crc16::Xmodem | Crc16::False => crc16_register,
    };

    Ok(Multi {
        base: stream.checksum(),
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: (f32_b << 16) | f32_a,
        crc16,
    })
}
