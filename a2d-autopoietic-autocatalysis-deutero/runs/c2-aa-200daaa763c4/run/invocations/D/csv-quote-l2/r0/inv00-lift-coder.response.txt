#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|column| column == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|column| column == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_table(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    parse_table(input, config)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_records(input, b',', b'"')
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_table(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut records = parse_records(input, config.delimiter, config.quote)?;
    let (header, rows) = if config.headers && !records.is_empty() {
        let header = records.remove(0);
        if config.duplicates == DuplicatePolicy::Reject {
            for second in 0..header.len() {
                if let Some(first) = header[..second]
                    .iter()
                    .position(|name| name == &header[second])
                {
                    return Err(CsvError::DuplicateHeader {
                        name: header[second].clone(),
                        first,
                        second,
                    });
                }
            }
        }
        for (index, row) in records.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
        (Some(header), records)
    } else {
        (None, records)
    };

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

fn parse_records(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = Vec::new();
    let mut field_started = false;
    let mut quoted = false;
    let mut quote_open = 0usize;
    let mut after_quote = false;
    let mut record_index = 0usize;
    let mut record_touched = false;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if !quoted && byte == b'\r' {
            if bytes.get(i + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: i,
                });
            }
            push_field(&mut record, &mut field);
            records.push(record);
            record = Vec::new();
            field = Vec::new();
            field_started = false;
            after_quote = false;
            record_touched = false;
            record_index += 1;
            i += 2;
            continue;
        }

        if quoted {
            if byte == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote);
                    i += 2;
                    continue;
                }
                quoted = false;
                after_quote = true;
                i += 1;
                continue;
            }
            field.push(byte);
            i += 1;
            continue;
        }

        if after_quote {
            if byte == delimiter {
                push_field(&mut record, &mut field);
                field_started = false;
                after_quote = false;
                record_touched = true;
                i += 1;
                continue;
            }
            if byte == b'\n' {
                push_field(&mut record, &mut field);
                records.push(record);
                record = Vec::new();
                field = Vec::new();
                field_started = false;
                after_quote = false;
                record_touched = false;
                record_index += 1;
                i += 1;
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: i,
            });
        }

        if byte == delimiter {
            push_field(&mut record, &mut field);
            field_started = false;
            record_touched = true;
            i += 1;
            continue;
        }
        if byte == b'\n' {
            push_field(&mut record, &mut field);
            records.push(record);
            record = Vec::new();
            field = Vec::new();
            field_started = false;
            after_quote = false;
            record_touched = false;
            record_index += 1;
            i += 1;
            continue;
        }
        if byte == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: i,
                });
            }
            quoted = true;
            quote_open = i;
            field_started = true;
            record_touched = true;
            i += 1;
            continue;
        }

        field.push(byte);
        field_started = true;
        record_touched = true;
        i += 1;
    }

    if quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_open,
        });
    }

    if record_touched || !record.is_empty() || !field.is_empty() || field_started || after_quote {
        push_field(&mut record, &mut field);
        records.push(record);
    }

    let mut decoded = Vec::with_capacity(records.len());
    for row in records {
        let mut decoded_row = Vec::with_capacity(row.len());
        for bytes in row {
            decoded_row.push(String::from_utf8(bytes).expect("input is valid UTF-8"));
        }
        decoded.push(decoded_row);
    }
    Ok(decoded)
}

fn push_field(record: &mut Vec<Vec<u8>>, field: &mut Vec<u8>) {
    record.push(std::mem::take(field));
}
