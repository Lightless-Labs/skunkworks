#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut cron = Self {
            minutes: [false; 60],
            hours: [false; 24],
        };
        Self::parse_field(&bytes[..split], 0, 59, &mut cron.minutes)?;
        Self::parse_field(&bytes[split + 1..], split + 1, 23, &mut cron.hours)?;
        Ok(cron)
    }

    fn parse_field(
        field: &[u8],
        offset: usize,
        max: usize,
        selected: &mut [bool],
    ) -> Result<(), CronError> {
        let mut start = 0;
        loop {
            let end = field[start..]
                .iter()
                .position(|&b| b == b',')
                .map(|i| start + i)
                .unwrap_or(field.len());
            Self::parse_item(&field[start..end], offset + start, max, selected)?;
            if end == field.len() {
                break;
            }
            start = end + 1;
        }
        Ok(())
    }

    fn parse_item(
        item: &[u8],
        offset: usize,
        max: usize,
        selected: &mut [bool],
    ) -> Result<(), CronError> {
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        if slash_positions.len() > 1 {
            return Err(CronError::Syntax(offset + slash_positions[1]));
        }

        let (base_end, step_start) = if let Some(&slash) = slash_positions.first() {
            (slash, Some(slash + 1))
        } else {
            (item.len(), None)
        };
        let base = &item[..base_end];

        let (range_start, range_end, dash_pos) = if base == b"*" {
            (0usize, max, None)
        } else if let Some(dash) = base.iter().position(|&b| b == b'-') {
            let left = Self::parse_number(&base[..dash], offset, max)?;
            let right = Self::parse_number(&base[dash + 1..], offset + dash + 1, max)?;
            if left > right {
                return Err(CronError::Descending(offset + dash));
            }
            (left, right, Some(dash))
        } else {
            if step_start.is_some() {
                return Err(CronError::Syntax(offset + base_end));
            }
            let n = Self::parse_number(base, offset, max)?;
            (n, n, None)
        };

        let step = if let Some(step_pos) = step_start {
            if base != b"*" && dash_pos.is_none() {
                return Err(CronError::Syntax(offset + base_end));
            }
            let step_bytes = &item[step_pos..];
            let value = Self::parse_number(step_bytes, offset + step_pos, usize::MAX)?;
            let step_max = if max == 59 { 60 } else { 24 };
            if value == 0 || value > step_max {
                return Err(CronError::Step(offset + step_pos));
            }
            value
        } else {
            1
        };

        let mut n = range_start;
        while n <= range_end {
            selected[n] = true;
            match n.checked_add(step) {
                Some(next) => n = next,
                None => break,
            }
        }
        Ok(())
    }

    fn parse_number(bytes: &[u8], offset: usize, max: usize) -> Result<usize, CronError> {
        if bytes.is_empty() {
            return Err(CronError::Syntax(offset));
        }
        for (i, &b) in bytes.iter().enumerate() {
            if !b.is_ascii_digit() {
                return Err(CronError::Syntax(offset + i));
            }
        }
        if bytes.len() > 1 && bytes[0] == b'0' {
            return Err(CronError::Number(offset));
        }

        let mut value = 0usize;
        for &b in bytes {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as usize))
                .unwrap_or(usize::MAX);
        }
        if value > max {
            return Err(CronError::Bounds(offset));
        }
        Ok(value)
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
    ids: Vec<String>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
            ids: Vec::new(),
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let owned_id = id.to_owned();
        self.ids.push(owned_id.clone());
        self.jobs.push(Job {
            id: owned_id,
            cron,
            priority,
            policy,
            order: self.jobs.len(),
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower = self.last_poll.map(|t| t / 60_000 + 1).unwrap_or(0);
        let upper = now / 60_000;
        let mut pending: Vec<(Occurrence, u8, usize)> = Vec::new();

        if lower <= upper {
            for job in &self.jobs {
                match job.policy {
                    Policy::All => {
                        for minute in lower..=upper {
                            if job.cron.matches(minute) {
                                pending.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: minute * 60_000,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                        }
                    }
                    Policy::Once => {
                        if let Some(minute) = (lower..=upper)
                            .rev()
                            .find(|&minute| job.cron.matches(minute))
                        {
                            pending.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                    Policy::Skip => {
                        if now % 60_000 == 0 {
                            let minute = now / 60_000;
                            if minute >= lower && minute <= upper && job.cron.matches(minute) {
                                pending.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: now,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                        }
                    }
                }
            }
        }

        pending.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(pending.into_iter().map(|entry| entry.0).collect())
    }
}
