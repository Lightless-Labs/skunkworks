```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }

    let expected = board.width * board.height;
    if board.cells.len() != expected {
        return Err(Error::Shape);
    }

    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    Ok(())
}

fn index_for(board: &Board, direction: Direction, line: usize, offset: usize) -> usize {
    match direction {
        Direction::Left => line * board.width + offset,
        Direction::Right => line * board.width + (board.width - 1 - offset),
        Direction::Up => offset * board.width + line,
        Direction::Down => (board.height - 1 - offset) * board.width + line,
    }
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;

    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    let (line_count, line_len) = match direction {
        Direction::Left | Direction::Right => (board.height, board.width),
        Direction::Up | Direction::Down => (board.width, board.height),
    };

    for line in 0..line_count {
        let tiles: Vec<(usize, u32)> = (0..line_len)
            .filter_map(|offset| {
                let index = index_for(board, direction, line, offset);
                let value = board.cells[index];
                (value != 0).then_some((index, value))
            })
            .collect();

        let mut packed_offset = 0;
        let mut tile_pos = 0;

        while tile_pos < tiles.len() {
            let (source_index, value) = tiles[tile_pos];

            if tile_pos + 1 < tiles.len() && tiles[tile_pos + 1].1 == value {
                let second_index = tiles[tile_pos + 1].0;

                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first: source_index,
                        second: second_index,
                    });
                }

                let merged_value = value * 2;
                let destination = index_for(board, direction, line, packed_offset);
                cells[destination] = merged_value;
                merges.push(Merge {
                    from: [source_index, second_index],
                    to: destination,
                    value: merged_value,
                });
                score += u64::from(merged_value);
                packed_offset += 1;
                tile_pos += 2;
            } else {
                let destination = index_for(board, direction, line, packed_offset);
                cells[destination] = value;
                packed_offset += 1;
                tile_pos += 1;
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;

    let mut legal = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        match move_board(board, direction) {
            Ok(result) if result.changed => legal.push(direction),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(error) => return Err(error),
        }
    }

    Ok(legal)
}
```
