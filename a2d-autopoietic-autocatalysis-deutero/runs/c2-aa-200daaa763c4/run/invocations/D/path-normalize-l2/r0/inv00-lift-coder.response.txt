#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.as_bytes().last() == Some(&b'/')
        || path
            .rsplit('/')
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn reduce_components<'a>(
    components: impl Iterator<Item = (usize, &'a str)>,
    absolute: bool,
) -> Result<Vec<&'a str>, PathError> {
    let mut result = Vec::new();

    for (at, component) in components {
        match component {
            "" | "." => {}
            ".." => {
                if result.last().is_some_and(|part| *part != "..") {
                    result.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    result.push("..");
                }
            }
            other => result.push(other),
        }
    }

    Ok(result)
}

fn join_result(parts: &[&str], absolute: bool, want_directory: bool) -> String {
    let mut result = if absolute {
        format!("/{}", parts.join("/"))
    } else if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };

    if want_directory && result != "/" && result != "." {
        result.push('/');
    }

    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let components = reduce_components(
        path.match_indices('/')
            .flat_map(|(slash_at, _)| {
                let start = slash_at + 1;
                let end = path[start..]
                    .find('/')
                    .map_or(path.len(), |offset| start + offset);
                if start < end {
                    Some((start, &path[start..end]))
                } else {
                    None
                }
            })
            .chain(if !path.starts_with('/') {
                let end = path.find('/').unwrap_or(path.len());
                if end > 0 {
                    Some((0, &path[..end]))
                } else {
                    None
                }
            } else {
                None
            }),
        absolute,
    )?;

    Ok(join_result(&components, absolute, directory_intent(path)))
}

fn validate_context(path: &str) -> Result<Vec<&str>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(normalized
        .split('/')
        .filter(|part| !part.is_empty())
        .collect())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;

    let home_parts = match home {
        Some(value) => {
            if !value.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(
                normalize(value)
                    .map_err(|_| PathError::InvalidHome)?
                    .split('/')
                    .filter(|part| !part.is_empty())
                    .map(str::to_owned)
                    .collect::<Vec<_>>(),
            )
        }
        None => None,
    };

    validate_raw(path)?;

    let tilde = path.starts_with('~');
    let remainder_start = if tilde {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        if home_parts.is_none() {
            return Err(PathError::MissingHome);
        }
        if bytes.get(1) == Some(&b'/') {
            2
        } else {
            1
        }
    } else {
        0
    };

    let absolute = !tilde && path.starts_with('/');
    let mut base: Vec<String> = if absolute {
        Vec::new()
    } else if tilde {
        home_parts.unwrap()
    } else {
        cwd_parts.into_iter().map(str::to_owned).collect()
    };

    let remainder = &path[remainder_start..];
    let mut components = Vec::new();
    let mut offset = remainder_start;

    for component in remainder.split('/') {
        if !component.is_empty() {
            components.push((offset, component));
        }
        offset += component.len() + 1;
    }

    for (at, component) in components {
        match component {
            "." => {}
            ".." => {
                if base.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            other => base.push(other.to_owned()),
        }
    }

    let parts: Vec<&str> = base.iter().map(String::as_str).collect();
    Ok(join_result(&parts, true, directory_intent(path)))
}
