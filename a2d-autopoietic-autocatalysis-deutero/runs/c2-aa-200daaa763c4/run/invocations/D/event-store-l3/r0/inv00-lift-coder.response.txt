#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Status {
    New,
    Open,
    Closed,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Account {
    pub status: Status,
    pub balance: i64,
    pub limit: i64,
    pub version: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { limit: i64 },
    Deposit { amount: i64 },
    Withdraw { amount: i64, floor: i64 },
    Close { expected_version: u64 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    VersionOverflow,
    InvalidLimit,
    AlreadyOpened,
    NotOpened,
    Closed,
    InvalidAmount,
    LimitExceeded,
    InvalidFloor,
    InsufficientFunds,
    FloorViolation,
    CloseVersion { expected: u64, actual: u64 },
    NonZeroBalance,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ReplayError {
    pub index: usize,
    pub kind: ApplyError,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Expected {
    Any,
    Missing,
    Exact(u64),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StoreError {
    InvalidId { byte: usize },
    Conflict {
        expected: Expected,
        actual: Option<u64>,
    },
    Rejected(ReplayError),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Snapshot {
    pub account: Account,
    pub tag: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SnapshotError {
    ZeroInterval,
    InvalidVersion,
    InvalidState,
    TagMismatch,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RestoreError {
    Snapshot(SnapshotError),
    Tail(ReplayError),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Store {
    snapshot_every: u64,
    streams: std::collections::BTreeMap<String, Stream>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Stream {
    events: Vec<Event>,
    account: Account,
    snapshot: Option<Snapshot>,
}

fn initial_account() -> Account {
    Account {
        status: Status::New,
        balance: 0,
        limit: 0,
        version: 0,
    }
}

fn apply_event(account: &mut Account, event: &Event) -> Result<(), ApplyError> {
    if account.version == u64::MAX {
        return Err(ApplyError::VersionOverflow);
    }

    match event {
        Event::Open { limit } => {
            if !(1..=1_000_000_000).contains(limit) {
                return Err(ApplyError::InvalidLimit);
            }
            match account.status {
                Status::New => {
                    account.status = Status::Open;
                    account.limit = *limit;
                    account.balance = 0;
                }
                Status::Open | Status::Closed => {
                    return Err(ApplyError::AlreadyOpened);
                }
            }
        }
        Event::Deposit { amount } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *amount > account.limit - account.balance {
                return Err(ApplyError::LimitExceeded);
            }
            account.balance += *amount;
        }
        Event::Withdraw { amount, floor } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *floor < 0 || *floor > account.limit {
                return Err(ApplyError::InvalidFloor);
            }
            if *amount > account.balance {
                return Err(ApplyError::InsufficientFunds);
            }
            if account.balance - *amount < *floor {
                return Err(ApplyError::FloorViolation);
            }
            account.balance -= *amount;
        }
        Event::Close { expected_version } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion {
                    expected: *expected_version,
                    actual: account.version,
                });
            }
            if account.balance != 0 {
                return Err(ApplyError::NonZeroBalance);
            }
            account.status = Status::Closed;
        }
    }

    account.version += 1;
    Ok(())
}

fn apply_events(account: &Account, events: &[Event]) -> Result<Account, ReplayError> {
    let mut result = account.clone();
    for (index, event) in events.iter().enumerate() {
        apply_event(&mut result, event).map_err(|kind| ReplayError { index, kind })?;
    }
    Ok(result)
}

pub fn replay(events: &[Event]) -> Result<Account, ReplayError> {
    apply_events(&initial_account(), events)
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum IdValidation {
    Valid,
}

fn validate_id(id: &str) -> Result<IdValidation, usize> {
    let bytes = id.as_bytes();
    if bytes.is_empty() {
        return Err(0);
    }

    for (index, byte) in bytes.iter().enumerate() {
        if index == 24 {
            return Err(24);
        }
        let valid = if index == 0 {
            byte.is_ascii_lowercase()
        } else {
            byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-'
        };
        if !valid {
            return Err(index);
        }
    }

    if bytes.len() > 24 {
        return Err(24);
    }
    Ok(IdValidation::Valid)
}

fn actual_version(stream: Option<&Stream>) -> Option<u64> {
    stream.map(|s| s.account.version)
}

fn expected_matches(expected: &Expected, actual: Option<u64>) -> bool {
    match expected {
        Expected::Any => true,
        Expected::Missing => actual.is_none(),
        Expected::Exact(version) => actual == Some(*version),
    }
}

impl Store {
    pub fn new(snapshot_every: u64) -> Result<Self, SnapshotError> {
        if snapshot_every == 0 {
            return Err(SnapshotError::ZeroInterval);
        }
        Ok(Self {
            snapshot_every,
            streams: std::collections::BTreeMap::new(),
        })
    }

    pub fn append(
        &mut self,
        id: &str,
        expected: Expected,
        events: &[Event],
    ) -> Result<Account, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;

        let actual = actual_version(self.streams.get(id));
        if !expected_matches(&expected, actual) {
            return Err(StoreError::Conflict { expected, actual });
        }

        let existing_account = self
            .streams
            .get(id)
            .map(|stream| stream.account.clone())
            .unwrap_or_else(initial_account);

        if events.is_empty() {
            return Ok(existing_account);
        }

        let mut account = existing_account.clone();
        let mut latest_snapshot = self
            .streams
            .get(id)
            .and_then(|stream| stream.snapshot.clone());

        let mut new_events = Vec::with_capacity(events.len());
        for (index, event) in events.iter().enumerate() {
            apply_event(&mut account, event)
                .map_err(|kind| StoreError::Rejected(ReplayError { index, kind }))?;
            new_events.push(event.clone());

            if account.version > 0 && account.version % self.snapshot_every == 0 {
                latest_snapshot = Some(Snapshot {
                    tag: snapshot_tag(&account),
                    account: account.clone(),
                });
            }
        }

        match self.streams.get_mut(id) {
            Some(stream) => {
                stream.events.extend(new_events);
                stream.account = account.clone();
                stream.snapshot = latest_snapshot;
            }
            None => {
                self.streams.insert(
                    id.to_owned(),
                    Stream {
                        events: new_events,
                        account: account.clone(),
                        snapshot: latest_snapshot,
                    },
                );
            }
        }

        Ok(account)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.streams.get(id).map(|stream| stream.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        let Some(stream) = self.streams.get(id) else {
            return Ok(None);
        };

        let start = usize::try_from(after).unwrap_or(usize::MAX);
        let events = if start >= stream.events.len() {
            Vec::new()
        } else {
            stream.events[start..].to_vec()
        };
        Ok(Some(events))
    }

    pub fn snapshot(&self, id: &str) -> Result<Option<Snapshot>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self
            .streams
            .get(id)
            .and_then(|stream| stream.snapshot.clone()))
    }

    pub fn tail(&self, id: &str) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        let Some(stream) = self.streams.get(id) else {
            return Ok(None);
        };

        let start = stream
            .snapshot
            .as_ref()
            .map(|snapshot| usize::try_from(snapshot.account.version).unwrap_or(usize::MAX))
            .unwrap_or(0);

        Ok(Some(if start >= stream.events.len() {
            Vec::new()
        } else {
            stream.events[start..].to_vec()
        }))
    }
}

pub fn snapshot_tag(account: &Account) -> u64 {
    let status_code = match account.status {
        Status::New => 0,
        Status::Open => 0xA5,
        Status::Closed => 0x5A,
    };
    account.version.rotate_left(7)
        ^ (account.limit as u64).wrapping_mul(31)
        ^ (account.balance as u64).rotate_left(17)
        ^ status_code
}

fn validate_snapshot(snapshot: &Snapshot) -> Result<(), SnapshotError> {
    let account = &snapshot.account;

    if account.version == 0 {
        return Err(SnapshotError::InvalidVersion);
    }

    if matches!(account.status, Status::New)
        || !(1..=1_000_000_000).contains(&account.limit)
        || account.balance < 0
        || account.balance > account.limit
        || (matches!(account.status, Status::Closed)
            && (account.balance != 0 || account.version < 2))
    {
        return Err(SnapshotError::InvalidState);
    }

    if snapshot.tag != snapshot_tag(account) {
        return Err(SnapshotError::TagMismatch);
    }

    Ok(())
}

pub fn rebuild(
    snapshot: Option<&Snapshot>,
    tail: &[Event],
) -> Result<Account, RestoreError> {
    match snapshot {
        None => replay(tail).map_err(RestoreError::Tail),
        Some(snapshot) => {
            validate_snapshot(snapshot).map_err(RestoreError::Snapshot)?;
            apply_events(&snapshot.account, tail).map_err(RestoreError::Tail)
        }
    }
}
