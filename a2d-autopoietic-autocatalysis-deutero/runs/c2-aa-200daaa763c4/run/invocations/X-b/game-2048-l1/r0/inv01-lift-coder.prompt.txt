#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && !(2..=2_147_483_648).contains(&value).then(|| value.is_power_of_two()).unwrap_or(false) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let mut scan: Vec<(usize, u32)> = match side {
        Side::Left => row
            .iter()
            .copied()
            .enumerate()
            .filter(|&(_, value)| value != 0)
            .collect(),
        Side::Right => row
            .iter()
            .copied()
            .enumerate()
            .rev()
            .filter(|&(_, value)| value != 0)
            .collect(),
    };

    let mut output_scan = Vec::with_capacity(scan.len());
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut i = 0;

    while i < scan.len() {
        let (index, value) = scan[i];
        if i + 1 < scan.len() && scan[i + 1].1 == value {
            let (second_index, _) = scan[i + 1];
            let Some(merged_value) = value.checked_mul(2) else {
                return Err(Error::MergeOverflow {
                    first: index,
                    second: second_index,
                });
            };

            let to = match side {
                Side::Left => output_scan.len(),
                Side::Right => row.len() - 1 - output_scan.len(),
            };
            merges.push(Merge {
                from: [index, second_index],
                to,
                value: merged_value,
            });
            score += u64::from(merged_value);
            output_scan.push(merged_value);
            i += 2;
        } else {
            output_scan.push(value);
            i += 1;
        }
    }

    let mut cells = vec![0; row.len()];
    match side {
        Side::Left => {
            cells[..output_scan.len()].copy_from_slice(&output_scan);
        }
        Side::Right => {
            for (offset, value) in output_scan.into_iter().enumerate() {
                cells[row.len() - 1 - offset] = value;
            }
        }
    }

    // Merge destinations are assigned during the scan; update them to final
    // compacted positions in encounter order.
    for (merge, output_position) in merges.iter_mut().zip(
        match side {
            Side::Left => (0..).take(merges.len()).collect::<Vec<_>>(),
            Side::Right => (row.len().saturating_sub(1)..).rev().take(merges.len()).collect::<Vec<_>>(),
        },
    ) {
        merge.to = output_position;
    }

    // `scan` is no longer needed; compare against the original input.
    let changed = cells.as_slice() != row;

    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}
