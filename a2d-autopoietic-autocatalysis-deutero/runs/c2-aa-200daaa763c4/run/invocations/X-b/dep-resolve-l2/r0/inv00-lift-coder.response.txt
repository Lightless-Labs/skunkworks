```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    (1..=24).contains(&bytes.len())
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn validate_names(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn canonicalize(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonicalize(constraints.to_vec())
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(value: &str) -> String {
    let mut encoded = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            encoded.push(*byte as char);
        } else {
            encoded.push_str(&format!("%{byte:02X}"));
        }
    }
    encoded
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode_text(site), encode_text(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode_text(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode_text(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn inspect_node(
    catalog: &[Package],
    roots: &[Requirement],
    assigned: &[(String, Version)],
) -> Result<
    (
        Vec<(String, Vec<Constraint>)>,
        Vec<(String, Vec<Constraint>)>,
    ),
    ResolveError,
> {
    let mut active: Vec<(String, Vec<Constraint>)> = Vec::new();

    fn add_requirement(active: &mut Vec<(String, Vec<Constraint>)>, req: &Requirement) {
        if let Some((_, constraints)) = active.iter_mut().find(|(name, _)| name == &req.package) {
            constraints.push(req.constraint);
        } else {
            active.push((req.package.clone(), vec![req.constraint]));
        }
    }

    for root in roots {
        add_requirement(&mut active, root);
    }
    for (name, version) in assigned {
        if let Some(package) = catalog.iter().find(|package| package.name == *name) {
            if let Some(release) = package.releases.iter().find(|release| release.version == *version) {
                for dep in &release.deps {
                    add_requirement(&mut active, dep);
                }
            }
        }
    }
    active.sort_by(|a, b| a.0.cmp(&b.0));
    for (_, constraints) in &mut active {
        *constraints = canonicalize(std::mem::take(constraints));
    }

    let missing = active
        .iter()
        .filter(|(name, _)| !catalog.iter().any(|package| package.name == *name))
        .map(|(name, _)| name)
        .min();
    if let Some(name) = missing {
        return Err(ResolveError::Missing {
            package: name.clone(),
        });
    }

    for (name, constraints) in &active {
        if let Some((_, version)) = assigned.iter().find(|(assigned_name, _)| assigned_name == name) {
            if !constraints.iter().all(|constraint| matches(*version, *constraint)) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    for (name, constraints) in &active {
        if assigned.iter().any(|(assigned_name, _)| assigned_name == name) {
            continue;
        }
        let package = catalog.iter().find(|package| package.name == *name).unwrap();
        if !package
            .releases
            .iter()
            .any(|release| constraints.iter().all(|constraint| matches(release.version, *constraint)))
        {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
    }

    Ok((active, Vec::new()))
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    assigned: &mut Vec<(String, Version)>,
    attempts: &mut usize,
    max_attempts: usize,
    first_error: &mut Option<ResolveError>,
) -> Result<Vec<Selection>, ResolveError> {
    let (active, _) = match inspect_node(catalog, roots, assigned) {
        Ok(value) => value,
        Err(error) => {
            if first_error.is_none() {
                *first_error = Some(error.clone());
            }
            return Err(error);
        }
    };

    let next = active
        .iter()
        .find(|(name, _)| !assigned.iter().any(|(assigned_name, _)| assigned_name == name));
    let Some((name, constraints)) = next else {
        let mut selections: Vec<Selection> = assigned
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect();
        selections.sort_by(|a, b| a.package.cmp(&b.package));
        return Ok(selections);
    };

    let package = catalog.iter().find(|package| package.name == *name).unwrap();
    let mut candidates: Vec<Version> = package
        .releases
        .iter()
        .filter(|release| constraints.iter().all(|constraint| matches(release.version, *constraint)))
        .map(|release| release.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    for version in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        assigned.push((name.clone(), version));
        match search(catalog, roots, assigned, attempts, max_attempts, first_error) {
            Ok(solution) => return Ok(solution),
            Err(ResolveError::Limit { attempts }) => {
                assigned.pop();
                return Err(ResolveError::Limit { attempts });
            }
            Err(error) => {
                assigned.pop();
                if first_error.is_none() {
                    *first_error = Some(error);
                }
            }
        }
    }

    Err(first_error.clone().unwrap_or_else(|| ResolveError::NoMatch {
        package: name.clone(),
        requirements: constraints.clone(),
    }))
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut attempts = 0;
    let mut first_error = None;
    match search(
        catalog,
        roots,
        &mut Vec::new(),
        &mut attempts,
        max_attempts,
        &mut first_error,
    ) {
        Ok(solution) => Ok(solution),
        Err(ResolveError::Limit { attempts }) => Err(ResolveError::Limit { attempts }),
        Err(_) => Err(first_error.unwrap()),
    }
}
```
