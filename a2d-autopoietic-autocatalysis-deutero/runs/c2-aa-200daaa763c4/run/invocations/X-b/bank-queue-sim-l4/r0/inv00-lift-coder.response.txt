#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let complement = denominator - r;
    if r > complement || (r == complement && q % 2 == 1) {
        Some(q + 1)
    } else {
        Some(q)
    }
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    total_wait: u64,
    started: bool,
    pause_time: Option<u64>,
}

#[derive(Clone)]
struct Active {
    job: Job,
    dispatch_time: u64,
    work_at: u64,
    finish: u64,
}

fn waiter_better(a: &Job, b: &Job) -> bool {
    if a.customer.priority != b.customer.priority {
        return a.customer.priority < b.customer.priority;
    }
    match (a.pause_time, b.pause_time) {
        (Some(pa), Some(pb)) => pa < pb || (pa == pb && a.customer.id < b.customer.id),
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            a.customer.arrival < b.customer.arrival
                || (a.customer.arrival == b.customer.arrival && a.customer.id > b.customer.id)
        }
    }
}

fn push_event(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn expire_waiters(
    now: u64,
    queue: &mut Vec<Job>,
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired = Vec::new();
    let mut kept = Vec::new();
    for job in queue.drain(..) {
        if job.deadline.is_some_and(|d| d <= now) {
            expired.push(job);
        } else {
            kept.push(job);
        }
    }
    expired.sort_by_key(|j| (j.deadline.unwrap(), j.customer.id));
    for job in expired {
        let waited = job.total_wait;
        stats.wait_ticks += waited as u128;
        stats.reneged += 1;
        push_event(
            events,
            now,
            job.customer.id,
            Kind::Renege { waited },
        );
    }
    *queue = kept;
}

fn best_waiter_index(queue: &[Job]) -> Option<usize> {
    let mut best = None;
    for i in 0..queue.len() {
        if best.is_none_or(|b| waiter_better(&queue[i], &queue[b])) {
            best = Some(i);
        }
    }
    best
}

fn finish_active(
    now: u64,
    active: &mut Option<Active>,
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    if active.as_ref().is_some_and(|a| a.finish == now) {
        let a = active.take().unwrap();
        stats.busy_ticks += (now - a.dispatch_time) as u128;
        stats.completed += 1;
        stats.turnaround_ticks += (now - a.job.customer.arrival) as u128;
        stats.completion_order.push(a.job.customer.id);
        push_event(events, now, a.job.customer.id, Kind::Finish);
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut seen = Vec::<u32>::new();
    let mut deadlines = Vec::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen.contains(&customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        seen.push(customer.id);
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
        let deadline = match customer.patience {
            Some(p) => Some(
                customer
                    .arrival
                    .checked_add(p)
                    .ok_or(Error::DeadlineOverflow {
                        index,
                        id: customer.id,
                    })?,
            ),
            None => None,
        };
        deadlines.push(deadline);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };
    let mut events = Vec::new();
    let mut queue: Vec<Job> = Vec::new();
    let mut active: Option<Active> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut have_time = false;

    while next_arrival < order.len() || active.is_some() || !queue.is_empty() {
        let mut candidate: Option<u64> = None;
        if let Some(&idx) = order.get(next_arrival) {
            candidate = Some(customers[idx].arrival);
        }
        if let Some(a) = &active {
            for t in [Some(a.finish), (a.work_at > now).then_some(a.work_at)]
                .into_iter()
                .flatten()
            {
                candidate = Some(candidate.map_or(t, |c| c.min(t)));
            }
        }
        for job in &queue {
            if let Some(d) = job.deadline {
                candidate = Some(candidate.map_or(d, |c| c.min(d)));
            }
        }
        let t = candidate.expect("pending simulation work has a next time");
        if have_time && t < now {
            unreachable!("event times are monotone");
        }
        now = t;
        have_time = true;

        // Phase A
        finish_active(now, &mut active, &mut events, &mut stats);

        // Phase B
        expire_waiters(now, &mut queue, &mut events, &mut stats);

        // Phase C
        while next_arrival < order.len() {
            let idx = order[next_arrival];
            let customer = &customers[idx];
            if customer.arrival != now {
                break;
            }
            next_arrival += 1;
            push_event(&mut events, now, customer.id, Kind::Arrive);
            let limit = config.waiting_capacity + usize::from(active.is_none());
            if queue.len() >= limit {
                stats.balked += 1;
                push_event(&mut events, now, customer.id, Kind::Balk);
                continue;
            }
            stats.accepted += 1;
            queue.push(Job {
                customer: customer.clone(),
                deadline: deadlines[idx],
                remaining: customer.service,
                total_wait: 0,
                started: false,
                pause_time: None,
            });
        }

        // Phase D
        let protected = active.as_ref().is_some_and(|a| now < a.work_at);
        if !protected {
            if let (Some(a), Some(wi)) = (&active, best_waiter_index(&queue)) {
                if queue[wi].customer.priority < a.job.customer.priority {
                    let by = queue[wi].customer.id;
                    let a = active.take().unwrap();
                    let remaining = a.finish - now;
                    stats.busy_ticks += (now - a.dispatch_time) as u128;
                    stats.preemptions += 1;
                    push_event(
                        &mut events,
                        now,
                        a.job.customer.id,
                        Kind::Preempt { remaining, by },
                    );
                    let mut job = a.job;
                    job.remaining = remaining;
                    job.pause_time = Some(now);
                    job.deadline = match job.customer.patience {
                        Some(budget) => Some(now.checked_add(budget).ok_or(Error::TimeOverflow {
                            customer: job.customer.id,
                            time: now,
                            stage: OverflowStage::Patience,
                        })?),
                        None => None,
                    };
                    queue.push(job);
                }
            }
        }

        // Phase E: dispatch until either occupied or no waiter.
        loop {
            if active.is_none() {
                let Some(wi) = best_waiter_index(&queue) else {
                    break;
                };
                let mut job = queue.remove(wi);
                let segment = now - match job.pause_time {
                    Some(_) => {
                        // The queue entry time is the latest preemption time.
                        job.pause_time.unwrap()
                    }
                    None => job.customer.arrival + job.total_wait,
                };
                job.total_wait += segment;
                stats.wait_ticks += segment as u128;

                let switching = last_dispatched
                    .is_some_and(|id| id != job.customer.id);
                let cost = if switching { config.switch_cost } else { 0 };
                let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                    customer: job.customer.id,
                    time: now,
                    stage: OverflowStage::WorkAt,
                })?;
                let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                    customer: job.customer.id,
                    time: now,
                    stage: OverflowStage::Finish,
                })?;

                let first = !job.started;
                if first {
                    job.started = true;
                    stats.started += 1;
                }
                push_event(
                    &mut events,
                    now,
                    job.customer.id,
                    if first {
                        Kind::Start {
                            remaining: job.remaining,
                            work_at,
                            finish,
                        }
                    } else {
                        Kind::Resume {
                            remaining: job.remaining,
                            work_at,
                            finish,
                        }
                    },
                );
                last_dispatched = Some(job.customer.id);
                active = Some(Active {
                    job,
                    dispatch_time: now,
                    work_at,
                    finish,
                });

                if finish == now {
                    finish_active(now, &mut active, &mut events, &mut stats);
                    continue;
                }
            }
            break;
        }

        // Phase F
        expire_waiters(now, &mut queue, &mut events, &mut stats);
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(
            stats.turnaround_ticks * 1000,
            stats.completed as u128,
        )
        .unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}
