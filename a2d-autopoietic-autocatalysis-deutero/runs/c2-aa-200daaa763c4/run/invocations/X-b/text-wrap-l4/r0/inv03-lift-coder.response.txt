#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    text: String,
    consumed: usize,
    pieces: usize,
    marker_ended: bool,
}

#[derive(Clone)]
struct Layout {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
}

fn better(a: &Layout, b: &Layout) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    for (x, y) in a.lines.iter().zip(&b.lines) {
        if x.consumed != y.consumed {
            return x.consumed > y.consumed;
        }
    }
    for (x, y) in a.lines.iter().zip(&b.lines) {
        if x.text != y.text {
            return x.text < y.text;
        }
    }
    false
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if ch == '~' {
            markers.push(chars.len());
        } else {
            chars.push(ch);
        }
    }

    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }
    words
}

fn line_cost(gap: usize, final_line: bool) -> u64 {
    let square = (gap * gap) as u64;
    if final_line { square / 4 } else { square }
}

fn optimize(words: &[Word], options: &WrapOptions) -> Layout {
    fn solve(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut [Option<Layout>],
    ) -> Layout {
        if wi == words.len() {
            return Layout {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
            };
        }

        let memo_index = (wi * 513 + off) * 2 + usize::from(first);
        if let Some(result) = &memo[memo_index] {
            return result.clone();
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let cap = options.width - indent;
        let mut best: Option<Layout> = None;

        let mut body = String::new();
        let mut used = 0usize;
        let mut pieces = 0usize;
        let mut cursor_w = wi;
        let mut cursor_off = off;
        let mut consumed = 0usize;

        loop {
            let word = &words[cursor_w];
            let remaining = word.chars.len() - cursor_off;
            let separator = usize::from(pieces > 0);

            if used + separator + remaining <= cap {
                if separator > 0 {
                    body.push(' ');
                    used += 1;
                }
                body.extend(word.chars[cursor_off..].iter());
                used += remaining;
                consumed += remaining;
                pieces += 1;

                let next = solve(cursor_w + 1, 0, false, words, options, memo);
                let final_line = next.lines.is_empty();
                let gap = options.width - (indent + used);
                let cost = next.cost
                    + line_cost(gap, final_line)
                    + if !final_line && pieces == 1 { 3 } else { 0 };

                let mut candidate = Layout {
                    cost,
                    marker_lines: next.marker_lines,
                    lines: vec![Line {
                        text: format!("{}{}", " ".repeat(indent), body),
                        consumed,
                        pieces,
                        marker_ended: false,
                    }],
                };
                candidate.lines.extend(next.lines);
                if best.as_ref().is_none_or(|current| better(&candidate, current)) {
                    best = Some(candidate);
                }

                body.truncate(body.len() - remaining);
                used -= remaining;
                consumed -= remaining;
                pieces -= 1;
                if separator > 0 {
                    body.pop();
                    used -= 1;
                }
            }

            for &marker in &word.markers {
                if marker <= cursor_off || marker >= word.chars.len() {
                    continue;
                }
                let prefix = marker - cursor_off;
                if used + separator + prefix + 1 > cap {
                    continue;
                }

                let old_len = body.len();
                if separator > 0 {
                    body.push(' ');
                }
                body.extend(word.chars[cursor_off..marker].iter());
                body.push('-');

                let line_used = used + separator + prefix + 1;
                let line_consumed = consumed + prefix;
                let next = solve(cursor_w, marker, false, words, options, memo);
                let final_line = next.lines.is_empty();
                let gap = options.width - (indent + line_used);
                let line_pieces = pieces + 1;
                let cost = next.cost
                    + line_cost(gap, final_line)
                    + options.hyphen_cost as u64
                    + if !final_line && line_pieces == 1 { 3 } else { 0 };

                let mut candidate = Layout {
                    cost,
                    marker_lines: next.marker_lines + 1,
                    lines: vec![Line {
                        text: format!("{}{}", " ".repeat(indent), body),
                        consumed: line_consumed,
                        pieces: line_pieces,
                        marker_ended: true,
                    }],
                };
                candidate.lines.extend(next.lines);
                if best.as_ref().is_none_or(|current| better(&candidate, current)) {
                    best = Some(candidate);
                }
                body.truncate(old_len);
            }

            if pieces == 0 && remaining > cap {
                let marker_fits = word.markers.iter().any(|&marker| {
                    marker > cursor_off
                        && marker < word.chars.len()
                        && marker - cursor_off + 1 <= cap
                });
                if !marker_fits {
                    let text: String = word.chars[cursor_off..cursor_off + cap]
                        .iter()
                        .collect();
                    let next = solve(cursor_w, cursor_off + cap, false, words, options, memo);
                    let gap = options.width - (indent + cap);
                    let final_line = next.lines.is_empty();
                    let cost = next.cost
                        + line_cost(gap, final_line)
                        + (2 * options.width + 1) as u64
                        + if !final_line { 3 } else { 0 };

                    let mut candidate = Layout {
                        cost,
                        marker_lines: next.marker_lines,
                        lines: vec![Line {
                            text: format!("{}{}", " ".repeat(indent), text),
                            consumed: cap,
                            pieces: 1,
                            marker_ended: false,
                        }],
                    };
                    candidate.lines.extend(next.lines);
                    if best.as_ref().is_none_or(|current| better(&candidate, current)) {
                        best = Some(candidate);
                    }
                }
            }

            if cursor_w + 1 >= words.len() {
                break;
            }

            let next_word = &words[cursor_w + 1];
            if used + 1 + next_word.chars.len() > cap {
                break;
            }

            body.push(' ');
            used += 1;
            cursor_w += 1;
            cursor_off = 0;
        }

        let result = best.unwrap_or(Layout {
            cost: u64::MAX / 4,
            marker_lines: usize::MAX / 4,
            lines: Vec::new(),
        });
        memo[memo_index] = Some(result.clone());
        result
    }

    let mut memo = vec![None; (words.len() + 1) * 513 * 2];
    solve(0, 0, true, words, options, &mut memo)
}

fn justify(layout: &mut Layout, options: &WrapOptions) {
    if !options.justify {
        return;
    }

    for i in 0..layout.lines.len().saturating_sub(1) {
        if layout.lines[i].pieces < 2 {
            continue;
        }

        let indent = layout.lines[i]
            .text
            .chars()
            .take_while(|&ch| ch == ' ')
            .count();
        let body: String = layout.lines[i].text.chars().skip(indent).collect();
        let gaps = body.matches(' ').count();
        if gaps == 0 {
            continue;
        }

        let extra = options.width - layout.lines[i].text.chars().count();
        let base = extra / gaps;
        let remainder = extra % gaps;
        let start = i % gaps;
        let mut gap_index = 0usize;
        let mut rendered = String::new();

        for ch in body.chars() {
            rendered.push(ch);
            if ch == ' ' {
                rendered.push_str(&" ".repeat(base));
                let distance = (gap_index + gaps - start) % gaps;
                if distance < remainder {
                    rendered.push(' ');
                }
                gap_index += 1;
            }
        }

        layout.lines[i].text = format!("{}{}", " ".repeat(indent), rendered);
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let char_count = input.chars().count();
    if char_count > 512 {
        return Err(WrapError::InputTooLong { chars: char_count });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let raw: Vec<char> = input.chars().collect();
    let byte_indexes: Vec<usize> = input.char_indices().map(|(index, _)| index).collect();
    for i in 0..raw.len() {
        if raw[i] == '~' {
            let left_ok = i > 0 && raw[i - 1] != '~' && !is_separator(raw[i - 1]);
            let right_ok = i + 1 < raw.len() && raw[i + 1] != '~' && !is_separator(raw[i + 1]);
            if !left_ok || !right_ok {
                return Err(WrapError::InvalidMarker {
                    index: byte_indexes[i],
                });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut iter = raw.iter().copied().peekable();
    while let Some(ch) = iter.next() {
        if ch == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut i = 0usize;

    while i < normalized.len() {
        if normalized[i] != '\n' {
            i += 1;
            continue;
        }

        let run_start = i;
        while i < normalized.len() && normalized[i] == '\n' {
            i += 1;
        }

        if i - run_start >= 2 {
            let words = parse_words(&normalized[start..run_start]);
            if !words.is_empty() {
                let mut layout = optimize(&words, options);
                justify(&mut layout, options);
                output.extend(layout.lines.into_iter().map(|line| line.text));
            }
            output.extend(std::iter::repeat_n(String::new(), i - run_start - 1));
            start = i;
        }
    }

    let words = parse_words(&normalized[start..]);
    if !words.is_empty() {
        let mut layout = optimize(&words, options);
        justify(&mut layout, options);
        output.extend(layout.lines.into_iter().map(|line| line.text));
    }

    Ok(output)
}
