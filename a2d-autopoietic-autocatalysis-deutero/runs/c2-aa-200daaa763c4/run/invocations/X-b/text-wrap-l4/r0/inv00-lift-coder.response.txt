#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Candidate {
    cost: u64,
    marker_lines: usize,
    lines: Vec<String>,
    consumed: Vec<usize>,
}

#[derive(Clone)]
struct Piece {
    text: String,
    scalars: usize,
    marker_ended: bool,
}

fn better(a: &Candidate, b: &Candidate) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    for (x, y) in a.consumed.iter().zip(&b.consumed) {
        if x != y {
            return x > y;
        }
    }
    for (x, y) in a.lines.iter().zip(&b.lines) {
        if x != y {
            return x < y;
        }
    }
    false
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if ch == '~' {
            markers.push(chars.len());
        } else {
            chars.push(ch);
        }
    }

    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }
    words
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn optimize(
    words: &[Word],
    first: bool,
    options: &WrapOptions,
    memo: &mut [Option<Candidate>],
) -> Candidate {
    let n = words.len();
    let slots = n + 1;

    fn solve(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut [Option<Candidate>],
        slots: usize,
    ) -> Candidate {
        if wi == words.len() {
            return Candidate {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed: Vec::new(),
            };
        }

        let key = wi * slots + off;
        if !first {
            if let Some(c) = &memo[key] {
                return c.clone();
            }
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let cap = options.width - indent;
        let mut pieces = Vec::<Piece>::new();
        let mut used = 0usize;
        let mut cursor_w = wi;
        let mut cursor_off = off;
        let mut best: Option<Candidate> = None;

        loop {
            let word = &words[cursor_w];
            let remaining = word.chars.len() - cursor_off;
            let gap = if pieces.is_empty() { 0 } else { 1 };

            if used + gap + remaining <= cap {
                let mut text: String = word.chars[cursor_off..].iter().collect();
                if !pieces.is_empty() {
                    text.insert(0, ' ');
                }
                let scalar_count = gap + remaining;
                pieces.push(Piece {
                    text,
                    scalars: scalar_count,
                    marker_ended: false,
                });
                used += scalar_count;

                let line_text = pieces.iter().map(|p| p.text.as_str()).collect::<String>();
                let consumed = pieces.iter().map(|p| p.scalars).sum::<usize>();
                let next = solve(cursor_w + 1, 0, false, words, options, memo, slots);
                let displayed = indent + used;
                let gap_width = options.width - displayed;
                let mut cand = Candidate {
                    cost: next.cost
                        + (gap_width as u64 * gap_width as u64) / 4
                        + if pieces.len() == 1 && !next.lines.is_empty() {
                            3
                        } else {
                            0
                        },
                    marker_lines: next.marker_lines,
                    lines: vec![format!("{}{}", " ".repeat(indent), line_text)],
                    consumed: vec![consumed],
                };
                cand.lines.extend(next.lines);
                cand.consumed.extend(next.consumed);
                if best.as_ref().is_none_or(|b| better(&cand, b)) {
                    best = Some(cand);
                }
                pieces.pop();
                used -= scalar_count;
            }

            for &m in &word.markers {
                if m <= cursor_off || m >= word.chars.len() {
                    continue;
                }
                let prefix_len = m - cursor_off;
                if used + gap + prefix_len + 1 > cap {
                    continue;
                }
                let mut text = String::new();
                if !pieces.is_empty() {
                    text.push(' ');
                }
                text.extend(word.chars[cursor_off..m].iter());
                text.push('-');
                let scalar_count = gap + prefix_len;
                pieces.push(Piece {
                    text,
                    scalars: scalar_count,
                    marker_ended: true,
                });
                let mut line_text = pieces.iter().map(|p| p.text.as_str()).collect::<String>();
                let consumed = pieces.iter().map(|p| p.scalars).sum::<usize>();
                let next = solve(cursor_w, m, false, words, options, memo, slots);
                let displayed = indent + used + gap + prefix_len + 1;
                let gap_width = options.width - displayed;
                let mut cand = Candidate {
                    cost: next.cost
                        + gap_width as u64 * gap_width as u64
                        + options.hyphen_cost as u64
                        + if pieces.len() == 1 && !next.lines.is_empty() {
                            3
                        } else {
                            0
                        },
                    marker_lines: next.marker_lines + 1,
                    lines: vec![format!("{}{}", " ".repeat(indent), line_text)],
                    consumed: vec![consumed],
                };
                cand.lines.extend(next.lines);
                cand.consumed.extend(next.consumed);
                if best.as_ref().is_none_or(|b| better(&cand, b)) {
                    best = Some(cand);
                }
                pieces.pop();
                line_text.clear();
            }

            if pieces.is_empty() && remaining > cap {
                let marker_fits = word.markers.iter().any(|&m| {
                    m > cursor_off
                        && m < word.chars.len()
                        && (m - cursor_off) + 1 <= cap
                });
                if !marker_fits {
                    let take = cap;
                    let text: String = word.chars[cursor_off..cursor_off + take]
                        .iter()
                        .collect();
                    let next = solve(cursor_w, cursor_off + take, false, words, options, memo, slots);
                    let displayed = indent + take;
                    let gap_width = options.width - displayed;
                    let mut cand = Candidate {
                        cost: next.cost
                            + gap_width as u64 * gap_width as u64
                            + (2 * options.width + 1) as u64
                            + if next.lines.is_empty() { 0 } else { 3 },
                        marker_lines: next.marker_lines,
                        lines: vec![format!("{}{}", " ".repeat(indent), text)],
                        consumed: vec![take],
                    };
                    cand.lines.extend(next.lines);
                    cand.consumed.extend(next.consumed);
                    if best.as_ref().is_none_or(|b| better(&cand, b)) {
                        best = Some(cand);
                    }
                }
            }

            if cursor_w + 1 >= words.len() {
                break;
            }
            let next_word = &words[cursor_w + 1];
            let needed = 1 + next_word.chars.len();
            if used + needed > cap {
                break;
            }

            used += 1;
            pieces.push(Piece {
                text: " ".to_string(),
                scalars: 1,
                marker_ended: false,
            });
            cursor_w += 1;
            cursor_off = 0;
            let _ = next_word;
        }

        let result = best.unwrap_or_else(|| Candidate {
            cost: u64::MAX / 4,
            marker_lines: usize::MAX / 4,
            lines: Vec::new(),
            consumed: Vec::new(),
        });
        if !first {
            memo[key] = Some(result.clone());
        }
        result
    }

    let _ = first;
    solve(0, 0, true, words, options, memo, slots)
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars = input.chars().count();
    if chars > 512 {
        return Err(WrapError::InputTooLong { chars });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let raw: Vec<char> = input.chars().collect();
    for i in 0..raw.len() {
        if raw[i] == '~' {
            let valid_left = i > 0 && raw[i - 1] != '~' && !is_separator(raw[i - 1]);
            let valid_right =
                i + 1 < raw.len() && raw[i + 1] != '~' && !is_separator(raw[i + 1]);
            if !valid_left || !valid_right {
                let index = input
                    .char_indices()
                    .nth(i)
                    .map(|(b, _)| b)
                    .unwrap_or(input.len());
                return Err(WrapError::InvalidMarker { index });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut iter = raw.iter().copied().peekable();
    while let Some(ch) = iter.next() {
        if ch == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    if normalized.is_empty() {
        return Ok(Vec::new());
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut i = 0usize;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let run_start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            let run_len = i - run_start;
            if run_len >= 2 {
                let words = parse_words(&normalized[start..run_start]);
                if !words.is_empty() {
                    let memo_len = (words.len() + 1) * (words.len() + 1);
                    let mut memo = vec![None; memo_len];
                    let mut candidate = optimize(&words, true, options, &mut memo);
                    if options.justify {
                        justify_lines(&mut candidate.lines, &words, options);
                    }
                    output.extend(candidate.lines);
                }
                output.extend(std::iter::repeat(String::new()).take(run_len - 1));
                start = i;
            }
        } else {
            i += 1;
        }
    }

    let words = parse_words(&normalized[start..]);
    if !words.is_empty() {
        let memo_len = (words.len() + 1) * (words.len() + 1);
        let mut memo = vec![None; memo_len];
        let mut candidate = optimize(&words, true, options, &mut memo);
        if options.justify {
            justify_lines(&mut candidate.lines, &words, options);
        }
        output.extend(candidate.lines);
    }

    output
}

fn justify_lines(lines: &mut [String], _words: &[Word], options: &WrapOptions) {
    if lines.len() < 2 {
        return;
    }
    // Justification is reconstructed from the visible line text. Hyphens remain
    // attached to their word piece.
    for line_index in 0..lines.len() - 1 {
        let line = &lines[line_index];
        let indent = line.chars().take_while(|&c| c == ' ').count();
        let body: String = line.chars().skip(indent).collect();
        let gaps = body.matches(' ').count();
        if gaps == 0 {
            continue;
        }
        let used = line.chars().count();
        let extra = options.width.saturating_sub(used);
        let base = extra / gaps;
        let rem = extra % gaps;
        let mut gap_index = 0usize;
        let mut result = String::new();
        for ch in body.chars() {
            if ch == ' ' {
                result.push(' ');
                result.push_str(&" ".repeat(base));
                if (gap_index + gaps - (line_index % gaps)) % gaps < rem {
                    result.push(' ');
                }
                gap_index += 1;
            } else {
                result.push(ch);
            }
        }
        lines[line_index] = format!("{}{}", " ".repeat(indent), result);
    }
}
