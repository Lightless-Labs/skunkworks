pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars = input.chars().count();
    if chars > 512 {
        return Err(WrapError::InputTooLong { chars });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let raw: Vec<char> = input.chars().collect();
    for i in 0..raw.len() {
        if raw[i] == '~' {
            let valid_left = i > 0 && raw[i - 1] != '~' && !is_separator(raw[i - 1]);
            let valid_right =
                i + 1 < raw.len() && raw[i + 1] != '~' && !is_separator(raw[i + 1]);
            if !valid_left || !valid_right {
                let index = input
                    .char_indices()
                    .nth(i)
                    .map(|(b, _)| b)
                    .unwrap_or(input.len());
                return Err(WrapError::InvalidMarker { index });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut iter = raw.iter().copied().peekable();
    while let Some(ch) = iter.next() {
        if ch == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    if normalized.is_empty() {
        return Ok(Vec::new());
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut i = 0usize;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let run_start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            let run_len = i - run_start;
            if run_len >= 2 {
                let words = parse_words(&normalized[start..run_start]);
                if !words.is_empty() {
                    let memo_len = (words.len() + 1) * (words.len() + 1);
                    let mut memo = vec![None; memo_len];
                    let mut candidate = optimize(&words, true, options, &mut memo);
                    if options.justify {
                        justify_lines(&mut candidate.lines, &words, options);
                    }
                    output.extend(candidate.lines);
                }
                output.extend(std::iter::repeat(String::new()).take(run_len - 1));
                start = i;
            }
        } else {
            i += 1;
        }
    }

    let words = parse_words(&normalized[start..]);
    if !words.is_empty() {
        let memo_len = (words.len() + 1) * (words.len() + 1);
        let mut memo = vec![None; memo_len];
        let mut candidate = optimize(&words, true, options, &mut memo);
        if options.justify {
            justify_lines(&mut candidate.lines, &words, options);
        }
        output.extend(candidate.lines);
    }

    output
}

fn justify_lines(lines: &mut [String], _words: &[Word], options: &WrapOptions) {
    if lines.len() < 2 {
        return;
    }
    // Justification is reconstructed from the visible line text. Hyphens remain
    // attached to their word piece.
    for line_index in 0..lines.len() - 1 {
        let line = &lines[line_index];
        let indent = line.chars().take_while(|&c| c == ' ').count();
        let body: String = line.chars().skip(indent).collect();
        let gaps = body.matches(' ').count();
        if gaps == 0 {
            continue;
        }
        let used = line.chars().count();
        let extra = options.width.saturating_sub(used);
        let base = extra / gaps;
        let rem = extra % gaps;
        let mut gap_index = 0usize;
        let mut result = String::new();
        for ch in body.chars() {
            if ch == ' ' {
                result.push(' ');
                result.push_str(&" ".repeat(base));
                if (gap_index + gaps - (line_index % gaps)) % gaps < rem {
                    result.push(' ');
                }
                gap_index += 1;
            } else {
                result.push(ch);
            }
        }
        lines[line_index] = format!("{}{}", " ".repeat(indent), result);
    }
}

    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    for (x, y) in a.consumed.iter().zip(&b.consumed) {
        if x != y {
            return x > y;
        }
    }
    for (x, y) in a.lines.iter().zip(&b.lines) {
        if x != y {
            return x < y;
        }
    }
    false
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if ch == '~' {
            markers.push(chars.len());
        } else {
            chars.push(ch);
        }
    }

    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }
    words
}

fn optimize(words: &[Word], options: &WrapOptions) -> Candidate {
    fn solve(
        wi: usize,
        off: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut Vec<Option<Candidate>>,
    ) -> Candidate {
        if wi == words.len() {
            return Candidate {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
                consumed: Vec::new(),
                pieces: Vec::new(),
            };
        }

        let key = (wi, off, first);
        let memo_index = (wi * 513 + off) * 2 + usize::from(first);
        if let Some(c) = &memo[memo_index] {
            return c.clone();
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let cap = options.width - indent;
        let mut best: Option<Candidate> = None;
        let mut body = String::new();
        let mut used = 0usize;
        let mut piece_count = 0usize;
        let mut cursor_w = wi;
        let mut cursor_off = off;
        let mut consumed = 0usize;

        loop {
            let word = &words[cursor_w];
            let remaining = word.chars.len() - cursor_off;
            let separator = usize::from(piece_count > 0);

            if used + separator + remaining <= cap {
                if separator > 0 {
                    body.push(' ');
                    used += 1;
                }
                body.extend(word.chars[cursor_off..].iter());
                used += remaining;
                consumed += separator + remaining;
                piece_count += 1;

                let next = solve(cursor_w + 1, 0, false, words, options, memo);
                let gap = options.width - (indent + used);
                let is_final = next.lines.is_empty();
                let mut cand = Candidate {
                    cost: next.cost
                        + if is_final {
                            (gap * gap / 4) as u64
                        } else {
                            (gap * gap) as u64
                        }
                        + if !is_final && piece_count == 1 { 3 } else { 0 },
                    marker_lines: next.marker_lines,
                    lines: vec![format!("{}{}", " ".repeat(indent), body)],
                    consumed: vec![consumed],
                    pieces: vec![piece_count],
                };
                cand.lines.extend(next.lines);
                cand.consumed.extend(next.consumed);
                cand.pieces.extend(next.pieces);
                if best.as_ref().is_none_or(|b| better(&cand, b)) {
                    best = Some(cand);
                }

                body.truncate(body.len() - remaining);
                used -= remaining;
                consumed -= separator + remaining;
                piece_count -= 1;
                if separator > 0 {
                    body.pop();
                    used -= 1;
                }
            }

            for &marker in &word.markers {
                if marker <= cursor_off || marker >= word.chars.len() {
                    continue;
                }
                let prefix = marker - cursor_off;
                if used + separator + prefix + 1 > cap {
                    continue;
                }

                let old_len = body.len();
                if separator > 0 {
                    body.push(' ');
                }
                body.extend(word.chars[cursor_off..marker].iter());
                body.push('-');
                let line_used = used + separator + prefix + 1;
                let line_consumed = consumed + separator + prefix;
                let next = solve(cursor_w, marker, false, words, options, memo);
                let gap = options.width - (indent + line_used);
                let mut cand = Candidate {
                    cost: next.cost
                        + (gap * gap) as u64
                        + options.hyphen_cost as u64
                        + if piece_count + 1 == 1 && !next.lines.is_empty() {
                            3
                        } else {
                            0
                        },
                    marker_lines: next.marker_lines + 1,
                    lines: vec![format!("{}{}", " ".repeat(indent), body)],
                    consumed: vec![line_consumed],
                    pieces: vec![piece_count + 1],
                };
                cand.lines.extend(next.lines);
                cand.consumed.extend(next.consumed);
                cand.pieces.extend(next.pieces);
                if best.as_ref().is_none_or(|b| better(&cand, b)) {
                    best = Some(cand);
                }
                body.truncate(old_len);
            }

            if piece_count == 0 && remaining > cap {
                let marker_fits = word.markers.iter().any(|&marker| {
                    marker > cursor_off
                        && marker < word.chars.len()
                        && marker - cursor_off + 1 <= cap
                });
                if !marker_fits {
                    let text: String = word.chars[cursor_off..cursor_off + cap].iter().collect();
                    let next = solve(cursor_w, cursor_off + cap, false, words, options, memo);
                    let gap = options.width - (indent + cap);
                    let mut cand = Candidate {
                        cost: next.cost
                            + (gap * gap) as u64
                            + (2 * options.width + 1) as u64
                            + if !next.lines.is_empty() { 3 } else { 0 },
                        marker_lines: next.marker_lines,
                        lines: vec![format!("{}{}", " ".repeat(indent), text)],
                        consumed: vec![cap],
                        pieces: vec![1],
                    };
                    cand.lines.extend(next.lines);
                    cand.consumed.extend(next.consumed);
                    cand.pieces.extend(next.pieces);
                    if best.as_ref().is_none_or(|b| better(&cand, b)) {
                        best = Some(cand);
                    }
                }
            }

            if cursor_w + 1 >= words.len() {
                break;
            }
            let next_word = &words[cursor_w + 1];
            let needed = 1 + next_word.chars.len();
            if used + needed > cap {
                break;
            }
            body.push(' ');
            used += 1;
            consumed += 1;
            piece_count += 1;
            cursor_w += 1;
            cursor_off = 0;
        }

        let result = best.unwrap_or(Candidate {
            cost: u64::MAX / 4,
            marker_lines: usize::MAX / 4,
            lines: Vec::new(),
            consumed: Vec::new(),
            pieces: Vec::new(),
        });
        memo[memo_index] = Some(result.clone());
        result
    }

    let mut memo = vec![None; (words.len() + 1) * 513 * 2];
    solve(0, 0, true, words, options, &mut memo)
}

fn justify(candidate: &mut Candidate, options: &WrapOptions) {
    if !options.justify {
        return;
    }
    for i in 0..candidate.lines.len().saturating_sub(1) {
        if candidate.pieces[i] < 2 {
            continue;
        }

        let indent = candidate.lines[i]
            .chars()
            .take_while(|&ch| ch == ' ')
            .count();
        let body: String = candidate.lines[i].chars().skip(indent).collect();
        let gaps = body.matches(' ').count();
        if gaps == 0 {
            continue;
        }

        let extra = options.width - candidate.lines[i].chars().count();
        let base = extra / gaps;
        let remainder = extra % gaps;
        let start = i % gaps;
        let mut gap_index = 0usize;
        let mut rendered = String::new();

        for ch in body.chars() {
            rendered.push(ch);
            if ch == ' ' {
                rendered.push_str(&" ".repeat(base));
                let distance = (gap_index + gaps - start) % gaps;
                if distance < remainder {
                    rendered.push(' ');
                }
                gap_index += 1;
            }
        }
        candidate.lines[i] = format!("{}{}", " ".repeat(indent), rendered);
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let char_count = input.chars().count();
    if char_count > 512 {
        return Err(WrapError::InputTooLong { chars: char_count });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let raw: Vec<char> = input.chars().collect();
    let byte_indexes: Vec<usize> = input.char_indices().map(|(index, _)| index).collect();
    for i in 0..raw.len() {
        if raw[i] == '~' {
            let left_ok = i > 0 && raw[i - 1] != '~' && !is_separator(raw[i - 1]);
            let right_ok = i + 1 < raw.len() && raw[i + 1] != '~' && !is_separator(raw[i + 1]);
            if !left_ok || !right_ok {
                return Err(WrapError::InvalidMarker {
                    index: byte_indexes[i],
                });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut iter = raw.iter().copied().peekable();
    while let Some(ch) = iter.next() {
        if ch == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut i = 0usize;
    while i < normalized.len() {
        if normalized[i] != '\n' {
            i += 1;
            continue;
        }

        let run_start = i;
        while i < normalized.len() && normalized[i] == '\n' {
            i += 1;
        }

        if i - run_start >= 2 {
            let words = parse_words(&normalized[start..run_start]);
            if !words.is_empty() {
                let mut candidate = optimize(&words, options);
                justify(&mut candidate, options);
                output.extend(candidate.lines);
            }
            output.extend(std::iter::repeat(String::new()).take(i - run_start - 1));
            start = i;
        }
    }

    let words = parse_words(&normalized[start..]);
    if !words.is_empty() {
        let mut candidate = optimize(&words, options);
        justify(&mut candidate, options);
        output.extend(candidate.lines);
    }

    Ok(output)
}
