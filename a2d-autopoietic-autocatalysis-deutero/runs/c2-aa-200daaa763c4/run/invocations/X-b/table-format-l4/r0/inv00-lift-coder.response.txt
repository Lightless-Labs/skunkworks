#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn validate_cell(s: &str, row: usize, column: usize) -> Result<(), TableError> {
    let mut iter = s.char_indices().peekable();
    while let Some((byte, ch)) = iter.next() {
        if ch == '\r' {
            if matches!(iter.peek(), Some((_, '\n'))) {
                iter.next();
                continue;
            }
            return Err(TableError::InvalidChar { row, column, byte });
        }
        if ch == '\n' {
            continue;
        }
        if ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}' {
            return Err(TableError::InvalidChar { row, column, byte });
        }
    }
    Ok(())
}

fn logical_lines(s: &str) -> Vec<String> {
    let normalized = s.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| line.trim_end_matches(' ').to_owned())
        .collect()
}

fn tokens(line: &str, markdown: bool) -> Vec<Token> {
    let mut out = Vec::new();
    for ch in line.chars() {
        if markdown && ch == '|' {
            out.push(Token { text: "\\|".to_owned(), width: 2 });
        } else if markdown && ch == '\\' {
            out.push(Token { text: "\\\\".to_owned(), width: 2 });
        } else {
            out.push(Token { text: ch.to_string(), width: 1 });
        }
    }
    out
}

fn token_width(ts: &[Token]) -> usize {
    ts.iter().map(|t| t.width).sum()
}

fn fragments(ts: &[Token], width: usize, rule: &Overflow) -> Vec<String> {
    if ts.is_empty() {
        return vec![String::new()];
    }
    match rule {
        Overflow::Clip => {
            let total = token_width(ts);
            if total <= width {
                return vec![ts.iter().map(|t| t.text.as_str()).collect()];
            }
            let mut result = String::new();
            let mut used = 0;
            let limit = width.saturating_sub(1);
            for token in ts {
                if used + token.width > limit {
                    break;
                }
                result.push_str(&token.text);
                used += token.width;
            }
            result.push('…');
            vec![result]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < ts.len() {
                let mut end = start;
                let mut used = 0;
                while end < ts.len() && used + ts[end].width <= width {
                    used += ts[end].width;
                    end += 1;
                }
                // All tokens fit under the mandated minimum widths.
                if end == start {
                    end += 1;
                }
                result.push(ts[start..end].iter().map(|t| t.text.as_str()).collect());
                start = end;
            }
            result
        }
    }
}

fn aligned(content: &str, width: usize, alignment: &Alignment) -> String {
    let content_width = content.chars().count();
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    format!("{}{}{}", " ".repeat(left), content, " ".repeat(right))
}

fn border(widths: &[usize]) -> String {
    let mut out = String::from("+");
    for &width in widths {
        out.push_str(&"-".repeat(width + 2));
        out.push('+');
    }
    out.push('\n');
    out
}

fn data_line(contents: &[String], widths: &[usize], aligns: &[Alignment], mode: &Mode) -> String {
    if matches!(mode, Mode::Compact) {
        let mut out = String::new();
        for i in 0..contents.len() {
            if i > 0 {
                out.push_str("  ");
            }
            let mut cell = aligned(&contents[i], widths[i], &aligns[i]);
            if i + 1 == contents.len() {
                let padding = widths[i].saturating_sub(contents[i].chars().count());
                if matches!(aligns[i], Alignment::Left) {
                    cell.truncate(cell.len().saturating_sub(padding));
                } else if matches!(aligns[i], Alignment::Center) {
                    let right_padding = padding - padding / 2;
                    cell.truncate(cell.len().saturating_sub(right_padding));
                }
            }
            out.push_str(&cell);
        }
        out.push('\n');
        return out;
    }

    let mut out = String::from("|");
    for i in 0..contents.len() {
        out.push(' ');
        out.push_str(&aligned(&contents[i], widths[i], &aligns[i]));
        out.push_str(" |");
    }
    out.push('\n');
    out
}

fn markdown_marker(widths: &[usize], aligns: &[Alignment]) -> String {
    let mut out = String::from("|");
    for (i, &width) in widths.iter().enumerate() {
        let marker = match aligns[i] {
            Alignment::Left => format!(":{}", "-".repeat(width - 1)),
            Alignment::Right => format!("{}:", "-".repeat(width - 1)),
            Alignment::Center => format!(":{}:", "-".repeat(width - 2)),
        };
        out.push(' ');
        out.push_str(&marker);
        out.push_str(" |");
    }
    out.push('\n');
    out
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged { row, expected: n, actual: cells.len() });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount { expected: n, actual: options.alignments.len() });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount { expected: n, actual: options.overflow.len() });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let overhead = match options.mode {
        Mode::Compact => 2 * n.saturating_sub(1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let min_width = if matches!(options.mode, Mode::Markdown) { 3 } else { 1 };
    let minimum = overhead + n * min_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow { minimum, provided: options.max_width });
    }

    for (r, cells) in rows.iter().enumerate() {
        for (c, cell) in cells.iter().enumerate() {
            validate_cell(cell, r, c)?;
        }
    }

    let aligns: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let markdown = matches!(options.mode, Mode::Markdown);
    let all_lines: Vec<Vec<Vec<Token>>> = rows
        .iter()
        .map(|row| row.iter().map(|cell| logical_lines(cell).iter().map(|line| tokens(line, markdown)).collect()).collect())
        .collect();

    let mut widths = vec![min_width; n];
    for row in &all_lines {
        for (col, lines) in row.iter().enumerate() {
            for line in lines {
                widths[col] = widths[col].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let candidate = (0..n)
                .filter(|&i| widths[i] > min_width)
                .max_by_key(|&i| (widths[i], i));
            if let Some(i) = candidate {
                widths[i] -= 1;
            } else {
                break;
            }
        }
    }

    let mut prepared: Vec<Vec<Vec<Vec<String>>>> = Vec::with_capacity(rows.len());
    for row in &all_lines {
        let mut prepared_row = Vec::with_capacity(n);
        for (col, lines) in row.iter().enumerate() {
            let mut cell_fragments = Vec::new();
            for line in lines {
                cell_fragments.push(fragments(line, widths[col], &overflows[col]));
            }
            prepared_row.push(cell_fragments);
        }
        prepared.push(prepared_row);
    }

    let mut out = String::new();
    if matches!(options.mode, Mode::Grid) {
        out.push_str(&border(&widths));
    }

    for (r, row) in prepared.iter().enumerate() {
        let height = row
            .iter()
            .map(|cell| cell.iter().map(Vec::len).sum::<usize>())
            .max()
            .unwrap_or(1);

        for physical in 0..height {
            let mut contents = Vec::with_capacity(n);
            for cell in row {
                let mut flat = Vec::new();
                for line_frags in cell {
                    flat.extend(line_frags.iter().cloned());
                }
                contents.push(flat.get(physical).cloned().unwrap_or_default());
            }
            out.push_str(&data_line(&contents, &widths, &aligns, &options.mode));
        }

        if matches!(options.mode, Mode::Markdown) && r == 0 {
            out.push_str(&markdown_marker(&widths, &aligns));
        }
        if matches!(options.mode, Mode::Grid) && options.header && r == 0 && rows.len() > 1 {
            out.push_str(&border(&widths));
        }
    }

    if matches!(options.mode, Mode::Grid) {
        out.push_str(&border(&widths));
    }
    Ok(out)
}
