```rust
use std::collections::HashSet;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = HashSet::new();
        for product in &products {
            if product.id.is_empty()
                || product.id.len() > 8
                || !product.id.bytes().all(|b| b.is_ascii_lowercase())
            {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if !seen.insert(product.id.clone()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: Self::denomination(index),
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let Some(index) = Self::denomination_index(coin) else {
            return Err(Error::BadCoin { coin });
        };
        let _ = index;

        if self.credit as u32 + coin as u32 > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(product) = self.products.iter().find(|product| product.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(product.id.clone());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(selected_id) = self.selected.as_ref() else {
            return Err(Error::NoSelection);
        };
        let product = self
            .products
            .iter()
            .find(|product| product.id == *selected_id)
            .expect("selected products always belong to the catalog");
        let id = product.id.clone();
        let price = product.price;

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = Self::choose_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for (index, count) in change.iter().enumerate() {
            proposed_tubes[index] -= count;
        }
        for coin in &self.escrow {
            let index = Self::denomination_index(*coin)
                .expect("escrow contains only accepted denominations");
            proposed_tubes[index] += 1;
        }
        for (index, count) in proposed_tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: Self::denomination(index),
                });
            }
        }

        let mut change_coins = Vec::new();
        for index in (0..4).rev() {
            let coin = Self::denomination(index);
            for _ in 0..change[index] {
                change_coins.push(coin);
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt {
            id,
            price,
            change: change_coins,
        })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }

    fn denomination_index(coin: u16) -> Option<usize> {
        match coin {
            1 => Some(0),
            3 => Some(1),
            4 => Some(2),
            10 => Some(3),
            _ => None,
        }
    }

    fn denomination(index: usize) -> u16 {
        [1, 3, 4, 10][index]
    }

    fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
        let denominations = [1u16, 3, 4, 10];

        let mut greedy = [0u16; 4];
        let mut remaining = amount;
        for index in (0..4).rev() {
            let take = tubes[index].min(remaining / denominations[index]);
            greedy[index] = take;
            remaining -= take * denominations[index];
        }
        if remaining == 0 {
            return Some(greedy);
        }

        let mut best: Option<[u16; 4]> = None;
        let mut best_count = u32::MAX;

        fn search(
            index: isize,
            remaining: u16,
            tubes: &[u16; 4],
            current: &mut [u16; 4],
            best: &mut Option<[u16; 4]>,
            best_count: &mut u32,
        ) {
            let denominations = [1u16, 3, 4, 10];
            if index < 0 {
                if remaining == 0 {
                    let count: u32 = current.iter().map(|&n| n as u32).sum();
                    let replace = match best {
                        None => true,
                        Some(previous) => {
                            count < *best_count
                                || (count == *best_count
                                    && (0..4).rev().find_map(|i| {
                                        if current[i] != previous[i] {
                                            Some(current[i] > previous[i])
                                        } else {
                                            None
                                        }
                                    }) == Some(true))
                        }
                    };
                    if replace {
                        *best = Some(*current);
                        *best_count = count;
                    }
                }
                return;
            }

            let i = index as usize;
            let coin = denominations[i];
            let max_take = tubes[i].min(remaining / coin);
            for take in 0..=max_take {
                let new_remaining = remaining - take * coin;
                let next_count = current
                    .iter()
                    .map(|&n| n as u32)
                    .sum::<u32>()
                    + take as u32;
                if next_count > *best_count {
                    continue;
                }
                current[i] = take;
                search(
                    index - 1,
                    new_remaining,
                    tubes,
                    current,
                    best,
                    best_count,
                );
            }
            current[i] = 0;
        }

        search(
            3,
            amount,
            tubes,
            &mut [0; 4],
            &mut best,
            &mut best_count,
        );
        best
    }
}
```
