use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn is_ascii_letter(b: u8) -> bool {
    b.is_ascii_alphabetic()
}

fn is_scheme_rest(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')
}

fn is_host_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')
}

fn global_byte_error(input: &str) -> Option<usize> {
    input
        .as_bytes()
        .iter()
        .position(|&b| !(0x21..=0x7e).contains(&b))
}

fn codec_to_parse(err: CodecError, offset: usize) -> ParseError {
    match err {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    if let Some(at) = global_byte_error(input) {
        return Err(ParseError::InvalidByte { at });
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !is_ascii_letter(bytes[0]) {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !is_scheme_rest(b) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    let scheme = input[..colon].to_ascii_lowercase();

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|p| authority_start + p)
        .unwrap_or(input.len());
    let authority = &input[authority_start..authority_end];
    let authority_bytes = authority.as_bytes();

    let ats: Vec<usize> = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + ats[1],
        });
    }
    if ats.first() == Some(&0) {
        return Err(ParseError::InvalidUserInfo { at: authority_start });
    }

    let (userinfo, host_port_start) = if let Some(&at) = ats.first() {
        (Some(authority[..at].to_string()), at + 1)
    } else {
        (None, 0)
    };

    let host_port = &authority[host_port_start..];
    let host_port_bytes = host_port.as_bytes();
    let host_start = authority_start + host_port_start;

    let (host, port, ipv6) = if host_port_bytes.first() == Some(&b'[') {
        let close = host_port_bytes[1..]
            .iter()
            .position(|&b| b == b']')
            .map(|p| p + 1);
        let close = match close {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &host_port[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }
        let suffix_start = close + 1;
        let port = if suffix_start == host_port.len() {
            None
        } else if host_port_bytes[suffix_start] == b':' {
            Some(parse_port(
                &host_port[suffix_start + 1..],
                host_start + suffix_start + 1,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + suffix_start,
            });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let host_port_colon = host_port_bytes.iter().position(|&b| b == b':');
        let host_end = host_port_colon.unwrap_or(host_port.len());
        let host_text = &host_port[..host_end];
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (i, &b) in host_text.as_bytes().iter().enumerate() {
            if !is_host_byte(b) {
                return Err(ParseError::InvalidHost {
                    at: host_start + i,
                });
            }
        }
        let port = if let Some(c) = host_port_colon {
            Some(parse_port(
                &host_port[c + 1..],
                host_start + c + 1,
            )?)
        } else {
            None
        };
        (host_text.to_ascii_lowercase(), port, false)
    };

    let after_authority = &input[authority_end..];
    let frag_rel = after_authority.find('#');
    let before_fragment = frag_rel.map(|i| &after_authority[..i]).unwrap_or(after_authority);
    let fragment = frag_rel.map(|i| after_authority[i + 1..].to_string());

    let query_rel = before_fragment.find('?');
    let path_raw = query_rel
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_rel.map(|i| before_fragment[i + 1..].to_string());
    let path = if path_raw.is_empty() {
        "/".to_string()
    } else {
        path_raw.to_string()
    };

    let path_offset = authority_end;
    let query_offset = query_rel.map(|i| authority_end + i + 1);
    let fragment_offset = frag_rel.map(|i| authority_end + i + 1);

    if let Some(ref info) = userinfo {
        decode_component(Component::UserInfo, info)
            .map_err(|e| codec_to_parse(e, authority_start))?;
    }
    if !path_raw.is_empty() {
        decode_component(Component::Path, path_raw)
            .map_err(|e| codec_to_parse(e, path_offset))?;
    }
    if let Some(ref q) = query {
        decode_component(Component::Query, q)
            .map_err(|e| codec_to_parse(e, query_offset.unwrap()))?;
    }
    if let Some(ref f) = fragment {
        decode_component(Component::Fragment, f)
            .map_err(|e| codec_to_parse(e, fragment_offset.unwrap()))?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(port: &str, start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + i });
        }
    }
    let bytes = port.as_bytes();
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut value: u32 = 0;
    for b in bytes {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: start }),
        };
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            if hi.is_none() || lo.is_none() {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hi.unwrap() << 4) | lo.unwrap());
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_string()),
        Err(err) => {
            let at = err.valid_up_to();
            Err(CodecError::InvalidUtf8 {
                at: origins.get(at).copied().unwrap_or(input.len()),
            })
        }
    }
}

fn safe_component_byte(component: Component, b: u8) -> bool {
    if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
        return true;
    }
    match component {
        Component::UserInfo => b == b':',
        Component::Path => b == b'/',
        Component::Query => matches!(b, b'/' | b'?'),
        Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        if safe_component_byte(component, b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<String> = Vec::new();
    let mut terminal_navigation = false;

    for (i, raw) in raw_segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, raw).map_err(|e| match e {
            CodecError::BadEscape { at } => PathError::BadEscape { at: at + 1 + segment_offset(&raw_segments, i) },
            CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at: at + 1 + segment_offset(&raw_segments, i) },
        })?;
        let is_last = i + 1 == raw_segments.len();
        if decoded == "." {
            if is_last {
                terminal_navigation = true;
            }
        } else if decoded == ".." {
            if !stack.is_empty() {
                stack.pop();
            }
            if is_last {
                terminal_navigation = true;
            }
        } else {
            stack.push((*raw).to_string());
        }
    }

    if terminal_navigation {
        stack.push(String::new());
    }
    if stack.is_empty() {
        Ok("/".to_string())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn segment_offset(segments: &[&str], index: usize) -> usize {
    segments[..index]
        .iter()
        .map(|s| s.len() + 1)
        .sum()
}

fn scan_reference_bytes(reference: &str) -> Option<usize> {
    global_byte_error(reference)
}

fn split_reference(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let (before_fragment, fragment) = match reference.find('#') {
        Some(i) => (&reference[..i], Some(&reference[i + 1..])),
        None => (reference, None),
    };
    match before_fragment.find('?') {
        Some(i) => (
            &before_fragment[..i],
            Some(&before_fragment[i + 1..]),
            fragment,
        ),
        None => (before_fragment, None, fragment),
    }
}

fn url_raw_authority(url: &Url) -> String {
    let mut out = String::new();
    if let Some(ref userinfo) = url.userinfo {
        out.push_str(userinfo);
        out.push('@');
    }
    if url.ipv6 {
        out.push('[');
        out.push_str(&url.host);
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        out.push(':');
        out.push_str(&port.to_string());
    }
    out
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    if let Some(at) = scan_reference_bytes(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    let first_boundary = reference
        .char_indices()
        .find(|(_, c)| matches!(c, '/' | '?' | '#'))
        .map(|(i, _)| i)
        .unwrap_or(reference.len());
    if reference[..first_boundary].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let (reference_path, reference_query, reference_fragment) = split_reference(reference);
    let selected_path;
    let selected_query;

    if reference_path.is_empty() {
        selected_path = base_path.clone();
        selected_query = if reference_query.is_some() {
            reference_query.map(str::to_string)
        } else {
            base_url.query.clone()
        };
    } else {
        selected_query = reference_query.map(str::to_string);
        if reference_path.starts_with('/') {
            selected_path = reference_path.to_string();
        } else {
            let slash = base_path.rfind('/').map(|i| i + 1).unwrap_or(0);
            selected_path = format!("{}{}", &base_path[..slash], reference_path);
        }
    }

    let mut constructed = format!("{}://{}", base_url.scheme, url_raw_authority(&base_url));
    constructed.push_str(&selected_path);
    if let Some(query) = selected_query.as_ref() {
        constructed.push('?');
        constructed.push_str(query);
    }
    if let Some(fragment) = reference_fragment {
        constructed.push('#');
        constructed.push_str(fragment);
    }

    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}
