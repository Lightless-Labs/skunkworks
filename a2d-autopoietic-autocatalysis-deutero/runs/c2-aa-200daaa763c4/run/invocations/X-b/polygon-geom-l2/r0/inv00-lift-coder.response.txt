#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

const COORD_LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;
const MAX_QUERIES: usize = 4096;

fn in_range(point: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&point.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&point.y)
}

fn prepare(points: &[Point]) -> Result<&[Point], GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective = if points.len() >= 2 && points[points.len() - 1] == points[0] {
        &points[..points.len() - 1]
    } else {
        points
    };

    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }

    Ok(effective)
}

fn cross(a: Point, b: Point, q: Point) -> i128 {
    let ab_x = i128::from(b.x) - i128::from(a.x);
    let ab_y = i128::from(b.y) - i128::from(a.y);
    let aq_x = i128::from(q.x) - i128::from(a.x);
    let aq_y = i128::from(q.y) - i128::from(a.y);
    ab_x * aq_y - ab_y * aq_x
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = (i128::from(b.x) - i128::from(a.x)).unsigned_abs();
    let dy = (i128::from(b.y) - i128::from(a.y)).unsigned_abs();
    dx * dx + dy * dy
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let scaled_squared = squared_distance(a, b) * 1_000_000;
    let mut low = 0_u128;
    let mut high = 3_000_000_000_u128;

    while low < high {
        let mid = low + (high - low + 1) / 2;
        if mid * mid <= scaled_squared {
            low = mid;
        } else {
            high = mid - 1;
        }
    }

    let remainder = scaled_squared - low * low;
    let threshold = 2 * low + 1;
    if 4 * remainder >= threshold * threshold {
        (low + 1) as u64
    } else {
        low as u64
    }
}

fn edge_pair(a: Point, b: Point) -> (Point, Point) {
    if a <= b {
        (a, b)
    } else {
        (b, a)
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepare(points)?;
    let n = polygon.len();

    let mut signed_area2 = 0_i128;
    let mut edge_milli = Vec::with_capacity(n);
    let mut perimeter_milli = 0_u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..n {
        let a = polygon[i];
        let b = polygon[(i + 1) % n];

        signed_area2 += i128::from(a.x) * i128::from(b.y)
            - i128::from(a.y) * i128::from(b.x);

        let length = rounded_edge_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        longest_edge = match longest_edge {
            None => Some(i),
            Some(best) => {
                let best_length = edge_milli[best];
                if length > best_length
                    || (length == best_length
                        && (edge_pair(a, b)
                            < edge_pair(polygon[best], polygon[(best + 1) % n])
                            || (edge_pair(a, b)
                                == edge_pair(polygon[best], polygon[(best + 1) % n])
                                && i < best)))
                {
                    Some(i)
                } else {
                    Some(best)
                }
            }
        };
    }

    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare(points)?;

    if queries.len() > MAX_QUERIES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }

    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let n = polygon.len();
    let mut results = Vec::with_capacity(queries.len());

    for &query in queries {
        let mut boundary_edge = None;
        for i in 0..n {
            let a = polygon[i];
            let b = polygon[(i + 1) % n];

            if cross(a, b, query) == 0
                && query.x >= a.x.min(b.x)
                && query.x <= a.x.max(b.x)
                && query.y >= a.y.min(b.y)
                && query.y <= a.y.max(b.y)
            {
                boundary_edge = Some(i);
                break;
            }
        }

        if let Some(edge) = boundary_edge {
            results.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..n {
            let a = polygon[i];
            let b = polygon[(i + 1) % n];

            if (a.y > query.y) != (b.y > query.y) {
                let side = cross(a, b, query);
                if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                    inside = !inside;
                }
            }
        }

        results.push(if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }

    Ok(results)
}
