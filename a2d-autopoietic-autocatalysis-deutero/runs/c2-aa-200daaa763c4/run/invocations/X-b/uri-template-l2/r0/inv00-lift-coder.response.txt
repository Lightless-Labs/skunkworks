#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression {
    operator: Option<char>,
    variables: Vec<String>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[cursor..start]);

        let reserved = matches!(expression.operator, Some('+') | Some('#'));
        let mut fields = Vec::new();

        for name in &expression.variables {
            if let Some(value) = vars.iter().rev().find_map(|(key, value)| {
                if *key == name {
                    Some(*value)
                } else {
                    None
                }
            }) {
                fields.push(encode_value(value, reserved));
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some('#') {
                output.push('#');
            }
            let separator = if expression.operator == Some(';') {
                ";"
            } else if expression.operator == Some('?') || expression.operator == Some('&') {
                "&"
            } else if expression.operator == Some('.') {
                "."
            } else if expression.operator == Some('/') {
                "/"
            } else {
                ","
            };
            output.push_str(&fields.join(separator));
        }

        cursor = end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

fn validate_template(template: &str) -> Result<Vec<(usize, usize, Expression)>, Error> {
    let mut expressions = Vec::new();
    let mut cursor = 0;
    let bytes = template.as_bytes();

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let start = cursor;
                let Some(relative_end) = bytes[cursor + 1..].iter().position(|&byte| byte == b'}')
                else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = cursor + 1 + relative_end;

                if let Some(relative_open) =
                    bytes[cursor + 1..end].iter().position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: cursor + 1 + relative_open,
                    });
                }

                let content_start = cursor + 1;
                let content = &template[content_start..end];
                let (operator, names_start) = match content.as_bytes().first().copied() {
                    Some(b'+') => (Some('+'), content_start + 1),
                    Some(b'#') => (Some('#'), content_start + 1),
                    Some(byte @ (b'.' | b'/' | b';' | b'?' | b'&')) => {
                        return Err(Error::UnsupportedOperator {
                            pos: content_start + usize::from(byte != 0),
                        });
                    }
                    _ => (None, content_start),
                };

                // The unsupported operator is always the first content byte.
                if matches!(content.as_bytes().first(), Some(b'.' | b'/' | b';' | b'?' | b'&')) {
                    return Err(Error::UnsupportedOperator { pos: content_start });
                }

                if names_start == end {
                    return Err(Error::EmptyExpression { pos: end });
                }

                let mut names = Vec::new();
                let mut seen = Vec::<String>::new();
                let mut spec_start = names_start;

                for spec in template[names_start..end].split(',') {
                    if spec.is_empty() {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let name_bytes = spec.as_bytes();
                    if !is_name_start(name_bytes[0]) {
                        return Err(Error::InvalidName { pos: spec_start });
                    }
                    for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
                        if !is_name_continue(byte) {
                            return Err(Error::InvalidName {
                                pos: spec_start + offset,
                            });
                        }
                    }

                    if seen.iter().any(|name| name == spec) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(spec.to_owned());
                    names.push(spec.to_owned());
                    spec_start += spec.len() + 1;
                }

                expressions.push((start, end, Expression { operator, variables: names }));
                cursor = end + 1;
            }
            _ => cursor += 1,
        }
    }

    Ok(expressions)
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn encode_value(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::with_capacity(bytes.len());
    let mut cursor = 0;

    while cursor < bytes.len() {
        let byte = bytes[cursor];

        if byte == b'%' && cursor + 2 < bytes.len() && is_hex(bytes[cursor + 1]) && is_hex(bytes[cursor + 2]) {
            output.push('%');
            output.push((bytes[cursor + 1] as char).to_ascii_uppercase());
            output.push((bytes[cursor + 2] as char).to_ascii_uppercase());
            cursor += 3;
            continue;
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        cursor += 1;
    }

    output
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}
