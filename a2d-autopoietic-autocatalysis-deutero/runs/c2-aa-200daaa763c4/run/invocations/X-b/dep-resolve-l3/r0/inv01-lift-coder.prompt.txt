fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(value: &str) -> String {
    let mut result = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            result.push(*byte as char);
        } else {
            result.push('%');
            result.push_str(&format!("{:02X}", byte));
        }
    }
    result
}

fn active_constraints(
    catalog: &[Package],
    roots: &[Requirement],
    assignments: &[(String, Version)],
) -> Vec<(String, Vec<Constraint>)> {
    let mut active: Vec<(String, Vec<Constraint>)> = Vec::new();

    fn add(active: &mut Vec<(String, Vec<Constraint>)>, req: &Requirement) {
        match active.binary_search_by(|(name, _)| name.cmp(&req.package)) {
            Ok(index) => active[index].1.push(req.constraint),
            Err(index) => active.insert(index, (req.package.clone(), vec![req.constraint])),
        }
    }

    for root in roots {
        add(&mut active, root);
    }

    for (name, version) in assignments {
        if let Ok(index) = catalog.binary_search_by(|p| p.name.cmp(name)) {
            if let Some(release) = catalog[index]
                .releases
                .iter()
                .find(|r| r.version == *version)
            {
                for dep in &release.deps {
                    add(&mut active, dep);
                }
            }
        }
    }

    for (_, constraints) in &mut active {
        canonicalize(constraints);
    }
    active
}

fn assignment_version(assignments: &[(String, Version)], name: &str) -> Option<Version> {
    assignments
        .binary_search_by(|(n, _)| n.as_str().cmp(name))
        .ok()
        .map(|i| assignments[i].1)
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<(String, Version)>)>,
    first_failure: Option<ResolveError>,
}

impl Search<'_> {
    fn record_failure(&mut self, error: ResolveError) {
        if self.first_failure.is_none() {
            self.first_failure = Some(error);
        }
    }

    fn lock_cost(&self, assignments: &[(String, Version)]) -> usize {
        self.locks
            .iter()
            .filter(|lock| assignment_version(assignments, &lock.package) != Some(lock.version))
            .count()
    }

    fn better(
        &self,
        cost: usize,
        candidate: &[(String, Version)],
        current: &(usize, Vec<(String, Version)>),
    ) -> bool {
        if cost != current.0 {
            return cost < current.0;
        }
        for package in self.catalog {
            let a = assignment_version(candidate, &package.name);
            let b = assignment_version(&current.1, &package.name);
            match (a, b) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(av), Some(bv)) if av != bv => return av > bv,
                _ => {}
            }
        }
        false
    }

    fn inspect(&mut self, assignments: &mut Vec<(String, Version)>) -> Result<(), ResolveError> {
        let active = active_constraints(self.catalog, self.roots, assignments);

        for (name, _) in &active {
            if self.catalog.binary_search_by(|p| p.name.cmp(name)).is_err() {
                return Err(ResolveError::Missing {
                    package: name.clone(),
                });
            }
        }

        for (name, constraints) in &active {
            if let Some(selected) = assignment_version(assignments, name) {
                if !constraints.iter().all(|c| matches(selected, *c)) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if assignment_version(assignments, name).is_none() {
                let package = &self.catalog
                    [self.catalog.binary_search_by(|p| p.name.cmp(name)).unwrap()];
                if !package
                    .releases
                    .iter()
                    .any(|r| constraints.iter().all(|c| matches(r.version, *c)))
                {
                    return Err(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        let next = active
            .iter()
            .find(|(name, _)| assignment_version(assignments, name).is_none())
            .map(|(name, constraints)| (name.clone(), constraints.clone()));

        let Some((name, constraints)) = next else {
            let cost = self.lock_cost(assignments);
            if self
                .best
                .as_ref()
                .map_or(true, |current| self.better(cost, assignments, current))
            {
                self.best = Some((cost, assignments.clone()));
            }
            return Ok(());
        };

        let package = &self.catalog[self.catalog.binary_search_by(|p| p.name.cmp(&name)).unwrap()];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .filter(|release| {
                constraints
                    .iter()
                    .all(|constraint| matches(release.version, *constraint))
            })
            .map(|release| release.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.iter().find(|lock| lock.package == name) {
            if let Some(index) = candidates.iter().position(|version| *version == lock.version) {
                let locked_version = candidates.remove(index);
                candidates.insert(0, locked_version);
            }
        }

        let mut first_child_failure = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            let insert_at = assignments
                .binary_search_by(|(n, _)| n.cmp(&name))
                .unwrap_err();
            assignments.insert(insert_at, (name.clone(), candidate));

            match self.inspect(assignments) {
                Ok(()) => {}
                Err(error @ ResolveError::Limit { .. }) => {
                    assignments.remove(insert_at);
                    return Err(error);
                }
                Err(error) => {
                    assignments.remove(insert_at);
                    if first_child_failure.is_none() {
                        first_child_failure = Some(error);
                    }
                }
            }
        }

        if let Some(error) = first_child_failure {
            self.record_failure(error.clone());
            Err(error)
        } else {
            unreachable!()
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }

    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut sorted_catalog = catalog.to_vec();
    sorted_catalog.sort_by(|a, b| a.name.cmp(&b.name));

    let mut search = Search {
        catalog: &sorted_catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_failure: None,
    };
    let mut assignments = Vec::new();
    let search_result = search.inspect(&mut assignments);

    if let Err(ResolveError::Limit { attempts }) = search_result {
        return Err(ResolveError::Limit { attempts });
    }

    let Some((_, solution)) = search.best else {
        return Err(search.first_failure.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = solution
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let final_active = active_constraints(&sorted_catalog, roots, &solution);
    let mut changes = Vec::new();
    let names: std::collections::BTreeSet<String> = locks
        .iter()
        .map(|lock| lock.package.clone())
        .chain(solution.iter().map(|(name, _)| name.clone()))
        .collect();

    for name in names {
        let before = locks
            .iter()
            .find(|lock| lock.package == name)
            .map(|lock| lock.version);
        let after = assignment_version(&solution, &name);
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = sorted_catalog
                    .binary_search_by(|p| p.name.cmp(&name))
                    .ok()
                    .map(|i| sorted_catalog[i].releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let constraints = final_active
                        .binary_search_by(|(package, _)| package.cmp(&name))
                        .ok()
                        .map(|i| &final_active[i].1);
                    if constraints.is_some_and(|cs| !cs.iter().all(|c| matches(old, *c))) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
