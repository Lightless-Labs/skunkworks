#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes.last() != Some(&b'-')
        && bytes
            .iter()
            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-')
}

fn version_matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(expected) => version == expected,
        Constraint::AtLeast(minimum) => version >= minimum,
        Constraint::Before(bound) => version < bound,
        Constraint::Caret(base) => {
            version >= base
                && if base.major > 0 {
                    version.major == base.major
                } else if base.minor > 0 {
                    version.major == 0 && version.minor == base.minor
                } else {
                    version == base
                }
        }
    }
}

fn constraint_key(constraint: Constraint) -> (u8, Version) {
    match constraint {
        Constraint::Exact(version) => (0, version),
        Constraint::AtLeast(version) => (1, version),
        Constraint::Before(version) => (2, version),
        Constraint::Caret(version) => (3, version),
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort_by_key(|constraint| constraint_key(*constraint));
    constraints.dedup();
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(version) => format!("={}", version_text(version)),
        Constraint::AtLeast(version) => format!(">={}", version_text(version)),
        Constraint::Before(version) => format!("<{}", version_text(version)),
        Constraint::Caret(version) => format!("^{}", version_text(version)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(value: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut encoded = String::new();

    for &byte in value.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'-') {
            encoded.push(byte as char);
        } else {
            encoded.push('%');
            encoded.push(HEX[(byte >> 4) as usize] as char);
            encoded.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }

    encoded
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn add_constraint(active: &mut Vec<(String, Vec<Constraint>)>, requirement: &Requirement) {
    match active.binary_search_by(|(name, _)| name.cmp(&requirement.package)) {
        Ok(index) => active[index].1.push(requirement.constraint),
        Err(index) => active.insert(
            index,
            (requirement.package.clone(), vec![requirement.constraint]),
        ),
    }
}

fn assignment_version(assignments: &[(String, Version)], name: &str) -> Option<Version> {
    assignments
        .binary_search_by(|(assigned, _)| assigned.as_str().cmp(name))
        .ok()
        .map(|index| assignments[index].1)
}

fn active_constraints(
    catalog: &[Package],
    roots: &[Requirement],
    assignments: &[(String, Version)],
) -> Vec<(String, Vec<Constraint>)> {
    let mut active = Vec::new();

    for root in roots {
        add_constraint(&mut active, root);
    }

    for (name, version) in assignments {
        if let Ok(index) = catalog.binary_search_by(|package| package.name.cmp(name)) {
            if let Some(release) = catalog[index]
                .releases
                .iter()
                .find(|release| release.version == *version)
            {
                for dependency in &release.deps {
                    add_constraint(&mut active, dependency);
                }
            }
        }
    }

    for (_, constraints) in &mut active {
        canonicalize(constraints);
    }

    active
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<(String, Version)>)>,
    first_failure: Option<ResolveError>,
}

impl Search<'_> {
    fn lock_cost(&self, assignments: &[(String, Version)]) -> usize {
        self.locks
            .iter()
            .filter(|lock| assignment_version(assignments, &lock.package) != Some(lock.version))
            .count()
    }

    fn is_better(
        &self,
        cost: usize,
        candidate: &[(String, Version)],
        current: &(usize, Vec<(String, Version)>),
    ) -> bool {
        if cost != current.0 {
            return cost < current.0;
        }

        for package in self.catalog {
            match (
                assignment_version(candidate, &package.name),
                assignment_version(&current.1, &package.name),
            ) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(left), Some(right)) if left != right => return left > right,
                _ => {}
            }
        }

        false
    }

    fn record_failure(&mut self, error: ResolveError) {
        if self.first_failure.is_none() {
            self.first_failure = Some(error);
        }
    }

    fn inspect(&mut self, assignments: &mut Vec<(String, Version)>) -> Result<(), ResolveError> {
        let active = active_constraints(self.catalog, self.roots, assignments);

        if let Some((name, _)) = active.iter().find(|(name, _)| {
            self.catalog
                .binary_search_by(|package| package.name.cmp(name))
                .is_err()
        }) {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, constraints) in &active {
            if let Some(selected) = assignment_version(assignments, name) {
                if !constraints
                    .iter()
                    .all(|constraint| version_matches(selected, *constraint))
                {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if assignment_version(assignments, name).is_none() {
                let index = self
                    .catalog
                    .binary_search_by(|package| package.name.cmp(name))
                    .unwrap();
                if !self.catalog[index].releases.iter().any(|release| {
                    constraints
                        .iter()
                        .all(|constraint| version_matches(release.version, *constraint))
                }) {
                    return Err(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        let next = active
            .iter()
            .find(|(name, _)| assignment_version(assignments, name).is_none())
            .map(|(name, constraints)| (name.clone(), constraints.clone()));

        let Some((name, constraints)) = next else {
            let cost = self.lock_cost(assignments);
            if self
                .best
                .as_ref()
                .map_or(true, |current| self.is_better(cost, assignments, current))
            {
                self.best = Some((cost, assignments.clone()));
            }
            return Ok(());
        };

        let index = self
            .catalog
            .binary_search_by(|package| package.name.cmp(&name))
            .unwrap();

        let mut candidates: Vec<Version> = self.catalog[index]
            .releases
            .iter()
            .filter(|release| {
                constraints
                    .iter()
                    .all(|constraint| version_matches(release.version, *constraint))
            })
            .map(|release| release.version)
            .collect();
        candidates.sort_by(|left, right| right.cmp(left));

        if let Some(lock) = self.locks.iter().find(|lock| lock.package == name) {
            if let Some(position) = candidates.iter().position(|version| *version == lock.version) {
                let locked_version = candidates.remove(position);
                candidates.insert(0, locked_version);
            }
        }

        let mut first_child_failure = None;

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }

            self.attempts += 1;
            let insert_at = assignments
                .binary_search_by(|(assigned_name, _)| assigned_name.cmp(&name))
                .unwrap_err();
            assignments.insert(insert_at, (name.clone(), candidate));

            let child_result = self.inspect(assignments);

            match child_result {
                Ok(()) => {}
                Err(error @ ResolveError::Limit { .. }) => {
                    assignments.remove(insert_at);
                    return Err(error);
                }
                Err(error) => {
                    assignments.remove(insert_at);
                    if first_child_failure.is_none() {
                        first_child_failure = Some(error.clone());
                    }
                    self.record_failure(error);
                }
            }
        }

        match first_child_failure {
            Some(error) => Err(error),
            None => Ok(()),
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (index, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{index}]"),
                name: root.package.clone(),
            });
        }
    }

    for (package_index, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{package_index}]"),
                name: package.name.clone(),
            });
        }

        for (release_index, release) in package.releases.iter().enumerate() {
            for (dependency_index, dependency) in release.deps.iter().enumerate() {
                if !valid_name(&dependency.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!(
                            "catalog[{package_index}].release[{release_index}].dep[{dependency_index}]"
                        ),
                        name: dependency.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (index, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{index}]"),
                name: lock.package.clone(),
            });
        }
    }

    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut sorted_catalog = catalog.to_vec();
    sorted_catalog.sort_by(|left, right| left.name.cmp(&right.name));

    let mut search = Search {
        catalog: &sorted_catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_failure: None,
    };

    let mut assignments = Vec::new();
    if let Err(error) = search.inspect(&mut assignments) {
        if matches!(error, ResolveError::Limit { .. }) {
            return Err(error);
        }
    }

    let Some((_, solution)) = search.best else {
        return Err(search.first_failure.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected = solution
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let final_active = active_constraints(&sorted_catalog, roots, &solution);
    let names: std::collections::BTreeSet<String> = locks
        .iter()
        .map(|lock| lock.package.clone())
        .chain(solution.iter().map(|(name, _)| name.clone()))
        .collect();

    let mut changes = Vec::new();

    for name in names {
        let before = locks
            .iter()
            .find(|lock| lock.package == name)
            .map(|lock| lock.version);
        let after = assignment_version(&solution, &name);

        if before == after {
            continue;
        }

        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = sorted_catalog
                    .binary_search_by(|package| package.name.cmp(&name))
                    .ok()
                    .is_some_and(|index| {
                        sorted_catalog[index]
                            .releases
                            .iter()
                            .any(|release| release.version == old)
                    });

                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let constraints = final_active
                        .binary_search_by(|(package, _)| package.cmp(&name))
                        .ok()
                        .map(|index| &final_active[index].1);

                    if constraints.is_some_and(|constraints| {
                        !constraints
                            .iter()
                            .all(|constraint| version_matches(old, *constraint))
                    }) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => unreachable!(),
        };

        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
