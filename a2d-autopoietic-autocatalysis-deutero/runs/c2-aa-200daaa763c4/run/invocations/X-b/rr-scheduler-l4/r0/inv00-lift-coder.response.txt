#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum EventKind {
    Arrival,
    Wake,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
    level: usize,
}

#[derive(Clone, Debug)]
struct JobState {
    burst_position: usize,
    remaining: u32,
    level: usize,
    wake_level: usize,
    completion: Option<u64>,
}

fn validate(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<(), Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }
    Ok(())
}

fn mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        return 0;
    }
    let numerator = total * 100;
    (numerator + count as u64 / 2) / count as u64
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    validate(jobs, quanta, switch_cost)?;

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut states: Vec<JobState> = jobs
        .iter()
        .map(|job| JobState {
            burst_position: 0,
            remaining: job.bursts[0],
            level: job.level,
            wake_level: job.level,
            completion: None,
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: job.arrival as u64,
            kind: EventKind::Arrival,
            index,
            level: job.level,
        })
        .collect();

    let mut queues: Vec<Vec<usize>> = vec![Vec::new(); quanta.len()];
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut completed_count = 0usize;

    while completed_count < jobs.len() {
        events.sort_by_key(|event| {
            (
                event.time,
                match event.kind {
                    EventKind::Arrival => 0u8,
                    EventKind::Wake => 1u8,
                },
                event.index,
            )
        });

        let mut admitted = Vec::new();
        let mut pending = Vec::new();
        for event in events.drain(..) {
            if event.time <= time {
                admitted.push(event);
            } else {
                pending.push(event);
            }
        }
        events = pending;

        for event in admitted {
            let level = match event.kind {
                EventKind::Arrival => event.level,
                EventKind::Wake => states[event.index].wake_level,
            };
            queues[level].push(event.index);
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if let Some(next_event_time) = events.iter().map(|event| event.time).min() {
                if next_event_time > time {
                    if let Some((_, previous_end)) = idle.last_mut() {
                        if *previous_end == time {
                            *previous_end = next_event_time;
                        } else {
                            idle.push((time, next_event_time));
                        }
                    } else {
                        idle.push((time, next_event_time));
                    }
                    time = next_event_time;
                    last_dispatch_id = None;
                }
                continue;
            }
            break;
        };

        let job_index = queues[level].remove(0);
        let job = &jobs[job_index];
        if last_dispatch_id.is_some_and(|last_id| last_id != job.id) {
            time += switch_cost as u64;
        }

        events.sort_by_key(|event| {
            (
                event.time,
                match event.kind {
                    EventKind::Arrival => 0u8,
                    EventKind::Wake => 1u8,
                },
                event.index,
            )
        });
        let mut admitted = Vec::new();
        let mut pending = Vec::new();
        for event in events.drain(..) {
            if event.time <= time {
                admitted.push(event);
            } else {
                pending.push(event);
            }
        }
        events = pending;

        for event in admitted {
            let event_level = match event.kind {
                EventKind::Arrival => event.level,
                EventKind::Wake => states[event.index].wake_level,
            };
            queues[event_level].push(event.index);
        }

        let start = time;
        let consumed = states[job_index]
            .remaining
            .min(quanta[level]) as u64;
        time += consumed;
        let end = time;

        let burst_finished = states[job_index].remaining == consumed as u32;
        let final_burst = burst_finished && states[job_index].burst_position + 1 == job.bursts.len() - 1;

        runs.push(Run {
            id: job.id,
            start,
            end,
            finished: final_burst,
            level,
        });

        if burst_finished {
            if final_burst {
                states[job_index].completion = Some(end);
                completed_count += 1;
            } else {
                let next_io_position = states[job_index].burst_position + 1;
                let delay = job.bursts[next_io_position] as u64;
                states[job_index].burst_position += 2;
                states[job_index].remaining = job.bursts[states[job_index].burst_position];
                states[job_index].wake_level = level.saturating_sub(1);
                events.push(Event {
                    time: end + delay,
                    kind: EventKind::Wake,
                    index: job_index,
                    level: states[job_index].wake_level,
                });
            }
        } else {
            states[job_index].remaining -= consumed as u32;
            states[job_index].level = (level + 1).min(quanta.len() - 1);
            queues[states[job_index].level].push(job_index);
        }

        last_dispatch_id = Some(job.id);

        events.sort_by_key(|event| {
            (
                event.time,
                match event.kind {
                    EventKind::Arrival => 0u8,
                    EventKind::Wake => 1u8,
                },
                event.index,
            )
        });
        let mut admitted = Vec::new();
        let mut pending = Vec::new();
        for event in events.drain(..) {
            if event.time <= time {
                admitted.push(event);
            } else {
                pending.push(event);
            }
        }
        events = pending;

        for event in admitted {
            let event_level = match event.kind {
                EventKind::Arrival => event.level,
                EventKind::Wake => states[event.index].wake_level,
            };
            queues[event_level].push(event.index);
        }
    }

    let stats: Vec<Stat> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| {
            let completion = states[index].completion.unwrap_or(time);
            let turnaround = completion - job.arrival as u64;
            let cpu: u64 = job
                .bursts
                .iter()
                .step_by(2)
                .map(|&burst| burst as u64)
                .sum();
            let io: u64 = job
                .bursts
                .iter()
                .skip(1)
                .step_by(2)
                .map(|&burst| burst as u64)
                .sum();
            Stat {
                id: job.id,
                completion,
                turnaround,
                waiting: turnaround - cpu - io,
            }
        })
        .collect();

    let total_waiting: u64 = stats.iter().map(|stat| stat.waiting).sum();
    let total_turnaround: u64 = stats.iter().map(|stat| stat.turnaround).sum();

    Ok(Schedule {
        end: time,
        runs,
        idle,
        mean_waiting_x100: mean_x100(total_waiting, stats.len()),
        mean_turnaround_x100: mean_x100(total_turnaround, stats.len()),
        stats,
    })
}
