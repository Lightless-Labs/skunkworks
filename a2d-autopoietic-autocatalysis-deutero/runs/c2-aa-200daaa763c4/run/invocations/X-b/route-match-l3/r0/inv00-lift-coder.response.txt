#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing_slash: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(bytes: &[u8], start: usize) -> Result<String, usize> {
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(start + i);
            }
            let Some(h) = hex_upper(bytes[i + 1]) else {
                return Err(start + i);
            };
            let Some(l) = hex_upper(bytes[i + 2]) else {
                return Err(start + i);
            };
            let decoded = (h << 4) | l;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(start + i);
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(start + i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded segments are ASCII"))
}

fn valid_capture_name(name: &[u8]) -> bool {
    if name.is_empty() || !name[0].is_ascii_lowercase() {
        return false;
    }
    name[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool, Vec<u8>, bool), usize> {
    let bytes = pattern.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if pattern == "/" {
        return Ok((Vec::new(), false, vec![3], false));
    }

    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut rank = Vec::new();
    let mut has_wildcard = false;
    let mut start = 1;

    while start < end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(start);
        }
        let raw = &bytes[start..stop];
        if raw[0] == b':' || raw[0] == b'*' {
            let wildcard = raw[0] == b'*';
            let name = &raw[1..];
            if !valid_capture_name(name) || !names.insert(name.to_vec()) {
                return Err(start);
            }
            if wildcard && stop < end {
                return Err(start);
            }
            let name = std::str::from_utf8(name).unwrap().to_string();
            if wildcard {
                segments.push(Segment::Wildcard(name));
                rank.push(0);
                has_wildcard = true;
            } else {
                segments.push(Segment::Param(name));
                rank.push(1);
            }
        } else {
            match decode_segment(raw, start) {
                Ok(value) => {
                    segments.push(Segment::Static(value));
                    rank.push(2);
                }
                Err(at) => return Err(at),
            }
        }
        if stop == end {
            break;
        }
        start = stop + 1;
    }

    if !has_wildcard {
        rank.push(3);
    }
    Ok((segments, trailing, rank, has_wildcard))
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), usize> {
    let bytes = path.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut start = 1;

    while start < end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(start);
        }
        match decode_segment(&bytes[start..stop], start) {
            Ok(value) => segments.push(value),
            Err(at) => return Err(at),
        }
        if stop == end {
            break;
        }
        start = stop + 1;
    }

    Ok((segments, trailing))
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut i = 0;

    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if path.get(i)? != value {
                    return None;
                }
                i += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(i)?.clone()));
                i += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[i..].join("/");
                params.push((name.clone(), value));
                i = path.len();
                break;
            }
        }
    }

    if i != path.len() {
        return None;
    }
    Some(params)
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing_slash, rank, has_wildcard) =
            parse_pattern(pattern).map_err(|at| Error::InvalidPattern { at })?;
        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_string(),
            segments,
            trailing_slash,
            rank,
            has_wildcard,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .bytes()
            .position(|b| b == b'?' || b == b'#')
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (segments, trailing_slash) =
            parse_path(head).map_err(|at| Error::InvalidPath { at })?;

        let path_candidates: Vec<(&Route, Vec<(String, String)>)> = self
            .routes
            .iter()
            .filter_map(|route| {
                if self.policy == SlashPolicy::Strict
                    && route.trailing_slash != trailing_slash
                {
                    return None;
                }
                match_route(route, &segments).map(|params| (route, params))
            })
            .collect();

        if path_candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && path_candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<_> = path_candidates
            .iter()
            .filter(|(route, _)| route.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = path_candidates
                .iter()
                .map(|(route, _)| route.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|(a, _), (b, _)| b.rank.cmp(&a.rank).then_with(|| a.id.cmp(&b.id)));
        let (route, params) = eligible[0];

        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing_slash != trailing_slash
            && !segments.is_empty()
        {
            if route.trailing_slash {
                Some(format!("{head}/"))
            } else {
                Some(head[..head.len() - 1].to_string())
            }
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }
}
