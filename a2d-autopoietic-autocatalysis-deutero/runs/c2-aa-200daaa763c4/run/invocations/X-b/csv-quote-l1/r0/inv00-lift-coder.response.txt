#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        3
    } else {
        0
    };

    if i == bytes.len() {
        return Ok(Vec::new());
    }

    #[derive(Clone, Copy, PartialEq, Eq)]
    enum State {
        Unquoted,
        Quoted,
        AfterQuote,
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = Vec::new();
    let mut state = State::Unquoted;
    let mut record_index = 0usize;
    let mut quote_start = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if state != State::Quoted && b == b'\r' {
            if bytes.get(i + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: i,
                });
            }
            record.push(String::from_utf8(field).expect("valid UTF-8 field"));
            records.push(record);
            record = Vec::new();
            field = Vec::new();
            state = State::Unquoted;
            record_started = false;
            record_index += 1;
            i += 2;
            continue;
        }

        match state {
            State::Unquoted => match b {
                b'"' => {
                    if !field.is_empty() {
                        return Err(CsvError::UnexpectedQuote {
                            record: record_index,
                            byte: i,
                        });
                    }
                    quote_start = i;
                    state = State::Quoted;
                    record_started = true;
                    i += 1;
                }
                b',' => {
                    record.push(String::from_utf8(field).expect("valid UTF-8 field"));
                    field = Vec::new();
                    record_started = true;
                    i += 1;
                }
                b'\n' => {
                    record.push(String::from_utf8(field).expect("valid UTF-8 field"));
                    records.push(record);
                    record = Vec::new();
                    field = Vec::new();
                    state = State::Unquoted;
                    record_started = false;
                    record_index += 1;
                    i += 1;
                }
                _ => {
                    field.push(b);
                    record_started = true;
                    i += 1;
                }
            },
            State::Quoted => {
                if b == b'"' {
                    if bytes.get(i + 1) == Some(&b'"') {
                        field.push(b'"');
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    field.push(b);
                    i += 1;
                }
            }
            State::AfterQuote => match b {
                b',' => {
                    record.push(String::from_utf8(field).expect("valid UTF-8 field"));
                    field = Vec::new();
                    state = State::Unquoted;
                    record_started = true;
                    i += 1;
                }
                b'\n' => {
                    record.push(String::from_utf8(field).expect("valid UTF-8 field"));
                    records.push(record);
                    record = Vec::new();
                    field = Vec::new();
                    state = State::Unquoted;
                    record_started = false;
                    record_index += 1;
                    i += 1;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            },
        }
    }

    if state == State::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if record_started || !record.is_empty() || !field.is_empty() {
        record.push(String::from_utf8(field).expect("valid UTF-8 field"));
        records.push(record);
    }

    Ok(records)
}
