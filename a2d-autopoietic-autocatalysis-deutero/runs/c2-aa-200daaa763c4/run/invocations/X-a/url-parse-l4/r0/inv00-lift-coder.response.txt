    InvalidScheme { at: usize },

fn is_hex(b: u8) -> bool {
    b.is_ascii_digit() || (b'a'..=b'f').contains(&b) || (b'A'..=b'F').contains(&b)
}

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'a'..=b'f' => b - b'a' + 10,
        _ => b - b'A' + 10,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() || !is_hex(bytes[i + 1]) || !is_hex(bytes[i + 2]) {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]));
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: origins.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

fn component_safe(component: Component, b: u8) -> bool {
    if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
        return true;
    }
    match component {
        Component::UserInfo => b == b':',
        Component::Path => b == b'/',
        Component::Query => matches!(b, b'/' | b'?'),
        Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        if component_safe(component, b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn map_codec_parse(err: CodecError, offset: usize) -> ParseError {
    match err {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

fn validate_component(
    component: Component,
    value: &str,
    offset: usize,
) -> Result<(), ParseError> {
    decode_component(component, value)
        .map(|_| ())
        .map_err(|e| map_codec_parse(e, offset))
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        if !(bytes[i].is_ascii_alphanumeric() || matches!(bytes[i], b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let mut auth_end = input.len();
    for i in auth_start..input.len() {
        if matches!(bytes[i], b'/' | b'?' | b'#') {
            auth_end = i;
            break;
        }
    }
    let authority = &input[auth_start..auth_end];
    let ats: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(i))
        .collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: auth_start + ats[1],
        });
    }

    let (userinfo, hostport_start) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: auth_start + at,
            });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let hostport = &authority[hostport_start..];
    let hostport_abs = auth_start + hostport_start;
    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let close = match hostport.as_bytes().iter().position(|&b| b == b']') {
            Some(i) => i,
            None => {
                return Err(ParseError::InvalidIpv6 {
                    at: hostport_abs,
                });
            }
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 {
                at: hostport_abs,
            });
        }
        ipv6 = true;
        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], hostport_abs + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_abs + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), port)
    } else {
        let split = hostport.find(':');
        let (host_part, port) = if let Some(p) = split {
            (
                &hostport[..p],
                Some(parse_port(&hostport[p + 1..], hostport_abs + p + 1)?),
            )
        } else {
            (hostport, None)
        };
        if host_part.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_abs });
        }
        for (i, b) in host_part.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_abs + i,
                });
            }
        }
        (host_part.to_ascii_lowercase(), port)
    };

    let remainder = &input[auth_end..];
    let fragment_pos = remainder.find('#');
    let before_fragment = fragment_pos
        .map(|i| &remainder[..i])
        .unwrap_or(remainder);
    let fragment = fragment_pos.map(|i| remainder[i + 1..].to_owned());

    let query_pos = before_fragment.find('?');
    let raw_path = query_pos
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_pos.map(|i| before_fragment[i + 1..].to_owned());
    let path = if raw_path.is_empty() {
        "/".to_owned()
    } else {
        raw_path.to_owned()
    };

    if let Some(ref u) = userinfo {
        validate_component(Component::UserInfo, u, auth_start)?;
    }
    if !raw_path.is_empty() {
        validate_component(Component::Path, raw_path, auth_end)?;
    }
    if let Some(qpos) = query_pos {
        validate_component(Component::Query, &before_fragment[qpos + 1..], auth_end + qpos + 1)?;
    }
    if let Some(fpos) = fragment_pos {
        validate_component(
            Component::Fragment,
            &remainder[fpos + 1..],
            auth_end + fpos + 1,
        )?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(port: &str, offset: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: offset });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: offset + i });
        }
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: offset });
    }
    let mut value: u32 = 0;
    for b in port.bytes() {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: offset }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: offset });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: offset });
    }
    Ok(value as u16)
}

fn map_codec_path(err: CodecError) -> PathError {
    match err {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(map_codec_path)?;

    let mut stack: Vec<String> = Vec::new();
    let mut segments = path[1..].split('/').peekable();
    let mut last_navigation = false;

    for segment in segments.by_ref() {
        let decoded = decode_component(Component::Path, segment).map_err(map_codec_path)?;
        if decoded == "." {
            last_navigation = true;
        } else if decoded == ".." {
            if !stack.is_empty() {
                stack.pop();
            }
            last_navigation = true;
        } else {
            stack.push(segment.to_owned());
            last_navigation = false;
        }
    }

    let final_segment = path.rsplit('/').next().unwrap_or("");
    let final_decoded = decode_component(Component::Path, final_segment).map_err(map_codec_path)?;
    if final_decoded == "." || final_decoded == ".." {
        stack.push(String::new());
    }

    let _ = last_navigation;
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn scan_reference(input: &str) -> Result<(), ParseError> {
    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn parse_reference_components(input: &str) -> (String, Option<String>, Option<String>) {
    let fragment_pos = input.find('#');
    let before_fragment = fragment_pos.map(|i| &input[..i]).unwrap_or(input);
    let fragment = fragment_pos.map(|i| input[i + 1..].to_owned());
    let query_pos = before_fragment.find('?');
    let path = query_pos
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment)
        .to_owned();
    let query = query_pos.map(|i| before_fragment[i + 1..].to_owned());
    (path, query, fragment)
}

fn path_of(url: &Url) -> Result<String, ResolveError> {
    normalise_path(&url.path).map_err(ResolveError::Path)
}

fn authority_of(url: &Url) -> String {
    let mut out = String::new();
    if let Some(ref u) = url.userinfo {
        out.push_str(u);
        out.push('@');
    }
    if url.ipv6 {
        out.push('[');
        out.push_str(&url.host);
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        out.push(':');
        out.push_str(&port.to_string());
    }
    out
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let normalized_base = path_of(&base_url)?;
    scan_reference(reference).map_err(ResolveError::Reference)?;

    let boundary = reference
        .char_indices()
        .find(|(_, c)| matches!(c, '/' | '?' | '#'))
        .map(|(i, _)| i)
        .unwrap_or(reference.len());
    if reference[..boundary].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let (ref_path, ref_query, ref_fragment) = parse_reference_components(reference);
    let (selected_path, selected_query) = if ref_path.is_empty() {
        (
            normalized_base.clone(),
            if ref_query.is_none() {
                base_url.query.clone()
            } else {
                ref_query
            },
        )
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path
        } else {
            let slash = normalized_base.rfind('/').unwrap_or(0);
            format!("{}{}", &normalized_base[..=slash], ref_path)
        };
        (merged, ref_query)
    };

    let mut absolute = format!("{}://{}", base_url.scheme, authority_of(&base_url));
    absolute.push_str(&selected_path);
    if let Some(q) = selected_query {
        absolute.push('?');
        absolute.push_str(&q);
    }
    if let Some(f) = ref_fragment {
        absolute.push('#');
        absolute.push_str(&f);
    }

    let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

fn map_codec_query(err: CodecError, offset: usize) -> QueryError {
    match err {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    }
}

fn replace_plus(value: &str) -> String {
    value.replace('+', " ")
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut field_offset = 0usize;

    for field in input.split('&') {
        let current_offset = field_offset;
        field_offset += field.len() + 1;
        if field.is_empty() {
            continue;
        }
        let eq = field.find('=');
        let (raw_key, raw_value) = match eq {
            Some(i) => (&field[..i], &field[i + 1..]),
            None => (field, ""),
        };
        let key_input = replace_plus(raw_key);
        let key = decode_component(Component::Query, &key_input)
            .map_err(|e| map_codec_query(e, current_offset))?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: current_offset });
        }
        let value_offset = current_offset + raw_key.len() + usize::from(eq.is_some());
        let value_input = replace_plus(raw_value);
        let value = decode_component(Component::Query, &value_input)
            .map_err(|e| map_codec_query(e, value_offset))?;

        if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }
    Ok(QueryMap { entries })
}

fn encode_form(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::new();
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing)) = merged.iter_mut().find(|(k, _)| k == key) {
            existing.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        for value in values {
            fields.push(format!("{}={}", encode_form(&key), encode_form(&value)));
        }
    }
    fields.join("&")
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match &url.query {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(userinfo) = &url.userinfo {
        let decoded =
            decode_component(Component::UserInfo, userinfo).map_err(|e| match e {
                CodecError::BadEscape { at } => {
                    CanonicalError::Parse(ParseError::BadEscape { at })
                }
                CodecError::InvalidUtf8 { at } => {
                    CanonicalError::Parse(ParseError::InvalidUtf8 { at })
                }
            })?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }

    if url.ipv6 {
        let addr = Ipv6Addr::from_str(&url.host).expect("parsed IPv6 host remains valid");
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }

    if let Some(port) = url.port {
        let default = (url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443);
        if !default {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let mut encoded_segments = Vec::new();
    for segment in path.split('/') {
        let decoded = decode_component(Component::Path, segment).map_err(|e| match e {
            CodecError::BadEscape { at } => {
                CanonicalError::Path(PathError::BadEscape { at })
            }
            CodecError::InvalidUtf8 { at } => {
                CanonicalError::Path(PathError::InvalidUtf8 { at })
            }
        })?;
        encoded_segments.push(encode_unreserved(&decoded));
    }
    out.push_str(&encoded_segments.join("/"));

    if let Some(map) = query_map {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }
    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment).map_err(|e| match e {
            CodecError::BadEscape { at } => {
                CanonicalError::Parse(ParseError::BadEscape { at })
            }
            CodecError::InvalidUtf8 { at } => {
                CanonicalError::Parse(ParseError::InvalidUtf8 { at })
            }
        })?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(out)
}

fn encode_unreserved(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::new();
    for &b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}
