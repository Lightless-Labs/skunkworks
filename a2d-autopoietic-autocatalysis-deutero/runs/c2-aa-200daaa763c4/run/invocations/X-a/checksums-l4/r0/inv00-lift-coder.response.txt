const ADLER_MOD: u64 = 65_521;

fn crc32_raw_update(mut crc: u32, data: &[u8]) -> u32 {
    for &byte in data {
        crc ^= byte as u32;
        for _ in 0..8 {
            let low = crc & 1;
            crc >>= 1;
            if low != 0 {
                crc ^= 0xedb8_8320;
            }
        }
    }
    crc
}

pub fn crc32(data: &[u8]) -> u32 {
    crc32_raw_update(0xffff_ffff, data) ^ 0xffff_ffff
}

fn adler_update(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *a = ((*a as u64 + byte as u64) % ADLER_MOD) as u32;
        *b = ((*b as u64 + *a as u64) % ADLER_MOD) as u32;
    }
}

pub fn adler32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (1, 0);
    adler_update(&mut a, &mut b, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }

    Ok(())
}

#[derive(Clone, Copy)]
struct BaseState {
    crc: u32,
    a: u32,
    b: u32,
}

impl BaseState {
    fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    fn update(&mut self, data: &[u8]) {
        self.crc = crc32_raw_update(self.crc, data);
        adler_update(&mut self.a, &mut self.b, data);
    }

    fn checksum(self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = BaseState::new();
    for span in spans {
        state.update(&data[span.start..span.end]);
    }
    Ok(state.checksum())
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    a: u32,
    b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= ADLER_MOD as u32 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= ADLER_MOD as u32 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc = crc32_raw_update(self.crc, data);
        adler_update(&mut self.a, &mut self.b, data);
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let checksums = self.checksum();
        format!("c{:08x}:a{:08x}", checksums.crc32, checksums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut crc = 0u32;
        for offset in 1..9 {
            crc = (crc << 4)
                | hex_value(bytes[offset]).ok_or(SnapshotError::Hex { offset })? as u32;
        }

        let mut adler = 0u32;
        for offset in 11..19 {
            adler = (adler << 4)
                | hex_value(bytes[offset]).ok_or(SnapshotError::Hex { offset })? as u32;
        }

        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

#[derive(Clone, Copy)]
struct FletcherState {
    a16: u32,
    b16: u32,
    a32: u32,
    b32: u32,
    pending_high: Option<u8>,
}

impl FletcherState {
    fn new() -> Self {
        Self {
            a16: 0,
            b16: 0,
            a32: 0,
            b32: 0,
            pending_high: None,
        }
    }

    fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.a16 = (self.a16 + byte as u32) % 255;
            self.b16 = (self.b16 + self.a16) % 255;

            if let Some(high) = self.pending_high.take() {
                let word = ((high as u32) << 8) | byte as u32;
                self.update_word(word);
            } else {
                self.pending_high = Some(byte);
            }
        }
    }

    fn update_word(&mut self, word: u32) {
        self.a32 = (self.a32 + word) % 65_535;
        self.b32 = (self.b32 + self.a32) % 65_535;
    }

    fn finish_fletcher32(&mut self) -> u32 {
        if let Some(high) = self.pending_high.take() {
            self.update_word((high as u32) << 8);
        }
        (self.b32 << 16) | self.a32
    }

    fn fletcher16(self) -> u16 {
        ((self.b16 << 8) | self.a16) as u16
    }

    fn fletcher32_value(self) -> u32 {
        (self.b32 << 16) | self.a32
    }
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut state = FletcherState::new();
    state.update(data);
    state.fletcher16()
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut state = FletcherState::new();
    state.update(data);
    state.finish_fletcher32()
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = FletcherState::new();
    for span in spans {
        state.update(&data[span.start..span.end]);
    }
    Ok(state.finish_fletcher32())
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut crc: u16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = crc & 0x8000;
                    crc = crc.wrapping_shl(1);
                    if top != 0 {
                        crc ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                crc ^= byte as u16;
                for _ in 0..8 {
                    let low = crc & 1;
                    crc >>= 1;
                    if low != 0 {
                        crc ^= 0x8408;
                    }
                }
            }
        }
    }

    if matches!(variant, Crc16::X25) {
        crc ^ 0xffff
    } else {
        crc
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut base = BaseState::new();
    let mut fletcher = FletcherState::new();
    let mut crc16_state = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        base.update(selected);
        fletcher.update(selected);
        crc16_state = crc16_update(crc16_state, selected, variant);
    }

    Ok(Multi {
        base: base.checksum(),
        fletcher16: fletcher.fletcher16(),
        fletcher32: fletcher.finish_fletcher32(),
        crc16: if matches!(variant, Crc16::X25) {
            crc16_state ^ 0xffff
        } else {
            crc16_state
        },
    })
}

fn crc16_update(mut crc: u16, data: &[u8], variant: Crc16) -> u16 {
    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = crc & 0x8000;
                    crc = crc.wrapping_shl(1);
                    if top != 0 {
                        crc ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                crc ^= byte as u16;
                for _ in 0..8 {
                    let low = crc & 1;
                    crc >>= 1;
                    if low != 0 {
                        crc ^= 0x8408;
                    }
                }
            }
        }
    }
    crc
}

fn crc32_zero_byte(mut crc: u32) -> u32 {
    for _ in 0..8 {
        let low = crc & 1;
        crc >>= 1;
        if low != 0 {
            crc ^= 0xedb8_8320;
        }
    }
    crc
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for i in 0..32 {
        square[i] = gf2_matrix_times(matrix, matrix[i]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    // Matrix for one zero byte, represented by its action on each basis bit.
    let mut power = [0u32; 32];
    for bit in 0..32 {
        power[bit] = crc32_zero_byte(1u32 << bit);
    }

    let mut value = left;
    let mut length = right_len;
    while length != 0 {
        if length & 1 != 0 {
            value = gf2_matrix_times(&power, value);
        }
        length >>= 1;
        if length != 0 {
            let mut squared = [0u32; 32];
            gf2_matrix_square(&mut squared, &power);
            power = squared;
        }
    }
    value ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut combined = 0u32;
    for part in parts {
        combined = crc32_combine(combined, part.crc32, part.len);
    }

    Ok(Part {
        crc32: combined,
        len: total,
    })
}

fn selected_len(spans: &[Span]) -> u64 {
    spans.iter().fold(0u64, |total, span| {
        total + (span.end - span.start) as u64
    })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    let hashes = checksums_spans(data, spans)?;
    let len = selected_len(spans);
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, hashes.crc32, hashes.adler32
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for range in [4..20, 25..33, 40..48] {
        for offset in range {
            if hex_value(bytes[offset]).is_none() {
                return Err(VerifyError::Hex { offset });
            }
        }
    }

    let mut len = 0u64;
    for &byte in &bytes[4..20] {
        len = (len << 4) | hex_value(byte).unwrap() as u64;
    }
    let mut expected_crc = 0u32;
    for &byte in &bytes[25..33] {
        expected_crc = (expected_crc << 4) | hex_value(byte).unwrap() as u32;
    }
    let mut expected_adler = 0u32;
    for &byte in &bytes[40..48] {
        expected_adler = (expected_adler << 4) | hex_value(byte).unwrap() as u32;
    }

    if len != selected_len(spans) {
        return Err(VerifyError::Length);
    }

    let actual = checksums_spans(data, spans).map_err(VerifyError::Span)?;
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
