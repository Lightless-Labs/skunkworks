#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Option<char>,
    names: Vec<String>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse(template)?;
    let mut output = String::new();
    let mut expression_index = 0;
    let mut cursor = 0;

    while cursor < template.len() {
        let Some(relative_open) = template[cursor..].find('{') else {
            output.push_str(&template[cursor..]);
            break;
        };
        let open = cursor + relative_open;
        output.push_str(&template[cursor..open]);

        let close = template[open..].find('}').map(|n| open + n).unwrap();
        let expression = &expressions[expression_index];
        append_expansion(&mut output, expression, vars);

        cursor = close + 1;
        expression_index += 1;
    }

    Ok(output)
}

fn parse(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = bytes[open + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative_close;

                if let Some(relative_nested) =
                    bytes[open + 1..close].iter().position(|&b| b == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative_nested,
                    });
                }

                expressions.push(parse_expression(template, open + 1, close)?);
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    Ok(expressions)
}

fn parse_expression(template: &str, start: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let operator = match bytes[start] {
        b'+' => Some('+'),
        b'#' => Some('#'),
        b'.' => return Err(Error::UnsupportedOperator { pos: start }),
        b'/' => return Err(Error::UnsupportedOperator { pos: start }),
        b';' => return Err(Error::UnsupportedOperator { pos: start }),
        b'?' => return Err(Error::UnsupportedOperator { pos: start }),
        b'&' => return Err(Error::UnsupportedOperator { pos: start }),
        _ => None,
    };

    let content_start = start + usize::from(operator.is_some());
    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen: Vec<String> = Vec::new();
    let mut spec_start = content_start;

    for spec_end in (content_start..=close).filter(|&i| i == close || bytes[i] == b',') {
        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..spec_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        if let Some(offset) = name_bytes[1..].iter().position(|&b| !is_name_continue(b)) {
            return Err(Error::InvalidName {
                pos: spec_start + 1 + offset,
            });
        }

        let name = template[spec_start..spec_end].to_owned();
        if seen.iter().any(|previous| previous == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        names.push(name);
        spec_start = spec_end + 1;
    }

    Ok(Expression { operator, names })
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn append_expansion(output: &mut String, expression: &Expression, vars: &[(&str, &str)]) {
    let mut fields = Vec::new();
    for name in &expression.names {
        if let Some(value) = vars
            .iter()
            .rev()
            .find_map(|(binding_name, value)| (*binding_name == name).then_some(*value))
        {
            fields.push(value);
        }
    }

    if fields.is_empty() {
        return;
    }

    if expression.operator == Some('#') {
        output.push('#');
    }

    let reserved = matches!(expression.operator, Some('+') | Some('#'));
    for (index, field) in fields.into_iter().enumerate() {
        if index != 0 {
            output.push(',');
        }
        encode_value(output, field, reserved);
    }
}

fn encode_value(output: &mut String, value: &str, allow_reserved: bool) {
    let bytes = value.as_bytes();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];
        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) =
                (hex_value(bytes[index + 1]), hex_value(bytes[index + 2]))
            {
                output.push('%');
                output.push(hex_digit(high));
                output.push(hex_digit(low));
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
