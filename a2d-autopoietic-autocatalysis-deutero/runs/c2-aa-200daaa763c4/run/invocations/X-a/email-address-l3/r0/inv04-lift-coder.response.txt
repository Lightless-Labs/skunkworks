#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let bytes = input.as_bytes();
    if let Some(at) = bytes.iter().position(|b| *b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut clean = Vec::<u8>::with_capacity(bytes.len());
    let mut positions = Vec::<usize>::with_capacity(bytes.len());
    let mut runs = Vec::<usize>::new();
    let mut i = 0usize;
    let mut depth = 0usize;
    let mut outer_comment = 0usize;
    let mut protected_quote = false;

    while i < bytes.len() {
        if protected_quote {
            let b = bytes[i];
            clean.push(b);
            positions.push(i);
            i += 1;
            if b == b'\\' {
                if i < bytes.len() {
                    clean.push(bytes[i]);
                    positions.push(i);
                    i += 1;
                }
            } else if b == b'"' {
                protected_quote = false;
            }
            continue;
        }

        if depth == 0 && clean.is_empty() && bytes[i] == b'"' {
            protected_quote = true;
            clean.push(bytes[i]);
            positions.push(i);
            i += 1;
            continue;
        }

        if depth == 0
            && (is_hws(bytes[i])
                || bytes[i] == b'('
                || bytes[i] == b'\r'
                || bytes[i] == b'\n')
        {
            let run_start = i;
            loop {
                if depth == 0 {
                    if i >= bytes.len() {
                        break;
                    }
                    if is_hws(bytes[i]) {
                        i += 1;
                        continue;
                    }
                    if bytes[i] == b'\r' || bytes[i] == b'\n' {
                        if let Some(end) = fold_end(bytes, i) {
                            i = end;
                            continue;
                        }
                        return Err(EmailError::BadFold { at: i });
                    }
                    if bytes[i] == b'(' {
                        outer_comment = i;
                        depth = 1;
                        i += 1;
                        continue;
                    }
                    break;
                }

                if i >= bytes.len() {
                    break;
                }
                let b = bytes[i];
                if b == b'\\' {
                    if i + 1 >= bytes.len() || !is_printable(bytes[i + 1]) {
                        return Err(EmailError::CommentEscape { at: i });
                    }
                    i += 2;
                    continue;
                }
                if b == b'\r' || b == b'\n' {
                    if let Some(end) = fold_end(bytes, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if b == b'(' {
                    if depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    depth += 1;
                    i += 1;
                    continue;
                }
                if b == b')' {
                    depth -= 1;
                    i += 1;
                    continue;
                }
                if !is_printable(b) && b != b'\t' {
                    return Err(EmailError::CommentChar { at: i });
                }
                i += 1;
            }
            runs.push(run_start);
            continue;
        }

        match bytes[i] {
            b')' => return Err(EmailError::UnexpectedClose { at: i }),
            b'\r' | b'\n' => return Err(EmailError::BadFold { at: i }),
            b => {
                clean.push(b);
                positions.push(i);
                i += 1;
            }
        }
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }

    for &start in &runs {
        let next = positions.iter().position(|p| *p >= start);
        let left_adjacent = positions
            .iter()
            .rposition(|p| *p < start)
            .is_some_and(|idx| clean[idx] == b'@');
        let right_adjacent = next.is_some_and(|idx| clean[idx] == b'@');
        let at_boundary = next.is_none() || left_adjacent || right_adjacent;
        if !at_boundary {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quoted_close = None;
    if clean.first() == Some(&b'"') {
        let mut j = 1usize;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                quoted_close = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if quoted_close.is_none() {
            return Err(EmailError::UnclosedQuote { at: positions[0] });
        }
    }

    let protected_end = quoted_close.unwrap_or(0);
    let ats: Vec<usize> = clean
        .iter()
        .enumerate()
        .filter_map(|(idx, b)| {
            if *b == b'@' && !(quoted_close.is_some() && idx <= protected_end) {
                Some(idx)
            } else {
                None
            }
        })
        .collect();

    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt { at: positions[ats[1]] });
    }

    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: positions[sep] });
    }

    let local_bytes = &clean[..sep];
    let local_positions = &positions[..sep];
    let (local, quoted) = if local_bytes.first() == Some(&b'"') {
        let close = quoted_close.unwrap();
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: positions[close + 1],
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1usize;
        while j < close {
            let b = local_bytes[j];
            if b == b'\\' {
                let Some(&escaped) = local_bytes.get(j + 1) else {
                    return Err(EmailError::BadEscape {
                        at: local_positions[j],
                    });
                };
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: local_positions[j],
                    });
                }
                decoded.push(escaped);
                origins.push(local_positions[j]);
                j += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: local_positions[j],
                    });
                }
                decoded.push(b);
                origins.push(local_positions[j]);
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: positions[close] });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_positions[64],
            });
        }
        for idx in 0..local_bytes.len() {
            let b = local_bytes[idx];
            if b == b'.' {
                if idx == 0
                    || idx + 1 == local_bytes.len()
                    || local_bytes[idx - 1] == b'.'
                {
                    return Err(EmailError::LocalDot {
                        at: local_positions[idx],
                    });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar {
                    at: local_positions[idx],
                });
            }
        }
        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start_idx = sep + 1;
    let domain_start = positions
        .get(domain_start_idx)
        .copied()
        .unwrap_or(input.len());
    let mut domain_end = clean.len();
    let mut removed_dot = None;
    if domain_end > domain_start_idx && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        removed_dot = Some(positions[domain_end]);
    }
    if domain_end == domain_start_idx {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_bytes = &clean[domain_start_idx..domain_end];
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start_idx + 253],
        });
    }

    let mut labels = Vec::<(usize, usize)>::new();
    let mut label_start = domain_start_idx;
    for idx in domain_start_idx..=domain_end {
        if idx == domain_end || clean[idx] == b'.' {
            if idx == label_start {
                let at = if idx == domain_end {
                    removed_dot.unwrap_or(input.len())
                } else {
                    positions[idx]
                };
                return Err(EmailError::EmptyLabel { at });
            }
            labels.push((label_start, idx));
            label_start = idx + 1;
        }
    }

    for &(start, end) in &labels {
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: positions[start + 63],
            });
        }
        if clean[start] == b'-' {
            return Err(EmailError::LabelHyphen { at: positions[start] });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[end - 1],
            });
        }
        for idx in start..end {
            let b = clean[idx];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: positions[idx] });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: positions[last_start],
        });
    }

    let domain = clean[domain_start_idx..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
