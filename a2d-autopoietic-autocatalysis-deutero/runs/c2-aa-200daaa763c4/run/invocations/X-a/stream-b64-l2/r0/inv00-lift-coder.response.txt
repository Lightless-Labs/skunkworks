#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn digit(byte: u8) -> Option<u8> {
    ALPHABET.iter().position(|&b| b == byte).map(|n| n as u8)
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut index = 0;

        if self.tail_len > 0 {
            while self.tail_len < 2 && index < input.len() {
                self.tail[self.tail_len] = input[index];
                self.tail_len += 1;
                index += 1;
            }

            if self.tail_len == 2 && index < input.len() {
                encode_triplet(
                    &mut output,
                    self.tail[0],
                    self.tail[1],
                    input[index],
                );
                self.tail_len = 0;
                index += 1;
            } else if self.tail_len < 2 {
                return Ok(output);
            }
        }

        while index + 3 <= input.len() {
            encode_triplet(
                &mut output,
                input[index],
                input[index + 1],
                input[index + 2],
            );
            index += 3;
        }

        self.tail_len = input.len() - index;
        for i in 0..self.tail_len {
            self.tail[i] = input[index + i];
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            1 => {
                let a = self.tail[0];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                output.push_str("==");
            }
            2 => {
                let a = self.tail[0];
                let b = self.tail[1];
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => {}
        }
        self.tail_len = 0;
        Ok(output)
    }
}

impl Default for Encoder {
    fn default() -> Self {
        Self::new()
    }
}

fn encode_triplet(output: &mut String, a: u8, b: u8, c: u8) {
    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    output.push(ALPHABET[(c & 0x3f) as usize] as char);
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    group_len: usize,
    pending_second_padding: bool,
    terminal_padding: bool,
    next_offset: usize,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            group_len: 0,
            pending_second_padding: false,
            terminal_padding: false,
            next_offset: 0,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let offset = self.next_offset;
            self.next_offset += 1;

            let error = if self.terminal_padding {
                Some(Error::BadPadding { offset })
            } else if self.pending_second_padding && byte != b'=' {
                Some(Error::BadPadding { offset })
            } else if self.group_len < 2 && byte == b'=' {
                Some(Error::BadPadding { offset })
            } else if byte != b'=' && digit(byte).is_none() {
                Some(Error::InvalidByte { offset, byte })
            } else {
                None
            };

            if let Some(error) = error {
                self.sticky_error = Some(error.clone());
                return Err(error);
            }

            let position = self.group_len;
            self.group[position] = byte;
            self.offsets[position] = offset;
            self.group_len += 1;

            if position == 2 && byte == b'=' {
                self.pending_second_padding = true;
            }

            if self.group_len == 4 {
                let a = digit(self.group[0]).expect("validated digit");
                let b = digit(self.group[1]).expect("validated digit");
                let c = if self.group[2] == b'=' {
                    0
                } else {
                    digit(self.group[2]).expect("validated digit")
                };

                output.push((a << 2) | (b >> 4));

                if self.group[2] != b'=' {
                    output.push((b << 4) | (c >> 2));
                }

                if self.group[3] != b'=' {
                    let d = digit(self.group[3]).expect("validated digit");
                    output.push((c << 6) | d);
                } else {
                    self.terminal_padding = true;
                }

                self.group_len = 0;
                self.pending_second_padding = false;
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if self.pending_second_padding {
            let error = Error::BadPadding {
                offset: self.offsets[2],
            };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        if self.group_len != 0 {
            let error = Error::Truncated {
                offset: self.offsets[0],
            };
            self.sticky_error = Some(error.clone());
            return Err(error);
        }

        self.finished = true;
        Ok(Vec::new())
    }
}

impl Default for Decoder {
    fn default() -> Self {
        Self::new()
    }
}
