#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    fields: [Vec<bool>; 5],
    day_stars: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        let min = domain_min(field);
        self.fields[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &selected)| {
                if selected {
                    Some((i as u8) + min)
                } else {
                    None
                }
            })
            .collect()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn domain_min(field: Field) -> u8 {
    match field {
        Field::Minute | Field::Hour | Field::DayOfWeek => 0,
        Field::DayOfMonth | Field::Month => 1,
    }
}

fn domain_max(field: Field) -> u8 {
    match field {
        Field::Minute => 59,
        Field::Hour => 23,
        Field::DayOfMonth => 31,
        Field::Month => 12,
        Field::DayOfWeek => 6,
    }
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let field_order = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut fields = [
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
    ];
    let mut day_stars = [false; 2];

    for (i, field) in field_order.iter().copied().enumerate() {
        let text = parts[i];
        let bytes = text.as_bytes();

        for (offset, &byte) in bytes.iter().enumerate() {
            if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        let min = domain_min(field) as u32;
        let max = domain_max(field) as u32;
        let width = max - min + 1;
        let mut selected = vec![false; width as usize];

        let mut item_start = 0;
        for (item, next_start) in text
            .split(',')
            .zip(text.match_indices(',').map(|(idx, _)| idx + 1).chain(std::iter::once(text.len())))
        {
            if item.is_empty() {
                return Err(ParseError::EmptyItem {
                    field,
                    offset: item_start,
                });
            }

            if (field == Field::DayOfMonth || field == Field::DayOfWeek)
                && item_start == 0
                && item == "*"
            {
                day_stars[if field == Field::DayOfMonth { 0 } else { 1 }] = true;
            }

            parse_item(item, item_start, field, min, max, width, &mut selected)?;
            item_start = next_start;
        }

        fields[i] = selected;
    }

    Ok(Cron { fields, day_stars })
}

fn parse_item(
    item: &str,
    item_offset: usize,
    field: Field,
    min: u32,
    max: u32,
    width: u32,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
        .collect();

    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (base, step_text, step_offset) = if let Some(&slash) = slash_positions.first() {
        (&item[..slash], Some(&item[slash + 1..]), Some(item_offset + slash + 1))
    } else {
        (item, None, None)
    };

    let base_bytes = base.as_bytes();
    let valid_base = if base == "*" {
        true
    } else {
        let hyphens: Vec<usize> = base_bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
            .collect();

        match hyphens.as_slice() {
            [] => !base.is_empty() && base_bytes.iter().all(u8::is_ascii_digit),
            [h] => {
                *h > 0
                    && *h + 1 < base_bytes.len()
                    && base_bytes[..*h].iter().all(u8::is_ascii_digit)
                    && base_bytes[*h + 1..].iter().all(u8::is_ascii_digit)
            }
            _ => false,
        }
    };

    if !valid_base || (step_text.is_some() && step_text.unwrap().is_empty()) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let hyphen = base_bytes.iter().position(|&b| b == b'-');
    let (start, end, start_offset, end_offset) = if base == "*" {
        (min, max, item_offset, item_offset)
    } else if let Some(h) = hyphen {
        let start_offset = item_offset;
        let end_offset = item_offset + h + 1;
        let start = parse_number(&base[..h], field, start_offset, false)?;
        let end = parse_number(&base[h + 1..], field, end_offset, false)?;
        validate_domain(start, field, start_offset)?;
        validate_domain(end, field, end_offset)?;
        if start > end {
            return Err(ParseError::ReversedRange {
                field,
                offset: end_offset,
            });
        }
        (start, end, start_offset, end_offset)
    } else {
        let start = parse_number(base, field, item_offset, false)?;
        validate_domain(start, field, item_offset)?;
        (start, if step_text.is_some() { max } else { start }, item_offset, item_offset)
    };

    let step = if let (Some(step_text), Some(step_offset)) = (step_text, step_offset) {
        let value = parse_number(step_text, field, step_offset, true)?;
        if value < 1 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: step_offset,
            });
        }
        value
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected[(value - min) as usize] = true;
        match value.checked_add(step) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }

    let _ = (start_offset, end_offset);
    Ok(())
}

fn parse_number(
    text: &str,
    field: Field,
    offset: usize,
    is_step: bool,
) -> Result<u32, ParseError> {
    let bytes = text.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value: u32 = 0;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    if is_step {
        Ok(value)
    } else {
        Ok(value)
    }
}

fn validate_domain(value: u32, field: Field, offset: usize) -> Result<(), ParseError> {
    if value < domain_min(field) as u32 || value > domain_max(field) as u32 {
        Err(ParseError::OutOfRange { field, offset })
    } else {
        Ok(())
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn c(s: &str) -> Cron {
        parse_cron(s).unwrap()
    }

    #[test]
    fn p_lists_and_ranges() {
        let q = c("4,10-12,4 8 * * *");
        assert_eq!(q.allowed(Field::Minute), vec![4, 10, 11, 12]);
        assert_eq!(q.allowed(Field::Hour), vec![8]);
    }

    #[test]
    fn p_star_steps() {
        let q = c("*/20 0 * */3 *");
        assert_eq!(q.allowed(Field::Minute), vec![0, 20, 40]);
        assert_eq!(q.allowed(Field::Month), vec![1, 4, 7, 10]);
    }

    #[test]
    fn p_parse_error() {
        assert_eq!(
            parse_cron("0 0 * *"),
            Err(ParseError::FieldCount { found: 4 })
        );
        assert_eq!(
            parse_cron("0 0 32 * *"),
            Err(ParseError::OutOfRange {
                field: Field::DayOfMonth,
                offset: 0
            })
        );
    }
}
