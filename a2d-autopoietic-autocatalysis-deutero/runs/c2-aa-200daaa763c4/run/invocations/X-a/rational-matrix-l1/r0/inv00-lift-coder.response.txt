```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_i128_fraction(mut num: i128, mut den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if den < 0 {
        num = -num;
        den = -den;
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
    num /= g;
    den /= g;

    if num < i64::MIN as i128 || num > i64::MAX as i128 || den < 1 || den > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: num as i64,
        den: den as i64,
    })
}

fn from_u128_signed(num: u128, negative: bool, den: u128) -> Result<Rational, RationalError> {
    let g = gcd_u128(num, den);
    let n = num / g;
    let d = den / g;

    if d > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }
    if negative {
        let limit = 1u128 << 63;
        if n > limit {
            return Err(RationalError::Overflow);
        }
        let value = if n == limit {
            i64::MIN
        } else {
            -(n as i64)
        };
        Ok(Rational {
            num: value,
            den: d as i64,
        })
    } else {
        if n > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        Ok(Rational {
            num: n as i64,
            den: d as i64,
        })
    }
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let mut n = num as i128;
        let mut d = den as i128;
        if d < 0 {
            n = -n;
            d = -d;
        }
        from_i128_fraction(n, d)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let left = rhs.den as i128 / g;
        let right = self.den as i128 / g;
        let num = self.num as i128 * left + rhs.num as i128 * right;
        let den = self.den as i128 * left;
        from_i128_fraction(num, den)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let left = rhs.den as i128 / g;
        let right = self.den as i128 / g;
        let num = self.num as i128 * left - rhs.num as i128 * right;
        let den = self.den as i128 * left;
        from_i128_fraction(num, den)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.den as u128);
        let g2 = gcd_u128(rhs.num.unsigned_abs() as u128, self.den as u128);

        let a = (self.num as i128) / g1 as i128;
        let b = (rhs.num as i128) / g2 as i128;
        let c = (self.den as i128) / g2 as i128;
        let d = (rhs.den as i128) / g1 as i128;

        from_i128_fraction(a * b, c * d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }

        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.num.unsigned_abs() as u128);
        let g2 = gcd_u128(self.den as u128, rhs.den as u128);

        let mut a = (self.num as i128) / g1 as i128;
        let mut b = (rhs.den as i128) / g2 as i128;
        let c = (self.den as i128) / g2 as i128;
        let mut d = (rhs.num as i128) / g1 as i128;

        if d < 0 {
            a = -a;
            d = -d;
        }
        from_i128_fraction(a * b, c * d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Rational {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i += 1;
        }

        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        let num_end = i;

        let mut slash = None;
        let mut den_start = 0;
        let mut den_end = 0;

        if i < bytes.len() {
            if bytes[i] == b'/' {
                slash = Some(i);
                i += 1;
                den_start = i;
                while i < bytes.len() && bytes[i].is_ascii_digit() {
                    i += 1;
                }
                den_end = i;
            } else {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        if num_start == num_end {
            return Err(RationalError::Parse {
                at: i,
                kind: if i == bytes.len() {
                    ParseKind::InvalidByte
                } else {
                    ParseKind::InvalidByte
                },
            });
        }

        if let Some(slash_at) = slash {
            if den_start == den_end {
                return Err(RationalError::Parse {
                    at: bytes.len(),
                    kind: ParseKind::InvalidByte,
                });
            }
            if i < bytes.len() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
            let _ = slash_at;
        }

        if negative && num_end == 1 {
            return Err(RationalError::Parse {
                at: 1,
                kind: ParseKind::InvalidByte,
            });
        }
        if !negative && num_end == 0 {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::InvalidByte,
            });
        }

        if num_end - num_start >= 2 && bytes[num_start] == b'0' {
            return Err(RationalError::Parse {
                at: num_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }
        if negative && &text[..num_end] == "-0" {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::NonCanonical,
            });
        }
        if slash.is_some() && den_end - den_start >= 2 && bytes[den_start] == b'0' {
            return Err(RationalError::Parse {
                at: den_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }

        let num_mag = parse_magnitude(&bytes[num_start..num_end]).ok_or(RationalError::Parse {
            at: 0,
            kind: ParseKind::OutOfRange,
        })?;
        let num_limit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        if num_mag > num_limit {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::OutOfRange,
            });
        }

        let den_mag = if slash.is_some() {
            let mag = parse_magnitude(&bytes[den_start..den_end]).ok_or(RationalError::Parse {
                at: den_start,
                kind: ParseKind::OutOfRange,
            })?;
            if mag > i64::MAX as u128 {
                return Err(RationalError::Parse {
                    at: den_start,
                    kind: ParseKind::OutOfRange,
                });
            }
            if mag == 0 {
                return Err(RationalError::Parse {
                    at: den_start,
                    kind: ParseKind::ZeroDenominator,
                });
            }
            mag
        } else {
            1
        };

        if let Some(slash_at) = slash {
            if den_mag == 1 {
                return Err(RationalError::Parse {
                    at: slash_at,
                    kind: ParseKind::NonCanonical,
                });
            }
            if num_mag == 0 {
                return Err(RationalError::Parse {
                    at: slash_at,
                    kind: ParseKind::NonCanonical,
                });
            }
            if gcd_u128(num_mag, den_mag) > 1 {
                return Err(RationalError::Parse {
                    at: slash_at,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        from_u128_signed(num_mag, negative, den_mag)
    }
}

fn parse_magnitude(bytes: &[u8]) -> Option<u128> {
    let mut value = 0u128;
    for &b in bytes {
        let digit = (b - b'0') as u128;
        value = value.checked_mul(10)?.checked_add(digit)?;
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
