#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    idle_since: u64,
    occupied_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|s| s.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index].iter().any(|c| c.id == customer.id) {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut ordered_customers: Vec<Customer> = customers.to_vec();
    ordered_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|s| ServerState {
            id: s.id,
            opens: s.opens,
            overhead: s.overhead,
            opened: false,
            idle_since: s.opens,
            occupied_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_customer = 0usize;
    let mut now = 0u64;

    loop {
        let mut finishes: Vec<(u32, usize)> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| {
                (s.occupied_until == Some(now)).then_some((s.id, i))
            })
            .collect();
        finishes.sort_by_key(|&(id, _)| id);

        for (_, i) in finishes {
            let server_id = states[i].id;
            let customer_id = match events.last() {
                Some(Event::Start {
                    customer, server, ..
                }) if *server == server_id => *customer,
                _ => find_finishing_customer(&events, server_id, now),
            };
            events.push(Event::Finish {
                time: now,
                customer: customer_id,
                server: server_id,
            });
            states[i].occupied_until = None;
            states[i].idle_since = now;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (!s.opened && s.opens == now).then_some(i))
            .collect();
        opening.sort_by(|&a, &b| states[b].id.cmp(&states[a].id));
        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time: now,
                server: states[i].id,
            });
        }

        let mut arriving = Vec::new();
        while next_customer < ordered_customers.len()
            && ordered_customers[next_customer].arrival == now
        {
            arriving.push(next_customer);
            next_customer += 1;
        }
        for &idx in &arriving {
            events.push(Event::Arrive {
                time: now,
                customer: ordered_customers[idx].id,
            });
            waiting.push(idx);
        }

        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, s)| s.opened && s.occupied_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };
            let customer_index = waiting.remove(0);
            let customer = &ordered_customers[customer_index];
            let server = &states[server_index];

            let finish = customer
                .service
                .checked_add(now)
                .and_then(|t| t.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                })?;

            let customer_id = customer.id;
            let server_id = server.id;
            events.push(Event::Start {
                time: now,
                customer: customer_id,
                server: server_id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer_id,
                    server: server_id,
                });
                states[server_index].idle_since = now;
                states[server_index].occupied_until = None;
            } else {
                states[server_index].occupied_until = Some(finish);
            }
        }

        let all_done = next_customer == ordered_customers.len()
            && waiting.is_empty()
            && states.iter().all(|s| s.occupied_until.is_none());
        let all_open = states.iter().all(|s| s.opened);
        if all_done && all_open {
            break;
        }

        let next_arrival = ordered_customers
            .get(next_customer)
            .map(|c| c.arrival)
            .filter(|&t| t > now);
        let next_open = states
            .iter()
            .filter(|s| !s.opened && s.opens > now)
            .map(|s| s.opens)
            .min();
        let next_finish = states
            .iter()
            .filter_map(|s| s.occupied_until)
            .filter(|&t| t > now)
            .min();

        now = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .min()
            .expect("simulation has a future event");
    }

    Ok(events)
}

fn find_finishing_customer(events: &[Event], server_id: u32, time: u64) -> u32 {
    events
        .iter()
        .rev()
        .find_map(|event| match event {
            Event::Start {
                time: start_time,
                customer,
                server,
                finish,
            } if *server == server_id && *finish == time && *start_time <= time => {
                let already_finished = events.iter().any(|e| {
                    matches!(
                        e,
                        Event::Finish {
                            customer: finished_customer,
                            server: finished_server,
                            time: finish_time
                        } if *finished_customer == *customer
                            && *finished_server == *server
                            && *finish_time == time
                    )
                });
                (!already_finished).then_some(*customer)
            }
            _ => None,
        })
        .expect("every occupied server has a matching start")
}
