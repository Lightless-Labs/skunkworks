use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let mut n = num as i128;
        let mut d = den as i128;
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Rational { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Rational {
            num: n as i64,
            den: d as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Rational {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut i = usize::from(bytes[0] == b'-');
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(err(i, ParseKind::InvalidByte));
        }

        let mut slash = None;
        let mut den_start = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(err(i, ParseKind::InvalidByte));
            }
            slash = Some(i);
            i += 1;
            den_start = Some(i);
            if i == bytes.len() {
                return Err(err(i, ParseKind::InvalidByte));
            }
            if !bytes[i].is_ascii_digit() {
                return Err(err(i, ParseKind::InvalidByte));
            }
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
        }
        if i < bytes.len() {
            return Err(err(i, ParseKind::InvalidByte));
        }

        let num_end = slash.unwrap_or(bytes.len());
        let num_digits = &bytes[num_start..num_end];
        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(err(num_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && num_digits == b"0" {
            return Err(err(0, ParseKind::NonCanonical));
        }

        let num_mag = parse_u128_digits(num_digits).unwrap_or(u128::MAX);
        let negative = bytes[0] == b'-';
        let num_limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if num_mag > num_limit {
            return Err(err(0, ParseKind::OutOfRange));
        }

        if let (Some(ds), Some(slash_at)) = (den_start, slash) {
            let den_digits = &bytes[ds..];
            if den_digits.len() >= 2 && den_digits[0] == b'0' {
                return Err(err(ds + 1, ParseKind::NonCanonical));
            }
            let den_mag = parse_u128_digits(den_digits).unwrap_or(u128::MAX);
            if den_mag > i64::MAX as u128 {
                return Err(err(ds, ParseKind::OutOfRange));
            }
            if den_mag == 0 {
                return Err(err(ds, ParseKind::ZeroDenominator));
            }
            if den_mag == 1 || num_mag == 0 {
                return Err(err(slash_at, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den_mag) > 1 {
                return Err(err(slash_at, ParseKind::NonCanonical));
            }

            let num = if negative {
                if num_mag == 1u128 << 63 {
                    i64::MIN
                } else {
                    -(num_mag as i64)
                }
            } else {
                num_mag as i64
            };
            return Ok(Rational {
                num,
                den: den_mag as i64,
            });
        }

        let num = if negative {
            if num_mag == 1u128 << 63 {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };
        Ok(Rational { num, den: 1 })
    }
}

fn parse_u128_digits(bytes: &[u8]) -> Option<u128> {
    let mut value = 0u128;
    for &b in bytes {
        value = value.checked_mul(10)?.checked_add((b - b'0') as u128)?;
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Matrix { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|x| x.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data.get(index).copied().ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Matrix {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let out_len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let zero = Rational { num: 0, den: 1 };
        let mut out = Vec::with_capacity(out_len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let term = self.data[r * self.cols + k]
                        .mul(&rhs.data[k * rhs.cols + c])
                        .map_err(|source| MatrixError::Arithmetic {
                            row: r,
                            col: c,
                            step: k,
                            source,
                        })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                out.push(acc);
            }
        }
        Ok(Matrix {
            rows: self.rows,
            cols: rhs.cols,
            data: out,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let mut work = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        eliminate(&mut work, self.cols, &mut pivots, &mut swaps)?;
        Ok(Reduction {
            matrix: work,
            pivots,
            swaps,
        })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };
        let one = Rational { num: 1, den: 1 };
        for r in 0..n {
            for c in 0..width {
                data.push(if c < n {
                    self.data[r * n + c]
                } else if c - n == r {
                    one
                } else {
                    zero
                });
            }
        }
        let mut work = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        eliminate(&mut work, n, &mut pivots, &mut swaps)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }
        let mut result = Vec::with_capacity(n.checked_mul(n).ok_or(MatrixError::ShapeOverflow)?);
        for r in 0..n {
            for c in 0..n {
                result.push(work.data[r * width + n + c]);
            }
        }
        Ok(Matrix {
            rows: n,
            cols: n,
            data: result,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        let m = self.rows;
        let n = self.cols;
        let k = rhs.cols;
        if m != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: m,
                rhs: rhs.rows,
            });
        }
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = m.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..m {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..k {
                data.push(rhs.data[r * k + c]);
            }
        }
        let mut work = Matrix {
            rows: m,
            cols: width,
            data,
        };
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        eliminate(&mut work, n, &mut pivots, &mut swaps).map_err(SolveError::Matrix)?;

        for r in 0..m {
            let coeff_zero = (0..n).all(|c| work.data[r * width + c].num == 0);
            if coeff_zero {
                for c in 0..k {
                    if work.data[r * width + n + c].num != 0 {
                        let witness = work.data[r * width..(r + 1) * width].to_vec();
                        return Ok(Solution::None {
                            row: r,
                            rhs_col: c,
                            witness,
                        });
                    }
                }
            }
        }

        let rank = pivots.len();
        let mut is_pivot = vec![false; n];
        for &p in &pivots {
            is_pivot[p] = true;
        }
        let free: Vec<usize> = (0..n).filter(|&c| !is_pivot[c]).collect();

        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let zero = Rational { num: 0, den: 1 };
        let mut particular_data = vec![zero; particular_len];
        for (j, &pcol) in pivots.iter().enumerate() {
            for c in 0..k {
                particular_data[pcol * k + c] = work.data[j * width + n + c];
            }
        }
        let particular = Matrix {
            rows: n,
            cols: k,
            data: particular_data,
        };

        if rank == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots,
            });
        }

        let f = free.len();
        let basis_len = n.checked_mul(f).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![zero; basis_len];
        let one = Rational { num: 1, den: 1 };
        for (j, &free_col) in free.iter().enumerate() {
            basis[free_col * f + j] = one;
            for (i, &pivot_col) in pivots.iter().enumerate() {
                let coeff = work.data[i * width + free_col];
                let neg = coeff.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col,
                        col: j,
                        step: rank,
                        source,
                    })
                })?;
                basis[pivot_col * f + j] = neg;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix {
                rows: n,
                cols: f,
                data: basis,
            },
            pivots,
        })
    }
}

fn pivot_key(value: Rational, row: usize) -> (i64, u64, usize) {
    (value.den, value.num.unsigned_abs(), row)
}

fn eliminate(
    work: &mut Matrix,
    candidate_cols: usize,
    pivots: &mut Vec<usize>,
    swaps: &mut Vec<(usize, usize)>,
) -> Result<(), MatrixError> {
    let mut p = 0usize;
    for col in 0..candidate_cols {
        if p == work.rows {
            break;
        }
        let mut chosen: Option<(usize, (i64, u64, usize))> = None;
        for row in p..work.rows {
            let value = work.data[row * work.cols + col];
            if value.num == 0 {
                continue;
            }
            let key = pivot_key(value, row);
            if chosen.map_or(true, |(_, best)| key < best) {
                chosen = Some((row, key));
            }
        }
        let Some((q, _)) = chosen else {
            continue;
        };
        if q != p {
            for c in 0..work.cols {
                work.data.swap(p * work.cols + c, q * work.cols + c);
            }
            swaps.push((p, q));
        }
        let pivot = work.data[p * work.cols + col];
        let mut normalized = Vec::with_capacity(work.cols);
        for c in 0..work.cols {
            let value = work.data[p * work.cols + c];
            normalized.push(value.div(&pivot).map_err(|source| MatrixError::Arithmetic {
                row: p,
                col: c,
                step: p,
                source,
            })?);
        }
        for c in 0..work.cols {
            work.data[p * work.cols + c] = normalized[c];
        }
        for row in 0..work.rows {
            if row == p {
                continue;
            }
            let factor = work.data[row * work.cols + col];
            if factor.num == 0 {
                continue;
            }
            for c in 0..work.cols {
                let product = factor.mul(&normalized[c]).map_err(|source| {
                    MatrixError::Arithmetic {
                        row,
                        col: c,
                        step: p,
                        source,
                    }
                })?;
                let updated = work.data[row * work.cols + c]
                    .sub(&product)
                    .map_err(|source| MatrixError::Arithmetic {
                        row,
                        col: c,
                        step: p,
                        source,
                    })?;
                work.data[row * work.cols + c] = updated;
            }
        }
        pivots.push(col);
        p += 1;
    }
    Ok(())
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}
