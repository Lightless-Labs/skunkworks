#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let mut payload_start = 0;
    if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &input[..hash] != expected {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = hash + 1;
        if payload_start == bytes.len() {
            return Err(BaseError::Empty { operand });
        }
    }

    let payload = &bytes[payload_start..];

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        if !matches!(digit_value(byte), Some(value) if value < radix) {
            return Err(BaseError::InvalidDigit {
                operand,
                index: payload_start + offset,
            });
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let has_digit_before = offset > 0 && payload[offset - 1] != b'_';
            let has_digit_after = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !has_digit_before || !has_digit_after {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    let digits: Vec<u8> = payload
        .iter()
        .filter(|&&byte| byte != b'_')
        .map(|&byte| digit_value(byte).unwrap())
        .collect();

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, radix: u8, digit: u8) {
    let mut carry = digit as u32;
    for limb in value.iter_mut() {
        let n = *limb as u32 * radix as u32 + carry;
        *limb = (n & 0xffff) as u16;
        carry = n >> 16;
    }
    while carry != 0 {
        value.push((carry & 0xffff) as u16);
        carry >>= 16;
    }
}

fn parse_value(digits: &[u8], radix: u8) -> Vec<u16> {
    let mut value = vec![0u16];
    for &digit in digits {
        multiply_add(&mut value, radix, digit);
    }
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    value
}

fn add_values(left: &mut Vec<u16>, right: &[u16]) {
    let length = left.len().max(right.len());
    left.resize(length, 0);
    let mut carry = 0u32;

    for i in 0..length {
        let sum = left[i] as u32 + right.get(i).copied().unwrap_or(0) as u32 + carry;
        left[i] = (sum & 0xffff) as u16;
        carry = sum >> 16;
    }
    if carry != 0 {
        left.push(carry as u16);
    }
}

fn divide_value(value: &mut Vec<u16>, divisor: u8) -> u8 {
    let mut remainder = 0u32;
    for limb in value.iter_mut().rev() {
        let current = (remainder << 16) | *limb as u32;
        *limb = (current / divisor as u32) as u16;
        remainder = current % divisor as u32;
    }
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    remainder as u8
}

fn format_value(mut value: Vec<u16>, radix: u8, style: BaseStyle) -> String {
    let mut digits = Vec::new();

    loop {
        let remainder = divide_value(&mut value, radix);
        let character = if remainder < 10 {
            b'0' + remainder
        } else if style.upper {
            b'A' + remainder - 10
        } else {
            b'a' + remainder - 10
        };
        digits.push(character);
        if value.len() == 1 && value[0] == 0 {
            break;
        }
    }

    while digits.len() < style.min_width {
        digits.push(b'0');
    }
    digits.reverse();

    if style.group == 0 {
        return String::from_utf8(digits).unwrap();
    }

    let mut output = Vec::with_capacity(digits.len() + digits.len() / style.group);
    for (index, &digit) in digits.iter().enumerate() {
        if index > 0 && (digits.len() - index) % style.group == 0 {
            output.push(b'_');
        }
        output.push(digit);
    }
    String::from_utf8(output).unwrap()
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;

    let digits = parse_operand(input, from, 0)?;
    let value = parse_value(&digits, from);
    Ok(format_value(value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let mut value = parse_value(&left_digits, base);
    let right_value = parse_value(&right_digits, base);
    add_values(&mut value, &right_value);

    Ok(format_value(value, base, style))
}
