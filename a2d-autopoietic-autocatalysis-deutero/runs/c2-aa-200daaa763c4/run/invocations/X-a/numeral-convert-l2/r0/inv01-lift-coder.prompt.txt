pub struct BaseStyle {

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, base: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let mut payload_start = 0;
    if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = base.to_string();
        if &bytes[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = hash + 1;
    }

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (i, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < base => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + i,
                })
            }
        }
    }

    for (i, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (i == 0
                || i + 1 == payload.len()
                || payload[i - 1] == b'_'
                || payload[i + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + i,
            });
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, base: u8, digit: u8) {
    let mut carry = digit as u16;
    for limb in value.iter_mut() {
        let n = *limb as u16 * base as u16 + carry;
        *limb = (n % 1_000_000_000) as u32;
        carry = n / 1_000_000_000;
    }
    while carry != 0 {
        value.push((carry % 1_000_000_000) as u32);
        carry /= 1_000_000_000;
    }
}

fn parse_to_big(digits: &[u8], base: u8) -> Vec<u32> {
    let mut value = vec![0u32];
    for &digit in digits {
        multiply_add(&mut value, base, digit);
    }
    value
}

fn add_big(left: &[u32], right: &[u32]) -> Vec<u32> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u64;
    for i in 0..left.len().max(right.len()) {
        let sum = left.get(i).copied().unwrap_or(0) as u64
            + right.get(i).copied().unwrap_or(0) as u64
            + carry;
        result.push((sum % 1_000_000_000) as u32);
        carry = sum / 1_000_000_000;
    }
    if carry != 0 {
        result.push(carry as u32);
    }
    result
}

fn divide_big(value: &mut Vec<u32>, divisor: u8) -> u8 {
    let mut remainder = 0u64;
    for limb in value.iter_mut().rev() {
        let current = remainder * 1_000_000_000 + *limb as u64;
        *limb = (current / divisor as u64) as u32;
        remainder = current % divisor as u64;
    }
    while value.len() > 1 && value.last() == Some(&0) {
        value.pop();
    }
    remainder as u8
}

fn format_big(mut value: Vec<u32>, base: u8, style: BaseStyle) -> String {
    let mut digits = Vec::new();
    if value.iter().all(|&limb| limb == 0) {
        digits.push(0);
    } else {
        while value.iter().any(|&limb| limb != 0) {
            digits.push(divide_big(&mut value, base));
        }
        digits.reverse();
    }

    let pad = style.min_width.saturating_sub(digits.len());
    let mut output = String::with_capacity(digits.len() + pad + digits.len() / style.group.max(1));
    for _ in 0..pad {
        output.push('0');
    }

    let total = pad + digits.len();
    for position in 0..total {
        if style.group != 0 && position != 0 && (total - position) % style.group == 0 {
            output.push('_');
        }

        if position < pad {
            continue;
        }

        let digit = digits[position - pad];
        let ch = match digit {
            0..=9 => (b'0' + digit) as char,
            _ if style.upper => (b'A' + digit - 10) as char,
            _ => (b'a' + digit - 10) as char,
        };
        output.push(ch);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let value = parse_to_big(&digits, from);
    Ok(format_big(value, to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;
    let left_value = parse_to_big(&left_digits, base);
    let right_value = parse_to_big(&right_digits, base);
    Ok(format_big(add_big(&left_value, &right_value), base, style))
}
