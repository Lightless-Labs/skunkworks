```rust
use std::collections::VecDeque;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    Ticks { index: usize },
}

struct PendingJob {
    id: u32,
    remaining: u32,
    level: usize,
}

pub fn schedule(
    jobs: &[Job],
    quanta: &[u32],
    switch_cost: u32,
) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut arrival_order: Vec<usize> = (0..jobs.len()).collect();
    arrival_order.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut queues: Vec<VecDeque<PendingJob>> =
        (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut next_arrival = 0;
    let mut completed: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut last_dispatched_id: Option<u32> = None;
    let mut completed_count = 0usize;

    while completed_count < jobs.len() {
        while next_arrival < arrival_order.len()
            && u64::from(jobs[arrival_order[next_arrival]].arrival) <= time
        {
            let index = arrival_order[next_arrival];
            let job = &jobs[index];
            if job.ticks == 0 {
                completed[index] = Some(u64::from(job.arrival));
                completed_count += 1;
            } else {
                queues[job.level].push_back(PendingJob {
                    id: job.id,
                    remaining: job.ticks,
                    level: job.level,
                });
            }
            next_arrival += 1;
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());

        let Some(level) = selected_level else {
            if next_arrival >= arrival_order.len() {
                break;
            }

            let next_time = u64::from(jobs[arrival_order[next_arrival]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched_id = None;
            }
            continue;
        };

        let mut selected = queues[level]
            .pop_front()
            .expect("selected queue must contain a job");

        if last_dispatched_id.is_some_and(|id| id != selected.id) {
            time += u64::from(switch_cost);
        }

        while next_arrival < arrival_order.len()
            && u64::from(jobs[arrival_order[next_arrival]].arrival) <= time
        {
            let index = arrival_order[next_arrival];
            let job = &jobs[index];
            if job.ticks == 0 {
                completed[index] = Some(u64::from(job.arrival));
                completed_count += 1;
            } else {
                queues[job.level].push_back(PendingJob {
                    id: job.id,
                    remaining: job.ticks,
                    level: job.level,
                });
            }
            next_arrival += 1;
        }

        let start = time;
        let consumed = selected.remaining.min(quanta[level]);
        time += u64::from(consumed);
        selected.remaining -= consumed;
        let finished = selected.remaining == 0;

        runs.push(Run {
            id: selected.id,
            start,
            end: time,
            finished,
            level,
        });

        while next_arrival < arrival_order.len()
            && u64::from(jobs[arrival_order[next_arrival]].arrival) <= time
        {
            let index = arrival_order[next_arrival];
            let job = &jobs[index];
            if job.ticks == 0 {
                completed[index] = Some(u64::from(job.arrival));
                completed_count += 1;
            } else {
                queues[job.level].push_back(PendingJob {
                    id: job.id,
                    remaining: job.ticks,
                    level: job.level,
                });
            }
            next_arrival += 1;
        }

        if finished {
            let index = jobs
                .iter()
                .position(|job| job.id == selected.id)
                .expect("selected job must come from input");
            completed[index] = Some(time);
            completed_count += 1;
        } else {
            selected.level = (level + 1).min(quanta.len() - 1);
            queues[selected.level].push_back(selected);
        }

        last_dispatched_id = Some(runs.last().expect("run was just added").id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| {
            (
                job.id,
                completed[index].expect("every job must have completed"),
            )
        })
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
```
