#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_records(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut position = if input.starts_with('\u{feff}') {
        '\u{feff}'.len_utf8()
    } else {
        0
    };

    if position == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = Vec::new();
    let mut record_index = 0usize;

    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::FieldStart;

    while position < bytes.len() {
        let byte = bytes[position];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: position,
            });
        }

        let outside_quotes = !matches!(state, State::Quoted { .. });
        if outside_quotes && byte == b'\r' {
            if bytes.get(position + 1) == Some(&b'\n') {
                match state {
                    State::AfterQuote | State::FieldStart | State::Unquoted => {}
                    State::Quoted { .. } => unreachable!(),
                }
                fields.push(String::from_utf8(field).expect("valid UTF-8 field"));
                records.push(std::mem::take(&mut fields));
                field.clear();
                position += 2;
                record_index += 1;
                state = State::FieldStart;
                continue;
            }
            return Err(CsvError::BareCr {
                record: record_index,
                byte: position,
            });
        }

        match state {
            State::FieldStart => {
                if byte == quote {
                    state = State::Quoted { opening: position };
                    position += 1;
                } else if byte == delimiter {
                    fields.push(String::new());
                    position += 1;
                } else if byte == b'\n' {
                    fields.push(String::new());
                    records.push(std::mem::take(&mut fields));
                    position += 1;
                    record_index += 1;
                    state = State::FieldStart;
                } else {
                    field.push(byte);
                    state = State::Unquoted;
                    position += 1;
                }
            }
            State::Unquoted => {
                if byte == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: position,
                    });
                } else if byte == delimiter {
                    fields.push(String::from_utf8(std::mem::take(&mut field))
                        .expect("valid UTF-8 field"));
                    position += 1;
                    state = State::FieldStart;
                } else if byte == b'\n' {
                    fields.push(String::from_utf8(std::mem::take(&mut field))
                        .expect("valid UTF-8 field"));
                    records.push(std::mem::take(&mut fields));
                    position += 1;
                    record_index += 1;
                    state = State::FieldStart;
                } else {
                    field.push(byte);
                    position += 1;
                }
            }
            State::Quoted { opening } => {
                if byte == quote {
                    if bytes.get(position + 1) == Some(&quote) {
                        field.push(quote);
                        position += 2;
                    } else {
                        state = State::AfterQuote;
                        position += 1;
                    }
                } else {
                    field.push(byte);
                    position += 1;
                }
                let _ = opening;
            }
            State::AfterQuote => {
                if byte == delimiter {
                    fields.push(String::from_utf8(std::mem::take(&mut field))
                        .expect("valid UTF-8 field"));
                    position += 1;
                    state = State::FieldStart;
                } else if byte == b'\n' {
                    fields.push(String::from_utf8(std::mem::take(&mut field))
                        .expect("valid UTF-8 field"));
                    records.push(std::mem::take(&mut fields));
                    position += 1;
                    record_index += 1;
                    state = State::FieldStart;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: position,
                    });
                }
            }
        }
    }

    match state {
        State::Quoted { opening } => {
            return Err(CsvError::UnclosedQuote {
                record: record_index,
                byte: opening,
            });
        }
        State::FieldStart => {
            if !fields.is_empty() || position > 0 {
                fields.push(String::new());
                records.push(fields);
            }
        }
        State::Unquoted | State::AfterQuote => {
            fields.push(String::from_utf8(field).expect("valid UTF-8 field"));
            records.push(fields);
        }
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_records(input, b',', b'"')
}

fn parse_table(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let mut records = parse_records(input, config.delimiter, config.quote)?;

    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(names) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            for second in 0..names.len() {
                if let Some(first) = names[..second].iter().position(|name| name == &names[second]) {
                    return Err(CsvError::DuplicateHeader {
                        name: names[second].clone(),
                        first,
                        second,
                    });
                }
            }
        }

        for (data_index, row) in records.iter().enumerate() {
            if row.len() != names.len() {
                return Err(CsvError::HeaderWidth {
                    record: data_index + 1,
                    expected: names.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: records,
        duplicates: config.duplicates,
    })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_table(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let digits = if bytes[0] == b'-' { &bytes[1..] } else { bytes };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return bytes[0] != b'-';
    }
    digits[0].is_ascii_digit()
        && digits[0] != b'0'
        && digits.iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    let (negative, body) = if let Some(rest) = bytes.strip_prefix(b"-") {
        (true, rest)
    } else {
        (false, bytes)
    };
    let Some(dot) = body.iter().position(|&byte| byte == b'.') else {
        return false;
    };
    let whole = &body[..dot];
    let fraction = &body[dot + 1..];
    if fraction.len() != 2 || !fraction.iter().all(u8::is_ascii_digit) || whole.is_empty() {
        return false;
    }
    let whole_valid = whole == b"0"
        || (whole[0].is_ascii_digit()
            && whole[0] != b'0'
            && whole.iter().all(u8::is_ascii_digit));
    if !whole_valid {
        return false;
    }
    !(negative && whole == b"0" && fraction == b"00")
}

fn parse_integer(value: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(value) {
        return Err(CellErrorKind::InvalidInteger);
    }
    value.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(value: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(value) {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let negative = value.starts_with('-');
    let body = if negative { &value[1..] } else { value };
    let dot = body.find('.').expect("validated decimal");
    let whole: i128 = body[..dot].parse().expect("validated digits");
    let fraction: i128 = body[dot + 1..].parse().expect("validated digits");
    let magnitude = whole * 100 + fraction;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate_cell(
    cell: &str,
    kind: ColumnType,
    nullable: bool,
) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if nullable {
            return Ok(Value::Null);
        }
        if kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut output_rows = Vec::new();
    let mut errors = Vec::new();

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];

        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let column = match resolved {
                Some(column) => column,
                None => {
                    errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: None,
                        kind: CellErrorKind::UnknownColumn,
                    });
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                    continue;
                }
            };

            let Some(cell) = row.get(column) else {
                errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column),
                    kind: CellErrorKind::MissingCell,
                });
                if policy != ErrorPolicy::Collect {
                    break;
                }
                continue;
            };

            match evaluate_cell(cell, spec.kind, spec.nullable) {
                Ok(value) => output[spec_index] = Some(value),
                Err(kind) => {
                    errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column),
                        kind,
                    });
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                }
            }
        }

        output_rows.push(output);
        if policy == ErrorPolicy::Stop && !errors.is_empty() {
            break;
        }
    }

    TypedResult {
        rows: output_rows,
        errors,
    }
}
