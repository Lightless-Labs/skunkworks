#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum ParseState {
    FieldStart,
    Unquoted,
    Quoted,
    AfterQuote,
}

fn parse_records(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let offset = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if offset == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = ParseState::FieldStart;
    let mut quote_start = 0usize;
    let mut record_index = 0usize;
    let mut i = offset;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        let inside_quote = state == ParseState::Quoted;
        if !inside_quote && byte == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        match state {
            ParseState::FieldStart => {
                if byte == quote {
                    quote_start = i;
                    state = ParseState::Quoted;
                    i += 1;
                } else if byte == delimiter {
                    record.push(String::new());
                    i += 1;
                } else if byte == b'\n' {
                    record.push(String::new());
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    i += 1;
                } else if byte == b'\r' {
                    record.push(String::new());
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    i += 2;
                } else {
                    let start = i;
                    while i < bytes.len()
                        && bytes[i] != 0
                        && bytes[i] != delimiter
                        && bytes[i] != quote
                        && bytes[i] != b'\n'
                        && bytes[i] != b'\r'
                    {
                        i += 1;
                    }
                    field.push_str(&input[start..i]);
                    state = ParseState::Unquoted;
                }
            }
            ParseState::Unquoted => {
                if byte == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else if byte == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = ParseState::FieldStart;
                    i += 1;
                } else if byte == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = ParseState::FieldStart;
                    i += 1;
                } else if byte == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = ParseState::FieldStart;
                    i += 2;
                } else {
                    let start = i;
                    while i < bytes.len()
                        && bytes[i] != 0
                        && bytes[i] != delimiter
                        && bytes[i] != quote
                        && bytes[i] != b'\n'
                        && bytes[i] != b'\r'
                    {
                        i += 1;
                    }
                    field.push_str(&input[start..i]);
                }
            }
            ParseState::Quoted => {
                if byte == quote {
                    if bytes.get(i + 1) == Some(&quote) {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = ParseState::AfterQuote;
                        i += 1;
                    }
                } else {
                    let start = i;
                    while i < bytes.len() && bytes[i] != 0 && bytes[i] != quote {
                        i += 1;
                    }
                    field.push_str(&input[start..i]);
                }
            }
            ParseState::AfterQuote => {
                if byte == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = ParseState::FieldStart;
                    i += 1;
                } else if byte == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = ParseState::FieldStart;
                    i += 1;
                } else if byte == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = ParseState::FieldStart;
                    i += 2;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
    }

    if state == ParseState::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if state == ParseState::FieldStart {
        // A nonempty input ending after a delimiter still has a final empty field.
        if bytes.len() > offset && i > offset {
            let ended_with_terminator = i >= 1
                && (bytes[i - 1] == b'\n'
                    || (i >= 2 && bytes[i - 2] == b'\r' && bytes[i - 1] == b'\n'));
            if !ended_with_terminator {
                record.push(String::new());
                records.push(record);
            }
        }
    } else {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_records(input, b',', b'"')
}

fn records_to_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let mut header = None;
    let mut rows = records;

    if config.headers && !rows.is_empty() {
        let first = rows.remove(0);

        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: Vec<(String, usize)> = Vec::new();
            for (second, name) in first.iter().enumerate() {
                if let Some((_, first_index)) = seen.iter().find(|(prior, _)| prior == name) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: *first_index,
                        second,
                    });
                }
                seen.push((name.clone(), second));
            }
        }

        let expected = first.len();
        for (record_index, row) in rows.iter().enumerate() {
            if row.len() != expected {
                return Err(CsvError::HeaderWidth {
                    record: record_index + 1,
                    expected,
                    actual: row.len(),
                });
            }
        }

        header = Some(first);
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }

        let result = parse_records(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| records_to_table(records, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}
