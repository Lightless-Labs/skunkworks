#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|n| n == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|n| n == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: after quote
    let mut quote_start = 0usize;
    let mut record = 0usize;
    let mut active = false;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: pos });
        }

        if state != 2 && b == b'\r' {
            if bytes.get(pos + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr { record, byte: pos });
            }
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            pos += 2;
            record += 1;
            state = 0;
            active = false;
            continue;
        }

        if state != 2 && b == b'\n' {
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            pos += 1;
            record += 1;
            state = 0;
            active = false;
            continue;
        }

        match state {
            0 => {
                if b == delimiter {
                    fields.push(String::new());
                    active = true;
                    pos += 1;
                } else if b == quote {
                    state = 2;
                    quote_start = pos;
                    active = true;
                    pos += 1;
                } else {
                    let start = pos;
                    while pos < bytes.len()
                        && bytes[pos] != delimiter
                        && bytes[pos] != quote
                        && bytes[pos] != b'\r'
                        && bytes[pos] != b'\n'
                        && bytes[pos] != 0
                    {
                        pos += 1;
                    }
                    if pos < bytes.len() && bytes[pos] == quote {
                        return Err(CsvError::UnexpectedQuote { record, byte: pos });
                    }
                    field.push_str(&input[start..pos]);
                    state = 1;
                    active = true;
                }
            }
            1 => {
                if b == delimiter {
                    fields.push(std::mem::take(&mut field));
                    state = 0;
                    active = true;
                    pos += 1;
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote { record, byte: pos });
                } else {
                    let start = pos;
                    while pos < bytes.len()
                        && bytes[pos] != delimiter
                        && bytes[pos] != quote
                        && bytes[pos] != b'\r'
                        && bytes[pos] != b'\n'
                        && bytes[pos] != 0
                    {
                        pos += 1;
                    }
                    field.push_str(&input[start..pos]);
                }
            }
            2 => {
                if b == quote {
                    if bytes.get(pos + 1) == Some(&quote) {
                        field.push(quote as char);
                        pos += 2;
                    } else {
                        state = 3;
                        pos += 1;
                    }
                } else {
                    let start = pos;
                    while pos < bytes.len()
                        && bytes[pos] != quote
                        && bytes[pos] != 0
                    {
                        pos += 1;
                    }
                    field.push_str(&input[start..pos]);
                }
            }
            3 => {
                if b == delimiter {
                    fields.push(std::mem::take(&mut field));
                    state = 0;
                    active = true;
                    pos += 1;
                } else {
                    return Err(CsvError::TrailingByte { record, byte: pos });
                }
            }
            _ => unreachable!(),
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_start,
        });
    }

    if active || !fields.is_empty() || !field.is_empty() {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    for second in 0..header.len() {
        if let Some(first) = header[..second].iter().position(|n| n == &header[second]) {
            return Some(CsvError::DuplicateHeader {
                name: header[second].clone(),
                first,
                second,
            });
        }
    }
    None
}

fn table_from_input(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut records = parse_dialect(input, config.delimiter, config.quote)?;
    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(ref header) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header(header) {
                return Err(error);
            }
        }
        for (i, row) in records.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: records,
        duplicates: config.duplicates,
    })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = table_from_input(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| CsvError::InvalidConfig { component })?;
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

    MissingCell,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    let valid = if start == b.len() {
        false
    } else if b[start] == b'0' {
        start + 1 == b.len() && start == 0
    } else {
        (b'1'..=b'9').contains(&b[start])
            && b[start + 1..].iter().all(u8::is_ascii_digit)
    };
    if !valid {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    let b = s.as_bytes();
    let neg = b.first() == Some(&b'-');
    let start = usize::from(neg);
    if start >= b.len() {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let dot = match b[start..].iter().position(|&x| x == b'.') {
        Some(i) => start + i,
        None => return Err(CellErrorKind::InvalidDecimal),
    };
    let int = &b[start..dot];
    let frac = &b[dot + 1..];
    let valid_int = if int == b"0" {
        true
    } else {
        !int.is_empty()
            && (b'1'..=b'9').contains(&int[0])
            && int[1..].iter().all(u8::is_ascii_digit)
    };
    if !valid_int || frac.len() != 2 || !frac.iter().all(u8::is_ascii_digit) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    if neg && int == b"0" && frac == b"00" {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let mut magnitude: i128 = 0;
    for &digit in int.iter().chain(frac.iter()) {
        magnitude = magnitude * 10 + i128::from(digit - b'0');
    }
    let signed = if neg { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate_cell(
    table: &Table,
    row: &[String],
    row_index: usize,
    spec: &ColumnSpec,
    spec_index: usize,
) -> (Option<Value>, Option<CellError>) {
    let resolved = match &spec.column {
        ColumnRef::Index(i) => Some(*i),
        ColumnRef::Name(name) => table.column_index(name),
    };
    let Some(column) = resolved else {
        return (
            None,
            Some(CellError {
                row: row_index,
                spec: spec_index,
                column: None,
                kind: CellErrorKind::UnknownColumn,
            }),
        );
    };
    let Some(cell) = row.get(column) else {
        return (
            None,
            Some(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(column),
                kind: CellErrorKind::MissingCell,
            }),
        );
    };

    if cell.is_empty() {
        if spec.nullable {
            return (Some(Value::Null), None);
        }
        if spec.kind != ColumnType::Text {
            return (
                None,
                Some(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column),
                    kind: CellErrorKind::Empty,
                }),
            );
        }
    }

    let value = match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.clone())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    };

    match value {
        Ok(value) => (Some(value), None),
        Err(kind) => (
            None,
            Some(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(column),
                kind,
            }),
        ),
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if policy == ErrorPolicy::FirstPerRow && row_failed {
                break;
            }
            let (value, error) = evaluate_cell(table, row, row_index, spec, spec_index);
            output[spec_index] = value;
            if let Some(error) = error {
                result.errors.push(error);
                row_failed = true;
                if policy == ErrorPolicy::Stop {
                    result.rows.push(output);
                    return result;
                }
            }
        }

        result.rows.push(output);
    }

    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {

    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns { record: usize },
    DuplicateHeader {
        name: String,
        first: usize,
        second: usize,
    },
}

fn serialize_field(
    out: &mut String,
    field: &str,
    delimiter: u8,
    quote: u8,
    first_field_first_record: bool,
    sole_empty: bool,
) {
    let bytes = field.as_bytes();
    let quote_it = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
        || (first_field_first_record && field.starts_with('\u{feff}'))
        || (sole_empty && field.is_empty());

    if !quote_it {
        out.push_str(field);
        return;
    }
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == u32::from(quote) {
            out.push(quote as char);
            out.push(quote as char);
        } else {
            out.push(ch);
        }
    }
    out.push(quote as char);
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original: Vec<&[String]> = Vec::new();
    if let Some(header) = &table.header {
        original.push(header);
    }
    original.extend(table.rows.iter().map(Vec::as_slice));

    for (record, row) in original.iter().enumerate() {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in original.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                records.push(header.clone());
            }
            records.extend(table.rows.clone());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                if table.duplicates == DuplicatePolicy::Reject {
                    if let Some(CsvError::DuplicateHeader {
                        name,
                        first,
                        second,
                    }) = duplicate_header(header)
                    {
                        return Err(WriteError::DuplicateHeader {
                            name,
                            first,
                            second,
                        });
                    }
                }

                let mut retained: Vec<(String, usize)> = Vec::new();
                for (index, name) in header.iter().enumerate() {
                    if retained.iter().any(|(n, _)| n == name) {
                        continue;
                    }
                    let selected = match table.duplicates {
                        DuplicatePolicy::KeepLast => header
                            .iter()
                            .rposition(|n| n == name)
                            .unwrap_or(index),
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => index,
                    };
                    retained.push((name.clone(), selected));
                }
                retained.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
                let canonical_header: Vec<String> =
                    retained.iter().map(|(name, _)| name.clone()).collect();
                let mut projected: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| retained.iter().map(|(_, i)| row[*i].clone()).collect())
                    .collect();
                projected.sort_by(|a, b| {
                    for (x, y) in a.iter().zip(b.iter()) {
                        let order = x.as_bytes().cmp(y.as_bytes());
                        if order != std::cmp::Ordering::Equal {
                            return order;
                        }
                    }
                    a.len().cmp(&b.len())
                });
                records.push(canonical_header);
                records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| {
                    for (x, y) in a.iter().zip(b.iter()) {
                        let order = x.as_bytes().cmp(y.as_bytes());
                        if order != std::cmp::Ordering::Equal {
                            return order;
                        }
                    }
                    a.len().cmp(&b.len())
                });
                records = rows;
            }
        }
    }

    if records.is_empty() {
        return Ok(String::new());
    }

    let canonical = config.mode == WriteMode::Canonical;
    let ending = if canonical {
        "\n"
    } else {
        match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        }
    };

    let mut out = String::new();
    for (record_index, row) in records.iter().enumerate() {
        if record_index > 0 {
            out.push_str(ending);
        }
        for (column_index, field) in row.iter().enumerate() {
            if column_index > 0 {
                out.push(config.delimiter as char);
            }
            serialize_field(
                &mut out,
                field,
                config.delimiter,
                config.quote,
                record_index == 0 && column_index == 0,
                row.len() == 1 && field.is_empty(),
            );
        }
    }

    if canonical || config.final_terminator {
        out.push_str(ending);
    }
    Ok(out)
}
