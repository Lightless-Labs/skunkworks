#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart { Delimiter, Quote }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy { Reject, KeepFirst, KeepLast }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = Vec::new();
    let mut record_index = 0usize;
    let mut field_started = false;
    let mut quoted = false;
    let mut closed = false;
    let mut quote_start = 0usize;
    let mut record_started = false;

    while pos < bytes.len() {
        let b = bytes[pos];
        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: pos });
        }

        if quoted {
            if b == quote {
                if pos + 1 < bytes.len() && bytes[pos + 1] == quote {
                    field.push(quote);
                    pos += 2;
                    continue;
                } else {
                    quoted = false;
                    closed = true;
                    pos += 1;
                    continue;
                }
            }
            field.push(b);
            pos += 1;
            continue;
        }

        if b == b'\r' && (pos + 1 >= bytes.len() || bytes[pos + 1] != b'\n') {
            return Err(CsvError::BareCr { record: record_index, byte: pos });
        }

        if closed {
            if b == delimiter {
                row.push(String::from_utf8(field).unwrap());
                field = Vec::new();
                field_started = false;
                closed = false;
                record_started = true;
                pos += 1;
            } else if b == b'\n' || b == b'\r' {
                row.push(String::from_utf8(field).unwrap());
                field = Vec::new();
                records.push(row);
                row = Vec::new();
                field_started = false;
                closed = false;
                record_started = false;
                record_index += 1;
                pos += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record: record_index, byte: pos });
            }
            continue;
        }

        if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote { record: record_index, byte: pos });
            }
            quoted = true;
            field_started = true;
            quote_start = pos;
            record_started = true;
            pos += 1;
        } else if b == delimiter {
            row.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            field_started = false;
            record_started = true;
            pos += 1;
        } else if b == b'\n' || b == b'\r' {
            row.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            records.push(row);
            row = Vec::new();
            field_started = false;
            record_started = false;
            record_index += 1;
            pos += if b == b'\r' { 2 } else { 1 };
        } else {
            field.push(b);
            field_started = true;
            record_started = true;
            pos += 1;
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: quote_start });
    }
    if record_started || !row.is_empty() || !field.is_empty() || closed || field_started {
        row.push(String::from_utf8(field).unwrap());
        records.push(row);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|x| x == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst =>
                header.iter().position(|x| x == name),
        }
    }
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table { header: None, rows: records, duplicates: config.duplicates });
    }
    if records.is_empty() {
        return Ok(Table { header: None, rows: Vec::new(), duplicates: config.duplicates });
    }

    let header = records[0].clone();
    if config.duplicates == DuplicatePolicy::Reject {
        let mut seen: Vec<(String, usize)> = Vec::new();
        for (i, name) in header.iter().enumerate() {
            if let Some((_, first)) = seen.iter().find(|(n, _)| n == name) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(), first: *first, second: i,
                });
            }
            seen.push((name.clone(), i));
        }
    }

    for (i, row) in records.iter().enumerate().skip(1) {
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth {
                record: i, expected: header.len(), actual: row.len(),
            });
        }
    }
    Ok(Table { header: Some(header), rows: records.into_iter().skip(1).collect(), duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    finished: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), finished: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.finished {
            return result.clone();
        }
        let result = parse_with(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ColumnRef { Index(usize), Name(String) }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType { Text, Integer, Decimal, Boolean }

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef,
    pub kind: ColumnType,
    pub nullable: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value { Null, Text(String), Integer(i64), Decimal(i64), Boolean(bool) }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn, MissingCell, Empty, InvalidInteger, IntegerOverflow,
    InvalidDecimal, DecimalOverflow, InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy { Collect, FirstPerRow, Stop }

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    if b[start] == b'0' { return start + 1 == b.len(); }
    b[start].is_ascii_digit() && b[start] != b'0' && b[start..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.starts_with(b"-"));
    let Some(dot) = b.iter().position(|&x| x == b'.') else { return false; };
    if dot < start || b.len() != dot + 3 { return false; }
    let whole = &b[start..dot];
    let frac = &b[dot + 1..];
    if whole.is_empty() || !frac.iter().all(u8::is_ascii_digit) { return false; }
    let int_ok = if whole[0] == b'0' { whole.len() == 1 }
        else { whole[0].is_ascii_digit() && whole[0] != b'0' && whole.iter().all(u8::is_ascii_digit) };
    if !int_ok { return false; }
    !(start == 1 && whole == b"0" && frac == b"00")
}

fn evaluate(cell: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if spec.nullable { return Ok(Value::Null); }
        if spec.kind != ColumnType::Text { return Err(CellErrorKind::Empty); }
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(cell) { return Err(CellErrorKind::InvalidInteger); }
            cell.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(cell) { return Err(CellErrorKind::InvalidDecimal); }
            let neg = cell.starts_with('-');
            let body = if neg { &cell[1..] } else { cell };
            let mut parts = body.split('.');
            let whole = parts.next().unwrap().parse::<i128>().unwrap();
            let frac = parts.next().unwrap().parse::<i128>().unwrap();
            let magnitude = whole * 100 + frac;
            let signed = if neg { -magnitude } else { magnitude };
            i64::try_from(signed).map(Value::Decimal).map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        }
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    for (row_index, row) in table.rows.iter().enumerate() {
        let mut out = Vec::with_capacity(columns.len());
        let mut failed = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            if failed {
                out.push(None);
                continue;
            }
            let index = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(n) => table.column_index(n),
            };
            let evaluated = match index {
                None => Err((None, CellErrorKind::UnknownColumn)),
                Some(i) => match row.get(i) {
                    None => Err((Some(i), CellErrorKind::MissingCell)),
                    Some(cell) => evaluate(cell, spec).map_err(|e| (Some(i), e)),
                }
            };
            match evaluated {
                Ok(value) => out.push(Some(value)),
                Err((column, kind)) => {
                    out.push(None);
                    result.errors.push(CellError { row: row_index, spec: spec_index, column, kind });
                    if policy != ErrorPolicy::Collect { failed = true; }
                    if policy == ErrorPolicy::Stop {
                        result.rows.push(out);
                        return result;
                    }
                    if policy == ErrorPolicy::FirstPerRow { break; }
                }
            }
        }
        if out.len() < columns.len() { out.resize(columns.len(), None); }
        result.rows.push(out);
    }
    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding { Lf, CrLf }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum WriteMode { Preserve, Canonical }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {
    pub delimiter: u8,
    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn duplicate_error(header: &[String]) -> Option<WriteError> {
    let mut seen: Vec<(String, usize)> = Vec::new();
    for (i, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(n, _)| n == name) {
            return Some(WriteError::DuplicateHeader { name: name.clone(), first: *first, second: i });
        }
        seen.push((name.clone(), i));
    }
    None
}

fn cmp_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    a.cmp(b)
}

fn encode_field(field: &str, first: bool, sole: bool, delimiter: u8, quote: u8) -> String {
    let bytes = field.as_bytes();
    let needs = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
        || (first && field.starts_with('\u{feff}'))
        || (sole && field.is_empty());
    if !needs { return field.to_string(); }
    let mut out = String::new();
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 { out.push(quote as char); }
        out.push(ch);
    }
    out.push(quote as char);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut originals: Vec<(usize, &Vec<String>)> = Vec::new();
    if let Some(header) = &table.header { originals.push((0, header)); }
    let data_start = usize::from(table.header.is_some());
    for (i, row) in table.rows.iter().enumerate() { originals.push((data_start + i, row)); }

    for (record, row) in &originals {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul { record: *record, column, byte });
            }
        }
    }
    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width { record: data_start + i, expected: header.len(), actual: row.len() });
            }
        }
    }
    for (record, row) in &originals {
        if row.is_empty() { return Err(WriteError::ZeroColumns { record: *record }); }
    }
    if config.mode == WriteMode::Canonical
        && table.header.is_some()
        && table.duplicates == DuplicatePolicy::Reject
    {
        if let Some(header) = &table.header {
            if let Some(error) = duplicate_error(header) { return Err(error); }
        }
    }

    let mut header_out: Option<Vec<String>> = table.header.clone();
    let mut rows_out = table.rows.clone();
    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            let mut retained: Vec<usize> = Vec::new();
            for (i, name) in header.iter().enumerate() {
                let positions: Vec<usize> = header.iter().enumerate()
                    .filter_map(|(j, n)| if n == name { Some(j) } else { None }).collect();
                let chosen = match table.duplicates {
                    DuplicatePolicy::KeepLast => *positions.last().unwrap(),
                    DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => positions[0],
                };
                if chosen == i { retained.push(i); }
            }
            retained.sort_by(|&a, &b| header[a].as_bytes().cmp(header[b].as_bytes()));
            header_out = Some(retained.iter().map(|&i| header[i].clone()).collect());
            rows_out = rows_out.iter().map(|row| retained.iter().map(|&i| row[i].clone()).collect()).collect();
        }
        rows_out.sort_by(|a, b| cmp_rows(a, b));
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    if let Some(h) = header_out { records.push(h); }
    records.extend(rows_out);
    if records.is_empty() { return Ok(String::new()); }

    let canonical = config.mode == WriteMode::Canonical;
    let term = if canonical { "\n" } else {
        match config.line_ending { LineEnding::Lf => "\n", LineEnding::CrLf => "\r\n" }
    };
    let mut out = String::new();
    for (ri, row) in records.iter().enumerate() {
        if ri > 0 { out.push_str(term); }
        for (ci, field) in row.iter().enumerate() {
            if ci > 0 { out.push(config.delimiter as char); }
            let sole = row.len() == 1;
            out.push_str(&encode_field(field, ci == 0 && ri == 0, sole, config.delimiter, config.quote));
        }
    }
    if canonical || config.final_terminator { out.push_str(term); }
    Ok(out)
}
