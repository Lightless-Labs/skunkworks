#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|column| column == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|column| column == name)
            }
        }
    }
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut index = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;
    let mut record_started = false;

    #[derive(Clone, Copy)]
    enum State {
        Start,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::Start;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: index,
            });
        }

        let in_quoted = matches!(state, State::Quoted { .. });
        if !in_quoted && byte == b'\r' {
            if bytes.get(index + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: index,
                });
            }
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            record_index += 1;
            record_started = false;
            state = State::Start;
            index += 2;
            continue;
        }

        match state {
            State::Start => {
                if byte == quote {
                    state = State::Quoted { opening: index };
                    record_started = true;
                    index += 1;
                } else if byte == delimiter {
                    fields.push(String::new());
                    record_started = true;
                    index += 1;
                } else if byte == b'\n' {
                    fields.push(String::new());
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    record_started = false;
                    index += 1;
                } else {
                    field.push(byte as char);
                    // ASCII syntax bytes are handled above; non-ASCII bytes are
                    // appended below as their original UTF-8 sequence.
                    if byte >= 0x80 {
                        let width = utf8_width(byte);
                        field.pop();
                        field.push_str(std::str::from_utf8(&bytes[index..index + width]).unwrap());
                        index += width;
                    } else {
                        state = State::Unquoted;
                        record_started = true;
                        index += 1;
                    }
                }
            }
            State::Unquoted => {
                if byte == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: index,
                    });
                } else if byte == delimiter {
                    fields.push(std::mem::take(&mut field));
                    state = State::Start;
                    record_started = true;
                    index += 1;
                } else if byte == b'\n' {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    record_started = false;
                    state = State::Start;
                    index += 1;
                } else if byte >= 0x80 {
                    let width = utf8_width(byte);
                    field.push_str(std::str::from_utf8(&bytes[index..index + width]).unwrap());
                    index += width;
                } else {
                    field.push(byte as char);
                    index += 1;
                }
            }
            State::Quoted { opening } => {
                if byte == quote {
                    if bytes.get(index + 1) == Some(&quote) {
                        field.push(quote as char);
                        index += 2;
                    } else {
                        state = State::AfterQuote;
                        index += 1;
                    }
                } else if byte >= 0x80 {
                    let width = utf8_width(byte);
                    field.push_str(std::str::from_utf8(&bytes[index..index + width]).unwrap());
                    index += width;
                } else {
                    let _ = opening;
                    field.push(byte as char);
                    index += 1;
                }
            }
            State::AfterQuote => {
                if byte == delimiter {
                    fields.push(std::mem::take(&mut field));
                    state = State::Start;
                    record_started = true;
                    index += 1;
                } else if byte == b'\n' {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    record_started = false;
                    state = State::Start;
                    index += 1;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: index,
                    });
                }
            }
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    if record_started || !fields.is_empty() || !field.is_empty() {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}

fn utf8_width(byte: u8) -> usize {
    if byte & 0b1111_1000 == 0b1111_0000 {
        4
    } else if byte & 0b1111_0000 == 0b1110_0000 {
        3
    } else {
        2
    }
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn build_table(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let mut iter = records.into_iter();
    let header = if config.headers {
        iter.next()
    } else {
        None
    };

    if let Some(ref names) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            for second in 0..names.len() {
                if let Some(first) = names[..second]
                    .iter()
                    .position(|name| name == &names[second])
                {
                    return Err(CsvError::DuplicateHeader {
                        name: names[second].clone(),
                        first,
                        second,
                    });
                }
            }
        }

        let expected = names.len();
        let mut rows = Vec::new();
        for (offset, row) in iter.enumerate() {
            if row.len() != expected {
                return Err(CsvError::HeaderWidth {
                    record: offset + 1,
                    expected,
                    actual: row.len(),
                });
            }
            rows.push(row);
        }

        Ok(Table {
            header,
            rows,
            duplicates: config.duplicates,
        })
    } else {
        Ok(Table {
            header: None,
            rows: iter.collect(),
            duplicates: config.duplicates,
        })
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}
