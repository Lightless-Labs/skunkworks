    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())

        self.result = Some(result.clone());
        result

        self.input.clear();
        self.result = None;

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        let mut stop_after_error = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if stop_after_error {
                continue;
            }

            let column_index = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let Some(column_index) = column_index else {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                if policy != ErrorPolicy::Collect {
                    stop_after_error = true;
                }
                continue;
            };

            let Some(cell) = row.get(column_index) else {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column_index),
                    kind: CellErrorKind::MissingCell,
                });
                if policy != ErrorPolicy::Collect {
                    stop_after_error = true;
                }
                continue;
            };

            if spec.nullable && cell.is_empty() {
                output[spec_index] = Some(Value::Null);
                continue;
            }

            let parsed = if cell.is_empty() && spec.kind != ColumnType::Text {
                Err(CellErrorKind::Empty)
            } else {
                parse_typed_cell(cell, spec.kind)
            };

            match parsed {
                Ok(value) => output[spec_index] = Some(value),
                Err(kind) => {
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column_index),
                        kind,
                    });
                    if policy != ErrorPolicy::Collect {
                        stop_after_error = true;
                    }
                }
            }
        }

        result.rows.push(output);
        if policy == ErrorPolicy::Stop && stop_after_error {
            break;
        }
    }

    result
}

fn parse_typed_cell(cell: &str, kind: ColumnType) -> Result<Value, CellErrorKind> {
    match kind {
        ColumnType::Text => Ok(Value::Text(cell.to_owned())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

fn integer_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let start = usize::from(bytes[0] == b'-');
    if start == bytes.len() {
        return false;
    }
    if bytes[start] == b'0' {
        return start + 1 == bytes.len();
    }
    if !(b'1'..=b'9').contains(&bytes[start]) {
        return false;
    }
    bytes[start + 1..].iter().all(u8::is_ascii_digit)
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn decimal_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    let start = usize::from(bytes.first() == Some(&b'-'));
    if bytes.len() < start + 4 {
        return false;
    }

    let dot = bytes.len() - 3;
    if bytes[dot] != b'.'
        || !bytes[dot + 1..].iter().all(u8::is_ascii_digit)
        || bytes[dot + 1..].len() != 2
    {
        return false;
    }

    let integral = &bytes[start..dot];
    if integral.is_empty() {
        return false;
    }
    if integral[0] == b'0' {
        if integral.len() != 1 {
            return false;
        }
    } else if !(b'1'..=b'9').contains(&integral[0])
        || !integral[1..].iter().all(u8::is_ascii_digit)
    {
        return false;
    }

    !(bytes[0] == b'-' && integral == b"0" && bytes[dot + 1..] == *b"00")
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let negative = s.starts_with('-');
    let body = if negative { &s[1..] } else { s };
    let dot = body.len() - 3;
    let whole = body[..dot]
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let fraction = body[dot + 1..]
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let magnitude = whole
        .checked_mul(100)
        .and_then(|value| value.checked_add(fraction))
        .ok_or(CellErrorKind::DecimalOverflow)?;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_configured(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let records = parse_records(input, config.delimiter, config.quote)?;

    let mut header = None;
    let mut rows = records;
    if config.headers && !rows.is_empty() {
        let parsed_header = rows.remove(0);
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen = std::collections::HashMap::<&str, usize>::new();
            for (second, name) in parsed_header.iter().enumerate() {
                if let Some(first) = seen.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: *first,
                        second,
                    });
                }
                seen.insert(name, second);
            }
        }
        for (index, row) in rows.iter().enumerate() {
            if row.len() != parsed_header.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: parsed_header.len(),
                    actual: row.len(),
                });
            }
        }
        header = Some(parsed_header);
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

#[derive(Clone, Copy)]
enum State {
    FieldStart,
    Unquoted,
    Quoted { opening: usize },
    Closed,
}

fn parse_records(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let skip = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if skip == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = State::FieldStart;
    let mut i = skip;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == 0 {
            return Err(CsvError::Nul { record: records.len(), byte: i });
        }

        let quoted = matches!(state, State::Quoted { .. });
        if !quoted && byte == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record: records.len(), byte: i });
        }

        if quoted {
            match byte {
                b if b == quote => {
                    if bytes.get(i + 1) == Some(&quote) {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = State::Closed;
                        i += 1;
                    }
                }
                _ => {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                }
            }
            continue;
        }

        match state {
            State::FieldStart => {
                if byte == delimiter {
                    record.push(String::new());
                    i += 1;
                } else if byte == quote {
                    state = State::Quoted { opening: i };
                    i += 1;
                } else if byte == b'\n' || byte == b'\r' {
                    record.push(String::new());
                    records.push(std::mem::take(&mut record));
                    field.clear();
                    state = State::FieldStart;
                    i += if byte == b'\r' { 2 } else { 1 };
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    state = State::Unquoted;
                    i += ch.len_utf8();
                }
            }
            State::Unquoted => {
                if byte == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    i += 1;
                } else if byte == quote {
                    return Err(CsvError::UnexpectedQuote { record: records.len(), byte: i });
                } else if byte == b'\n' || byte == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    state = State::FieldStart;
                    i += if byte == b'\r' { 2 } else { 1 };
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                }
            }
            State::Closed => {
                if byte == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    i += 1;
                } else if byte == b'\n' || byte == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    state = State::FieldStart;
                    i += if byte == b'\r' { 2 } else { 1 };
                } else {
                    return Err(CsvError::TrailingByte { record: records.len(), byte: i });
                }
            }
            State::Quoted { .. } => unreachable!(),
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote {
            record: records.len(),
            byte: opening,
        });
    }

    if !matches!(state, State::FieldStart) || !record.is_empty() {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

fn parse_with_config(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = Vec::new();
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;
    let mut record = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !in_quotes && b == b'\r' {
            if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                return Err(CsvError::BareCr { record, byte: i });
            }
            row.push(String::from_utf8(field).expect("input is valid UTF-8"));
            records.push(row);
            row = Vec::new();
            field = Vec::new();
            i += 2;
            record += 1;
            record_started = false;
            after_quote = false;
            continue;
        }

        if in_quotes {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote);
                    i += 2;
                    continue;
                }
                in_quotes = false;
                after_quote = true;
                i += 1;
                continue;
            }
            field.push(b);
            i += 1;
            continue;
        }

        if after_quote {
            if b == delimiter {
                row.push(String::from_utf8(field).expect("input is valid UTF-8"));
                field = Vec::new();
                after_quote = false;
                record_started = true;
                i += 1;
                continue;
            }
            if b == b'\n' {
                row.push(String::from_utf8(field).expect("input is valid UTF-8"));
                records.push(row);
                row = Vec::new();
                field = Vec::new();
                after_quote = false;
                record_started = false;
                i += 1;
                record += 1;
                continue;
            }
            return Err(CsvError::TrailingByte { record, byte: i });
        }

        if b == delimiter {
            row.push(String::from_utf8(field).expect("input is valid UTF-8"));
            field = Vec::new();
            record_started = true;
            i += 1;
            continue;
        }
        if b == b'\n' {
            row.push(String::from_utf8(field).expect("input is valid UTF-8"));
            records.push(row);
            row = Vec::new();
            field = Vec::new();
            record_started = false;
            i += 1;
            record += 1;
            continue;
        }
        if b == quote {
            if field.is_empty() {
                in_quotes = true;
                quote_open = i;
                record_started = true;
                i += 1;
                continue;
            }
            return Err(CsvError::UnexpectedQuote { record, byte: i });
        }

        field.push(b);
        record_started = true;
        i += 1;
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_open,
        });
    }

    if record_started || after_quote || !field.is_empty() || !row.is_empty() {
        row.push(String::from_utf8(field).expect("input is valid UTF-8"));
        records.push(row);
    }

    Ok(records)
}

impl Table {

        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn table_from_records(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table {
            header: None,
            rows: records,
            duplicates: config.duplicates,
        });
    }

    let mut records = records.into_iter();
    let Some(header) = records.next() else {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    };

    if config.duplicates == DuplicatePolicy::Reject {
        for second in 0..header.len() {
            if let Some(first) = header[..second]
                .iter()
                .position(|value| value == &header[second])
            {
                return Err(CsvError::DuplicateHeader {
                    name: header[second].clone(),
                    first,
                    second,
                });
            }
        }
    }

    let mut rows = Vec::new();
    for (record_index, row) in records.enumerate() {
        let input_record = record_index + 1;
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth {
                record: input_record,
                expected: header.len(),
                actual: row.len(),
            });
        }
        rows.push(row);
    }

    Ok(Table {
        header: Some(header),
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }

        let result = parse_with_config(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let start = usize::from(bytes[0] == b'-');
    if start == bytes.len() {
        return false;
    }
    if bytes[start] == b'0' {
        return start + 1 == bytes.len();
    }
    if !(b'1'..=b'9').contains(&bytes[start]) {
        return false;
    }
    bytes[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let start = usize::from(bytes[0] == b'-');
    if start >= bytes.len() {
        return false;
    }

    let mut i = start;
    if bytes[i] == b'0' {
        i += 1;
    } else if (b'1'..=b'9').contains(&bytes[i]) {
        i += 1;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
    } else {
        return false;
    }

    if i + 3 != bytes.len() || bytes[i] != b'.' {
        return false;
    }
    bytes[i + 1..].iter().all(u8::is_ascii_digit)
}

fn parse_integer(value: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(value) {
        return Err(CellErrorKind::InvalidInteger);
    }
    value
        .parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(value: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(value) {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let negative = value.starts_with('-');
    let unsigned = if negative { &value[1..] } else { value };
    let (whole, fractional) = unsigned.split_once('.').expect("validated decimal");
    let magnitude = whole
        .parse::<i128>()
        .expect("validated decimal integer")
        .checked_mul(100)
        .and_then(|n| {
            n.checked_add(
                fractional
                    .parse::<i128>()
                    .expect("validated decimal fraction"),
            )
        })
        .expect("decimal magnitude fits i128");

    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate_cell(
    cell: &str,
    spec: &ColumnSpec,
) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_owned())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];

        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let error = if let Some(column_index) = resolved {
                if let Some(cell) = row.get(column_index) {
                    match evaluate_cell(cell, spec) {
                        Ok(value) => {
                            output[spec_index] = Some(value);
                            None
                        }
                        Err(kind) => Some(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: Some(column_index),
                            kind,
                        }),
                    }
                } else {
                    Some(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column_index),
                        kind: CellErrorKind::MissingCell,
                    })
                }
            } else {
                Some(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                })
            };

            if let Some(error) = error {
                result.errors.push(error);
                match policy {
                    ErrorPolicy::Collect => {}
                    ErrorPolicy::FirstPerRow => break,
                    ErrorPolicy::Stop => {
                        result.rows.push(output);
                        break 'rows;
                    }
                }
            }
        }

        result.rows.push(output);
    }

    result
}

    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();

    let mut i = start;
    let mut record_index = 0usize;
    let mut at_field_start = true;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut opening_quote = 0usize;

                byte: i,
            });

                record_index += 1;
                at_field_start = true;
                after_quote = false;
                record_started = false;
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: i,
            });
        }

            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            i += 1;
            record_index += 1;
            at_field_start = true;
            record_started = false;
            continue;

                opening_quote = i;
                at_field_start = false;
                record_started = true;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: i,
                });
            }
            continue;
        }

        let ch = input[i..].chars().next().expect("valid UTF-8 boundary");
        field.push(ch);
        i += ch.len_utf8();
        at_field_start = false;
        record_started = true;
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if record_started || !record.is_empty() || !field.is_empty() || after_quote {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_config(input, b',', b'"')
}

fn table_from_input(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let parsed = parse_with_config(input, config.delimiter, config.quote)?;

    let (header, rows) = if config.headers {
        if parsed.is_empty() {
            (None, Vec::new())
        } else {
            let mut iter = parsed.into_iter();
            (iter.next(), iter.collect::<Vec<_>>())
        }
    } else {
        (None, parsed)
    };

    if let Some(header_values) = header.as_ref() {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: HashMap<&str, usize> = HashMap::new();
            for (second, name) in header_values.iter().enumerate() {
                if let Some(&first) = seen.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first,
                        second,
                    });
                }
                seen.insert(name.as_str(), second);
            }
        }

        for (index, row) in rows.iter().enumerate() {
            if row.len() != header_values.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: header_values.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = table_from_input(&self.input, self.config);
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ColumnRef {
    Index(usize),
    Name(String),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType {
    Text,
    Integer,
    Decimal,
    Boolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef,
    pub kind: ColumnType,
    pub nullable: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Null,
    Text(String),
    Integer(i64),
    Decimal(i64),
    Boolean(bool),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(text: &str) -> bool {
    let bytes = text.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let digits = if bytes[0] == b'-' { &bytes[1..] } else { bytes };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return bytes[0] != b'-';
    }
    (b'1'..=b'9').contains(&digits[0]) && digits[1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(text: &str) -> bool {
    let bytes = text.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let body = if bytes[0] == b'-' { &bytes[1..] } else { bytes };
    let Some(dot) = body.iter().position(|&b| b == b'.') else {
        return false;
    };
    if body[dot + 1..].len() != 2
        || !body[dot + 1..].iter().all(u8::is_ascii_digit)
    {
        return false;
    }
    let integer = &body[..dot];
    if integer.is_empty() {
        return false;
    }
    let valid_integer = integer == b"0"
        || ((b'1'..=b'9').contains(&integer[0])
            && integer[1..].iter().all(u8::is_ascii_digit));
    if !valid_integer {
        return false;
    }
    !(bytes[0] == b'-' && integer == b"0" && body[dot + 1..] == *b"00")
}

fn parse_integer(text: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(text) {
        return Err(CellErrorKind::InvalidInteger);
    }
    text.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(text: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(text) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = text.starts_with('-');
    let unsigned = if negative { &text[1..] } else { text };
    let dot = unsigned.find('.').expect("validated decimal");
    let whole: i128 = unsigned[..dot]
        .parse()
        .expect("validated decimal digits");
    let fraction: i128 = unsigned[dot + 1..]
        .parse()
        .expect("validated decimal digits");
    let magnitude = whole * 100 + fraction;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate_cell(
    table: &Table,
    row_index: usize,
    row: &[String],
    spec_index: usize,
    spec: &ColumnSpec,
) -> (Option<Value>, Option<CellError>) {
    let column = match &spec.column {
        ColumnRef::Index(index) => Some(*index),
        ColumnRef::Name(name) => table.column_index(name),
    };

    let Some(column) = column else {
        return (
            None,
            Some(CellError {
                row: row_index,
                spec: spec_index,
                column: None,
                kind: CellErrorKind::UnknownColumn,
            }),
        );
    };

    let Some(cell) = row.get(column) else {
        return (
            None,
            Some(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(column),
                kind: CellErrorKind::MissingCell,
            }),
        );
    };

    if cell.is_empty() {
        if spec.nullable {
            return (Some(Value::Null), None);
        }
        if spec.kind != ColumnType::Text {
            return (
                None,
                Some(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column),
                    kind: CellErrorKind::Empty,
                }),
            );
        }
    }

    let value = match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.clone())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    };

    match value {
        Ok(value) => (Some(value), None),
        Err(kind) => (
            None,
            Some(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(column),
                kind,
            }),
        ),
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut output = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut values = Vec::with_capacity(columns.len());
        let mut failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if failed {
                values.push(None);
                continue;
            }

            let (value, error) = evaluate_cell(table, row_index, row, spec_index, spec);
            values.push(value);
            if let Some(error) = error {
                output.errors.push(error);
                failed = true;
                if policy == ErrorPolicy::Stop {
                    for _ in spec_index + 1..columns.len() {
                        values.push(None);
                    }
                    break;
                }
            }

            if failed && policy == ErrorPolicy::FirstPerRow {
                for _ in spec_index + 1..columns.len() {
                    values.push(None);
                }
                break;
            }
        }

        output.rows.push(values);
        if failed && policy == ErrorPolicy::Stop {
            break;
        }
    }

    output
}
