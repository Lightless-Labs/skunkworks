#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut field_started = false;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_start = 0;
    let mut record = 0;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !in_quotes && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if in_quotes {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let start = i;
                i += 1;
                while i < bytes.len() && bytes[i] != quote && bytes[i] != 0 {
                    i += 1;
                }
                field.push_str(&input[start..i]);
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                fields.push(std::mem::take(&mut field));
                field_started = false;
                after_quote = false;
                i += 1;
            } else if b == b'\n' || (b == b'\r' && bytes.get(i + 1) == Some(&b'\n')) {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                field_started = false;
                after_quote = false;
                record += 1;
                i += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record, byte: i });
            }
            continue;
        }

        if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote { record, byte: i });
            }
            field_started = true;
            in_quotes = true;
            quote_start = i;
            i += 1;
        } else if b == delimiter {
            fields.push(std::mem::take(&mut field));
            field_started = false;
            i += 1;
        } else if b == b'\n' || (b == b'\r' && bytes.get(i + 1) == Some(&b'\n')) {
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            field_started = false;
            record += 1;
            i += if b == b'\r' { 2 } else { 1 };
        } else {
            let start = i;
            i += 1;
            while i < bytes.len()
                && bytes[i] != 0
                && bytes[i] != delimiter
                && bytes[i] != quote
                && bytes[i] != b'\n'
                && bytes[i] != b'\r'
            {
                i += 1;
            }
            field.push_str(&input[start..i]);
            field_started = true;
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_start,
        });
    }

    if !fields.is_empty() || field_started || !field.is_empty() {
        fields.push(field);
        records.push(fields);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn build_table(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut iter = records.into_iter();
            (iter.next(), iter.collect::<Vec<_>>())
        }
    } else {
        (None, records)
    };

    if let Some(ref header) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            for second in 0..header.len() {
                if let Some(first) = header[..second].iter().position(|s| s == &header[second]) {
                    return Err(CsvError::DuplicateHeader {
                        name: header[second].clone(),
                        first,
                        second,
                    });
                }
            }
        }
        for (index, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let parsed = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.result = Some(parsed.clone());
        parsed
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    (b'1'..=b'9').contains(&b[start]) && b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let Some(dot) = s.find('.') else {
        return false;
    };
    let (whole, fraction_with_dot) = s.split_at(dot);
    let fraction = &fraction_with_dot[1..];
    if fraction.len() != 2 || !fraction.bytes().all(|b| b.is_ascii_digit()) {
        return false;
    }
    let b = whole.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        if start + 1 != b.len() {
            return false;
        }
        return whole != "-0";
    }
    (b'1'..=b'9').contains(&b[start]) && b[start + 1..].iter().all(u8::is_ascii_digit)
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut output = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };
    let mut stopped = false;

    for (row_index, row) in table.rows.iter().enumerate() {
        if stopped {
            break;
        }
        let mut out_row = vec![None; columns.len()];
        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (column, value) = match resolved {
                None => (None, Err(CellErrorKind::UnknownColumn)),
                Some(index) => match row.get(index) {
                    None => (Some(index), Err(CellErrorKind::MissingCell)),
                    Some(cell) if cell.is_empty() && spec.nullable => {
                        out_row[spec_index] = Some(Value::Null);
                        continue;
                    }
                    Some(cell) if cell.is_empty() && spec.kind != ColumnType::Text => {
                        (Some(index), Err(CellErrorKind::Empty))
                    }
                    Some(cell) => {
                        let parsed = match spec.kind {
                            ColumnType::Text => Ok(Value::Text(cell.clone())),
                            ColumnType::Integer => {
                                if !integer_syntax(cell) {
                                    Err(CellErrorKind::InvalidInteger)
                                } else {
                                    cell.parse::<i64>()
                                        .map(Value::Integer)
                                        .map_err(|_| CellErrorKind::IntegerOverflow)
                                }
                            }
                            ColumnType::Decimal => {
                                if !decimal_syntax(cell) {
                                    Err(CellErrorKind::InvalidDecimal)
                                } else {
                                    let neg = cell.starts_with('-');
                                    let unsigned = if neg { &cell[1..] } else { cell.as_str() };
                                    let (whole, frac) = unsigned.split_once('.').unwrap();
                                    let combined = format!("{whole}{frac}");
                                    let magnitude = combined.parse::<u128>();
                                    match magnitude {
                                        Ok(m) if neg && m == (i64::MAX as u128) + 1 => {
                                            Ok(Value::Decimal(i64::MIN))
                                        }
                                        Ok(m) if neg && m <= i64::MAX as u128 => {
                                            Ok(Value::Decimal(-(m as i64)))
                                        }
                                        Ok(m) if !neg && m <= i64::MAX as u128 => {
                                            Ok(Value::Decimal(m as i64))
                                        }
                                        _ => Err(CellErrorKind::DecimalOverflow),
                                    }
                                }
                            }
                            ColumnType::Boolean => match cell.as_str() {
                                "yes" | "1" => Ok(Value::Boolean(true)),
                                "no" | "0" => Ok(Value::Boolean(false)),
                                _ => Err(CellErrorKind::InvalidBoolean),
                            },
                        };
                        (Some(index), parsed)
                    }
                },
            };

            match value {
                Ok(v) => out_row[spec_index] = Some(v),
                Err(kind) => {
                    output.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column,
                        kind,
                    });
                    if policy == ErrorPolicy::Stop {
                        stopped = true;
                        break;
                    }
                    if policy == ErrorPolicy::FirstPerRow {
                        break;
                    }
                }
            }
        }
        output.rows.push(out_row);
    }
    output
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {

    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn duplicate_header(header: &[String]) -> Option<(String, usize, usize)> {
    for second in 0..header.len() {
        if let Some(first) = header[..second].iter().position(|s| s == &header[second]) {
            return Some((header[second].clone(), first, second));
        }
    }
    None
}

fn quote_field(
    field: &str,
    first_field_first_record: bool,
    sole_empty: bool,
    delimiter: u8,
    quote: u8,
) -> String {
    let bytes = field.as_bytes();
    let starts_or_ends = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
    let bom = first_field_first_record && field.starts_with('\u{feff}');
    let must_quote = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || starts_or_ends
        || bom
        || sole_empty;

    if !must_quote {
        return field.to_string();
    }

    let q = quote as char;
    let mut out = String::new();
    out.push(q);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(q);
            out.push(q);
        } else {
            out.push(ch);
        }
    }
    out.push(q);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original_records: Vec<&[String]> = Vec::new();
    if let Some(header) = &table.header {
        original_records.push(header);
    }
    for row in &table.rows {
        original_records.push(row);
    }

    for (record, fields) in original_records.iter().enumerate() {
        for (column, field) in fields.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|b| *b == 0) {
                return Err(WriteError::Nul {
                    record,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, fields) in original_records.iter().enumerate() {
        if fields.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some((name, first, second)) = duplicate_header(header) {
                    return Err(WriteError::DuplicateHeader {
                        name,
                        first,
                        second,
                    });
                }
            }
        }
    }

    let (headers, mut rows): (Option<Vec<String>>, Vec<Vec<String>>) =
        if config.mode == WriteMode::Preserve {
            (table.header.clone(), table.rows.clone())
        } else if let Some(header) = &table.header {
            let mut keep = Vec::new();
            for (index, name) in header.iter().enumerate() {
                let retained = match table.duplicates {
                    DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                        !header[..index].iter().any(|earlier| earlier == name)
                    }
                    DuplicatePolicy::KeepLast => {
                        !header[index + 1..].iter().any(|later| later == name)
                    }
                };
                if retained {
                    keep.push(index);
                }
            }
            keep.sort_by(|a, b| header[*a].as_bytes().cmp(header[*b].as_bytes()));
            let projected_header = keep.iter().map(|i| header[*i].clone()).collect();
            let projected_rows = table
                .rows
                .iter()
                .map(|row| keep.iter().map(|i| row[*i].clone()).collect::<Vec<_>>())
                .collect::<Vec<_>>();
            (Some(projected_header), projected_rows)
        } else {
            (None, table.rows.clone())
        };

    if config.mode == WriteMode::Canonical {
        rows.sort_by(|a, b| {
            for (x, y) in a.iter().zip(b.iter()) {
                let ordering = x.as_bytes().cmp(y.as_bytes());
                if ordering != std::cmp::Ordering::Equal {
                    return ordering;
                }
            }
            a.len().cmp(&b.len())
        });
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    if let Some(header) = headers {
        records.push(header);
    }
    records.extend(rows);

    if records.is_empty() {
        return Ok(String::new());
    }

    let canonical = config.mode == WriteMode::Canonical;
    let separator = if canonical || config.line_ending == LineEnding::Lf {
        "\n"
    } else {
        "\r\n"
    };

    let mut rendered_records = Vec::with_capacity(records.len());
    for (record_index, fields) in records.iter().enumerate() {
        let mut rendered = Vec::new();
        for (field_index, field) in fields.iter().enumerate() {
            rendered.push(quote_field(
                field,
                record_index == 0 && field_index == 0,
                fields.len() == 1 && field.is_empty(),
                config.delimiter,
                config.quote,
            ));
        }
        rendered_records.push(rendered.join(&(config.delimiter as char).to_string()));
    }

    let mut output = rendered_records.join(separator);
    if canonical || config.final_terminator {
        output.push_str(separator);
    }
    Ok(output)
}
