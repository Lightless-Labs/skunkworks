#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

pub struct Encoder {
    buffered: Vec<u8>,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            buffered: Vec::new(),
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut data = Vec::with_capacity(self.buffered.len() + input.len());
        data.extend_from_slice(&self.buffered);
        data.extend_from_slice(input);

        let complete_len = data.len() / 3 * 3;
        let mut output = String::with_capacity(complete_len / 3 * 4);

        for chunk in data[..complete_len].chunks_exact(3) {
            encode_triplet(chunk[0], chunk[1], chunk[2], &mut output);
        }

        self.buffered.clear();
        self.buffered.extend_from_slice(&data[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        match self.buffered.as_slice() {
            [] => {}
            [a] => {
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[((a & 0x03) << 4) as usize] as char);
                output.push_str("==");
            }
            [a, b] => {
                output.push(ALPHABET[(a >> 2) as usize] as char);
                output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
                output.push(ALPHABET[((b & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => unreachable!(),
        }

        self.buffered.clear();
        self.finished = true;
        Ok(output)
    }
}

pub struct Decoder {
    group: [u8; 4],
    group_offsets: [usize; 4],
    group_len: usize,
    offset: usize,
    terminal_padding: bool,
    pending_second_padding: Option<usize>,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            group_offsets: [0; 4],
            group_len: 0,
            offset: 0,
            terminal_padding: false,
            pending_second_padding: None,
            finished: false,
            sticky_error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let current_offset = self.offset;
            self.offset += 1;

            if self.terminal_padding {
                return self.fail(Error::BadPadding {
                    offset: current_offset,
                });
            }

            if let Some(pending_offset) = self.pending_second_padding {
                if byte != b'=' {
                    return self.fail(Error::BadPadding {
                        offset: current_offset,
                    });
                }

                self.pending_second_padding = None;
                self.group[3] = b'=';
                self.group_offsets[3] = current_offset;
                self.group_len = 0;
                self.terminal_padding = true;
                output.push((self.value(self.group[0]) << 2) | (self.value(self.group[1]) >> 4));
                let _ = pending_offset;
                continue;
            }

            match self.group_len {
                0 | 1 => {
                    if byte == b'=' {
                        return self.fail(Error::BadPadding {
                            offset: current_offset,
                        });
                    }
                    let Some(_) = alphabet_value(byte) else {
                        return self.fail(Error::InvalidByte {
                            offset: current_offset,
                            byte,
                        });
                    };
                    self.store(byte, current_offset);
                }
                2 => {
                    if byte == b'=' {
                        self.store(byte, current_offset);
                        self.pending_second_padding = Some(current_offset);
                    } else {
                        let Some(_) = alphabet_value(byte) else {
                            return self.fail(Error::InvalidByte {
                                offset: current_offset,
                                byte,
                            });
                        };
                        self.store(byte, current_offset);
                    }
                }
                3 => {
                    if byte == b'=' {
                        if self.group[2] == b'=' {
                            self.group[3] = byte;
                            self.group_offsets[3] = current_offset;
                            self.group_len = 0;
                            self.terminal_padding = true;
                            output.push((self.value(self.group[0]) << 2)
                                | (self.value(self.group[1]) >> 4));
                        } else {
                            self.group[3] = byte;
                            self.group_offsets[3] = current_offset;
                            self.group_len = 0;
                            self.terminal_padding = true;
                            output.push((self.value(self.group[0]) << 2)
                                | (self.value(self.group[1]) >> 4));
                            output.push((self.value(self.group[1]) << 4)
                                | (self.value(self.group[2]) >> 2));
                        }
                    } else {
                        let Some(_) = alphabet_value(byte) else {
                            return self.fail(Error::InvalidByte {
                                offset: current_offset,
                                byte,
                            });
                        };
                        self.store(byte, current_offset);
                        output.push((self.value(self.group[0]) << 2)
                            | (self.value(self.group[1]) >> 4));
                        output.push((self.value(self.group[1]) << 4)
                            | (self.value(self.group[2]) >> 2));
                        output.push((self.value(self.group[2]) << 6) | self.value(self.group[3]));
                        self.group_len = 0;
                    }
                }
                _ => unreachable!(),
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        if let Some(offset) = self.pending_second_padding {
            return self.fail(Error::BadPadding { offset });
        }
        if self.group_len != 0 {
            return self.fail(Error::Truncated {
                offset: self.group_offsets[0],
            });
        }

        self.finished = true;
        Ok(Vec::new())
    }

    fn store(&mut self, byte: u8, offset: usize) {
        self.group[self.group_len] = byte;
        self.group_offsets[self.group_len] = offset;
        self.group_len += 1;
    }

    fn value(&self, byte: u8) -> u8 {
        alphabet_value(byte).unwrap_or(0)
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky_error = Some(error.clone());
        Err(error)
    }
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

fn alphabet_value(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}

fn encode_triplet(a: u8, b: u8, c: u8, output: &mut String) {
    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    output.push(ALPHABET[(c & 0x3f) as usize] as char);
}

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).unwrap();
    output.push_str(&encoder.finish().unwrap());
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}
