#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|part| !part.is_empty())
            .last()
            .is_some_and(|part| part == "." || part == "..")
}

fn components(path: &str) -> Vec<(String, usize)> {
    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((path[start..i].to_string(), start));
    }
    result
}

fn reduce_component(
    stack: &mut Vec<String>,
    part: &str,
    at: usize,
    absolute: bool,
) -> Result<(), PathError> {
    match part {
        "." => {}
        ".." => {
            if stack.last().is_some_and(|s| s != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_string());
            }
        }
        _ => stack.push(part.to_string()),
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut out = if absolute {
        if stack.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", stack.join("/"))
        }
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };

    if intent && out != "/" && out != "." {
        out.push('/');
    }
    out
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();

    for (part, at) in components(path) {
        reduce_component(&mut stack, &part, at, absolute)?;
    }

    Ok(render(&stack, absolute, intent))
}

fn validate_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }
    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized).into_iter().map(|(p, _)| p).collect()),
        Err(_) => Err(error),
    }
}

fn validate_request(path: &str) -> Result<(), PathError> {
    validate_raw(path)
}

fn tilde_info(path: &str) -> Result<Option<usize>, PathError> {
    if !path.starts_with('~') {
        return Ok(None);
    }
    if path == "~" || path.as_bytes().get(1) == Some(&b'/') {
        Ok(Some(if path == "~" { 1 } else { 2 }))
    } else {
        Err(PathError::UnsupportedTilde { at: 1 })
    }
}

fn resolve_base(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<(Vec<String>, bool, usize), PathError> {
    let cwd_parts = validate_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(h) => Some(validate_context(h, PathError::InvalidHome)?),
        None => None,
    };

    validate_request(path)?;
    let tilde = tilde_info(path)?;

    if let Some(consumed) = tilde {
        let base = home_parts.ok_or(PathError::MissingHome)?;
        Ok((base, true, consumed))
    } else if path.starts_with('/') {
        Ok((Vec::new(), true, 0))
    } else {
        Ok((cwd_parts, true, 0))
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let (mut stack, absolute, consumed) = resolve_base(path, cwd, home)?;
    let intent = directory_intent(path);

    for (part, at) in components(&path[consumed..]) {
        reduce_component(&mut stack, &part, at + consumed, absolute)?;
    }

    Ok(render(&stack, absolute, intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();

    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }

        if let Some((_, first)) = seen.iter().find(|(p, _)| *p == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }
    Ok(())
}

fn absolute_key(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", stack.join("/"))
    }
}

struct Traversal<'a> {
    links: &'a [Link],
    max_expansions: usize,
    expansions: usize,
    active: Vec<usize>,
}

impl<'a> Traversal<'a> {
    fn process(
        &mut self,
        stack: &mut Vec<String>,
        raw: &str,
        offset: usize,
        absolute: bool,
    ) -> Result<(), PathError> {
        for (part, at) in components(raw) {
            let original_at = at + offset;
            match part.as_str() {
                "." => continue,
                ".." => {
                    if stack.last().is_some_and(|s| s != "..") {
                        stack.pop();
                    } else if absolute {
                        return Err(PathError::Escape { at: original_at });
                    } else {
                        stack.push("..".to_string());
                    }
                }
                _ => {
                    stack.push(part);
                    if absolute {
                        let key = absolute_key(stack);
                        if let Some((index, link)) = self
                            .links
                            .iter()
                            .enumerate()
                            .find(|(_, link)| link.path == key)
                        {
                            if self.active.contains(&index) {
                                return Err(PathError::Loop { index });
                            }
                            if self.expansions >= self.max_expansions {
                                return Err(PathError::Limit { index });
                            }
                            self.expansions += 1;
                            self.active.push(index);

                            // Relative targets begin at the link's parent.
                            stack.pop();
                            let target_absolute = link.target.starts_with('/');
                            if target_absolute {
                                stack.clear();
                            }
                            let result = self.process(stack, &link.target, 0, true);
                            self.active.pop();
                            result?;
                        }
                    }
                }
            }
        }
        Ok(())
    }
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(h) => Some(validate_context(h, PathError::InvalidHome)?),
        None => None,
    };

    validate_links(links)?;
    validate_request(path)?;
    let tilde = tilde_info(path)?;

    let (mut stack, consumed) = if let Some(consumed) = tilde {
        (home_parts.ok_or(PathError::MissingHome)?, consumed)
    } else if path.starts_with('/') {
        (Vec::new(), 0)
    } else {
        (cwd_parts, 0)
    };

    let intent = directory_intent(path);
    let mut traversal = Traversal {
        links,
        max_expansions,
        expansions: 0,
        active: Vec::new(),
    };

    traversal.process(&mut stack, &path[consumed..], consumed, true)?;
    Ok(render(&stack, true, intent))
}

fn main() {}
