```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    #[derive(Clone)]
    struct State {
        burst_pos: usize,
        remaining: u32,
        level: usize,
        completion: u64,
    }

    #[derive(Clone, Copy)]
    struct Event {
        time: u64,
        kind: u8,
        index: usize,
    }

    let mut states: Vec<State> = jobs
        .iter()
        .map(|job| State {
            burst_pos: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: 0,
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: u64::from(job.arrival),
            kind: 0,
            index,
        })
        .collect();
    events.sort_by_key(|event| (event.time, event.kind, event.index));
    let mut event_cursor = 0usize;

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut time = 0u64;
    let mut last_id: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut completed = 0usize;

    while completed < jobs.len() {
        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor];
            queues[states[event.index].level].push_back(event.index);
            event_cursor += 1;
        }

        let Some(level) = queues.iter().position(|queue| !queue.is_empty()) else {
            if event_cursor < events.len() {
                let next_time = events[event_cursor].time;
                if next_time > time {
                    if let Some(last) = idle.last_mut() {
                        if last.1 == time {
                            last.1 = next_time;
                        } else {
                            idle.push((time, next_time));
                        }
                    } else {
                        idle.push((time, next_time));
                    }
                    time = next_time;
                    last_id = None;
                }
                continue;
            }
            break;
        };

        let index = queues[level].pop_front().unwrap();
        let job = &jobs[index];

        if last_id.is_some_and(|id| id != job.id) {
            time += u64::from(switch_cost);
        }

        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor];
            queues[states[event.index].level].push_back(event.index);
            event_cursor += 1;
        }

        let dispatch_level = states[index].level;
        let consumed = states[index].remaining.min(quanta[dispatch_level]);
        let start = time;
        time += u64::from(consumed);
        let end = time;

        states[index].remaining -= consumed;
        let burst_finished = states[index].remaining == 0;
        let final_burst = states[index].burst_pos + 1 == job.bursts.len();

        runs.push(Run {
            id: job.id,
            start,
            end,
            finished: burst_finished && final_burst,
            level: dispatch_level,
        });

        while event_cursor < events.len() && events[event_cursor].time <= end {
            let event = events[event_cursor];
            queues[states[event.index].level].push_back(event.index);
            event_cursor += 1;
        }

        if burst_finished && final_burst {
            states[index].completion = end;
            completed += 1;
        } else if burst_finished {
            let io_pos = states[index].burst_pos + 1;
            let wake_time = end + u64::from(job.bursts[io_pos]);
            states[index].burst_pos += 2;
            states[index].remaining = job.bursts[states[index].burst_pos];
            states[index].level = dispatch_level.saturating_sub(1);

            events.push(Event {
                time: wake_time,
                kind: 1,
                index,
            });
            events[event_cursor..].sort_by_key(|event| (event.time, event.kind, event.index));

            while event_cursor < events.len() && events[event_cursor].time <= end {
                let event = events[event_cursor];
                queues[states[event.index].level].push_back(event.index);
                event_cursor += 1;
            }
        } else {
            states[index].level = (dispatch_level + 1).min(quanta.len() - 1);
            queues[states[index].level].push_back(index);
        }

        last_id = Some(job.id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut waiting_total = 0u64;
    let mut turnaround_total = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = states[index].completion;
        let turnaround = completion - u64::from(job.arrival);
        let cpu_total: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&value| u64::from(value))
            .sum();
        let io_total: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&value| u64::from(value))
            .sum();
        let waiting = turnaround - cpu_total - io_total;

        waiting_total += waiting;
        turnaround_total += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = (waiting_total * 100 + count / 2) / count;
    let mean_turnaround_x100 = (turnaround_total * 100 + count / 2) / count;

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end: time,
    })
}

fn main() {}
```
