#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_prefix_end = if bytes[0] == b'"' {
        let mut i = 1;
        let mut close = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => {
                    i += 2;
                }
                b'"' => {
                    close = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match close {
            Some(i) => Some(i),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut i = 0;
    if let Some(close) = quoted_prefix_end {
        i = close + 1;
    }
    while i < bytes.len() {
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: separators[1],
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let (local, quoted) = if let Some(close) = quoted_prefix_end {
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut pos = 1;
        while pos < close {
            let b = bytes[pos];
            if b == b'\\' {
                if pos + 1 >= close {
                    return Err(EmailError::BadEscape { at: pos });
                }
                let escaped = bytes[pos + 1];
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: pos });
                }
                decoded.push(escaped);
                origins.push(pos);
                pos += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: pos });
                }
                decoded.push(b);
                origins.push(pos);
                pos += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }

        (
            String::from_utf8(decoded).expect("validated ASCII"),
            true,
        )
    } else {
        let local_bytes = &bytes[..sep];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for (idx, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if idx == 0
                    || idx + 1 == local_bytes.len()
                    || local_bytes[idx - 1] == b'.'
                {
                    return Err(EmailError::LocalDot { at: idx });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: idx });
            }
        }

        (
            String::from_utf8(local_bytes.to_vec()).expect("validated ASCII"),
            false,
        )
    };

    let domain_start = sep + 1;
    let mut domain_end = bytes.len();
    let root_dot = if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels = Vec::new();
    let mut label_start = domain_start;
    let mut pos = domain_start;
    while pos <= domain_end {
        if pos == domain_end || bytes[pos] == b'.' {
            let label_end = pos;
            if label_start == label_end {
                let at = if label_start == domain_end {
                    root_dot.unwrap_or(label_start)
                } else {
                    label_start
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let label_len = label_end - label_start;
            if label_len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: label_start + 63,
                });
            }
            if bytes[label_start] == b'-' {
                return Err(EmailError::LabelHyphen { at: label_start });
            }
            if bytes[label_end - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: label_end - 1 });
            }
            for j in label_start..label_end {
                let b = bytes[j];
                if !b.is_ascii_alphanumeric() && b != b'-' {
                    return Err(EmailError::DomainChar { at: j });
                }
            }

            labels.push((label_start, label_end));
            label_start = pos + 1;
        }
        pos += 1;
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = *labels.last().expect("at least one label");
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let normalized_domain: String = bytes[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain: normalized_domain,
        quoted,
    })
}

fn main() {}
