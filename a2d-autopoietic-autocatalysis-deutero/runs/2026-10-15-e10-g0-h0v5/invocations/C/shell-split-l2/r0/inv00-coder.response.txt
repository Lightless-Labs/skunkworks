#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single { opening: usize },
    Double { opening: usize },
}

fn name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn first_env_value<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    if name.is_empty() {
        return None;
    }
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }

    result
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let after_open = dollar + 2;
    let mut close = None;
    let mut iter = input[after_open..].char_indices().peekable();

    while let Some((offset, c)) = iter.next() {
        let absolute = after_open + offset;
        if c == '\\' {
            if iter.next().is_none() {
                break;
            }
        } else if c == '}' {
            close = Some(absolute);
            break;
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[after_open..close];

    let mut chars = contents.char_indices();
    let mut name_end = 0;
    let mut first = true;

    for (offset, c) in chars.by_ref() {
        if (first && name_start(c)) || (!first && name_continue(c)) {
            name_end = offset + c.len_utf8();
            first = false;
        } else {
            break;
        }
    }

    if first {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_end];
    let remainder = &contents[name_end..];
    let value = if remainder.is_empty() {
        first_env_value(name, env).unwrap_or("").to_string()
    } else if let Some(default) = remainder.strip_prefix(":-") {
        match first_env_value(name, env) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single { .. } => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double { .. } => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        word.push(next);
                        i = after_next;
                    } else if next == '\n' {
                        i = after_next;
                    } else {
                        word.push('\\');
                        i = next_i;
                    }
                } else if c == '$' {
                    active = true;
                    if next_i < input.len() {
                        let n = input[next_i..].chars().next().unwrap();
                        if n == '{' {
                            let (value, end) = parse_braced(input, i, env)?;
                            word.push_str(&value);
                            i = end;
                        } else if name_start(n) {
                            let mut end = next_i;
                            while end < input.len() {
                                let nc = input[end..].chars().next().unwrap();
                                if name_continue(nc) {
                                    end += nc.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next_i..end];
                            word.push_str(first_env_value(name, env).unwrap_or(""));
                            i = end;
                        } else {
                            word.push('$');
                            i = next_i;
                        }
                    } else {
                        word.push('$');
                        i = next_i;
                    }
                } else {
                    word.push(c);
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single { opening: i };
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double { opening: i };
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        active = true;
                        word.push(next);
                        i = after_next;
                    }
                } else if c == '$' {
                    active = true;
                    if next_i < input.len() {
                        let n = input[next_i..].chars().next().unwrap();
                        if n == '{' {
                            let (value, end) = parse_braced(input, i, env)?;
                            word.push_str(&value);
                            i = end;
                        } else if name_start(n) {
                            let mut end = next_i;
                            while end < input.len() {
                                let nc = input[end..].chars().next().unwrap();
                                if name_continue(nc) {
                                    end += nc.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next_i..end];
                            word.push_str(first_env_value(name, env).unwrap_or(""));
                            i = end;
                        } else {
                            word.push('$');
                            i = next_i;
                        }
                    } else {
                        word.push('$');
                        i = next_i;
                    }
                } else {
                    active = true;
                    word.push(c);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single { opening } => {
            return Err(SplitError::UnterminatedSingle { pos: opening });
        }
        Quote::Double { opening } => {
            return Err(SplitError::UnterminatedDouble { pos: opening });
        }
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}

fn main() {}
