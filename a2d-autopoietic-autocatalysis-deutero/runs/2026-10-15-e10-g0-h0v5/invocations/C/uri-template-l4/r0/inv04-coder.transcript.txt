#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Spec<'a> {
    name: &'a str,
    modifier: Modifier,
    pos: usize,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression<'a>(
    content: &'a str,
    content_pos: usize,
    close_pos: usize,
) -> Result<(char, Vec<Spec<'a>>), Error> {
    let bytes = content.as_bytes();
    let mut operator = '\0';
    let mut start = 0;

    if let Some(&first) = bytes.first() {
        if matches!(first, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
            operator = first as char;
            start = 1;
        } else if matches!(first, b'=' | b'!' | b'@' | b'|' | b'$' | b'^') {
            return Err(Error::UnsupportedOperator { pos: content_pos });
        }
    }

    let rest = &content[start..];
    let rest_pos = content_pos + start;
    if rest.is_empty() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specs = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut offset = 0;

    for part in rest.split(',') {
        let spec_pos = rest_pos + offset;
        if part.is_empty() {
            return Err(Error::EmptyVariable { pos: spec_pos });
        }

        let part_bytes = part.as_bytes();
        let modifier_at = part_bytes
            .iter()
            .position(|&byte| byte == b'*' || byte == b':')
            .unwrap_or(part.len());
        let name = &part[..modifier_at];

        if name.is_empty() {
            return Err(Error::InvalidName { pos: spec_pos });
        }

        let name_bytes = name.as_bytes();
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_pos });
        }
        if let Some(index) = name_bytes
            .iter()
            .position(|&byte| !is_name_continue(byte))
        {
            return Err(Error::InvalidName {
                pos: spec_pos + index,
            });
        }

        let modifier = if modifier_at < part.len() {
            let suffix = &part_bytes[modifier_at..];
            match suffix {
                [b'*'] => Modifier::Explode,
                [b':', first @ b'1'..=b'9', rest]
                    if rest.len() <= 3 && rest.iter().all(u8::is_ascii_digit) =>
                {
                    let mut number = (first - b'0') as usize;
                    for digit in rest {
                        number = number * 10 + (digit - b'0') as usize;
                    }
                    Modifier::Prefix(number)
                }
                _ => {
                    return Err(Error::InvalidModifier {
                        pos: spec_pos + modifier_at,
                    });
                }
            }
        } else {
            Modifier::None
        };

        if !seen.insert(name) {
            return Err(Error::DuplicateVariable { pos: spec_pos });
        }

        specs.push(Spec {
            name,
            modifier,
            pos: spec_pos,
        });
        offset += part.len() + 1;
    }

    Ok((operator, specs))
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut result = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%'
            && index + 2 < bytes.len()
            && bytes[index + 1].is_ascii_hexdigit()
            && bytes[index + 2].is_ascii_hexdigit()
        {
            result.push('%');
            result.push((bytes[index + 1] as char).to_ascii_uppercase());
            result.push((bytes[index + 2] as char).to_ascii_uppercase());
            index += 3;
            continue;
        }

        let unreserved =
            byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_char = reserved
            && matches!(
                byte,
                b':' | b'/'
                    | b'?'
                    | b'#'
                    | b'['
                    | b']'
                    | b'@'
                    | b'!'
                    | b'$'
                    | b'&'
                    | b'\''
                    | b'('
                    | b')'
                    | b'*'
                    | b'+'
                    | b','
                    | b';'
                    | b'='
            );

        if unreserved || reserved_char {
            result.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 0x0f) as usize] as char);
        }

        index += 1;
    }

    result
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn named_scalar(name: &str, value: &str, operator: char, reserved: bool) -> String {
    let encoded = encode(value, reserved);
    match operator {
        ';' if encoded.is_empty() => name.to_owned(),
        ';' | '?' | '&' => format!("{name}={encoded}"),
        _ => encoded,
    }
}

fn expand_expression(
    operator: char,
    specs: &[Spec<'_>],
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let reserved = matches!(operator, '+' | '#');
    let separator = match operator {
        '.' => ".",
        '/' => "/",
        ';' | '?' | '&' => ";",
        _ => ",",
    };

    let mut fields = Vec::new();

    for spec in specs {
        let Some(value) = lookup(spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.pos });
        }

        match value {
            Value::Scalar(original) => {
                let selected = match spec.modifier {
                    Modifier::Prefix(count) => original.chars().take(count).collect::<String>(),
                    _ => original.clone(),
                };
                fields.push(named_scalar(
                    spec.name,
                    &selected,
                    operator,
                    reserved,
                ));
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }

                if matches!(spec.modifier, Modifier::Explode) {
                    for item in items {
                        fields.push(named_scalar(spec.name, item, operator, reserved));
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(match operator {
                        ';' | '?' | '&' => {
                            named_scalar(spec.name, &joined, operator, reserved)
                        }
                        _ => joined,
                    });
                }
            }
            Value::Map(entries) => {
                let mut keys = std::collections::HashSet::new();
                for (key, _) in entries {
                    if !keys.insert(key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                }

                if entries.is_empty() {
                    continue;
                }

                if matches!(spec.modifier, Modifier::Explode) {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        if operator == ';' && encoded_value.is_empty() {
                            fields.push(encoded_key);
                        } else {
                            fields.push(format!("{encoded_key}={encoded_value}"));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(match operator {
                        ';' | '?' | '&' => {
                            named_scalar(spec.name, &joined, operator, reserved)
                        }
                        _ => joined,
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let prefix = match operator {
        '#' => "#",
        '.' => ".",
        '/' => "/",
        ';' => ";",
        '?' => "?",
        '&' => "&",
        _ => "",
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < template.len() {
        match template.as_bytes()[index] {
            b'{' => {
                let opening = index;
                let tail = &template[index + 1..];
                let Some(relative_close) = tail.find('}') else {
                    return Err(Error::Unclosed { pos: opening });
                };
                let closing = index + 1 + relative_close;

                if let Some(inner) = template[index + 1..closing].find('{') {
                    return Err(Error::NestedOpen {
                        pos: index + 1 + inner,
                    });
                }

                let content_start = index + 1;
                let parsed = parse_expression(
                    &template[content_start..closing],
                    content_start,
                    closing,
                )?;
                expressions.push((opening, closing + 1, parsed));
                index = closing + 1;
            }
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            _ => {
                let character = template[index..]
                    .chars()
                    .next()
                    .expect("index is within template");
                index += character.len_utf8();
            }
        }
    }

    let mut output = String::new();
    let mut cursor = 0;

    for (opening, after_close, (operator, specs)) in expressions {
        output.push_str(&template[cursor..opening]);
        output.push_str(&expand_expression(operator, &specs, vars)?);
        cursor = after_close;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

            operator = first as char;
            start = 1;
        } else if matches!(first, b'=' | b'!' | b'@' | b'|' | b'$' | b'^') {
            return Err(Error::UnsupportedOperator { pos: content_pos });
        }
    }

    let rest = &content[start..];
    let rest_pos = content_pos + start;
    if rest.is_empty() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specs = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut offset = 0;

    for part in rest.split(',') {
        let spec_pos = rest_pos + offset;
        if part.is_empty() {
            return Err(Error::EmptyVariable { pos: spec_pos });
        }

        let part_bytes = part.as_bytes();
        let modifier_at = part_bytes
            .iter()
            .position(|&byte| byte == b'*' || byte == b':')
            .unwrap_or(part.len());
        let name = &part[..modifier_at];

        if name.is_empty() {
            return Err(Error::InvalidName { pos: spec_pos });
        }

        let name_bytes = name.as_bytes();
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_pos });
        }
        if let Some(index) = name_bytes
            .iter()
            .position(|&byte| !is_name_continue(byte))
        {
            return Err(Error::InvalidName {
                pos: spec_pos + index,
            });
        }

        let modifier = if modifier_at < part.len() {
            let suffix = &part_bytes[modifier_at..];
            match suffix {
                [b'*'] => Modifier::Explode,
                [b':', first @ b'1'..=b'9', digits]
                    if digits.len() <= 3 && digits.iter().all(u8::is_ascii_digit) =>
                {
                    let mut number = (first - b'0') as usize;
                    for digit in *digits {
                        number = number * 10 + (*digit - b'0') as usize;
                    }
                    Modifier::Prefix(number)
                }
                _ => {
                    return Err(Error::InvalidModifier {
                        pos: spec_pos + modifier_at,
                    });
                }
            }
        } else {
            Modifier::None
        };

        if !seen.insert(name) {
            return Err(Error::DuplicateVariable { pos: spec_pos });
        }

        specs.push(Spec {
            name,
            modifier,
            pos: spec_pos,
        });
        offset += part.len() + 1;
    }

    Ok((operator, specs))
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut result = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%'
            && index + 2 < bytes.len()
            && bytes[index + 1].is_ascii_hexdigit()
            && bytes[index + 2].is_ascii_hexdigit()
        {
            result.push('%');
            result.push((bytes[index + 1] as char).to_ascii_uppercase());
            result.push((bytes[index + 2] as char).to_ascii_uppercase());
            index += 3;
            continue;
        }

        let unreserved =
            byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_char = reserved
            && matches!(
                byte,
                b':' | b'/'
                    | b'?'
                    | b'#'
                    | b'['
                    | b']'
                    | b'@'
                    | b'!'
                    | b'$'
                    | b'&'
                    | b'\''
                    | b'('
                    | b')'
                    | b'*'
                    | b'+'
                    | b','
                    | b';'
                    | b'='
            );

        if unreserved || reserved_char {
            result.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 0x0f) as usize] as char);
        }

        index += 1;
    }

    result
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn named_scalar(name: &str, value: &str, operator: char, reserved: bool) -> String {
    let encoded = encode(value, reserved);
    match operator {
        ';' if encoded.is_empty() => name.to_owned(),
        ';' | '?' | '&' => format!("{name}={encoded}"),
        _ => encoded,
    }
}

fn expand_expression(
    operator: char,
    specs: &[Spec<'_>],
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let reserved = matches!(operator, '+' | '#');
    let separator = match operator {
        '.' => ".",
        '/' => "/",
        ';' | '?' | '&' => ";",
        _ => ",",
    };

    let mut fields = Vec::new();

    for spec in specs {
        let Some(value) = lookup(spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.pos });
        }

        match value {
            Value::Scalar(original) => {
                let selected = match spec.modifier {
                    Modifier::Prefix(count) => original.chars().take(count).collect::<String>(),
                    _ => original.clone(),
                };
                fields.push(named_scalar(spec.name, &selected, operator, reserved));
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }

                if matches!(spec.modifier, Modifier::Explode) {
                    for item in items {
                        fields.push(named_scalar(spec.name, item, operator, reserved));
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(match operator {
                        ';' | '?' | '&' => {
                            named_scalar(spec.name, &joined, operator, reserved)
                        }
                        _ => joined,
                    });
                }
            }
            Value::Map(entries) => {
                let mut keys = std::collections::HashSet::new();
                for (key, _) in entries {
                    if !keys.insert(key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                }

                if entries.is_empty() {
                    continue;
                }

                if matches!(spec.modifier, Modifier::Explode) {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        if operator == ';' && encoded_value.is_empty() {
                            fields.push(encoded_key);
                        } else {
                            fields.push(format!("{encoded_key}={encoded_value}"));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(match operator {
                        ';' | '?' | '&' => {
                            named_scalar(spec.name, &joined, operator, reserved)
                        }
                        _ => joined,
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let prefix = match operator {
        '#' => "#",
        '.' => ".",
        '/' => "/",
        ';' => ";",
        '?' => "?",
        '&' => "&",
        _ => "",
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < template.len() {
        match template.as_bytes()[index] {
            b'{' => {
                let opening = index;
                let tail = &template[index + 1..];
                let Some(relative_close) = tail.find('}') else {
                    return Err(Error::Unclosed { pos: opening });
                };
                let closing = index + 1 + relative_close;

                if let Some(inner) = template[index + 1..closing].find('{') {
                    return Err(Error::NestedOpen {
                        pos: index + 1 + inner,
                    });
                }

                let content_start = index + 1;
                let parsed = parse_expression(
                    &template[content_start..closing],
                    content_start,
                    closing,
                )?;
                expressions.push((opening, closing + 1, parsed));
                index = closing + 1;
            }
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            _ => {
                let character = template[index..]
                    .chars()
                    .next()
                    .expect("index is within template");
                index += character.len_utf8();
            }
        }
    }

    let mut output = String::new();
    let mut cursor = 0;

    for (opening, after_close, (operator, specs)) in expressions {
        output.push_str(&template[cursor..opening]);
        output.push_str(&expand_expression(operator, &specs, vars)?);
        cursor = after_close;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}
