fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes.iter().all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
        && bytes.last() != Some(&b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort_by_key(|c| constraint_rank(*c));
    constraints.dedup();
    constraints
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical(cs.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|c| match *c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    })
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

struct Search<'a> {
    packages: &'a [Package],
    package_indices: std::collections::BTreeMap<String, usize>,
    locks: &'a [Lock],
    attempts: usize,
    max_attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<(usize, Vec<Selection>)>,
}

impl<'a> Search<'a> {
    fn visit(&mut self, selected: &mut std::collections::BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
            std::collections::BTreeMap::new();

        for (name, index) in &self.package_indices {
            let _ = (name, index);
        }

        // Root requirements are supplied separately through the search object's seed map.
        // The caller stores them in an auxiliary map by inserting them before recursion.
        // This placeholder is replaced by roots in `root_constraints`.
        self.collect_active(selected, &mut active);

        let missing: Vec<String> = active
            .keys()
            .filter(|name| !self.package_indices.contains_key(*name))
            .cloned()
            .collect();
        if let Some(package) = missing.first() {
            return Err(ResolveError::Missing {
                package: package.clone(),
            });
        }

        for (package, version) in selected.iter() {
            if let Some(cs) = active.get(package) {
                if !matches(*version, cs) {
                    return Err(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: canonical(cs.clone()),
                    });
                }
            }
        }

        let mut unselected: Vec<String> = active
            .keys()
            .filter(|name| !selected.contains_key(*name))
            .cloned()
            .collect();
        unselected.sort();

        for package in &unselected {
            let idx = self.package_indices[package];
            let cs = active.get(package).unwrap();
            let any = self.packages[idx]
                .releases
                .iter()
                .any(|r| matches(r.version, cs));
            if !any {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: canonical(cs.clone()),
                });
            }
        }

        if unselected.is_empty() {
            let solution: Vec<Selection> = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            let cost = self
                .locks
                .iter()
                .filter(|lock| selected.get(&lock.package) != Some(&lock.version))
                .count();
            if self
                .best
                .as_ref()
                .map_or(true, |(best_cost, best_solution)| {
                    cost < *best_cost
                        || (cost == *best_cost
                            && self.solution_better(&solution, best_solution))
                })
            {
                self.best = Some((cost, solution));
            }
            return Ok(());
        }

        let package = unselected[0].clone();
        let idx = self.package_indices[&package];
        let cs = active.get(&package).unwrap();
        let mut candidates: Vec<Version> = self.packages[idx]
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| matches(*v, cs))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.iter().find(|l| l.package == package) {
            if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
                let locked = candidates.remove(pos);
                candidates.insert(0, locked);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            match self.visit(selected) {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    selected.remove(&package);
                    return Err(ResolveError::Limit { attempts });
                }
                Err(error) => {
                    if self.first_error.is_none() {
                        self.first_error = Some(error);
                    }
                }
            }
            selected.remove(&package);
        }
        Ok(())
    }

    fn solution_better(&self, a: &[Selection], b: &[Selection]) -> bool {
        let av: std::collections::BTreeMap<&str, Version> =
            a.iter().map(|s| (s.package.as_str(), s.version)).collect();
        let bv: std::collections::BTreeMap<&str, Version> =
            b.iter().map(|s| (s.package.as_str(), s.version)).collect();
        for name in self.package_indices.keys() {
            match (av.get(name.as_str()), bv.get(name.as_str())) {
                (Some(x), Some(y)) if x != y => return x > y,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                _ => {}
            }
        }
        false
    }

    fn collect_active(
        &self,
        selected: &std::collections::BTreeMap<String, Version>,
        active: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
    ) {
        // Root constraints are in a thread-local-free field maintained by root_requirements.
        ROOTS.with(|roots| {
            for req in roots.borrow().iter() {
                active.entry(req.package.clone()).or_default().push(req.constraint);
            }
        });
        for (package, version) in selected {
            if let Some(&idx) = self.package_indices.get(package) {
                if let Some(release) = self.packages[idx]
                    .releases
                    .iter()
                    .find(|r| r.version == *version)
                {
                    for req in &release.deps {
                        active.entry(req.package.clone()).or_default().push(req.constraint);
                    }
                }
            }
        }
        for constraints in active.values_mut() {
            *constraints = canonical(std::mem::take(constraints));
        }
    }
}

    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if let Some(error) = invalid_name(&root.package, format!("root[{}]", i)) {
            return Err(error);
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if let Some(error) = invalid_name(&package.name, format!("catalog[{}]", i)) {
            return Err(error);
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if let Some(error) = invalid_name(
                    &dep.package,
                    format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                ) {
                    return Err(error);
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if let Some(error) = invalid_name(&lock.package, format!("lock[{}]", i)) {
            return Err(error);
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    ROOTS.with(|r| *r.borrow_mut() = roots.to_vec());

    let package_indices = catalog
        .iter()
        .enumerate()
        .map(|(i, p)| (p.name.clone(), i))
        .collect();

    let mut search = Search {
        packages: catalog,
        package_indices,
        locks,
        attempts: 0,
        max_attempts,
        first_error: None,
        best: None,
    };
    let mut selected = std::collections::BTreeMap::new();
    match search.visit(&mut selected) {
        Err(ResolveError::Limit { attempts }) => Err(ResolveError::Limit { attempts }),
        Err(error) => Err(error),
        Ok(()) => {
            if let Some((_, chosen)) = search.best {
                let final_constraints = {
                    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
                        std::collections::BTreeMap::new();
                    for req in roots {
                        active.entry(req.package.clone()).or_default().push(req.constraint);
                    }
                    for selection in &chosen {
                        if let Some(&idx) = search.package_indices.get(&selection.package) {
                            if let Some(release) = catalog[idx]
                                .releases
                                .iter()
                                .find(|r| r.version == selection.version)
                            {
                                for req in &release.deps {
                                    active.entry(req.package.clone()).or_default().push(req.constraint);
                                }
                            }
                        }
                    }
                    for cs in active.values_mut() {
                        *cs = canonical(std::mem::take(cs));
                    }
                    active
                };

                let chosen_map: std::collections::BTreeMap<String, Version> =
                    chosen.iter().map(|s| (s.package.clone(), s.version)).collect();
                let lock_map: std::collections::BTreeMap<String, Version> =
                    locks.iter().map(|l| (l.package.clone(), l.version)).collect();
                let mut names = std::collections::BTreeSet::new();
                names.extend(chosen_map.keys().cloned());
                names.extend(lock_map.keys().cloned());

                let mut changes = Vec::new();
                for package in names {
                    let before = lock_map.get(&package).copied();
                    let after = chosen_map.get(&package).copied();
                    if before == after {
                        continue;
                    }
                    let reason = match (before, after) {
                        (None, Some(_)) => ChangeReason::Added,
                        (Some(_), None) => ChangeReason::Removed,
                        (Some(old), Some(_)) => {
                            let exists = search.package_indices.get(&package).is_some_and(|idx| {
                                catalog[*idx].releases.iter().any(|r| r.version == old)
                            });
                            if !exists {
                                ChangeReason::Unavailable
                            } else if final_constraints
                                .get(&package)
                                .is_some_and(|cs| !matches(old, cs))
                            {
                                ChangeReason::Rejected
                            } else {
                                ChangeReason::Rebalanced
                            }
                        }
                        (None, None) => unreachable!(),
                    };
                    changes.push(Change {
                        package,
                        before,
                        after,
                        reason,
                    });
                }
                Ok(Resolution {
                    selected: chosen,
                    changes,
                })
            } else {
                Err(search.first_error.unwrap_or(ResolveError::Missing {
                    package: String::new(),
                }))
            }
        }
    }
}

fn main() {}
