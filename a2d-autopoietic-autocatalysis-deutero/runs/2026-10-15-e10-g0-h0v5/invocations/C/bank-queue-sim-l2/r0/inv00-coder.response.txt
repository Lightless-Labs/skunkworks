```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone, Copy)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    occupied: bool,
    idle_since: u64,
    finish: Option<(u64, u32)>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|prior| prior.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|prior| prior.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            occupied: false,
            idle_since: server.opens,
            finish: None,
        })
        .collect();

    let mut arrival_order: Vec<usize> = (0..customers.len()).collect();
    arrival_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut arrival_pos = 0;
    let mut finished_customers = 0;
    let mut time = 0;
    let mut events = Vec::new();

    while finished_customers < customers.len() || states.iter().any(|s| !s.opened) {
        // Complete all work due at this tick, in ascending server ID order.
        let mut due: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| {
                s.finish
                    .filter(|(finish_time, _)| *finish_time == time)
                    .map(|_| i)
            })
            .collect();
        due.sort_by_key(|&i| states[i].id);
        for i in due {
            let (finish_time, customer_id) = states[i].finish.take().unwrap();
            states[i].occupied = false;
            states[i].idle_since = finish_time;
            finished_customers += 1;
            events.push(Event::Finish {
                time,
                customer: customer_id,
                server: states[i].id,
            });
        }

        // Open servers in descending ID order.
        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (!s.opened && s.opens == time).then_some(i))
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].id));
        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time,
                server: states[i].id,
            });
        }

        // Emit the complete arrival cohort in descending customer ID order.
        while arrival_pos < arrival_order.len()
            && customers[arrival_order[arrival_pos]].arrival == time
        {
            let customer = &customers[arrival_order[arrival_pos]];
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            arrival_pos += 1;
        }

        let mut waiting_start = arrival_pos;
        // Include arrivals from this tick in the FIFO. Since they have been
        // emitted, find the earliest not-yet-dispatched arrival.
        if waiting_start > 0 {
            waiting_start = 0;
        }
        let mut waiting: Vec<usize> = (0..arrival_order.len())
            .filter(|&pos| {
                let customer = &customers[arrival_order[pos]];
                customer.arrival <= time
                    && !events.iter().any(|event| {
                        matches!(event, Event::Start { customer: id, .. } if *id == customer.id)
                    })
            })
            .collect();
        waiting.sort_by(|&a, &b| a.cmp(&b));

        loop {
            let Some(&customer_pos) = waiting.first() else {
                break;
            };
            let customer = &customers[arrival_order[customer_pos]];

            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, s)| s.opened && !s.occupied)
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };
            let server = states[server_index];

            let service_end = time.checked_add(customer.service).ok_or(Error::TimeOverflow {
                customer: customer.id,
                server: server.id,
                time,
            })?;
            let finish = service_end.checked_add(server.overhead).ok_or(Error::TimeOverflow {
                customer: customer.id,
                server: server.id,
                time,
            })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server.id,
                finish,
            });
            waiting.remove(0);

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server.id,
                });
                finished_customers += 1;
                states[server_index].idle_since = time;
            } else {
                states[server_index].occupied = true;
                states[server_index].finish = Some((finish, customer.id));
            }
        }

        if finished_customers >= customers.len() && states.iter().all(|s| s.opened) {
            break;
        }

        let next_arrival = arrival_order
            .get(arrival_pos)
            .map(|&i| customers[i].arrival);
        let next_open = states
            .iter()
            .filter(|s| !s.opened)
            .map(|s| s.opens)
            .min();
        let next_finish = states
            .iter()
            .filter_map(|s| s.finish.map(|(t, _)| t))
            .min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&t| t > time)
            .min();

        if let Some(next_time) = next_time {
            time = next_time;
        } else if finished_customers < customers.len() {
            // A customer may be waiting while all servers are still closed,
            // with an opening at the current tick already processed only if
            // the opening schedule has progressed. Any remaining time must
            // come from an unprocessed opening or finish.
            let next = [next_open, next_finish].into_iter().flatten().min();
            if let Some(next) = next {
                time = next;
            } else {
                break;
            }
        } else {
            break;
        }
    }

    Ok(events)
}

fn main() {}
```
