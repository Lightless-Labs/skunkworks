#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone, Debug)]
struct Candidate {
    cost: u64,
    marker_lines: usize,
    lines: Vec<String>,
    consumed: Vec<usize>,
    pieces: Vec<usize>,
}

fn better(a: &Candidate, b: &Candidate) -> bool {
    a.cost < b.cost
        || (a.cost == b.cost
            && (a.marker_lines < b.marker_lines
                || (a.marker_lines == b.marker_lines
                    && (a.lines.len() < b.lines.len()
                        || (a.lines.len() == b.lines.len()
                            && (a.consumed > b.consumed
                                || (a.consumed == b.consumed && a.lines < b.lines)))))))
}

fn solve(
    words: &[Word],
    wi: usize,
    offset: usize,
    first: bool,
    options: &WrapOptions,
    memo: &mut Vec<Vec<[Option<Candidate>; 2]>>,
) -> Option<Candidate> {
    if wi == words.len() {
        return Some(Candidate {
            cost: 0,
            marker_lines: 0,
            lines: Vec::new(),
            consumed: Vec::new(),
            pieces: Vec::new(),
        });
    }

    let slot = usize::from(first);
    if let Some(cached) = &memo[wi][offset][slot] {
        return Some(cached.clone());
    }

    let indent = if first {
        options.first_indent
    } else {
        options.rest_indent
    };
    let capacity = options.width - indent;
    let mut best: Option<Candidate> = None;

    let mut pieces: Vec<String> = Vec::new();
    let mut display_count = 0usize;
    let mut consumed_count = 0usize;
    let mut cur_wi = wi;
    let mut cur_off = offset;

    loop {
        let word = &words[cur_wi];
        let remaining = word.chars.len() - cur_off;
        let sep = usize::from(!pieces.is_empty());
        if display_count + sep >= capacity {
            break;
        }

        let room = capacity - display_count - sep;
        if remaining <= room {
            pieces.push(word.chars[cur_off..].iter().collect());
            display_count += sep + remaining;
            consumed_count += remaining;

            let line = format!("{}{}", " ".repeat(indent), pieces.join(" "));
            let gap = options.width - (indent + display_count);
            let final_line = cur_wi + 1 == words.len();
            let mut line_cost = (gap * gap) as u64;
            if final_line {
                line_cost /= 4;
            }

            if let Some(mut candidate) =
                solve(words, cur_wi + 1, 0, false, options, memo)
            {
                candidate.cost += line_cost;
                if !final_line && pieces.len() == 1 {
                    candidate.cost += 3;
                }
                candidate.lines.insert(0, line);
                candidate.consumed.insert(0, consumed_count);
                candidate.pieces.insert(0, pieces.len());
                if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                    best = Some(candidate);
                }
            }
        }

        for &marker in &word.markers {
            if marker <= cur_off || marker >= word.chars.len() {
                continue;
            }
            let prefix_len = marker - cur_off;
            let used = display_count + sep + prefix_len + 1;
            if used > capacity {
                continue;
            }
            let mut line = " ".repeat(indent);
            if !pieces.is_empty() {
                line.push_str(&pieces.join(" "));
                line.push(' ');
            }
            line.extend(word.chars[cur_off..marker].iter());
            line.push('-');

            if let Some(mut candidate) =
                solve(words, cur_wi, marker, false, options, memo)
            {
                let gap = options.width - (indent + used);
                candidate.cost +=
                    (gap * gap) as u64 + options.hyphen_cost as u64;
                if pieces.len() + 1 == 1 {
                    candidate.cost += 3;
                }
                candidate.marker_lines += 1;
                candidate.lines.insert(0, line);
                candidate.consumed.insert(0, consumed_count + prefix_len);
                candidate.pieces.insert(0, pieces.len() + 1);
                if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                    best = Some(candidate);
                }
            }
        }

        if pieces.is_empty() && remaining > capacity {
            let has_fitting_marker = word.markers.iter().any(|&marker| {
                marker > cur_off
                    && marker < word.chars.len()
                    && marker - cur_off + 1 <= capacity
            });
            if !has_fitting_marker {
                let take = capacity;
                let line = format!(
                    "{}{}",
                    " ".repeat(indent),
                    word.chars[cur_off..cur_off + take]
                        .iter()
                        .collect::<String>()
                );
                if let Some(mut candidate) =
                    solve(words, cur_wi, cur_off + take, false, options, memo)
                {
                    let gap = options.width - (indent + take);
                    candidate.cost +=
                        (gap * gap) as u64 + (2 * options.width + 1) as u64;
                    candidate.lines.insert(0, line);
                    candidate.consumed.insert(0, take);
                    candidate.pieces.insert(0, 1);
                    if best.as_ref().map_or(true, |b| better(&candidate, b)) {
                        best = Some(candidate);
                    }
                }
            }
        }

        if cur_wi + 1 >= words.len() {
            break;
        }
        if display_count + 1 >= capacity {
            break;
        }
        cur_wi += 1;
        cur_off = 0;
    }

    memo[wi][offset][slot] = best.clone();
    best
}

fn layout_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }

    let mut memo: Vec<Vec<[Option<Candidate>; 2]>> = words
        .iter()
        .map(|w| vec![[None, None]; w.chars.len() + 1])
        .collect();

    let Some(mut result) = solve(words, 0, 0, true, options, &mut memo) else {
        return Vec::new();
    };

    if options.justify {
        for i in 0..result.lines.len().saturating_sub(1) {
            let gaps = result.pieces[i].saturating_sub(1);
            if gaps == 0 {
                continue;
            }
            let count = result.lines[i].chars().count();
            let extra = options.width - count;
            let each = extra / gaps;
            let remainder = extra % gaps;
            let start = i % gaps;
            let indent = if i == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };

            let body: String = result.lines[i].chars().skip(indent).collect();
            let parts: Vec<&str> = body.split(' ').collect();
            let mut rebuilt = " ".repeat(indent);
            for (j, part) in parts.iter().enumerate() {
                if j > 0 {
                    let gap_index = j - 1;
                    let bonus =
                        usize::from((gap_index + gaps - start) % gaps < remainder);
                    rebuilt.push_str(&" ".repeat(1 + each + bonus));
                }
                rebuilt.push_str(part);
            }
            result.lines[i] = rebuilt;
        }
    }

    result.lines
}

fn parse_words(segment: &str) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    let flush = |words: &mut Vec<Word>,
                 chars: &mut Vec<char>,
                 markers: &mut Vec<usize>| {
        if !chars.is_empty() {
            words.push(Word {
                chars: std::mem::take(chars),
                markers: std::mem::take(markers),
            });
        }
    };

    for ch in segment.chars() {
        if ch == ' '
            || ch == '\t'
            || ch == '\n'
            || ch == '\r'
            || ch == '\u{00A0}'
        {
            flush(&mut words, &mut chars, &mut markers);
        } else if ch == '~' {
            markers.push(chars.len());
        } else {
            chars.push(ch);
        }
    }
    flush(&mut words, &mut chars, &mut markers);
    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let char_count = input.chars().count();
    if char_count > 512 {
        return Err(WrapError::InputTooLong { chars: char_count });
    }

    let mut byte_index = 0usize;
    for ch in input.chars() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index: byte_index });
        }
        byte_index += ch.len_utf8();
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    let is_sep = |c: char| {
        c == ' ' || c == '\t' || c == '\r' || c == '\n' || c == '\u{00A0}'
    };
    for i in 0..original.len() {
        if original[i].1 == '~' {
            let valid_left =
                i > 0 && original[i - 1].1 != '~' && !is_sep(original[i - 1].1);
            let valid_right = i + 1 < original.len()
                && original[i + 1].1 != '~'
                && !is_sep(original[i + 1].1);
            if !valid_left || !valid_right {
                return Err(WrapError::InvalidMarker {
                    index: original[i].0,
                });
            }
        }
    }

    let mut normalized = String::new();
    let mut iter = input.chars().peekable();
    while let Some(ch) = iter.next() {
        if ch == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut segment_start = 0usize;
    let mut i = 0usize;
    while i < chars.len() {
        if chars[i] == '\n' {
            let run_start = i;
            while i < chars.len() && chars[i] == '\n' {
                i += 1;
            }
            if i - run_start >= 2 {
                let segment: String =
                    chars[segment_start..run_start].iter().collect();
                output.extend(layout_paragraph(
                    &parse_words(&segment),
                    options,
                ));
                for _ in 1..(i - run_start) {
                    output.push(String::new());
                }
                segment_start = i;
            }
        } else {
            i += 1;
        }
    }

    let segment: String = chars[segment_start..].iter().collect();
    output.extend(layout_paragraph(&parse_words(&segment), options));
    Ok(output)
}

fn main() {}
