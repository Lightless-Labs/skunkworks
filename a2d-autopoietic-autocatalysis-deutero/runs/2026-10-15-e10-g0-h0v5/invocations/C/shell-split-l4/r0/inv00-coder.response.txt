#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find_map(|(key, value)| (*key == name).then_some(*value))
}

fn append_text(word: &mut Word, text: &str, active_glob: bool) {
    for c in text.chars() {
        if active_glob && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn split_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let bytes = input.as_bytes();
    let after_open = dollar + 2;
    let mut close = after_open;
    let mut found = None;

    while close < bytes.len() {
        if bytes[close] == b'\\' {
            close += 1;
            if close < bytes.len() {
                let c = input[close..].chars().next().unwrap();
                close += c.len_utf8();
            }
        } else if bytes[close] == b'}' {
            found = Some(close);
            break;
        } else {
            let c = input[close..].chars().next().unwrap();
            close += c.len_utf8();
        }
    }

    let close = found.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let body = &input[after_open..close];

    let name_end = body
        .char_indices()
        .find_map(|(i, c)| (!is_name_continue(c)).then_some(i))
        .unwrap_or(body.len());
    let name = &body[..name_end];

    if name.is_empty() || !name.chars().next().is_some_and(is_name_start) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let remainder = &body[name_end..];
    if remainder.is_empty() {
        return Ok((close + 1, lookup(name, env).unwrap_or("").to_string()));
    }

    if let Some(default) = remainder.strip_prefix(":-") {
        let selected = lookup(name, env).filter(|value| !value.is_empty());
        if let Some(value) = selected {
            Ok((close + 1, value.to_string()))
        } else {
            let mut decoded = String::new();
            let mut chars = default.chars();
            while let Some(c) = chars.next() {
                if c == '\\' {
                    if let Some(next) = chars.next() {
                        decoded.push(next);
                    } else {
                        decoded.push('\\');
                    }
                } else {
                    decoded.push(c);
                }
            }
            Ok((close + 1, decoded))
        }
    } else {
        Err(SplitError::InvalidExpansion { pos: dollar })
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.text.push(c);
                }
                i = next_i;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            word.text.push(next);
                        }
                        i = after_next;
                    } else {
                        word.text.push('\\');
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    } else if input[next_i..].starts_with('{') {
                        active = true;
                        let (end, value) = split_expansion(input, i, env)?;
                        append_text(&mut word, &value, false);
                        i = end;
                    } else if next_i < input.len()
                        && input[next_i..].chars().next().is_some_and(is_name_start)
                    {
                        active = true;
                        let mut end = next_i;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if is_name_continue(ch) {
                                end += ch.len_utf8();
                            } else {
                                break;
                            }
                        }
                        let name = &input[next_i..end];
                        let value = lookup(name, env).unwrap_or("");
                        append_text(&mut word, value, false);
                        i = end;
                    } else {
                        word.text.push('$');
                        i = next_i;
                    }
                } else {
                    word.text.push(c);
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        active = true;
                        word.text.push(next);
                        i = after_next;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    } else if input[next_i..].starts_with('{') {
                        active = true;
                        let (end, value) = split_expansion(input, i, env)?;
                        append_text(&mut word, &value, true);
                        i = end;
                    } else if next_i < input.len()
                        && input[next_i..].chars().next().is_some_and(is_name_start)
                    {
                        active = true;
                        let mut end = next_i;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if is_name_continue(ch) {
                                end += ch.len_utf8();
                            } else {
                                break;
                            }
                        }
                        let name = &input[next_i..end];
                        let value = lookup(name, env).unwrap_or("");
                        append_text(&mut word, value, true);
                        i = end;
                    } else {
                        active = true;
                        word.text.push('$');
                        i = next_i;
                    }
                } else {
                    active = true;
                    if matches!(c, '*' | '?' | '[') {
                        word.glob.push(word.text.len());
                    }
                    word.text.push(c);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn is_bare_safe(c: char) -> bool {
    c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c)
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let bare = if !word.is_empty() && !word.contains('\n') {
            Some(
                word.chars()
                    .map(|c| {
                        if is_bare_safe(c) {
                            c.to_string()
                        } else {
                            format!("\\{c}")
                        }
                    })
                    .collect::<String>(),
            )
        } else {
            None
        };

        let single = if !word.contains('\'') {
            Some(format!("'{word}'"))
        } else {
            None
        };

        let double = if !word.contains('`') {
            let mut result = String::from("\"");
            for c in word.chars() {
                if matches!(c, '\\' | '"' | '$') {
                    result.push('\\');
                }
                result.push(c);
            }
            result.push('"');
            Some(result)
        } else {
            None
        };

        let chosen = [bare, single, double]
            .into_iter()
            .flatten()
            .min_by_key(|candidate| candidate.len())
            .unwrap_or_else(|| {
                let mut fallback = String::from("'");
                for c in word.chars() {
                    if c == '\'' {
                        fallback.push_str("'\\''");
                    } else {
                        fallback.push(c);
                    }
                }
                fallback.push('\'');
                fallback
            });

        encoded.push(chosen);
    }

    Ok(encoded.join(" "))
}

fn main() {}
