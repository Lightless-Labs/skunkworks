use std::collections::BTreeSet;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [BTreeSet<u8>; 5],
    day_unrestricted: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn names(field: Field) -> Option<&'static [&'static str]> {
    match field {
        Field::Month => Some(&[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ]),
        Field::DayOfWeek => Some(&["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"]),
        _ => None,
    }
}

fn parse_base(
    text: &str,
    field: Field,
    offset: usize,
) -> Result<(u8, Option<usize>), ParseError> {
    let bytes = text.as_bytes();
    if bytes.iter().all(u8::is_ascii_digit) && !bytes.is_empty() {
        if bytes.len() > 1 && bytes[0] == b'0' {
            return Err(ParseError::LeadingZero { field, offset });
        }
        let mut value: u32 = 0;
        for &b in bytes {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or(ParseError::NumberOverflow { field, offset })?;
        }
        let (min, max) = domain(field);
        if value < min as u32 || value > max as u32 {
            return Err(ParseError::OutOfRange { field, offset });
        }
        return Ok((value as u8, None));
    }

    if let Some(list) = names(field) {
        if text.len() == 3 && text.is_ascii() {
            let upper = text.to_ascii_uppercase();
            if let Some(pos) = list.iter().position(|name| *name == upper) {
                return Ok((
                    if field == Field::Month {
                        (pos + 1) as u8
                    } else {
                        pos as u8
                    },
                    None,
                ));
            }
        }
    }

    Err(ParseError::InvalidSyntax { field, offset })
}

fn parse_number(text: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    let bytes = text.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value: u32 = 0;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<BTreeSet<u8>, ParseError> {
    for (offset, &b) in text.as_bytes().iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = domain(field);
    let mut selected = BTreeSet::new();
    let mut item_start = 0usize;

    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item_offset = item_start;
        let slash_count = item.bytes().filter(|&b| b == b'/').count();
        if slash_count > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }

        let (base, step_text, step_offset) = if let Some(slash) = item.find('/') {
            (&item[..slash], Some(&item[slash + 1..]), item_offset + slash + 1)
        } else {
            (item, None, 0)
        };

        let wildcard = base == "*";
        let hyphen_count = base.bytes().filter(|&b| b == b'-').count();
        let range_parts: Vec<&str> = if hyphen_count == 0 {
            vec![base]
        } else if hyphen_count == 1 {
            base.split('-').collect()
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        };

        if (wildcard && hyphen_count != 0)
            || (!wildcard && (range_parts.is_empty() || range_parts.iter().any(|s| s.is_empty())))
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }

        let mut parsed_range: Option<(u8, u8)> = None;
        if !wildcard {
            let start_off = item_offset;
            let (start, _) = parse_base(range_parts[0], field, start_off)?;
            let end = if range_parts.len() == 2 {
                let end_off = item_offset + range_parts[0].len() + 1;
                let (end, _) = parse_base(range_parts[1], field, end_off)?;
                if start > end {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_off,
                    });
                }
                end
            } else {
                start
            };
            parsed_range = Some((start, end));
        }

        let step = if let Some(step_text) = step_text {
            let n = parse_number(step_text, field, step_offset)?;
            let width = (max - min + 1) as u32;
            if n == 0 || n > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_offset,
                });
            }
            Some(n as u8)
        } else {
            None
        };

        let (start, end) = if wildcard {
            (min, max)
        } else {
            let (start, end) = parsed_range.unwrap();
            if range_parts.len() == 1 && step.is_some() {
                (start, max)
            } else {
                (start, end)
            }
        };

        if let Some(step) = step {
            let mut value = start as u16;
            while value <= end as u16 {
                selected.insert(value as u8);
                value += step as u16;
            }
        } else {
            for value in start..=end {
                selected.insert(value);
            }
        }

        item_start += item.len() + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.as_bytes().first() == Some(&b'@') {
        match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        input
    };

    let parts: Vec<&str> = expanded.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets: [BTreeSet<u8>; 5] = std::array::from_fn(|_| BTreeSet::new());
    for i in 0..5 {
        sets[i] = parse_field(parts[i], FIELDS[i])?;
    }

    Ok(Cron {
        sets,
        day_unrestricted: [parts[2] == "*", parts[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.sets[field_index(field)].iter().copied().collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut current = after;
        loop {
            if current.minute < 59 {
                current.minute += 1;
            } else {
                current.minute = 0;
                if current.hour < 23 {
                    current.hour += 1;
                } else {
                    current.hour = 0;
                    if !advance_day(&mut current) {
                        return Ok(None);
                    }
                }
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut current = before;
        loop {
            if current.minute > 0 {
                current.minute -= 1;
            } else {
                current.minute = 59;
                if current.hour > 0 {
                    current.hour -= 1;
                } else {
                    current.hour = 23;
                    if !retreat_day(&mut current) {
                        return Ok(None);
                    }
                }
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }

        let mut fires = Vec::new();
        let mut cursor = start;
        while cursor < end {
            if self.matches_valid(cursor) {
                if fires.len() == cap {
                    return Ok(FireWindow {
                        fires,
                        truncated: true,
                    });
                }
                fires.push(cursor);
            }
            if !advance_minute(&mut cursor) {
                break;
            }
        }
        Ok(FireWindow {
            fires,
            truncated: false,
        })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::with_capacity(5);
        for i in 0..5 {
            let field = FIELDS[i];
            let (min, max) = domain(field);
            let full = self.sets[i].len() == (max - min + 1) as usize;
            let value = if i == 2 && self.day_unrestricted[0] {
                "*".to_string()
            } else if i == 4 && self.day_unrestricted[1] {
                "*".to_string()
            } else if full && i != 2 && i != 4 {
                "*".to_string()
            } else if full {
                "*/1".to_string()
            } else {
                self.sets[i]
                    .iter()
                    .map(u8::to_string)
                    .collect::<Vec<_>>()
                    .join(",")
            };
            fields.push(value);
        }
        fields.join(" ")
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.sets[0].contains(&at.minute)
            || !self.sets[1].contains(&at.hour)
            || !self.sets[3].contains(&at.month)
        {
            return false;
        }

        let dom = self.sets[2].contains(&at.day);
        let dow = self.sets[4].contains(&weekday(at));
        match self.day_unrestricted {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    let invalid = if !(2000..=2399).contains(&at.year) {
        Some(DatePart::Year)
    } else if !(1..=12).contains(&at.month) {
        Some(DatePart::Month)
    } else if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        Some(DatePart::Day)
    } else if at.hour > 23 {
        Some(DatePart::Hour)
    } else if at.minute > 59 {
        Some(DatePart::Minute)
    } else {
        None
    };
    match invalid {
        Some(component) => Err(DateError::InvalidDateTime { component }),
        None => Ok(()),
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(at: CivilDateTime) -> u8 {
    let mut days = 0u32;
    for year in 2000..at.year {
        days += if leap(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        days += days_in_month(at.year, month) as u32;
    }
    days += at.day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn advance_day(at: &mut CivilDateTime) -> bool {
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        true
    } else if at.year < 2399 {
        at.month = 1;
        at.year += 1;
        true
    } else {
        false
    }
}

fn retreat_day(at: &mut CivilDateTime) -> bool {
    if at.day > 1 {
        at.day -= 1;
        return true;
    }
    if at.month > 1 {
        at.month -= 1;
    } else if at.year > 2000 {
        at.year -= 1;
        at.month = 12;
    } else {
        return false;
    }
    at.day = days_in_month(at.year, at.month);
    true
}

fn advance_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        true
    } else {
        at.minute = 0;
        if at.hour < 23 {
            at.hour += 1;
            true
        } else {
            at.hour = 0;
            advance_day(at)
        }
    }
}

fn main() {}
