#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let separators: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'@').then_some(index))
        .collect();

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let separator = separators[0];
    let local = &bytes[..separator];
    let raw_domain = &bytes[separator + 1..];

    if local.is_empty() {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    if local.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (index, &byte) in local.iter().enumerate() {
        let original_at = index;

        if byte == b'.' {
            if index == 0
                || index + 1 == local.len()
                || local[index - 1] == b'.'
            {
                return Err(EmailError::LocalDot { at: original_at });
            }
        } else if !byte.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&byte)
        {
            return Err(EmailError::LocalChar { at: original_at });
        }
    }

    let domain_start = separator + 1;
    let root_dot = raw_domain.last() == Some(&b'.');
    let domain = if root_dot {
        &raw_domain[..raw_domain.len() - 1]
    } else {
        raw_domain
    };

    if domain.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let removed_dot_at = domain_start + domain.len();
    let mut label_starts = Vec::new();
    let mut label_start = 0;

    for index in 0..=domain.len() {
        if index == domain.len() || domain[index] == b'.' {
            let label = &domain[label_start..index];
            if label.is_empty() {
                let at = if index == domain.len() && root_dot {
                    removed_dot_at
                } else {
                    domain_start + label_start
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let label_at = domain_start + label_start;
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong { at: label_at + 63 });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: label_at });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: label_at + label.len() - 1,
                });
            }
            for (offset, &byte) in label.iter().enumerate() {
                if !byte.is_ascii_alphanumeric() && byte != b'-' {
                    return Err(EmailError::DomainChar {
                        at: label_at + offset,
                    });
                }
            }

            label_starts.push((label_start, index));
            label_start = index + 1;
        }
    }

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = *label_starts.last().unwrap();
    let last_label = &domain[last_start..last_end];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let normalized_domain = domain
        .iter()
        .map(|&byte| byte.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local: String::from_utf8(local.to_vec()).unwrap(),
        domain: normalized_domain,
        quoted: false,
    })
}

fn main() {}
