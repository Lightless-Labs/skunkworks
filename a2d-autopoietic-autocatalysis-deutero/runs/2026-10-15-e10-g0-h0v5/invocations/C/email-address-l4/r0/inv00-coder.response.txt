#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if i + 2 >= bytes.len() || bytes[i] != b'\r' || bytes[i + 1] != b'\n' || !is_hws(bytes[i + 2]) {
        return None;
    }
    let mut j = i + 2;
    while j < bytes.len() && is_hws(bytes[j]) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let n = input.len();
    let mut out = Vec::new();
    let mut map = Vec::new();
    let mut runs = Vec::new();
    let mut i = 0;
    let mut protected = false;
    let mut quote_open = None;
    let mut quote_escaped = false;

    while i < n {
        if protected {
            let b = input[i];
            out.push(b);
            map.push(i);
            if quote_escaped {
                quote_escaped = false;
            } else if b == b'\\' {
                quote_escaped = true;
            } else if b == b'"' {
                protected = false;
            }
            i += 1;
            continue;
        }

        if !out.is_empty() && out[0] == b'"' {
            if out.iter().all(|_| true) {
                // Quote protection is active only until its first closing quote.
                let closed = out.len() >= 2 && {
                    let mut k = 1;
                    let mut found = false;
                    while k < out.len() {
                        if out[k] == b'\\' {
                            k += 2;
                        } else if out[k] == b'"' {
                            found = true;
                            break;
                        } else {
                            k += 1;
                        }
                    }
                    found
                };
                if !closed {
                    protected = true;
                    quote_open = Some(map[0]);
                }
            }
        }

        // Once the initial quote has closed, later bytes are ordinary CFWS syntax.
        if protected {
            continue;
        }

        let b = input[i];
        if b == b'\r' || b == b'\n' {
            let end = is_fold(input, i).ok_or(EmailError::BadFold { at: i })?;
            let start = i;
            i = end;
            if let Some(last) = runs.last_mut() {
                if last.end == start {
                    last.end = end;
                    continue;
                }
            }
            runs.push(Run { start, end });
            continue;
        }
        if is_hws(b) {
            let start = i;
            while i < n && is_hws(input[i]) {
                i += 1;
            }
            if let Some(last) = runs.last_mut() {
                if last.end == start {
                    last.end = i;
                    continue;
                }
            }
            runs.push(Run { start, end: i });
            continue;
        }
        if b == b'(' {
            let start = i;
            let mut depth = 0usize;
            let outer = i;
            while i < n {
                let c = input[i];
                if c == b'\\' {
                    if i + 1 >= n || !(0x20..=0x7e).contains(&input[i + 1]) {
                        return Err(EmailError::CommentEscape { at: i });
                    }
                    i += 2;
                } else if c == b'(' {
                    depth += 1;
                    if depth > 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    i += 1;
                } else if c == b')' {
                    depth -= 1;
                    i += 1;
                    if depth == 0 {
                        break;
                    }
                } else if c == b'\r' || c == b'\n' {
                    let end = is_fold(input, i).ok_or(EmailError::BadFold { at: i })?;
                    i = end;
                } else if c == b'\t' || (0x20..=0x7e).contains(&c) {
                    i += 1;
                } else {
                    return Err(EmailError::CommentChar { at: i });
                }
            }
            if depth != 0 {
                return Err(EmailError::UnclosedComment { at: outer });
            }
            if let Some(last) = runs.last_mut() {
                if last.end == start {
                    last.end = i;
                    continue;
                }
            }
            runs.push(Run { start, end: i });
            continue;
        }
        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        out.push(b);
        map.push(i);
        i += 1;

        if out.len() == 1 && b == b'"' {
            protected = true;
            quote_open = Some(i - 1);
        }
    }

    let _ = quote_open;
    for run in &runs {
        let before = out.iter().zip(&map).find(|(_, p)| **p >= run.start);
        let after_idx = map.iter().rposition(|p| *p < run.start);
        let legal_start = before.is_none();
        let legal_end = after_idx.is_none();
        let left_at = after_idx.map(|idx| out[idx] == b'@').unwrap_or(false);
        let right_at = before.map(|(b, _)| *b == b'@').unwrap_or(false);
        if !(legal_start || legal_end || left_at || right_at) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    Ok((out, map, runs))
}

fn original(map: &[usize], idx: usize, input_len: usize) -> usize {
    map.get(idx).copied().unwrap_or(input_len)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut prev_dot = true;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if prev_dot || i + 1 == bytes.len() {
                return false;
            }
            prev_dot = true;
        } else if b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
            prev_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_impl(input: &str, policy: Option<&Policy>) -> Result<(Address, usize), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = raw.iter().position(|b| *b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }
    let (bytes, map, _) = cleanup(raw)?;
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: original(&map, 0, raw.len()) });
        }
    }

    let mut ats = Vec::new();
    let mut i = 0;
    if let Some(qe) = quote_end {
        i = qe + 1;
    }
    while i < bytes.len() {
        if bytes[i] == b'@' {
            ats.push(i);
        }
        i += 1;
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt { at: original(&map, ats[1], raw.len()) });
    }
    let at = ats[0];
    if at == 0 {
        return Err(EmailError::EmptyLocal { at: original(&map, at, raw.len()) });
    }
    let local_bytes = &bytes[..at];
    let mut local = Vec::new();
    let quoted;
    let mut origins = Vec::new();

    if local_bytes[0] == b'"' {
        let qe = quote_end.unwrap();
        if qe + 1 != at {
            return Err(EmailError::TrailingLocal { at: original(&map, qe + 1, raw.len()) });
        }
        quoted = true;
        let mut j = 1;
        while j < qe {
            if local_bytes[j] == b'\\' {
                if j + 1 >= qe || ![b'"', b'\\', b'@'].contains(&local_bytes[j + 1]) {
                    return Err(EmailError::BadEscape { at: original(&map, j, raw.len()) });
                }
                local.push(local_bytes[j + 1]);
                origins.push(original(&map, j, raw.len()));
                j += 2;
            } else {
                let b = local_bytes[j];
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: original(&map, j, raw.len()) });
                }
                local.push(b);
                origins.push(original(&map, j, raw.len()));
                j += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: original(&map, qe, raw.len()) });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        quoted = false;
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: original(&map, 64, raw.len()) });
        }
        let mut prev_dot = false;
        for (j, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if j == 0 || prev_dot || j + 1 == local_bytes.len() {
                    return Err(EmailError::LocalDot { at: original(&map, j, raw.len()) });
                }
                prev_dot = true;
            } else {
                prev_dot = false;
                if !(b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)) {
                    return Err(EmailError::LocalChar { at: original(&map, j, raw.len()) });
                }
            }
            local.push(b);
            origins.push(original(&map, j, raw.len()));
        }
    }

    let domain_start = at + 1;
    let mut dom_end = bytes.len();
    let mut root_pos = None;
    if dom_end > domain_start && bytes[dom_end - 1] == b'.' {
        root_pos = Some(original(&map, dom_end - 1, raw.len()));
        dom_end -= 1;
    }
    if dom_end == domain_start {
        return Err(EmailError::EmptyDomain { at: original(&map, domain_start, raw.len()) });
    }
    if dom_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong { at: original(&map, domain_start + 253, raw.len()) });
    }

    let domain = &bytes[domain_start..dom_end];
    let mut labels = Vec::new();
    let mut start = 0;
    for k in 0..=domain.len() {
        if k == domain.len() || domain[k] == b'.' {
            let end = k;
            if start == end {
                let pos = if start == domain.len() {
                    root_pos.unwrap_or_else(|| original(&map, domain_start + start, raw.len()))
                } else {
                    original(&map, domain_start + start, raw.len())
                };
                return Err(EmailError::EmptyLabel { at: pos });
            }
            let len = end - start;
            if len > 63 {
                return Err(EmailError::LabelTooLong { at: original(&map, domain_start + start + 63, raw.len()) });
            }
            if domain[start] == b'-' {
                return Err(EmailError::LabelHyphen { at: original(&map, domain_start + start, raw.len()) });
            }
            if domain[end - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: original(&map, domain_start + end - 1, raw.len()) });
            }
            for q in start..end {
                let b = domain[q];
                if !(b.is_ascii_alphanumeric() || b == b'-') {
                    return Err(EmailError::DomainChar { at: original(&map, domain_start + q, raw.len()) });
                }
            }
            labels.push((start, end));
            start = k + 1;
        }
    }
    if labels.len() < 2 && !policy.map(|p| p.allow_single_label).unwrap_or(false) {
        return Err(EmailError::DomainNeedsDot { at: original(&map, domain_start, raw.len()) });
    }
    let (ls, le) = *labels.last().unwrap();
    if domain[ls..le].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: original(&map, domain_start + ls, raw.len()) });
    }

    let mut result_local = String::from_utf8(local.clone()).unwrap();
    let mut result_quoted = quoted;
    let mut strip_position = None;
    if let Some(p) = policy {
        if p.local_case == LocalCase::Lower {
            for b in &mut local {
                b.make_ascii_lowercase();
            }
            result_local = String::from_utf8(local.clone()).unwrap();
        }
        if p.strip_subaddress {
            if let Some(pos) = local.iter().position(|b| *b == b'+') {
                strip_position = Some(origins[pos]);
                local.truncate(pos);
                origins.truncate(pos);
                result_local = String::from_utf8(local.clone()).unwrap();
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyNormalizedLocal {
                at: strip_position.unwrap_or(original(&map, 0, raw.len())),
            });
        }
        let safe = valid_dot_atom(&local);
        if !safe {
            result_quoted = true;
        } else if p.unquote_safe {
            result_quoted = false;
        }
    }

    let domain_string = domain.iter().map(|b| (*b as char).to_ascii_lowercase()).collect();
    Ok((Address { local: result_local, domain: domain_string, quoted: result_quoted }, 0))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_impl(input, None).map(|v| v.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_impl(input, Some(policy)).map(|v| v.0)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' {
                out.push('\\');
            }
            out.push(c);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (idx, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(e) => errors.push((idx, e)),
            Ok(address) => {
                if let Some((_, _, first)) = keys.iter().find(|(l, d, _)| l == &address.local && d == &address.domain) {
                    duplicate_of[idx] = Some(*first);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), idx));
                    source_indices.push(idx);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult { addresses, source_indices, duplicate_of, errors }
}
