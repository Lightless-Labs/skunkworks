const ADLER_MOD: u32 = 65521;

fn crc32_step(mut crc: u32, byte: u8) -> u32 {
    crc ^= byte as u32;
    for _ in 0..8 {
        crc = (crc >> 1) ^ if crc & 1 != 0 { 0xedb88320 } else { 0 };
    }
    crc
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffff;
    for &byte in data {
        crc = crc32_step(crc, byte);
    }
    crc ^ 0xffff_ffff
}

fn adler_update(a: &mut u32, b: &mut u32, byte: u8) {
    *a = (*a + byte as u32) % ADLER_MOD;
    *b = (*b + *a) % ADLER_MOD;
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1;
    let mut b = 0;
    for &byte in data {
        adler_update(&mut a, &mut b, byte);
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            crc = crc32_step(crc, byte);
            adler_update(&mut a, &mut b, byte);
        }
    }
    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    a: u32,
    b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc = crc32_step(self.crc, byte);
            adler_update(&mut self.a, &mut self.b, byte);
        }
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        let crc = parse_hex_u32(bytes, 1, 9)?;
        let adler = parse_hex_u32(bytes, 11, 19)?;
        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn hex_nibble(byte: u8, offset: usize) -> Result<u32, SnapshotError> {
    match byte {
        b'0'..=b'9' => Ok((byte - b'0') as u32),
        b'a'..=b'f' => Ok((byte - b'a' + 10) as u32),
        _ => Err(SnapshotError::Hex { offset }),
    }
}

fn parse_hex_u32(
    bytes: &[u8],
    start: usize,
    end: usize,
) -> Result<u32, SnapshotError> {
    let mut value = 0u32;
    for offset in start..end {
        value = (value << 4) | hex_nibble(bytes[offset], offset)?;
    }
    Ok(value)
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

fn fletcher32_word(a: &mut u32, b: &mut u32, word: u16) {
    *a = (*a + word as u32) % 65535;
    *b = (*b + *a) % 65535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0;
    let mut b = 0;
    let mut chunks = data.chunks_exact(2);
    for chunk in &mut chunks {
        fletcher32_word(&mut a, &mut b, u16::from_be_bytes([chunk[0], chunk[1]]));
    }
    if let Some(&last) = chunks.remainder().first() {
        fletcher32_word(&mut a, &mut b, (last as u16) << 8);
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0;
    let mut b = 0;
    let mut pending = None;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending.take() {
                fletcher32_word(&mut a, &mut b, ((high as u16) << 8) | byte as u16);
            } else {
                pending = Some(byte);
            }
        }
    }
    if let Some(high) = pending {
        fletcher32_word(&mut a, &mut b, (high as u16) << 8);
    }
    Ok((b << 16) | a)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut crc: u16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    crc = (crc << 1) ^ if crc & 0x8000 != 0 { 0x1021 } else { 0 };
                }
            }
            Crc16::X25 => {
                crc ^= byte as u16;
                for _ in 0..8 {
                    crc = (crc >> 1) ^ if crc & 1 != 0 { 0x8408 } else { 0 };
                }
            }
        }
    }
    if matches!(variant, Crc16::X25) {
        crc ^ 0xffff
    } else {
        crc
    }
}

#[derive(Clone, Copy)]
struct Accumulators {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
    f16_a: u32,
    f16_b: u32,
    f32_a: u32,
    f32_b: u32,
    f32_pending: Option<u8>,
    crc16: u16,
}

impl Accumulators {
    fn new(variant: Crc16) -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
            f16_a: 0,
            f16_b: 0,
            f32_a: 0,
            f32_b: 0,
            f32_pending: None,
            crc16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
        }
    }

    fn update(&mut self, byte: u8, variant: Crc16) {
        self.crc = crc32_step(self.crc, byte);
        adler_update(&mut self.adler_a, &mut self.adler_b, byte);

        self.f16_a = (self.f16_a + byte as u32) % 255;
        self.f16_b = (self.f16_b + self.f16_a) % 255;

        if let Some(high) = self.f32_pending.take() {
            fletcher32_word(
                &mut self.f32_a,
                &mut self.f32_b,
                ((high as u16) << 8) | byte as u16,
            );
        } else {
            self.f32_pending = Some(byte);
        }

        match variant {
            Crc16::Xmodem | Crc16::False => {
                self.crc16 ^= (byte as u16) << 8;
                for _ in 0..8 {
                    self.crc16 = (self.crc16 << 1)
                        ^ if self.crc16 & 0x8000 != 0 { 0x1021 } else { 0 };
                }
            }
            Crc16::X25 => {
                self.crc16 ^= byte as u16;
                for _ in 0..8 {
                    self.crc16 = (self.crc16 >> 1)
                        ^ if self.crc16 & 1 != 0 { 0x8408 } else { 0 };
                }
            }
        }
    }

    fn multi(self, variant: Crc16) -> Multi {
        let (f32_a, f32_b) = if let Some(high) = self.f32_pending {
            let mut a = self.f32_a;
            let mut b = self.f32_b;
            fletcher32_word(&mut a, &mut b, (high as u16) << 8);
            (a, b)
        } else {
            (self.f32_a, self.f32_b)
        };
        Multi {
            base: Hashes {
                crc32: self.crc ^ 0xffff_ffff,
                adler32: (self.adler_b << 16) | self.adler_a,
            },
            fletcher16: ((self.f16_b << 8) | self.f16_a) as u16,
            fletcher32: (f32_b << 16) | f32_a,
            crc16: if matches!(variant, Crc16::X25) {
                self.crc16 ^ 0xffff
            } else {
                self.crc16
            },
        }
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut accumulators = Accumulators::new(variant);
    for span in spans {
        for &byte in &data[span.start..span.end] {
            accumulators.update(byte, variant);
        }
    }
    Ok(accumulators.multi(variant))
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
}

fn crc32_zero_byte_operator(mut value: u32) -> u32 {
    for _ in 0..8 {
        value = (value >> 1) ^ if value & 1 != 0 { 0xedb88320 } else { 0 };
    }
    value
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut matrix = [0u32; 32];
    for bit in 0..32 {
        matrix[bit] = crc32_zero_byte_operator(1u32 << bit);
    }

    let mut result = left;
    let mut length = right_len;
    while length != 0 {
        if length & 1 != 0 {
            result = gf2_matrix_times(&matrix, result);
        }
        length >>= 1;
        if length != 0 {
            let mut square = [0u32; 32];
            gf2_matrix_square(&mut square, &matrix);
            matrix = square;
        }
    }
    result ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }
    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut combined = 0;
    for part in parts {
        combined = crc32_combine(combined, part.crc32, part.len);
    }
    Ok(Part {
        crc32: combined,
        len: total,
    })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut count = 0u64;
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for span in spans {
        count += (span.end - span.start) as u64;
        for &byte in &data[span.start..span.end] {
            crc = crc32_step(crc, byte);
            adler_update(&mut a, &mut b, byte);
        }
    }
    Ok(format!(
        "len={count:016x};crc={:08x};adler={:08x}",
        crc ^ 0xffff_ffff,
        (b << 16) | a
    ))
}

pub fn verify(
    data: &[u8],
    spans: &[Span],
    token: &str,
) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let length = parse_verify_hex(bytes, 4, 20)?;
    let expected_crc = parse_verify_hex(bytes, 25, 33)? as u32;
    let expected_adler = parse_verify_hex(bytes, 40, 48)? as u32;

    let mut actual_len = 0u64;
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    for span in spans {
        actual_len += (span.end - span.start) as u64;
        for &byte in &data[span.start..span.end] {
            crc = crc32_step(crc, byte);
            adler_update(&mut a, &mut b, byte);
        }
    }
    if length != actual_len {
        return Err(VerifyError::Length);
    }
    let actual_crc = crc ^ 0xffff_ffff;
    let actual_adler = (b << 16) | a;
    if expected_crc != actual_crc {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual_crc,
        });
    }
    if expected_adler != actual_adler {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual_adler,
        });
    }
    Ok(())
}

fn parse_verify_hex(bytes: &[u8], start: usize, end: usize) -> Result<u64, VerifyError> {
    let mut value = 0u64;
    for offset in start..end {
        let digit = match bytes[offset] {
            b'0'..=b'9' => (bytes[offset] - b'0') as u64,
            b'a'..=b'f' => (bytes[offset] - b'a' + 10) as u64,
            _ => return Err(VerifyError::Hex { offset }),
        };
        value = (value << 4) | digit;
    }
    Ok(value)
}
