use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = -n;
        d = -d;
    }

    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;

    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128 * rhs.den as i128;
        let b = rhs.num as i128 * self.den as i128;
        rational_from_i128(a + b, self.den as i128 * rhs.den as i128)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128 * rhs.den as i128;
        let b = rhs.num as i128 * self.den as i128;
        rational_from_i128(a - b, self.den as i128 * rhs.den as i128)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Rational {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(parse_err(i, ParseKind::InvalidByte));
                }
                slash = Some(i);
            } else if b.is_ascii_digit() {
                continue;
            } else if b == b'-' && i == 0 {
                continue;
            } else {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }

        let num_end = slash.unwrap_or(bytes.len());
        let num_start = if bytes[0] == b'-' { 1 } else { 0 };
        if num_start == num_end {
            return Err(parse_err(bytes.len(), ParseKind::InvalidByte));
        }
        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(parse_err(bytes.len(), ParseKind::InvalidByte));
            }
        }

        let num_digits = &bytes[num_start..num_end];
        let den_digits = slash.map(|s| &bytes[s + 1..]);

        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(parse_err(num_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && num_digits == b"0" {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }
        if let Some(d) = den_digits {
            if d.len() >= 2 && d[0] == b'0' {
                return Err(parse_err(slash.unwrap() + 2, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let mut nmag: u128 = 0;
        for &b in num_digits {
            nmag = nmag * 10 + (b - b'0') as u128;
            let limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
            if nmag > limit {
                return Err(parse_err(0, ParseKind::OutOfRange));
            }
        }

        let mut dval: u128 = 1;
        if let Some(d) = den_digits {
            dval = 0;
            for &b in d {
                dval = dval * 10 + (b - b'0') as u128;
                if dval > i64::MAX as u128 {
                    return Err(parse_err(slash.unwrap() + 1, ParseKind::OutOfRange));
                }
            }
            if dval == 0 {
                return Err(parse_err(slash.unwrap() + 1, ParseKind::ZeroDenominator));
            }
            if dval == 1 {
                return Err(parse_err(slash.unwrap(), ParseKind::NonCanonical));
            }
            if nmag == 0 {
                return Err(parse_err(slash.unwrap(), ParseKind::NonCanonical));
            }
            if gcd_u128(nmag, dval) > 1 {
                return Err(parse_err(slash.unwrap(), ParseKind::NonCanonical));
            }
        }

        let n = if negative {
            if nmag == (1u128 << 63) {
                i64::MIN
            } else {
                -(nmag as i64)
            }
        } else {
            nmag as i64
        };
        Ok(Rational {
            num: n,
            den: dval as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

fn zero() -> Rational {
    Rational { num: 0, den: 1 }
}

fn one() -> Rational {
    Rational { num: 1, den: 1 }
}

fn arith_err(row: usize, col: usize, step: usize, source: RationalError) -> MatrixError {
    MatrixError::Arithmetic {
        row,
        col,
        step,
        source,
    }
}

fn checked_entries(rows: usize, cols: usize) -> Result<usize, MatrixError> {
    rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)
}

fn eliminate(
    data: &mut [Rational],
    rows: usize,
    cols: usize,
    pivot_limit: usize,
) -> Result<(Vec<usize>, Vec<(usize, usize)>), MatrixError> {
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;
    let mut candidate = 0usize;

    while candidate < pivot_limit {
        if p == rows {
            break;
        }

        let mut chosen: Option<usize> = None;
        for r in p..rows {
            let v = data[r * cols + candidate];
            if v.num == 0 {
                continue;
            }
            let key = (v.den as u64, v.num.unsigned_abs(), r);
            if chosen
                .map(|old| {
                    let ov = data[old * cols + candidate];
                    key < (ov.den as u64, ov.num.unsigned_abs(), old)
                })
                .unwrap_or(true)
            {
                chosen = Some(r);
            }
        }

        let Some(q) = chosen else {
            candidate += 1;
            continue;
        };

        if q != p {
            for c in 0..cols {
                data.swap(p * cols + c, q * cols + c);
            }
            swaps.push((p, q));
        }

        let pivot = data[p * cols + candidate];
        for c in 0..cols {
            let v = data[p * cols + c];
            data[p * cols + c] = v
                .div(&pivot)
                .map_err(|e| arith_err(p, c, p, e))?;
        }

        for r in 0..rows {
            if r == p {
                continue;
            }
            let factor = data[r * cols + candidate];
            if factor.num == 0 {
                continue;
            }
            for c in 0..cols {
                let product = factor
                    .mul(&data[p * cols + c])
                    .map_err(|e| arith_err(r, c, p, e))?;
                data[r * cols + c] = data[r * cols + c]
                    .sub(&product)
                    .map_err(|e| arith_err(r, c, p, e))?;
            }
        }

        pivots.push(candidate);
        p += 1;
        candidate += 1;
    }

    Ok((pivots, swaps))
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = checked_entries(rows, cols)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Matrix { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let len = self.data.len();
        let mut data = vec![zero(); len];
        if self.rows != 0 && self.cols != 0 {
            for r in 0..self.rows {
                for c in 0..self.cols {
                    data[c * self.rows + r] = self.data[r * self.cols + c];
                }
            }
        }
        Matrix {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let out_len = checked_entries(self.rows, rhs.cols)?;
        let mut data = vec![zero(); out_len];

        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero();
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left
                        .mul(&right)
                        .map_err(|e| arith_err(r, c, k, e))?;
                    acc = acc.add(&term).map_err(|e| arith_err(r, c, k, e))?;
                }
                data[r * rhs.cols + c] = acc;
            }
        }

        Ok(Matrix {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let mut data = self.data.clone();
        let (pivots, swaps) = eliminate(&mut data, self.rows, self.cols, self.cols)?;
        Ok(Reduction {
            matrix: Matrix {
                rows: self.rows,
                cols: self.cols,
                data,
            },
            pivots,
            swaps,
        })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }

        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut augmented = vec![zero(); len];

        for r in 0..n {
            for c in 0..n {
                augmented[r * width + c] = self.data[r * n + c];
            }
            augmented[r * width + n + r] = one();
        }

        let (pivots, _) = eliminate(&mut augmented, n, width, n)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }

        let out_len = checked_entries(n, n)?;
        let mut inverse = vec![zero(); out_len];
        for r in 0..n {
            for c in 0..n {
                inverse[r * n + c] = augmented[r * width + n + c];
            }
        }
        Ok(Matrix {
            rows: n,
            cols: n,
            data: inverse,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }

        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut augmented = vec![zero(); len];
        for r in 0..self.rows {
            for c in 0..n {
                augmented[r * width + c] = self.data[r * n + c];
            }
            for c in 0..k {
                augmented[r * width + n + c] = rhs.data[r * k + c];
            }
        }

        let (pivots, _) =
            eliminate(&mut augmented, self.rows, width, n).map_err(SolveError::Matrix)?;

        for r in 0..self.rows {
            let coeff_zero = (0..n).all(|c| augmented[r * width + c].num == 0);
            if coeff_zero {
                for c in 0..k {
                    if augmented[r * width + n + c].num != 0 {
                        let witness = augmented[r * width..r * width + width].to_vec();
                        return Ok(Solution::None {
                            row: r,
                            rhs_col: c,
                            witness,
                        });
                    }
                }
            }
        }

        let rank = pivots.len();
        let mut is_pivot = vec![false; n];
        for &p in &pivots {
            is_pivot[p] = true;
        }
        let free: Vec<usize> = (0..n).filter(|&c| !is_pivot[c]).collect();

        let particular_len = n.checked_mul(k).ok_or_else(|| {
            SolveError::Matrix(MatrixError::ShapeOverflow)
        })?;
        let mut particular = vec![zero(); particular_len];
        for (j, &pivot_col) in pivots.iter().enumerate() {
            for c in 0..k {
                particular[pivot_col * k + c] = augmented[j * width + n + c];
            }
        }

        let particular = Matrix {
            rows: n,
            cols: k,
            data: particular,
        };

        if rank == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots,
            });
        }

        let f = free.len();
        let basis_len = n.checked_mul(f).ok_or_else(|| {
            SolveError::Matrix(MatrixError::ShapeOverflow)
        })?;
        let mut basis = vec![zero(); basis_len];

        for (j, &free_col) in free.iter().enumerate() {
            basis[free_col * f + j] = one();
            for (i, &pivot_col) in pivots.iter().enumerate() {
                let value = augmented[i * width + free_col];
                let neg = value.neg().map_err(|e| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col,
                        col: j,
                        step: rank,
                        source: e,
                    })
                })?;
                basis[pivot_col * f + j] = neg;
            }
        }

        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix {
                rows: n,
                cols: f,
                data: basis,
            },
            pivots,
        })
    }
}

fn main() {}
