fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort();
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {

}

fn constraint_text_list(cs: &[Constraint]) -> String {
    cs.iter().map(|c| constraint_text(*c)).collect::<Vec<_>>().join(",")

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

    v.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    v
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
    active_edges: BTreeMap<String, Vec<(String, usize, Constraint, Vec<String>, Option<String>, Platform)>>,
}

fn path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn make_closure(
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    catalog: &BTreeMap<String, &Package>,
    target: Target,
) -> Closure {
    let mut c = Closure::default();
    let mut queue: Vec<(String, String, Vec<String>)> = Vec::new();

    for (i, root) in roots.iter().enumerate() {
        let fs = root.features.iter().cloned().collect::<BTreeSet<_>>().into_iter().collect::<Vec<_>>().join(",");
        let token = format!(
            "root[{}]:{}@{}+{{{}}}",
            i, root.package, constraint_text(root.constraint), fs
        );
        let path = vec![token];
        c.constraints.entry(root.package.clone()).or_default().push(root.constraint);
        c.origins.entry(root.package.clone()).or_default().push(Origin {
            constraint: root.constraint,
            path: path.clone(),
        });
        c.requests.entry(root.package.clone()).or_default().extend(root.features.iter().cloned());
        queue.push((root.package.clone(), String::new(), path));
    }

    // Repeatedly propagate feature enables and active dependency edges until no new
    // feature request or shorter canonical path remains.
    loop {
        let mut changed = false;

        for (pkg, version) in selected {
            let Some(package) = catalog.get(pkg) else { continue };
            let Some(release) = package.releases.iter().find(|r| r.version == *version) else { continue };
            let reqs = c.requests.get(pkg).cloned().unwrap_or_default();
            for feat_name in reqs {
                if let Some(feat) = release.features.iter().find(|f| f.name == feat_name) {
                    let set = c.requests.entry(pkg.clone()).or_default();
                    for enabled in &feat.enables {
                        if set.insert(enabled.clone()) {
                            changed = true;
                        }
                    }
                }
            }
        }

        // Build canonical package paths from root origins and active edges found so far.
        let mut best: BTreeMap<String, Vec<String>> = BTreeMap::new();
        for (pkg, origins) in &c.origins {
            for o in origins {
                let candidate = o.path.clone();
                if best.get(pkg).map_or(true, |old| path_better(&candidate, old)) {
                    best.insert(pkg.clone(), candidate);
                }
            }
        }
        // Edges are derived below in a fixed point; use current source best paths.
        for (src, edges) in c.active_edges.clone() {
            if let Some(src_path) = best.get(&src).cloned() {
                for (dst, _, _, token_tail, _, _) in edges {
                    let mut p = src_path.clone();
                    p.extend(token_tail);
                    if best.get(&dst).map_or(true, |old| path_better(&p, old)) {
                        best.insert(dst, p);
                        changed = true;
                    }
                }
            }
        }
        c.paths = best.clone();

        let mut new_edges = Vec::new();
        for (src, version) in selected {
            let Some(package) = catalog.get(src) else { continue };
            let Some(release) = package.releases.iter().find(|r| r.version == *version) else { continue };
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target) {
                    continue;
                }
                if let Some(gate) = &dep.gate {
                    if !c.requests.get(src).map_or(false, |s| s.contains(gate)) {
                        continue;
                    }
                }
                let mut fs = dep.features.iter().cloned().collect::<BTreeSet<_>>().into_iter().collect::<Vec<_>>().join(",");
                if fs.is_empty() { fs = String::new(); }
                let gate_text = dep.gate.as_deref().unwrap_or("*");
                let platform_text = match dep.platform {
                    Platform::Any => "any",
                    Platform::Linux => "linux",
                    Platform::Mac => "mac",
                    Platform::Windows => "windows",
                };
                let token = format!(
                    "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                    src, version_text(*version), j, dep.package, constraint_text(dep.constraint), fs,
                    gate_text, platform_text
                );
                let tail = vec![token];
                let edge_key = (src.clone(), j);
                if !c.active_edges.get(src).map_or(false, |es| es.iter().any(|e| e.1 == j)) {
                    c.constraints.entry(dep.package.clone()).or_default().push(dep.constraint);
                    c.requests.entry(dep.package.clone()).or_default().extend(dep.features.iter().cloned());
                    c.active_edges.entry(src.clone()).or_default().push((
                        dep.package.clone(), j, dep.constraint, tail.clone(), dep.gate.clone(), dep.platform
                    ));
                    let src_path = c.paths.get(src).cloned().unwrap_or_default();
                    let mut p = src_path;
                    p.extend(tail);
                    c.origins.entry(dep.package.clone()).or_default().push(Origin {
                        constraint: dep.constraint,
                        path: p,
                    });
                    new_edges.push(edge_key);
                    changed = true;
                }
            }
        }

        if !changed {
            break;
        }
    }

    // Refresh paths over all edges to shortest-path closure.
    loop {
        let mut changed = false;
        let old = c.paths.clone();
        for (src, edges) in &c.active_edges {
            let Some(src_path) = old.get(src) else { continue };
            for (dst, _, _, tail, _, _) in edges {
                let mut p = src_path.clone();
                p.extend(tail.clone());
                if c.paths.get(dst).map_or(true, |cur| path_better(&p, cur)) {
                    c.paths.insert(dst.clone(), p);
                    changed = true;
                }
            }
        }
        if !changed { break; }
    }
    c
}

fn candidate_versions(
    name: &str,
    constraints: &[Constraint],
    catalog: &BTreeMap<String, &Package>,
    lockmap: &BTreeMap<String, Version>,
) -> Vec<Version> {
    let Some(pkg) = catalog.get(name) else { return vec![] };
    let mut vs = pkg.releases.iter()
        .map(|r| r.version)
        .filter(|v| constraints.iter().all(|c| matches(*c, *v)))
        .collect::<Vec<_>>();
    vs.sort_by(|a, b| b.cmp(a));
    vs.dedup();
    if let Some(lock) = lockmap.get(name) {
        if let Some(i) = vs.iter().position(|v| v == lock) {
            let v = vs.remove(i);
            vs.insert(0, v);
        }
    }
    vs
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: BTreeMap<String, &'a Package>,
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    budget: usize,
    attempts: usize,
    best: Option<(usize, Vec<Version>, BTreeMap<String, Version>, Closure)>,
    first_error: Option<ResolveError>,
    all_catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn remember_error(&mut self, err: ResolveError) {
        if self.first_error.is_none() { self.first_error = Some(err); }
    }

    fn visit(&mut self, selected: BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let closure = make_closure(self.roots, &selected, &self.catalog, self.target);
        let active_names = closure.constraints.keys().cloned().collect::<Vec<_>>();

        for pkg in &active_names {
            if !self.catalog.contains_key(pkg) {
                let evidence = closure.origins.get(pkg).cloned().unwrap_or_default()
                    .into_iter().map(|o| Evidence { constraint: o.constraint, path: o.path }).collect();
                self.remember_error(ResolveError::Missing { package: pkg.clone(), evidence });
                return Ok(());
            }
        }

        // Selected feature validity precedes numeric conflicts.
        for (pkg, version) in &selected {
            let Some(requests) = closure.requests.get(pkg) else { continue };
            let Some(package) = self.catalog.get(pkg) else { continue };
            let Some(release) = package.releases.iter().find(|r| r.version == *version) else { continue };
            for feature in requests {
                if !release.features.iter().any(|f| &f.name == feature) {
                    self.remember_error(ResolveError::MissingFeature {
                        package: pkg.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: closure.paths.get(pkg).cloned().unwrap_or_default(),
                    });
                    return Ok(());
                }
            }
        }

        for pkg in &active_names {
            if let Some(version) = selected.get(pkg) {
                let reqs = canonical(closure.constraints.get(pkg).cloned().unwrap_or_default());
                if !reqs.iter().all(|c| matches(*c, *version)) {
                    let evidence = closure.origins.get(pkg).cloned().unwrap_or_default()
                        .into_iter().map(|o| Evidence { constraint: o.constraint, path: o.path }).collect();
                    self.remember_error(ResolveError::Conflict {
                        package: pkg.clone(), selected: *version, requirements: reqs, evidence
                    });
                    return Ok(());
                }
            }
        }

        for pkg in &active_names {
            if !selected.contains_key(pkg) {
                let reqs = closure.constraints.get(pkg).cloned().unwrap_or_default();
                if candidate_versions(pkg, &reqs, &self.catalog, self.locks).is_empty() {
                    let evidence = closure.origins.get(pkg).cloned().unwrap_or_default()
                        .into_iter().map(|o| Evidence { constraint: o.constraint, path: o.path }).collect();
                    self.remember_error(ResolveError::NoMatch {
                        package: pkg.clone(), requirements: canonical(reqs), evidence
                    });
                    return Ok(());
                }
            }
        }

        let next = active_names.iter().find(|p| !selected.contains_key(*p)).cloned();
        if let Some(pkg) = next {
            let reqs = closure.constraints.get(&pkg).cloned().unwrap_or_default();
            let candidates = candidate_versions(&pkg, &reqs, &self.catalog, self.locks);
            for v in candidates {
                if self.attempts == self.budget {
                    return Err(ResolveError::Limit { attempts: self.budget });
                }
                self.attempts += 1;
                let mut child = selected.clone();
                child.insert(pkg.clone(), v);
                self.visit(child)?;
            }
            return Ok(());
        }

        let lock_cost = self.locks.iter()
            .filter(|(p, v)| selected.get(*p) != Some(*v))
            .count();
        let vector = self.all_catalog_names.iter().map(|p| selected.get(p).copied()).collect::<Vec<_>>();
        let better = match &self.best {
            None => true,
            Some((cost, old_vec, _, _)) => {
                lock_cost < *cost || (lock_cost == *cost && vector_cmp(&vector, old_vec) == std::cmp::Ordering::Greater)
            }
        };
        if better {
            self.best = Some((lock_cost, vector, selected, closure));
        }
        Ok(())
    }
}

fn vector_cmp(a: &[Option<Version>], b: &[Option<Version>]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let ord = match (x, y) {
            (None, None) => std::cmp::Ordering::Equal,
            (None, Some(_)) => std::cmp::Ordering::Less,
            (Some(_), None) => std::cmp::Ordering::Greater,
            (Some(x), Some(y)) => x.cmp(y),
        };
        if ord != std::cmp::Ordering::Equal { return ord; }
    }
    std::cmp::Ordering::Equal
}

fn validate_name(site: String, name: &str) -> Result<(), ResolveError> {
    if valid_name(name) { Ok(()) } else {
        Err(ResolveError::InvalidName { site, name: name.to_string() })
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        validate_name(format!("root[{}]", i), &r.package)?;
        for (f, feature) in r.features.iter().enumerate() {
            validate_name(format!("root[{}].feature[{}]", i, f), feature)?;
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        validate_name(format!("catalog[{}]", i), &p.name)?;
        for (j, r) in p.releases.iter().enumerate() {
            for (k, d) in r.deps.iter().enumerate() {
                validate_name(format!("catalog[{}].release[{}].dep[{}]", i, j, k), &d.package)?;
                if let Some(g) = &d.gate {
                    validate_name(format!("catalog[{}].release[{}].dep[{}].gate", i, j, k), g)?;
                }
                for (f, feature) in d.features.iter().enumerate() {
                    validate_name(format!("catalog[{}].release[{}].dep[{}].feature[{}]", i, j, k, f), feature)?;
                }
            }
            for (f, feature) in r.features.iter().enumerate() {
                validate_name(format!("catalog[{}].release[{}].feature[{}]", i, j, f), &feature.name)?;
                for (e, enabled) in feature.enables.iter().enumerate() {
                    validate_name(format!("catalog[{}].release[{}].feature[{}].enable[{}]", i, j, f, e), enabled)?;
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for p in catalog {
        if !seen_packages.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage { package: p.name.clone() });
        }
    }

    for p in catalog {
        let mut seen_versions = BTreeSet::new();
        for r in &p.releases {
            if !seen_versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion { package: p.name.clone(), version: r.version });
            }
        }
    }

    for p in catalog {
        for r in &p.releases {
            let mut seen = BTreeSet::new();
            for f in &r.features {
                if !seen.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(), version: r.version, feature: f.name.clone()
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        validate_name(format!("lock[{}]", i), &lock.package)?;
    }
    let mut lockmap = BTreeMap::new();
    for lock in locks {
        if lockmap.insert(lock.package.clone(), lock.version).is_some() {
            return Err(ResolveError::DuplicateLock { package: lock.package.clone() });
        }
    }

    let catalog_map = catalog.iter().map(|p| (p.name.clone(), p)).collect::<BTreeMap<_, _>>();
    let mut names = catalog.iter().map(|p| p.name.clone()).collect::<Vec<_>>();
    names.sort();
    let lock_map = lockmap.clone();

    let mut search = Search {
        roots, catalog: catalog_map, locks: &lock_map, target, budget: max_attempts,
        attempts: 0, best: None, first_error: None, all_catalog_names: names,
    };
    search.visit(BTreeMap::new())?;

    let Some((_, _, chosen, final_closure)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::Missing {
            package: String::new(), evidence: vec![]
        }));
    };

    let selected = chosen.iter().map(|(package, version)| Selection {
        package: package.clone(), version: *version
    }).collect::<Vec<_>>();

    let mut all_names = BTreeSet::new();
    all_names.extend(lockmap.keys().cloned());
    all_names.extend(chosen.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = lockmap.get(&package).copied();
        let after = chosen.get(&package).copied();
        if before == after { continue; }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map.get(&package)
                    .map_or(false, |p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let reqs = final_closure.constraints.get(&package).cloned().unwrap_or_default();
                    if !reqs.iter().all(|c| matches(*c, old)) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => continue,
        };
        changes.push(Change { package, before, after, reason });
    }

    Ok(Resolution { selected, changes })
}

fn main() {}
