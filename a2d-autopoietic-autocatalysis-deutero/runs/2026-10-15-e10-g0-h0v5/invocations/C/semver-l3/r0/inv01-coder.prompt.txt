use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    let field = &s[start..*pos];
    if field.len() > 1 && field.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    field.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut id_start = start;
    let mut i = start;
    loop {
        if i == end || bytes[i] == b'.' {
            if i == id_start {
                return Err(VersionError::EmptyIdentifier { at: id_start });
            }
            let id = &s[id_start..i];
            if prerelease && id.bytes().all(|b| b.is_ascii_digit())
                && id.len() > 1 && id.starts_with('0')
            {
                return Err(VersionError::LeadingZero { at: id_start });
            }
            out.push(id.to_string());
            if i == end {
                break;
            }
            i += 1;
            id_start = i;
        } else {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;
    let minor = parse_core_number(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        if pos >= s.len() || matches!(s.as_bytes()[pos], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;
    let patch = parse_core_number(s, &mut pos)?;
    let bytes = s.as_bytes();
    if pos < s.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < s.len() && bytes[pos] == b'-' {
        let begin = pos + 1;
        let plus = bytes[begin..].iter().position(|&b| b == b'+').map(|n| begin + n);
        let end = plus.unwrap_or(s.len());
        pre = parse_identifiers(s, begin, end, true)?;
        pos = end;
    }
    if pos < s.len() && bytes[pos] == b'+' {
        build = parse_identifiers(s, pos + 1, s.len(), false)?;
        pos = s.len();
    }
    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn is_numeric(id: &str) -> bool {
    !id.is_empty() && id.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = is_numeric(x);
        let yn = is_numeric(y);
        let c = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn token_ranges(s: &str) -> Vec<(usize, usize)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && whitespace(b[i]) { i += 1; }
        if i == b.len() { break; }
        let start = i;
        while i < b.len() && !whitespace(b[i]) { i += 1; }
        out.push((start, i));
    }
    out
}

fn version_operand(s: &str, start: usize, full: bool) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at: start, error })
}

fn parse_partial(s: &str, start: usize) -> Result<(Vec<Option<u64>>, Option<Version>), RangeError> {
    let fullish = s.contains('-') || s.contains('+')
        || (s.split('.').count() >= 3 && !s.split('.').any(|x| matches!(x, "*" | "x" | "X")));
    if fullish {
        return version_operand(s, start, true).map(|v| (Vec::new(), Some(v)));
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at: start });
    }
    let mut vals = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
            vals.push(None);
        } else {
            if wildcard_seen || !f.bytes().all(|b| b.is_ascii_digit())
                || (f.len() > 1 && f.starts_with('0'))
            {
                return Err(RangeError::Syntax { at: start });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: start })?;
            vals.push(Some(n));
        }
    }
    Ok((vals, None))
}

fn version_tuple(v: &Version) -> (u64, u64, u64) { (v.major, v.minor, v.patch) }

fn bound_cmp(op: Op, version: Version) -> Bound { Bound::Cmp { op, version } }

fn endpoint(vals: &[Option<u64>], full: Option<Version>) -> Version {

    let mut nums = [0u64; 3];
    for (i, v) in vals.iter().enumerate() {
        if let Some(n) = v { nums[i] = *n; }
    }
    Version { major: nums[0], minor: nums[1], patch: nums[2], pre: vec![], build: vec![] }
}

fn explicit_anchor(v: &Version, anchors: &mut Vec<(u64,u64,u64)>) {
    if !v.pre.is_empty() {
        anchors.push(version_tuple(v));
    }
}

fn next_upper(vals: &[Option<u64>], kind: u8, at: usize) -> Result<Version, RangeError> {
    let k = vals.iter().take_while(|x| x.is_some()).count();
    let mut v = endpoint(vals, None);
    let idx = if kind == b'~' {
        if k == 1 { 0 } else { 1 }
    } else if v.major > 0 || k == 1 {
        0
    } else if v.minor > 0 || k == 2 {
        1
    } else { 2 };

    v.major = nums[0]; v.minor = nums[1]; v.patch = nums[2];

}

fn parse_arm(s: &str, toks: &[(usize,usize)]) -> Result<Arm, RangeError> {
    if toks.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let mut checks = Vec::new();
    let mut anchors = Vec::new();
    let standalone_dash = toks.iter().position(|&(a,b)| &s[a..b] == "-");
    if let Some(dash_i) = standalone_dash {
        if toks.len() != 3 || dash_i != 1 {
            return Err(RangeError::Syntax { at: toks[dash_i].0 });
        }
        let (la, lb) = toks[0];
        let (ua, ub) = toks[2];
        let (lv, lf) = parse_partial(&s[la..lb], la)?;
        let (uv, uf) = parse_partial(&s[ua..ub], ua)?;
        if lv.iter().all(Option::is_none) {
            return Err(RangeError::Syntax { at: la });
        }
        if uv.iter().all(Option::is_none) {
            return Err(RangeError::Syntax { at: ua });
        }
        let lower = endpoint(&lv, lf.clone());
        explicit_anchor(&lower, &mut anchors);
        checks.push(bound_cmp(Op::Ge, lower));
        if let Some(full) = uf {
            explicit_anchor(&full, &mut anchors);
            checks.push(bound_cmp(Op::Le, full));
        } else {
            let k = uv.iter().take_while(|x| x.is_some()).count();
            if k == 3 {
                checks.push(bound_cmp(Op::Le, endpoint(&uv, None)));
            } else {
                let upper = next_upper(&uv, b'~', ua)?;
                checks.push(bound_cmp(Op::Lt, upper));
            }
        }
        return Ok(Arm { checks, anchors });
    }

    for &(a,b) in toks {
        let token = &s[a..b];
        if token == "||" { return Err(RangeError::Syntax { at: a }); }
        let bytes = token.as_bytes();
        let first = bytes[0];
        if matches!(first, b'^' | b'~') {
            let operand = &token[1..];
            let op_start = a + 1;
            if operand.is_empty() { return Err(RangeError::Syntax { at: a }); }
            let (vals, full) = parse_partial(operand, op_start)?;
            let k = vals.iter().take_while(|x| x.is_some()).count();
            if full.is_none() && k == 0 { return Err(RangeError::Syntax { at: a }); }
            let lower = endpoint(&vals, full.clone());
            explicit_anchor(&lower, &mut anchors);
            checks.push(bound_cmp(Op::Ge, lower));
            if full.is_some() {
                let mut p = vals.clone();
                let v = full.unwrap();
                let fields = vec![Some(v.major), Some(v.minor), Some(v.patch)];
                let upper = next_upper(&fields, first, op_start)?;
                checks.push(bound_cmp(Op::Lt, upper));
            } else {
                checks.push(bound_cmp(Op::Lt, next_upper(&vals, first, op_start)?));
            }
        } else if matches!(first, b'<' | b'>' | b'=') {
            let (op, n) = if token.starts_with("<=") { (Op::Le,2) }
                else if token.starts_with(">=") { (Op::Ge,2) }
                else if first == b'<' { (Op::Lt,1) }
                else if first == b'>' { (Op::Gt,1) }
                else { (Op::Eq,1) };
            if token.len() == n || !token.as_bytes()[n].is_ascii_digit() {
                return Err(RangeError::Syntax { at: a });
            }
            let v = version_operand(&token[n..], a+n, true)?;
            explicit_anchor(&v, &mut anchors);
            checks.push(bound_cmp(op, v));
        } else if first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X') {
            let (vals, full) = parse_partial(token, a)?;
            if let Some(v) = full {
                explicit_anchor(&v, &mut anchors);
                checks.push(Bound::Exact(v));
            } else {
                let k = vals.iter().take_while(|x| x.is_some()).count();
                if k == 0 { checks.push(Bound::Any); }
                else {
                    let lower = endpoint(&vals, None);
                    checks.push(bound_cmp(Op::Ge, lower));
                    checks.push(bound_cmp(Op::Lt, next_upper(&vals, b'~', a)?));
                }
            }
        } else {
            return Err(RangeError::Syntax { at: a });
        }
    }
    Ok(Arm { checks, anchors })
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = token_ranges(s);
    if tokens.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        let (a,b) = tokens[i];
        if &s[a..b] == "||" {
            if current.is_empty() { return Err(RangeError::Empty { at: a }); }
            arms.push(parse_arm(s, &current)?);
            current.clear();
            if i + 1 == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
        } else {
            current.push((a,b));
        }
        i += 1;
    }
    if !current.is_empty() { arms.push(parse_arm(s, &current)?); }
    Ok(Range { arms })
}

fn one_match(v: &Version, arm: &Arm, include_pre: bool) -> bool {
    if !include_pre && !v.pre.is_empty()
        && !arm.anchors.contains(&version_tuple(v))
    {
        return false;
    }
    arm.checks.iter().all(|c| match c {
        Bound::Any => true,
        Bound::Exact(x) => compare_versions(v, x) == Ordering::Equal,
        Bound::Cmp { op, version } => {
            let ord = compare_versions(v, version);
            match op {
                Op::Eq => ord == Ordering::Equal,
                Op::Lt => ord == Ordering::Less,
                Op::Le => ord != Ordering::Greater,
                Op::Gt => ord == Ordering::Greater,
                Op::Ge => ord != Ordering::Less,
            }
        }
    })
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| one_match(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool { matches_with(v, r, false) }

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

fn extrema(candidates: &[String], r: &Range, maximum: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (i, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index: i, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) { continue; }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                best = Some(i);
            }
        } else { best = Some(i); }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

#[derive(Clone, Debug)]
struct Interval {
    lower: Option<(Version, bool)>,
    upper: Option<(Version, bool)>,
}

fn arm_interval(arm: &Arm) -> Interval {
    let mut lower: Option<(Version, bool)> = None;
    let mut upper: Option<(Version, bool)> = None;
    for c in &arm.checks {
        let (op, v) = match c {
            Bound::Any => continue,
            Bound::Exact(v) => {
                let l = (v.clone(), true);
                let u = (v.clone(), true);
                lower = Some(l); upper = Some(u);
                continue;
            }
            Bound::Cmp { op, version } => (*op, version.clone()),
        };
        match op {
            Op::Eq => { lower = Some((v.clone(), true)); upper = Some((v, true)); }
            Op::Ge | Op::Gt => {
                let inclusive = op == Op::Ge;
                if lower.as_ref().map_or(true, |(x, inc)| {
                    compare_versions(&v, x) == Ordering::Greater
                        || (compare_versions(&v, x) == Ordering::Equal && !inclusive && *inc)
                }) { lower = Some((v, inclusive)); }
            }
            Op::Le | Op::Lt => {
                let inclusive = op == Op::Le;
                if upper.as_ref().map_or(true, |(x, inc)| {
                    compare_versions(&v, x) == Ordering::Less
                        || (compare_versions(&v, x) == Ordering::Equal && !inclusive && *inc)
                }) { upper = Some((v, inclusive)); }
            }
        }
    }
    Interval { lower, upper }
}

fn interval_simple_nonempty(iv: &Interval) -> bool {
    match (&iv.lower, &iv.upper) {
        (Some((l, li)), Some((u, ui))) => {
            let c = compare_versions(l, u);
            c == Ordering::Less || (c == Ordering::Equal && *li && *ui)
        }
        _ => true,
    }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let ia = arm_interval(aa);
            let ib = arm_interval(bb);
            let lower = match (&ia.lower, &ib.lower) {
                (Some(x), Some(y)) => Some(if compare_versions(&x.0, &y.0) == Ordering::Greater { x.clone() } else { y.clone() }),
                (Some(x), None) | (None, Some(x)) => Some(x.clone()),
                _ => None,
            };
            let upper = match (&ia.upper, &ib.upper) {
                (Some(x), Some(y)) => Some(if compare_versions(&x.0, &y.0) == Ordering::Less { x.clone() } else { y.clone() }),
                (Some(x), None) | (None, Some(x)) => Some(x.clone()),
                _ => None,
            };
            if let (Some((l, li)), Some((u, ui))) = (&lower, &upper) {
                let c = compare_versions(l, u);
                if c == Ordering::Greater || (c == Ordering::Equal && !(*li && *ui)) { continue; }
            }
            if include_prerelease { return true; }
            let anchors: Vec<_> = aa.anchors.iter().filter(|x| bb.anchors.contains(x)).collect();
            let admitted = !anchors.is_empty() && anchors.iter().any(|t| {
                let v = Version { major:t.0, minor:t.1, patch:t.2, pre:vec!["0".into()], build:vec![] };
                let in_a = one_match(&v, aa, true);
                let in_b = one_match(&v, bb, true);
                in_a && in_b
            });
            let stable_possible = true;
            if stable_possible || admitted { return true; }
        }
    }
    false
}

fn version_display(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() { s.push('-'); s.push_str(&v.pre.join(".")); }
    s
}

pub fn canonical(r: &Range) -> String {
    let mut strings = Vec::new();
    for arm in &r.arms {
        let iv = arm_interval(arm);
        if !interval_simple_nonempty(&iv) { continue; }
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anchor_text = if anchors.is_empty() { "-".to_string() } else {
            anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
        };
        let lo = iv.lower.as_ref().map_or("*".to_string(), |(v,inc)| format!("{}{}", if *inc { ">=" } else { ">" }, version_display(v)));
        let hi = iv.upper.as_ref().map_or("*".to_string(), |(v,inc)| format!("{}{}", if *inc { "<=" } else { "<" }, version_display(v)));
        strings.push(format!("[{lo};{hi};pre={anchor_text}]"));
    }
    strings.sort();
    strings.dedup();
    if strings.is_empty() { "EMPTY".to_string() } else { strings.join(" || ") }
}
