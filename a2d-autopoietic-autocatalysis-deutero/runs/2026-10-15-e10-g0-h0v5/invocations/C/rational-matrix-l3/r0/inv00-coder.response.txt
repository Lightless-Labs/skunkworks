use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let sign_negative = (num < 0) ^ (den < 0);
        let an = num.unsigned_abs();
        let ad = den.unsigned_abs();
        let g = gcd_u128(an, ad);
        let rn = an / g;
        let rd = ad / g;
        if rd > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        let max_negative = (i64::MAX as u128) + 1;
        if (!sign_negative && rn > i64::MAX as u128) || (sign_negative && rn > max_negative) {
            return Err(RationalError::Overflow);
        }
        let n = if sign_negative {
            if rn == max_negative { i64::MIN } else { -(rn as i64) }
        } else {
            rn as i64
        };
        Ok(Self { num: n, den: rd as i64 })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.num as i128;
        let d = self.den as i128 * rhs.den as i128;
        Self::from_i128(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let n = self.num as i128 * rhs.den as i128;
        let d = self.den as i128 * rhs.num as i128;
        Self::from_i128(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: -self.num, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        fn err(at: usize, kind: ParseKind) -> RationalError {
            RationalError::Parse { at, kind }
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }

        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(err(i, ParseKind::InvalidByte));
        }

        let mut den_start = None;
        let slash_at;
        if i < bytes.len() && bytes[i] == b'/' {
            slash_at = i;
            i += 1;
            den_start = Some(i);
            if i == bytes.len() {
                return Err(err(i, ParseKind::InvalidByte));
            }
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(err(i, ParseKind::InvalidByte));
            }
        } else {
            slash_at = usize::MAX;
        }
        if i < bytes.len() {
            return Err(err(i, ParseKind::InvalidByte));
        }

        let num_end = if den_start.is_some() { slash_at } else { bytes.len() };
        let negative = bytes[0] == b'-';
        let digits_start = if negative { 1 } else { 0 };
        let num_digits = num_end - digits_start;

        if num_digits >= 2 && bytes[digits_start] == b'0' {
            return Err(err(digits_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == 1 && bytes[1] == b'0' {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if let Some(ds) = den_start {
            let de = bytes.len();
            if de - ds >= 2 && bytes[ds] == b'0' {
                return Err(err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let num_mag = parse_u128_digits(&bytes[digits_start..num_end]);
        let num_limit = if negative { (i64::MAX as u128) + 1 } else { i64::MAX as u128 };
        if num_mag > num_limit {
            return Err(err(0, ParseKind::OutOfRange));
        }

        let den_mag = if let Some(ds) = den_start {
            let value = parse_u128_digits(&bytes[ds..]);
            if value > i64::MAX as u128 {
                return Err(err(ds, ParseKind::OutOfRange));
            }
            value
        } else {
            1
        };

        if den_start.is_some() && den_mag == 0 {
            return Err(err(den_start.unwrap(), ParseKind::ZeroDenominator));
        }

        if den_start.is_some() && den_mag == 1 {
            return Err(err(slash_at, ParseKind::NonCanonical));
        }
        if den_start.is_some() && num_mag == 0 {
            return Err(err(slash_at, ParseKind::NonCanonical));
        }

        if den_start.is_some() && gcd_u128(num_mag, den_mag) > 1 {
            return Err(err(slash_at, ParseKind::NonCanonical));
        }

        let num = if negative {
            if num_mag == (i64::MAX as u128) + 1 {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };
        Ok(Self { num, den: den_mag as i64 })
    }
}

fn parse_u128_digits(bytes: &[u8]) -> u128 {
    bytes.iter().fold(0u128, |acc, &b| {
        acc.saturating_mul(10).saturating_add((b - b'0') as u128)
    })
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let idx = row.checked_mul(self.cols)
            .and_then(|x| x.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data.get(idx).copied().ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let out_len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(out_len);
        let zero = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.rref_limited(self.cols)
    }

    fn rref_limited(&self, candidate_cols: usize) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..candidate_cols {
            if p == matrix.rows {
                break;
            }
            let mut chosen: Option<(i64, u128, usize)> = None;
            for row in p..matrix.rows {
                let value = matrix.data[row * matrix.cols + col];
                if value.num == 0 {
                    continue;
                }
                let key = (value.den, value.num.unsigned_abs(), row);
                if chosen.map_or(true, |current| key < current) {
                    chosen = Some(key);
                }
            }
            let Some((_, _, q)) = chosen else { continue };

            if q != p {
                for c in 0..matrix.cols {
                    matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = matrix.data[p * matrix.cols + col];
            let mut normalized = Vec::with_capacity(matrix.cols);
            for c in 0..matrix.cols {
                let value = matrix.data[p * matrix.cols + c];
                let result = value.div(&pivot).map_err(|source| MatrixError::Arithmetic {
                    row: p, col: c, step: p, source,
                })?;
                normalized.push(result);
            }
            for c in 0..matrix.cols {
                matrix.data[p * matrix.cols + c] = normalized[c];
            }

            for row in 0..matrix.rows {
                if row == p {
                    continue;
                }
                let factor = matrix.data[row * matrix.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..matrix.cols {
                    let product = factor.mul(&normalized[c]).map_err(|source| MatrixError::Arithmetic {
                        row, col: c, step: p, source,
                    })?;
                    let current = matrix.data[row * matrix.cols + c];
                    let result = current.sub(&product).map_err(|source| MatrixError::Arithmetic {
                        row, col: c, step: p, source,
                    })?;
                    matrix.data[row * matrix.cols + c] = result;
                }
            }

            pivots.push(col);
            p += 1;
        }

        Ok(Reduction { matrix, pivots, swaps })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(if r == c {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }
        let augmented = Self { rows: n, cols: width, data };
        let red = augmented.rref_limited(n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank: red.pivots.len() });
        }
        let mut result = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                result.push(red.matrix.data[r * width + n + c]);
            }
        }
        Ok(Self { rows: n, cols: n, data: result })
    }
}
