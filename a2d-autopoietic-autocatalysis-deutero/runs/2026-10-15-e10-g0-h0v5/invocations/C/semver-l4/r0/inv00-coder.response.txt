use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        return Err(match bytes.get(start).copied() {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: start },
            _ => VersionError::InvalidChar { at: start },
        });
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end && bytes[p] != b'.' {
            let b = bytes[p];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if pre && id.bytes().all(is_digit) && id.len() > 1 && id.as_bytes()[0] == b'0' {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_string());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        return Err(match s.as_bytes().get(p).copied() {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: p },
            _ => VersionError::InvalidChar { at: p },
        });
    }
    p += 1;
    let minor = parse_core(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        return Err(match s.as_bytes().get(p).copied() {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: p },
            _ => VersionError::InvalidChar { at: p },
        });
    }
    p += 1;
    let patch = parse_core(s, &mut p)?;
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < bytes.len() && bytes[p] == b'-' {
        let pre_start = p + 1;
        let plus = bytes[pre_start..].iter().position(|&b| b == b'+').map(|x| pre_start + x);
        let pre_end = plus.unwrap_or(bytes.len());
        pre = parse_identifiers(s, pre_start, pre_end, true)?;
        p = pre_end;
    }
    if p < bytes.len() && bytes[p] == b'+' {
        build = parse_identifiers(s, p + 1, bytes.len(), false)?;
        p = bytes.len();
    }
    if p != bytes.len() {
        return Err(VersionError::InvalidChar { at: p });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    if a.is_empty() && b.is_empty() { return Ordering::Equal; }
    if a.is_empty() { return Ordering::Greater; }
    if b.is_empty() { return Ordering::Less; }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let c = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal { return c; }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core(a).cmp(&core(b)).then_with(|| cmp_pre(&a.pre, &b.pre))
}

fn mkver(a: u64, b: u64, c: u64, pre: Vec<String>) -> Version {
    Version { major: a, minor: b, patch: c, pre, build: Vec::new() }
}

fn endpoint(v: Version, op: Op, anchor: bool) -> Cmp {
    Cmp { op, v, anchor }
}

fn ws(b: u8) -> bool { matches!(b, b' ' | b'\t' | b'\r' | b'\n') }

fn tokens_with_offsets(s: &str, base: usize) -> Vec<(String, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = 0;
    while p < bytes.len() {
        while p < bytes.len() && ws(bytes[p]) { p += 1; }
        if p == bytes.len() { break; }
        let st = p;
        while p < bytes.len() && !ws(bytes[p]) { p += 1; }
        out.push((s[st..p].to_string(), base + st));
    }
    out
}

fn operand_version(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

#[derive(Clone, Debug)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
}

fn parse_partial_operand(s: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix = s.contains('-') || s.contains('+');
    let fields: Vec<&str> = s.split('.').collect();
    let has_wild = fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));
    if has_suffix || (fields.len() >= 3 && !has_wild) {
        return Ok(Partial { nums: Vec::new(), full: Some(operand_version(s, at)?) });
    }
    if fields.len() > 3 { return Err(RangeError::Syntax { at }); }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen || f.is_empty() || !f.bytes().all(is_digit)
                || (f.len() > 1 && f.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            match f.parse::<u64>() {
                Ok(n) => nums.push(n),
                Err(_) => return Err(RangeError::Syntax { at }),
            }
        }
    }
    Ok(Partial { nums, full: None })
}

fn padded(p: &Partial) -> Version {
    mkver(*p.nums.get(0).unwrap_or(&0), *p.nums.get(1).unwrap_or(&0), *p.nums.get(2).unwrap_or(&0), Vec::new())
}

fn increment_upper(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let (mut a, mut b, mut c) = core(v);
    let target = match field {
        0 => &mut a,
        1 => &mut b,
        _ => &mut c,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    if field == 0 { b = 0; c = 0; }
    if field == 1 { c = 0; }
    Ok(mkver(a, b, c, vec!["0".into()]))
}

fn partial_expansion(p: &Partial, at: usize) -> Result<Vec<Cmp>, RangeError> {
    if p.nums.is_empty() { return Ok(Vec::new()); }
    if let Some(v) = &p.full {
        return Ok(vec![endpoint(v.clone(), Op::Eq, !v.pre.is_empty())]);
    }
    let lo = padded(p);
    if p.nums.len() == 3 {
        return Ok(vec![endpoint(lo, Op::Eq, false)]);
    }
    let up = increment_upper(&lo, p.nums.len() - 1, at)?;
    Ok(vec![endpoint(lo, Op::Ge, false), endpoint(up, Op::Lt, false)])
}

fn parse_general_token(tok: &str, at: usize) -> Result<Vec<Cmp>, RangeError> {
    if tok.starts_with('^') || tok.starts_with('~') {
        let op = tok.as_bytes()[0];
        let p = parse_partial_operand(&tok[1..], at + 1)?;
        if p.nums.is_empty() && p.full.is_none() { return Err(RangeError::Syntax { at }); }
        let lo = p.full.clone().unwrap_or_else(|| padded(&p));
        let k = if p.full.is_some() { 3 } else { p.nums.len() };
        let field = if op == b'^' {
            if lo.major > 0 || k == 1 { 0 }
            else if lo.minor > 0 || k == 2 { 1 }
            else { 2 }
        } else if k == 1 { 0 } else { 1 };
        let up = increment_upper(&lo, field, at + 1)?;
        return Ok(vec![
            endpoint(lo.clone(), Op::Ge, p.full.as_ref().map_or(false, |v| !v.pre.is_empty())),
            endpoint(up, Op::Lt, false),
        ]);
    }
    if tok.starts_with('<') || tok.starts_with('>') || tok.starts_with('=') {
        let (op, n) = if tok.starts_with("<=") { (Op::Le, 2) }
            else if tok.starts_with(">=") { (Op::Ge, 2) }
            else { (match tok.as_bytes()[0] { b'<' => Op::Lt, b'>' => Op::Gt, _ => Op::Eq }, 1) };
        if n == tok.len() || !tok.as_bytes()[n].is_ascii_digit() {
            return Err(RangeError::Syntax { at });
        }
        let v = operand_version(&tok[n..], at + n)?;
        return Ok(vec![endpoint(v.clone(), op, !v.pre.is_empty())]);
    }
    if !(tok.as_bytes()[0].is_ascii_digit() || matches!(tok, "*" | "x" | "X")) {
        return Err(RangeError::Syntax { at });
    }
    let p = parse_partial_operand(tok, at)?;
    partial_expansion(&p, at)
}

fn parse_hyphen(tokens: &[(String, usize)], arm_start: usize) -> Result<Arm, RangeError> {
    let dashes: Vec<usize> = tokens.iter().filter(|(t, _)| t == "-").map(|(_, at)| *at).collect();
    if !dashes.is_empty() && (tokens.len() != 3 || tokens[1].0 != "-") {
        return Err(RangeError::Syntax { at: dashes[0] });
    }
    if tokens.len() != 3 || tokens[1].0 != "-" {
        return Err(RangeError::Syntax { at: arm_start });
    }
    let (lt, la) = (&tokens[0].0, tokens[0].1);
    let (ut, ua) = (&tokens[2].0, tokens[2].1);
    let lp = parse_partial_operand(lt, la)?;
    let up = parse_partial_operand(ut, ua)?;
    if lp.nums.is_empty() && lp.full.is_none() { return Err(RangeError::Syntax { at: la }); }
    if up.nums.is_empty() && up.full.is_none() { return Err(RangeError::Syntax { at: ua }); }
    let mut cmps = Vec::new();
    let lv = lp.full.clone().unwrap_or_else(|| padded(&lp));
    cmps.push(endpoint(lv.clone(), Op::Ge, lp.full.as_ref().map_or(false, |v| !v.pre.is_empty())));
    if let Some(v) = up.full {
        cmps.push(endpoint(v.clone(), Op::Le, !v.pre.is_empty()));
    } else if up.nums.len() == 3 {
        cmps.push(endpoint(padded(&up), Op::Le, false));
    } else {
        let upper = increment_upper(&padded(&up), up.nums.len() - 1, ua)?;
        cmps.push(endpoint(upper, Op::Lt, false));
    }
    Ok(Arm { anchors: cmps.iter().filter(|c| c.anchor).map(|c| core(&c.v)).collect(), cmps })
}

fn parse_arm(s: &str, base: usize) -> Result<Arm, RangeError> {
    let tokens = tokens_with_offsets(s, base);
    if tokens.is_empty() { return Err(RangeError::Empty { at: base }); }
    if tokens.iter().any(|(t, _)| t == "-") {
        return parse_hyphen(&tokens, tokens[0].1);
    }
    let mut cmps = Vec::new();
    for (t, at) in tokens {
        cmps.extend(parse_general_token(&t, at)?);
    }
    let anchors = cmps.iter().filter(|c| c.anchor).map(|c| core(&c.v)).collect();
    Ok(Arm { cmps, anchors })
}

fn split_ranges(s: &str) -> Result<Vec<(String, usize)>, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut start = 0;
    let mut p = 0;
    let mut had_sep = false;
    while p < bytes.len() {
        if p + 1 < bytes.len() && bytes[p] == b'|' && bytes[p + 1] == b'|' {
            let left_end = p;
            let left = &s[start..left_end];
            if tokens_with_offsets(left, start).is_empty() {
                return Err(RangeError::Empty { at: if had_sep { start } else { 0 } });
            }
            arms.push((left.to_string(), start));
            p += 2;
            while p < bytes.len() && ws(bytes[p]) { p += 1; }
            start = p;
            had_sep = true;
        } else {
            p += 1;
        }
    }
    let tail = &s[start..];
    if tokens_with_offsets(tail, start).is_empty() {
        if had_sep { return Err(RangeError::Empty { at: s.len() }); }
        if tokens_with_offsets(s, 0).is_empty() { return Err(RangeError::Empty { at: 0 }); }
    }
    arms.push((tail.to_string(), start));
    Ok(arms)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let arms = split_ranges(s)?;
    let mut out = Vec::new();
    for (a, at) in arms { out.push(parse_arm(&a, at)?); }
    Ok(Range { arms: out })
}

fn arm_matches(v: &Version, a: &Arm, include: bool) -> bool {
    if !include && !v.pre.is_empty() && !a.anchors.contains(&core(v)) { return false; }
    a.cmps.iter().all(|c| {
        let ord = compare_versions(v, &c.v);
        match c.op {
            Op::Eq => ord == Ordering::Equal,
            Op::Lt => ord == Ordering::Less,
            Op::Le => ord != Ordering::Greater,
            Op::Gt => ord == Ordering::Greater,
            Op::Ge => ord != Ordering::Less,
        }
    })
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}
pub fn matches(v: &Version, r: &Range) -> bool { matches_with(v, r, false) }

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index: usize, pub error: VersionError }

fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (index, s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index, error })?);
    }
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) { continue; }
        if let Some(j) = best {
            let ord = compare_versions(&parsed[i], &parsed[j]);
            if (max && ord == Ordering::Greater) || (!max && ord == Ordering::Less) { best = Some(i); }
        } else { best = Some(i); }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let anchors = if include_prerelease { Vec::new() } else {
                aa.anchors.iter().copied().filter(|x| bb.anchors.contains(x)).collect()
            };
            if !anchors.is_empty() {
                for t in anchors {
                    let v = mkver(t.0, t.1, t.2, vec!["0".into()]);
                    if arm_matches(&v, aa, true) && arm_matches(&v, bb, true) { return true; }
                }
            }
            let mut candidates = Vec::new();
            candidates.push(mkver(0, 0, 0, Vec::new()));
            for c in aa.cmps.iter().chain(bb.cmps.iter()) {
                candidates.push(c.v.clone());
                let (x, y, z) = core(&c.v);
                candidates.push(mkver(x, y, z, Vec::new()));
            }
            for v in candidates {
                if v.pre.is_empty() && arm_matches(&v, aa, true) && arm_matches(&v, bb, true) { return true; }
            }
        }
    }
    false
}

fn strip_build(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() { s.push('-'); s.push_str(&v.pre.join(".")); }
    s
}

pub fn canonical(r: &Range) -> String {
    let mut rows = Vec::new();
    for a in &r.arms {
        let mut lower: Option<(Version, bool)> = None;
        let mut upper: Option<(Version, bool)> = None;
        let mut eq: Option<Version> = None;
        for c in &a.cmps {
            match c.op {
                Op::Eq => eq = Some(c.v.clone()),
                Op::Gt | Op::Ge => {
                    let inc = c.op == Op::Ge;
                    if lower.as_ref().map_or(true, |(v, i)| compare_versions(&c.v, v) == Ordering::Greater || (compare_versions(&c.v, v) == Ordering::Equal && !inc && *i)) {
                        lower = Some((c.v.clone(), inc));
                    }
                }
                Op::Lt | Op::Le => {
                    let inc = c.op == Op::Le;
                    if upper.as_ref().map_or(true, |(v, i)| compare_versions(&c.v, v) == Ordering::Less || (compare_versions(&c.v, v) == Ordering::Equal && !inc && *i)) {
                        upper = Some((c.v.clone(), inc));
                    }
                }
            }
        }
        if let Some(v) = eq { lower = Some((v.clone(), true)); upper = Some((v, true)); }
        let lo = lower.map_or("*".to_string(), |(v, i)| format!("{}{}", if i { ">=" } else { ">" }, strip_build(&v)));
        let hi = upper.map_or("*".to_string(), |(v, i)| format!("{}{}", if i { "<=" } else { "<" }, strip_build(&v)));
        let mut anchors = a.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let an = if anchors.is_empty() { "-".into() } else {
            anchors.iter().map(|(x,y,z)| format!("{x}.{y}.{z}")).collect::<Vec<_>>().join(",")
        };
        rows.push(format!("[{lo};{hi};pre={an}]"));
    }
    rows.sort();
    rows.dedup();
    if rows.is_empty() { "EMPTY".into() } else { rows.join(" || ") }
}

#[derive(Clone, Debug)]
pub enum SetExpr { Range(String), Union(Vec<SetExpr>), Intersection(Vec<SetExpr>) }
#[derive(Clone, Debug)]
pub struct Constraint { pub label: String, pub expr: SetExpr }
#[derive(Clone, Debug)]
pub struct Package { pub name: String, pub candidates: Vec<String>, pub constraints: Vec<Constraint> }
#[derive(Clone, Copy, Debug)]
pub enum Preference { Highest, Lowest, StableFirst }
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection { pub package: String, pub version: String }
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr(
    expr: &SetExpr, path: &mut Vec<usize>, package: &str, label: &str,
    out: &mut Vec<Range>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            out.push(parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.into(), label: label.into(), path: path.clone(), error
            })?);
        }
        SetExpr::Union(xs) | SetExpr::Intersection(xs) => {
            for (i, x) in xs.iter().enumerate() {
                path.push(i);
                validate_expr(x, path, package, label, out)?;
                path.pop();
            }
        }
    }
    Ok(())
}

fn eval_expr(expr: &SetExpr, v: &Version, include: bool, ranges: &mut impl Iterator<Item=&Range>) -> bool {
    match expr {
        SetExpr::Range(_) => matches_with(v, ranges.next().unwrap(), include),
        SetExpr::Union(xs) => xs.iter().any(|x| eval_expr(x, v, include, ranges)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| eval_expr(x, v, include, ranges)),
    }
}

pub fn select_packages(
    packages: &[Package], preference: Preference, include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) { return Err(SelectError::DuplicatePackage { name: p.name.clone() }); }
    }
    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }
    let mut parsed: Vec<Vec<Version>> = Vec::new();
    for p in packages {
        let mut vs = Vec::new();
        for (index, s) in p.candidates.iter().enumerate() {
            vs.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index, error
            })?);
        }
        parsed.push(vs);
    }
    let mut all_ranges: Vec<Vec<Vec<Range>>> = Vec::new();
    for p in packages {
        let mut pc = Vec::new();
        for c in &p.constraints {
            let mut rs = Vec::new();
            validate_expr(&c.expr, &mut Vec::new(), &p.name, &c.label, &mut rs)?;
            pc.push(rs);
        }
        all_ranges.push(pc);
    }
    let mut selections = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let mut accepted = Vec::new();
        for (ci, v) in parsed[pi].iter().enumerate() {
            let ok = p.constraints.iter().enumerate().all(|(ki, c)| {
                eval_expr(&c.expr, v, include_prerelease, &mut all_ranges[pi][ki].iter())
            });
            if ok { accepted.push(ci); }
        }
        if accepted.is_empty() {
            let mut conflicts: Vec<Vec<String>> = Vec::new();
            if !p.candidates.is_empty() {
                let n = p.constraints.len();
                for mask in 1usize..(1usize << n) {
                    let count = mask.count_ones() as usize;
                    if conflicts.first().map_or(false, |x| x.len() < count) { continue; }
                    let sat = accepted_subset(mask, p, &parsed[pi], include_prerelease, &all_ranges[pi]);
                    if !sat {
                        let mut labs = (0..n).filter(|i| mask & (1 << i) != 0).map(|i| p.constraints[i].label.clone()).collect::<Vec<_>>();
                        labs.sort_by(|a,b| a.as_bytes().cmp(b.as_bytes()));
                        conflicts.push(labs);
                    }
                }
            }
            conflicts.sort_by(|a,b| a.len().cmp(&b.len()).then_with(|| a.join("\0").as_bytes().cmp(b.join("\0").as_bytes())));
            return Err(SelectError::NoSolution {
                package: p.name.clone(),
                labels: conflicts.into_iter().next().unwrap_or_default(),
            });
        }
        let is_better = |i: usize, j: usize| -> bool {
            let a = &parsed[pi][i]; let b = &parsed[pi][j];
            match preference {
                Preference::Highest => compare_versions(a,b) == Ordering::Greater,
                Preference::Lowest => compare_versions(a,b) == Ordering::Less,
                Preference::StableFirst => {
                    if a.pre.is_empty() != b.pre.is_empty() { a.pre.is_empty() }
                    else { compare_versions(a,b) == Ordering::Greater }
                }
            }
        };
        let mut best = accepted[0];
        for &i in &accepted[1..] { if is_better(i, best) { best = i; } }
        selections.push(Selection { package: p.name.clone(), version: p.candidates[best].clone() });
    }
    Ok(selections)
}

fn accepted_subset(mask: usize, p: &Package, parsed: &[Version], include: bool, ranges: &[Vec<Range>]) -> bool {
    parsed.iter().enumerate().any(|(ci, v)| {
        p.constraints.iter().enumerate().all(|(ki, c)| {
            mask & (1 << ki) == 0 || eval_expr(&c.expr, v, include, &mut ranges[ki].iter())
        })
    })
}
