#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanCase {
    Upper,
    Lower,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RomanError {
    Empty,
    InvalidByte { index: usize },
    MixedCase { index: usize },
    NonCanonical { index: usize },
    Range,
}

pub fn roman_encode(value: u32, case: RomanCase) -> Result<String, RomanError> {
    if value > 3999 {
        return Err(RomanError::Range);
    }

    if value == 0 {
        return Ok(match case {
            RomanCase::Upper => "N".to_string(),
            RomanCase::Lower => "n".to_string(),
        });
    }

    let tokens: &[(u32, &str)] = &[
        (1000, "M"),
        (900, "CM"),
        (500, "D"),
        (400, "CD"),
        (100, "C"),
        (90, "XC"),
        (50, "L"),
        (40, "XL"),
        (10, "X"),
        (9, "IX"),
        (5, "V"),
        (4, "IV"),
        (1, "I"),
    ];

    let mut remaining = value;
    let mut result = String::new();

    for &(amount, token) in tokens {
        while remaining >= amount {
            result.push_str(token);
            remaining -= amount;
        }
    }

    if let RomanCase::Lower = case {
        result.make_ascii_lowercase();
    }

    Ok(result)
}

pub fn roman_decode(input: &str) -> Result<u32, RomanError> {
    if input.is_empty() {
        return Err(RomanError::Empty);
    }

    let bytes = input.as_bytes();

    for (index, &byte) in bytes.iter().enumerate() {
        if !matches!(
            byte,
            b'I' | b'V' | b'X' | b'L' | b'C' | b'D' | b'M' | b'N'
                | b'i' | b'v' | b'x' | b'l' | b'c' | b'd' | b'm' | b'n'
        ) {
            return Err(RomanError::InvalidByte { index });
        }
    }

    let mut case = None;
    for (index, &byte) in bytes.iter().enumerate() {
        let current = if byte.is_ascii_uppercase() {
            Some(true)
        } else {
            Some(false)
        };

        if let Some(first_case) = case {
            if first_case != current.unwrap() {
                return Err(RomanError::MixedCase { index });
            }
        } else {
            case = current;
        }
    }

    if input == "N" || input == "n" {
        return Ok(0);
    }

    if let Some(index) = bytes.iter().position(|&byte| byte == b'N' || byte == b'n') {
        let _ = index;
        return Err(RomanError::NonCanonical { index: 0 });
    }

    let mut provisional = 0u32;
    let mut i = 0;
    while i < bytes.len() {
        let current = roman_symbol_value(bytes[i]);
        if i + 1 < bytes.len() {
            let next = roman_symbol_value(bytes[i + 1]);
            if next > current {
                provisional = provisional.saturating_add(next - current).min(4000);
                i += 2;
                if provisional > 3999 {
                    return Err(RomanError::Range);
                }
                continue;
            }
        }
        provisional = provisional.saturating_add(current).min(4000);
        i += 1;
        if provisional > 3999 {
            return Err(RomanError::Range);
        }
    }

    let input_case = if bytes[0].is_ascii_uppercase() {
        RomanCase::Upper
    } else {
        RomanCase::Lower
    };
    let canonical = roman_encode(provisional, input_case)?;

    let canonical_bytes = canonical.as_bytes();
    let common_len = bytes.len().min(canonical_bytes.len());
    for index in 0..common_len {
        if bytes[index] != canonical_bytes[index] {
            return Err(RomanError::NonCanonical { index });
        }
    }
    if bytes.len() != canonical_bytes.len() {
        return Err(RomanError::NonCanonical { index: common_len });
    }

    Ok(provisional)
}

fn roman_symbol_value(byte: u8) -> u32 {
    match byte.to_ascii_uppercase() {
        b'I' => 1,
        b'V' => 5,
        b'X' => 10,
        b'L' => 50,
        b'C' => 100,
        b'D' => 500,
        b'M' => 1000,
        _ => 0,
    }
}

fn main() {}
