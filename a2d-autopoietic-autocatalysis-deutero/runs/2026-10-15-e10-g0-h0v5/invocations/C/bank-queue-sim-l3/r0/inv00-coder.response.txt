#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    remaining: u64,
    started: bool,
    paused_at: Option<u64>,
}

#[derive(Clone)]
struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiting_better(a: &Waiting, b: &Waiting) -> bool {
    if a.customer.priority != b.customer.priority {
        return a.customer.priority < b.customer.priority;
    }
    match (a.paused_at, b.paused_at) {
        (Some(ta), Some(tb)) if ta != tb => ta < tb,
        (Some(_), None) => true,
        (None, Some(_)) => false,
        _ => {
            if a.started != b.started {
                a.started
            } else if a.started {
                a.customer.id < b.customer.id
            } else if a.customer.arrival != b.customer.arrival {
                a.customer.arrival < b.customer.arrival
            } else {
                a.customer.id > b.customer.id
            }
        }
    }
}

fn best_waiter_index(waiting: &[Waiting]) -> Option<usize> {
    let mut best = None;
    for i in 0..waiting.len() {
        if best.map_or(true, |b| waiting_better(&waiting[i], &waiting[b])) {
            best = Some(i);
        }
    }
    best
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::new();
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;

    while next_arrival < arrivals.len() || running.is_some() || !waiting.is_empty() {
        let next_time = {
            let mut t = arrivals.get(next_arrival).map(|c| c.arrival);
            if let Some(r) = &running {
                for candidate in [Some(r.finish), (r.work_at > 0).then_some(r.work_at)] {
                    if let Some(x) = candidate {
                        t = Some(t.map_or(x, |cur| cur.min(x)));
                    }
                }
            }
            match t {
                Some(x) => x,
                None => break,
            }
        };

        if running.as_ref().is_some_and(|r| r.finish == next_time) {
            let r = running.take().unwrap();
            events.push(Event {
                time: next_time,
                customer: r.customer.id,
                kind: Kind::Finish,
            });
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == next_time {
            let c = arrivals[next_arrival].clone();
            events.push(Event {
                time: next_time,
                customer: c.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                remaining: c.service,
                customer: c,
                started: false,
                paused_at: None,
            });
            next_arrival += 1;
        }

        let can_preempt = running
            .as_ref()
            .is_some_and(|r| next_time >= r.work_at && next_time < r.finish);
        if can_preempt {
            if let Some(best) = best_waiter_index(&waiting) {
                let superior = waiting[best].customer.priority
                    < running.as_ref().unwrap().customer.priority;
                if superior {
                    let by = waiting[best].customer.id;
                    let r = running.take().unwrap();
                    let remaining = r.finish - next_time;
                    events.push(Event {
                        time: next_time,
                        customer: r.customer.id,
                        kind: Kind::Preempt { remaining, by },
                    });
                    waiting.push(Waiting {
                        customer: r.customer,
                        remaining,
                        started: true,
                        paused_at: Some(next_time),
                    });
                }
            }
        }

        loop {
            if running.is_some() {
                break;
            }
            let Some(index) = best_waiter_index(&waiting) else {
                break;
            };
            let w = waiting.remove(index);
            let cost = if last_dispatched == Some(w.customer.id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };
            let work_at = next_time.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: w.customer.id,
                time: next_time,
            })?;
            let finish = work_at.checked_add(w.remaining).ok_or(Error::TimeOverflow {
                customer: w.customer.id,
                time: next_time,
            })?;

            let kind = if w.started {
                Kind::Resume {
                    remaining: w.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: w.remaining,
                    work_at,
                    finish,
                }
            };
            events.push(Event {
                time: next_time,
                customer: w.customer.id,
                kind,
            });
            last_dispatched = Some(w.customer.id);
            running = Some(Running {
                customer: w.customer,
                remaining: w.remaining,
                work_at,
                finish,
            });

            if finish == next_time {
                let r = running.take().unwrap();
                events.push(Event {
                    time: next_time,
                    customer: r.customer.id,
                    kind: Kind::Finish,
                });
                continue;
            }
            break;
        }
    }

    Ok(events)
}

fn main() {}
