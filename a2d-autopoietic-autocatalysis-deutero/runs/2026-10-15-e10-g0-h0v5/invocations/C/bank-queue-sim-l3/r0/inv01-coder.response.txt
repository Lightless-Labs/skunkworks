```rust
use std::collections::HashSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    remaining: u64,
    started: bool,
    paused_at: Option<u64>,
}

struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiting_better(a: &Waiting, b: &Waiting) -> bool {
    if a.customer.priority != b.customer.priority {
        return a.customer.priority < b.customer.priority;
    }

    match (a.paused_at, b.paused_at) {
        (Some(a_time), Some(b_time)) if a_time != b_time => a_time < b_time,
        (Some(_), None) => true,
        (None, Some(_)) => false,
        _ => {
            if a.started != b.started {
                a.started
            } else if a.started {
                a.customer.id < b.customer.id
            } else if a.customer.arrival != b.customer.arrival {
                a.customer.arrival < b.customer.arrival
            } else {
                a.customer.id > b.customer.id
            }
        }
    }
}

fn best_waiter_index(waiting: &[Waiting]) -> Option<usize> {
    let mut best = None;
    for i in 0..waiting.len() {
        if best.map_or(true, |j| waiting_better(&waiting[i], &waiting[j])) {
            best = Some(i);
        }
    }
    best
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut ids = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;

    loop {
        if next_arrival == arrivals.len() && running.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time = arrivals.get(next_arrival).map(|c| c.arrival);
        if let Some(active) = running.as_ref() {
            for time in [Some(active.finish), Some(active.work_at)] {
                if let Some(time) = time {
                    next_time = Some(next_time.map_or(time, |current| current.min(time)));
                }
            }
        }

        let Some(time) = next_time else {
            break;
        };

        if running.as_ref().is_some_and(|active| active.finish == time) {
            let active = running.take().unwrap();
            events.push(Event {
                time,
                customer: active.customer.id,
                kind: Kind::Finish,
            });
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                remaining: customer.service,
                customer,
                started: false,
                paused_at: None,
            });
            next_arrival += 1;
        }

        let can_preempt = running
            .as_ref()
            .is_some_and(|active| time >= active.work_at && time < active.finish);

        if can_preempt {
            if let Some(index) = best_waiter_index(&waiting) {
                let should_preempt = waiting[index].customer.priority
                    < running.as_ref().unwrap().customer.priority;
                if should_preempt {
                    let by = waiting[index].customer.id;
                    let active = running.take().unwrap();
                    let remaining = active.finish - time;
                    events.push(Event {
                        time,
                        customer: active.customer.id,
                        kind: Kind::Preempt { remaining, by },
                    });
                    waiting.push(Waiting {
                        customer: active.customer,
                        remaining,
                        started: true,
                        paused_at: Some(time),
                    });
                }
            }
        }

        while running.is_none() {
            let Some(index) = best_waiter_index(&waiting) else {
                break;
            };
            let customer = waiting.remove(index);
            let cost = if last_dispatched.is_some_and(|id| id != customer.customer.id) {
                switch_cost
            } else {
                0
            };
            let work_at = time.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: customer.customer.id,
                time,
            })?;
            let finish = work_at.checked_add(customer.remaining).ok_or(Error::TimeOverflow {
                customer: customer.customer.id,
                time,
            })?;

            let kind = if customer.started {
                Kind::Resume {
                    remaining: customer.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: customer.remaining,
                    work_at,
                    finish,
                }
            };
            events.push(Event {
                time,
                customer: customer.customer.id,
                kind,
            });
            last_dispatched = Some(customer.customer.id);
            running = Some(Running {
                customer: customer.customer,
                remaining: customer.remaining,
                work_at,
                finish,
            });

            if finish == time {
                let finished = running.take().unwrap();
                events.push(Event {
                    time,
                    customer: finished.customer.id,
                    kind: Kind::Finish,
                });
            } else {
                break;
            }
        }
    }

    Ok(events)
}

fn main() {}
```
