#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    ContextTooLarge { context: usize },
    CarriageReturn { side: Side, byte: usize },
    MissingFinalLf { side: Side },
    TooLarge { old_len: usize, new_len: usize },
}

fn split_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        return Vec::new();
    }

    text.strip_suffix('\n')
        .unwrap()
        .split('\n')
        .collect()
}

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }

    if let Some(byte) = old.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn {
            side: Side::Old,
            byte,
        });
    }
    if let Some(byte) = new.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn {
            side: Side::New,
            byte,
        });
    }

    if !old.is_empty() && !old.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::Old });
    }
    if !new.is_empty() && !new.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::New });
    }

    let old_lines = split_lines(old);
    let new_lines = split_lines(new);

    if old_lines.len() > 128 || new_lines.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old_lines.len(),
            new_len: new_lines.len(),
        });
    }

    let m = old_lines.len();
    let n = new_lines.len();
    let mut lcs = vec![vec![0usize; n + 1]; m + 1];

    for i in (0..m).rev() {
        for j in (0..n).rev() {
            lcs[i][j] = if old_lines[i] == new_lines[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut edits = Vec::new();
    let (mut i, mut j) = (0, 0);
    while i < m && j < n {
        if old_lines[i] == new_lines[j] {
            edits.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old_lines[i].to_string(),
            });
            i += 1;
            j += 1;
        } else if lcs[i + 1][j] > lcs[i][j + 1]
            || (lcs[i + 1][j] == lcs[i][j + 1] && (i + j) % 2 == 0)
        {
            edits.push(Edit::Delete {
                old_index: i,
                line: old_lines[i].to_string(),
            });
            i += 1;
        } else {
            edits.push(Edit::Insert {
                new_index: j,
                line: new_lines[j].to_string(),
            });
            j += 1;
        }
    }
    while i < m {
        edits.push(Edit::Delete {
            old_index: i,
            line: old_lines[i].to_string(),
        });
        i += 1;
    }
    while j < n {
        edits.push(Edit::Insert {
            new_index: j,
            line: new_lines[j].to_string(),
        });
        j += 1;
    }

    let is_change = |edit: &Edit| !matches!(edit, Edit::Keep { .. });
    if !edits.iter().any(is_change) {
        return Ok(String::new());
    }

    let mut groups = Vec::new();
    let mut k = 0;
    while k < edits.len() {
        if !is_change(&edits[k]) {
            k += 1;
            continue;
        }
        let start = k;
        while k < edits.len() && is_change(&edits[k]) {
            k += 1;
        }
        groups.push((start, k));
    }

    let merge_distance = 2 * context;
    let mut merged = Vec::new();
    for group in groups {
        if let Some(last) = merged.last_mut() {
            let (last_start, last_end): &mut (usize, usize) = last;
            let keeps_between = edits[*last_end..group.0]
                .iter()
                .filter(|edit| matches!(edit, Edit::Keep { .. }))
                .count();
            if keeps_between <= merge_distance {
                *last_end = group.1;
                continue;
            }
            let _ = last_start;
        }
        merged.push(group);
    }

    let mut out = String::from("--- before\n+++ after\n");
    for (group_start, group_end) in merged {
        let mut start = group_start;
        let mut preceding = 0;
        while start > 0 && preceding < context {
            if !matches!(edits[start - 1], Edit::Keep { .. }) {
                break;
            }
            start -= 1;
            preceding += 1;
        }

        let mut end = group_end;
        let mut following = 0;
        while end < edits.len() && following < context {
            if !matches!(edits[end], Edit::Keep { .. }) {
                break;
            }
            end += 1;
            following += 1;
        }

        let old_count = edits[start..end]
            .iter()
            .filter(|edit| !matches!(edit, Edit::Insert { .. }))
            .count();
        let new_count = edits[start..end]
            .iter()
            .filter(|edit| !matches!(edit, Edit::Delete { .. }))
            .count();

        let old_before = edits[..start]
            .iter()
            .filter(|edit| !matches!(edit, Edit::Insert { .. }))
            .count();
        let new_before = edits[..start]
            .iter()
            .filter(|edit| !matches!(edit, Edit::Delete { .. }))
            .count();

        let old_start = if old_count > 0 {
            old_before + 1
        } else {
            old_before
        };
        let new_start = if new_count > 0 {
            new_before + 1
        } else {
            new_before
        };

        out.push_str(&format!(
            "@@ -{old_start},{old_count} +{new_start},{new_count} @@\n"
        ));

        for edit in &edits[start..end] {
            match edit {
                Edit::Keep { line, .. } => {
                    out.push(' ');
                    out.push_str(line);
                    out.push('\n');
                }
                Edit::Delete { line, .. } => {
                    out.push('-');
                    out.push_str(line);
                    out.push('\n');
                }
                Edit::Insert { line, .. } => {
                    out.push('+');
                    out.push_str(line);
                    out.push('\n');
                }
            }
        }
    }

    Ok(out)
}
