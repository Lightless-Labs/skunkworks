#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, base: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let payload_start = if let Some(tag_end) = bytes.iter().position(|&b| b == b'#') {
        let expected = base.to_string();
        if &bytes[..tag_end] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        tag_end + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    // Validate every digit before checking separator placement.
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }

        match digit_value(byte) {
            Some(value) if value < base => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || payload[offset - 1] == b'_'
                || payload[offset + 1] == b'_')
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add(value: &mut Vec<u8>, multiplier: u8, addend: u8) {
    let mut carry = addend as u16;
    for digit in value.iter_mut() {
        let next = (*digit as u16) * (multiplier as u16) + carry;
        *digit = (next % 10) as u8;
        carry = next / 10;
    }
    while carry != 0 {
        value.push((carry % 10) as u8);
        carry /= 10;
    }
}

fn add_values(left: &[u8], right: &[u8], base: u8) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    let mut index = 0;

    while index < left.len() || index < right.len() || carry != 0 {
        let sum = left.get(index).copied().unwrap_or(0) as u16
            + right.get(index).copied().unwrap_or(0) as u16
            + carry;
        result.push((sum % base as u16) as u8);
        carry = sum / base as u16;
        index += 1;
    }

    if result.is_empty() {
        result.push(0);
    }
    result
}

fn render(mut digits: Vec<u8>, style: BaseStyle) -> String {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }

    while digits.len() < style.min_width {
        digits.push(0);
    }

    let mut chars = Vec::with_capacity(digits.len());
    for &digit in digits.iter().rev() {
        let byte = if digit < 10 {
            b'0' + digit
        } else if style.upper {
            b'A' + (digit - 10)
        } else {
            b'a' + (digit - 10)
        };
        chars.push(byte as char);
    }

    if style.group == 0 || chars.len() <= style.group {
        return chars.into_iter().collect();
    }

    let mut output = String::with_capacity(chars.len() + chars.len() / style.group);
    let first_group = {
        let remainder = chars.len() % style.group;
        if remainder == 0 { style.group } else { remainder }
    };

    for (index, ch) in chars.into_iter().enumerate() {
        if index != 0 && (index - first_group) % style.group == 0 {
            output.push('_');
        }
        output.push(ch);
    }

    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let mut decimal = vec![0u8];
    for digit in digits {
        multiply_add(&mut decimal, from, digit);
    }

    let mut converted = Vec::new();
    let mut work = decimal;
    while !(work.len() == 1 && work[0] == 0) {
        let mut quotient = Vec::with_capacity(work.len());
        let mut remainder = 0u16;
        for &digit in work.iter().rev() {
            let current = remainder * 10 + digit as u16;
            let q = current / to as u16;
            remainder = current % to as u16;
            if !quotient.is_empty() || q != 0 {
                quotient.push(q as u8);
            }
        }
        converted.push(remainder as u8);
        work = quotient;
    }
    if converted.is_empty() {
        converted.push(0);
    }

    Ok(render(converted, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let mut left_value = vec![0u8];
    for digit in left_digits {
        multiply_add(&mut left_value, base, digit);
    }

    let mut right_value = vec![0u8];
    for digit in right_digits {
        multiply_add(&mut right_value, base, digit);
    }

    let decimal_sum = add_values(&left_value, &right_value, 10);
    let mut converted = Vec::new();
    let mut work = decimal_sum;

    while !(work.len() == 1 && work[0] == 0) {
        let mut quotient = Vec::with_capacity(work.len());
        let mut remainder = 0u16;
        for &digit in work.iter().rev() {
            let current = remainder * 10 + digit as u16;
            let q = current / base as u16;
            remainder = current % base as u16;
            if !quotient.is_empty() || q != 0 {
                quotient.push(q as u8);
            }
        }
        converted.push(remainder as u8);
        work = quotient;
    }

    if converted.is_empty() {
        converted.push(0);
    }

    Ok(render(converted, style))
}
