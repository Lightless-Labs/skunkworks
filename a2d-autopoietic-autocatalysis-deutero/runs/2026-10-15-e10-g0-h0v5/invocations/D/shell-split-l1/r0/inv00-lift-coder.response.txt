#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match ch {
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    words.push(std::mem::take(&mut word));
                    active = false;
                }
            }
            '\'' => {
                active = true;
                let mut closed = false;
                while let Some((_, inner)) = chars.next() {
                    if inner == '\'' {
                        closed = true;
                        break;
                    }
                    word.push(inner);
                }
                if !closed {
                    return Err(SplitError::UnterminatedSingle { pos });
                }
            }
            '"' => {
                active = true;
                let mut closed = false;
                while let Some((inner_pos, inner)) = chars.next() {
                    match inner {
                        '"' => {
                            closed = true;
                            break;
                        }
                        '\\' => {
                            let Some(&(next_pos, next)) = chars.peek() else {
                                return Err(SplitError::TrailingEscape { pos: inner_pos });
                            };
                            if matches!(next, '"' | '\\' | '$' | '\n') {
                                chars.next();
                                if next != '\n' {
                                    word.push(next);
                                }
                            } else {
                                word.push('\\');
                                let _ = next_pos;
                            }
                        }
                        _ => word.push(inner),
                    }
                }
                if !closed {
                    return Err(SplitError::UnterminatedDouble { pos });
                }
            }
            '\\' => {
                if let Some(&(next_pos, next)) = chars.peek() {
                    if next == '\n' {
                        chars.next();
                    } else {
                        chars.next();
                        word.push(next);
                        active = true;
                    }
                    let _ = next_pos;
                } else {
                    return Err(SplitError::TrailingEscape { pos });
                }
            }
            _ => {
                word.push(ch);
                active = true;
            }
        }
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
