#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

fn encode_group(n: u16) -> String {
    let mut out = String::new();
    let hundreds = n / 100;
    let remainder = n % 100;

    if hundreds != 0 {
        out.push_str(SMALL[hundreds as usize]);
        out.push_str(" hundred");
        if remainder != 0 {
            out.push_str(" and ");
        }
    }

    if remainder != 0 {
        if remainder < 20 {
            out.push_str(SMALL[remainder as usize]);
        } else {
            out.push_str(TENS[(remainder / 10 - 2) as usize]);
            if remainder % 10 != 0 {
                out.push('-');
                out.push_str(SMALL[(remainder % 10) as usize]);
            }
        }
    }
    out
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in groups.iter_mut().rev() {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let mut parts: Vec<String> = Vec::new();
    for index in 0..7 {
        let n = groups[index];
        if n == 0 {
            continue;
        }
        let mut part = encode_group(n);
        if index < 6 {
            part.push(' ');
            part.push_str(SCALES[5 - index]);
        }
        parts.push(part);
    }

    let mut result = String::new();
    for (i, part) in parts.iter().enumerate() {
        if i != 0 {
            result.push(' ');
        }
        result.push_str(part);
    }

    let units = groups[6];
    if units != 0 && units < 100 && parts.len() > 1 {
        let separator_index = result.rfind(' ').unwrap();
        result.replace_range(separator_index..=separator_index, " and ");
    }

    if value < 0 {
        Ok(format!("minus {result}"))
    } else {
        Ok(result)
    }
}

fn word_value(word: &str) -> Option<(u128, WordKind)> {
    if word == "zero" {
        return Some((0, WordKind::Number));
    }
    for (i, candidate) in SMALL.iter().enumerate().skip(1) {
        if word == *candidate {
            return Some((i as u128, WordKind::Number));
        }
    }
    for (i, candidate) in TENS.iter().enumerate() {
        if word == *candidate {
            return Some(((i as u128 + 2) * 10, WordKind::Number));
        }
    }
    match word {
        "hundred" => Some((100, WordKind::Hundred)),
        "thousand" => Some((1_000, WordKind::Scale)),
        "million" => Some((1_000_000, WordKind::Scale)),
        "billion" => Some((1_000_000_000, WordKind::Scale)),
        "trillion" => Some((1_000_000_000_000, WordKind::Scale)),
        "quadrillion" => Some((1_000_000_000_000_000, WordKind::Scale)),
        "quintillion" => Some((1_000_000_000_000_000_000, WordKind::Scale)),
        "and" => Some((0, WordKind::And)),
        _ => None,
    }
}

#[derive(Clone, Copy)]
enum WordKind {
    Number,
    Hundred,
    Scale,
    And,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            let left_ok = index > 0 && bytes[index - 1].is_ascii_lowercase();
            let right_ok = index + 1 < bytes.len() && bytes[index + 1].is_ascii_lowercase();
            if !left_ok || !right_ok {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (i, (token, index)) in tokens.iter().enumerate() {
        if *token == "minus" && i == 0 {
            continue;
        }
        if word_value(token).is_none() {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let negative = tokens.first().map(|(word, _)| *word == "minus").unwrap_or(false);
    let arithmetic = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for (token, _) in arithmetic {
        let (value, kind) = word_value(token).ok_or(WordsError::Range)?;
        match kind {
            WordKind::Number => group += value,
            WordKind::Hundred => group *= 100,
            WordKind::Scale => {
                total += group * value;
                group = 0;
            }
            WordKind::And => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let signed = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    if canonical.as_bytes() != input.as_bytes() {
        let mismatch = canonical
            .bytes()
            .zip(input.bytes())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index: mismatch });
    }
    Ok(signed)
}
