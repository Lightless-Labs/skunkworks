#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

fn err_at(kind: fn(usize) -> EmailError, at: usize) -> EmailError {
    kind(at)
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, Option<usize>, Option<usize>)>,
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();

    let mut i = 0;
    let mut comment_depth = 0usize;
    let mut outer_comment = 0usize;
    let mut quote_mode = false;
    let mut quote_escape = false;
    let mut run_start: Option<usize> = None;
    let mut run_left: Option<usize> = None;
    let mut run_right: Option<usize> = None;

    while i < input.len() {
        let b = input[i];

        if quote_mode {
            bytes.push(b);
            positions.push(i);
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                quote_mode = false;
            }
            i += 1;
            continue;
        }

        if comment_depth > 0 {
            if is_hws(b) {
                if run_start.is_none() {
                    run_start = Some(i);
                    run_left = bytes.len().checked_sub(1);
                }
                i += 1;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if b == b'\r'
                    && i + 2 < input.len()
                    && input[i + 1] == b'\n'
                    && is_hws(input[i + 2])
                {
                    if run_start.is_none() {
                        run_start = Some(i);
                        run_left = bytes.len().checked_sub(1);
                    }
                    i += 2;
                    while i < input.len() && is_hws(input[i]) {
                        i += 1;
                    }
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'\\' {
                if i + 1 >= input.len()
                    || !(0x20..=0x7e).contains(&input[i + 1])
                {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b < 0x20 || b > 0x7e {
                return Err(EmailError::CommentChar { at: i });
            }
            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_depth += 1;
            } else if b == b')' {
                comment_depth -= 1;
                if comment_depth == 0 {
                    let start = outer_comment;
                    if run_start.is_none() {
                        run_start = Some(start);
                        run_left = bytes.len().checked_sub(1);
                    }
                    i += 1;
                    continue;
                }
            }
            i += 1;
            continue;
        }

        let is_fold = b == b'\r'
            && i + 2 < input.len()
            && input[i + 1] == b'\n'
            && is_hws(input[i + 2]);

        if is_hws(b) || is_fold {
            if run_start.is_none() {
                run_start = Some(i);
                run_left = bytes.len().checked_sub(1);
            }
            if is_fold {
                i += 2;
                while i < input.len() && is_hws(input[i]) {
                    i += 1;
                }
            } else {
                i += 1;
            }
            continue;
        }

        if b == b'\r' || b == b'\n' {
            return Err(EmailError::BadFold { at: i });
        }

        if b == b'(' {
            if run_start.is_none() {
                run_start = Some(i);
                run_left = bytes.len().checked_sub(1);
            }
            outer_comment = i;
            comment_depth = 1;
            i += 1;
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if let Some(start) = run_start.take() {
            run_right = Some(bytes.len());
            runs.push((start, run_left.take(), run_right.take()));
        }

        if bytes.is_empty() && b == b'"' {
            quote_mode = true;
        }
        bytes.push(b);
        positions.push(i);
        i += 1;
    }

    if comment_depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }
    if let Some(start) = run_start.take() {
        runs.push((start, run_left.take(), None));
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn parse_internal(input: &str, allow_single_label: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    let b = &cleaned.bytes;
    let pos = &cleaned.positions;

    for &(start, left, right) in &cleaned.runs {
        let legal = left.is_none()
            || right.is_none()
            || left.map(|n| b[n] == b'@').unwrap_or(false)
            || right.map(|n| b[n] == b'@').unwrap_or(false);
        if !legal {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }

    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut protected_end = 0usize;
    if b[0] == b'"' {
        let mut i = 1usize;
        let mut closed = false;
        while i < b.len() {
            if b[i] == b'\\' {
                i += 2;
            } else if b[i] == b'"' {
                protected_end = i + 1;
                closed = true;
                break;
            } else {
                i += 1;
            }
        }
        if !closed {
            return Err(EmailError::UnclosedQuote { at: pos[0] });
        }
    }

    let separators: Vec<usize> = (protected_end..b.len())
        .filter(|&i| b[i] == b'@')
        .collect();
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: pos[separators[1]],
        });
    }
    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: pos[sep] });
    }

    let mut local = String::new();
    let mut local_origins = Vec::new();
    let quoted;

    if b[0] == b'"' {
        let close = protected_end - 1;
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: pos[close + 1],
            });
        }
        quoted = true;
        let mut i = 1usize;
        while i < close {
            if b[i] == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape { at: pos[i] });
                }
                match b[i + 1] {
                    b'"' | b'\\' | b'@' => {
                        local.push(b[i + 1] as char);
                        local_origins.push(pos[i]);
                        i += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: pos[i] }),
                }
            } else {
                if !(0x20..=0x7e).contains(&b[i]) {
                    return Err(EmailError::QuotedChar { at: pos[i] });
                }
                local.push(b[i] as char);
                local_origins.push(pos[i]);
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: pos[close] });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_origins[64],
            });
        }
    } else {
        quoted = false;
        if sep > 64 {
            return Err(EmailError::LocalTooLong { at: pos[64] });
        }
        for i in 0..sep {
            let c = b[i];
            if c == b'.' {
                if i == 0 || i + 1 == sep || b[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: pos[i] });
                }
            } else if !c.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&c)
            {
                return Err(EmailError::LocalChar { at: pos[i] });
            }
            local.push(c as char);
            local_origins.push(pos[i]);
        }
    }

    let domain_start = sep + 1;
    let domain_start_pos = if domain_start < b.len() {
        pos[domain_start]
    } else {
        input.len()
    };
    if domain_start == b.len() {
        return Err(EmailError::EmptyDomain {
            at: domain_start_pos,
        });
    }

    let mut domain_end = b.len();
    let has_root_dot = b[domain_end - 1] == b'.';
    if has_root_dot {
        domain_end -= 1;
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: domain_start_pos,
        });
    }

    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos[domain_start + 253],
        });
    }

    let mut labels = Vec::new();
    let mut start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || b[i] == b'.' {
            if start == i {
                let at = if i < b.len() {
                    pos[i]
                } else if has_root_dot {
                    pos[b.len() - 1]
                } else {
                    input.len()
                };
                return Err(EmailError::EmptyLabel { at });
            }
            let len = i - start;
            if len > 63 {
                return Err(EmailError::LabelTooLong { at: pos[start + 63] });
            }
            if b[start] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos[start] });
            }
            if b[i - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos[i - 1] });
            }
            for j in start..i {
                if !b[j].is_ascii_alphanumeric() && b[j] != b'-' {
                    return Err(EmailError::DomainChar { at: pos[j] });
                }
            }
            labels.push((start, i));
            start = i + 1;
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: domain_start_pos,
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if b[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos[last_start] });
    }

    let domain: String = b[domain_start..domain_end]
        .iter()
        .map(|&c| (c as char).to_ascii_lowercase())
        .collect();

    Ok((
        Address {
            local,
            domain,
            quoted,
        },
        local_origins,
    ))
}

fn valid_dot_atom(local: &str) -> bool {
    let b = local.as_bytes();
    if b.is_empty() || b.len() > 64 {
        return false;
    }
    for (i, &c) in b.iter().enumerate() {
        if c == b'.' {
            if i == 0 || i + 1 == b.len() || b[i - 1] == b'.' {
                return false;
            }
        } else if !c.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&c) {
            return false;
        }
    }
    true
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_internal(input, false).map(|x| x.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_internal(input, policy.allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(index) = address.local.find('+') {
            if index == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.get(index).copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(index);
        }
    }

    let safe = valid_dot_atom(&address.local);
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let local = if address.quoted {
        let mut out = String::with_capacity(address.local.len() + 2);
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
        out
    } else {
        address.local.clone()
    };
    format!("{}@{}", local, address.domain)
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: Vec<(String, String, usize)> = Vec::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                if let Some((_, _, first_index)) = seen
                    .iter()
                    .find(|(local, domain, _)| local == &address.local && domain == &address.domain)
                {
                    duplicate_of[index] = Some(*first_index);
                } else {
                    seen.push((address.local.clone(), address.domain.clone(), index));
                    addresses.push(address);
                    source_indices.push(index);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
