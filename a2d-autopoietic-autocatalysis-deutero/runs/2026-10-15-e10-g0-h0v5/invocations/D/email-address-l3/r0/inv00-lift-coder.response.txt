#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let mut clean = Vec::<u8>::new();
    let mut positions = Vec::<usize>::new();
    let mut runs = Vec::<Run>::new();
    let mut i = 0usize;
    let mut first_retained = true;

    while i < bytes.len() {
        if first_retained && bytes[i] == b'"' {
            let mut j = i;
            while j < bytes.len() {
                let b = bytes[j];
                clean.push(b);
                positions.push(j);
                j += 1;
                if b == b'\\' {
                    if j < bytes.len() {
                        clean.push(bytes[j]);
                        positions.push(j);
                        j += 1;
                    }
                } else if b == b'"' && j > i + 1 {
                    break;
                }
            }
            first_retained = false;
            i = j;
            continue;
        }

        let start = i;
        let mut removed = false;
        loop {
            if i >= bytes.len() {
                break;
            }
            if is_hws(bytes[i]) {
                removed = true;
                i += 1;
                continue;
            }
            if bytes[i] == b'\r' || bytes[i] == b'\n' {
                let at = i;
                if bytes[i] == b'\r'
                    && i + 2 < bytes.len()
                    && bytes[i + 1] == b'\n'
                    && is_hws(bytes[i + 2])
                {
                    removed = true;
                    i += 2;
                    while i < bytes.len() && is_hws(bytes[i]) {
                        i += 1;
                    }
                    continue;
                }
                return Err(EmailError::BadFold { at });
            }
            if bytes[i] == b'(' {
                removed = true;
                let outer = i;
                let mut stack = vec![i];
                i += 1;
                while !stack.is_empty() {
                    if i >= bytes.len() {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                    let b = bytes[i];
                    if b == b'\\' {
                        let slash = i;
                        if i + 1 >= bytes.len() || !printable(bytes[i + 1]) {
                            return Err(EmailError::CommentEscape { at: slash });
                        }
                        i += 2;
                        continue;
                    }
                    if b == b'\r' || b == b'\n' {
                        let at = i;
                        if b == b'\r'
                            && i + 2 < bytes.len()
                            && bytes[i + 1] == b'\n'
                            && is_hws(bytes[i + 2])
                        {
                            i += 2;
                            while i < bytes.len() && is_hws(bytes[i]) {
                                i += 1;
                            }
                            continue;
                        }
                        return Err(EmailError::BadFold { at });
                    }
                    if b == b'(' {
                        if stack.len() == 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        stack.push(i);
                        i += 1;
                        continue;
                    }
                    if b == b')' {
                        stack.pop();
                        i += 1;
                        continue;
                    }
                    if !(printable(b) || b == b'\t') {
                        return Err(EmailError::CommentChar { at: i });
                    }
                    i += 1;
                }
                continue;
            }
            if bytes[i] == b')' {
                return Err(EmailError::UnexpectedClose { at: i });
            }
            break;
        }
        if removed {
            runs.push(Run { start, end: i });
        }
        if i < bytes.len() {
            clean.push(bytes[i]);
            positions.push(i);
            first_retained = false;
            i += 1;
        }
    }

    for run in runs {
        let left_at = clean
            .iter()
            .enumerate()
            .filter(|(_, _)| true)
            .take_while(|(k, _)| positions[*k] < run.start)
            .last()
            .map(|(k, _)| k);
        let right_at = positions.iter().position(|&p| p >= run.end);
        let left_boundary = left_at.is_none()
            || left_at.map(|k| clean[k] == b'@').unwrap_or(false);
        let right_boundary = right_at.is_none()
            || right_at.map(|k| clean[k] == b'@').unwrap_or(false);
        if !(left_boundary || right_boundary) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut j = 1usize;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                quote_end = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: positions[0],
            });
        }
    }

    let protected_end = quote_end.unwrap_or(0);
    let mut ats = Vec::new();
    for idx in 0..clean.len() {
        if clean[idx] == b'@' && !(quote_end.is_some() && idx <= protected_end) {
            ats.push(idx);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: positions[ats[1]],
        });
    }
    let at_idx = ats[0];
    let sep_pos = positions[at_idx];
    if at_idx == 0 {
        return Err(EmailError::EmptyLocal { at: sep_pos });
    }

    let local_bytes = &clean[..at_idx];
    let local_positions = &positions[..at_idx];
    let (local, quoted) = if local_bytes[0] == b'"' {
        let end = quote_end.expect("quoted prefix must have a closing quote");
        if end + 1 != at_idx {
            return Err(EmailError::TrailingLocal {
                at: positions[end + 1],
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1usize;
        while j < end {
            if local_bytes[j] == b'\\' {
                if j + 1 >= end {
                    return Err(EmailError::BadEscape {
                        at: local_positions[j],
                    });
                }
                let escaped = local_bytes[j + 1];
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: local_positions[j],
                    });
                }
                decoded.push(escaped);
                origins.push(local_positions[j]);
                j += 2;
            } else {
                let b = local_bytes[j];
                if !printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: local_positions[j],
                    });
                }
                decoded.push(b);
                origins.push(local_positions[j]);
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: positions[end],
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_positions[64],
            });
        }
        let allowed_special = b"!#$%&'*+-/=?^_`{|}~";
        for j in 0..local_bytes.len() {
            let b = local_bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || local_bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: local_positions[j],
                    });
                }
            } else if !b.is_ascii_alphanumeric() && !allowed_special.contains(&b) {
                return Err(EmailError::LocalChar {
                    at: local_positions[j],
                });
            }
        }
        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start = at_idx + 1;
    let domain_start_pos = positions.get(domain_start).copied().unwrap_or(input.len());
    if domain_start == clean.len() {
        return Err(EmailError::EmptyDomain {
            at: domain_start_pos,
        });
    }
    let mut domain_end = clean.len();
    let mut removed_root_pos = None;
    if clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        removed_root_pos = Some(positions[domain_end]);
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: domain_start_pos,
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start + 253],
        });
    }

    let mut labels = Vec::<(usize, usize)>::new();
    let mut label_start = domain_start;
    for idx in domain_start..domain_end {
        if clean[idx] == b'.' {
            labels.push((label_start, idx));
            label_start = idx + 1;
        }
    }
    labels.push((label_start, domain_end));

    let mut last_label = (0usize, 0usize);
    for (n, &(start, end)) in labels.iter().enumerate() {
        if start == end {
            let pos = if n + 1 == labels.len() {
                removed_root_pos.unwrap_or_else(|| {
                    positions.get(start).copied().unwrap_or(input.len())
                })
            } else {
                positions.get(start).copied().unwrap_or(input.len())
            };
            return Err(EmailError::EmptyLabel { at: pos });
        }
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: positions[start + 63],
            });
        }
        if clean[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[start],
            });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[end - 1],
            });
        }
        for idx in start..end {
            let b = clean[idx];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: positions[idx] });
            }
        }
        if n + 1 == labels.len() {
            last_label = (start, end);
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: domain_start_pos,
        });
    }
    let (last_start, last_end) = last_label;
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: positions[last_start],
        });
    }

    let mut domain = clean[domain_start..domain_end].to_vec();
    domain.make_ascii_lowercase();
    Ok(Address {
        local,
        domain: String::from_utf8(domain).unwrap(),
        quoted,
    })
}
