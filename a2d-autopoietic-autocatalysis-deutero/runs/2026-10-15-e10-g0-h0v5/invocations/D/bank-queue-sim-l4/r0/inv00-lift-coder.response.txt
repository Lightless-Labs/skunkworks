#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let other_half = denominator - r;
    Some(if r > other_half || (r == other_half && q & 1 == 1) {
        q + 1
    } else {
        q
    })
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    queue_since: u64,
    waited: u64,
    paused: bool,
    pause_time: u64,
    ever_started: bool,
}

#[derive(Clone)]
struct Active {
    idx: usize,
    work_at: u64,
    finish: u64,
    dispatch_at: u64,
}

fn event(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event { time, customer: id, kind });
}

fn better(a: &Job, b: &Job) -> bool {
    if a.customer.priority != b.customer.priority {
        return a.customer.priority < b.customer.priority;
    }
    if a.paused != b.paused {
        return a.paused;
    }
    if a.paused {
        if a.pause_time != b.pause_time {
            return a.pause_time < b.pause_time;
        }
        a.customer.id < b.customer.id
    } else if a.customer.arrival != b.customer.arrival {
        a.customer.arrival < b.customer.arrival
    } else {
        a.customer.id > b.customer.id
    }
}

fn best_waiter(waiting: &[usize], jobs: &[Job]) -> Option<usize> {
    waiting.iter().copied().min_by(|&a, &b| {
        if better(&jobs[a], &jobs[b]) {
            std::cmp::Ordering::Less
        } else if better(&jobs[b], &jobs[a]) {
            std::cmp::Ordering::Greater
        } else {
            std::cmp::Ordering::Equal
        }
    })
}

fn fail_time(id: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow { customer: id, time, stage }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut seen = Vec::new();
    let mut deadlines = Vec::with_capacity(customers.len());
    for (i, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index: i });
        }
        if seen.iter().any(|&id| id == c.id) {
            return Err(Error::DuplicateId { index: i, id: c.id });
        }
        seen.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index: i, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow { index: i, id: c.id })?),
            None => None,
        };
        deadlines.push(deadline);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a].arrival.cmp(&customers[b].arrival)
            .then(customers[a].priority.cmp(&customers[b].priority))
            .then(customers[b].id.cmp(&customers[a].id))
    });

    let mut jobs: Vec<Job> = customers.iter().enumerate().map(|(i, c)| Job {
        customer: c.clone(),
        deadline: deadlines[i],
        remaining: c.service,
        queue_since: c.arrival,
        waited: 0,
        paused: false,
        pause_time: 0,
        ever_started: false,
    }).collect();

    let (mut events, mut waiting) = (Vec::new(), Vec::new());
    let (mut active, mut now, mut next_arrival) = (None::<Active>, 0u64, 0usize);
    let (mut last_dispatched, mut any_dispatch) = (None::<u32>, false);
    let (mut accepted, mut balked, mut reneged, mut completed, mut started, mut preemptions) =
        (0usize, 0usize, 0usize, 0usize, 0usize, 0usize);
    let (mut busy_ticks, mut wait_ticks, mut turnaround_ticks) = (0u128, 0u128, 0u128);
    let mut completion_order = Vec::new();
    let mut last_event_time = 0u64;

    while next_arrival < order.len() || active.is_some() || !waiting.is_empty() {
        if active.is_none() && waiting.is_empty() && next_arrival < order.len() {
            now = now.max(customers[order[next_arrival]].arrival);
        }

        // Phase A: a completion at this tick wins over every other event.
        if let Some(a) = active.as_ref() {
            if a.finish == now {
                let idx = a.idx;
                let id = jobs[idx].customer.id;
                event(&mut events, now, id, Kind::Finish);
                last_event_time = now;
                completed += 1;
                turnaround_ticks += (now - jobs[idx].customer.arrival) as u128;
                completion_order.push(id);
                active = None;
            }
        }

        // Phase B: expire those who were waiting at the start of the tick.
        let mut expired: Vec<usize> = waiting.iter().copied()
            .filter(|&i| jobs[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| jobs[a].deadline.cmp(&jobs[b].deadline).then(jobs[a].customer.id.cmp(&jobs[b].customer.id)));
        for i in expired {
            if let Some(pos) = waiting.iter().position(|&x| x == i) {
                waiting.remove(pos);
                let segment = now - jobs[i].queue_since;
                jobs[i].waited += segment;
                wait_ticks += segment as u128;
                event(&mut events, now, jobs[i].customer.id, Kind::Renege { waited: jobs[i].waited });
                last_event_time = now;
                reneged += 1;
            }
        }

        // Phase C: process the complete arrival cohort.
        while next_arrival < order.len() && customers[order[next_arrival]].arrival == now {
            let i = order[next_arrival];
            next_arrival += 1;
            let id = jobs[i].customer.id;
            event(&mut events, now, id, Kind::Arrive);
            last_event_time = now;
            let limit = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= limit {
                event(&mut events, now, id, Kind::Balk);
                last_event_time = now;
                balked += 1;
            } else {
                accepted += 1;
                jobs[i].queue_since = now;
                waiting.push(i);
            }
        }

        // Phase D: preempt only after the nonpreemptible switching interval.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(w) = best_waiter(&waiting, &jobs) {
                    if jobs[waiting.iter().copied().find(|&x| x == w).unwrap()].customer.priority
                        < jobs[a.idx].customer.priority
                    {
                        let a = active.take().unwrap();
                        let id = jobs[a.idx].customer.id;
                        let remaining = a.finish - now;
                        event(&mut events, now, id, Kind::Preempt { remaining, by: jobs[w].customer.id });
                        last_event_time = now;
                        preemptions += 1;
                        busy_ticks += (now - a.dispatch_at) as u128;
                        jobs[a.idx].remaining = remaining;
                        jobs[a.idx].queue_since = now;
                        jobs[a.idx].paused = true;
                        jobs[a.idx].pause_time = now;
                        if let Some(budget) = jobs[a.idx].customer.patience {
                            let spent = jobs[a.idx].waited;
                            let remaining_budget = budget.saturating_sub(spent);
                            jobs[a.idx].deadline = Some(now.checked_add(remaining_budget).ok_or_else(|| fail_time(id, now, OverflowStage::Patience))?);
                        }
                        waiting.push(a.idx);
                    }
                }
            }
        }

        // Phase E: dispatch repeatedly, allowing inline zero-duration finishes.
        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&waiting, &jobs) else { break };
            let pos = waiting.iter().position(|&x| x == i).unwrap();
            waiting.remove(pos);

            let segment = now - jobs[i].queue_since;
            jobs[i].waited += segment;
            wait_ticks += segment as u128;
            if let Some(budget) = jobs[i].customer.patience {
                let remaining_budget = budget.saturating_sub(jobs[i].waited);
                jobs[i].deadline = Some(now.checked_add(remaining_budget).ok_or_else(|| fail_time(jobs[i].customer.id, now, OverflowStage::Patience))?);
            }

            let id = jobs[i].customer.id;
            let cost = if !any_dispatch || last_dispatched == Some(id) { 0 } else { config.switch_cost };
            let work_at = now.checked_add(cost).ok_or_else(|| fail_time(id, now, OverflowStage::WorkAt))?;
            let finish = work_at.checked_add(jobs[i].remaining).ok_or_else(|| fail_time(id, now, OverflowStage::Finish))?;
            let was_started = jobs[i].ever_started;
            let kind = if was_started {
                Kind::Resume { remaining: jobs[i].remaining, work_at, finish }
            } else {
                started += 1;
                Kind::Start { remaining: jobs[i].remaining, work_at, finish }
            };
            jobs[i].ever_started = true;
            any_dispatch = true;
            last_dispatched = Some(id);
            event(&mut events, now, id, kind);
            last_event_time = now;
            active = Some(Active { idx: i, work_at, finish, dispatch_at: now });

            if finish == now {
                event(&mut events, now, id, Kind::Finish);
                last_event_time = now;
                completed += 1;
                turnaround_ticks += (now - jobs[i].customer.arrival) as u128;
                completion_order.push(id);
                active = None;
                continue;
            }
            break;
        }

        // Phase F: expire newly queued waiters, after dispatches.
        let mut expired: Vec<usize> = waiting.iter().copied()
            .filter(|&i| jobs[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| jobs[a].deadline.cmp(&jobs[b].deadline).then(jobs[a].customer.id.cmp(&jobs[b].customer.id)));
        for i in expired {
            if let Some(pos) = waiting.iter().position(|&x| x == i) {
                waiting.remove(pos);
                let segment = now - jobs[i].queue_since;
                jobs[i].waited += segment;
                wait_ticks += segment as u128;
                event(&mut events, now, jobs[i].customer.id, Kind::Renege { waited: jobs[i].waited });
                last_event_time = now;
                reneged += 1;
            }
        }

        if next_arrival >= order.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time = None::<u64>;
        let mut consider = |t: u64| {
            if t > now && next_time.map_or(true, |n| t < n) {
                next_time = Some(t);
            }
        };
        if next_arrival < order.len() {
            consider(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = active.as_ref() {
            consider(a.finish);
            consider(a.work_at);
        }
        for &i in &waiting {
            if let Some(d) = jobs[i].deadline {
                consider(d);
            }
        }
        if let Some(t) = next_time {
            now = t;
        } else {
            // At u64::MAX, all remaining work must finish or expire at this tick.
            if now == u64::MAX {
                if active.is_some() {
                    return Err(fail_time(jobs[active.as_ref().unwrap().idx].customer.id, now, OverflowStage::Finish));
                }
                break;
            }
        }
    }

    let mean_wait_milli = if accepted == 0 { 0 } else { rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap() };
    let mean_turnaround_milli = if completed == 0 { 0 } else { rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap() };
    let utilization_bps = if last_event_time == 0 {
        if busy_ticks == 0 { 0 } else { 10000 }
    } else {
        rounded_ratio(busy_ticks * 10000, last_event_time as u128).unwrap().min(u32::MAX as u128) as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted, balked, reneged, completed, started, preemptions,
            elapsed: last_event_time,
            busy_ticks, wait_ticks, turnaround_ticks,
            mean_wait_milli, mean_turnaround_milli, utilization_bps,
            completion_order,
        },
    })
}
