#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn append_text(word: &mut Word, text: &str, wildcards_active: bool) {
    for c in text.chars() {
        if wildcards_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn next_char(input: &str, pos: usize) -> Option<(char, usize)> {
    let c = input.get(pos..)?.chars().next()?;
    Some((c, pos + c.len_utf8()))
}

fn decode_default(input: &str, start: usize, end: usize) -> String {
    let mut out = String::new();
    let mut pos = start;
    while pos < end {
        let (c, next) = next_char(input, pos).unwrap();
        if c == '\\' && next < end {
            let (escaped, after) = next_char(input, next).unwrap();
            out.push(escaped);
            pos = after;
        } else {
            out.push(c);
            pos = next;
        }
    }
    out
}

fn parse_braced(
    input: &str,
    dollar_pos: usize,
    after_open: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let mut close = None;
    let mut pos = after_open;

    while pos < input.len() {
        let (c, next) = next_char(input, pos).unwrap();
        if c == '\\' {
            if next < input.len() {
                let (_, after) = next_char(input, next).unwrap();
                pos = after;
            } else {
                pos = next;
            }
        } else if c == '}' {
            close = Some(pos);
            break;
        } else {
            pos = next;
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar_pos })?;
    let contents = &input[after_open..close];

    let mut name_end = 0;
    let mut chars = contents.char_indices();
    let Some((first_offset, first)) = chars.next() else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };
    if first_offset != 0 || !is_name_start(first) {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }
    name_end = first.len_utf8();

    for (offset, c) in chars {
        if is_name_continue(c) {
            name_end = offset + c.len_utf8();
        } else {
            break;
        }
    }

    let name = &contents[..name_end];
    let suffix = &contents[name_end..];

    let value = if suffix.is_empty() {
        env_value(env, name).unwrap_or("").to_string()
    } else if let Some(default) = suffix.strip_prefix(":-") {
        let selected = env_value(env, name).filter(|value| !value.is_empty());
        match selected {
            Some(value) => value.to_string(),
            None => {
                let default_start = after_open + name_end + 2;
                let _ = default;
                decode_default(input, default_start, close)
            }
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    Ok((value, close + 1))
}

fn expand_unbraced(
    input: &str,
    dollar_pos: usize,
    after_dollar: usize,
    env: &[(&str, &str)],
) -> Option<(String, usize)> {
    let (first, _) = next_char(input, after_dollar)?;
    if !is_name_start(first) {
        return None;
    }

    let mut end = after_dollar;
    while let Some((c, after)) = next_char(input, end) {
        if is_name_continue(c) {
            end = after;
        } else {
            break;
        }
    }

    let name = &input[after_dollar..end];
    let _ = dollar_pos;
    Some((env_value(env, name).unwrap_or("").to_string(), end))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0;
    let mut pos = 0;

    while pos < input.len() {
        let (c, next) = next_char(input, pos).unwrap();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.text.push(c);
                }
                pos = next;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    pos = next;
                } else if c == '\\' {
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let (escaped, after) = next_char(input, next).unwrap();
                    if matches!(escaped, '"' | '\\' | '$') {
                        append_text(&mut word, &escaped.to_string(), false);
                        pos = after;
                    } else if escaped == '\n' {
                        pos = after;
                    } else {
                        word.text.push('\\');
                        pos = next;
                    }
                } else if c == '$' {
                    if let Some((following, after_dollar)) = next_char(input, next) {
                        if following == '(' {
                            return Err(SplitError::CommandSubstitution { pos });
                        }
                        if following == '{' {
                            let (value, after) = parse_braced(input, pos, after_dollar, env)?;
                            active = true;
                            append_text(&mut word, &value, false);
                            pos = after;
                            continue;
                        }
                        if is_name_start(following) {
                            let (value, end) =
                                expand_unbraced(input, pos, next, env).unwrap();
                            active = true;
                            append_text(&mut word, &value, false);
                            pos = end;
                            continue;
                        }
                    }
                    word.text.push('$');
                    pos = next;
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else {
                    word.text.push(c);
                    pos = next;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    pos = next;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single;
                    quote_start = pos;
                    pos = next;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double;
                    quote_start = pos;
                    pos = next;
                } else if c == '\\' {
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let (escaped, after) = next_char(input, next).unwrap();
                    if escaped == '\n' {
                        pos = after;
                    } else {
                        active = true;
                        word.text.push(escaped);
                        pos = after;
                    }
                } else if c == '$' {
                    if let Some((following, after_dollar)) = next_char(input, next) {
                        if following == '(' {
                            return Err(SplitError::CommandSubstitution { pos });
                        }
                        if following == '{' {
                            let (value, after) = parse_braced(input, pos, after_dollar, env)?;
                            active = true;
                            append_text(&mut word, &value, true);
                            pos = after;
                            continue;
                        }
                        if is_name_start(following) {
                            let (value, end) =
                                expand_unbraced(input, pos, next, env).unwrap();
                            active = true;
                            append_text(&mut word, &value, true);
                            pos = end;
                            continue;
                        }
                    }
                    active = true;
                    word.text.push('$');
                    pos = next;
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else {
                    active = true;
                    append_text(&mut word, &c.to_string(), true);
                    pos = next;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }

    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{word}'"))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }

    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

fn fallback_candidate(word: &str) -> String {
    let mut out = String::from("'");
    for c in word.chars() {
        if c == '\'' {
            out.push('\'');
            out.push('\\');
            out.push('\'');
            out.push('\'');
        } else {
            out.push(c);
        }
    }
    out.push('\'');
    out
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let candidates = [
            bare_candidate(word),
            single_candidate(word),
            double_candidate(word),
        ];

        let best = candidates
            .into_iter()
            .flatten()
            .min_by_key(String::len)
            .unwrap_or_else(|| fallback_candidate(word));

        encoded.push(best);
    }

    Ok(encoded.join(" "))
}
