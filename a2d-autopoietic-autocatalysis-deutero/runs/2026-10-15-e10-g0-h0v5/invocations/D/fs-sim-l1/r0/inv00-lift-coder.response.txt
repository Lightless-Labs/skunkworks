```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn parse_path(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut start = 1;
        let mut components = Vec::new();

        for end in 1..=bytes.len() {
            if end == bytes.len() || bytes[end] == b'/' {
                let component = &path[start..end];
                if component.is_empty() {
                    return Err(Error::InvalidPath(if end == bytes.len() {
                        bytes.len() - 1
                    } else {
                        end
                    }));
                }
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                components.push(component);
                start = end + 1;
            }
        }

        Ok(components)
    }

    fn prefix(components: &[&str], count: usize) -> String {
        if count == 0 {
            return "/".to_owned();
        }
        let mut result = String::new();
        for component in components.iter().take(count) {
            result.push('/');
            result.push_str(component);
        }
        result
    }

    fn lookup<'a>(&'a self, components: &[&str]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        for (i, component) in components.iter().enumerate() {
            match current {
                Entry::Dir(children) => match children.get(*component) {
                    Some(entry) => current = entry,
                    None => return Err(Error::Missing(Self::prefix(components, i + 1))),
                },
                Entry::File(_) => return Err(Error::NotDir(Self::prefix(components, i)),
            }
        }
        Ok(current)
    }

    fn parent_mut<'a>(
        &'a mut self,
        components: &[&str],
    ) -> Result<&'a mut BTreeMap<String, Entry>, Error> {
        let parent_components = &components[..components.len() - 1];
        let mut current = &mut self.root;

        for (i, component) in parent_components.iter().enumerate() {
            match current {
                Entry::Dir(children) => match children.get_mut(*component) {
                    Some(entry) => current = entry,
                    None => return Err(Error::Missing(Self::prefix(components, i + 1))),
                },
                Entry::File(_) => return Err(Error::NotDir(Self::prefix(components, i)),
            }
        }

        match current {
            Entry::Dir(children) => Ok(children),
            Entry::File(_) => Err(Error::NotDir(Self::prefix(
                components,
                components.len() - 1,
            ))),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::parse_path(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let full_path = Self::prefix(&components, components.len());
        let parent = self.parent_mut(&components)?;
        if parent.contains_key(*components.last().unwrap()) {
            return Err(Error::Exists(full_path));
        }

        parent.insert(
            components.last().unwrap().to_string(),
            Entry::Dir(BTreeMap::new()),
        );
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::parse_path(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let full_path = Self::prefix(&components, components.len());
        let parent = self.parent_mut(&components)?;
        let name = components.last().unwrap();

        match parent.get_mut(*name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(full_path)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                parent.insert(name.to_string(), Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::parse_path(path)?;
        match self.lookup(&components)? {
            Entry::File(contents) => Ok(contents.clone()),
            Entry::Dir(_) => Err(Error::IsDir(Self::prefix(&components, components.len()))),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::parse_path(path)?;
        match self.lookup(&components)? {
            Entry::File(_) => Err(Error::NotDir(Self::prefix(&components, components.len()))),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
