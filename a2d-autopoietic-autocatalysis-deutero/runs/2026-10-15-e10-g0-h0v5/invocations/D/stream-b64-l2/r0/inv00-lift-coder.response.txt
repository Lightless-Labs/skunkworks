#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
}

const ALPHABET: &[u8; 64] =
    b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";

pub fn encode(input: &[u8]) -> String {
    let mut encoder = Encoder::new();
    let mut output = encoder.push(input).expect("new encoder is open");
    output.push_str(&encoder.finish().expect("new encoder is open"));
    output
}

pub fn decode(input: &[u8]) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new();
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    tail: [u8; 2],
    tail_len: usize,
    finished: bool,
}

impl Encoder {
    pub fn new() -> Self {
        Self {
            tail: [0; 2],
            tail_len: 0,
            finished: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = String::new();
        let mut pos = 0;

        if self.tail_len != 0 {
            while self.tail_len < 3 && pos < input.len() {
                if self.tail_len < 2 {
                    self.tail[self.tail_len] = input[pos];
                }
                self.tail_len += 1;
                pos += 1;
            }
            if self.tail_len == 3 {
                encode_triplet(
                    &mut output,
                    self.tail[0],
                    self.tail[1],
                    input[pos - 1],
                );
                self.tail_len = 0;
            } else {
                return Ok(output);
            }
        }

        while pos + 3 <= input.len() {
            encode_triplet(&mut output, input[pos], input[pos + 1], input[pos + 2]);
            pos += 3;
        }

        self.tail_len = input.len() - pos;
        if self.tail_len > 0 {
            self.tail[0] = input[pos];
        }
        if self.tail_len > 1 {
            self.tail[1] = input[pos + 1];
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;

        let mut output = String::new();
        match self.tail_len {
            1 => {
                output.push(ALPHABET[(self.tail[0] >> 2) as usize] as char);
                output.push(ALPHABET[((self.tail[0] & 0x03) << 4) as usize] as char);
                output.push_str("==");
            }
            2 => {
                output.push(ALPHABET[(self.tail[0] >> 2) as usize] as char);
                output.push(
                    ALPHABET[(((self.tail[0] & 0x03) << 4) | (self.tail[1] >> 4)) as usize]
                        as char,
                );
                output.push(ALPHABET[((self.tail[1] & 0x0f) << 2) as usize] as char);
                output.push('=');
            }
            _ => {}
        }
        self.tail_len = 0;
        Ok(output)
    }
}

fn encode_triplet(output: &mut String, a: u8, b: u8, c: u8) {
    output.push(ALPHABET[(a >> 2) as usize] as char);
    output.push(ALPHABET[(((a & 0x03) << 4) | (b >> 4)) as usize] as char);
    output.push(ALPHABET[(((b & 0x0f) << 2) | (c >> 6)) as usize] as char);
    output.push(ALPHABET[(c & 0x3f) as usize] as char);
}

#[derive(Clone, Copy)]
enum DecodeState {
    Open,
    PendingSecondPadding { offset: usize },
    Terminal,
}

pub struct Decoder {
    group: [u8; 4],
    offsets: [usize; 4],
    len: usize,
    state: DecodeState,
    next_offset: usize,
    finished: bool,
    error: Option<Error>,
}

impl Decoder {
    pub fn new() -> Self {
        Self {
            group: [0; 4],
            offsets: [0; 4],
            len: 0,
            state: DecodeState::Open,
            next_offset: 0,
            finished: false,
            error: None,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();

        for &byte in input {
            let offset = self.next_offset;
            self.next_offset += 1;

            let error = match self.state {
                DecodeState::Terminal => Some(Error::BadPadding { offset }),
                DecodeState::PendingSecondPadding { offset: padding_offset } => {
                    if byte == b'=' {
                        self.state = DecodeState::Terminal;
                        let mut values = [0u8; 4];
                        values[0] = alphabet_value(self.group[0]).unwrap();
                        values[1] = alphabet_value(self.group[1]).unwrap();
                        output.push((values[0] << 2) | (values[1] >> 4));
                        self.len = 0;
                        self.group = [0; 4];
                        self.offsets = [0; 4];
                        None
                    } else {
                        Some(Error::BadPadding { offset })
                    }
                }
                DecodeState::Open => {
                    let position = self.len;
                    if position < 2 && byte == b'=' {
                        Some(Error::BadPadding { offset })
                    } else if byte == b'=' && position == 2 {
                        self.group[position] = byte;
                        self.offsets[position] = offset;
                        self.len = 3;
                        self.state = DecodeState::PendingSecondPadding { offset };
                        None
                    } else if byte == b'=' && position == 3 {
                        let values = [
                            alphabet_value(self.group[0]).unwrap(),
                            alphabet_value(self.group[1]).unwrap(),
                            alphabet_value(self.group[2]).unwrap(),
                        ];
                        output.push((values[0] << 2) | (values[1] >> 4));
                        output.push((values[1] << 4) | (values[2] >> 2));
                        self.state = DecodeState::Terminal;
                        self.len = 0;
                        self.group = [0; 4];
                        self.offsets = [0; 4];
                        None
                    } else if let Some(value) = alphabet_value(byte) {
                        self.group[position] = byte;
                        self.offsets[position] = offset;
                        self.len += 1;
                        if self.len == 4 {
                            let a = alphabet_value(self.group[0]).unwrap();
                            let b = alphabet_value(self.group[1]).unwrap();
                            let c = alphabet_value(self.group[2]).unwrap();
                            let d = alphabet_value(self.group[3]).unwrap();
                            output.push((a << 2) | (b >> 4));
                            output.push((b << 4) | (c >> 2));
                            output.push((c << 6) | d);
                            self.len = 0;
                            self.group = [0; 4];
                            self.offsets = [0; 4];
                        }
                        None
                    } else {
                        Some(Error::InvalidByte { offset, byte })
                    }
                }
            };

            if let Some(error) = error {
                self.error = Some(error.clone());
                return Err(error);
            }
        }

        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let result = match self.state {
            DecodeState::PendingSecondPadding { offset } => {
                Err(Error::BadPadding { offset })
            }
            _ if self.len != 0 => Err(Error::Truncated {
                offset: self.offsets[0],
            }),
            _ => Ok(Vec::new()),
        };

        match result {
            Ok(output) => {
                self.finished = true;
                Ok(output)
            }
            Err(error) => {
                self.error = Some(error.clone());
                Err(error)
            }
        }
    }
}

fn alphabet_value(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' => Some(62),
        b'/' => Some(63),
        _ => None,
    }
}
