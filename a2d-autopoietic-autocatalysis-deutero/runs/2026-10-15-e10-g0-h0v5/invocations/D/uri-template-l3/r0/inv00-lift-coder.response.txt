#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone)]
struct Expression<'a> {
    operator: Option<char>,
    variables: Vec<&'a str>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression<'a>(
    content: &'a str,
    content_pos: usize,
    close_pos: usize,
) -> Result<Expression<'a>, Error> {
    let mut operator = None;
    let mut body = content;
    let mut body_pos = content_pos;

    if let Some(first) = content.as_bytes().first().copied() {
        if b"+#./;?&".contains(&first) {
            operator = Some(first as char);
            body = &content[1..];
            body_pos += 1;
        } else if first == b'!' || first == b'=' || first == b'<' || first == b'>' || first == b'@'
            || first == b'|' || first == b'$' || first == b'-' || first == b'%' || first == b'^'
            || first == b'\\' || first == b'(' || first == b')'
        {
            return Err(Error::UnsupportedOperator { pos: content_pos });
        }
    }

    if body.is_empty() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut offset = 0;

    for specifier in body.split(',') {
        let specifier_pos = body_pos + offset;

        if specifier.is_empty() {
            return Err(Error::EmptyVariable { pos: specifier_pos });
        }

        let bytes = specifier.as_bytes();
        if !is_name_start(bytes[0]) {
            return Err(Error::InvalidName { pos: specifier_pos });
        }

        for (index, byte) in bytes.iter().copied().enumerate().skip(1) {
            if !is_name_byte(byte) {
                return Err(Error::InvalidName {
                    pos: specifier_pos + index,
                });
            }
        }

        if seen.iter().any(|name| *name == specifier) {
            return Err(Error::DuplicateVariable { pos: specifier_pos });
        }

        seen.push(specifier);
        variables.push(specifier);
        offset += specifier.len() + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn validate(template: &str) -> Result<Vec<Expression<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let open_pos = pos;
                let mut close_pos = pos + 1;
                while close_pos < bytes.len() && bytes[close_pos] != b'}' {
                    close_pos += 1;
                }

                if close_pos == bytes.len() {
                    return Err(Error::Unclosed { pos: open_pos });
                }

                if let Some(inner_offset) = bytes[open_pos + 1..close_pos]
                    .iter()
                    .position(|byte| *byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open_pos + 1 + inner_offset,
                    });
                }

                let content = &template[open_pos + 1..close_pos];
                expressions.push(parse_expression(
                    content,
                    open_pos + 1,
                    close_pos,
                )?);
                pos = close_pos + 1;
            }
            _ => {
                let ch = template[pos..].chars().next().unwrap();
                pos += ch.len_utf8();
            }
        }
    }

    Ok(expressions)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn upper_hex(byte: u8) -> u8 {
    match byte {
        b'a'..=b'f' => byte - b'a' + b'A',
        _ => byte,
    }
}

fn encode(value: &str, reserved_allowed: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut pos = 0;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == b'%'
            && pos + 2 < bytes.len()
            && hex_value(bytes[pos + 1]).is_some()
            && hex_value(bytes[pos + 2]).is_some()
        {
            output.push('%');
            output.push(upper_hex(bytes[pos + 1]) as char);
            output.push(upper_hex(bytes[pos + 2]) as char);
            pos += 3;
            continue;
        }

        if is_unreserved(byte) || (reserved_allowed && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        pos += 1;
    }

    output
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let reserved_allowed = matches!(operator, Some('+') | Some('#'));
    let mut fields = Vec::new();

    for name in &expression.variables {
        if let Some(value) = lookup(name, vars) {
            let encoded = encode(value, reserved_allowed);
            let field = match operator {
                Some(';') if encoded.is_empty() => (*name).to_owned(),
                Some(';') | Some('?') | Some('&') => format!("{name}={encoded}"),
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match operator {
        Some('#') => ("#", ","),
        Some('.') => (".", "."),
        Some('/') => ("/", "/"),
        Some(';') => (";", ";"),
        Some('?') => ("?", "&"),
        Some('&') => ("&", "&"),
        _ => ("", ","),
    };

    let mut output = String::from(prefix);
    output.push_str(&fields.join(separator));
    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut output = String::new();
    let mut pos = 0;
    let mut expression_index = 0;
    let bytes = template.as_bytes();

    while pos < bytes.len() {
        if bytes[pos] == b'{' {
            let mut close_pos = pos + 1;
            while close_pos < bytes.len() && bytes[close_pos] != b'}' {
                close_pos += 1;
            }

            output.push_str(&expand_expression(
                &expressions[expression_index],
                vars,
            ));
            expression_index += 1;
            pos = close_pos + 1;
        } else {
            let ch = template[pos..].chars().next().unwrap();
            output.push(ch);
            pos += ch.len_utf8();
        }
    }

    Ok(output)
}
