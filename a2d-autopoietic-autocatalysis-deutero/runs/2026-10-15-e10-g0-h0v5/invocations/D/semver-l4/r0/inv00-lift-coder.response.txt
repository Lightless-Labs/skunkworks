use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}
fn valid_ident_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}
fn split_identifiers(s: &str, base: usize, prerelease: bool) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut start = 0;
    loop {
        let end = bytes[start..].iter().position(|b| *b == b'.').map(|n| start + n).unwrap_or(bytes.len());
        if end == start {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
        for i in start..end {
            if !valid_ident_byte(bytes[i]) {
                return Err(VersionError::InvalidChar { at: base + i });
            }
        }
        if prerelease && end - start > 1 && bytes[start] == b'0' && bytes[start..end].iter().all(|b| is_digit(*b)) {
            return Err(VersionError::LeadingZero { at: base + start });
        }
        out.push(s[start..end].to_string());
        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: base + start });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut fields = [0u64; 3];
    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && is_digit(bytes[pos]) {
            pos += 1;
        }
        if pos == start {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        fields[field] = s[start..pos].parse::<u64>().map_err(|_| VersionError::Overflow { at: start })?;
        if field < 2 {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        } else if pos < bytes.len() && !matches!(bytes[pos], b'-' | b'+') {
            return Err(VersionError::InvalidChar { at: pos });
        }
    }
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() && bytes[pos] == b'-' {
        let start = pos + 1;
        let end = bytes[start..].iter().position(|b| *b == b'+').map(|n| start + n).unwrap_or(bytes.len());
        pre = split_identifiers(&s[start..end], start, true)?;
        pos = end;
    }
    if pos < bytes.len() && bytes[pos] == b'+' {
        let start = pos + 1;
        build = split_identifiers(&s[start..], start, false)?;
        pos = bytes.len();
    }
    if pos != bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version { major: fields[0], minor: fields[1], patch: fields[2], pre, build })
}
fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}
fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}
fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b.iter()) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let c = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.len().cmp(&b.len())
}
pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core(a).cmp(&core(b)).then_with(|| cmp_pre(&a.pre, &b.pre))
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}
#[derive(Clone)]
struct Tok {
    s: String,
    at: usize,
}
fn tokenize(s: &str) -> Vec<Tok> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && ws(b[i]) { i += 1; }
        if i == b.len() { break; }
        let start = i;
        while i < b.len() && !ws(b[i]) { i += 1; }
        out.push(Tok { s: s[start..i].to_string(), at: start });
    }
    out
}
fn mkver(a: u64, b: u64, c: u64, pre: Vec<String>, build: Vec<String>) -> Version {
    Version { major: a, minor: b, patch: c, pre, build }
}
fn zero_pre() -> Vec<String> { vec!["0".to_string()] }
fn parsed_full(tok: &Tok, offset: usize, operand: &str) -> Result<Version, RangeError> {
    parse_version(operand).map_err(|error| RangeError::Version { at: offset, error })
}
#[derive(Clone)]
enum Partial {
    Full(Version),
    Part { nums: Vec<u64>, k: usize },
}
fn partial(tok: &Tok, offset: usize, s: &str) -> Result<Partial, RangeError> {
    if s.contains('-') || s.contains('+') {
        return parsed_full(tok, offset, s).map(Partial::Full);
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() >= 3 && !fields.iter().any(|x| matches!(*x, "*" | "x" | "X")) {
        return parsed_full(tok, offset, s).map(Partial::Full);
    }
    if fields.len() > 3 || fields.iter().any(|x| x.is_empty()) {
        return Err(RangeError::Syntax { at: tok.at });
    }
    let mut nums = Vec::new();
    let mut wildcard = false;
    for f in &fields {
        if matches!(*f, "*" | "x" | "X") {
            wildcard = true;
        } else {
            if wildcard || !f.bytes().all(is_digit) || (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at: tok.at });
            }
            let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: tok.at })?;
            nums.push(n);
        }
    }
    Ok(Partial::Part { k: nums.len(), nums })
}
fn padded(nums: &[u64]) -> Version {
    mkver(*nums.get(0).unwrap_or(&0), *nums.get(1).unwrap_or(&0), *nums.get(2).unwrap_or(&0), vec![], vec![])
}
fn overflow(tok: &Tok) -> RangeError { RangeError::Syntax { at: tok.at } }
fn bump(v: &Version, field: usize, tok: &Tok) -> Result<Version, RangeError> {
    let mut a = v.major;
    let mut b = v.minor;
    let mut c = v.patch;
    match field {
        0 => { a = a.checked_add(1).ok_or_else(|| overflow(tok))?; b = 0; c = 0; }
        1 => { b = b.checked_add(1).ok_or_else(|| overflow(tok))?; c = 0; }
        _ => { c = c.checked_add(1).ok_or_else(|| overflow(tok))?; }
    }
    Ok(mkver(a,b,c,zero_pre(),vec![]))
}
fn add(arm: &mut Arm, op: &str, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() { arm.anchors.push(core(&v)); }
    arm.cmps.push(Cmp { op: op.to_string(), v, anchor });
}
fn parse_operand(tok: &Tok, arm: &mut Arm) -> Result<(), RangeError> {
    let s = tok.s.as_str();
    let (prefix, body) = if s.starts_with('^') || s.starts_with('~') {
        (&s[..1], &s[1..])
    } else { ("", s) };
    let off = tok.at + prefix.len();
    let p = partial(tok, off, body)?;
    match (prefix, p) {
        ("", Partial::Full(v)) => add(arm, "=", v, true),
        ("", Partial::Part { nums, k: 0 }) => {}
        ("", Partial::Part { nums, k }) => {
            let lo = padded(&nums);
            add(arm, ">=", lo.clone(), true);
            if k < 3 {
                let upper = if k == 1 { bump(&lo, 0, tok)? } else { bump(&lo, 1, tok)? };
                add(arm, "<", upper, false);
            }
        }
        ("^", Partial::Part { nums, k: 0 }) | ("~", Partial::Part { nums, k: 0 }) => return Err(RangeError::Syntax { at: tok.at }),
        ("^", Partial::Part { nums, k }) => {
            let lo = padded(&nums);
            add(arm, ">=", lo.clone(), true);
            let f = if lo.major > 0 || k == 1 { 0 } else if lo.minor > 0 || k == 2 { 1 } else { 2 };
            add(arm, "<", bump(&lo, f, tok)?, false);
        }
        ("~", Partial::Part { nums, k }) => {
            let lo = padded(&nums);
            add(arm, ">=", lo.clone(), true);
            add(arm, "<", bump(&lo, if k == 1 { 0 } else { 1 }, tok)?, false);
        }
        ("^", Partial::Full(v)) | ("~", Partial::Full(v)) => {
            let lo = v.clone();
            add(arm, ">=", lo.clone(), true);
            let f = if prefix == "~" { if lo.patch == lo.patch { 1 } else { 1 } }
                    else if lo.major > 0 { 0 } else if lo.minor > 0 { 1 } else { 2 };
            add(arm, "<", bump(&lo, f, tok)?, false);
        }
        _ => return Err(RangeError::Syntax { at: tok.at }),
    }
    Ok(())
}
fn cmp_token(tok: &Tok, arm: &mut Arm) -> Result<(), RangeError> {
    let s = tok.s.as_str();
    let (op, n) = if s.starts_with(">=") { (">=",2) } else if s.starts_with("<=") { ("<=",2) }
        else if s.starts_with('>') { (">",1) } else if s.starts_with('<') { ("<",1) }
        else if s.starts_with('=') { ("=",1) } else { return Err(RangeError::Syntax { at: tok.at }); };
    if s.len() <= n || !s.as_bytes()[n].is_ascii_digit() {
        return Err(RangeError::Syntax { at: tok.at });
    }
    let v = parsed_full(tok, tok.at+n, &s[n..])?;
    add(arm, op, v, true);
    Ok(())
}
fn parse_hyphen(tokens: &[Tok]) -> Result<Arm, RangeError> {
    let mut arm = Arm { cmps: vec![], anchors: vec![] };
    if tokens.len() != 3 || tokens[1].s != "-" {
        let at = tokens.iter().find(|t| t.s == "-").unwrap().at;
        return Err(RangeError::Syntax { at });
    }
    let lo = partial(&tokens[0], tokens[0].at, &tokens[0].s)?;
    let hi = partial(&tokens[2], tokens[2].at, &tokens[2].s)?;
    let to_full = |p: Partial| -> Option<(Version, usize)> {
        match p {
            Partial::Full(v) => Some((v,3)),
            Partial::Part { nums,k } if k > 0 => Some((padded(&nums),k)),
            _ => None,
        }
    };
    let (lv, lk) = to_full(lo).ok_or(RangeError::Syntax { at: tokens[0].at })?;
    let (hv, hk) = to_full(hi).ok_or(RangeError::Syntax { at: tokens[2].at })?;
    add(&mut arm, ">=", lv, true);
    if hk == 3 && !tokens[2].s.contains('*') && !tokens[2].s.contains('x') && !tokens[2].s.contains('X') && tokens[2].s.matches('.').count() >= 2 {
        add(&mut arm, "<=", hv, true);
    } else {
        add(&mut arm, "<", bump(&hv, if hk == 1 { 0 } else { 1 }, &tokens[2])?, false);
    }
    let _ = lk;
    Ok(arm)
}
pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let mut segments: Vec<Vec<Tok>> = vec![Vec::new()];
    for t in tokens {
        if t.s == "||" {
            if segments.last().unwrap().is_empty() {
                return Err(RangeError::Empty { at: t.at });
            }
            segments.push(Vec::new());
        } else {
            segments.last_mut().unwrap().push(t);
        }
    }
    if segments.last().unwrap().is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    let mut arms = Vec::new();
    for seg in segments {
        if seg.iter().any(|t| t.s == "-") {
            arms.push(parse_hyphen(&seg)?);
            continue;
        }
        let mut arm = Arm { cmps: vec![], anchors: vec![] };
        for tok in &seg {
            let b = tok.s.as_bytes()[0];
            if matches!(b, b'<' | b'>' | b'=') {
                cmp_token(tok, &mut arm)?;
            } else if matches!(b, b'^' | b'~') {
                parse_operand(tok, &mut arm)?;
            } else if b.is_ascii_digit() || matches!(b,b'*'|b'x'|b'X') {
                parse_operand(tok, &mut arm)?;
            } else {
                return Err(RangeError::Syntax { at: tok.at });
            }
        }
        arms.push(arm);
    }
    Ok(Range { arms })
}
fn compare(c: &Cmp, v: &Version) -> bool {
    let o = compare_versions(v, &c.v);
    match c.op.as_str() {
        "=" => o == Ordering::Equal, "<" => o == Ordering::Less, "<=" => o != Ordering::Greater,
        ">" => o == Ordering::Greater, ">=" => o != Ordering::Less, _ => false,
    }
}
fn arm_match(v: &Version, arm: &Arm, include: bool) -> bool {
    if !arm.cmps.iter().all(|c| compare(c,v)) { return false; }
    include || v.pre.is_empty() || arm.anchors.contains(&core(v))
}
pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_match(v,a,include_prerelease))
}
pub fn matches(v: &Version, r: &Range) -> bool { matches_with(v,r,false) }

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index: usize, pub error: VersionError }
pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates,r,true)
}
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates,r,false)
}
fn extrema(candidates: &[String], r: &Range, high: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::new();
    for (i,s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index:i,error })?);
    }
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i],r) { continue; }
        if let Some(j) = best {
            let o = compare_versions(&parsed[i],&parsed[j]);
            if (high && o == Ordering::Greater) || (!high && o == Ordering::Less) { best = Some(i); }
        } else { best = Some(i); }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn core_string(c: (u64,u64,u64)) -> String { format!("{}.{}.{}",c.0,c.1,c.2) }
fn version_string(v: &Version) -> String {
    let mut s = core_string(core(v));
    if !v.pre.is_empty() { s.push('-'); s.push_str(&v.pre.join(".")); }
    s
}
fn arm_bounds(a: &Arm) -> (Option<(Version,bool)>, Option<(Version,bool)>, bool) {
    let mut lo: Option<(Version,bool)> = None;
    let mut hi: Option<(Version,bool)> = None;
    let mut eq = false;
    for c in &a.cmps {
        match c.op.as_str() {
            "=" => {
                eq = true;
                lo = Some((c.v.clone(),true));
                hi = Some((c.v.clone(),true));
            }
            ">" | ">=" => {
                let inc = c.op == ">=";
                let replace = lo.as_ref().map(|(v,i)| {
                    let o=compare_versions(&c.v,v);
                    o==Ordering::Greater || (o==Ordering::Equal && !inc && *i)
                }).unwrap_or(true);
                if replace { lo=Some((c.v.clone(),inc)); }
            }
            "<" | "<=" => {
                let inc = c.op == "<=";
                let replace = hi.as_ref().map(|(v,i)| {
                    let o=compare_versions(&c.v,v);
                    o==Ordering::Less || (o==Ordering::Equal && !inc && *i)
                }).unwrap_or(true);
                if replace { hi=Some((c.v.clone(),inc)); }
            }
            _ => {}
        }
    }
    (lo,hi,eq)
}
fn render_bound(b: &Option<(Version,bool)>, lower: bool) -> String {
    match b {
        None => "*".into(),
        Some((v,inc)) => format!("{}{}", if lower { if *inc { ">=" } else { ">" } } else if *inc { "<=" } else { "<" }, version_string(v)),
    }
}
pub fn canonical(r: &Range) -> String {
    let mut rows = Vec::new();
    for a in &r.arms {
        let (lo,hi,_) = arm_bounds(a);
        let mut anchors = a.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let pre = if anchors.is_empty() { "-".into() } else { anchors.iter().map(|c|core_string(*c)).collect::<Vec<_>>().join(",") };
        let row = format!("[{};{};pre={}]",render_bound(&lo,true),render_bound(&hi,false),pre);
        if !rows.contains(&row) { rows.push(row); }
    }
    rows.sort();
    if rows.is_empty() { "EMPTY".into() } else { rows.join(" || ") }
}
pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let (al,ah,_) = arm_bounds(aa);
            let (bl,bh,_) = arm_bounds(bb);
            let lower = match (&al,&bl) {
                (Some(x),Some(y)) => Some(if compare_versions(&x.0,&y.0)==Ordering::Greater { x.clone() } else { y.clone() }),
                (Some(x),None)|(None,Some(x)) => Some(x.clone()),
                _ => None,
            };
            let upper = match (&ah,&bh) {
                (Some(x),Some(y)) => Some(if compare_versions(&x.0,&y.0)==Ordering::Less { x.clone() } else { y.clone() }),
                (Some(x),None)|(None,Some(x)) => Some(x.clone()),
                _ => None,
            };
            if let (Some((l,li)),Some((u,ui))) = (&lower,&upper) {
                let o=compare_versions(l,u);
                if o==Ordering::Greater || (o==Ordering::Equal && !(*li && *ui)) { continue; }
            }
            if include_prerelease || (aa.anchors.iter().any(|x|bb.anchors.contains(x))) { return true; }
            let mut candidate = lower.as_ref().map(|x|x.0.clone()).unwrap_or_else(||mkver(0,0,0,vec![],vec![]));
            if let Some((_,false)) = &lower {
                if candidate.pre.is_empty() { candidate.pre=zero_pre(); }
            }
            if candidate.pre.is_empty() {
                if !lower.as_ref().map(|(v,i)| {
                    let o=compare_versions(&candidate,v); o==Ordering::Greater || (*i && o==Ordering::Equal)
                }).unwrap_or(true) {
                    let (x,y,z)=core(&candidate);
                    candidate=if z<u64::MAX { mkver(x,y,z+1,vec![],vec![]) }
                    else if y<u64::MAX { mkver(x,y+1,0,vec![],vec![]) }
                    else if x<u64::MAX { mkver(x+1,0,0,vec![],vec![]) }
                    else { continue };
                }
            }
            if upper.as_ref().map(|(v,i)| {
                let o=compare_versions(&candidate,v); o==Ordering::Less || (*i && o==Ordering::Equal)
            }).unwrap_or(true) { return true; }
        }
    }
    false
}

#[derive(Clone, Debug)]
pub enum SetExpr { Range(String), Union(Vec<SetExpr>), Intersection(Vec<SetExpr>) }
#[derive(Clone, Debug)]
pub struct Constraint { pub label: String, pub expr: SetExpr }
#[derive(Clone, Debug)]
pub struct Package { pub name: String, pub candidates: Vec<String>, pub constraints: Vec<Constraint> }
#[derive(Clone, Copy, Debug)]
pub enum Preference { Highest, Lowest, StableFirst }
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection { pub package: String, pub version: String }
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}
fn expr_valid(expr: &SetExpr, package: &str, label: &str, path: &mut Vec<usize>) -> Result<(),SelectError> {
    match expr {
        SetExpr::Range(s) => parse_range(s).map(|_|()).map_err(|error|SelectError::InvalidRange{package:package.into(),label:label.into(),path:path.clone(),error}),
        SetExpr::Union(xs)|SetExpr::Intersection(xs) => {
            for (i,x) in xs.iter().enumerate() {
                path.push(i);
                expr_valid(x,package,label,path)?;
                path.pop();
            }
            Ok(())
        }
    }
}
fn eval(expr: &SetExpr, v: &Version, ranges: &mut Vec<Range>, include: bool) -> bool {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).expect("validated range");
            matches_with(v,&r,include)
        }
        SetExpr::Union(xs) => xs.iter().any(|x|eval(x,v,ranges,include)),
        SetExpr::Intersection(xs) => xs.iter().all(|x|eval(x,v,ranges,include)),
    }
}
fn subset_satisfiable(indices: &[usize], constraints: &[Constraint], versions: &[Version], include: bool) -> bool {
    versions.iter().any(|v| indices.iter().all(|i| {
        let mut tmp=Vec::new();
        eval(&constraints[*i].expr,v,&mut tmp,include)
    }))
}
fn choose_conflict(constraints: &[Constraint], versions: &[Version], include: bool) -> Vec<String> {
    if versions.is_empty() { return vec![]; }
    let n=constraints.len();
    for size in 1..=n {
        let mut answer: Option<Vec<String>>=None;
        fn walk(start:usize,size:usize,cur:&mut Vec<usize>,n:usize,constraints:&[Constraint],versions:&[Version],include:bool,answer:&mut Option<Vec<String>>) {
            if cur.len()==size {
                if !subset_satisfiable(cur,constraints,versions,include) {
                    let mut labels:Vec<String>=cur.iter().map(|i|constraints[*i].label.clone()).collect();
                    labels.sort_by(|a,b|a.as_bytes().cmp(b.as_bytes()));
                    if answer.as_ref().map(|x|labels.as_slice()<x.as_slice()).unwrap_or(true) { *answer=Some(labels); }
                }
                return;
            }
            for i in start..n {
                cur.push(i); walk(i+1,size,cur,n,constraints,versions,include,answer); cur.pop();
            }
        }
        walk(0,size,&mut Vec::new(),n,constraints,versions,include,&mut answer);
        if let Some(x)=answer { return x; }
    }
    vec![]
}
pub fn select_packages(packages: &[Package], preference: Preference, include_prerelease: bool) -> Result<Vec<Selection>, SelectError> {
    let mut names=std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) { return Err(SelectError::DuplicatePackage{name:p.name.clone()}); }
    }
    for p in packages {
        let mut labels=std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) { return Err(SelectError::DuplicateLabel{package:p.name.clone(),label:c.label.clone()}); }
        }
    }
    let mut all_versions:Vec<Vec<Version>>=Vec::new();
    for p in packages {
        let mut vs=Vec::new();
        for (i,s) in p.candidates.iter().enumerate() {
            vs.push(parse_version(s).map_err(|error|SelectError::InvalidCandidate{package:p.name.clone(),index:i,error})?);
        }
        all_versions.push(vs);
    }
    for p in packages {
        for c in &p.constraints { expr_valid(&c.expr,&p.name,&c.label,&mut Vec::new())?; }
    }
    let mut result=Vec::new();
    for (pi,p) in packages.iter().enumerate() {
        let vs=&all_versions[pi];
        let mut best:Option<usize>=None;
        for i in 0..vs.len() {
            if !p.constraints.iter().all(|c|eval(&c.expr,&vs[i],&mut Vec::new(),include_prerelease)) { continue; }
            if let Some(j)=best {
                let stable_pref=preference==Preference::StableFirst && vs[i].pre.is_empty()!=vs[j].pre.is_empty();
                let better=if stable_pref { vs[i].pre.is_empty() }
                    else { match preference {
                        Preference::Lowest => compare_versions(&vs[i],&vs[j])==Ordering::Less,
                        _ => compare_versions(&vs[i],&vs[j])==Ordering::Greater,
                    }};
                if better { best=Some(i); }
            } else { best=Some(i); }
        }
        if let Some(i)=best {
            result.push(Selection{package:p.name.clone(),version:p.candidates[i].clone()});
        } else {
            let labels=choose_conflict(&p.constraints,vs,include_prerelease);
            return Err(SelectError::NoSolution{package:p.name.clone(),labels});
        }
    }
    Ok(result)
}
