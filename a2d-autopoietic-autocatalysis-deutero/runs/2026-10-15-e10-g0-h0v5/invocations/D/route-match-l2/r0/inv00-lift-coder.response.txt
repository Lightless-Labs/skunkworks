#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parts = parse_path(head)?;

        let mut best: Option<(usize, Vec<u8>, Vec<(String, String)>)> = None;
        for route in &self.routes {
            if let Some(params) = match_route(&route.segments, &parts) {
                let replace = match &best {
                    None => true,
                    Some((_, rank, _)) => route.rank > *rank,
                };
                if replace {
                    best = Some((route.id, route.rank.clone(), params));
                }
            }
        }

        match best {
            Some((route, _, params)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let bytes = pattern.as_bytes();
    let mut ranges = Vec::new();
    let mut start = 1;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                if start == i {
                    return Err(Error::InvalidPattern { at: i });
                }
                ranges.push((start, i));
                return Err(Error::InvalidPattern { at: i });
            }
            if i == start {
                return Err(Error::InvalidPattern { at: start });
            }
            ranges.push((start, i));
            start = i + 1;
        }
    }
    if start == bytes.len() {
        return Err(Error::InvalidPattern { at: bytes.len() - 1 });
    }
    ranges.push((start, bytes.len()));

    let mut segments = Vec::with_capacity(ranges.len());
    let mut names = std::collections::HashSet::new();

    for (begin, end) in ranges {
        let segment = &pattern[begin..end];
        if let Some(name) = segment.strip_prefix(':') {
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: begin });
            }
            segments.push(Segment::Param(name.to_owned()));
        } else if let Some(name) = segment.strip_prefix('*') {
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: begin });
            }
            if end != bytes.len() {
                return Err(Error::InvalidPattern { at: begin });
            }
            segments.push(Segment::Wildcard(name.to_owned()));
        } else {
            segments.push(Segment::Static(decode_segment(pattern, begin, end)?));
        }
    }
    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut ranges = Vec::new();
    let mut start = 1;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                if start == i {
                    return Err(Error::InvalidPath { at: i });
                }
                ranges.push((start, i));
                return Err(Error::InvalidPath { at: i });
            }
            if i == start {
                return Err(Error::InvalidPath { at: start });
            }
            ranges.push((start, i));
            start = i + 1;
        }
    }
    if start == bytes.len() {
        return Err(Error::InvalidPath {
            at: bytes.len().saturating_sub(1),
        });
    }
    ranges.push((start, bytes.len()));

    ranges
        .into_iter()
        .map(|(begin, end)| decode_segment(path, begin, end).map_err(|e| match e {
            Error::InvalidPattern { at } | Error::InvalidPath { at } => Error::InvalidPath { at },
            Error::NotFound => Error::NotFound,
        }))
        .collect()
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(input: &str, begin: usize, end: usize) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(end - begin);
    let mut i = begin;

    while i < end {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= end {
                return Err(Error::InvalidPattern { at: i });
            }
            let hi = hex_upper(bytes[i + 1]);
            let lo = hex_upper(bytes[i + 2]);
            let value = match (hi, lo) {
                (Some(h), Some(l)) => (h << 4) | l,
                _ => return Err(Error::InvalidPattern { at: i }),
            };
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(Error::InvalidPattern { at: i });
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(Error::InvalidPattern { at: i });
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded segment is ASCII"))
}

fn hex_upper(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn match_route(segments: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for (i, segment) in segments.iter().enumerate() {
        match segment {
            Segment::Static(expected) => {
                if path.get(path_index)? != expected {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                return Some(params);
            }
        }
        if i + 1 == segments.len() && path_index != path.len() {
            return None;
        }
    }

    if path_index == path.len() {
        Some(params)
    } else {
        None
    }
}
