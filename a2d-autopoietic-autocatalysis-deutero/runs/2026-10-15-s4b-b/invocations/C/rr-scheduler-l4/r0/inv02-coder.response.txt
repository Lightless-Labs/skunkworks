```rust
use std::collections::VecDeque;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    level: usize,
}

#[derive(Clone, Debug)]
struct State {
    burst_position: usize,
    remaining: u32,
    level: usize,
    completion: u64,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let mut queues: Vec<VecDeque<usize>> =
        (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut states: Vec<State> = jobs
        .iter()
        .map(|job| State {
            burst_position: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: 0,
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: job.arrival as u64,
            kind: 0,
            index,
            level: job.level,
        })
        .collect();

    let mut time = 0u64;
    let mut last_dispatch: Option<u32> = None;
    let mut runs: Vec<Run> = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut completed = 0usize;

    while completed < jobs.len() {
        events.sort_by_key(|event| (event.time, event.kind, event.index));

        while events.first().is_some_and(|event| event.time <= time) {
            let event = events.remove(0);
            states[event.index].level = event.level;
            queues[event.level].push_back(event.index);
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if let Some(event) = events.first() {
                let next_time = event.time;
                if next_time > time {
                    if let Some(last) = idle.last_mut() {
                        if last.1 == time {
                            last.1 = next_time;
                        } else {
                            idle.push((time, next_time));
                        }
                    } else {
                        idle.push((time, next_time));
                    }
                    time = next_time;
                    last_dispatch = None;
                }
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let id = jobs[index].id;

        if last_dispatch.is_some_and(|last_id| last_id != id) {
            time += switch_cost as u64;
        }

        events.sort_by_key(|event| (event.time, event.kind, event.index));
        while events.first().is_some_and(|event| event.time <= time) {
            let event = events.remove(0);
            states[event.index].level = event.level;
            queues[event.level].push_back(event.index);
        }

        let quantum = quanta[level];
        let consumed = states[index].remaining.min(quantum);
        let start = time;
        time += consumed as u64;
        states[index].remaining -= consumed;

        let burst_ended = states[index].remaining == 0;
        let final_burst =
            burst_ended && states[index].burst_position + 1 == jobs[index].bursts.len();
        runs.push(Run {
            id,
            start,
            end: time,
            finished: final_burst,
            level,
        });

        if burst_ended && !final_burst {
            let io_position = states[index].burst_position + 1;
            let io_delay = jobs[index].bursts[io_position] as u64;
            let next_cpu_position = io_position + 1;
            states[index].burst_position = next_cpu_position;
            states[index].remaining = jobs[index].bursts[next_cpu_position];
            let wake_level = level.saturating_sub(1);
            events.push(Event {
                time: time + io_delay,
                kind: 1,
                index,
                level: wake_level,
            });
        }

        events.sort_by_key(|event| (event.time, event.kind, event.index));
        while events.first().is_some_and(|event| event.time <= time) {
            let event = events.remove(0);
            states[event.index].level = event.level;
            queues[event.level].push_back(event.index);
        }

        if final_burst {
            states[index].completion = time;
            completed += 1;
        } else if burst_ended {
            // The job is blocked until its wake event.
        } else {
            states[index].level = (level + 1).min(quanta.len() - 1);
            queues[states[index].level].push_back(index);
        }

        last_dispatch = Some(id);
    }

    let stats: Vec<Stat> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| {
            let completion = states[index].completion;
            let turnaround = completion - job.arrival as u64;
            let cpu: u64 = job.bursts.iter().step_by(2).map(|&burst| burst as u64).sum();
            let io: u64 = job
                .bursts
                .iter()
                .skip(1)
                .step_by(2)
                .map(|&delay| delay as u64)
                .sum();
            Stat {
                id: job.id,
                completion,
                turnaround,
                waiting: turnaround - cpu - io,
            }
        })
        .collect();

    let count = jobs.len() as u64;
    let round_mean_x100 = |total: u64| {
        if count == 0 {
            0
        } else {
            (total * 100 + count / 2) / count
        }
    };
    let total_waiting: u64 = stats.iter().map(|stat| stat.waiting).sum();
    let total_turnaround: u64 = stats.iter().map(|stat| stat.turnaround).sum();

    Ok(Schedule {
        end: time,
        runs,
        idle,
        stats,
        mean_waiting_x100: round_mean_x100(total_waiting),
        mean_turnaround_x100: round_mean_x100(total_turnaround),
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn cpu_only_demotion() {
        assert_eq!(
            schedule(
                &[
                    Job {
                        id: 1,
                        arrival: 0,
                        bursts: vec![6],
                        level: 0
                    },
                    Job {
                        id: 2,
                        arrival: 0,
                        bursts: vec![2],
                        level: 0
                    }
                ],
                &[2, 4],
                0
            ),
            Ok(Schedule {
                runs: vec![
                    Run {
                        id: 1,
                        start: 0,
                        end: 2,
                        finished: false,
                        level: 0
                    },
                    Run {
                        id: 2,
                        start: 2,
                        end: 4,
                        finished: true,
                        level: 0
                    },
                    Run {
                        id: 1,
                        start: 4,
                        end: 8,
                        finished: true,
                        level: 1
                    }
                ],
                idle: vec![],
                stats: vec![
                    Stat {
                        id: 1,
                        completion: 8,
                        turnaround: 8,
                        waiting: 2
                    },
                    Stat {
                        id: 2,
                        completion: 4,
                        turnaround: 4,
                        waiting: 2
                    }
                ],
                mean_waiting_x100: 200,
                mean_turnaround_x100: 600,
                end: 8
            })
        );
    }

    #[test]
    fn io_idle_and_promotion() {
        assert_eq!(
            schedule(
                &[Job {
                    id: 5,
                    arrival: 2,
                    bursts: vec![1, 3, 2],
                    level: 1
                }],
                &[3, 4],
                0
            ),
            Ok(Schedule {
                runs: vec![
                    Run {
                        id: 5,
                        start: 2,
                        end: 3,
                        finished: false,
                        level: 1
                    },
                    Run {
                        id: 5,
                        start: 6,
                        end: 8,
                        finished: true,
                        level: 0
                    }
                ],
                idle: vec![(0, 2), (3, 6)],
                stats: vec![Stat {
                    id: 5,
                    completion: 8,
                    turnaround: 6,
                    waiting: 0
                }],
                mean_waiting_x100: 0,
                mean_turnaround_x100: 600,
                end: 8
            })
        );
    }

    #[test]
    fn empty_statistics() {
        assert_eq!(
            schedule(&[], &[1], 0),
            Ok(Schedule {
                runs: vec![],
                idle: vec![],
                stats: vec![],
                mean_waiting_x100: 0,
                mean_turnaround_x100: 0,
                end: 0
            })
        );
    }

    #[test]
    fn invalid_cpu_burst() {
        assert_eq!(
            schedule(
                &[Job {
                    id: 1,
                    arrival: 0,
                    bursts: vec![0],
                    level: 0
                }],
                &[2],
                0
            ),
            Err(Error::Burst {
                index: 0,
                position: 0
            })
        );
    }

    #[test]
    fn zero_duration_io_promotes_before_selection() {
        let result = schedule(
            &[
                Job {
                    id: 1,
                    arrival: 0,
                    bursts: vec![1, 0, 1],
                    level: 1,
                },
                Job {
                    id: 2,
                    arrival: 1,
                    bursts: vec![2],
                    level: 0,
                },
            ],
            &[2, 2],
            0,
        )
        .unwrap();

        assert_eq!(
            result.runs,
            vec![
                Run {
                    id: 1,
                    start: 0,
                    end: 1,
                    finished: false,
                    level: 1,
                },
                Run {
                    id: 2,
                    start: 1,
                    end: 3,
                    finished: true,
                    level: 0,
                },
                Run {
                    id: 1,
                    start: 3,
                    end: 4,
                    finished: true,
                    level: 0,
                },
            ]
        );
    }

    #[test]
    fn validation_precedence() {
        assert_eq!(
            schedule(
                &[
                    Job {
                        id: 1,
                        arrival: 1001,
                        bursts: vec![1],
                        level: 0,
                    },
                    Job {
                        id: 1,
                        arrival: 0,
                        bursts: vec![1],
                        level: 0,
                    },
                ],
                &[1],
                0,
            ),
            Err(Error::Arrival { index: 0 })
        );
    }
}

fn main() {}
```
