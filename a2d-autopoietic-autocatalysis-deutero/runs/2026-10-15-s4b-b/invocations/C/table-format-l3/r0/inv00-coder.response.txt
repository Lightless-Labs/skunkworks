#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    let fixed = n.checked_mul(3).and_then(|v| v.checked_add(1)).unwrap_or(usize::MAX);
    let minimum = fixed.checked_add(n).unwrap_or(usize::MAX);
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }
        }
    }

    let trimmed: Vec<Vec<String>> = rows
        .iter()
        .map(|cells| {
            cells
                .iter()
                .map(|cell| cell.trim_end_matches(' ').to_owned())
                .collect()
        })
        .collect();

    let mut widths = vec![1usize; n];
    for cells in &trimmed {
        for (column, cell) in cells.iter().enumerate() {
            widths[column] = widths[column].max(cell.chars().count());
        }
    }

    if options.max_width != 0 {
        let mut total = fixed
            .checked_add(widths.iter().copied().sum::<usize>())
            .unwrap_or(usize::MAX);
        while total > options.max_width {
            let widest = (0..n)
                .filter(|&i| widths[i] > 1)
                .max_by_key(|&i| (widths[i], i));
            let Some(column) = widest else { break };
            widths[column] -= 1;
            total -= 1;
        }
    }

    let align_for = |column: usize| {
        if options.alignments.is_empty() {
            Alignment::Left
        } else {
            options.alignments[column].clone()
        }
    };

    let mut output = String::new();
    output.push('+');
    for &width in &widths {
        output.extend(std::iter::repeat('-').take(width + 2));
        output.push('+');
    }
    output.push('\n');

    for (row_index, cells) in trimmed.iter().enumerate() {
        output.push('|');
        for (column, cell) in cells.iter().enumerate() {
            let width = widths[column];
            let chars: Vec<char> = cell.chars().collect();
            let content: String = if chars.len() > width {
                let mut s: String = chars[..width - 1].iter().collect();
                s.push('…');
                s
            } else {
                cell.clone()
            };
            let content_width = content.chars().count();
            let extra = width - content_width;
            let (left, right) = match align_for(column) {
                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),
            };
            output.push(' ');
            output.extend(std::iter::repeat(' ').take(left));
            output.push_str(&content);
            output.extend(std::iter::repeat(' ').take(right));
            output.push(' ');
            output.push('|');
        }
        output.push('\n');

        if options.header && row_index == 0 && rows.len() > 1 {
            output.push('+');
            for &width in &widths {
                output.extend(std::iter::repeat('-').take(width + 2));
                output.push('+');
            }
            output.push('\n');
        }
    }

    output.push('+');
    for &width in &widths {
        output.extend(std::iter::repeat('-').take(width + 2));
        output.push('+');
    }
    output.push('\n');
    Ok(output)
}

fn main() {}
