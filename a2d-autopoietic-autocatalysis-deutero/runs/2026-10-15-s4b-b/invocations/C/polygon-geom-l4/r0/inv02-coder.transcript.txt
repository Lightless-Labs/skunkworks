const LIMIT: i64 = 1_000_000_000;

fn in_range(p: Point) -> bool {
    (-LIMIT..=LIMIT).contains(&p.x) && (-LIMIT..=LIMIT).contains(&p.y)
}

fn prepare(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > 4096 {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }
    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective.first() == effective.last() {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn shoelace(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn rounded_sqrt_scaled(dx: i128, dy: i128) -> u64 {
    let d2 = (dx * dx + dy * dy) as u128;
    let value = d2 * 1_000_000;
    let mut lo = 0u128;
    let mut hi = 3_000_000_000u128;
    while lo < hi {
        let mid = (lo + hi + 1) / 2;
        if mid * mid <= value {
            lo = mid;
        } else {
            hi = mid - 1;
        }
    }
    if value - lo * lo >= lo * 2 + 1 {
        (lo + 1) as u64
    } else {
        lo as u64
    }
}

fn edge_length(a: Point, b: Point) -> u64 {
    rounded_sqrt_scaled(
        b.x as i128 - a.x as i128,
        b.y as i128 - a.y as i128,
    )
}

fn measures_for(points: &[Point]) -> Measures {
    let signed = shoelace(points);
    let n = points.len();
    let edge_milli: Vec<u64> = (0..n)
        .map(|i| edge_length(points[i], points[(i + 1) % n]))
        .collect();
    let perimeter_milli: u64 = edge_milli.iter().copied().sum();
    let longest_edge = (0..n).max_by(|&i, &j| {
        edge_milli[i].cmp(&edge_milli[j]).then_with(|| {
            let pair = |k: usize| {
                let a = points[k];
                let b = points[(k + 1) % n];
                if a <= b { (a, b) } else { (b, a) }
            };
            pair(j).cmp(&pair(i)).then_with(|| j.cmp(&i))
        })
    });
    Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation: if signed > 0 {
            Orientation::CounterClockwise
        } else if signed < 0 {
            Orientation::Clockwise
        } else {
            Orientation::Degenerate
        },
        edge_milli,
        perimeter_milli,
        longest_edge,
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare(points)?;
    if queries.len() > 4096 {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let n = polygon.len();
    Ok(queries
        .iter()
        .map(|&query| {
            for i in 0..n {
                if on_segment(polygon[i], polygon[(i + 1) % n], query) {
                    return Location::Boundary { edge: i };
                }
            }
            let mut inside = false;
            for i in 0..n {
                let a = polygon[i];
                let b = polygon[(i + 1) % n];
                if (a.y > query.y) != (b.y > query.y) {
                    let c = cross(a, b, query);
                    if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                        inside = !inside;
                    }
                }
            }
            if inside {
                Location::Inside
            } else {
                Location::Outside
            }
        })
        .collect())
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > 4096 {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|(existing, _)| *existing == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);
    let m = unique.len();
    let mut hull: Vec<usize> = Vec::new();

    if m <= 2 {
        hull.extend(0..m);
    } else {
        let all_collinear =
            (2..m).all(|i| cross(unique[0].0, unique[m - 1].0, unique[i].0) == 0);
        if all_collinear {
            if mode == Collinear::Include {
                hull.extend(0..m);
            } else {
                hull.push(0);
                hull.push(m - 1);
            }
        } else {
            let mut lower: Vec<usize> = Vec::new();
            for i in 0..m {
                while lower.len() >= 2 {
                    let a = unique[lower[lower.len() - 2]].0;
                    let b = unique[lower[lower.len() - 1]].0;
                    let c = unique[i].0;
                    let turn = cross(a, b, c);
                    if (mode == Collinear::Include && turn < 0)
                        || (mode == Collinear::Exclude && turn <= 0)
                    {
                        lower.pop();
                    } else {
                        break;
                    }
                }
                lower.push(i);
            }

            let mut upper: Vec<usize> = Vec::new();
            for i in (0..m).rev() {
                while upper.len() >= 2 {
                    let a = unique[upper[upper.len() - 2]].0;
                    let b = unique[upper[upper.len() - 1]].0;
                    let c = unique[i].0;
                    let turn = cross(a, b, c);
                    if (mode == Collinear::Include && turn < 0)
                        || (mode == Collinear::Exclude && turn <= 0)
                    {
                        upper.pop();
                    } else {
                        break;
                    }
                }
                upper.push(i);
            }
            lower.pop();
            upper.pop();
            lower.extend(upper);
            hull = lower;
        }
    }

    let vertices: Vec<Point> = hull.iter().map(|&i| unique[i].0).collect();
    let source_indices: Vec<usize> = hull.iter().map(|&i| unique[i].1).collect();
    let area2 = if vertices.len() >= 3 {
        shoelace(&vertices).unsigned_abs()
    } else {
        0
    };
    let perimeter_milli = if vertices.is_empty() {
        0
    } else {
        (0..vertices.len())
            .map(|i| edge_length(vertices[i], vertices[(i + 1) % vertices.len()]))
            .sum()
    };
    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}

fn segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool {
    let ab_c = cross(a, b, c);
    let ab_d = cross(a, b, d);
    let cd_a = cross(c, d, a);
    let cd_b = cross(c, d, b);
    if ab_c == 0 && on_segment(a, b, c) {
        return true;
    }
    if ab_d == 0 && on_segment(a, b, d) {
        return true;
    }
    if cd_a == 0 && on_segment(c, d, a) {
        return true;
    }
    if cd_b == 0 && on_segment(c, d, b) {
        return true;
    }
    (ab_c > 0) != (ab_d > 0) && (cd_a > 0) != (cd_b > 0)
}

pub fn validate(points: &[Point]) -> Result<Validated, GeomError> {
    let effective = prepare(points)?;
    let n = effective.len();

    for i in 0..n {
        for j in (i + 1)..n {
            if effective[i] == effective[j] {
                return Err(GeomError::Duplicate {
                    first: i,
                    second: j,
                });
            }
        }
    }

    let signed = shoelace(&effective);
    if signed == 0 {
        return Err(GeomError::ZeroArea);
    }

    for i in 0..n {
        let prev = effective[(i + n - 1) % n];
        let curr = effective[i];
        let next = effective[(i + 1) % n];
        if cross(prev, curr, next) == 0 {
            let dot = (prev.x as i128 - curr.x as i128)
                * (next.x as i128 - curr.x as i128)
                + (prev.y as i128 - curr.y as i128)
                    * (next.y as i128 - curr.y as i128);
            if dot > 0 {
                return Err(GeomError::AdjacentOverlap { vertex: i });
            }
        }
    }

    for i in 0..n {
        for j in (i + 1)..n {
            if j == i + 1 || (i == 0 && j == n - 1) {
                continue;
            }
            if segments_intersect(
                effective[i],
                effective[(i + 1) % n],
                effective[j],
                effective[(j + 1) % n],
            ) {
                return Err(GeomError::Intersection {
                    edge_a: i,
                    edge_b: j,
                });
            }
        }
    }

    let min_index = (0..n).min_by_key(|&i| effective[i]).unwrap();
    let indices: Vec<usize> = if signed > 0 {
        (0..n).map(|k| (min_index + k) % n).collect()
    } else {
        (0..n).map(|k| (min_index + n - k) % n).collect()
    };
    let vertices: Vec<Point> = indices.iter().map(|&i| effective[i]).collect();
    let measures = measures_for(&vertices);
    Ok(Validated {
        vertices,
        source_indices: indices,
        measures,
    })
}
