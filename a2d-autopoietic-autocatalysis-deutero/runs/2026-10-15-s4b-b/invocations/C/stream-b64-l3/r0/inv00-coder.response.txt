#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn digit(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn encoded_digit(n: u8, alphabet: Alphabet) -> u8 {
    match n {
        0..=25 => b'A' + n,
        26..=51 => b'a' + (n - 26),
        52..=61 => b'0' + (n - 52),
        62 => if alphabet == Alphabet::Standard { b'+' } else { b'-' },
        _ => if alphabet == Alphabet::Standard { b'/' } else { b'_' },
    }
}

fn encode_raw(input: &[u8], options: Options) -> Vec<u8> {
    let mut out = Vec::with_capacity(input.len().saturating_add(2) / 3 * 4);
    let mut i = 0;
    while i + 3 <= input.len() {
        let a = input[i];
        let b = input[i + 1];
        let c = input[i + 2];
        out.push(encoded_digit(a >> 2, options.alphabet));
        out.push(encoded_digit(((a & 3) << 4) | (b >> 4), options.alphabet));
        out.push(encoded_digit(((b & 15) << 2) | (c >> 6), options.alphabet));
        out.push(encoded_digit(c & 63, options.alphabet));
        i += 3;
    }
    match input.len() - i {
        1 => {
            let a = input[i];
            out.push(encoded_digit(a >> 2, options.alphabet));
            out.push(encoded_digit((a & 3) << 4, options.alphabet));
            if options.padding != Padding::Forbidden {
                out.extend_from_slice(b"==");
            }
        }
        2 => {
            let a = input[i];
            let b = input[i + 1];
            out.push(encoded_digit(a >> 2, options.alphabet));
            out.push(encoded_digit(((a & 3) << 4) | (b >> 4), options.alphabet));
            out.push(encoded_digit((b & 15) << 2, options.alphabet));
            if options.padding != Padding::Forbidden {
                out.push(b'=');
            }
        }
        _ => {}
    }
    out
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate(options)?;
    let raw = encode_raw(input, options);
    let bytes = if let Whitespace::Wrapped { width, ending } = options.whitespace {
        let sep: &[u8] = match ending {
            LineEnding::Lf => b"\n",
            LineEnding::CrLf => b"\r\n",
        };
        let rows = raw.len().saturating_sub(1) / width;
        let mut wrapped = Vec::with_capacity(raw.len().saturating_add(rows * sep.len()));
        for (i, chunk) in raw.chunks(width).enumerate() {
            if i != 0 {
                wrapped.extend_from_slice(sep);
            }
            wrapped.extend_from_slice(chunk);
        }
        wrapped
    } else {
        raw
    };
    Ok(String::from_utf8(bytes).expect("base64 alphabet is ASCII"))
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            tail: Vec::new(),
            closed: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut data = Vec::with_capacity(self.tail.len() + input.len());
        data.extend_from_slice(&self.tail);
        data.extend_from_slice(input);
        let complete_len = data.len() / 3 * 3;
        let mut out = encode_raw(&data[..complete_len], self.options);
        self.tail.clear();
        self.tail.extend_from_slice(&data[complete_len..]);
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            // Insert separators at row boundaries across the entire stream.
            let prior = self.emitted_symbols;
            let mut wrapped = Vec::with_capacity(out.len() + out.len() / width * 2);
            for (i, &b) in out.iter().enumerate() {
                if prior + i > 0 && (prior + i) % width == 0 {
                    wrapped.extend_from_slice(match ending {
                        LineEnding::Lf => b"\n",
                        LineEnding::CrLf => b"\r\n",
                    });
                }
                wrapped.push(b);
            }
            out = wrapped;
            self.emitted_symbols += complete_len / 3 * 4;
        } else {
            self.emitted_symbols += out.len();
        }
        Ok(String::from_utf8(out).expect("base64 alphabet is ASCII"))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;
        let out = encode_raw(&self.tail, self.options);
        self.tail.clear();
        if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
            let mut wrapped = Vec::with_capacity(out.len() + out.len() / width * 2);
            for (i, &b) in out.iter().enumerate() {
                if self.emitted_symbols + i > 0
                    && (self.emitted_symbols + i) % width == 0
                {
                    wrapped.extend_from_slice(match ending {
                        LineEnding::Lf => b"\n",
                        LineEnding::CrLf => b"\r\n",
                    });
                }
                wrapped.push(b);
            }
            self.emitted_symbols += out.len();
            Ok(String::from_utf8(wrapped).expect("base64 alphabet is ASCII"))
        } else {
            Ok(String::from_utf8(out).expect("base64 alphabet is ASCII"))
        }
    }
}

impl Encoder {
    // Number of raw symbols already emitted; private stream layout state.
    fn set_emitted_symbols_placeholder(&mut self) {}
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum FrameState {
    Normal,
    NeedLf(usize),
    NeedSymbol(usize),
}

pub struct Decoder {
    options: Options,
    group: Vec<(u8, usize)>,
    offset: usize,
    terminal: bool,
    sticky: Option<Error>,
    closed: bool,
    row_symbols: usize,
    frame: FrameState,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate(options)?;
        Ok(Self {
            options,
            group: Vec::new(),
            offset: 0,
            terminal: false,
            sticky: None,
            closed: false,
            row_symbols: 0,
            frame: FrameState::Normal,
        })
    }

    fn fail(&mut self, error: Error) -> Result<Vec<u8>, Error> {
        self.sticky = Some(error.clone());
        Err(error)
    }

    fn symbol(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        let pos = self.group.len();
        if self.terminal {
            return Err(Error::BadPadding { offset });
        }
        if pos == 2 && self.group[2].0 == b'=' && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }
        if byte == b'=' {
            if pos < 2 || pos > 3 {
                return Err(Error::BadPadding { offset });
            }
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
        } else if digit(byte, self.options.alphabet).is_none() {
            return Err(Error::InvalidByte { offset, byte });
        }

        self.group.push((byte, offset));
        if self.group.len() == 4 {
            let g = &self.group;
            let a = digit(g[0].0, self.options.alphabet).ok_or(Error::BadPadding { offset: g[0].1 })?;
            let b = digit(g[1].0, self.options.alphabet).ok_or(Error::BadPadding { offset: g[1].1 })?;
            out.push((a << 2) | (b >> 4));
            if g[2].0 == b'=' {
                self.terminal = true;
            } else {
                let c = digit(g[2].0, self.options.alphabet).ok_or(Error::BadPadding { offset: g[2].1 })?;
                out.push((b << 4) | (c >> 2));
                if g[3].0 == b'=' {
                    self.terminal = true;
                } else {
                    let d = digit(g[3].0, self.options.alphabet).ok_or(Error::BadPadding { offset: g[3].1 })?;
                    out.push((c << 6) | d);
                }
            }
            self.group.clear();
        }
        Ok(())
    }

    fn process_byte(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        match self.options.whitespace {
            Whitespace::None => self.symbol(byte, offset, out),
            Whitespace::IgnoreAscii => {
                if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                    Ok(())
                } else {
                    self.symbol(byte, offset, out)
                }
            }
            Whitespace::Wrapped { width, ending } => {
                match self.frame {
                    FrameState::NeedLf(cr) => {
                        if byte != b'\n' {
                            return Err(Error::BadWrap { offset });
                        }
                        self.frame = FrameState::NeedSymbol(cr);
                        return Ok(());
                    }
                    FrameState::NeedSymbol(first) => {
                        self.frame = FrameState::Normal;
                        if byte == b'\n' || byte == b'\r' || byte.is_ascii_whitespace() {
                            return Err(Error::BadWrap { offset });
                        }
                        self.row_symbols = 0;
                        self.symbol(byte, offset, out)?;
                        self.row_symbols = 1;
                        let _ = first;
                        return Ok(());
                    }
                    FrameState::Normal => {}
                }
                if self.row_symbols == width {
                    match ending {
                        LineEnding::Lf if byte == b'\n' => {
                            self.frame = FrameState::NeedSymbol(offset);
                            return Ok(());
                        }
                        LineEnding::CrLf if byte == b'\r' => {
                            self.frame = FrameState::NeedLf(offset);
                            return Ok(());
                        }
                        _ => return Err(Error::BadWrap { offset }),
                    }
                }
                if byte == b'\n' || byte == b'\r' || byte.is_ascii_whitespace() {
                    return Err(Error::BadWrap { offset });
                }
                self.symbol(byte, offset, out)?;
                self.row_symbols += 1;
                Ok(())
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut out = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;
            if let Err(error) = self.process_byte(byte, offset, &mut out) {
                return self.fail(error);
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky {
            return Err(error.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        if let Whitespace::Wrapped { .. } = self.options.whitespace {
            match self.frame {
                FrameState::NeedLf(cr) => return self.fail(Error::BadWrap { offset: cr }),
                FrameState::NeedSymbol(first) => return self.fail(Error::BadWrap { offset: first }),
                FrameState::Normal => {}
            }
        }
        if self.group.len() == 3 && self.group[2].0 == b'=' {
            let offset = self.group[2].1;
            return self.fail(Error::BadPadding { offset });
        }
        if !self.group.is_empty() {
            match self.options.padding {
                Padding::Required => {
                    return self.fail(Error::Truncated { offset: self.group[0].1 });
                }
                Padding::Forbidden | Padding::Optional => match self.group.len() {
                    1 => return self.fail(Error::Truncated { offset: self.group[0].1 }),
                    2 => {
                        let a = digit(self.group[0].0, self.options.alphabet).unwrap();
                        let b = digit(self.group[1].0, self.options.alphabet).unwrap();
                        self.closed = true;
                        return Ok(vec![(a << 2) | (b >> 4)]);
                    }
                    3 => {
                        let a = digit(self.group[0].0, self.options.alphabet).unwrap();
                        let b = digit(self.group[1].0, self.options.alphabet).unwrap();
                        let c = digit(self.group[2].0, self.options.alphabet).unwrap();
                        self.closed = true;
                        return Ok(vec![(a << 2) | (b >> 4), (b << 4) | (c >> 2)]);
                    }
                    _ => {}
                },
            }
        }
        self.closed = true;
        Ok(Vec::new())
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate(options)?;
    let mut decoder = Decoder::new(options)?;
    let out = decoder.push(input)?;
    let tail = decoder.finish()?;
    let mut all = out;
    all.extend(tail);
    Ok(all)
}

fn main() {}
