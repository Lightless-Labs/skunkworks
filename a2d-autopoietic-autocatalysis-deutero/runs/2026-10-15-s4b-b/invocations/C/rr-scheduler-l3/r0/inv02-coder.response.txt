#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut arrival_order: Vec<usize> = (0..jobs.len()).collect();
    arrival_order.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut completion: Vec<(u32, u64)> = jobs.iter().map(|job| (job.id, 0)).collect();
    let mut completed: Vec<bool> = vec![false; jobs.len()];
    let mut runs: Vec<Run> = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut time = 0u64;
    let mut next_arrival = 0usize;
    let mut last_dispatched_id: Option<u32> = None;
    let mut finished_count = 0usize;

    fn admit_through(
        jobs: &[Job],
        arrival_order: &[usize],
        next_arrival: &mut usize,
        time: u64,
        queues: &mut [std::collections::VecDeque<usize>],
        completed: &mut [bool],
        completion: &mut [(u32, u64)],
        finished_count: &mut usize,
    ) {
        while *next_arrival < arrival_order.len() {
            let index = arrival_order[*next_arrival];
            let job = &jobs[index];
            if u64::from(job.arrival) > time {
                break;
            }
            *next_arrival += 1;

            if job.ticks == 0 {
                completed[index] = true;
                completion[index].1 = u64::from(job.arrival);
                *finished_count += 1;
            } else {
                queues[job.level].push_back(index);
            }
        }
    }

    while finished_count < jobs.len() {
        admit_through(
            jobs,
            &arrival_order,
            &mut next_arrival,
            time,
            &mut queues,
            &mut completed,
            &mut completion,
            &mut finished_count,
        );

        let selected_level = queues.iter().position(|queue| !queue.is_empty());

        if selected_level.is_none() {
            if next_arrival >= arrival_order.len() {
                break;
            }

            let next_time = u64::from(jobs[arrival_order[next_arrival]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched_id = None;
            }
            continue;
        }

        let level = selected_level.unwrap();
        let index = queues[level].pop_front().unwrap();
        let job = &jobs[index];

        if last_dispatched_id.is_some_and(|id| id != job.id) {
            time += u64::from(switch_cost);
        }

        admit_through(
            jobs,
            &arrival_order,
            &mut next_arrival,
            time,
            &mut queues,
            &mut completed,
            &mut completion,
            &mut finished_count,
        );

        let start = time;
        let consumed = remaining[index].min(quanta[level]);
        time += u64::from(consumed);
        remaining[index] -= consumed;
        let is_finished = remaining[index] == 0;

        runs.push(Run {
            id: job.id,
            start,
            end: time,
            finished: is_finished,
            level,
        });

        admit_through(
            jobs,
            &arrival_order,
            &mut next_arrival,
            time,
            &mut queues,
            &mut completed,
            &mut completion,
            &mut finished_count,
        );

        if is_finished {
            completed[index] = true;
            completion[index].1 = time;
            finished_count += 1;
        } else {
            let next_level = (level + 1).min(quanta.len() - 1);
            queues[next_level].push_back(index);
        }

        last_dispatched_id = Some(job.id);
    }

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn demotion_basics() {
        assert_eq!(
            schedule(
                &[
                    Job { id: 1, arrival: 0, ticks: 7, level: 0 },
                    Job { id: 2, arrival: 0, ticks: 2, level: 0 },
                ],
                &[2, 4],
                0,
            ),
            Ok(Schedule {
                runs: vec![
                    Run { id: 1, start: 0, end: 2, finished: false, level: 0 },
                    Run { id: 2, start: 2, end: 4, finished: true, level: 0 },
                    Run { id: 1, start: 4, end: 8, finished: false, level: 1 },
                    Run { id: 1, start: 8, end: 9, finished: true, level: 1 },
                ],
                idle: vec![],
                completion: vec![(1, 9), (2, 4)],
                end: 9,
            })
        );
    }

    #[test]
    fn priority_and_idle() {
        assert_eq!(
            schedule(
                &[
                    Job { id: 3, arrival: 3, ticks: 1, level: 1 },
                    Job { id: 4, arrival: 3, ticks: 1, level: 0 },
                ],
                &[2, 3],
                0,
            ),
            Ok(Schedule {
                runs: vec![
                    Run { id: 4, start: 3, end: 4, finished: true, level: 0 },
                    Run { id: 3, start: 4, end: 5, finished: true, level: 1 },
                ],
                idle: vec![(0, 3)],
                completion: vec![(3, 5), (4, 4)],
                end: 5,
            })
        );
    }

    #[test]
    fn empty_priority_schedule() {
        assert_eq!(
            schedule(&[], &[1], 0),
            Ok(Schedule {
                runs: vec![],
                idle: vec![],
                completion: vec![],
                end: 0,
            })
        );
    }

    #[test]
    fn reject_invalid_level() {
        assert_eq!(
            schedule(
                &[Job { id: 5, arrival: 0, ticks: 1, level: 2 }],
                &[2, 3],
                0,
            ),
            Err(Error::InitialLevel { index: 0 })
        );
    }

    #[test]
    fn zero_tick_arrival_during_run_keeps_its_arrival_time() {
        let result = schedule(
            &[
                Job { id: 1, arrival: 0, ticks: 3, level: 0 },
                Job { id: 2, arrival: 1, ticks: 0, level: 0 },
            ],
            &[3],
            0,
        )
        .unwrap();

        assert_eq!(result.completion, vec![(1, 3), (2, 1)]);
        assert_eq!(result.runs.len(), 1);
    }
}

fn main() {}
