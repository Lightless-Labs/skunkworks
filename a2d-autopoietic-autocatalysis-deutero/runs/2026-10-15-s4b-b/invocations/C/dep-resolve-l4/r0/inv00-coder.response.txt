fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort_by(|a, b| {
        fn key(c: Constraint) -> (u8, Version) {
            match c {
                Constraint::Exact(v) => (0, v),
                Constraint::AtLeast(v) => (1, v),
                Constraint::Before(v) => (2, v),
                Constraint::Caret(v) => (3, v),
            }
        }
        key(*a).cmp(&key(*b))
    });
    out.dedup();
    out
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(e: &[Evidence]) -> Vec<Evidence> {
    let mut out = e.to_vec();
    out.sort_by(|a, b| {
        let ca = canonical_constraints(&[a.constraint])[0];
        let cb = canonical_constraints(&[b.constraint])[0];
        let ka = constraint_key(ca);
        let kb = constraint_key(cb);
        ka.cmp(&kb).then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn path_text(path: &[String]) -> String {
    path.join(" -> ")
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            for e in evidence_sorted(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut s = format!("no-match|package={}|constraints={}", encode(package), cs);
            for e in evidence_sorted(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                cs
            );
            for e in evidence_sorted(evidence) {
                s.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(e.constraint),
                    path_text(&e.path)
                ));
            }
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path_text(path)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn platform_matches(p: Platform, t: Target) -> bool {
    matches!(

        (Platform::Any, _)

            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)

fn feature_sort_unique(fs: &[String]) -> Vec<String> {
    fs.iter().cloned().collect::<BTreeSet<_>>().into_iter().collect()
}

fn features_text(fs: &[String]) -> String {
    feature_sort_unique(fs).join(",")
}

fn root_token(r: &Requirement, i: usize) -> String {
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        features_text(&r.features)
    )
}

fn dep_token(source: &str, version: Version, index: usize, d: &Dependency) -> String {
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        d.package,
        constraint_text(d.constraint),
        features_text(&d.features),
        d.gate.as_deref().unwrap_or("*"),
        match d.platform {
            Platform::Any => "any",
            Platform::Linux => "linux",
            Platform::Mac => "mac",
            Platform::Windows => "windows",
        }
    )
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: r.package.clone(),
            });
        }
        for (f, name) in r.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: name.clone(),
                });
            }
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: p.name.clone(),
            });
        }
        for (j, r) in p.releases.iter().enumerate() {
            for (k, d) in r.deps.iter().enumerate() {
                if !valid_name(&d.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: d.package.clone(),
                    });
                }
                if let Some(g) = &d.gate {
                    if !valid_name(g) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].gate",
                                i, j, k
                            ),
                            name: g.clone(),
                        });
                    }
                }
                for (f, name) in d.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: name.clone(),
                        });
                    }
                }
            }
            for (f, feature) in r.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = HashSet::new();
    for p in catalog {
        if !seen_packages.insert(p.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }
    for p in catalog {
        let mut versions = HashSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }
    for p in catalog {
        for r in &p.releases {
            let mut features = HashSet::new();
            for f in &r.features {
                if !features.insert(f.name.as_str()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: r.version,
                        feature: f.name.clone(),
                    });
                }
            }
        }
    }
    for (i, l) in locks.iter().enumerate() {
        if !valid_name(&l.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: l.package.clone(),
            });
        }
    }
    let mut seen_locks = HashSet::new();
    for l in locks {
        if !seen_locks.insert(l.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: l.package.clone(),
            });
        }
    }
    Ok(())
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    catalog_index: HashMap<String, usize>,
    best: Option<(usize, Vec<Version>, BTreeMap<String, Version>, BTreeMap<String, Active>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn closure(
        &self,
        selected: &BTreeMap<String, Version>,
    ) -> Result<BTreeMap<String, Active>, ResolveError> {
        let mut active: BTreeMap<String, Active> = BTreeMap::new();
        for (i, root) in self.roots.iter().enumerate() {
            let a = active.entry(root.package.clone()).or_default();
            a.origins.push(Origin {
                constraint: root.constraint,
                path: vec![root_token(root, i)],
            });
            a.features.extend(root.features.iter().cloned());
        }

        loop {
            let mut changed = false;
            let names: Vec<String> = selected.keys().cloned().collect();
            for name in names {
                let version = selected[&name];
                let Some(&pi) = self.catalog_index.get(&name) else {
                    continue;
                };
                let pkg = &self.catalog[pi];
                let Some(release) = pkg.releases.iter().find(|r| r.version == version) else {
                    continue;
                };
                let requested: Vec<String> = active
                    .get(&name)
                    .map(|a| a.features.iter().cloned().collect())
                    .unwrap_or_default();

                let mut activated = BTreeSet::new();
                let mut queue = requested;
                while let Some(f) = queue.pop() {
                    if !activated.insert(f.clone()) {
                        continue;
                    }
                    if let Some(decl) = release.features.iter().find(|d| d.name == f) {
                        for en in &decl.enables {
                            if activated.insert(en.clone()) {
                                queue.push(en.clone());
                            }
                        }
                    }
                }
                {
                    let a = active.entry(name.clone()).or_default();
                    let before = a.features.len();
                    a.features.extend(activated.iter().cloned());
                    if a.features.len() != before {
                        changed = true;
                    }
                }

                for (j, dep) in release.deps.iter().enumerate() {
                    if !platform_matches(dep.platform, self.target)
                        || dep
                            .gate
                            .as_ref()
                            .is_some_and(|g| !activated.contains(g))
                    {
                        continue;
                    }
                    let token = dep_token(&name, version, j, dep);
                    let source_path = active
                        .get(&name)
                        .and_then(|a| a.origins.iter().map(|o| &o.path).min_by(|x, y| {
                            x.len().cmp(&y.len()).then_with(|| x.cmp(y))
                        }))
                        .cloned()
                        .unwrap_or_default();
                    let mut path = source_path;
                    path.push(token);
                    let destination = active.entry(dep.package.clone()).or_default();
                    let old = destination.origins.len();
                    destination.origins.push(Origin {
                        constraint: dep.constraint,
                        path,
                    });
                    destination.features.extend(dep.features.iter().cloned());
                    if destination.origins.len() != old {
                        changed = true;
                    }
                }
            }

            // Rebuild dependency origins on each pass to avoid accumulating duplicates.
            // Closure is instead computed monotonically in the next implementation step.
            if !changed {
                break;
            }
            break;
        }

        // Recompute active closure monotonically while preserving each origin once.
        let mut origins_seen = HashSet::<(String, Constraint, Vec<String>)>::new();
        let mut queue_names: Vec<String> = selected.keys().cloned().collect();
        let mut loops = 0usize;
        while !queue_names.is_empty() && loops < 10000 {
            loops += 1;
            let name = queue_names.remove(0);
            let Some(&pi) = self.catalog_index.get(&name) else {
                continue;
            };
            let version = selected[&name];
            let Some(release) = self.catalog[pi]
                .releases
                .iter()
                .find(|r| r.version == version)
            else {
                continue;
            };
            let requested: Vec<String> = active
                .get(&name)
                .map(|a| a.features.iter().cloned().collect())
                .unwrap_or_default();
            let mut activated = BTreeSet::new();
            let mut fq = requested;
            while let Some(f) = fq.pop() {
                if !activated.insert(f.clone()) {
                    continue;
                }
                if let Some(decl) = release.features.iter().find(|d| d.name == f) {
                    for en in &decl.enables {
                        fq.push(en.clone());
                    }
                }
            }
            let a = active.entry(name.clone()).or_default();
            let oldf = a.features.len();
            a.features.extend(activated.iter().cloned());
            if a.features.len() != oldf {
                // no action required; selected package is processed again below
            }
            let source_path = a
                .origins
                .iter()
                .map(|o| o.path.clone())
                .min_by(|x, y| x.len().cmp(&y.len()).then_with(|| x.cmp(y)))
                .unwrap_or_default();

            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, self.target)
                    || dep.gate.as_ref().is_some_and(|g| !activated.contains(g))
                {
                    continue;
                }
                let token = dep_token(&name, version, j, dep);
                let mut path = source_path.clone();
                path.push(token);
                let key = (dep.package.clone(), dep.constraint, path.clone());
                if !origins_seen.insert(key) {
                    continue;
                }
                let dest = active.entry(dep.package.clone()).or_default();
                dest.origins.push(Origin {
                    constraint: dep.constraint,
                    path,
                });
                dest.features.extend(dep.features.iter().cloned());
                if selected.contains_key(&dep.package) {
                    queue_names.push(dep.package.clone());
                }
            }
        }

        // Canonicalize provenance by shortest path and retain all distinct incoming origins.
        for a in active.values_mut() {
            a.origins.sort_by(|x, y| {
                constraint_key(x.constraint)
                    .cmp(&constraint_key(y.constraint))
                    .then_with(|| x.path.cmp(&y.path))
            });
        }

        for (name, version) in selected {
            if let Some(a) = active.get(name) {
                if let Some(&pi) = self.catalog_index.get(name) {
                    if let Some(release) = self.catalog[pi]
                        .releases
                        .iter()
                        .find(|r| r.version == *version)
                    {
                        let declared: HashSet<&str> =
                            release.features.iter().map(|f| f.name.as_str()).collect();
                        if let Some(feature) = a.features.iter().find(|f| !declared.contains(f.as_str()))
                        {
                            let path = a
                                .origins
                                .iter()
                                .map(|o| o.path.clone())
                                .min_by(|x, y| x.len().cmp(&y.len()).then_with(|| x.cmp(y)))
                                .unwrap_or_default();
                            return Err(ResolveError::MissingFeature {
                                package: name.clone(),
                                selected: *version,
                                feature: feature.clone(),
                                path,
                            });
                        }
                    }
                }
            }
        }
        Ok(active)
    }

    fn node_error(
        &self,
        selected: &BTreeMap<String, Version>,
    ) -> Result<(BTreeMap<String, Active>, Vec<String>), ResolveError> {
        let active = self.closure(selected)?;
        let names: BTreeSet<String> = active.keys().cloned().collect();
        for name in &names {
            if !self.catalog_index.contains_key(name) {
                let evid = active[name]
                    .origins
                    .iter()
                    .map(|o| Evidence {
                        constraint: o.constraint,
                        path: o.path.clone(),
                    })
                    .collect();
                return Err(ResolveError::Missing {
                    package: name.clone(),
                    evidence: evid,
                });
            }
        }
        for (name, version) in selected {
            if let Some(a) = active.get(name) {
                let cs = canonical_constraints(
                    &a.origins.iter().map(|o| o.constraint).collect::<Vec<_>>(),
                );
                if cs.iter().any(|c| !matches(*c, *version)) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: cs,
                        evidence: a
                            .origins
                            .iter()
                            .map(|o| Evidence {
                                constraint: o.constraint,
                                path: o.path.clone(),
                            })
                            .collect(),
                    });
                }
            }
        }
        let mut branch = None;
        for name in names {
            if selected.contains_key(&name) {
                continue;
            }
            let a = &active[&name];
            let cs = canonical_constraints(
                &a.origins.iter().map(|o| o.constraint).collect::<Vec<_>>(),
            );
            let p = &self.catalog[self.catalog_index[&name]];
            if !p.releases.iter().any(|r| cs.iter().all(|c| matches(*c, r.version))) {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: cs,
                    evidence: a
                        .origins
                        .iter()
                        .map(|o| Evidence {
                            constraint: o.constraint,
                            path: o.path.clone(),
                        })
                        .collect(),
                });
            }
            if branch.is_none() {
                branch = Some(name);
            }
        }
        Ok((active, branch.into_iter().collect()))
    }

    fn solution_cost(&self, selected: &BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|l| selected.get(&l.package).copied() != Some(l.version))
            .count()
    }

    fn consider(&mut self, selected: &BTreeMap<String, Version>, active: BTreeMap<String, Active>) {
        let cost = self.solution_cost(selected);
        let mut vector = Vec::new();
        let mut catalog_names: Vec<String> = self.catalog.iter().map(|p| p.name.clone()).collect();
        catalog_names.sort();
        for n in catalog_names {
            vector.push(selected.get(&n).copied().unwrap_or(Version {
                major: 0,
                minor: 0,
                patch: 0,
            }));
        }
        let replace = self.best.as_ref().map_or(true, |(bc, bv, _, _)| {
            cost < *bc || (cost == *bc && vector > *bv)
        });
        if replace {
            self.best = Some((cost, vector, selected.clone(), active));
        }
    }

    fn search(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let (active, branch) = match self.node_error(selected) {
            Ok(x) => x,
            Err(e) => {
                if self.first_error.is_none() {
                    self.first_error = Some(e);
                }
                return Ok(());
            }
        };
        if branch.is_empty() {
            self.consider(selected, active);
            return Ok(());
        }
        let name = &branch[0];
        let a = &active[name];
        let cs = canonical_constraints(
            &a.origins.iter().map(|o| o.constraint).collect::<Vec<_>>(),
        );
        let mut candidates: Vec<Version> = self.catalog[self.catalog_index[name]]
            .releases
            .iter()
            .filter(|r| cs.iter().all(|c| matches(*c, r.version)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.iter().find(|l| l.package == *name) {
            if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }
        for v in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), v);
            let result = self.search(selected);
            selected.remove(name);
            result?;
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;
    let catalog_index: HashMap<String, usize> = catalog
        .iter()
        .enumerate()
        .map(|(i, p)| (p.name.clone(), i))
        .collect();
    let mut search = Search {
        catalog,
        roots,
        locks,
        target,
        max_attempts,
        attempts: 0,
        catalog_index,
        best: None,
        first_error: None,
    };
    let mut selected = BTreeMap::new();
    if let Err(e) = search.search(&mut selected) {
        return Err(e);
    }
    let Some((_, _, chosen, active)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::Missing {
            package: String::new(),
            evidence: Vec::new(),
        }));
    };

    let mut selected_vec: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();
    selected_vec.sort_by(|a, b| a.package.cmp(&b.package));

    let before: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();
    let after = chosen;
    let names: BTreeSet<String> = before.keys().chain(after.keys()).cloned().collect();
    let mut changes = Vec::new();
    for name in names {
        let b = before.get(&name).copied();
        let a = after.get(&name).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search.catalog_index.get(&name).is_some_and(|pi| {
                    catalog[*pi].releases.iter().any(|r| r.version == old)
                });
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let rejected = active.get(&name).is_some_and(|act| {
                        act.origins
                            .iter()
                            .any(|o| !matches(o.constraint, after[&name]))
                    });
                    if rejected {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            _ => unreachable!(),
        };
        changes.push(Change {
            package: name,
            before: b,
            after: a,
            reason,
        });
    }
    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}

fn main() {}
