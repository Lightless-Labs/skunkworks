#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

fn parse_fragment(bytes: &[u8], start: usize, end: usize, max: u32, step: bool) -> Result<u32, CronError> {
    if start == end {
        return Err(CronError::Syntax(start));
    }

    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0u32;
    for &b in &bytes[start..end] {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
            Some(v) => v,
            None => return Err(if step { CronError::Step(start) } else { CronError::Bounds(start) }),
        };
    }

    if value > max {
        return Err(if step { CronError::Step(start) } else { CronError::Bounds(start) });
    }
    Ok(value)
}

fn parse_field(text: &str, offset: usize, max: u32, selected: &mut [bool]) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut item_start = 0;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(CronError::Syntax(offset + item_start));
        }

        let item = &bytes[item_start..item_end];
        let slash_pos = item.iter().position(|&b| b == b'/');
        if let Some(slash) = slash_pos {
            if let Some(second) = item[slash + 1..].iter().position(|&b| b == b'/') {
                return Err(CronError::Syntax(offset + item_start + slash + 1 + second));
            }
        }

        let base_end = slash_pos.unwrap_or(item.len());
        let base = &item[..base_end];

        if let Some(slash) = slash_pos {
            if base != b"*" && !base.contains(&b'-') {
                return Err(CronError::Syntax(offset + item_start + slash));
            }
        }

        let (range_start, range_end) = if base == b"*" {
            (0, max)
        } else if let Some(dash) = base.iter().position(|&b| b == b'-') {
            let left = parse_fragment(
                bytes,
                offset + item_start,
                offset + item_start + dash,
                max,
                false,
            )?;
            let right = parse_fragment(
                bytes,
                offset + item_start + dash + 1,
                offset + item_start + base_end,
                max,
                false,
            )?;
            if left > right {
                return Err(CronError::Descending(offset + item_start + dash));
            }
            (left, right)
        } else {
            let n = parse_fragment(
                bytes,
                offset + item_start,
                offset + item_start + base_end,
                max,
                false,
            )?;
            (n, n)
        };

        let step = if let Some(slash) = slash_pos {
            parse_fragment(
                bytes,
                offset + item_start + slash + 1,
                offset + item_end,
                if max == 59 { 60 } else { 24 },
                true,
            )?
        } else {
            1
        };

        if step == 0 {
            return Err(CronError::Step(
                slash_pos
                    .map(|s| offset + item_start + s + 1)
                    .unwrap_or(offset + item_end),
            ));
        }

        let mut n = range_start;
        loop {
            if let Some(slot) = selected.get_mut(n as usize) {
                *slot = true;
            }
            if range_end - n < step {
                break;
            }
            n += step;
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(())
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let sep = spaces[0];
        if sep == 0 || sep + 1 == bytes.len() {
            return Err(CronError::Layout(sep));
        }

        let mut cron = Cron {
            minutes: [false; 60],
            hours: [false; 24],
        };

        parse_field(&text[..sep], 0, 59, &mut cron.minutes)?;
        parse_field(&text[sep + 1..], sep + 1, 23, &mut cron.hours)?;
        Ok(cron)
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

#[derive(Clone)]
struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
    added_after: Option<u64>,
}

pub struct Calendar {
    jobs: Vec<Job>,
    ids: Vec<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;

        self.ids.push(id.to_owned());
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order: self.jobs.len(),
            added_after: self.last_poll,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let first_minute = match self.last_poll {
            None => 0,
            Some(last) => last / 60_000 + 1,
        };
        let last_minute = now / 60_000;
        let now_is_boundary = now % 60_000 == 0;
        let mut out: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let start = match job.added_after {
                Some(added) => first_minute.max(added / 60_000 + 1),
                None => first_minute,
            };
            if start > last_minute {
                continue;
            }

            match job.policy {
                Policy::All => {
                    let mut minute = start;
                    loop {
                        if job.cron.matches(minute) {
                            out.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                        if minute == last_minute {
                            break;
                        }
                        minute += 1;
                    }
                }
                Policy::Once => {
                    let mut minute = last_minute;
                    loop {
                        if minute < start {
                            break;
                        }
                        if job.cron.matches(minute) {
                            out.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                            break;
                        }
                        if minute == 0 {
                            break;
                        }
                        minute -= 1;
                    }
                }
                Policy::Skip => {
                    if now_is_boundary && now / 60_000 >= start && job.cron.matches(now / 60_000) {
                        out.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: now,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
            }
        }

        out.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(out.into_iter().map(|entry| entry.0).collect())
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn parse_ranges_steps_and_unions() {
        let cron = Cron::parse("5-15/5,1,1 2,4").unwrap();
        assert!(cron.matches(130));
        assert!(cron.matches(250));
        assert!(!cron.matches(131));
    }

    #[test]
    fn parse_errors_are_safe_and_ordered() {
        assert_eq!(Cron::parse(""), Err(CronError::Layout(0)));
        assert_eq!(Cron::parse("1 2 3"), Err(CronError::Layout(3)));
        assert_eq!(Cron::parse("1, 2"), Err(CronError::Layout(2)));
        assert_eq!(Cron::parse("1/2 0"), Err(CronError::Syntax(1)));
        assert_eq!(Cron::parse("1-2/0 0"), Err(CronError::Step(4)));
        assert_eq!(Cron::parse("1--2 0"), Err(CronError::Syntax(3)));
    }

    #[test]
    fn all_once_skip_and_watermark() {
        let mut calendar = Calendar::new();
        calendar.add("all", "*/1 0-23", 1, Policy::All).unwrap();
        calendar.add("once", "*/1 0-23", 2, Policy::Once).unwrap();
        calendar.add("skip", "*/1 0-23", 3, Policy::Skip).unwrap();

        let got = calendar.poll(90_000).unwrap();
        assert!(got.iter().any(|o| o.id == "all" && o.at == 0));
        assert!(got.iter().any(|o| o.id == "once" && o.at == 60_000));
        assert!(!got.iter().any(|o| o.id == "skip"));
        assert!(calendar.poll(90_000).unwrap().is_empty());
    }

    #[test]
    fn add_precedence_and_poll_errors() {
        let mut calendar = Calendar::new();
        calendar.add("one", "0 0", 1, Policy::All).unwrap();
        assert_eq!(
            calendar.add("one", "bad", 10, Policy::All),
            Err(Error::BadPriority)
        );
        assert_eq!(
            calendar.add("one", "bad", 1, Policy::All),
            Err(Error::Duplicate)
        );
        calendar.poll(10).unwrap();
        assert_eq!(calendar.poll(9), Err(Error::TimeReversed));
        assert_eq!(calendar.poll(604_800_001), Err(Error::TimeRange));
    }
}
