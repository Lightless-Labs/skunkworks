pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],

fn parse_number(bytes: &[u8], start: usize, end: usize, limit: u32, step: bool) -> Result<u32, CronError> {
    if start >= end {
        return Err(CronError::Syntax(start));
    }

    let mut i = start;
    while i < end {
        if bytes.get(i).is_none_or(|b| !b.is_ascii_digit()) {
            return Err(CronError::Syntax(i));
        }
        i += 1;
    }

    if end - start > 1 && bytes.get(start) == Some(&b'0') {
        return Err(CronError::Number(start));
    }

    let mut value = 0u32;
    i = start;
    while i < end {
        let digit = match bytes.get(i) {
            Some(b) => (b - b'0') as u32,
            None => return Err(CronError::Syntax(i)),
        };
        value = match value.checked_mul(10).and_then(|v| v.checked_add(digit)) {
            Some(v) => v,
            None => return Err(if step { CronError::Step(start) } else { CronError::Bounds(start) }),
        };
        i += 1;
    }

    if value > limit {
        return Err(if step { CronError::Step(start) } else { CronError::Bounds(start) });
    }
    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    start: usize,
    end: usize,
    offset: usize,
    max: u32,
    step_max: u32,
    selected: &mut [bool],
) -> Result<(), CronError> {
    if start == end {
        return Err(CronError::Syntax(offset + start));
    }

    let mut slash = None;
    let mut i = start;
    while i < end {
        if bytes.get(i) == Some(&b'/') {
            if slash.is_some() {
                return Err(CronError::Syntax(offset + i));
            }
            slash = Some(i);
        }
        i += 1;
    }

    let base_end = slash.unwrap_or(end);
    if let Some(slash_at) = slash {
        let is_star = base_end == start + 1 && bytes.get(start) == Some(&b'*');
        let has_dash = bytes.get(start..base_end).is_some_and(|s| s.contains(&b'-'));
        if !is_star && !has_dash {
            return Err(CronError::Syntax(offset + slash_at));
        }
    }

    let (range_start, range_end) = if base_end == start + 1 && bytes.get(start) == Some(&b'*') {
        (0, max)
    } else {
        let mut dash = None;
        i = start;
        while i < base_end {
            if bytes.get(i) == Some(&b'-') {
                dash = Some(i);
                break;
            }
            i += 1;
        }

        if let Some(dash_at) = dash {
            let left = parse_number(bytes, start, dash_at, max, false)
                .map_err(|e| adjust_error(e, offset))?;
            let right = parse_number(bytes, dash_at + 1, base_end, max, false)
                .map_err(|e| adjust_error(e, offset))?;
            if left > right {
                return Err(CronError::Descending(offset + dash_at));
            }
            (left, right)
        } else {
            let n = parse_number(bytes, start, base_end, max, false)
                .map_err(|e| adjust_error(e, offset))?;
            (n, n)
        }
    };

    let step = if let Some(slash_at) = slash {
        parse_number(bytes, slash_at + 1, end, step_max, true)
            .map_err(|e| adjust_error(e, offset))?
    } else {
        1
    };

    let mut value = range_start;
    loop {
        if let Some(slot) = selected.get_mut(value as usize) {
            *slot = true;
        }
        if range_end - value < step {
            break;
        }
        value += step;
    }
    Ok(())
}

fn adjust_error(error: CronError, offset: usize) -> CronError {
    match error {
        CronError::Layout(i) => CronError::Layout(offset + i),
        CronError::Syntax(i) => CronError::Syntax(offset + i),
        CronError::Number(i) => CronError::Number(offset + i),
        CronError::Bounds(i) => CronError::Bounds(offset + i),
        CronError::Descending(i) => CronError::Descending(offset + i),
        CronError::Step(i) => CronError::Step(offset + i),
    }
}

fn parse_field(
    text: &str,
    offset: usize,
    max: u32,
    step_max: u32,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut start = 0usize;
    loop {
        let end = match bytes.get(start..).and_then(|s| s.iter().position(|&b| b == b',')) {
            Some(relative) => start + relative,
            None => bytes.len(),
        };
        parse_item(bytes, start, end, offset, max, step_max, selected)?;
        if end == bytes.len() {
            break;
        }
        start = end + 1;
    }
    Ok(())
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return spaces.get(1).copied().map(CronError::Layout).ok_or(CronError::Layout(text.len()));
        }

        let sep = match spaces.first().copied() {
            Some(s) => s,
            None => return Err(CronError::Layout(text.len())),
        };
        if sep == 0 || sep + 1 == bytes.len() {
            return Err(CronError::Layout(sep));
        }

        let mut cron = Self {
            minutes: [false; 60],
            hours: [false; 24],
        };
        parse_field(&text[..sep], 0, 59, 60, &mut cron.minutes)?;
        parse_field(&text[sep + 1..], sep + 1, 23, 24, &mut cron.hours)?;
        Ok(cron)
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes.get((minute % 60) as usize).copied().unwrap_or(false)
            && self.hours.get(((minute / 60) % 24) as usize).copied().unwrap_or(false)
    }
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id.bytes().all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let order = self.jobs.len();
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order,
            added_after: self.last_poll,
        });
        self.ids.push(id.to_owned());
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let first_minute = match self.last_poll {
            None => 0,
            Some(last) => last / 60_000 + 1,
        };
        let last_minute = now / 60_000;
        let boundary_now = now % 60_000 == 0;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let start = match job.added_after {
                Some(added) => first_minute.max(added / 60_000 + 1),
                None => first_minute,
            };
            if start > last_minute {
                continue;
            }

            match job.policy {
                Policy::All => {
                    let mut minute = start;
                    loop {
                        if job.cron.matches(minute) {
                            emitted.push((
                                Occurrence { id: job.id.clone(), at: minute * 60_000 },
                                job.priority,
                                job.order,
                            ));
                        }
                        if minute == last_minute {
                            break;
                        }
                        minute += 1;
                    }
                }
                Policy::Once => {
                    let mut minute = last_minute;
                    loop {
                        if minute < start {
                            break;
                        }
                        if job.cron.matches(minute) {
                            emitted.push((
                                Occurrence { id: job.id.clone(), at: minute * 60_000 },
                                job.priority,
                                job.order,
                            ));
                            break;
                        }
                        if minute == 0 {
                            break;
                        }
                        minute -= 1;
                    }
                }
                Policy::Skip => {
                    if boundary_now && now / 60_000 >= start && job.cron.matches(now / 60_000) {
                        emitted.push((
                            Occurrence { id: job.id.clone(), at: now },
                            job.priority,
                            job.order,
                        ));
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|entry| entry.0).collect())
    }
}
