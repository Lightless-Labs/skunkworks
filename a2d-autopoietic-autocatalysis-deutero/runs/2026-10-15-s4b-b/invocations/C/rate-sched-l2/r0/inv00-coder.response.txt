#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    limit: u32,
    error_kind: u8,
) -> Result<u32, CronError> {
    if start == end {
        return Err(CronError::Syntax(start));
    }
    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value: u32 = 0;
    for i in start..end {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((bytes[i] - b'0') as u32))
            .unwrap_or(u32::MAX);
    }
    if value > limit {
        return Err(if error_kind == 0 {
            CronError::Bounds(start)
        } else {
            CronError::Step(start)
        });
    }
    Ok(value)
}

fn parse_field(
    text: &str,
    offset: usize,
    limit: u32,
    selected: &mut [bool],
    is_minute: bool,
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut item_start = 0;

    while item_start <= bytes.len() {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(CronError::Syntax(offset + item_start));
        }

        let item = &bytes[item_start..item_end];
        let slash = item.iter().position(|&b| b == b'/');
        if let Some(s) = slash {
            if let Some(second) = item[s + 1..].iter().position(|&b| b == b'/') {
                return Err(CronError::Syntax(offset + item_start + s + 1 + second));
            }
        }

        let base_end = slash.unwrap_or(item.len());
        let base = &item[..base_end];
        let step_start = slash.map(|s| offset + item_start + s + 1);

        let mut range_start: u32;
        let mut range_end: u32;

        if base == b"*" {
            range_start = 0;
            range_end = limit;
        } else {
            if let Some(s) = slash {
                if !base.contains(&b'-') {
                    return Err(CronError::Syntax(offset + item_start + s));
                }
            }

            if let Some(dash) = base.iter().position(|&b| b == b'-') {
                let left = parse_number(
                    bytes,
                    offset + item_start,
                    offset + item_start + dash,
                    limit,
                    0,
                )?;
                let right = parse_number(
                    bytes,
                    offset + item_start + dash + 1,
                    offset + item_start + base_end,
                    limit,
                    0,
                )?;
                if left > right {
                    return Err(CronError::Descending(offset + item_start + dash));
                }
                range_start = left;
                range_end = right;
            } else {
                let n = parse_number(
                    bytes,
                    offset + item_start,
                    offset + item_start + base_end,
                    limit,
                    0,
                )?;
                range_start = n;
                range_end = n;
            }
        }

        let step = if let Some(start) = step_start {
            parse_number(
                bytes,
                start,
                offset + item_end,
                if is_minute { 60 } else { 24 },
                1,
            )?
        } else {
            1
        };

        if step == 0 {
            return Err(CronError::Step(step_start.unwrap_or(offset + item_end)));
        }

        let mut n = range_start;
        loop {
            selected[n as usize] = true;
            if range_end - n < step {
                break;
            }
            n += step;
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(())
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let sep = spaces[0];
        if sep == 0 || sep + 1 == bytes.len() {
            return Err(CronError::Layout(sep));
        }

        let mut cron = Cron {
            minutes: [false; 60],
            hours: [false; 24],
        };
        parse_field(&text[..sep], 0, 59, &mut cron.minutes, true)?;
        parse_field(&text[sep + 1..], sep + 1, 23, &mut cron.hours, false)?;
        Ok(cron)
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

#[derive(Clone)]
struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
    added_after: Option<u64>,
}

pub struct Calendar {
    jobs: Vec<Job>,
    ids: Vec<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;

        self.ids.push(id.to_owned());
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order: self.jobs.len(),
            added_after: self.last_poll,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower = self.last_poll.unwrap_or(0);
        let first_minute = if self.last_poll.is_none() {
            0
        } else {
            lower / 60_000 + 1
        };
        let last_minute = now / 60_000;
        let now_is_boundary = now % 60_000 == 0;
        let mut out: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let start = if let Some(added) = job.added_after {
                first_minute.max(added / 60_000 + 1)
            } else {
                first_minute
            };

            if start > last_minute {
                continue;
            }

            match job.policy {
                Policy::All => {
                    let mut minute = start;
                    loop {
                        if job.cron.matches(minute) {
                            out.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                        if minute == last_minute {
                            break;
                        }
                        minute += 1;
                    }
                }
                Policy::Once => {
                    let mut minute = last_minute;
                    loop {
                        if minute < start {
                            break;
                        }
                        if job.cron.matches(minute) {
                            out.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                            break;
                        }
                        if minute == 0 {
                            break;
                        }
                        minute -= 1;
                    }
                }
                Policy::Skip => {
                    if now_is_boundary
                        && now / 60_000 >= start
                        && job.cron.matches(now / 60_000)
                    {
                        out.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: now,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
            }
        }

        out.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(out.into_iter().map(|entry| entry.0).collect())
    }
}

fn main() {}
