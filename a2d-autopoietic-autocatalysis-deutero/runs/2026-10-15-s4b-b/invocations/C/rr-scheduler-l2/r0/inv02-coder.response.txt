#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&i| (jobs[i].arrival, i));

    let mut runs: Vec<Run> = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut completion: Vec<(u32, u64)> = vec![(0, 0); jobs.len()];
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut ready: Vec<usize> = Vec::new();
    let mut completed: Vec<bool> = vec![false; jobs.len()];
    let mut completed_count: usize = 0;
    let mut event_pos: usize = 0;
    let mut time: u64 = 0;
    let mut last_id: Option<u32> = None;

    while completed_count < jobs.len() {
        while event_pos < events.len() && jobs[events[event_pos]].arrival as u64 <= time {
            let i = events[event_pos];
            event_pos += 1;
            if jobs[i].ticks == 0 {
                completion[i] = (jobs[i].id, jobs[i].arrival as u64);
                completed[i] = true;
                completed_count += 1;
            } else {
                ready.push(i);
            }
        }

        if ready.is_empty() {
            if event_pos >= events.len() {
                break;
            }
            let next_time = jobs[events[event_pos]].arrival as u64;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_id = None;
            }
            continue;
        }

        let i = ready.remove(0);
        let id = jobs[i].id;

        if last_id.is_some_and(|previous| previous != id) {
            time += switch_cost as u64;
        }

        while event_pos < events.len() && jobs[events[event_pos]].arrival as u64 <= time {
            let arriving = events[event_pos];
            event_pos += 1;
            if jobs[arriving].ticks == 0 {
                completion[arriving] = (jobs[arriving].id, jobs[arriving].arrival as u64);
                completed[arriving] = true;
                completed_count += 1;
            } else {
                ready.push(arriving);
            }
        }

        let start = time;
        let consumed = remaining[i].min(quantum);
        time += consumed as u64;
        remaining[i] -= consumed;
        let finished = remaining[i] == 0;

        runs.push(Run {
            id,
            start,
            end: time,
            finished,
        });

        while event_pos < events.len() && jobs[events[event_pos]].arrival as u64 <= time {
            let arriving = events[event_pos];
            event_pos += 1;
            if jobs[arriving].ticks == 0 {
                completion[arriving] = (jobs[arriving].id, jobs[arriving].arrival as u64);
                completed[arriving] = true;
                completed_count += 1;
            } else {
                ready.push(arriving);
            }
        }

        if finished {
            completion[i] = (id, time);
            completed[i] = true;
            completed_count += 1;
        } else {
            ready.push(i);
        }

        last_id = Some(id);
    }

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn arrivals_rotate() {
        assert_eq!(
            schedule(
                &[
                    Job {
                        id: 1,
                        arrival: 0,
                        ticks: 3,
                    },
                    Job {
                        id: 2,
                        arrival: 1,
                        ticks: 2,
                    },
                ],
                2,
                0
            ),
            Ok(Schedule {
                runs: vec![
                    Run {
                        id: 1,
                        start: 0,
                        end: 2,
                        finished: false,
                    },
                    Run {
                        id: 2,
                        start: 2,
                        end: 4,
                        finished: true,
                    },
                    Run {
                        id: 1,
                        start: 4,
                        end: 5,
                        finished: true,
                    },
                ],
                idle: vec![],
                completion: vec![(1, 5), (2, 4)],
                end: 5,
            })
        );
    }

    #[test]
    fn idle_before_arrival() {
        assert_eq!(
            schedule(
                &[Job {
                    id: 5,
                    arrival: 4,
                    ticks: 1,
                }],
                3,
                0
            ),
            Ok(Schedule {
                runs: vec![Run {
                    id: 5,
                    start: 4,
                    end: 5,
                    finished: true,
                }],
                idle: vec![(0, 4)],
                completion: vec![(5, 5)],
                end: 5,
            })
        );
    }

    #[test]
    fn dispatch_gap() {
        assert_eq!(
            schedule(
                &[
                    Job {
                        id: 6,
                        arrival: 0,
                        ticks: 1,
                    },
                    Job {
                        id: 8,
                        arrival: 0,
                        ticks: 1,
                    },
                ],
                2,
                2
            ),
            Ok(Schedule {
                runs: vec![
                    Run {
                        id: 6,
                        start: 0,
                        end: 1,
                        finished: true,
                    },
                    Run {
                        id: 8,
                        start: 3,
                        end: 4,
                        finished: true,
                    },
                ],
                idle: vec![],
                completion: vec![(6, 1), (8, 4)],
                end: 4,
            })
        );
    }

    #[test]
    fn reject_arrival() {
        assert_eq!(
            schedule(
                &[Job {
                    id: 1,
                    arrival: 1001,
                    ticks: 1,
                }],
                2,
                0
            ),
            Err(Error::Arrival { index: 0 })
        );
    }
}

fn main() {}
