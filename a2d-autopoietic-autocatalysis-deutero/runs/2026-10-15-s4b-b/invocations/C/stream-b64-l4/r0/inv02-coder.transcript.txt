                }
                FrameState::NeedSymbol { .. } => {
                    if byte == b'\r' || byte == b'\n' || matches!(byte, 9 | 11 | 12 | 32) {
                        return Err(Error::BadWrap { offset });
                    }
                    self.frame_state = FrameState::Row;
                    self.row_count = 0;
                    Ok(true)
                }
                FrameState::Row => {
                    if self.row_count == width {
                        match ending {
                            LineEnding::Lf if byte == b'\n' => {
                                self.frame_state = FrameState::NeedSymbol {
                                    first_offset: offset,
                                };
                                Ok(false)
                            }
                            LineEnding::CrLf if byte == b'\r' => {
                                self.frame_state = FrameState::NeedLf { cr_offset: offset };
                                Ok(false)
                            }
                            _ => Err(Error::BadWrap { offset }),
                        }
                    } else if byte == b'\r'
                        || byte == b'\n'
                        || matches!(byte, 9 | 11 | 12 | 32)
                    {
                        Err(Error::BadWrap { offset })
                    } else {
                        self.row_count += 1;
                        Ok(true)
                    }
                }
            },
        }
    }

    fn process_symbol(&mut self, byte: u8, offset: usize, out: &mut Vec<u8>) -> Result<(), Error> {
        if let Some(limit) = self.options.max_symbols {
            if self.symbols_seen >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.symbols_seen += 1;

        if self.terminal_pad {
            return Err(Error::BadPadding { offset });
        }
        if let Some(pad_offset) = self.pending_second_pad.take() {
            if byte != b'=' {
                return Err(Error::BadPadding { offset });
            }
            self.group.push(Sym { byte, offset, value: 0 });
            self.terminal_pad = true;
            let q: [Sym; 4] = self.group.clone().try_into().unwrap();
            out.extend(decode_complete_group(&q, self.options, true)?);
            let _ = pad_offset;
            self.group.clear();
            return Ok(());
        }

        let pos = self.group.len();
        if byte == b'=' {
            if pos < 2 {
                return Err(Error::BadPadding { offset });
            }
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            if pos == 2 {
                self.group.push(Sym { byte, offset, value: 0 });
                self.pending_second_pad = Some(offset);
                return Ok(());
            }
            self.group.push(Sym { byte, offset, value: 0 });
            self.terminal_pad = true;
            let q: [Sym; 4] = self.group.clone().try_into().unwrap();
            out.extend(decode_complete_group(&q, self.options, true)?);
            self.group.clear();
            return Ok(());
        }

        let value = alphabet_value(self.options, byte, offset)?;
        self.group.push(Sym { byte, offset, value });
        if self.group.len() == 4 {
            let q: [Sym; 4] = self.group.clone().try_into().unwrap();
            out.extend(decode_complete_group(&q, self.options, false)?);
            self.group.clear();
        }
        Ok(())
    }

    fn fail<T>(&mut self, error: Error) -> Result<T, Error> {
        self.sticky = Some(error.clone());
        Err(error)
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(e) = &self.sticky {
            return Err(e.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut out = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;
            match self.frame_byte(byte, offset) {
                Ok(false) => continue,
                Ok(true) => {
                    if let Err(e) = self.process_symbol(byte, offset, &mut out) {
                        return self.fail(e);
                    }
                }
                Err(e) => return self.fail(e),
            }
        }
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(e) = &self.sticky {
            return Err(e.clone());
        }
        if self.closed {
            return Err(Error::AlreadyFinished);
        }

        if let Whitespace::Wrapped { .. } = self.options.whitespace {
            match self.frame_state {
                FrameState::NeedLf { cr_offset } => {
                    return self.fail(Error::BadWrap { offset: cr_offset });
                }
                FrameState::NeedSymbol { first_offset } => {
                    return self.fail(Error::BadWrap { offset: first_offset });
                }
                FrameState::Row => {}
            }
        }

        if let Some(offset) = self.pending_second_pad {
            return self.fail(Error::BadPadding { offset });
        }

        let mut out = Vec::new();
        match self.group.len() {
            0 => {}
            1 => return self.fail(Error::Truncated { offset: self.group[0].offset }),
            2 | 3 if self.options.padding != Padding::Required => {
                if self.group.len() == 2 {
                    if self.options.strict && self.group[1].value & 15 != 0 {
                        return self.fail(Error::NonCanonical { offset: self.group[1].offset });
                    }
                    out.push((self.group[0].value << 2) | (self.group[1].value >> 4));
                } else {
                    if self.options.strict && self.group[2].value & 3 != 0 {
                        return self.fail(Error::NonCanonical { offset: self.group[2].offset });
                    }
                    out.push((self.group[0].value << 2) | (self.group[1].value >> 4));
                    out.push((self.group[1].value << 4) | (self.group[2].value >> 2));
                }
            }
            _ => return self.fail(Error::Truncated { offset: self.group[0].offset }),
        }

        self.closed = true;
        Ok(out)
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new(options)?;
    let mut out = decoder.push(input)?;
    out.extend(decoder.finish()?);
    Ok(out)
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    closed: bool,
    emitted_in_row: usize,
    pending_separator: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            tail: Vec::with_capacity(2),
            closed: false,
            emitted_in_row: 0,
            pending_separator: false,
        })
    }

    fn emit_symbols(&mut self, symbols: &[u8], out: &mut String) {
        for &b in symbols {
            if let Whitespace::Wrapped { width, ending } = self.options.whitespace {
                if self.pending_separator {
                    match ending {
                        LineEnding::Lf => out.push('\n'),
                        LineEnding::CrLf => out.push_str("\r\n"),
                    }
                    self.pending_separator = false;
                    self.emitted_in_row = 0;
                }
                out.push(b as char);
                self.emitted_in_row += 1;
                if self.emitted_in_row == width {
                    self.pending_separator = true;
                }
            } else {
                out.push(b as char);
            }
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        let mut combined = Vec::with_capacity(self.tail.len() + input.len());
        combined.extend_from_slice(&self.tail);
        combined.extend_from_slice(input);
        let complete_len = combined.len() / 3 * 3;
        let symbols = encode_symbols(&combined[..complete_len], self.options);
        let mut out = String::new();
        self.emit_symbols(&symbols, &mut out);
        self.tail.clear();
        self.tail.extend_from_slice(&combined[complete_len..]);
        Ok(out)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.closed {
            return Err(Error::AlreadyFinished);
        }
        self.closed = true;
        let symbols = encode_symbols(&self.tail, self.options);
        self.tail.clear();
        let mut out = String::new();
        self.emit_symbols(&symbols, &mut out);
        Ok(out)
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn opts() -> Options {
        Options {
            alphabet: Alphabet::Standard,
            padding: Padding::Required,
            whitespace: Whitespace::None,
            strict: false,
            max_symbols: None,
        }
    }

    #[test]
    fn encode_and_decode_basics() {
        let o = opts();
        assert_eq!(encode(b"Cat", o), Ok("Q2F0".to_string()));
        assert_eq!(decode(b"T2s=", o), Ok(b"Ok".to_vec()));
        assert_eq!(decode(b"", o), Ok(Vec::new()));
    }

    #[test]
    fn optional_unpadded_and_strict_canonicality() {
        let mut o = opts();
        o.padding = Padding::Optional;
        assert_eq!(decode(b"Zg", o), Ok(b"f".to_vec()));
        o.strict = true;
        assert_eq!(decode(b"Zh==", o), Err(Error::NonCanonical { offset: 1 }));
    }

    #[test]
    fn sticky_decoder_errors_and_offsets() {
        let mut d = Decoder::new(opts()).unwrap();
        let error = d.push(b"YWJj!").unwrap_err();
        assert_eq!(error, Error::InvalidByte { offset: 4, byte: b'!' });
        assert_eq!(d.finish(), Err(error.clone()));
        assert_eq!(d.push(b""), Err(error));
    }

    #[test]
    fn wrapped_encoder_has_no_trailing_separator() {
        let mut o = opts();
        o.whitespace = Whitespace::Wrapped {
            width: 4,
            ending: LineEnding::Lf,
        };
        let mut e = Encoder::new(o).unwrap();
        assert_eq!(e.push(b"Cat"), Ok("Q2F0".to_string()));
        assert_eq!(e.finish(), Ok(String::new()));
    }
}
