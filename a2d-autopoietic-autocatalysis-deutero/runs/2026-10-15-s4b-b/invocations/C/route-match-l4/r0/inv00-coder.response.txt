#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    has_wildcard: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b[0].is_ascii_lowercase()
        && b[1..]
            .iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_' || *c == b'-')
}

fn valid_capture_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b[0].is_ascii_lowercase()
        && b[1..]
            .iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(s: &str, base: usize, pattern: bool) -> Result<String, usize> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        if b[i] == b'%' {
            if i + 2 >= b.len() {
                return Err(base + i);
            }
            let (Some(h), Some(l)) = (hex_upper(b[i + 1]), hex_upper(b[i + 2])) else {
                return Err(base + i);
            };
            let v = (h << 4) | l;
            if !(0x20..=0x7e).contains(&v) || v == b'/' || unreserved(v) {
                return Err(base + i);
            }
            out.push(v);
            i += 3;
        } else if unreserved(b[i]) {
            out.push(b[i]);
            i += 1;
        } else {
            let _ = pattern;
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn parse_path(input: &str, is_pattern: bool) -> Result<(Vec<Segment>, bool), usize> {
    let b = input.as_bytes();
    if b.first() != Some(&b'/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut parts: Vec<(usize, &str)> = Vec::new();
    let mut start = 1;
    for i in 1..b.len() {
        if b[i] == b'/' {
            if i == start {
                return Err(start);
            }
            parts.push((start, &input[start..i]));
            start = i + 1;
        }
    }

    let trailing = b.last() == Some(&b'/');
    if trailing {
        if start == b.len() {
            // The final slash is the permitted trailing slash.
        } else {
            parts.push((start, &input[start..]));
        }
    } else {
        if start == b.len() {
            return Err(start);
        }
        parts.push((start, &input[start..]));
    }

    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();

    for (idx, raw) in parts.iter().copied().enumerate() {
        let (at, text) = raw;
        if is_pattern && (text.starts_with(':') || text.starts_with('*')) {
            let marker = text.as_bytes()[0];
            let name = &text[1..];
            if !valid_capture_name(name) || !names.insert(name.to_string()) {
                return Err(at);
            }
            if marker == b'*' && idx + 1 != parts.len() {
                return Err(at);
            }
            segments.push(if marker == b':' {
                Segment::Param(name.to_string())
            } else {
                Segment::Wildcard(name.to_string())
            });
        } else {
            let decoded = decode_segment(text, at, is_pattern)?;
            segments.push(Segment::Static(decoded));
        }
    }
    Ok((segments, trailing))
}

fn shape_conflict(a: &Route, b: &Route, policy: SlashPolicy) -> bool {
    if a.method != b.method || a.segments.len() != b.segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b.trailing {
        return false;
    }
    a.segments.iter().zip(&b.segments).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) =
            parse_path(pattern, true).map_err(|at| Error::InvalidPattern { at })?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        let has_wildcard = segments.iter().any(|s| matches!(s, Segment::Wildcard(_)));
        let mut rank = Vec::new();
        for seg in &segments {
            rank.push(match seg {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
            if matches!(seg, Segment::Wildcard(_)) {
                break;
            }
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        let route = Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            segments,
            trailing,
            has_wildcard,
            rank,
        };
        if let Some(existing) = self.routes.iter().find(|r| shape_conflict(r, &route, self.policy)) {
            return Err(Error::Conflict { with: existing.id });
        }
        self.routes.push(route);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let cut = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..cut];
        let (request_segments, request_trailing) =
            parse_path(head, false).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            let mut captures = Vec::new();
            let mut ok = true;
            let mut i = 0;
            for seg in &route.segments {
                match seg {
                    Segment::Static(s) => {
                        if request_segments.get(i) != Some(&Segment::Static(s.clone())) {
                            ok = false;
                            break;
                        }
                        i += 1;
                    }
                    Segment::Param(name) => {
                        match request_segments.get(i) {
                            Some(Segment::Static(value)) => captures.push((name.clone(), value.clone())),
                            _ => {
                                ok = false;
                                break;
                            }
                        }
                        i += 1;
                    }
                    Segment::Wildcard(name) => {
                        let rest = request_segments[i..]
                            .iter()
                            .map(|s| match s {
                                Segment::Static(v) => v.as_str(),
                                _ => "",
                            })
                            .collect::<Vec<_>>();
                        if rest.iter().any(|v| v.is_empty()) {
                            ok = false;
                            break;
                        }
                        captures.push((name.clone(), rest.join("/")));
                        i = request_segments.len();
                    }
                }
            }
            if ok && i == request_segments.len() {
                candidates.push((route, captures));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<usize> = if method == "HEAD" {
            if candidates.iter().any(|(r, _)| r.method == "HEAD") {
                candidates
                    .iter()
                    .enumerate()
                    .filter_map(|(i, (r, _))| (r.method == "HEAD").then_some(i))
                    .collect()
            } else {
                candidates
                    .iter()
                    .enumerate()
                    .filter_map(|(i, (r, _))| (r.method == "GET").then_some(i))
                    .collect()
            }
        } else {
            candidates
                .iter()
                .enumerate()
                .filter_map(|(i, (r, _))| (r.method == method).then_some(i))
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed = std::collections::BTreeSet::new();
            for (r, _) in &candidates {
                allowed.insert(r.method.clone());
                if r.method == "GET" {
                    allowed.insert("HEAD".to_string());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        let best = *eligible
            .iter()
            .max_by(|&&a, &&b| {
                candidates[b]
                    .0
                    .rank
                    .cmp(&candidates[a].0.rank)
                    .then_with(|| candidates[b].0.id.cmp(&candidates[a].0.id))
            })
            .unwrap();
        let (route, params) = &candidates[best];
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != request_trailing
            && !head.is_empty()
            && head != "/"
        {
            let mut target = head.to_string();
            if route.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };
        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|r| r.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_string(),
                });
            }
            supplied.insert(*key, *value);
        }

        let mut capture_names = Vec::new();
        for seg in &route.segments {
            match seg {
                Segment::Param(n) | Segment::Wildcard(n) => capture_names.push(n.as_str()),
                _ => {}
            }
        }
        for key in &capture_names {
            if !supplied.contains_key(*key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }
        let mut extras: Vec<&str> = supplied
            .keys()
            .copied()
            .filter(|k| !capture_names.contains(k))
            .collect();
        extras.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for seg in &route.segments {
            if let Segment::Param(n) | Segment::Wildcard(n) = seg {
                let value = supplied[n.as_str()];
                if !value.bytes().all(|b| (0x20..=0x7e).contains(&b)) {
                    return Err(Error::InvalidValue { name: n.clone() });
                }
                match seg {
                    Segment::Param(_) => {
                        if value.is_empty() || value.contains('/') {
                            return Err(Error::InvalidValue { name: n.clone() });
                        }
                    }
                    Segment::Wildcard(_) => {
                        if !value.is_empty()
                            && (value.starts_with('/')
                                || value.ends_with('/')
                                || value.contains("//"))
                        {
                            return Err(Error::InvalidValue { name: n.clone() });
                        }
                    }
                    _ => {}
                }
            }
        }

        fn encode_component(s: &str) -> String {
            let mut out = String::new();
            for b in s.bytes() {
                if unreserved(b) {
                    out.push(b as char);
                } else {
                    const HEX: &[u8; 16] = b"0123456789ABCDEF";
                    out.push('%');
                    out.push(HEX[(b >> 4) as usize] as char);
                    out.push(HEX[(b & 15) as usize] as char);
                }
            }
            out
        }

        let mut pieces = Vec::new();
        for seg in &route.segments {
            match seg {
                Segment::Static(s) => pieces.push(encode_component(s)),
                Segment::Param(n) => pieces.push(encode_component(supplied[n.as_str()])),
                Segment::Wildcard(n) => {
                    let value = supplied[n.as_str()];
                    if !value.is_empty() {
                        pieces.extend(value.split('/').map(encode_component));
                    }
                }
            }
        }
        let mut output = if pieces.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", pieces.join("/"))
        };
        if route.trailing && output != "/" {
            output.push('/');
        }
        Ok(output)
    }
}

fn main() {}
