    pub priority: u8,

pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        for (i, job) in jobs.iter().enumerate() {
            let valid_id = !job.id.is_empty()
                && job.id.len() <= 16
                && job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-');
            if !valid_id {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if jobs[..i].iter().any(|earlier| earlier.id == job.id) {
                return Err(Error::Duplicate(i));
            }
        }

        let mut records = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut dependencies = Vec::with_capacity(job.dependencies.len());
            for (di, dep) in job.dependencies.iter().enumerate() {
                if job.dependencies[..di].iter().any(|prior| prior == dep) {
                    return Err(Error::DuplicateDependency(i, di));
                }
                let Some(index) = jobs.iter().position(|candidate| candidate.id == *dep) else {
                    return Err(Error::MissingDependency(i, di));
                };
                dependencies.push(index);
            }
            records.push(JobRecord {
                id: job.id.clone(),
                priority: job.priority,
                dependencies,
                max_attempts: job.max_attempts,
                attempts: 0,
                state: State::Waiting { ready_at: 0 },
            });
        }

        let mut indegree = vec![0usize; records.len()];
        let mut dependents = vec![Vec::new(); records.len()];
        for (i, record) in records.iter().enumerate() {
            indegree[i] = record.dependencies.len();
            for &dep in &record.dependencies {
                dependents[dep].push(i);
            }
        }
        let mut queue = Vec::new();
        for (i, &degree) in indegree.iter().enumerate() {
            if degree == 0 {
                queue.push(i);
            }
        }
        let mut visited = 0usize;
        let mut cursor = 0usize;
        while cursor < queue.len() {
            let node = queue[cursor];
            cursor += 1;
            visited += 1;
            for &child in &dependents[node] {
                indegree[child] -= 1;
                if indegree[child] == 0 {
                    queue.push(child);
                }
            }
        }

            return Err(Error::Cycle);
        }

        Ok(Self {
            jobs: records,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            has_cycle: false,
        })

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.has_cycle
            && self.jobs.iter().any(|job| {
                matches!(job.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for job in &mut self.jobs {
            job.attempts = 0;
            job.state = State::Waiting { ready_at: now };
        }
        self.has_cycle = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for (i, job) in self.jobs.iter().enumerate() {
            let ready_at = match job.state {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            if !job.dependencies.iter().all(|&dep| {
                matches!(self.jobs[dep].state, State::Succeeded)
            }) {
                continue;
            }
            let effective = if self.aging_ms == 0 {
                job.priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                (job.priority as u64 + age).min(9) as u8
            };
            if best.map_or(true, |(_, p)| effective > p) {
                best = Some((i, effective));
            }
        }

        self.last_time = now;
        let Some((index, _)) = best else {
            return Ok(None);
        };

        let job = &mut self.jobs[index];
        job.attempts += 1;
        job.state = State::Running {
            attempt: job.attempts,
        };
        Ok(Some(Ticket {
            id: job.id.clone(),
            attempt: job.attempts,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }
        let Some(index) = self.jobs.iter().position(|job| job.id == id) else {
            return Err(Error::Unknown);
        };
        if !matches!(self.jobs[index].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        self.last_time = now;
        if success {
            self.jobs[index].state = State::Succeeded;
        } else if self.jobs[index].attempts == self.jobs[index].max_attempts {
            self.jobs[index].state = State::Failed;
            self.propagate_blocked();
        } else {
            let exponent = self.jobs[index].attempts.saturating_sub(1);
            let mut delay = self.retry_base;
            for _ in 0..exponent {
                if delay >= self.retry_cap {
                    break;
                }
                delay = delay.saturating_mul(2).min(self.retry_cap);
            }
            self.jobs[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }
        self.jobs
            .iter()
            .find(|job| job.id == id)
            .map(|job| job.state.clone())
            .ok_or(Error::Unknown)
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut changed = false;
            for i in 0..self.jobs.len() {
                if matches!(self.jobs[i].state, State::Waiting { .. })
                    && self.jobs[i].dependencies.iter().any(|&dep| {
                        matches!(self.jobs[dep].state, State::Failed | State::Blocked)
                    })
                {
                    self.jobs[i].state = State::Blocked;
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
    }
}
