```rust
use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, PartialEq, Eq)]
pub struct Big {
    // Little-endian base-10 digits; zero is represented as [0].
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

impl fmt::Debug for Big {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_tuple("Big").field(&self.format(0, 0)).finish()
    }
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = usize::from(bytes[0] == b'+');
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Scan every magnitude byte before checking separator structure.
        if let Some(index) = (start..bytes.len())
            .find(|&i| !bytes[i].is_ascii_digit() && bytes[i] != b'_')
        {
            return Err(Error::InvalidDigit { index });
        }

        // Underscores must have an ASCII digit immediately on both sides.
        if let Some(index) = (start..bytes.len()).find(|&i| {
            bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || !bytes[i - 1].is_ascii_digit()
                    || !bytes[i + 1].is_ascii_digit())
        }) {
            return Err(Error::Separator { index });
        }

        let first_digit = (start..bytes.len())
            .find(|&i| bytes[i].is_ascii_digit())
            .expect("nonempty, lexically valid magnitude contains a digit");

        let digit_count = bytes[start..]
            .iter()
            .filter(|&&b| b.is_ascii_digit())
            .count();

        if digit_count > 1 && bytes[first_digit] == b'0' {
            return Err(Error::LeadingZero { index: first_digit });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            let index = (start..bytes.len())
                .find(|&i| {
                    if bytes[i].is_ascii_digit() {
                        seen += 1;
                        seen == excess_number
                    } else {
                        false
                    }
                })
                .expect("digit count exceeds the limit");
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index,
            });
        }

        let digits = bytes[start..]
            .iter()
            .filter(|&&b| b.is_ascii_digit())
            .map(|&b| b - b'0')
            .rev()
            .collect();

        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let natural_len = self.digits.len().max(1);
        let width = natural_len.max(min_digits);
        let pad = width - natural_len;
        let mut out = String::with_capacity(width.saturating_add(width / group.max(1)));
        let mut emitted = 0usize;

        for i in (0..width).rev() {
            if group > 0 && emitted > 0 && (width - i) % group == 0 {
                out.push('_');
            }
            let digit = if i >= self.digits.len() {
                0
            } else {
                self.digits[i]
            };
            out.push(char::from(b'0' + digit));
            emitted += 1;
        }

        let _ = pad;
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        let left_len = self.normalized_len();
        let right_len = rhs.normalized_len();
        match left_len.cmp(&right_len) {
            Ordering::Equal => {
                for i in (0..left_len).rev() {
                    match self.digit_at(i).cmp(&rhs.digit_at(i)) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let max_len = self.normalized_len().max(rhs.normalized_len());
        let mut digits = Vec::with_capacity(max_len.saturating_add(1));
        let mut carry = 0u8;

        for i in 0..max_len {
            let sum = self.digit_at(i) + rhs.digit_at(i) + carry;
            digits.push(sum % 10);
            carry = sum / 10;
        }
        if carry != 0 {
            digits.push(carry);
        }
        Self::from_digits(digits)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let len = self.normalized_len();
        let mut digits = Vec::with_capacity(len);
        let mut borrow = 0i16;

        for i in 0..len {
            let mut value = i16::from(self.digit_at(i)) - i16::from(rhs.digit_at(i)) - borrow;
            if value < 0 {
                value += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            digits.push(value as u8);
        }

        Ok(Self::from_digits(digits))
    }

    fn digit_at(&self, index: usize) -> u8 {
        self.digits.get(index).copied().unwrap_or(0)
    }

    fn normalized_len(&self) -> usize {
        let mut len = self.digits.len();
        while len > 1 && self.digits[len - 1] == 0 {
            len -= 1;
        }
        len.max(1)
    }

    fn from_digits(mut digits: Vec<u8>) -> Self {
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }
        if digits.is_empty() {
            digits.push(0);
        }
        Self { digits }
    }
}

fn main() {}
```
