#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| matches!(b, b'a'..=b'z' | b'0'..=b'9' | b'_' | b'-'))
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| matches!(b, b'a'..=b'z' | b'0'..=b'9' | b'_'))
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_static_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let value = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || is_unreserved(value) {
                return Err(base + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(base + i);
            }
            decoded.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn split_path<'a>(path: &'a str) -> Result<(Vec<(&'a str, usize)>, bool), usize> {
    if !path.starts_with('/') {
        return Err(0);
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = path.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut parts = Vec::new();
    let mut start = 1;
    let mut i = 1;
    while i <= end {
        if i == end || bytes[i] == b'/' {
            if start == i {
                return Err(start);
            }
            parts.push((&path[start..i], start));
            start = i + 1;
        }
        i += 1;
    }
    Ok((parts, trailing))
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool, Vec<u8>), Error> {
    let (raw_parts, trailing) =
        split_path(pattern).map_err(|at| Error::InvalidPattern { at })?;
    let mut segments = Vec::with_capacity(raw_parts.len());
    let mut names = Vec::<String>::new();

    for (index, (raw, start)) in raw_parts.iter().enumerate() {
        let bytes = raw.as_bytes();
        if bytes.first() == Some(&b':') || bytes.first() == Some(&b'*') {
            let wildcard = bytes[0] == b'*';
            let name = &raw[1..];
            if !valid_capture_name(name) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if names.iter().any(|existing| existing == name) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if wildcard && index + 1 != raw_parts.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            names.push(name.to_string());
            segments.push(if wildcard {
                Segment::Wildcard(name.to_string())
            } else {
                Segment::Param(name.to_string())
            });
        } else {
            let decoded = decode_static_segment(raw, *start)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }

    let mut rank = Vec::with_capacity(segments.len() + 1);
    let has_wildcard = segments.iter().any(|s| matches!(s, Segment::Wildcard(_)));
    for segment in &segments {
        rank.push(match segment {
            Segment::Static(_) => 2,
            Segment::Param(_) => 1,
            Segment::Wildcard(_) => 0,
        });
    }
    if !has_wildcard {
        rank.push(3);
    }
    Ok((segments, trailing, rank))
}

fn parse_request(path: &str) -> Result<(Vec<String>, bool), Error> {
    let head_end = path
        .char_indices()
        .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
        .unwrap_or(path.len());
    let head = &path[..head_end];
    let (raw_parts, trailing) =
        split_path(head).map_err(|at| Error::InvalidPath { at })?;
    let mut segments = Vec::with_capacity(raw_parts.len());
    for (raw, start) in raw_parts {
        segments.push(
            decode_static_segment(raw, start).map_err(|at| Error::InvalidPath { at })?,
        );
    }
    Ok((segments, trailing))
}

fn same_shape(a: &Route, segments: &[Segment], trailing: bool, policy: SlashPolicy) -> bool {
    if a.segments.len() != segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != trailing {
        return false;
    }
    a.segments.iter().zip(segments).all(|(left, right)| match (left, right) {
        (Segment::Static(x), Segment::Static(y)) => x == y,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let has_wildcard = matches!(route.segments.last(), Some(Segment::Wildcard(_)));
    if !has_wildcard && route.segments.len() != request.len() {
        return None;
    }
    if has_wildcard && request.len() + 1 < route.segments.len() {
        return None;
    }

    let mut params = Vec::new();
    for (i, segment) in route.segments.iter().enumerate() {
        match segment {
            Segment::Static(expected) => {
                if request.get(i) != Some(expected) {
                    return None;
                }
            }
            Segment::Param(name) => {
                let value = request.get(i)?;
                params.push((name.clone(), value.clone()));
            }
            Segment::Wildcard(name) => {
                let value = request[i..].join("/");
                params.push((name.clone(), value));
                break;
            }
        }
    }
    Some(params)
}

fn encode_segment(value: &str) -> String {
    let mut out = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing, rank) = parse_pattern(pattern)?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }
        if let Some(route) = self.routes.iter().find(|route| {
            route.method == method && same_shape(route, &segments, trailing, self.policy)
        }) {
            return Err(Error::Conflict { with: route.id });
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (request, trailing) = parse_request(path)?;

        let mut candidates = Vec::<(&Route, Vec<(String, String)>)>::new();
        for route in &self.routes {
            if let Some(params) = match_route(route, &request) {
                if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                    continue;
                }
                candidates.push((route, params));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<_> = candidates
            .iter()
            .filter(|(route, _)| route.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = Vec::<String>::new();
            for (route, _) in &candidates {
                if !allowed.iter().any(|m| m == &route.method) {
                    allowed.push(route.method.clone());
                }
                if route.method == "GET" && !allowed.iter().any(|m| m == "HEAD") {
                    allowed.push("HEAD".to_string());
                }
            }
            allowed.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|(a, _), (b, _)| b.rank.cmp(&a.rank).then_with(|| a.id.cmp(&b.id)));
        let (route, params) = eligible[0];
        let redirect = if self.policy == SlashPolicy::Redirect && route.trailing != trailing {
            let head_end = path
                .char_indices()
                .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
                .unwrap_or(path.len());
            let mut head = path[..head_end].to_string();
            if route.trailing {
                head.push('/');
            } else if head.ends_with('/') && head != "/" {
                head.pop();
            }
            Some(head)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = Vec::<(&str, &str)>::new();
        for &(key, value) in params {
            if supplied.iter().any(|(existing, _)| *existing == key) {
                return Err(Error::DuplicateParam {
                    name: key.to_string(),
                });
            }
            supplied.push((key, value));
        }

        let mut required = Vec::<&str>::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(key) | Segment::Wildcard(key) => required.push(key),
                Segment::Static(_) => {}
            }
        }
        for key in &required {
            if !supplied.iter().any(|(given, _)| given == key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut extras: Vec<&str> = supplied
            .iter()
            .map(|(key, _)| *key)
            .filter(|key| !required.iter().any(|required| required == key))
            .collect();
        extras.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for key in &required {
            let value = supplied.iter().find(|(given, _)| given == key).unwrap().1;
            let valid_ascii = value.bytes().all(|b| (0x20..=0x7e).contains(&b));
            let is_wildcard = route
                .segments
                .iter()
                .any(|segment| matches!(segment, Segment::Wildcard(n) if n == key));
            let valid_value = if !valid_ascii {
                false
            } else if is_wildcard {
                value.is_empty()
                    || value
                        .split('/')
                        .all(|part| !part.is_empty())
            } else {
                !value.is_empty() && !value.contains('/')
            };
            if !valid_value {
                return Err(Error::InvalidValue {
                    name: (*key).to_string(),
                });
            }
        }

        let mut output = String::from("/");
        let mut first = true;
        for segment in &route.segments {
            if !first {
                output.push('/');
            }
            first = false;
            match segment {
                Segment::Static(value) => output.push_str(&encode_segment(value)),
                Segment::Param(key) => {
                    let value = supplied.iter().find(|(given, _)| given == key).unwrap().1;
                    output.push_str(&encode_segment(value));
                }
                Segment::Wildcard(key) => {
                    let value = supplied.iter().find(|(given, _)| given == key).unwrap().1;
                    if !value.is_empty() {
                        for (i, part) in value.split('/').enumerate() {
                            if i != 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_segment(part));
                        }
                    }
                }
            }
        }
        if route.trailing && !route.segments.is_empty() {
            output.push('/');
        }
        Ok(output)
    }
}

fn main() {}
