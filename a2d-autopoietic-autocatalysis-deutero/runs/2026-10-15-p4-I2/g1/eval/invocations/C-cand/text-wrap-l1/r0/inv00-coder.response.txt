#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    for (index, &byte) in input.as_bytes().iter().enumerate() {
        if !(32..=126).contains(&byte) && byte != b'\t' && byte != b'\n' && byte != b'\r' {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();

    for word in input.split(|c: char| c == ' ' || c == '\t' || c == '\n' || c == '\r') {
        if word.is_empty() {
            continue;
        }

        let bytes = word.as_bytes();
        let mut consumed = 0;

        while consumed < bytes.len() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            let remaining = bytes.len() - consumed;
            let line_len = line.len();

            if remaining <= width {
                let separator_len = usize::from(!line.is_empty());
                if line_len + separator_len + remaining <= width {
                    if separator_len != 0 {
                        line.push(' ');
                    }
                    line.push_str(&word[consumed..]);
                    consumed = bytes.len();
                } else if !line.is_empty() {
                    lines.push(std::mem::take(&mut line));
                } else {
                    line.push_str(&word[consumed..]);
                    consumed = bytes.len();
                }
            } else if line.is_empty() {
                line.push_str(&word[consumed..consumed + width]);
                consumed += width;
                lines.push(std::mem::take(&mut line));
            } else {
                let available = width - line_len - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(&word[consumed..consumed + available]);
                    consumed += available;
                    lines.push(std::mem::take(&mut line));
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}

fn main() {}
