#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(q + u128::from(round_up))
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    waited: u64,
    first_started: bool,
    paused_at: Option<u64>,
}

#[derive(Clone)]
struct Active {
    index: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
    dispatched_at: u64,
    protected: bool,
}

fn waiting_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.paused_at, b.paused_at) {
            (Some(x), Some(y)) => x.cmp(&y).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn add_u64(a: u64, b: u64) -> Option<u64> {
    a.checked_add(b)
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (i, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index: i });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index: i, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index: i, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow {
                index: i,
                id: c.id,
            })?),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let n = customers.len();
    let mut jobs: Vec<Job> = customers
        .iter()
        .cloned()
        .zip(initial_deadlines)
        .map(|(customer, deadline)| Job {
            remaining: customer.service,
            customer,
            deadline,
            waited: 0,
            first_started: false,
            paused_at: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut last_dispatch: Option<u32> = None;

    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();
    let mut last_event_time = 0u64;

    let emit = |events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind, last: &mut u64| {
        events.push(Event { time, customer, kind });
        *last = time;
    };

    while next_arrival < n || active.is_some() || !waiting.is_empty() {
        if active.is_none() && waiting.is_empty() {
            if next_arrival >= n {
                break;
            }
            now = now.max(customers[order[next_arrival]].arrival);
        }

        // Phase A: finish at this time.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let id = jobs[a.index].customer.id;
            emit(&mut events, now, id, Kind::Finish, &mut last_event_time);
            completed += 1;
            turnaround_ticks += (now - jobs[a.index].customer.arrival) as u128;
            completion_order.push(id);
        }

        // Phase B: expire those waiting at the start of the tick.
        let mut expired: Vec<(u64, u32, usize)> = waiting
            .iter()
            .filter_map(|&i| jobs[i].deadline.filter(|&d| d <= now).map(|d| (d, jobs[i].customer.id, i)))
            .collect();
        expired.sort_by_key(|x| (x.0, x.1));
        for (_, _, i) in expired {
            if let Some(pos) = waiting.iter().position(|&x| x == i) {
                waiting.remove(pos);
                let seg = now - jobs[i].customer.arrival;
                let _ = seg;
                let total = jobs[i].waited;
                emit(&mut events, now, jobs[i].customer.id, Kind::Renege { waited: total }, &mut last_event_time);
                reneged += 1;
            }
        }

        // Phase C: full arrival cohort.
        while next_arrival < n && customers[order[next_arrival]].arrival == now {
            let i = order[next_arrival];
            next_arrival += 1;
            emit(&mut events, now, jobs[i].customer.id, Kind::Arrive, &mut last_event_time);
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= gate {
                emit(&mut events, now, jobs[i].customer.id, Kind::Balk, &mut last_event_time);
                balked += 1;
            } else {
                accepted += 1;
                waiting.push(i);
            }
        }

        // Phase D: reconsider preemption after protection ends.
        if let Some(a) = active.as_mut() {
            if now >= a.work_at {
                a.protected = false;
            }
        }
        if active.as_ref().is_some_and(|a| !a.protected) && !waiting.is_empty() {
            let best = *waiting.iter().min_by(|&&x, &&y| waiting_cmp(&jobs[x], &jobs[y])).unwrap();
            let a = active.as_ref().unwrap();
            if jobs[best].customer.priority < jobs[a.index].customer.priority {
                let a = active.take().unwrap();
                let id = jobs[a.index].customer.id;
                let by = jobs[best].customer.id;
                let remaining = a.finish - now;
                emit(&mut events, now, id, Kind::Preempt { remaining, by }, &mut last_event_time);
                preemptions += 1;
                busy_ticks += (now - a.dispatched_at) as u128;
                jobs[a.index].remaining = remaining;
                let elapsed_wait = now - jobs[a.index].customer.arrival;
                jobs[a.index].waited = jobs[a.index].waited.saturating_add(0);
                jobs[a.index].deadline = match jobs[a.index].customer.patience {
                    Some(budget) => Some(now.checked_add(budget).ok_or(Error::TimeOverflow {
                        customer: id,
                        time: now,
                        stage: OverflowStage::Patience,
                    })?),
                    None => None,
                };
                jobs[a.index].paused_at = Some(now);
                let _ = elapsed_wait;
                waiting.push(a.index);
            }
        }

        // Phase E: dispatch until occupied or queue empty; inline finishes repeat.
        loop {
            if active.is_some() || waiting.is_empty() {
                break;
            }
            let pos = waiting
                .iter()
                .enumerate()
                .min_by(|(_, &x), (_, &y)| waiting_cmp(&jobs[x], &jobs[y]))
                .map(|(p, _)| p)
                .unwrap();
            let i = waiting.remove(pos);
            let id = jobs[i].customer.id;
            let segment = now - jobs[i].customer.arrival;
            jobs[i].waited = jobs[i].waited.saturating_add(segment);
            wait_ticks += segment as u128;

            let switching = last_dispatch.is_some_and(|prev| prev != id);
            let cost = if switching { config.switch_cost } else { 0 };
            let work_at = add_u64(now, cost).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = add_u64(work_at, jobs[i].remaining).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::Finish,
            })?;

            let remaining = jobs[i].remaining;
            let first = !jobs[i].first_started;
            if first {
                jobs[i].first_started = true;
                started += 1;
            }
            let kind = if first {
                Kind::Start { remaining, work_at, finish }
            } else {
                Kind::Resume { remaining, work_at, finish }
            };
            emit(&mut events, now, id, kind, &mut last_event_time);
            last_dispatch = Some(id);

            if finish == now {
                emit(&mut events, now, id, Kind::Finish, &mut last_event_time);
                completed += 1;
                turnaround_ticks += (now - jobs[i].customer.arrival) as u128;
                completion_order.push(id);
            } else {
                active = Some(Active {
                    index: i,
                    remaining,
                    work_at,
                    finish,
                    dispatched_at: now,
                    protected: now < work_at,
                });
            }
        }

        // Phase F: expire newly queued jobs.
        let mut newly_expired: Vec<(u64, u32, usize)> = waiting
            .iter()
            .filter_map(|&i| jobs[i].deadline.filter(|&d| d <= now).map(|d| (d, jobs[i].customer.id, i)))
            .collect();
        newly_expired.sort_by_key(|x| (x.0, x.1));
        for (_, _, i) in newly_expired {
            if let Some(pos) = waiting.iter().position(|&x| x == i) {
                waiting.remove(pos);
                let id = jobs[i].customer.id;
                let final_segment = now - jobs[i].customer.arrival;
                jobs[i].waited = jobs[i].waited.saturating_add(final_segment);
                wait_ticks += final_segment as u128;
                emit(&mut events, now, id, Kind::Renege { waited: jobs[i].waited }, &mut last_event_time);
                reneged += 1;
            }
        }

        if next_arrival >= n && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |t: u64| {
            if t > now {
                next_time = Some(next_time.map_or(t, |x| x.min(t)));
            }
        };
        if next_arrival < n {
            consider(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = &active {
            consider(a.finish);
            if a.protected {
                consider(a.work_at);
            }
        }
        for &i in &waiting {
            if let Some(d) = jobs[i].deadline {
                consider(d);
            }
        }
        if let Some(t) = next_time {
            now = t;
        } else if active.is_none() && waiting.is_empty() && next_arrival < n {
            now = customers[order[next_arrival]].arrival;
        } else {
            break;
        }
    }

    let mean_wait_milli = if accepted == 0 { 0 } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 { 0 } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap()
    };
    let utilization_bps = if last_event_time == 0 { 0 } else {
        rounded_ratio(busy_ticks * 10_000, last_event_time as u128).unwrap() as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed: last_event_time,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}

fn main() {}
