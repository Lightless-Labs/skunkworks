#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut runs: Vec<Run> = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut completion: Vec<(u32, u64)> = jobs
        .iter()
        .map(|job| (job.id, 0))
        .collect();
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut ready: Vec<usize> = Vec::new();
    let mut event_index = 0usize;
    let mut time = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut completed_count = 0usize;

    for (index, job) in jobs.iter().enumerate() {
        if job.ticks == 0 {
            completion[index].1 = u64::from(job.arrival);
            completed_count += 1;
        }
    }

    while completed_count < jobs.len() {
        while event_index < events.len()
            && u64::from(jobs[events[event_index]].arrival) <= time
        {
            let index = events[event_index];
            event_index += 1;
            if jobs[index].ticks > 0 {
                ready.push(index);
            }
        }

        if ready.is_empty() {
            if event_index >= events.len() {
                break;
            }

            let next_time = u64::from(jobs[events[event_index]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched = None;
            }
            continue;
        }

        let index = ready.remove(0);
        let id = jobs[index].id;

        if last_dispatched.is_some_and(|last_id| last_id != id) {
            time += u64::from(switch_cost);
        }

        while event_index < events.len()
            && u64::from(jobs[events[event_index]].arrival) <= time
        {
            let arriving = events[event_index];
            event_index += 1;
            if jobs[arriving].ticks > 0 {
                ready.push(arriving);
            }
        }

        let consumed = remaining[index].min(quantum);
        let start = time;
        time += u64::from(consumed);
        remaining[index] -= consumed;
        let finished = remaining[index] == 0;

        runs.push(Run {
            id,
            start,
            end: time,
            finished,
        });

        while event_index < events.len()
            && u64::from(jobs[events[event_index]].arrival) <= time
        {
            let arriving = events[event_index];
            event_index += 1;
            if jobs[arriving].ticks > 0 {
                ready.push(arriving);
            }
        }

        if finished {
            completion[index].1 = time;
            completed_count += 1;
        } else {
            ready.push(index);
        }

        last_dispatched = Some(id);
    }

    let end = completion.iter().map(|entry| entry.1).max().unwrap_or(0);

    Ok(Schedule {
        runs,
        idle,
        completion,
        end,
    })
}

fn main() {}
