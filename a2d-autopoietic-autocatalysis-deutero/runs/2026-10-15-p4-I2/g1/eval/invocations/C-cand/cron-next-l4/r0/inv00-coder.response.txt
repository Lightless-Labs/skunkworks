use std::fmt;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_dom: bool,
    unrestricted_dow: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

const MINIMUMS: [u8; 5] = [0, 0, 1, 1, 0];
const MAXIMUMS: [u8; 5] = [59, 23, 31, 12, 6];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(bytes: &[u8], start: usize, field: Field) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }
    Ok(value)
}

fn name_value(bytes: &[u8], index: usize) -> Option<u32> {
    let names: &[&str] = match index {
        3 => &["JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"],
        4 => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    if bytes.len() != 3 {
        return None;
    }
    let upper = [
        bytes[0].to_ascii_uppercase(),
        bytes[1].to_ascii_uppercase(),
        bytes[2].to_ascii_uppercase(),
    ];
    names.iter().position(|name| name.as_bytes() == upper).map(|n| {
        if index == 3 { n as u32 + 1 } else { n as u32 }
    })
}

fn parse_atom(
    bytes: &[u8],
    start: usize,
    index: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) {
        let value = parse_number(bytes, start, field)?;
        if value < MINIMUMS[index] as u32 || value > MAXIMUMS[index] as u32 {
            return Err(ParseError::OutOfRange { field, offset: start });
        }
        Ok(value)
    } else if let Some(value) = name_value(bytes, index) {
        Ok(value)
    } else {
        Err(syntax(field, start))
    }
}

fn parse_step(bytes: &[u8], start: usize, field: Field, index: usize) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(syntax(field, start));
    }
    let value = parse_number(bytes, start, field)?;
    if value == 0 || value > (MAXIMUMS[index] - MINIMUMS[index] + 1) as u32 {
        return Err(ParseError::BadStep { field, offset: start });
    }
    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    index: usize,
    field: Field,
    output: &mut [bool],
) -> Result<(), ParseError> {
    if bytes.is_empty() {
        return Err(ParseError::EmptyItem { field, offset: item_start });
    }

    let slash_positions: Vec<usize> = bytes.iter().enumerate().filter_map(|(i, b)| (*b == b'/').then_some(i)).collect();
    if slash_positions.len() > 1 {
        return Err(syntax(field, item_start));
    }
    let (base, step, step_start) = if let Some(&slash) = slash_positions.first() {
        (&bytes[..slash], Some(&bytes[slash + 1..]), item_start + slash + 1)
    } else {
        (bytes, None, item_start)
    };
    if base.is_empty() || (step.is_some() && step.unwrap().is_empty()) {
        return Err(syntax(field, item_start));
    }

    let hyphens: Vec<usize> = base.iter().enumerate().filter_map(|(i, b)| (*b == b'-').then_some(i)).collect();
    if hyphens.len() > 1 {
        return Err(syntax(field, item_start));
    }
    let is_star = base == b"*";
    let range = if let Some(&hyphen) = hyphens.first() {
        if hyphen == 0 || hyphen + 1 == base.len() {
            return Err(syntax(field, item_start));
        }
        Some((&base[..hyphen], item_start, &base[hyphen + 1..], item_start + hyphen + 1))
    } else {
        None
    };
    if !is_star && range.is_none() && base.contains(&b'*') {
        return Err(syntax(field, item_start));
    }
    if is_star && range.is_some() {
        return Err(syntax(field, item_start));
    }

    let parsed = if is_star {
        (MINIMUMS[index] as u32, MAXIMUMS[index] as u32)
    } else if let Some((left, left_start, right, right_start)) = range {
        let start = parse_atom(left, left_start, index, field)?;
        let end = parse_atom(right, right_start, index, field)?;
        if start > end {
            return Err(ParseError::ReversedRange { field, offset: right_start });
        }
        (start, end)
    } else {
        let value = parse_atom(base, item_start, index, field)?;
        (value, if step.is_some() { MAXIMUMS[index] as u32 } else { value })
    };

    let step_value = if let Some(step_bytes) = step {
        Some(parse_step(step_bytes, step_start, field, index)?)
    } else {
        None
    };
    let stride = step_value.unwrap_or(1);
    let mut value = parsed.0;
    while value <= parsed.1 {
        output[value as usize] = true;
        match value.checked_add(stride) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }
    Ok(())
}

fn parse_field(text: &str, index: usize) -> Result<Vec<bool>, ParseError> {
    let field = FIELDS[index];
    for (offset, &byte) in text.as_bytes().iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let max = MAXIMUMS[index] as usize;
    let mut output = vec![false; max + 1];
    let bytes = text.as_bytes();
    let mut start = 0;
    for (position, &byte) in bytes.iter().enumerate() {
        if byte == b',' {
            parse_item(&bytes[start..position], start, index, field, &mut output)?;
            start = position + 1;
        }
    }
    parse_item(&bytes[start..], start, index, field, &mut output)?;
    Ok(output)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    if input.as_bytes().first() == Some(&b'@') {
        let expansion = match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        };
        return parse_cron(expansion);
    }

    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let mut selected = Vec::with_capacity(5);
    for (index, part) in parts.iter().enumerate() {
        selected.push(parse_field(part, index)?);
    }
    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        unrestricted_dom: parts[2] == "*",
        unrestricted_dow: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        (MINIMUMS[index]..=MAXIMUMS[index])
            .filter(|&value| self.selected[index][value as usize])
            .collect()
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        let date = at.day as usize;
        let weekday = weekday(at.year, at.month, at.day) as usize;
        let minute = self.selected[0][at.minute as usize];
        let hour = self.selected[1][at.hour as usize];
        let month = self.selected[3][at.month as usize];
        let dom = self.selected[2][date];
        let dow = self.selected[4][weekday];
        let day_matches = match (self.unrestricted_dom, self.unrestricted_dow) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) if at.day == days_in_month(at.year, at.month) => dom && dow,
            (false, false) => dom || dow,
        };
        minute && hour && month && day_matches
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut current = after;
        loop {
            current = match advance_minute(current) {
                Some(value) => value,
                None => return Ok(None),
            };
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut current = before;
        loop {
            current = match retreat_minute(current) {
                Some(value) => value,
                None => return Ok(None),
            };
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        let mut fires = Vec::new();
        let mut current = start;
        while current < end {
            if self.matches_valid(current) {
                if fires.len() == cap {
                    return Ok(FireWindow { fires, truncated: true });
                }
                fires.push(current);
            }
            current = match advance_minute(current) {
                Some(value) => value,
                None => break,
            };
        }
        Ok(FireWindow { fires, truncated: false })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::with_capacity(5);
        for index in 0..5 {
            let min = MINIMUMS[index];
            let max = MAXIMUMS[index];
            let full = (min..=max).all(|value| self.selected[index][value as usize]);
            if index == 2 && self.unrestricted_dom || index == 4 && self.unrestricted_dow {
                fields.push("*".to_string());
            } else if full && (index == 0 || index == 1 || index == 3) {
                fields.push("*".to_string());
            } else if full {
                fields.push("*/1".to_string());
            } else {
                fields.push(
                    (min..=max)
                        .filter(|&value| self.selected[index][value as usize])
                        .map(|value| value.to_string())
                        .collect::<Vec<_>>()
                        .join(","),
                );
            }
        }
        fields.join(" ")
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn advance_minute(mut at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute < 59 {
        at.minute += 1;
    } else {
        at.minute = 0;
        if at.hour < 23 {
            at.hour += 1;
        } else {
            at.hour = 0;
            if at.day < days_in_month(at.year, at.month) {
                at.day += 1;
            } else {
                at.day = 1;
                if at.month < 12 {
                    at.month += 1;
                } else if at.year < 2399 {
                    at.month = 1;
                    at.year += 1;
                } else {
                    return None;
                }
            }
        }
    }
    Some(at)
}

fn retreat_minute(mut at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute > 0 {
        at.minute -= 1;
    } else {
        at.minute = 59;
        if at.hour > 0 {
            at.hour -= 1;
        } else {
            at.hour = 23;
            if at.day > 1 {
                at.day -= 1;
            } else if at.month > 1 {
                at.month -= 1;
                at.day = days_in_month(at.year, at.month);
            } else if at.year > 2000 {
                at.year -= 1;
                at.month = 12;
                at.day = 31;
            } else {
                return None;
            }
        }
    }
    Some(at)
}

impl fmt::Display for Field {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        write!(f, "{:?}", self)
    }
}
