#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    started: bool,
    remaining: u64,
    pause_time: Option<u64>,
}

#[derive(Clone, Copy)]
struct Running {
    index: usize,
    work_at: u64,
    finish: u64,
}

fn waiting_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(at), Some(bt)) => at.cmp(&bt).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => std::cmp::Ordering::Less,
            (None, Some(_)) => std::cmp::Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut seen = std::collections::HashSet::new();

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut jobs: Vec<Job> = customers
        .iter()
        .cloned()
        .map(|customer| Job {
            remaining: customer.service,
            customer,
            started: false,
            pause_time: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut running: Option<Running> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut next_arrival = 0usize;
    let mut finished_count = 0usize;
    let mut now = 0u64;

    while finished_count < customers.len() {
        // A scheduled finish takes precedence over arrivals and work boundaries.
        if let Some(active) = running {
            if active.finish == now {
                events.push(Event {
                    time: now,
                    customer: jobs[active.index].customer.id,
                    kind: Kind::Finish,
                });
                running = None;
                finished_count += 1;
            }
        }

        // Add the complete arrival cohort before making scheduling decisions.
        while next_arrival < arrivals.len()
            && customers[arrivals[next_arrival]].arrival == now
        {
            let index = arrivals[next_arrival];
            events.push(Event {
                time: now,
                customer: jobs[index].customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(index);
            next_arrival += 1;
        }

        if let Some(active) = running {
            if now >= active.work_at && !waiting.is_empty() {
                let best_pos = waiting
                    .iter()
                    .enumerate()
                    .min_by(|(_, &a), (_, &b)| waiting_cmp(&jobs[a], &jobs[b]))
                    .map(|(pos, _)| pos)
                    .unwrap();

                let best = waiting[best_pos];
                if jobs[best].customer.priority < jobs[active.index].customer.priority {
                    waiting.remove(best_pos);
                    let paused = active.index;
                    jobs[paused].remaining = active.finish - now;
                    jobs[paused].pause_time = Some(now);
                    events.push(Event {
                        time: now,
                        customer: jobs[paused].customer.id,
                        kind: Kind::Preempt {
                            remaining: jobs[paused].remaining,
                            by: jobs[best].customer.id,
                        },
                    });
                    waiting.push(paused);
                    running = None;
                }
            }
        }

        // Dispatch until occupied, allowing zero-length jobs to finish inline.
        while running.is_none() && !waiting.is_empty() {
            let best_pos = waiting
                .iter()
                .enumerate()
                .min_by(|(_, &a), (_, &b)| waiting_cmp(&jobs[a], &jobs[b]))
                .map(|(pos, _)| pos)
                .unwrap();
            let index = waiting.remove(best_pos);
            let id = jobs[index].customer.id;
            let cost = if last_dispatched.map_or(false, |previous| previous != id) {
                switch_cost
            } else {
                0
            };

            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
            })?;
            let finish = work_at
                .checked_add(jobs[index].remaining)
                .ok_or(Error::TimeOverflow {
                    customer: id,
                    time: now,
                })?;

            let kind = if jobs[index].started {
                Kind::Resume {
                    remaining: jobs[index].remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: jobs[index].remaining,
                    work_at,
                    finish,
                }
            };
            events.push(Event {
                time: now,
                customer: id,
                kind,
            });

            jobs[index].started = true;
            jobs[index].pause_time = None;
            last_dispatched = Some(id);
            running = Some(Running {
                index,
                work_at,
                finish,
            });

            if finish == now {
                events.push(Event {
                    time: now,
                    customer: id,
                    kind: Kind::Finish,
                });
                running = None;
                finished_count += 1;
            }
        }

        if finished_count == customers.len() {
            break;
        }

        let mut next_time: Option<u64> = None;
        if next_arrival < arrivals.len() {
            next_time = Some(customers[arrivals[next_arrival]].arrival);
        }
        if let Some(active) = running {
            for t in [active.finish, active.work_at] {
                if t > now && next_time.map_or(true, |current| t < current) {
                    next_time = Some(t);
                }
            }
        }

        match next_time {
            Some(t) => now = t,
            None => break,
        }
    }

    Ok(events)
}

fn main() {}
