```rust
use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian decimal digits. Zero is represented by [0] and is never negative.
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

impl fmt::Debug for Big {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        f.debug_struct("Big")
            .field("negative", &self.negative)
            .field("digits", &self.digits)
            .finish()
    }
}

impl Big {
    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn from_parts(negative: bool, digits: Vec<u8>) -> Self {
        let mut value = Self { negative, digits };
        value.normalize();
        value
    }

    fn normalize(&mut self) {
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.is_empty() {
            self.digits.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => {
                for (a, b) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let n = a.len().max(b.len());
        let mut out = Vec::with_capacity(n + 1);
        let mut carry = 0u8;
        for i in 0..n {
            let sum = a.get(i).copied().unwrap_or(0)
                + b.get(i).copied().unwrap_or(0)
                + carry;
            out.push(sum % 10);
            carry = sum / 10;
        }
        if carry != 0 {
            out.push(carry);
        }
        out
    }

    // Requires a >= b.
    fn sub_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i8;
        for i in 0..a.len() {
            let mut digit = a[i] as i8 - borrow - b.get(i).copied().unwrap_or(0) as i8;
            if digit < 0 {
                digit += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(digit as u8);
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut accum = vec![0u32; a.len() + b.len()];
        for (i, &x) in a.iter().enumerate() {
            for (j, &y) in b.iter().enumerate() {
                accum[i + j] += (x as u32) * (y as u32);
            }
        }
        for i in 0..accum.len().saturating_sub(1) {
            let carry = accum[i] / 10;
            accum[i] %= 10;
            accum[i + 1] += carry;
        }
        let mut out: Vec<u8> = accum.into_iter().map(|x| (x % 10) as u8).collect();
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_abs_digit(a: &[u8], digit: u8) -> Vec<u8> {
        if digit == 0 {
            return vec![0];
        }
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u16;
        for &x in a {
            let n = x as u16 * digit as u16 + carry;
            out.push((n % 10) as u8);
            carry = n / 10;
        }
        while carry > 0 {
            out.push((carry % 10) as u8);
            carry /= 10;
        }
        out
    }

    fn shift_add_digit(a: &[u8], digit: u8) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len() + 1);
        out.push(digit);
        out.extend_from_slice(a);
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over malformed separators.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let mut first_separator_error = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                first_separator_error = Some(index);
                break;
            }
        }
        if let Some(index) = first_separator_error {
            return Err(Error::Separator { index });
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len())
            .find(|&i| bytes[i].is_ascii_digit())
            .expect("nonempty magnitude with validated separators");
        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        let mut seen = 0usize;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte.is_ascii_digit() {
                seen += 1;
                if seen > max_digits {
                    return Err(Error::TooManyDigits {
                        limit: max_digits,
                        index,
                    });
                }
            }
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes[start..].iter().rev() {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }
        Ok(Self::from_parts(negative, digits))
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let raw_len = self.digits.len().max(1);
        let magnitude_len = raw_len.max(min_digits).max(1);
        let padding = magnitude_len - raw_len;
        let mut out = String::with_capacity(magnitude_len + magnitude_len / group.max(1) + 1);
        if self.negative {
            out.push('-');
        }
        for i in (0..magnitude_len).rev() {
            if group > 0 && i + 1 < magnitude_len && (magnitude_len - 1 - i + 1) % group == 0 {
                out.push('_');
            }
            if i >= self.digits.len() {
                out.push('0');
            } else {
                out.push((b'0' + self.digits[i]) as char);
            }
        }
        let _ = padding;
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => self.cmp_abs(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            Self::from_parts(
                self.negative,
                Self::add_abs(&self.digits, &rhs.digits),
            )
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Greater => Self::from_parts(
                    self.negative,
                    Self::sub_abs(&self.digits, &rhs.digits),
                ),
                Ordering::Less => Self::from_parts(
                    rhs.negative,
                    Self::sub_abs(&rhs.digits, &self.digits),
                ),
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            Self::zero()
        } else {
            Self::from_parts(!self.negative, self.digits.clone())
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        Self::from_parts(
            self.negative ^ rhs.negative,
            Self::mul_abs(&self.digits, &rhs.digits),
        )
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let mut remainder = vec![0];
        let mut quotient_be = Vec::with_capacity(self.digits.len());
        for &digit in self.digits.iter().rev() {
            remainder = Self::shift_add_digit(&remainder, digit);
            let mut qdigit = 0u8;
            for candidate in (1..=9).rev() {
                let product = Self::mul_abs_digit(&rhs.digits, candidate);
                if Self::cmp_digits(&product, &remainder) != Ordering::Greater {
                    qdigit = candidate;
                    remainder = Self::sub_abs(&remainder, &product);
                    break;
                }
            }
            quotient_be.push(qdigit);
        }

        quotient_be.reverse();
        let quotient = Self::from_parts(
            self.negative ^ rhs.negative,
            quotient_be,
        );
        let remainder = Self::from_parts(self.negative, remainder);
        Ok((quotient, remainder))
    }

    fn cmp_digits(a: &[u8], b: &[u8]) -> Ordering {
        let mut a_len = a.len();
        while a_len > 1 && a[a_len - 1] == 0 {
            a_len -= 1;
        }
        let mut b_len = b.len();
        while b_len > 1 && b[b_len - 1] == 0 {
            b_len -= 1;
        }
        match a_len.cmp(&b_len) {
            Ordering::Equal => {
                for i in (0..a_len).rev() {
                    match a[i].cmp(&b[i]) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }
}

fn main() {}
```
