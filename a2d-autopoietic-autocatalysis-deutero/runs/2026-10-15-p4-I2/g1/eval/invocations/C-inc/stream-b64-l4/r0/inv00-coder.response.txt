#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
    pub strict: bool,
    pub max_symbols: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
    NonCanonical { offset: usize },
    MixedAlphabet { offset: usize },
    Overlong { offset: usize, limit: usize },
}

fn validate_options(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_value(alphabet: Alphabet, byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn foreign_alphabet_byte(alphabet: Alphabet, byte: u8) -> bool {
    match alphabet {
        Alphabet::Standard => byte == b'-' || byte == b'_',
        Alphabet::UrlSafe => byte == b'+' || byte == b'/',
    }
}

fn encode_digit(alphabet: Alphabet, value: u8) -> u8 {
    match value {
        0..=25 => b'A' + value,
        26..=51 => b'a' + value - 26,
        52..=61 => b'0' + value - 52,
        62 if alphabet == Alphabet::Standard => b'+',
        62 => b'-',
        63 if alphabet == Alphabet::Standard => b'/',
        _ => b'_',
    }
}

struct EncoderState {
    options: Options,
    tail: Vec<u8>,
    symbols_in_row: usize,
    closed: bool,
}

pub struct Encoder {
    state: EncoderState,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            state: EncoderState {
                options,
                tail: Vec::new(),
                symbols_in_row: 0,
                closed: false,
            },
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.state.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut data = Vec::with_capacity(self.state.tail.len() + input.len());
        data.extend_from_slice(&self.state.tail);
        data.extend_from_slice(input);
        self.state.tail.clear();

        let complete_len = data.len() / 3 * 3;
        let mut output = Vec::new();
        for chunk in data[..complete_len].chunks_exact(3) {
            self.emit_symbol(encode_digit(
                self.state.options.alphabet,
                chunk[0] >> 2,
            ), &mut output);
            self.emit_symbol(
                encode_digit(
                    self.state.options.alphabet,
                    ((chunk[0] & 0x03) << 4) | (chunk[1] >> 4),
                ),
                &mut output,
            );
            self.emit_symbol(
                encode_digit(
                    self.state.options.alphabet,
                    ((chunk[1] & 0x0f) << 2) | (chunk[2] >> 6),
                ),
                &mut output,
            );
            self.emit_symbol(
                encode_digit(self.state.options.alphabet, chunk[2] & 0x3f),
                &mut output,
            );
        }
        self.state.tail.extend_from_slice(&data[complete_len..]);
        Ok(String::from_utf8(output).expect("base64 output is ASCII"))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.state.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        let alphabet = self.state.options.alphabet;
        match self.state.tail.as_slice() {
            [] => {}
            [a] => {
                self.emit_symbol(encode_digit(alphabet, a >> 2), &mut output);
                self.emit_symbol(encode_digit(alphabet, (a & 0x03) << 4), &mut output);
                if self.state.options.padding != Padding::Forbidden {
                    self.emit_symbol(b'=', &mut output);
                    self.emit_symbol(b'=', &mut output);
                }
            }
            [a, b] => {
                self.emit_symbol(encode_digit(alphabet, a >> 2), &mut output);
                self.emit_symbol(
                    encode_digit(alphabet, ((a & 0x03) << 4) | (b >> 4)),
                    &mut output,
                );
                self.emit_symbol(encode_digit(alphabet, (b & 0x0f) << 2), &mut output);
                if self.state.options.padding != Padding::Forbidden {
                    self.emit_symbol(b'=', &mut output);
                }
            }
            _ => unreachable!(),
        }

        self.state.tail.clear();
        self.state.closed = true;
        Ok(String::from_utf8(output).expect("base64 output is ASCII"))
    }

    fn emit_symbol(&mut self, byte: u8, output: &mut Vec<u8>) {
        if let Whitespace::Wrapped { width, ending } = self.state.options.whitespace {
            if self.state.symbols_in_row == width {
                match ending {
                    LineEnding::Lf => output.push(b'\n'),
                    LineEnding::CrLf => output.extend_from_slice(b"\r\n"),
                }
                self.state.symbols_in_row = 0;
            }
            self.state.symbols_in_row += 1;
        }
        output.push(byte);
    }
}

#[derive(Clone, Copy)]
enum WrapState {
    Row,
    NeedSeparatorFirst { offset: usize },
    NeedLf { offset: usize },
    NeedSymbol { offset: usize },
}

struct DecoderState {
    options: Options,
    offset: usize,
    symbol_count: usize,
    group: Vec<(u8, usize)>,
    pending_second_pad: Option<usize>,
    terminal_padding: bool,
    wrap_state: WrapState,
    closed: bool,
    sticky: Option<Error>,
}

pub struct Decoder {
    state: DecoderState,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            state: DecoderState {
                options,
                offset: 0,
                symbol_count: 0,
                group: Vec::with_capacity(4),
                pending_second_pad: None,
                terminal_padding: false,
                wrap_state: WrapState::Row,
                closed: false,
                sticky: None,
            },
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.state.sticky {
            return Err(error.clone());
        }
        if self.state.closed {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.state.offset;
            self.state.offset += 1;

            if let Err(error) = self.process_byte(byte, offset, &mut output) {
                self.state.sticky = Some(error.clone());
                return Err(error);
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.state.sticky {
            return Err(error.clone());
        }
        if self.state.closed {
            return Err(Error::AlreadyFinished);
        }

        let result = self.finish_inner();
        match result {
            Ok(output) => {
                self.state.closed = true;
                Ok(output)
            }
            Err(error) => {
                self.state.sticky = Some(error.clone());
                Err(error)
            }
        }
    }

    fn process_byte(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        let whitespace = self.state.options.whitespace;

        if whitespace == Whitespace::IgnoreAscii && matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
            return Ok(());
        }

        if let Whitespace::Wrapped { width, ending } = whitespace {
            match self.state.wrap_state {
                WrapState::NeedSeparatorFirst { offset: first } => {
                    let expected = match ending {
                        LineEnding::Lf => b'\n',
                        LineEnding::CrLf => b'\r',
                    };
                    if byte != expected {
                        return Err(Error::BadWrap { offset });
                    }
                    match ending {
                        LineEnding::Lf => {
                            self.state.wrap_state = WrapState::NeedSymbol { offset: first };
                        }
                        LineEnding::CrLf => {
                            self.state.wrap_state = WrapState::NeedLf { offset: first };
                        }
                    }
                    return Ok(());
                }
                WrapState::NeedLf { offset: first } => {
                    if byte != b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    self.state.wrap_state = WrapState::NeedSymbol { offset: first };
                    return Ok(());
                }
                WrapState::NeedSymbol { offset: first } => {
                    self.state.wrap_state = WrapState::Row;
                    if self.state.symbol_count == 0 {
                        // symbol_count is global, so row position is tracked by group-independent
                        // framing below; this state itself guarantees a symbol must follow.
                    }
                    if byte == b'\r' || byte == b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    let _ = first;
                }
                WrapState::Row => {}
            }

            let row_symbols = self.row_symbol_count();
            if row_symbols == width {
                let expected = match ending {
                    LineEnding::Lf => b'\n',
                    LineEnding::CrLf => b'\r',
                };
                if byte != expected {
                    return Err(Error::BadWrap { offset });
                }
                self.state.wrap_state = match ending {
                    LineEnding::Lf => WrapState::NeedSymbol { offset },
                    LineEnding::CrLf => WrapState::NeedLf { offset },
                };
                return Ok(());
            }

            if matches!(byte, b'\r' | b'\n' | 9 | 11 | 12 | 32) {
                return Err(Error::BadWrap { offset });
            }
        }

        if let Some(limit) = self.state.options.max_symbols {
            if self.state.symbol_count >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.state.symbol_count += 1;

        if self.state.terminal_padding {
            return Err(Error::BadPadding { offset });
        }
        if self.state.pending_second_pad.is_some() && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        let position = self.state.group.len();
        if byte == b'=' {
            if position < 2 {
                return Err(Error::BadPadding { offset });
            }
            if self.state.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            if position == 2 {
                self.state.pending_second_pad = Some(offset);
                self.state.group.push((byte, offset));
                self.bump_row_count();
                return Ok(());
            }
            if position == 3 {
                if self.state.pending_second_pad.is_some() {
                    self.state.group.push((byte, offset));
                    self.complete_padded_group(output)?;
                    return Ok(());
                }

                self.state.group.push((byte, offset));
                self.complete_padded_group(output)?;
                return Ok(());
            }
        }

        let value = match alphabet_value(self.state.options.alphabet, byte) {
            Some(value) => value,
            None if self.state.options.strict
                && foreign_alphabet_byte(self.state.options.alphabet, byte) =>
            {
                return Err(Error::MixedAlphabet { offset });
            }
            None => return Err(Error::InvalidByte { offset, byte }),
        };

        if self.state.pending_second_pad.is_some() {
            return Err(Error::BadPadding { offset });
        }
        if position >= 4 {
            return Err(Error::BadPadding { offset });
        }
        self.state.group.push((value, offset));
        self.bump_row_count();

        if self.state.group.len() == 4 {
            let vals = [
                self.state.group[0].0,
                self.state.group[1].0,
                self.state.group[2].0,
                self.state.group[3].0,
            ];
            output.extend_from_slice(&[
                (vals[0] << 2) | (vals[1] >> 4),
                (vals[1] << 4) | (vals[2] >> 2),
                (vals[2] << 6) | vals[3],
            ]);
            self.state.group.clear();
        }
        Ok(())
    }

    fn complete_padded_group(&mut self, output: &mut Vec<u8>) -> Result<(), Error> {
        let group = &self.state.group;
        if group.len() != 4 {
            return Err(Error::BadPadding {
                offset: group.last().map_or(self.state.offset, |x| x.1),
            });
        }

        let a = group[0].0;
        let b = group[1].0;
        if group[2].0 == b'=' && group[3].0 == b'=' {
            if self.state.options.strict && b & 0x0f != 0 {
                return Err(Error::NonCanonical { offset: group[1].1 });
            }
            output.push((a << 2) | (b >> 4));
        } else if group[2].0 != b'=' && group[3].0 == b'=' {
            let c = group[2].0;
            if self.state.options.strict && c & 0x03 != 0 {
                return Err(Error::NonCanonical { offset: group[2].1 });
            }
            output.extend_from_slice(&[(a << 2) | (b >> 4), (b << 4) | (c >> 2)]);
        } else {
            return Err(Error::BadPadding {
                offset: group[3].1,
            });
        }

        self.state.group.clear();
        self.state.pending_second_pad = None;
        self.state.terminal_padding = true;
        Ok(())
    }

    fn finish_inner(&mut self) -> Result<Vec<u8>, Error> {
        let mut output = Vec::new();

        if let WrapState::NeedSeparatorFirst { offset }
        | WrapState::NeedSymbol { offset }
        | WrapState::NeedLf { offset } = self.state.wrap_state
        {
            return Err(Error::BadWrap { offset });
        }

        if let Some(offset) = self.state.pending_second_pad {
            return Err(Error::BadPadding { offset });
        }

        let len = self.state.group.len();
        if len == 0 {
            return Ok(output);
        }

        if len == 4 {
            return Ok(output);
        }

        if self.state.options.padding == Padding::Required {
            return Err(Error::Truncated {
                offset: self.state.group[0].1,
            });
        }

        if len == 1 {
            return Err(Error::Truncated {
                offset: self.state.group[0].1,
            });
        }

        let values = &self.state.group;
        if len == 2 {
            if self.state.options.strict && values[1].0 & 0x0f != 0 {
                return Err(Error::NonCanonical { offset: values[1].1 });
            }
            output.push((values[0].0 << 2) | (values[1].0 >> 4));
        } else if len == 3 {
            if self.state.options.strict && values[2].0 & 0x03 != 0 {
                return Err(Error::NonCanonical { offset: values[2].1 });
            }
            output.extend_from_slice(&[
                (values[0].0 << 2) | (values[1].0 >> 4),
                (values[1].0 << 4) | (values[2].0 >> 2),
            ]);
        }
        Ok(output)
    }

    fn row_symbol_count(&self) -> usize {
        // Wrapped widths are multiples of four, so the number of symbols already
        // consumed modulo width determines the current row length.
        if let Whitespace::Wrapped { width, .. } = self.state.options.whitespace {
            self.state.symbol_count % width
        } else {
            0
        }
    }

    fn bump_row_count(&mut self) {}
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate_options(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut result = encoder.push(input)?;
    result.push_str(&encoder.finish()?);
    Ok(result)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate_options(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut result = decoder.push(input)?;
    result.extend_from_slice(&decoder.finish()?);
    Ok(result)
}

fn main() {}
