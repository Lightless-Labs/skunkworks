#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

pub struct Fs {
    entries: std::collections::BTreeMap<String, Entry>,
    cwd: Vec<String>,
}

enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = std::collections::BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: Vec::new(),
        }
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            self.cwd.clone()
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                other => components.push(other.to_string()),
            }
        }
        Ok(components)
    }

    fn lookup(&self, components: &[String]) -> Result<&Entry, Error> {
        let mut prefix = Vec::new();

        for component in components {
            let prefix_path = Self::canonical(&prefix);
            match self.entries.get(&prefix_path) {
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix_path)),
                None => return Err(Error::Missing(prefix_path)),
            }

            prefix.push(component.clone());
            let path = Self::canonical(&prefix);
            if !self.entries.contains_key(&path) {
                return Err(Error::Missing(path));
            }
        }

        self.entries
            .get(&Self::canonical(components))
            .ok_or_else(|| Error::Missing(Self::canonical(components)))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        let canonical = Self::canonical(&components);

        if components.is_empty() {
            return Err(Error::Exists(canonical));
        }

        let parent = &components[..components.len() - 1];
        match self.lookup(parent)? {
            Entry::Dir => {}
            Entry::File(_) => return Err(Error::NotDir(Self::canonical(parent))),
        }

        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }

        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = self.normalize(path)?;
        let canonical = Self::canonical(&components);

        if components.is_empty() {
            return Err(Error::IsDir(canonical));
        }

        let parent = &components[..components.len() - 1];
        match self.lookup(parent)? {
            Entry::Dir => {}
            Entry::File(_) => return Err(Error::NotDir(Self::canonical(parent))),
        }

        match self.entries.get(&canonical) {
            Some(Entry::Dir) => return Err(Error::IsDir(canonical)),
            Some(Entry::File(_)) | None => {}
        }

        self.entries
            .insert(canonical, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = self.normalize(path)?;
        let canonical = Self::canonical(&components);

        match self.lookup(&components)? {
            Entry::Dir => Err(Error::IsDir(canonical)),
            Entry::File(data) => Ok(data.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = self.normalize(path)?;
        let canonical = Self::canonical(&components);

        match self.lookup(&components)? {
            Entry::File(_) => return Err(Error::NotDir(canonical.clone())),
            Entry::Dir => {}
        }

        let mut children = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == &canonical {
                continue;
            }

            let basename = if canonical == "/" {
                match entry_path.strip_prefix('/') {
                    Some(rest) => rest,
                    None => continue,
                }
            } else {
                let prefix = format!("{}/", canonical);
                match entry_path.strip_prefix(&prefix) {
                    Some(rest) => rest,
                    None => continue,
                }
            };

            if basename.is_empty() || basename.contains('/') {
                continue;
            }

            let kind = match entry {
                Entry::Dir => Kind::Dir,
                Entry::File(_) => Kind::File,
            };
            children.push((basename.to_string(), kind));
        }

        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        Self::canonical(&self.cwd)
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        let canonical = Self::canonical(&components);

        match self.lookup(&components)? {
            Entry::Dir => {
                self.cwd = components;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(canonical)),
        }
    }
}

fn main() {}
