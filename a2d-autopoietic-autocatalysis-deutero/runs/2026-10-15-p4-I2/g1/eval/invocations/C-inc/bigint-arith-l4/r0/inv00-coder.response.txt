```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    digits: Vec<u8>, // little-endian, base 10
}

impl Big {
    fn zero() -> Self {
        Self { negative: false, digits: vec![0] }
    }

    fn one() -> Self {
        Self { negative: false, digits: vec![1] }
    }

    fn normalize(&mut self) {
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.is_empty() {
            self.digits.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => {
                for (a, b) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u16;
        for i in 0..a.len().max(b.len()) {
            let sum = a.get(i).copied().unwrap_or(0) as u16
                + b.get(i).copied().unwrap_or(0) as u16
                + carry;
            out.push((sum % 10) as u8);
            carry = sum / 10;
        }
        if carry != 0 {
            out.push(carry as u8);
        }
        out
    }

    // Requires a >= b.
    fn sub_abs(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i16;
        for i in 0..a.len() {
            let mut d = a[i] as i16 - b.get(i).copied().unwrap_or(0) as i16 - borrow;
            if d < 0 {
                d += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(d as u8);
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_small(&self, n: u32) -> Self {
        if n == 0 || self.is_zero() {
            return Self::zero();
        }
        let mut out = Vec::with_capacity(self.digits.len() + 10);
        let mut carry = 0u64;
        for &d in &self.digits {
            let v = d as u64 * n as u64 + carry;
            out.push((v % 10) as u8);
            carry = v / 10;
        }
        while carry > 0 {
            out.push((carry % 10) as u8);
            carry /= 10;
        }
        let mut result = Self { negative: false, digits: out };
        result.normalize();
        result
    }

    fn add_small(&mut self, n: u32) {
        let mut carry = n as u64;
        let mut i = 0;
        while carry > 0 {
            if i == self.digits.len() {
                self.digits.push(0);
            }
            let v = self.digits[i] as u64 + carry;
            self.digits[i] = (v % 10) as u8;
            carry = v / 10;
            i += 1;
        }
        self.normalize();
    }

    fn div_rem_abs(&self, rhs: &Self) -> (Self, Self) {
        let mut quotient_be = Vec::with_capacity(self.digits.len());
        let mut rem = Self::zero();
        for &digit in self.digits.iter().rev() {
            rem.digits.insert(0, digit);
            rem.normalize();

            let mut qdigit = 0u8;
            while rem.abs_cmp(rhs) != Ordering::Less {
                rem.digits = Self::sub_abs(&rem.digits, &rhs.digits);
                qdigit += 1;
            }
            quotient_be.push(qdigit);
        }
        quotient_be.reverse();
        let mut q = Self { negative: false, digits: quotient_be };
        q.normalize();
        rem.normalize();
        (q, rem)
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (start, negative) = match bytes[0] {
            b'+' => (1, false),
            b'-' => (1, true),
            _ => (0, false),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                continue;
            }
            let v = match bytes[i] {
                b'0'..=b'9' => (bytes[i] - b'0') as u32,
                b'a'..=b'z' => (bytes[i] - b'a' + 10) as u32,
                b'A'..=b'Z' => (bytes[i] - b'A' + 10) as u32,
                _ => radix,
            };
            if v >= radix {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || bytes[i - 1] == b'_'
                    || bytes[i + 1] == b'_')
            {
                return Err(Error::Separator { index: i });
            }
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        let first_value = match bytes[first_digit_index] {
            b'0' => 0,
            b'1'..=b'9' => 1,
            b'a'..=b'z' => 10,
            b'A'..=b'Z' => 10,
            _ => unreachable!(),
        };
        if digit_count > 1 && first_value == 0 {
            return Err(Error::LeadingZero { index: first_digit_index });
        }
        if digit_count > max_digits {
            let index = (start..bytes.len())
                .filter(|&i| bytes[i] != b'_')
                .nth(max_digits)
                .unwrap();
            return Err(Error::TooManyDigits { limit: max_digits, index });
        }

        let mut value = Self::zero();
        for &b in &bytes[start..] {
            if b == b'_' {
                continue;
            }
            let d = match b {
                b'0'..=b'9' => (b - b'0') as u32,
                b'a'..=b'z' => (b - b'a' + 10) as u32,
                b'A'..=b'Z' => (b - b'A' + 10) as u32,
                _ => unreachable!(),
            };
            value = value.mul_small(radix);
            value.add_small(d);
        }
        value.negative = negative && !value.is_zero();
        Ok(value)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut n = Self { negative: false, digits: self.digits.clone() };
        n.normalize();
        let mut out_digits = Vec::new();
        if n.is_zero() {
            out_digits.push(0);
        } else {
            let divisor = Self::from_u32(radix);
            while !n.is_zero() {
                let (q, r) = n.div_rem_abs(&divisor);
                out_digits.push(r.digits.first().copied().unwrap_or(0) as u32);
                n = q;
            }
        }
        while out_digits.len() < min_digits.max(1) {
            out_digits.push(0);
        }
        out_digits.reverse();

        let mut s = String::new();
        if self.negative && !self.is_zero() {
            s.push('-');
        }
        for (i, &d) in out_digits.iter().enumerate() {
            if group > 0 && i > 0 && (out_digits.len() - i) % group == 0 {
                s.push('_');
            }
            s.push(if d < 10 { (b'0' + d as u8) as char } else { (b'a' + (d - 10) as u8) as char });
        }
        Ok(s)
    }

    fn from_u32(mut n: u32) -> Self {
        if n == 0 {
            return Self::zero();
        }
        let mut digits = Vec::new();
        while n > 0 {
            digits.push((n % 10) as u8);
            n /= 10;
        }
        Self { negative: false, digits }
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut result = if self.negative == rhs.negative {
            Self { negative: self.negative, digits: Self::add_abs(&self.digits, &rhs.digits) }
        } else {
            match self.abs_cmp(rhs) {
                Ordering::Greater => Self { negative: self.negative, digits: Self::sub_abs(&self.digits, &rhs.digits) },
                Ordering::Less => Self { negative: rhs.negative, digits: Self::sub_abs(&rhs.digits, &self.digits) },
                Ordering::Equal => Self::zero(),
            }
        };
        result.normalize();
        result
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            let mut carry = 0u32;
            for (j, &b) in rhs.digits.iter().enumerate() {
                let k = i + j;
                let v = out[k] + a as u32 * b as u32 + carry;
                out[k] = v % 10;
                carry = v / 10;
            }
            let mut k = i + rhs.digits.len();
            while carry > 0 {
                let v = out[k] + carry;
                out[k] = v % 10;
                carry = v / 10;
                k += 1;
                if k == out.len() && carry > 0 {
                    out.push(0);
                }
            }
        }
        let mut result = Self {
            negative: self.negative != rhs.negative,
            digits: out.into_iter().map(|d| d as u8).collect(),
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let a = Self { negative: false, digits: self.digits.clone() };
        let b = Self { negative: false, digits: rhs.digits.clone() };
        let (mut q, mut r) = a.div_rem_abs(&b);
        q.negative = !q.is_zero() && (self.negative != rhs.negative);
        r.negative = !r.is_zero() && self.negative;
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > 1024 {
            return Err(Error::ExponentTooLarge { exponent, limit: 1024 });
        }
        let mut n = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while n > 0 {
            if n & 1 == 1 {
                result = result.mul(&base);
            }
            n >>= 1;
            if n > 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = Self { negative: false, digits: self.digits.clone() };
        let mut b = Self { negative: false, digits: rhs.digits.clone() };
        while !b.is_zero() {
            let (_, r) = a.div_rem_abs(&b);
            a = b;
            b = r;
        }
        a.negative = false;
        a.normalize();
        a
    }
}

fn main() {}
```
