```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum EventKind {
    Arrival,
    Wake,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
    level: usize,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let count = jobs.len();
    if count == 0 {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut events = Vec::new();
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: job.arrival as u64,
            kind: EventKind::Arrival,
            index,
            level: job.level,
        });
    }
    events.sort_by_key(|event| {
        (
            event.time,
            match event.kind {
                EventKind::Arrival => 0_u8,
                EventKind::Wake => 1_u8,
            },
            event.index,
        )
    });

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut burst_position = vec![0usize; count];
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.bursts[0]).collect();
    let mut levels: Vec<usize> = jobs.iter().map(|job| job.level).collect();
    let mut completions = vec![0u64; count];

    let mut runs = Vec::new();
    let mut idle: Vec<(u64, u64)> = Vec::new();
    let mut now = 0u64;
    let mut event_pos = 0usize;
    let mut completed = 0usize;
    let mut last_id: Option<u32> = None;

    while completed < count {
        while event_pos < events.len() && events[event_pos].time <= now {
            let event = events[event_pos];
            queues[event.level].push_back(event.index);
            event_pos += 1;
        }

        let Some(level) = queues.iter().position(|queue| !queue.is_empty()) else {
            if event_pos < events.len() {
                let next_time = events[event_pos].time;
                if next_time > now {
                    if let Some(last) = idle.last_mut() {
                        if last.1 == now {
                            last.1 = next_time;
                        } else {
                            idle.push((now, next_time));
                        }
                    } else {
                        idle.push((now, next_time));
                    }
                    now = next_time;
                    last_id = None;
                }
                continue;
            }
            break;
        };

        let index = queues[level].pop_front().unwrap();
        let id = jobs[index].id;

        if last_id.is_some() && last_id != Some(id) {
            now += switch_cost as u64;
        }

        while event_pos < events.len() && events[event_pos].time <= now {
            let event = events[event_pos];
            queues[event.level].push_back(event.index);
            event_pos += 1;
        }

        let position = burst_position[index];
        let consumed = remaining[index].min(quanta[level]);
        let start = now;
        now += consumed as u64;
        remaining[index] -= consumed;

        let burst_ended = remaining[index] == 0;
        let finished = burst_ended && position + 1 == jobs[index].bursts.len();

        runs.push(Run {
            id,
            start,
            end: now,
            finished,
            level,
        });

        while event_pos < events.len() && events[event_pos].time <= now {
            let event = events[event_pos];
            queues[event.level].push_back(event.index);
            event_pos += 1;
        }

        if finished {
            completed += 1;
            completions[index] = now;
        } else if burst_ended {
            let io_delay = jobs[index].bursts[position + 1] as u64;
            burst_position[index] += 2;
            remaining[index] = jobs[index].bursts[position + 2];

            events.push(Event {
                time: now + io_delay,
                kind: EventKind::Wake,
                index,
                level: level.saturating_sub(1),
            });
            events[event_pos..].sort_by_key(|event| {
                (
                    event.time,
                    match event.kind {
                        EventKind::Arrival => 0_u8,
                        EventKind::Wake => 1_u8,
                    },
                    event.index,
                )
            });

            while event_pos < events.len() && events[event_pos].time <= now {
                let event = events[event_pos];
                queues[event.level].push_back(event.index);
                event_pos += 1;
            }
        } else {
            levels[index] = (level + 1).min(quanta.len() - 1);
            queues[levels[index]].push_back(index);
        }

        last_id = Some(id);
    }

    let mut stats = Vec::with_capacity(count);
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = completions[index];
        let turnaround = completion - job.arrival as u64;
        let cpu_total: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&value| value as u64)
            .sum();
        let io_total: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&value| value as u64)
            .sum();
        let waiting = turnaround - cpu_total - io_total;

        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let divisor = count as u64;
    let mean_waiting_x100 = (total_waiting * 100 + divisor / 2) / divisor;
    let mean_turnaround_x100 = (total_turnaround * 100 + divisor / 2) / divisor;

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end: now,
    })
}

fn main() {}
```
