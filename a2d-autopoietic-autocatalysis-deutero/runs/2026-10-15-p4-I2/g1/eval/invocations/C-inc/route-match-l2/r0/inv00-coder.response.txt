#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    segments: Vec<Segment>,
    has_wildcard: bool,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let parsed = parse_pattern(pattern)?;
        let id = self.routes.len();
        let has_wildcard = parsed.iter().any(|s| matches!(s, Segment::Wildcard(_)));
        let mut rank: Vec<u8> = parsed
            .iter()
            .map(|s| match s {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            })
            .collect();
        if !has_wildcard {
            rank.push(3);
        }
        self.routes.push(Route {
            id,
            segments: parsed,
            has_wildcard,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let request = parse_path(head)?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;

        for route in &self.routes {
            let mut params = Vec::new();
            let mut pi = 0usize;
            let mut ri = 0usize;
            let mut matched = true;

            while pi < route.segments.len() {
                match &route.segments[pi] {
                    Segment::Wildcard(name) => {
                        let value = request[ri..]
                            .iter()
                            .map(|s| s.as_str())
                            .collect::<Vec<_>>()
                            .join("/");
                        params.push((name.clone(), value));
                        ri = request.len();
                        pi += 1;
                        break;
                    }
                    Segment::Static(expected) => {
                        if ri >= request.len() || request[ri] != *expected {
                            matched = false;
                            break;
                        }
                    }
                    Segment::Param(name) => {
                        if ri >= request.len() {
                            matched = false;
                            break;
                        }
                        params.push((name.clone(), request[ri].clone()));
                    }
                }
                pi += 1;
                ri += 1;
            }

            if matched && ri == request.len() && pi == route.segments.len() {
                match best {
                    None => best = Some((route, params)),
                    Some((old, _)) if route.rank > old.rank => best = Some((route, params)),
                    _ => {}
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    let bytes = pattern.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let mut ranges = Vec::new();
    let mut start = 1usize;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                return Err(Error::InvalidPattern { at: i });
            }
            if i == start {
                return Err(Error::InvalidPattern { at: start });
            }
            ranges.push((start, i));
            start = i + 1;
        }
    }
    if start == bytes.len() {
        return Err(Error::InvalidPattern { at: bytes.len() - 1 });
    }
    ranges.push((start, bytes.len()));

    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();

    for (index, (begin, end)) in ranges.iter().copied().enumerate() {
        let raw = &pattern[begin..end];
        if raw.starts_with(':') || raw.starts_with('*') {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: begin });
            }
            if marker == b'*' && index + 1 != ranges.len() {
                return Err(Error::InvalidPattern { at: begin });
            }
            if marker == b':' {
                segments.push(Segment::Param(name.to_owned()));
            } else {
                segments.push(Segment::Wildcard(name.to_owned()));
            }
        } else {
            segments.push(Segment::Static(decode_segment(pattern, begin, end, true)?));
        }
    }
    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    let bytes = path.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let mut ranges = Vec::new();
    let mut start = 1usize;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                return Err(Error::InvalidPath { at: i });
            }
            if i == start {
                return Err(Error::InvalidPath { at: start });
            }
            ranges.push((start, i));
            start = i + 1;
        }
    }
    if start == bytes.len() {
        return Err(Error::InvalidPath {
            at: bytes.len() - 1,
        });
    }
    ranges.push((start, bytes.len()));

    ranges
        .into_iter()
        .map(|(begin, end)| decode_segment(path, begin, end, false).map_err(|e| match e {
            Error::InvalidPattern { at } | Error::InvalidPath { at } => Error::InvalidPath { at },
            Error::NotFound => Error::NotFound,
        }))
        .collect()
}

fn decode_segment(source: &str, begin: usize, end: usize, pattern: bool) -> Result<String, Error> {
    let bytes = source.as_bytes();
    let mut out = Vec::new();
    let mut i = begin;
    while i < end {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= end {
                return invalid(pattern, i);
            }
            let hi = hex_upper(bytes[i + 1]);
            let lo = hex_upper(bytes[i + 2]);
            let Some(value) = hi.zip(lo).map(|(h, l)| h * 16 + l) else {
                return invalid(pattern, i);
            };
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return invalid(pattern, i);
            }
            out.push(value);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return invalid(pattern, i);
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn invalid<T>(pattern: bool, at: usize) -> Result<T, Error> {
    if pattern {
        Err(Error::InvalidPattern { at })
    } else {
        Err(Error::InvalidPath { at })
    }
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn main() {}
