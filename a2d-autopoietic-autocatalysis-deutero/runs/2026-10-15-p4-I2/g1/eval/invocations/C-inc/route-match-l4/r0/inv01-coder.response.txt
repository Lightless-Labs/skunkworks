use std::collections::{BTreeSet, HashMap, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
    names: HashMap<String, usize>,
}

struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn decode_segment(s: &str, base: usize) -> Result<String, usize> {
    let bytes = s.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hex = |x: u8| -> Option<u8> {
                match x {
                    b'0'..=b'9' => Some(x - b'0'),
                    b'A'..=b'F' => Some(x - b'A' + 10),
                    _ => None,
                }
            };
            let (Some(a), Some(c)) = (hex(bytes[i + 1]), hex(bytes[i + 2])) else {
                return Err(base + i);
            };
            let decoded = (a << 4) | c;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(out).unwrap())
}

fn valid_capture_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b[0].is_ascii_lowercase()
        && b[1..]
            .iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_')
}

fn valid_route_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b[0].is_ascii_lowercase()
        && b[1..].iter().all(|c| {
            c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_' || *c == b'-'
        })
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.as_bytes().iter().all(u8::is_ascii_uppercase)
}

fn parse_path_head(input: &str, path_error: bool) -> Result<ParsedPath, Error> {
    let invalid = |at| {
        if path_error {
            Error::InvalidPath { at }
        } else {
            Error::InvalidPattern { at }
        }
    };

    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing: false,
        });
    }

    let trailing = input.ends_with('/');
    let body_end = if trailing { input.len() - 1 } else { input.len() };
    let body = &input[1..body_end];
    if body.is_empty() {
        return Err(invalid(1));
    }

    let mut segments = Vec::new();
    let mut start = 0;
    for part in body.split('/') {
        if part.is_empty() {
            return Err(invalid(1 + start));
        }
        let decoded = decode_segment(part, 1 + start).map_err(invalid)?;
        segments.push(decoded);
        start += part.len() + 1;
    }

    Ok(ParsedPath { segments, trailing })
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool, Vec<u8>), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false, vec![3]));
    }

    let trailing = pattern.ends_with('/');
    let body_end = if trailing {
        pattern.len() - 1
    } else {
        pattern.len()
    };
    let body = &pattern[1..body_end];
    if body.is_empty() {
        return Err(Error::InvalidPattern { at: 1 });
    }

    let parts: Vec<&str> = body.split('/').collect();
    let mut segments = Vec::new();
    let mut seen = HashSet::new();
    let mut rank = Vec::new();
    let mut start = 1usize;

    for (idx, part) in parts.iter().enumerate() {
        if part.is_empty() {
            return Err(Error::InvalidPattern { at: start });
        }
        let is_last = idx + 1 == parts.len();

        if let Some(name) = part.strip_prefix(':') {
            if !valid_capture_name(name) || !seen.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(Segment::Param(name.to_string()));
            rank.push(1);
        } else if let Some(name) = part.strip_prefix('*') {
            if !is_last {
                return Err(Error::InvalidPattern { at: start });
            }
            if !valid_capture_name(name) || !seen.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(Segment::Wildcard(name.to_string()));
            rank.push(0);
        } else {
            let decoded =
                decode_segment(part, start).map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }
        start += part.len() + 1;
    }

    if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
        rank.push(3);
    }
    Ok((segments, trailing, rank))
}

fn same_shape(a: &Route, b_segments: &[Segment], b_trailing: bool, policy: SlashPolicy) -> bool {
    if a.segments.len() != b_segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b_trailing {
        return false;
    }
    a.segments.iter().zip(b_segments).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn encode_segment(value: &str) -> String {
    let mut out = String::new();
    for &b in value.as_bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
            names: HashMap::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing, rank) = parse_pattern(pattern)?;

        if self.names.contains_key(name) {
            return Err(Error::DuplicateName);
        }

        if let Some(route) = self.routes.iter().find(|route| {
            route.method == method && same_shape(route, &segments, trailing, self.policy)
        }) {
            return Err(Error::Conflict { with: route.id });
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            segments,
            trailing,
            rank,
        });
        self.names.insert(name.to_string(), id);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_path_head(head, true)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                continue;
            }
            if let Some(params) = match_route(route, &parsed.segments) {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<_> = candidates
            .iter()
            .filter(|(route, _)| route.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (route, _) in &candidates {
                allowed.insert(route.method.clone());
                if route.method == "GET" {
                    allowed.insert("HEAD".to_string());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        eligible.sort_by(|(a, _), (b, _)| {
            b.rank.cmp(&a.rank).then_with(|| a.id.cmp(&b.id))
        });
        let (route, params) = eligible[0];

        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != parsed.trailing
            && !parsed.segments.is_empty()
        {
            let mut target = head.to_string();
            if route.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let Some(&id) = self.names.get(name) else {
            return Err(Error::UnknownName);
        };
        let route = &self.routes[id];

        let mut supplied = HashMap::new();
        let mut repeated = HashSet::new();
        for &(key, value) in params {
            if !repeated.insert(key.to_string()) {
                return Err(Error::DuplicateParam {
                    name: key.to_string(),
                });
            }
            supplied.insert(key.to_string(), value);
        }

        let mut required = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(key) | Segment::Wildcard(key) => required.push(key.as_str()),
                Segment::Static(_) => {}
            }
        }
        for key in &required {
            if !supplied.contains_key(*key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let required_set: HashSet<&str> = required.iter().copied().collect();
        if let Some(extra) = supplied
            .keys()
            .filter(|key| !required_set.contains(key.as_str()))
            .min_by(|a, b| a.as_bytes().cmp(b.as_bytes()))
        {
            return Err(Error::ExtraParam {
                name: extra.clone(),
            });
        }

        for key in &required {
            let value = supplied.get(*key).unwrap();
            let bytes = value.as_bytes();
            let valid_ascii = bytes.iter().all(|b| (0x20..=0x7e).contains(b));
            let is_wildcard = route
                .segments
                .iter()
                .any(|segment| matches!(segment, Segment::Wildcard(n) if n == key));
            let valid = if is_wildcard {
                valid_ascii && (value.is_empty() || value.split('/').all(|part| !part.is_empty()))
            } else {
                valid_ascii && !value.is_empty() && !value.contains('/')
            };
            if !valid {
                return Err(Error::InvalidValue {
                    name: (*key).to_string(),
                });
            }
        }

        let mut output = String::new();
        for segment in &route.segments {
            output.push('/');
            match segment {
                Segment::Static(value) => output.push_str(&encode_segment(value)),
                Segment::Param(key) => output.push_str(&encode_segment(supplied[key.as_str()])),
                Segment::Wildcard(key) => {
                    let value = supplied[key.as_str()];
                    if !value.is_empty() {
                        for (i, component) in value.split('/').enumerate() {
                            if i != 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_segment(component));
                        }
                    }
                }
            }
        }
        if output.is_empty() {
            output.push('/');
        } else if route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard = matches!(route.segments.last(), Some(Segment::Wildcard(_)));
    let fixed_len = if wildcard {
        route.segments.len() - 1
    } else {
        route.segments.len()
    };

    if (!wildcard && path.len() != fixed_len) || (wildcard && path.len() < fixed_len) {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed_len {
        match &route.segments[i] {
            Segment::Static(value) if value == &path[i] => {}
            Segment::Param(name) => params.push((name.clone(), path[i].clone())),
            _ => return None,
        }
    }

    if wildcard {
        if let Segment::Wildcard(name) = &route.segments[fixed_len] {
            params.push((name.clone(), path[fixed_len..].join("/")));
        }
    }
    Some(params)
}

fn main() {}
