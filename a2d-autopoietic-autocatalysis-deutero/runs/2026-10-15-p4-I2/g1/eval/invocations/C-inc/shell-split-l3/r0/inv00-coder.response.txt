#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn append(word: &mut Word, value: &str, glob_active: bool) {
    for c in value.chars() {
        if glob_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn append_scalar(word: &mut Word, c: char, glob_active: bool) {
    if glob_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                out.push(next);
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let n = c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_scalar(&mut word, c, false);
                }
                i += n;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    i += n;
                } else if c == '\\' {
                    let pos = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_scalar(&mut word, next, false);
                        }
                        i += next.len_utf8();
                    } else {
                        append_scalar(&mut word, '\\', false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[i + 1..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + 1..].starts_with('{') {
                        let dollar = i;
                        let content_start = i + 2;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let d = input[j..].chars().next().unwrap();
                            if d == '\\' {
                                j += 1;
                                if j < input.len() {
                                    j += input[j..].chars().next().unwrap().len_utf8();
                                }
                            } else if d == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j += d.len_utf8();
                            }
                        }
                        let Some(close) = close else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar });
                        };
                        let contents = &input[content_start..close];
                        let (name, default) = if let Some(k) = contents.find(":-") {
                            (&contents[..k], Some(&contents[k + 2..]))
                        } else {
                            (contents, None)
                        };
                        if name.is_empty() || !name.chars().next().is_some_and(is_name_start)
                            || !name.chars().all(is_name_continue)
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let selected = match default {
                            Some(text) => match env_value(env, name) {
                                Some(value) if !value.is_empty() => value.to_string(),
                                _ => decode_default(text),
                            },
                            None => env_value(env, name).unwrap_or("").to_string(),
                        };
                        append(&mut word, &selected, false);
                        active = true;
                        i = close + 1;
                    } else if i + 1 < input.len()
                        && input[i + 1..].chars().next().is_some_and(is_name_start)
                    {
                        let start = i + 1;
                        let mut end = start;
                        while end < input.len() {
                            let d = input[end..].chars().next().unwrap();
                            if !is_name_continue(d) {
                                break;
                            }
                            end += d.len_utf8();
                        }
                        append(&mut word, env_value(env, &input[start..end]).unwrap_or(""), false);
                        active = true;
                        i = end;
                    } else {
                        append_scalar(&mut word, '$', false);
                        i += 1;
                    }
                } else {
                    append_scalar(&mut word, c, false);
                    i += n;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += n;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i += n;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i += n;
                } else if c == '\\' {
                    let pos = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next != '\n' {
                        append_scalar(&mut word, next, false);
                        active = true;
                    }
                    i += next.len_utf8();
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[i + 1..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + 1..].starts_with('{') {
                        let dollar = i;
                        let content_start = i + 2;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let d = input[j..].chars().next().unwrap();
                            if d == '\\' {
                                j += 1;
                                if j < input.len() {
                                    j += input[j..].chars().next().unwrap().len_utf8();
                                }
                            } else if d == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j += d.len_utf8();
                            }
                        }
                        let Some(close) = close else {
                            return Err(SplitError::UnterminatedExpansion { pos: dollar });
                        };
                        let contents = &input[content_start..close];
                        let (name, default) = if let Some(k) = contents.find(":-") {
                            (&contents[..k], Some(&contents[k + 2..]))
                        } else {
                            (contents, None)
                        };
                        if name.is_empty() || !name.chars().next().is_some_and(is_name_start)
                            || !name.chars().all(is_name_continue)
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let selected = match default {
                            Some(text) => match env_value(env, name) {
                                Some(value) if !value.is_empty() => value.to_string(),
                                _ => decode_default(text),
                            },
                            None => env_value(env, name).unwrap_or("").to_string(),
                        };
                        append(&mut word, &selected, true);
                        active = true;
                        i = close + 1;
                    } else if i + 1 < input.len()
                        && input[i + 1..].chars().next().is_some_and(is_name_start)
                    {
                        let start = i + 1;
                        let mut end = start;
                        while end < input.len() {
                            let d = input[end..].chars().next().unwrap();
                            if !is_name_continue(d) {
                                break;
                            }
                            end += d.len_utf8();
                        }
                        append(&mut word, env_value(env, &input[start..end]).unwrap_or(""), true);
                        active = true;
                        i = end;
                    } else {
                        append_scalar(&mut word, '$', false);
                        active = true;
                        i += 1;
                    }
                } else {
                    append_scalar(&mut word, c, true);
                    active = true;
                    i += n;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn main() {}
