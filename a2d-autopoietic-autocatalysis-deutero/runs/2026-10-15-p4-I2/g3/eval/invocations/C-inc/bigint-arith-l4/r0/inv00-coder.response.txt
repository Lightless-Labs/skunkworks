```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1,000,000,000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;

    fn zero() -> Self {
        Self { negative: false, limbs: vec![0] }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn normalize(&mut self) {
        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.limbs.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn abs(&self) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(&self, rhs: &Self) -> Self {
        let n = self.limbs.len().max(rhs.limbs.len());
        let mut out = Vec::with_capacity(n + 1);
        let mut carry = 0u64;
        for i in 0..n {
            let a = self.limbs.get(i).copied().unwrap_or(0) as u64;
            let b = rhs.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = a + b + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        let mut result = Self { negative: false, limbs: out };
        result.normalize();
        result
    }

    // Requires |self| >= |rhs|.
    fn sub_abs(&self, rhs: &Self) -> Self {
        let mut out = Vec::with_capacity(self.limbs.len());
        let mut borrow = 0i64;
        for i in 0..self.limbs.len() {
            let a = self.limbs[i] as i64;
            let b = rhs.limbs.get(i).copied().unwrap_or(0) as i64;
            let mut digit = a - b - borrow;
            if digit < 0 {
                digit += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(digit as u32);
        }
        let mut result = Self { negative: false, limbs: out };
        result.normalize();
        result
    }

    fn mul_small(&mut self, factor: u32) {
        if factor == 0 {
            *self = Self::zero();
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = *limb as u64 * factor as u64 + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
        self.normalize();
    }

    fn add_small(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let sum = self.limbs[i] as u64 + carry;
            self.limbs[i] = (sum % Self::BASE) as u32;
            carry = sum / Self::BASE;
            i += 1;
        }
    }

    fn div_rem_small(&self, divisor: u32) -> (Self, u32) {
        let mut quotient = vec![0; self.limbs.len()];
        let mut remainder = 0u64;
        for i in (0..self.limbs.len()).rev() {
            let current = remainder * Self::BASE + self.limbs[i] as u64;
            quotient[i] = (current / divisor as u64) as u32;
            remainder = current % divisor as u64;
        }
        let mut q = Self { negative: false, limbs: quotient };
        q.normalize();
        (q, remainder as u32)
    }

    fn mul_abs(&self, rhs: &Self) -> Self {
        let mut out = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let k = i + j;
                let value = out[k] as u64 + a as u64 * b as u64 + carry;
                out[k] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
            }
            let mut k = i + rhs.limbs.len();
            while carry != 0 {
                if k == out.len() {
                    out.push(0);
                }
                let value = out[k] as u64 + carry;
                out[k] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
                k += 1;
            }
        }
        let mut result = Self { negative: false, limbs: out };
        result.normalize();
        result
    }

    fn shift_base_add(&mut self, digit: u32) {
        if self.is_zero() {
            self.limbs[0] = digit;
        } else {
            self.limbs.insert(0, digit);
        }
        self.normalize();
    }

    fn div_rem_abs(&self, rhs: &Self) -> (Self, Self) {
        let mut quotient = vec![0u32; self.limbs.len()];
        let mut remainder = Self::zero();
        for i in (0..self.limbs.len()).rev() {
            remainder.shift_base_add(self.limbs[i]);
            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = (low + high) / 2;
                let product = rhs.mul_small_copy(mid as u32);
                match product.abs_cmp(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 { break; }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        low = mid + 1;
                    }
                }
            }
            quotient[i] = best as u32;
            if best != 0 {
                remainder = remainder.sub_abs(&rhs.mul_small_copy(best as u32));
            }
        }
        let mut q = Self { negative: false, limbs: quotient };
        q.normalize();
        remainder.normalize();
        (q, remainder)
    }

    fn mul_small_copy(&self, factor: u32) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result.mul_small(factor);
        result
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut digit_values = vec![None; bytes.len()];
        let mut invalid = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                continue;
            }
            let value = match bytes[i] {
                b'0'..=b'9' => Some((bytes[i] - b'0') as u32),
                b'a'..=b'z' => Some((bytes[i] - b'a' + 10) as u32),
                b'A'..=b'Z' => Some((bytes[i] - b'A' + 10) as u32),
                _ => None,
            };
            match value {
                Some(v) if v < radix => digit_values[i] = Some(v),
                _ => {
                    if invalid.is_none() {
                        invalid = Some(i);
                    }
                }
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let before_ok = i > start && digit_values[i - 1].is_some();
                let after_ok = i + 1 < bytes.len() && digit_values[i + 1].is_some();
                if !before_ok || !after_ok {
                    return Err(Error::Separator { index: i });
                }
            }
        }

        let digit_count = digit_values[start..].iter().filter(|v| v.is_some()).count();
        let first_digit_index = (start..bytes.len()).find(|&i| digit_values[i].is_some()).unwrap();
        if digit_count > 1 && digit_values[first_digit_index] == Some(0) {
            return Err(Error::LeadingZero { index: first_digit_index });
        }
        if digit_count > max_digits {
            let excess = max_digits;
            let index = (start..bytes.len())
                .filter(|&i| digit_values[i].is_some())
                .nth(excess)
                .unwrap();
            return Err(Error::TooManyDigits { limit: max_digits, index });
        }

        let mut result = Self::zero();
        for i in start..bytes.len() {
            if let Some(value) = digit_values[i] {
                result.mul_small(radix);
                result.add_small(value);
            }
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let mut n = self.abs();
        let mut digits = Vec::new();
        if n.is_zero() {
            digits.push(0u8);
        } else {
            while !n.is_zero() {
                let (q, r) = n.div_rem_small(radix);
                digits.push(r as u8);
                n = q;
            }
            digits.reverse();
        }
        while digits.len() < min_digits {
            digits.insert(0, 0);
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(match *digit {
                0..=9 => (b'0' + *digit) as char,
                _ => (b'a' + (*digit - 10)) as char,
            });
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => rhs.abs_cmp(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut result = match (self.negative, rhs.negative) {
            (false, false) => self.add_abs(rhs),
            (true, true) => {
                let mut n = self.add_abs(rhs);
                n.negative = !n.is_zero();
                n
            }
            (false, true) => match self.abs_cmp(rhs) {
                Ordering::Greater | Ordering::Equal => self.sub_abs(rhs),
                Ordering::Less => {
                    let mut n = rhs.sub_abs(self);
                    n.negative = !n.is_zero();
                    n
                }
            },
            (true, false) => match self.abs_cmp(rhs) {
                Ordering::Greater => {
                    let mut n = self.sub_abs(rhs);
                    n.negative = !n.is_zero();
                    n
                }
                Ordering::Equal => Self::zero(),
                Ordering::Less => rhs.sub_abs(self),
            },
        };
        result.normalize();
        result
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut result = self.mul_abs(rhs);
        result.negative = self.negative ^ rhs.negative;
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut q, mut r) = self.abs().div_rem_abs(&rhs.abs());
        q.negative = !q.is_zero() && (self.negative ^ rhs.negative);
        r.negative = !r.is_zero() && self.negative;
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: LIMIT });
        }
        let mut result = Self::one();
        let mut base = self.clone();
        let mut exp = exponent;
        while exp != 0 {
            if exp & 1 == 1 {
                result = result.mul(&base);
            }
            exp >>= 1;
            if exp != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, r) = a.div_rem(&b).expect("nonzero divisor");
            a = b;
            b = r.abs();
        }
        a.negative = false;
        a
    }
}

fn main() {}
```
