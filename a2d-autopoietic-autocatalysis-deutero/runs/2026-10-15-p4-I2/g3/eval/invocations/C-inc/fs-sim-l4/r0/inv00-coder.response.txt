```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn normalize(path: &str, base: &str) -> String {
        let mut parts: Vec<&str> = if path.starts_with('/') {
            Vec::new()
        } else if base == "/" {
            Vec::new()
        } else {
            base[1..].split('/').collect()
        };

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                name => parts.push(name),
            }
        }

        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn parent(path: &str) -> String {
        if path == "/" {
            return "/".to_string();
        }
        match path.rfind('/') {
            Some(0) | None => "/".to_string(),
            Some(i) => path[..i].to_string(),
        }
    }

    fn basename(path: &str) -> &str {
        path.rsplit('/').next().unwrap_or("")
    }

    fn join(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn is_within(path: &str, dir: &str) -> bool {
        if dir == "/" {
            return path.starts_with('/');
        }
        path == dir || path.strip_prefix(dir).is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| *candidate != path && Self::parent(candidate) == path)
            .cloned()
            .collect()
    }

    fn resolve(&self, path: &str, follow_final: bool, create_final: bool) -> Result<String, Error> {
        let mut current = path.to_string();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        loop {
            let components: Vec<&str> = if current == "/" {
                Vec::new()
            } else {
                current[1..].split('/').collect()
            };
            let mut physical = String::from("/");
            let mut restarted = false;

            for (index, component) in components.iter().enumerate() {
                let candidate = Self::join(&physical, component);
                let is_final = index + 1 == components.len();

                match self.entries.get(&candidate) {
                    None => {
                        if is_final && create_final {
                            return Ok(candidate);
                        }
                        return Err(Error::Missing(candidate));
                    }
                    Some(entry) => {
                        if let Entry::Link(target) = entry {
                            if !is_final || follow_final {
                                if !visited.insert(candidate.clone()) {
                                    return Err(Error::Loop(candidate));
                                }
                                if expansions >= 8 {
                                    return Err(Error::Depth(candidate));
                                }
                                expansions += 1;

                                let base = Self::parent(&candidate);
                                let target_path = Self::normalize(target, &base);
                                let suffix = components[index + 1..].join("/");
                                current = if suffix.is_empty() {
                                    target_path
                                } else if target_path == "/" {
                                    format!("/{}", suffix)
                                } else {
                                    format!("{}/{}", target_path, suffix)
                                };
                                current = Self::normalize(&current, "/");
                                restarted = true;
                                break;
                            }
                        }

                        if !is_final && !matches!(entry, Entry::Dir) {
                            return Err(Error::NotDir(candidate));
                        }
                        physical = candidate;
                    }
                }
            }

            if restarted {
                continue;
            }
            return Ok(physical);
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<String, Error> {
        self.resolve(&Self::parent(path), true, false)
    }

    fn get_kind(&self, path: &str) -> Option<Kind> {
        self.entries.get(path).map(Entry::kind)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let parent = Self::resolve_parent(&canonical)?;
        if !matches!(self.entries.get(&parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(parent));
        }
        let target = Self::join(&parent, Self::basename(&canonical));
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, true)?;
        let parent = Self::parent(&resolved);
        if !matches!(self.entries.get(&parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(parent));
        }

        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) | None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }

        let mut result: Vec<(String, Kind)> = self
            .children(&resolved)
            .into_iter()
            .filter_map(|child| {
                self.entries
                    .get(&child)
                    .map(|entry| (Self::basename(&child).to_string(), entry.kind()))
            })
            .collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        Self::validate(from)?;
        let source = Self::normalize(from, &self.cwd);
        Self::validate(to)?;
        let destination = Self::normalize(to, &self.cwd);

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let source_parent = self.resolve_parent(&source)?;
        let source_path = Self::join(&source_parent, Self::basename(&source));
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let destination_parent = self.resolve_parent(&destination)?;
        let destination_path =
            Self::join(&destination_parent, Self::basename(&destination));

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &destination_path)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&destination_path, &source_path) {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(dest) = &destination_entry {
            match (&source_entry, dest) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&destination_path).is_empty() {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| Self::is_within(path, &source_path))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if destination_entry.is_some() {
            self.entries.retain(|path, _| !Self::is_within(path, &destination_path));
        }
        self.entries.retain(|path, _| !Self::is_within(path, &source_path));

        for (old_path, entry) in moved {
            let suffix = &old_path[source_path.len()..];
            self.entries
                .insert(format!("{}{}", destination_path, suffix), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Root);
        }

        let parent = self.resolve_parent(&canonical)?;
        let resolved = Self::join(&parent, Self::basename(&canonical));
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &resolved) {
            return Err(Error::Busy(resolved));
        }

        if matches!(entry, Entry::Dir) && !self.children(&resolved).is_empty() && !recursive {
            return Err(Error::NotEmpty(resolved));
        }

        self.entries.retain(|candidate, _| !Self::is_within(candidate, &resolved));
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        Self::validate(link)?;
        let canonical = Self::normalize(link, &self.cwd);

        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        if canonical == "/" {
            return Err(Error::Exists("/".to_string()));
        }

        let parent = self.resolve_parent(&canonical)?;
        if !matches!(self.entries.get(&parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(parent));
        }
        let resolved = Self::join(&parent, Self::basename(&canonical));
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries
            .insert(resolved, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, false, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let descendants: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| Self::is_within(candidate, &resolved))
            .cloned()
            .collect();

        let mut sizes = BTreeMap::new();
        for candidate in descendants {
            let size = match self.entries.get(&candidate) {
                Some(Entry::File(data)) => data.len() as u64,
                Some(Entry::Link(target)) => target.len() as u64,
                Some(Entry::Dir) => self
                    .entries
                    .iter()
                    .filter(|(other, _)| {
                        *other != &candidate && Self::is_within(other, &candidate)
                    })
                    .map(|(_, entry)| match entry {
                        Entry::File(data) => data.len() as u64,
                        Entry::Link(target) => target.len() as u64,
                        Entry::Dir => 0,
                    })
                    .sum(),
                None => 0,
            };
            sizes.insert(candidate, size);
        }

        Ok(sizes.into_iter().collect())
    }
}

fn main() {}
```
