#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(
    jobs: &[Job],
    quantum: u32,
    switch_cost: u32,
) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut event_order: Vec<usize> = (0..jobs.len()).collect();
    event_order.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut ready: Vec<usize> = Vec::new();
    let mut ready_head = 0usize;
    let mut event_head = 0usize;
    let mut time = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let admit_through = |limit: u64,
                         event_head: &mut usize,
                         ready: &mut Vec<usize>,
                         completion: &mut Vec<Option<u64>>| {
        while *event_head < event_order.len() {
            let index = event_order[*event_head];
            let job = &jobs[index];
            let arrival = u64::from(job.arrival);
            if arrival > limit {
                break;
            }
            if job.ticks == 0 {
                completion[index] = Some(arrival);
            } else {
                ready.push(index);
            }
            *event_head += 1;
        }
    };

    loop {
        admit_through(time, &mut event_head, &mut ready, &mut completion);

        if ready_head == ready.len() {
            ready.clear();
            ready_head = 0;

            if event_head == event_order.len() {
                break;
            }

            let next_time = u64::from(jobs[event_order[event_head]].arrival);
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched = None;
            }
            continue;
        }

        let selected = ready[ready_head];
        ready_head += 1;

        let id = jobs[selected].id;
        if let Some(last_id) = last_dispatched {
            if last_id != id {
                time += u64::from(switch_cost);
            }
        }

        admit_through(time, &mut event_head, &mut ready, &mut completion);

        let consumed = remaining[selected].min(quantum);
        let start = time;
        time += u64::from(consumed);
        remaining[selected] -= consumed;
        let finished = remaining[selected] == 0;

        runs.push(Run {
            id,
            start,
            end: time,
            finished,
        });

        admit_through(time, &mut event_head, &mut ready, &mut completion);

        if finished {
            completion[selected] = Some(time);
        } else {
            ready.push(selected);
        }

        last_dispatched = Some(id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].unwrap()))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}

fn main() {}
