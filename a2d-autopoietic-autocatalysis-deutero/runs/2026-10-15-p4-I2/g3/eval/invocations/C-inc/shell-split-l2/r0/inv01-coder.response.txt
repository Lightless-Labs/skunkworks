#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let bytes = input.as_bytes();
    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b' ' | b'\t' | b'\r' | b'\n' => {
                if active {
                    words.push(std::mem::take(&mut word));
                    active = false;
                }
                i += 1;
            }
            b'\'' => {
                active = true;
                let quote_pos = i;
                i += 1;
                let mut closed = false;
                while i < bytes.len() {
                    if bytes[i] == b'\'' {
                        i += 1;
                        closed = true;
                        break;
                    }
                    let ch = input[i..].chars().next().unwrap();
                    word.push(ch);
                    i += ch.len_utf8();
                }
                if !closed {
                    return Err(SplitError::UnterminatedSingle { pos: quote_pos });
                }
            }
            b'"' => {
                active = true;
                let quote_pos = i;
                i += 1;
                let mut closed = false;
                while i < bytes.len() {
                    match bytes[i] {
                        b'"' => {
                            i += 1;
                            closed = true;
                            break;
                        }
                        b'\\' => {
                            let slash_pos = i;
                            i += 1;
                            if i == bytes.len() {
                                return Err(SplitError::TrailingEscape { pos: slash_pos });
                            }
                            if bytes[i] == b'"' || bytes[i] == b'\\' || bytes[i] == b'$' {
                                word.push(bytes[i] as char);
                                i += 1;
                            } else if bytes[i] == b'\n' {
                                i += 1;
                            } else {
                                word.push('\\');
                            }
                        }
                        b'$' => {
                            let (next_i, expanded) = expand(input, i, env)?;
                            if let Some(value) = expanded {
                                active = true;
                                word.push_str(&value);
                                i = next_i;
                            } else {
                                word.push('$');
                                i += 1;
                            }
                        }
                        _ => {
                            let ch = input[i..].chars().next().unwrap();
                            word.push(ch);
                            i += ch.len_utf8();
                        }
                    }
                }
                if !closed {
                    return Err(SplitError::UnterminatedDouble { pos: quote_pos });
                }
            }
            b'\\' => {
                let slash_pos = i;
                i += 1;
                if i == bytes.len() {
                    return Err(SplitError::TrailingEscape { pos: slash_pos });
                }
                if bytes[i] == b'\n' {
                    i += 1;
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    word.push(ch);
                    active = true;
                    i += ch.len_utf8();
                }
            }
            b'$' => {
                let (next_i, expanded) = expand(input, i, env)?;
                if let Some(value) = expanded {
                    active = true;
                    word.push_str(&value);
                    i = next_i;
                } else {
                    word.push('$');
                    active = true;
                    i += 1;
                }
            }
            _ => {
                let ch = input[i..].chars().next().unwrap();
                word.push(ch);
                active = true;
                i += ch.len_utf8();
            }
        }
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, Option<String>), SplitError> {
    let bytes = input.as_bytes();
    let next = dollar + 1;

    if next >= bytes.len() {
        return Ok((next, None));
    }

    if bytes[next] == b'{' {
        let mut close = next + 1;
        while close < bytes.len() {
            if bytes[close] == b'\\' {
                close += 1;
                if close < bytes.len() {
                    let ch = input[close..].chars().next().unwrap();
                    close += ch.len_utf8();
                }
            } else if bytes[close] == b'}' {
                break;
            } else {
                let ch = input[close..].chars().next().unwrap();
                close += ch.len_utf8();
            }
        }

        if close == bytes.len() {
            return Err(SplitError::UnterminatedExpansion { pos: dollar });
        }

        let contents = &input[next + 1..close];
        let content_bytes = contents.as_bytes();
        let mut name_end = 0;

        while name_end < content_bytes.len()
            && (content_bytes[name_end].is_ascii_alphanumeric()
                || content_bytes[name_end] == b'_')
        {
            name_end += 1;
        }

        let name = &contents[..name_end];
        let valid_name = !name.is_empty()
            && (name.as_bytes()[0].is_ascii_alphabetic() || name.as_bytes()[0] == b'_');

        let default_start = if name_end == content_bytes.len() {
            None
        } else if content_bytes.get(name_end..name_end + 2) == Some(b":-") {
            Some(name_end + 2)
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        if !valid_name {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let value = env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value);
        let result = match default_start {
            None => value.unwrap_or("").to_string(),
            Some(start) => match value {
                Some(v) if !v.is_empty() => v.to_string(),
                _ => decode_default(&contents[start..]),
            },
        };

        Ok((close + 1, Some(result)))
    } else {
        let first = input[next..].chars().next().unwrap();
        if !(first.is_ascii_alphabetic() || first == '_') {
            return Ok((dollar + 1, None));
        }

        let mut end = next + first.len_utf8();
        while end < bytes.len() {
            let ch = input[end..].chars().next().unwrap();
            if ch.is_ascii_alphanumeric() || ch == '_' {
                end += ch.len_utf8();
            } else {
                break;
            }
        }

        let name = &input[next..end];
        let value = env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value);
        Ok((end, Some(value.unwrap_or("").to_string())))
    }
}

fn decode_default(text: &str) -> String {
    let bytes = text.as_bytes();
    let mut result = String::new();
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'\\' && i + 1 < bytes.len() {
            i += 1;
            let ch = text[i..].chars().next().unwrap();
            result.push(ch);
            i += ch.len_utf8();
        } else {
            let ch = text[i..].chars().next().unwrap();
            result.push(ch);
            i += ch.len_utf8();
        }
    }

    result
}

fn main() {}
