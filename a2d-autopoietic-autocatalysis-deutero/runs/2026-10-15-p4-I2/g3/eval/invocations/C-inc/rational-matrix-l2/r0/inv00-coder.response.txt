use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn abs_i64(value: i64) -> u128 {
    if value < 0 {
        (-(value as i128)) as u128
    } else {
        value as u128
    }
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_wide(num as i128, den as i128)
    }

    fn from_wide(mut num: i128, mut den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if den < 0 {
            num = -num;
            den = -den;
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let divisor = gcd(num.unsigned_abs(), den as u128) as i128;
        num /= divisor;
        den /= divisor;

        if num < i64::MIN as i128 || num > i64::MAX as i128 || den > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: num as i64,
            den: den as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let d1 = self.den as i128;
        let d2 = rhs.den as i128;
        let common = gcd(d1 as u128, d2 as u128) as i128;
        let left_scale = d2 / common;
        let right_scale = d1 / common;
        Self::from_wide(a * left_scale + b * right_scale, d1 * left_scale)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let d1 = self.den as i128;
        let d2 = rhs.den as i128;
        let common = gcd(d1 as u128, d2 as u128) as i128;
        let left_scale = d2 / common;
        let right_scale = d1 / common;
        Self::from_wide(a * left_scale - b * right_scale, d1 * left_scale)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let an = self.num as i128;
        let ad = self.den as i128;
        let bn = rhs.num as i128;
        let bd = rhs.den as i128;

        let g1 = gcd(an.unsigned_abs(), bd as u128) as i128;
        let g2 = gcd(bn.unsigned_abs(), ad as u128) as i128;
        Self::from_wide((an / g1) * (bn / g2), (ad / g2) * (bd / g1))
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let an = self.num as i128;
        let ad = self.den as i128;
        let bn = rhs.num as i128;
        let bd = rhs.den as i128;

        let g1 = gcd(an.unsigned_abs(), bn.unsigned_abs()) as i128;
        let g2 = gcd(ad as u128, bd as u128) as i128;
        Self::from_wide((an / g1) * (bd / g2), (ad / g2) * (bn / g1))
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i += 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(RationalError::Parse {
                at: i,
                kind: ParseKind::InvalidByte,
            });
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let slash = if den_start.is_some() {
            Some(num_end)
        } else {
            None
        };

        let num_digits = &bytes[num_start..num_end];
        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(RationalError::Parse {
                at: num_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }
        if negative && num_digits == b"0" {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::NonCanonical,
            });
        }

        if let (Some(start), Some(end)) = (den_start, den_end) {
            let digits = &bytes[start..end];
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(RationalError::Parse {
                    at: start + 1,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let numerator_mag = parse_magnitude(num_digits);
        let numerator_limit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        if numerator_mag.map_or(true, |n| n > numerator_limit) {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::OutOfRange,
            });
        }

        let denominator_mag = if let (Some(start), Some(end)) = (den_start, den_end) {
            let digits = &bytes[start..end];
            match parse_magnitude(digits) {
                Some(n) if n <= i64::MAX as u128 => Some(n),
                _ => {
                    return Err(RationalError::Parse {
                        at: start,
                        kind: ParseKind::OutOfRange,
                    });
                }
            }
        } else {
            None
        };

        if let (Some(start), Some(0)) = (den_start, denominator_mag) {
            return Err(RationalError::Parse {
                at: start,
                kind: ParseKind::ZeroDenominator,
            });
        }

        if let (Some(start), Some(1)) = (den_start, denominator_mag) {
            return Err(RationalError::Parse {
                at: slash.unwrap_or(start),
                kind: ParseKind::NonCanonical,
            });
        }

        if let (Some(_), Some(_)) = (den_start, denominator_mag) {
            if numerator_mag == Some(0) {
                return Err(RationalError::Parse {
                    at: slash.unwrap_or(0),
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        if let (Some(start), Some(denominator)) = (den_start, denominator_mag) {
            let numerator = numerator_mag.unwrap();
            if gcd(numerator, denominator) > 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap_or(start),
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let magnitude = numerator_mag.unwrap() as i128;
        let num = if negative { -magnitude } else { magnitude };
        let den = denominator_mag.unwrap_or(1) as i128;
        Self::from_wide(num, den)
    }
}

fn parse_magnitude(digits: &[u8]) -> Option<u128> {
    let mut value = 0u128;
    for &digit in digits {
        value = value.checked_mul(10)?.checked_add((digit - b'0') as u128)?;
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|base| base.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data
            .get(index)
            .copied()
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for row in 0..self.cols {
            for col in 0..self.rows {
                data.push(self.data[col * self.cols + row]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let output_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(output_len);
        let zero = Rational { num: 0, den: 1 };

        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut sum = zero;
                for step in 0..self.cols {
                    let left_index = row * self.cols + step;
                    let right_index = step * rhs.cols + col;
                    let term = self.data[left_index]
                        .mul(&rhs.data[right_index])
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col,
                            step,
                            source,
                        })?;
                    sum = sum.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                }
                data.push(sum);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }
}
