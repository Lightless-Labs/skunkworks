#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
    pub strict: bool,
    pub max_symbols: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
    NonCanonical { offset: usize },
    MixedAlphabet { offset: usize },
    Overlong { offset: usize, limit: usize },
}

fn validate_options(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 64] {
    const STANDARD: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    const URL_SAFE: &[u8; 64] =
        b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_";

    match alphabet {
        Alphabet::Standard => STANDARD,
        Alphabet::UrlSafe => URL_SAFE,
    }
}

fn encode_symbols(input: &[u8], alphabet: Alphabet, padding: Padding) -> Vec<u8> {
    let table = alphabet_bytes(alphabet);
    let mut out = Vec::with_capacity(input.len().div_ceil(3) * 4);

    for chunk in input.chunks_exact(3) {
        let a = chunk[0];
        let b = chunk[1];
        let c = chunk[2];
        out.push(table[(a >> 2) as usize]);
        out.push(table[(((a & 0x03) << 4) | (b >> 4)) as usize]);
        out.push(table[(((b & 0x0f) << 2) | (c >> 6)) as usize]);
        out.push(table[(c & 0x3f) as usize]);
    }

    match input.len() % 3 {
        1 => {
            let a = input[input.len() - 1];
            out.push(table[(a >> 2) as usize]);
            out.push(table[((a & 0x03) << 4) as usize]);
            if padding != Padding::Forbidden {
                out.extend_from_slice(b"==");
            }
        }
        2 => {
            let a = input[input.len() - 2];
            let b = input[input.len() - 1];
            out.push(table[(a >> 2) as usize]);
            out.push(table[(((a & 0x03) << 4) | (b >> 4)) as usize]);
            out.push(table[((b & 0x0f) << 2) as usize]);
            if padding != Padding::Forbidden {
                out.push(b'=');
            }
        }
        _ => {}
    }

    out
}

fn wrap_symbols(symbols: &[u8], whitespace: Whitespace) -> Vec<u8> {
    let Whitespace::Wrapped { width, ending } = whitespace else {
        return symbols.to_vec();
    };

    let separator: &[u8] = match ending {
        LineEnding::Lf => b"\n",
        LineEnding::CrLf => b"\r\n",
    };

    let mut out = Vec::with_capacity(symbols.len() + symbols.len() / width * separator.len());
    for (row, chunk) in symbols.chunks(width).enumerate() {
        if row != 0 {
            out.extend_from_slice(separator);
        }
        out.extend_from_slice(chunk);
    }
    out
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate_options(options)?;
    let symbols = encode_symbols(input, options.alphabet, options.padding);
    let wrapped = wrap_symbols(&symbols, options.whitespace);
    Ok(String::from_utf8(wrapped).expect("base64 output is ASCII"))
}

fn digit_value(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn is_foreign_alphabet_byte(byte: u8, alphabet: Alphabet) -> bool {
    match alphabet {
        Alphabet::Standard => byte == b'-' || byte == b'_',
        Alphabet::UrlSafe => byte == b'+' || byte == b'/',
    }
}

fn decode_error_for_byte(byte: u8, offset: usize, options: Options) -> Error {
    if options.strict && is_foreign_alphabet_byte(byte, options.alphabet) {
        Error::MixedAlphabet { offset }
    } else {
        Error::InvalidByte { offset, byte }
    }
}

#[derive(Clone, Copy)]
struct Symbol {
    byte: u8,
    offset: usize,
    value: Option<u8>,
}

fn append_decoded_group(
    group: &[Symbol; 4],
    output: &mut Vec<u8>,
    strict: bool,
) -> Result<(), Error> {
    let a = group[0].value.expect("validated digit");
    let b = group[1].value.expect("validated digit");

    if group[2].byte == b'=' {
        if strict && b & 0x0f != 0 {
            return Err(Error::NonCanonical {
                offset: group[1].offset,
            });
        }
        output.push((a << 2) | (b >> 4));
        return Ok(());
    }

    let c = group[2].value.expect("validated digit");
    if group[3].byte == b'=' {
        if strict && c & 0x03 != 0 {
            return Err(Error::NonCanonical {
                offset: group[2].offset,
            });
        }
        output.push((a << 2) | (b >> 4));
        output.push((b << 4) | (c >> 2));
        return Ok(());
    }

    let d = group[3].value.expect("validated digit");
    output.push((a << 2) | (b >> 4));
    output.push((b << 4) | (c >> 2));
    output.push((c << 6) | d);
    Ok(())
}

#[derive(Clone, Copy)]
enum WrapState {
    Row,
    NeedLf { cr_offset: usize },
    NeedSeparator { first_offset: usize },
    NeedSymbol { first_offset: usize },
}

pub struct Decoder {
    options: Options,
    offset: usize,
    symbols_seen: usize,
    group: Vec<Symbol>,
    terminal_padding: bool,
    wrap_state: WrapState,
    row_symbols: usize,
    finished: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            offset: 0,
            symbols_seen: 0,
            group: Vec::with_capacity(4),
            terminal_padding: false,
            wrap_state: WrapState::Row,
            row_symbols: 0,
            finished: false,
            sticky_error: None,
        })
    }

    fn fail(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    fn is_ascii_whitespace(byte: u8) -> bool {
        matches!(byte, b' ' | b'\t' | b'\r' | b'\n' | 0x0b | 0x0c)
    }

    fn framing_byte(&mut self, byte: u8, offset: usize) -> Result<bool, Error> {
        match self.options.whitespace {
            Whitespace::None => Ok(true),
            Whitespace::IgnoreAscii => {
                Ok(!matches!(byte, 9 | 10 | 11 | 12 | 13 | 32))
            }
            Whitespace::Wrapped { width, ending } => match self.wrap_state {
                WrapState::NeedLf { cr_offset } => {
                    if byte != b'\n' {
                        return Err(Error::BadWrap { offset });
                    }
                    self.wrap_state = WrapState::NeedSymbol {
                        first_offset: cr_offset,
                    };
                    Ok(false)
                }
                WrapState::NeedSeparator { first_offset } => match ending {
                    LineEnding::Lf if byte == b'\n' => {
                        self.wrap_state = WrapState::NeedSymbol { first_offset };
                        Ok(false)
                    }
                    LineEnding::CrLf if byte == b'\r' => {
                        self.wrap_state = WrapState::NeedLf {
                            cr_offset: offset,
                        };
                        Ok(false)
                    }
                    _ => Err(Error::BadWrap { offset }),
                },
                WrapState::NeedSymbol { first_offset } => {
                    if Self::is_ascii_whitespace(byte) {
                        return Err(Error::BadWrap { offset });
                    }
                    self.wrap_state = WrapState::Row;
                    self.row_symbols = 1;
                    let _ = first_offset;
                    Ok(true)
                }
                WrapState::Row => {
                    if Self::is_ascii_whitespace(byte) {
                        return Err(Error::BadWrap { offset });
                    }
                    if self.row_symbols == width {
                        self.wrap_state = WrapState::NeedSeparator {
                            first_offset: offset,
                        };
                        self.framing_byte(byte, offset)
                    } else {
                        self.row_symbols += 1;
                        Ok(true)
                    }
                }
            },
        }
    }

    fn consume_symbol(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if let Some(limit) = self.options.max_symbols {
            if self.symbols_seen >= limit {
                return Err(Error::Overlong { offset, limit });
            }
        }
        self.symbols_seen += 1;

        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.group.len() == 2 && self.group[1].byte == b'=' && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        let position = self.group.len();
        if byte == b'=' {
            if position <= 1 || self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            self.group.push(Symbol {
                byte,
                offset,
                value: None,
            });
        } else {
            let value = digit_value(byte, self.options.alphabet)
                .ok_or_else(|| decode_error_for_byte(byte, offset, self.options))?;
            self.group.push(Symbol {
                byte,
                offset,
                value: Some(value),
            });
        }

        if self.group.len() == 4 {
            let padded = self.group[2].byte == b'=' || self.group[3].byte == b'=';
            append_decoded_group(
                &[self.group[0], self.group[1], self.group[2], self.group[3]],
                output,
                self.options.strict,
            )?;
            self.group.clear();
            if padded {
                self.terminal_padding = true;
            }
        }
        Ok(())
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;

            match self.framing_byte(byte, offset) {
                Ok(false) => continue,
                Ok(true) => {}
                Err(error) => return Err(self.fail(error)),
            }

            if let Err(error) = self.consume_symbol(byte, offset, &mut output) {
                return Err(self.fail(error));
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        match self.wrap_state {
            WrapState::NeedLf { cr_offset } => {
                return Err(self.fail(Error::BadWrap { offset: cr_offset }));
            }
            WrapState::NeedSeparator { first_offset }
            | WrapState::NeedSymbol { first_offset } => {
                return Err(self.fail(Error::BadWrap { offset: first_offset }));
            }
            WrapState::Row => {}
        }

        if self.group.len() == 3 && self.group[2].byte == b'=' {
            return Err(self.fail(Error::BadPadding {
                offset: self.group[2].offset,
            }));
        }

        let mut output = Vec::new();
        match self.group.len() {
            0 => {}
            1 => {
                return Err(self.fail(Error::Truncated {
                    offset: self.group[0].offset,
                }));
            }
            2 => {
                if self.options.padding == Padding::Required {
                    return Err(self.fail(Error::Truncated {
                        offset: self.group[0].offset,
                    }));
                }
                if self.options.strict && self.group[1].value.unwrap() & 0x0f != 0 {
                    return Err(self.fail(Error::NonCanonical {
                        offset: self.group[1].offset,
                    }));
                }
                output.push(
                    (self.group[0].value.unwrap() << 2)
                        | (self.group[1].value.unwrap() >> 4),
                );
            }
            3 => {
                if self.options.padding == Padding::Required {
                    return Err(self.fail(Error::Truncated {
                        offset: self.group[0].offset,
                    }));
                }
                if self.options.strict && self.group[2].value.unwrap() & 0x03 != 0 {
                    return Err(self.fail(Error::NonCanonical {
                        offset: self.group[2].offset,
                    }));
                }
                let a = self.group[0].value.unwrap();
                let b = self.group[1].value.unwrap();
                let c = self.group[2].value.unwrap();
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
            }
            _ => unreachable!(),
        }

        self.group.clear();
        self.finished = true;
        Ok(output)
    }
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    let mut decoder = Decoder::new(options)?;
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    options: Options,
    pending: Vec<u8>,
    row_symbols: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            pending: Vec::with_capacity(2),
            row_symbols: 0,
            finished: false,
        })
    }

    fn separator(&self) -> &'static [u8] {
        match self.options.whitespace {
            Whitespace::Wrapped {
                ending: LineEnding::Lf,
                ..
            } => b"\n",
            Whitespace::Wrapped {
                ending: LineEnding::CrLf,
                ..
            } => b"\r\n",
            _ => b"",
        }
    }

    fn emit_symbols(&mut self, symbols: &[u8], output: &mut Vec<u8>) {
        let Whitespace::Wrapped { width, .. } = self.options.whitespace else {
            output.extend_from_slice(symbols);
            return;
        };

        for &symbol in symbols {
            if self.row_symbols == width {
                output.extend_from_slice(self.separator());
                self.row_symbols = 0;
            }
            output.push(symbol);
            self.row_symbols += 1;
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let mut data = Vec::with_capacity(self.pending.len() + input.len());
        data.extend_from_slice(&self.pending);
        data.extend_from_slice(input);

        let complete_len = data.len() / 3 * 3;
        let symbols = encode_symbols(
            &data[..complete_len],
            self.options.alphabet,
            self.options.padding,
        );
        self.pending.clear();
        self.pending.extend_from_slice(&data[complete_len..]);

        let mut output = Vec::new();
        self.emit_symbols(&symbols, &mut output);
        Ok(String::from_utf8(output).expect("base64 output is ASCII"))
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }

        let symbols = encode_symbols(
            &self.pending,
            self.options.alphabet,
            self.options.padding,
        );
        self.pending.clear();

        let mut output = Vec::new();
        self.emit_symbols(&symbols, &mut output);
        self.finished = true;
        Ok(String::from_utf8(output).expect("base64 output is ASCII"))
    }
}

fn main() {}
