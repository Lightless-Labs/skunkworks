```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        let (num, den) = if den < 0 { (-num, -den) } else { (num, den) };
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
        let reduced_num = num / g;
        let reduced_den = den / g;

        if reduced_num < i64::MIN as i128
            || reduced_num > i64::MAX as i128
            || reduced_den < 1
            || reduced_den > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }

        Ok(Self {
            num: reduced_num as i64,
            den: reduced_den as i64,
        })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.den as i128;
        let b = rhs.den as i128;
        let g = gcd_u128(a as u128, b as u128) as i128;
        let left_scale = b / g;
        let right_scale = a / g;
        let num = self.num as i128 * left_scale + rhs.num as i128 * right_scale;
        let den = a * left_scale;
        Self::from_i128(num, den)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.den as i128;
        let b = rhs.den as i128;
        let g = gcd_u128(a as u128, b as u128) as i128;
        let left_scale = b / g;
        let right_scale = a / g;
        let num = self.num as i128 * left_scale - rhs.num as i128 * right_scale;
        let den = a * left_scale;
        Self::from_i128(num, den)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let c = self.den as i128;
        let d = rhs.den as i128;

        let g1 = gcd_u128(a.unsigned_abs(), d as u128) as i128;
        let g2 = gcd_u128(b.unsigned_abs(), c as u128) as i128;

        Self::from_i128((a / g1) * (b / g2), (c / g2) * (d / g1))
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }

        let a = self.num as i128;
        let b = rhs.den as i128;
        let c = self.den as i128;
        let d = rhs.num as i128;

        let g1 = gcd_u128(a.unsigned_abs(), d.unsigned_abs()) as i128;
        let g2 = gcd_u128(b as u128, c as u128) as i128;

        Self::from_i128((a / g1) * (b / g2), (c / g2) * (d / g1))
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        Self::from_i128(n, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
        }

        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            i += 1;
            let start = i;
            den_start = Some(start);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }

        let negative = bytes[0] == b'-';
        if num_end - num_start >= 2 && bytes[num_start] == b'0' {
            return Err(parse_err(if negative { 2 } else { 1 }, ParseKind::NonCanonical));
        }
        if negative && num_end - num_start == 1 && bytes[num_start] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }
        if let (Some(start), Some(end)) = (den_start, den_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_err(start + 1, ParseKind::NonCanonical));
            }
        }

        let num_limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let mut num_mag = 0u128;
        for &byte in &bytes[num_start..num_end] {
            num_mag = num_mag
                .saturating_mul(10)
                .saturating_add((byte - b'0') as u128);
            if num_mag > num_limit {
                return Err(parse_err(0, ParseKind::OutOfRange));
            }
        }

        let mut den_value = 1u128;
        if let (Some(start), Some(end)) = (den_start, den_end) {
            den_value = 0;
            for &byte in &bytes[start..end] {
                den_value = den_value
                    .saturating_mul(10)
                    .saturating_add((byte - b'0') as u128);
                if den_value > i64::MAX as u128 {
                    return Err(parse_err(start, ParseKind::OutOfRange));
                }
            }
        }

        if let Some(start) = den_start {
            if den_value == 0 {
                return Err(parse_err(start, ParseKind::ZeroDenominator));
            }

            let slash = start - 1;
            if den_value == 1 {
                return Err(parse_err(slash, ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_err(slash, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den_value) > 1 {
                return Err(parse_err(slash, ParseKind::NonCanonical));
            }
        }

        let num = if negative {
            if num_mag == (1u128 << 63) {
                i64::MIN as i128
            } else {
                -(num_mag as i128)
            }
        } else {
            num_mag as i128
        };
        Self::from_i128(num, den_value as i128)
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

fn main() {}
```
