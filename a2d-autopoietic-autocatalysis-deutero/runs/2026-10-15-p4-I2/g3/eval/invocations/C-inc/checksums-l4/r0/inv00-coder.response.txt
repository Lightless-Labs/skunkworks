#[derive(Debug, Clone)]
pub struct Stream {
    crc_raw: u32,
    adler_a: u32,
    adler_b: u32,
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffffu32;
    for &byte in data {
        crc ^= byte as u32;
        for _ in 0..8 {
            let low = crc & 1;
            crc >>= 1;
            if low != 0 {
                crc ^= 0xedb8_8320;
            }
        }
    }
    crc ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 65_521;
        b = (b + a) % 65_521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

#[derive(Clone, Copy)]
struct Accumulator {
    crc_raw: u32,
    adler_a: u32,
    adler_b: u32,
    f16_a: u32,
    f16_b: u32,
    f32_a: u32,
    f32_b: u32,
    f32_pending: Option<u8>,
    crc16: u16,
    crc16_variant: Crc16,
}

impl Accumulator {
    fn new(variant: Crc16) -> Self {
        Self {
            crc_raw: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
            f16_a: 0,
            f16_b: 0,
            f32_a: 0,
            f32_b: 0,
            f32_pending: None,
            crc16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
            crc16_variant: variant,
        }
    }

    fn update_byte(&mut self, byte: u8) {
        self.crc_raw ^= byte as u32;
        for _ in 0..8 {
            let low = self.crc_raw & 1;
            self.crc_raw >>= 1;
            if low != 0 {
                self.crc_raw ^= 0xedb8_8320;
            }
        }

        self.adler_a = (self.adler_a + byte as u32) % 65_521;
        self.adler_b = (self.adler_b + self.adler_a) % 65_521;

        self.f16_a = (self.f16_a + byte as u32) % 255;
        self.f16_b = (self.f16_b + self.f16_a) % 255;

        if let Some(high) = self.f32_pending.take() {
            let word = ((high as u32) << 8) | byte as u32;
            self.update_fletcher32_word(word);
        } else {
            self.f32_pending = Some(byte);
        }

        match self.crc16_variant {
            Crc16::Xmodem | Crc16::False => {
                self.crc16 ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = self.crc16 & 0x8000;
                    self.crc16 = self.crc16.wrapping_shl(1);
                    if top != 0 {
                        self.crc16 ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                self.crc16 ^= byte as u16;
                for _ in 0..8 {
                    let low = self.crc16 & 1;
                    self.crc16 >>= 1;
                    if low != 0 {
                        self.crc16 ^= 0x8408;
                    }
                }
            }
        }
    }

    fn update_fletcher32_word(&mut self, word: u32) {
        self.f32_a = (self.f32_a + word) % 65_535;
        self.f32_b = (self.f32_b + self.f32_a) % 65_535;
    }

    fn update_slice(&mut self, data: &[u8]) {
        for &byte in data {
            self.update_byte(byte);
        }
    }

    fn hashes(&self) -> Hashes {
        Hashes {
            crc32: self.crc_raw ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    fn fletcher32_value(&self) -> u32 {
        let mut a = self.f32_a;
        let mut b = self.f32_b;
        if let Some(high) = self.f32_pending {
            a = (a + ((high as u32) << 8)) % 65_535;
            b = (b + a) % 65_535;
        }
        (b << 16) | a
    }

    fn crc16_value(&self) -> u16 {
        match self.crc16_variant {
            Crc16::X25 => self.crc16 ^ 0xffff,
            Crc16::Xmodem | Crc16::False => self.crc16,
        }
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut accumulator = Accumulator::new(Crc16::Xmodem);
    for span in spans {
        accumulator.update_slice(&data[span.start..span.end]);
    }
    Ok(accumulator.hashes())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        let word = ((pair[0] as u32) << 8) | pair[1] as u32;
        a = (a + word) % 65_535;
        b = (b + a) % 65_535;
    }
    if let Some(&byte) = chunks.remainder().first() {
        a = (a + ((byte as u32) << 8)) % 65_535;
        b = (b + a) % 65_535;
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut accumulator = Accumulator::new(Crc16::Xmodem);
    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = accumulator.f32_pending.take() {
                accumulator.update_fletcher32_word(((high as u32) << 8) | byte as u32);
            } else {
                accumulator.f32_pending = Some(byte);
            }
        }
    }
    Ok(accumulator.fletcher32_value())
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut register = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                register ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = register & 0x8000;
                    register = register.wrapping_shl(1);
                    if top != 0 {
                        register ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                register ^= byte as u16;
                for _ in 0..8 {
                    let low = register & 1;
                    register >>= 1;
                    if low != 0 {
                        register ^= 0x8408;
                    }
                }
            }
        }
    }
    if variant == Crc16::X25 {
        register ^ 0xffff
    } else {
        register
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut accumulator = Accumulator::new(variant);
    for span in spans {
        accumulator.update_slice(&data[span.start..span.end]);
    }
    Ok(Multi {
        base: accumulator.hashes(),
        fletcher16: ((accumulator.f16_b << 8) | accumulator.f16_a) as u16,
        fletcher32: accumulator.fletcher32_value(),
        crc16: accumulator.crc16_value(),
    })
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
}

fn crc32_zero_byte_operator(value: u32) -> u32 {
    let mut crc = value;
    for _ in 0..8 {
        let low = crc & 1;
        crc >>= 1;
        if low != 0 {
            crc ^= 0xedb8_8320;
        }
    }
    crc
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for slot in odd.iter_mut().skip(1) {
        *slot = row;
        row <<= 1;
    }

    let mut even = [0u32; 32];
    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut len = right_len;
    let mut crc = left;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }

        gf2_matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }

    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total_len = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total_len = total_len
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut combined = 0u32;
    for part in parts {
        combined = crc32_combine(combined, part.crc32, part.len);
    }
    Ok(Part {
        crc32: combined,
        len: total_len,
    })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut accumulator = Accumulator::new(Crc16::Xmodem);
    let mut selected_len = 0u64;
    for span in spans {
        let length = span.end - span.start;
        selected_len += length as u64;
        accumulator.update_slice(&data[span.start..span.end]);
    }
    let hashes = accumulator.hashes();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        selected_len, hashes.crc32, hashes.adler32
    ))
}

fn parse_hex(bytes: &[u8], start: usize, end: usize) -> Result<u64, usize> {
    let mut value = 0u64;
    for offset in start..end {
        let digit = match bytes[offset] {
            b'0'..=b'9' => bytes[offset] - b'0',
            b'a'..=b'f' => bytes[offset] - b'a' + 10,
            _ => return Err(offset),
        };
        value = (value << 4) | digit as u64;
    }
    Ok(value)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for (start, end) in [(4usize, 20usize), (25, 33), (40, 48)] {
        for offset in start..end {
            if !matches!(bytes[offset], b'0'..=b'9' | b'a'..=b'f') {
                return Err(VerifyError::Hex { offset });
            }
        }
    }

    let length = parse_hex(bytes, 4, 20).unwrap() as u64;
    let expected_crc = parse_hex(bytes, 25, 33).unwrap() as u32;
    let expected_adler = parse_hex(bytes, 40, 48).unwrap() as u32;

    let mut selected_len = 0u64;
    let mut accumulator = Accumulator::new(Crc16::Xmodem);
    for span in spans {
        selected_len += (span.end - span.start) as u64;
        accumulator.update_slice(&data[span.start..span.end]);
    }

    if length != selected_len {
        return Err(VerifyError::Length);
    }
    let actual = accumulator.hashes();
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_raw: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_raw ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_raw: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc_raw ^= byte as u32;
            for _ in 0..8 {
                let low = self.crc_raw & 1;
                self.crc_raw >>= 1;
                if low != 0 {
                    self.crc_raw ^= 0xedb8_8320;
                }
            }
            self.adler_a = (self.adler_a + byte as u32) % 65_521;
            self.adler_b = (self.adler_b + self.adler_a) % 65_521;
        }
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        for offset in (1..9).chain(11..19) {
            if !matches!(bytes[offset], b'0'..=b'9' | b'a'..=b'f') {
                return Err(SnapshotError::Hex { offset });
            }
        }

        let crc = parse_hex(bytes, 1, 9).unwrap() as u32;
        let adler = parse_hex(bytes, 11, 19).unwrap() as u32;
        Self::from_seed(crc, adler)
    }
}
