#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    let (min, max) = bounds(field);
    if step {
        if value < 1 || value > max - min + 1 {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else if value < min || value > max {
        return Err(ParseError::OutOfRange {
            field,
            offset: start,
        });
    }

    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = bounds(field);
    let mut selected = vec![false; (max + 1) as usize];
    let mut item_start = 0;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&byte| byte == b',')
            .map(|offset| item_start + offset)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash = item.iter().position(|&byte| byte == b'/');

        let (base_end, step_range) = match slash {
            None => (item.len(), None),
            Some(pos) if item[pos + 1..].contains(&b'/') => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            Some(pos) => (pos, Some((pos + 1, item.len()))),
        };

        let base = &item[..base_end];
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &byte)| (byte == b'-').then_some(i))
            .collect();

        let base_kind = if base == b"*" {
            0
        } else if !base.is_empty()
            && hyphens.is_empty()
            && base.iter().all(u8::is_ascii_digit)
        {
            1
        } else if let [hyphen] = hyphens.as_slice() {
            if *hyphen > 0
                && *hyphen + 1 < base.len()
                && base[..*hyphen].iter().all(u8::is_ascii_digit)
                && base[*hyphen + 1..].iter().all(u8::is_ascii_digit)
            {
                2
            } else {
                -1
            }
        } else {
            -1
        };

        if base_kind < 0 || (step_range.is_some() && base.is_empty()) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (start_value, mut end_value) = match base_kind {
            0 => (min, max),
            1 => {
                let value = parse_number(bytes, item_start, item_start + base.len(), field, false)?;
                (value, value)
            }
            2 => {
                let hyphen = hyphens[0];
                let start_value =
                    parse_number(bytes, item_start, item_start + hyphen, field, false)?;
                let end_start = item_start + hyphen + 1;
                let end_value =
                    parse_number(bytes, end_start, item_end - (item.len() - hyphen - 1), field, false)?;

                if start_value > end_value {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: end_start,
                    });
                }

                (start_value, end_value)
            }
            _ => unreachable!(),
        };

        let step = if let Some((step_start, step_end)) = step_range {
            parse_number(
                bytes,
                item_start + step_start,
                item_start + step_end,
                field,
                true,
            )?
        } else {
            1
        };

        if base_kind == 1 && step_range.is_some() {
            end_value = max;
        }

        let mut value = start_value;
        while value <= end_value {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let mut selected = [Vec::new(), Vec::new(), Vec::new(), Vec::new(), Vec::new()];
    for (index, field) in FIELDS.into_iter().enumerate() {
        selected[index] = parse_field(pieces[index], field)?;
    }

    Ok(Cron {
        selected,
        day_of_month_is_star: pieces[2] == "*",
        day_of_week_is_star: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, max) = bounds(field);
        (min..=max)
            .filter(|&value| self.selected[field_index(field)][value as usize])
            .map(|value| value as u8)
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let day_of_week = weekday(at.year, at.month, at.day);
        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize]
            && self.selected[3][at.month as usize]
            && self.selected[4][day_of_week as usize])
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            current = advance_minute(current);
            if current.year > 2399 {
                return Ok(None);
            }
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn invalid(component: DatePart) -> DateError {
    DateError::InvalidDateTime { component }
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(invalid(DatePart::Year));
    }
    if !(1..=12).contains(&at.month) {
        return Err(invalid(DatePart::Month));
    }
    if at.day < 1 || at.day > days_in_month(at.year, at.month) {
        return Err(invalid(DatePart::Day));
    }
    if at.hour > 23 {
        return Err(invalid(DatePart::Hour));
    }
    if at.minute > 59 {
        return Err(invalid(DatePart::Minute));
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut result = 6u8;
    let mut y = 2000u16;
    let mut m = 1u8;
    let mut d = 1u8;

    while (y, m, d) < (year, month, day) {
        d += 1;
        if d > days_in_month(y, m) {
            d = 1;
            m += 1;
            if m > 12 {
                m = 1;
                y += 1;
            }
        }
        result = (result + 1) % 7;
    }

    result
}

fn advance_minute(mut at: CivilDateTime) -> CivilDateTime {
    at.minute += 1;
    if at.minute <= 59 {
        return at;
    }

    at.minute = 0;
    at.hour += 1;
    if at.hour <= 23 {
        return at;
    }

    at.hour = 0;
    at.day += 1;
    if at.day <= days_in_month(at.year, at.month) {
        return at;
    }

    at.day = 1;
    at.month += 1;
    if at.month <= 12 {
        return at;
    }

    at.month = 1;
    at.year += 1;
    at
}
