#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn render_tokens(tokens: &[Token]) -> String {
    let mut s = String::new();
    for token in tokens {
        s.push_str(&token.text);
    }
    s
}

fn trim_trailing_ascii_spaces(s: &str) -> &str {
    s.trim_end_matches(' ')
}

fn invalid_char(c: char) -> bool {
    (c.is_control() && c != '\n' && c != '\r') || c == '\u{2028}' || c == '\u{2029}'
}

fn validate_cell(
    cell: &str,
    row: usize,
    column: usize,
) -> Result<(), TableError> {
    let mut iter = cell.char_indices().peekable();
    while let Some((byte, c)) = iter.next() {
        if c == '\r' {
            if matches!(iter.peek(), Some((_, '\n'))) {
                iter.next();
            } else {
                return Err(TableError::InvalidChar { row, column, byte });
            }
        } else if invalid_char(c) {
            return Err(TableError::InvalidChar { row, column, byte });
        }
    }
    Ok(())
}

fn tokenize_line(line: &str, markdown: bool) -> Vec<Token> {
    let mut out = Vec::new();
    for c in line.chars() {
        if markdown && c == '|' {
            out.push(Token { text: "\\|".to_owned(), width: 2 });
        } else if markdown && c == '\\' {
            out.push(Token { text: "\\\\".to_owned(), width: 2 });
        } else {
            out.push(Token { text: c.to_string(), width: 1 });
        }
    }
    out
}

fn split_cell(cell: &str, markdown: bool) -> Vec<Vec<Token>> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| tokenize_line(trim_trailing_ascii_spaces(line), markdown))
        .collect()
}

fn total_width(widths: &[usize], overhead: usize) -> usize {
    widths.iter().sum::<usize>().saturating_add(overhead)
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    let total = tokens.iter().map(|t| t.width).sum::<usize>();
    match overflow {
        Overflow::Clip if total <= width => vec![render_tokens(tokens)],
        Overflow::Clip => {
            let limit = width.saturating_sub(1);
            let mut used = 0;
            let mut end = 0;
            for token in tokens {
                if used + token.width > limit {
                    break;
                }
                used += token.width;
                end += 1;
            }
            let mut s = render_tokens(&tokens[..end]);
            s.push('\u{2026}');
            vec![s]
        }
        Overflow::Wrap => {
            let mut out = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                out.push(render_tokens(&tokens[start..end]));
                start = end;
            }
            out
        }
    }
}

fn align_fragment(
    content: &str,
    content_width: usize,
    width: usize,
    alignment: &Alignment,
) -> String {
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    let mut out = String::new();
    out.extend(std::iter::repeat_n(' ', left));
    out.push_str(content);
    out.extend(std::iter::repeat_n(' ', right));
    out
}

fn emit_grid_or_markdown_line(
    row: &[Vec<String>],
    widths: &[usize],
    alignments: &[Alignment],
    compact: bool,
) -> String {
    let mut out = String::new();
    if !compact {
        out.push('|');
    }

    for col in 0..widths.len() {
        if compact {
            if col > 0 {
                out.push_str("  ");
            }
        } else {
            out.push(' ');
        }

        let content = row.get(col).map(String::as_str).unwrap_or("");
        let content_width = content.chars().count();
        let mut aligned = align_fragment(content, content_width, widths[col], &alignments[col]);
        if compact && col + 1 == widths.len() {
            let right_padding = match alignments[col] {
                Alignment::Left => widths[col].saturating_sub(content_width),
                Alignment::Right => 0,
                Alignment::Center => (widths[col].saturating_sub(content_width) + 1) / 2,
            };
            for _ in 0..right_padding {
                aligned.pop();
            }
        }
        out.push_str(&aligned);

        if !compact {
            out.push_str(" |");
        }
    }
    out.push('\n');
    out
}

fn grid_border(widths: &[usize]) -> String {
    let mut out = String::from("+");
    for &width in widths {
        out.extend(std::iter::repeat_n('-', width + 2));
        out.push('+');
    }
    out.push('\n');
    out
}

fn markdown_marker(widths: &[usize], alignments: &[Alignment]) -> String {
    let mut out = String::from("|");
    for (i, &width) in widths.iter().enumerate() {
        out.push(' ');
        match alignments[i] {
            Alignment::Left => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 1));
            }
            Alignment::Right => {
                out.extend(std::iter::repeat_n('-', width - 1));
                out.push(':');
            }
            Alignment::Center => {
                out.push(':');
                out.extend(std::iter::repeat_n('-', width - 2));
                out.push(':');
            }
        }
        out.push_str(" |");
    }
    out.push('\n');
    out
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let (overhead, minimum_width) = match options.mode {
        Mode::Grid | Mode::Markdown => (3 * n + 1, if matches!(options.mode, Mode::Markdown) { 3 } else { 1 }),
        Mode::Compact => (2 * (n - 1), 1),
    };
    let minimum = overhead + n * minimum_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            validate_cell(cell, row, column)?;
        }
    }

    let alignments = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };
    let markdown = matches!(options.mode, Mode::Markdown);

    let tokenized: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| row.iter().map(|cell| split_cell(cell, markdown)).collect())
        .collect();

    let mut widths = vec![minimum_width; n];
    for row in &tokenized {
        for (column, lines) in row.iter().enumerate() {
            for line in lines {
                let line_width = line.iter().map(|t| t.width).sum::<usize>();
                widths[column] = widths[column].max(line_width);
            }
        }
    }

    if options.max_width != 0 {
        while total_width(&widths, overhead) > options.max_width {
            let candidate = (0..n)
                .rev()
                .find(|&i| widths[i] > minimum_width)
                .unwrap();
            let widest = (0..n)
                .filter(|&i| widths[i] > minimum_width)
                .map(|i| widths[i])
                .max()
                .unwrap();
            let candidate = (0..n)
                .rev()
                .find(|&i| widths[i] == widest && widths[i] > minimum_width)
                .unwrap_or(candidate);
            widths[candidate] -= 1;
        }
    }

    let mut out = String::new();
    if matches!(options.mode, Mode::Grid) {
        out.push_str(&grid_border(&widths));
    }

    for row_index in 0..rows.len() {
        let mut cell_fragments: Vec<Vec<String>> = Vec::with_capacity(n);
        for column in 0..n {
            let mut combined = Vec::new();
            for line in &tokenized[row_index][column] {
                combined.extend(fragments(line, widths[column], &overflows[column]));
            }
            cell_fragments.push(combined);
        }

        let height = cell_fragments.iter().map(Vec::len).max().unwrap_or(1);
        for fragment_index in 0..height {
            let physical: Vec<String> = (0..n)
                .map(|column| {
                    cell_fragments[column]
                        .get(fragment_index)
                        .cloned()
                        .unwrap_or_default()
                })
                .collect();
            out.push_str(&emit_grid_or_markdown_line(
                &physical,
                &widths,
                &alignments,
                matches!(options.mode, Mode::Compact),
            ));
        }

        match options.mode {
            Mode::Grid if options.header && row_index == 0 && row_index + 1 < rows.len() => {
                out.push_str(&grid_border(&widths));
            }
            Mode::Markdown if row_index == 0 => {
                out.push_str(&markdown_marker(&widths, &alignments));
            }
            Mode::Grid if row_index + 1 == rows.len() => {
                out.push_str(&grid_border(&widths));
            }
            _ => {}
        }
    }

    Ok(out)
}

fn main() {}
