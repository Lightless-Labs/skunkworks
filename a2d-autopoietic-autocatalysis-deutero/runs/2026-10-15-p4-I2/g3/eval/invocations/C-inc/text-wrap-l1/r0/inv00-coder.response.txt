#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(32..=126).contains(&byte) && !matches!(byte, b'\t' | b'\n' | b'\r') {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let is_separator = |byte: u8| matches!(byte, b' ' | b'\t' | b'\n' | b'\r');
    let mut lines = Vec::new();
    let mut line = String::new();

    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_separator(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && !is_separator(bytes[i]) {
            i += 1;
        }

        let mut word_pos = start;
        while word_pos < i {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            let remaining = i - word_pos;
            let line_len = line.len();

            if remaining <= width {
                let needed = remaining + usize::from(!line.is_empty());
                if line_len + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(std::str::from_utf8(&bytes[word_pos..i]).unwrap());
                    word_pos = i;
                } else {
                    if !line.is_empty() {
                        lines.push(std::mem::take(&mut line));
                    }
                }
            } else if line.is_empty() {
                let end = word_pos + width;
                line.push_str(std::str::from_utf8(&bytes[word_pos..end]).unwrap());
                lines.push(std::mem::take(&mut line));
                word_pos = end;
            } else {
                let available = width - line_len - 1;
                if available >= 2 {
                    line.push(' ');
                    let end = word_pos + available;
                    line.push_str(std::str::from_utf8(&bytes[word_pos..end]).unwrap());
                    lines.push(std::mem::take(&mut line));
                    word_pos = end;
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}

fn main() {}
