#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn append_literal(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_expansion(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append_literal(word, c, active_glob);
    }
}

fn char_at(input: &str, pos: usize) -> Option<char> {
    input.get(pos..)?.chars().next()
}

fn next_pos(input: &str, pos: usize) -> usize {
    pos + char_at(input, pos).map(char::len_utf8).unwrap_or(0)
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let mut close = dollar + 2;

    while close < input.len() {
        let c = char_at(input, close).unwrap();
        if c == '\\' {
            close = next_pos(input, close);
            if close < input.len() {
                close = next_pos(input, close);
            }
        } else if c == '}' {
            break;
        } else {
            close = next_pos(input, close);
        }
    }

    if close >= input.len() {
        return Err(SplitError::UnterminatedExpansion { pos: dollar });
    }

    let content = &input[dollar + 2..close];
    let name_end = content
        .char_indices()
        .find(|(_, c)| !is_name_continue(*c))
        .map(|(i, _)| i)
        .unwrap_or(content.len());

    let name = &content[..name_end];
    let valid_name = !name.is_empty()
        && name.chars().next().map(is_name_start).unwrap_or(false)
        && name.chars().all(is_name_continue);

    if !valid_name {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let remainder = &content[name_end..];
    let selected: String = if remainder.is_empty() {
        env_value(env, name).unwrap_or("").to_string()
    } else if let Some(default) = remainder.strip_prefix(":-") {
        match env_value(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => {
                let mut decoded = String::new();
                let mut chars = default.chars();
                while let Some(c) = chars.next() {
                    if c == '\\' {
                        if let Some(next) = chars.next() {
                            decoded.push(next);
                        } else {
                            decoded.push('\\');
                        }
                    } else {
                        decoded.push(c);
                    }
                }
                decoded
            }
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close + 1, selected))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut pos = 0usize;

    while pos < input.len() {
        let c = char_at(input, pos).unwrap();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.text.push(c);
                }
                pos = next_pos(input, pos);
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    pos = next_pos(input, pos);
                } else if c == '\\' {
                    let slash = pos;
                    pos = next_pos(input, pos);
                    if pos >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = char_at(input, pos).unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            word.text.push(next);
                        }
                        pos = next_pos(input, pos);
                    } else {
                        word.text.push('\\');
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if c == '$' {
                    let after = next_pos(input, pos);
                    if char_at(input, after) == Some('(') {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if char_at(input, after) == Some('{') {
                        let (end, value) = parse_braced(input, pos, env)?;
                        append_expansion(&mut word, &value, false);
                        active = true;
                        pos = end;
                    } else if char_at(input, after).map(is_name_start).unwrap_or(false) {
                        let mut end = after;
                        while char_at(input, end).map(is_name_continue).unwrap_or(false) {
                            end = next_pos(input, end);
                        }
                        let name = &input[after..end];
                        append_expansion(&mut word, env_value(env, name).unwrap_or(""), false);
                        active = true;
                        pos = end;
                    } else {
                        word.text.push('$');
                        pos = after;
                    }
                } else {
                    word.text.push(c);
                    pos = next_pos(input, pos);
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    pos = next_pos(input, pos);
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_start = pos;
                    active = true;
                    pos = next_pos(input, pos);
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_start = pos;
                    active = true;
                    pos = next_pos(input, pos);
                } else if c == '\\' {
                    let slash = pos;
                    pos = next_pos(input, pos);
                    if pos >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = char_at(input, pos).unwrap();
                    if next == '\n' {
                        pos = next_pos(input, pos);
                    } else {
                        word.text.push(next);
                        active = true;
                        pos = next_pos(input, pos);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if c == '$' {
                    let after = next_pos(input, pos);
                    if char_at(input, after) == Some('(') {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if char_at(input, after) == Some('{') {
                        let (end, value) = parse_braced(input, pos, env)?;
                        append_expansion(&mut word, &value, true);
                        active = true;
                        pos = end;
                    } else if char_at(input, after).map(is_name_start).unwrap_or(false) {
                        let mut end = after;
                        while char_at(input, end).map(is_name_continue).unwrap_or(false) {
                            end = next_pos(input, end);
                        }
                        let name = &input[after..end];
                        append_expansion(&mut word, env_value(env, name).unwrap_or(""), true);
                        active = true;
                        pos = end;
                    } else {
                        word.text.push('$');
                        active = true;
                        pos = after;
                    }
                } else {
                    append_literal(&mut word, c, true);
                    active = true;
                    pos = next_pos(input, pos);
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }

    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric()
            || matches!(c, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%')
        {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }

    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

fn fallback_candidate(word: &str) -> String {
    let mut out = String::from("'");
    for c in word.chars() {
        if c == '\'' {
            out.push('\'');
            out.push('\\');
            out.push('\'');
            out.push('\'');
        } else {
            out.push(c);
        }
    }
    out.push('\'');
    out
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul { word: index, pos });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates = Vec::new();

        if let Some(candidate) = bare_candidate(word) {
            candidates.push((candidate, 0usize));
        }
        if let Some(candidate) = single_candidate(word) {
            candidates.push((candidate, 1usize));
        }
        if let Some(candidate) = double_candidate(word) {
            candidates.push((candidate, 2usize));
        }

        let chosen = if candidates.is_empty() {
            fallback_candidate(word)
        } else {
            candidates
                .into_iter()
                .min_by_key(|(candidate, priority)| (candidate.len(), *priority))
                .unwrap()
                .0
        };
        encoded.push(chosen);
    }

    Ok(encoded.join(" "))
}
