use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = n.checked_neg().ok_or(RationalError::Overflow)?;
        d = d.checked_neg().ok_or(RationalError::Overflow)?;
    }

    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let abs_n = n.unsigned_abs();
    let abs_d = d as u128;
    let g = gcd_u128(abs_n, abs_d);
    let reduced_abs_n = abs_n / g;
    let reduced_d = abs_d / g;

    if reduced_d > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }

    let reduced_n = if n < 0 {
        if reduced_abs_n > (i64::MAX as u128) + 1 {
            return Err(RationalError::Overflow);
        }
        if reduced_abs_n == (i64::MAX as u128) + 1 {
            i64::MIN
        } else {
            -(reduced_abs_n as i64)
        }
    } else {
        if reduced_abs_n > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        reduced_abs_n as i64
    };

    Ok(Rational {
        num: reduced_n,
        den: reduced_d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let n = num as i128;
        let d = den as i128;
        rational_from_i128(n, d)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128)
            .checked_mul(rhs.den as i128)
            .and_then(|a| {
                (rhs.num as i128)
                    .checked_mul(self.den as i128)
                    .and_then(|b| a.checked_add(b))
            })
            .ok_or(RationalError::Overflow)?;
        let d = (self.den as i128)
            .checked_mul(rhs.den as i128)
            .ok_or(RationalError::Overflow)?;
        rational_from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128)
            .checked_mul(rhs.den as i128)
            .and_then(|a| {
                (rhs.num as i128)
                    .checked_mul(self.den as i128)
                    .and_then(|b| a.checked_sub(b))
            })
            .ok_or(RationalError::Overflow)?;
        let d = (self.den as i128)
            .checked_mul(rhs.den as i128)
            .ok_or(RationalError::Overflow)?;
        rational_from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128)
            .checked_mul(rhs.num as i128)
            .ok_or(RationalError::Overflow)?;
        let d = (self.den as i128)
            .checked_mul(rhs.den as i128)
            .ok_or(RationalError::Overflow)?;
        rational_from_i128(n, d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let n = (self.num as i128)
            .checked_mul(rhs.den as i128)
            .ok_or(RationalError::Overflow)?;
        let d = (self.den as i128)
            .checked_mul(rhs.num as i128)
            .ok_or(RationalError::Overflow)?;
        rational_from_i128(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = self.num.checked_neg().ok_or(RationalError::Overflow)?;
        Ok(Self {
            num: n,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        fn parse_err(at: usize, kind: ParseKind) -> RationalError {
            RationalError::Parse { at, kind }
        }

        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }

        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let num_end = i;
        let mut slash = None;
        let mut den_start = 0;
        let mut den_end = 0;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            slash = Some(i);
            i += 1;
            den_start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            den_end = i;
        }

        if i < bytes.len() {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let num_digits = &bytes[num_start..num_end];
        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(parse_err(num_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == b"0" {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        if let Some(slash_at) = slash {
            let den_digits = &bytes[den_start..den_end];
            if den_digits.len() >= 2 && den_digits[0] == b'0' {
                return Err(parse_err(den_start + 1, ParseKind::NonCanonical));
            }
        }

        let mut num_mag: u128 = 0;
        for &b in num_digits {
            num_mag = num_mag
                .saturating_mul(10)
                .saturating_add((b - b'0') as u128);
        }
        let num_limit = if negative {
            (i64::MAX as u128) + 1
        } else {
            i64::MAX as u128
        };
        if num_mag > num_limit {
            return Err(parse_err(0, ParseKind::OutOfRange));
        }

        let mut den_mag: u128 = 1;
        if slash.is_some() {
            den_mag = 0;
            for &b in &bytes[den_start..den_end] {
                den_mag = den_mag
                    .saturating_mul(10)
                    .saturating_add((b - b'0') as u128);
            }
            if den_mag > i64::MAX as u128 {
                return Err(parse_err(den_start, ParseKind::OutOfRange));
            }
            if den_mag == 0 {
                return Err(parse_err(den_start, ParseKind::ZeroDenominator));
            }
        }

        if let Some(slash_at) = slash {
            if den_mag == 1 {
                return Err(parse_err(slash_at, ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_err(slash_at, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den_mag) > 1 {
                return Err(parse_err(slash_at, ParseKind::NonCanonical));
            }
        }

        let num = if negative {
            if num_mag == (i64::MAX as u128) + 1 {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };
        Ok(Self {
            num,
            den: den_mag as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch {
        coefficients: usize,
        rhs: usize,
    },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}

fn zero() -> Rational {
    Rational { num: 0, den: 1 }
}

fn one() -> Rational {
    Rational { num: 1, den: 1 }
}

fn arithmetic_error(row: usize, col: usize, step: usize, source: RationalError) -> MatrixError {
    MatrixError::Arithmetic {
        row,
        col,
        step,
        source,
    }
}

fn elimination(
    matrix: &mut Matrix,
    candidate_cols: usize,
) -> Result<(Vec<usize>, Vec<(usize, usize)>), MatrixError> {
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;
    let cols = matrix.cols;

    for candidate in 0..candidate_cols {
        if p == matrix.rows {
            break;
        }

        let mut best: Option<(i64, u128, usize)> = None;
        for row in p..matrix.rows {
            let value = matrix.data[row * cols + candidate];
            if value.num == 0 {
                continue;
            }
            let key = (value.den, value.num.unsigned_abs(), row);
            if best.map_or(true, |old| key < old) {
                best = Some(key);
            }
        }

        let Some((_, _, q)) = best else {
            continue;
        };

        if q != p {
            for col in 0..cols {
                matrix.data.swap(p * cols + col, q * cols + col);
            }
            swaps.push((p, q));
        }

        let pivot = matrix.data[p * cols + candidate];
        for col in 0..cols {
            let idx = p * cols + col;
            matrix.data[idx] = matrix.data[idx]
                .div(&pivot)
                .map_err(|e| arithmetic_error(p, col, p, e))?;
        }

        for row in 0..matrix.rows {
            if row == p {
                continue;
            }
            let factor = matrix.data[row * cols + candidate];
            if factor.num == 0 {
                continue;
            }
            for col in 0..cols {
                let product = factor
                    .mul(&matrix.data[p * cols + col])
                    .map_err(|e| arithmetic_error(row, col, p, e))?;
                let idx = row * cols + col;
                matrix.data[idx] = matrix.data[idx]
                    .sub(&product)
                    .map_err(|e| arithmetic_error(row, col, p, e))?;
            }
        }

        pivots.push(candidate);
        p += 1;
    }

    Ok((pivots, swaps))
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for row in 0..self.cols {
            for col in 0..self.rows {
                data.push(self.data[col * self.cols + row]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }

        let out_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(out_len);

        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut acc = zero();
                for step in 0..self.cols {
                    let left = self.data[row * self.cols + step];
                    let right = rhs.data[step * rhs.cols + col];
                    let term = left
                        .mul(&right)
                        .map_err(|e| arithmetic_error(row, col, step, e))?;
                    acc = acc
                        .add(&term)
                        .map_err(|e| arithmetic_error(row, col, step, e))?;
                }
                data.push(acc);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let (pivots, swaps) = elimination(&mut matrix, self.cols)?;
        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }

        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        for row in 0..n {
            for col in 0..n {
                data.push(self.data[row * n + col]);
            }
            for col in 0..n {
                data.push(if row == col { one() } else { zero() });
            }
        }

        let mut augmented = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let (pivots, _) = elimination(&mut augmented, n)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }

        let mut inverse_data = Vec::with_capacity(total / 2);
        for row in 0..n {
            for col in 0..n {
                inverse_data.push(augmented.data[row * width + n + col]);
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: inverse_data,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }

        let width = self
            .cols
            .checked_add(rhs.cols)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let total = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut augmented_data = Vec::with_capacity(total);
        for row in 0..self.rows {
            for col in 0..self.cols {
                augmented_data.push(self.data[row * self.cols + col]);
            }
            for col in 0..rhs.cols {
                augmented_data.push(rhs.data[row * rhs.cols + col]);
            }
        }

        let mut augmented = Matrix {
            rows: self.rows,
            cols: width,
            data: augmented_data,
        };
        let (pivots, _) =
            elimination(&mut augmented, self.cols).map_err(SolveError::Matrix)?;

        for row in 0..self.rows {
            let coeff_zero = (0..self.cols)
                .all(|col| augmented.data[row * width + col].num == 0);
            if coeff_zero {
                if let Some(rhs_col) =
                    (0..rhs.cols).find(|&col| augmented.data[row * width + self.cols + col].num != 0)
                {
                    let witness = augmented.data[row * width..(row + 1) * width].to_vec();
                    return Ok(Solution::None {
                        row,
                        rhs_col,
                        witness,
                    });
                }
            }
        }

        let value_len = self
            .cols
            .checked_mul(rhs.cols)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![zero(); value_len];
        for (pivot_row, &pivot_col) in pivots.iter().enumerate() {
            for rhs_col in 0..rhs.cols {
                particular_data[pivot_col * rhs.cols + rhs_col] =
                    augmented.data[pivot_row * width + self.cols + rhs_col];
            }
        }
        let particular = Matrix {
            rows: self.cols,
            cols: rhs.cols,
            data: particular_data,
        };

        if pivots.len() == self.cols {
            return Ok(Solution::Unique {
                values: particular,
                pivots,
            });
        }

        let free: Vec<usize> = (0..self.cols)
            .filter(|col| !pivots.contains(col))
            .collect();

        let basis_len = self
            .cols
            .checked_mul(free.len())
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![zero(); basis_len];

        for (basis_col, &free_col) in free.iter().enumerate() {
            basis_data[free_col * free.len() + basis_col] = one();
            for (pivot_row, &pivot_col) in pivots.iter().enumerate() {
                let value = augmented.data[pivot_row * width + free_col]
                    .neg()
                    .map_err(|source| {
                        SolveError::Matrix(MatrixError::Arithmetic {
                            row: pivot_col,
                            col: basis_col,
                            step: pivots.len(),
                            source,
                        })
                    })?;
                basis_data[pivot_col * free.len() + basis_col] = value;
            }
        }

        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix {
                rows: self.cols,
                cols: basis_len.checked_div(self.cols).unwrap_or(0),
                data: basis_data,
            },
            pivots,
        })
    }
}

fn main() {}
