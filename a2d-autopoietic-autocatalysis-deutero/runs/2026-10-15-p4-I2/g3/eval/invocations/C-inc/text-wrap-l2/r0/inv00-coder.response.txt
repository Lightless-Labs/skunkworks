#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let mut normalized = String::with_capacity(input.len());
    let mut chars = input.chars().peekable();
    while let Some(ch) = chars.next() {
        if ch == '\r' {
            if chars.peek() == Some(&'\n') {
                chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut i = 0;

    while i < chars.len() {
        if chars[i] == '\n' {
            let run_start = i;
            while i < chars.len() && chars[i] == '\n' {
                i += 1;
            }
            let run_len = i - run_start;
            if run_len >= 2 {
                wrap_segment(&chars[segment_start..run_start], options, &mut output);
                output.extend(std::iter::repeat_n(String::new(), run_len - 1));
                segment_start = i;
            }
        } else {
            i += 1;
        }
    }

    wrap_segment(&chars[segment_start..], options, &mut output);
    Ok(output)
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn wrap_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words: Vec<Vec<char>> = Vec::new();
    let mut word = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !word.is_empty() {
                words.push(std::mem::take(&mut word));
            }
        } else {
            word.push(ch);
        }
    }
    if !word.is_empty() {
        words.push(word);
    }

    if words.is_empty() {
        return;
    }

    let mut line = String::new();
    let mut line_content_len = 0usize;
    let mut line_number = 0usize;

    for word in words {
        let mut offset = 0usize;

        while offset < word.len() {
            let indent = if line_number == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            let capacity = options.width - indent;
            let suffix_len = word.len() - offset;

            if suffix_len <= capacity {
                let separator_len = usize::from(line_content_len > 0);
                if line_content_len + separator_len + suffix_len <= capacity {
                    if line_content_len > 0 {
                        line.push(' ');
                        line_content_len += 1;
                    }
                    for &ch in &word[offset..] {
                        line.push(ch);
                    }
                    line_content_len += suffix_len;
                    offset = word.len();
                } else {
                    emit_line(
                        &mut line,
                        line_content_len,
                        indent,
                        output,
                    );
                    line_content_len = 0;
                    line_number += 1;
                }
            } else if line_content_len == 0 {
                let end = offset + capacity;
                for &ch in &word[offset..end] {
                    line.push(ch);
                }
                line_content_len = capacity;
                offset = end;
                emit_line(&mut line, line_content_len, indent, output);
                line_content_len = 0;
                line_number += 1;
            } else {
                let available = capacity - line_content_len - 1;
                if available >= 2 {
                    line.push(' ');
                    line_content_len += 1;
                    let end = offset + available;
                    for &ch in &word[offset..end] {
                        line.push(ch);
                    }
                    line_content_len += available;
                    offset = end;
                    emit_line(&mut line, line_content_len, indent, output);
                    line_content_len = 0;
                    line_number += 1;
                } else {
                    emit_line(&mut line, line_content_len, indent, output);
                    line_content_len = 0;
                    line_number += 1;
                }
            }
        }
    }

    if line_content_len > 0 {
        let indent = if line_number == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        emit_line(&mut line, line_content_len, indent, output);
    }
}

fn emit_line(
    content: &mut String,
    content_len: usize,
    indent: usize,
    output: &mut Vec<String>,
) {
    let mut rendered = String::with_capacity(indent + content.len());
    rendered.extend(std::iter::repeat_n(' ', indent));
    rendered.push_str(content);
    output.push(rendered);
    content.clear();
    let _ = content_len;
}

fn main() {}
