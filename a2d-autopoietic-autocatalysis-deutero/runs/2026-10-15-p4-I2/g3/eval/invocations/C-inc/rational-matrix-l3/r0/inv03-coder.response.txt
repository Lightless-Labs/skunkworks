use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let negative = (num < 0) != (den < 0);
    let numerator_magnitude = num.unsigned_abs();
    let denominator_magnitude = den.unsigned_abs();
    let gcd = gcd_u128(numerator_magnitude, denominator_magnitude);
    let n = numerator_magnitude / gcd;
    let d = denominator_magnitude / gcd;

    if d > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }

    let signed_n = if negative {
        if n == (1u128 << 63) {
            i64::MIN
        } else if n <= i64::MAX as u128 {
            -(n as i64)
        } else {
            return Err(RationalError::Overflow);
        }
    } else if n <= i64::MAX as u128 {
        n as i64
    } else {
        return Err(RationalError::Overflow);
    };

    Ok(Rational {
        num: signed_n,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        rational_from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(err(i, ParseKind::InvalidByte));
                }
                slash = Some(i);
            } else if b.is_ascii_digit() || (b == b'-' && i == 0) {
                // Valid grammar byte.
            } else {
                return Err(err(i, ParseKind::InvalidByte));
            }
        }

        let num_end = slash.unwrap_or(bytes.len());
        let num_start = usize::from(bytes[0] == b'-');
        if num_start == num_end {
            return Err(err(bytes.len(), ParseKind::InvalidByte));
        }
        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(err(bytes.len(), ParseKind::InvalidByte));
            }
        }

        let num_digits = &bytes[num_start..num_end];
        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(err(num_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && num_digits == b"0" {
            return Err(err(0, ParseKind::NonCanonical));
        }

        let den_range = slash.map(|s| (s + 1, bytes.len()));
        if let Some((start, end)) = den_range {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(err(start + 1, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let numerator_limit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let mut numerator_magnitude = 0u128;
        for &b in num_digits {
            let digit = (b - b'0') as u128;
            if numerator_magnitude > (numerator_limit - digit) / 10 {
                return Err(err(0, ParseKind::OutOfRange));
            }
            numerator_magnitude = numerator_magnitude * 10 + digit;
        }

        let mut denominator_value = 1u64;
        if let Some((start, end)) = den_range {
            for &b in &bytes[start..end] {
                let digit = (b - b'0') as u64;
                if denominator_value > (i64::MAX as u64 - digit) / 10 {
                    return Err(err(start, ParseKind::OutOfRange));
                }
                denominator_value = denominator_value * 10 + digit;
            }

            if denominator_value == 0 {
                return Err(err(start, ParseKind::ZeroDenominator));
            }
            if denominator_value == 1 {
                return Err(err(start - 1, ParseKind::NonCanonical));
            }
            if numerator_magnitude == 0 {
                return Err(err(start - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(numerator_magnitude, denominator_value as u128) > 1 {
                return Err(err(start - 1, ParseKind::NonCanonical));
            }
        }

        let numerator = if negative {
            if numerator_magnitude == (1u128 << 63) {
                i64::MIN
            } else {
                -(numerator_magnitude as i64)
            }
        } else {
            numerator_magnitude as i64
        };

        Ok(Rational {
            num: numerator,
            den: denominator_value as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|base| base.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data
            .get(index)
            .copied()
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for row in 0..self.cols {
            for col in 0..self.rows {
                data.push(self.data[col * self.cols + row]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }

        let len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };

        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let left = self.data[row * self.cols + k];
                    let right = rhs.data[k * rhs.cols + col];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    fn eliminate(
        matrix: &mut Matrix,
        pivot_cols: usize,
    ) -> Result<(Vec<usize>, Vec<(usize, usize)>), MatrixError> {
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut pivot_row = 0usize;

        for pivot_col in 0..pivot_cols {
            if pivot_row == matrix.rows {
                break;
            }

            let mut chosen: Option<usize> = None;
            for row in pivot_row..matrix.rows {
                let value = matrix.data[row * matrix.cols + pivot_col];
                if value.num == 0 {
                    continue;
                }
                let key = (value.den, value.num.unsigned_abs(), row);
                if chosen.map_or(true, |best| {
                    let current = matrix.data[best * matrix.cols + pivot_col];
                    key < (current.den, current.num.unsigned_abs(), best)
                }) {
                    chosen = Some(row);
                }
            }

            let Some(chosen_row) = chosen else {
                continue;
            };

            if chosen_row != pivot_row {
                for col in 0..matrix.cols {
                    matrix
                        .data
                        .swap(pivot_row * matrix.cols + col, chosen_row * matrix.cols + col);
                }
                swaps.push((pivot_row, chosen_row));
            }

            let pivot = matrix.data[pivot_row * matrix.cols + pivot_col];
            for col in 0..matrix.cols {
                let index = pivot_row * matrix.cols + col;
                matrix.data[index] = matrix.data[index].div(&pivot).map_err(|source| {
                    MatrixError::Arithmetic {
                        row: pivot_row,
                        col,
                        step: pivot_row,
                        source,
                    }
                })?;
            }

            for row in 0..matrix.rows {
                if row == pivot_row {
                    continue;
                }

                let factor = matrix.data[row * matrix.cols + pivot_col];
                if factor.num == 0 {
                    continue;
                }

                for col in 0..matrix.cols {
                    let pivot_entry = matrix.data[pivot_row * matrix.cols + col];
                    let product = factor.mul(&pivot_entry).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col,
                            step: pivot_row,
                            source,
                        }
                    })?;
                    let index = row * matrix.cols + col;
                    matrix.data[index] = matrix.data[index].sub(&product).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col,
                            step: pivot_row,
                            source,
                        }
                    })?;
                }
            }

            pivots.push(pivot_col);
            pivot_row += 1;
        }

        Ok((pivots, swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let (pivots, swaps) = Self::eliminate(&mut matrix, self.cols)?;
        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }

        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };
        let one = Rational { num: 1, den: 1 };

        for row in 0..n {
            for col in 0..n {
                data.push(self.data[row * n + col]);
            }
            for col in 0..n {
                data.push(if row == col { one } else { zero });
            }
        }

        let mut augmented = Self {
            rows: n,
            cols: width,
            data,
        };
        let (pivots, _) = Self::eliminate(&mut augmented, n)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }

        let mut inverse_data = Vec::with_capacity(n * n);
        for row in 0..n {
            for col in 0..n {
                inverse_data.push(augmented.data[row * width + n + col]);
            }
        }

        Ok(Self {
            rows: n,
            cols: n,
            data: inverse_data,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

fn main() {}
