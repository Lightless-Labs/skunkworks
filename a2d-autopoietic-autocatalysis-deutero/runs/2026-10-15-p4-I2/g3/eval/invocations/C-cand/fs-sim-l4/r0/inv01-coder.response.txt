```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

struct MissingInfo {
    path: String,
    final_component: bool,
}

struct Resolver {
    visited: HashSet<String>,
    expansions: usize,
}

impl Resolver {
    fn new() -> Self {
        Self {
            visited: HashSet::new(),
            expansions: 0,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn normalize(path: &str, base: &str) -> String {
    let mut parts: Vec<&str> = if path.starts_with('/') {
        Vec::new()
    } else {
        base.split('/').filter(|p| !p.is_empty()).collect()
    };

    for component in path.split('/') {
        match component {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other),
        }
    }

    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_string)
        .collect()
}

fn from_components(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn parent_and_name(path: &str) -> (String, String) {
    let parts = components(path);
    if parts.is_empty() {
        return ("/".to_string(), String::new());
    }
    let name = parts[parts.len() - 1].clone();
    (from_components(&parts[..parts.len() - 1]), name)
}

fn join_child(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{name}")
    } else {
        format!("{parent}/{name}")
    }
}

fn is_within(path: &str, directory: &str) -> bool {
    path == directory
        || directory == "/"
        || path
            .strip_prefix(directory)
            .is_some_and(|rest| rest.starts_with('/'))
}

fn has_children(entries: &BTreeMap<String, Entry>, path: &str) -> bool {
    entries.keys().any(|candidate| {
        candidate != path
            && if path == "/" {
                candidate.starts_with('/')
            } else {
                candidate
                    .strip_prefix(path)
                    .is_some_and(|rest| rest.starts_with('/'))
            }
    })
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn resolve(
        &self,
        path: &str,
        resolver: &mut Resolver,
    ) -> Result<String, (Error, Option<MissingInfo>)> {
        let mut pending = components(path);
        let mut resolved: Vec<String> = Vec::new();
        let mut index = 0;

        loop {
            if index >= pending.len() {
                return Ok(from_components(&resolved));
            }

            let component = pending[index].clone();
            let candidate = join_child(&from_components(&resolved), &component);
            let is_final = index + 1 == pending.len();

            let Some(entry) = self.entries.get(&candidate) else {
                return Err((
                    Error::Missing(candidate.clone()),
                    Some(MissingInfo {
                        path: candidate,
                        final_component: is_final,
                    }),
                ));
            };

            if let Entry::Link(target) = entry {
                if !resolver.visited.insert(candidate.clone()) {
                    return Err((Error::Loop(candidate), None));
                }
                if resolver.expansions >= 8 {
                    return Err((Error::Depth(candidate), None));
                }
                resolver.expansions += 1;

                let parent = from_components(&resolved);
                let target_path = normalize(target, &parent);
                let mut expanded = components(&target_path);
                expanded.extend(pending[index + 1..].iter().cloned());
                pending = components(&from_components(&expanded));
                resolved.clear();
                index = 0;
                continue;
            }

            if !is_final && !matches!(entry, Entry::Dir) {
                return Err((Error::NotDir(candidate), None));
            }

            resolved.push(component);
            index += 1;
        }
    }

    fn resolve_path(&self, path: &str) -> Result<String, Error> {
        self.resolve(path, &mut Resolver::new())
            .map_err(|(error, _)| error)
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = parent_and_name(path);
        let resolved_parent = self.resolve_path(&parent)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, name)),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        validate_path(path)?;
        let canonical = normalize(path, &self.cwd);
        let (parent, name) = self.resolve_parent(&canonical)?;
        let destination = join_child(&parent, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        validate_path(path)?;
        let canonical = normalize(path, &self.cwd);
        let mut resolver = Resolver::new();

        match self.resolve(&canonical, &mut resolver) {
            Ok(resolved) => match self.entries.get_mut(&resolved) {
                Some(Entry::Dir) => Err(Error::IsDir(resolved)),
                Some(Entry::File(bytes)) => {
                    *bytes = data.to_vec();
                    Ok(())
                }
                Some(Entry::Link(_)) | None => unreachable!(),
            },
            Err((Error::Missing(_), Some(info))) if info.final_component => {
                let (parent, name) = parent_and_name(&info.path);
                let resolved_parent = self.resolve_path(&parent)?;
                match self.entries.get(&resolved_parent) {
                    Some(Entry::Dir) => {
                        let destination = join_child(&resolved_parent, &name);
                        self.entries
                            .insert(destination, Entry::File(data.to_vec()));
                        Ok(())
                    }
                    Some(_) => Err(Error::NotDir(resolved_parent)),
                    None => Err(Error::Missing(resolved_parent)),
                }
            }
            Err((error, _)) => Err(error),
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        validate_path(path)?;
        let canonical = normalize(path, &self.cwd);
        let resolved = self.resolve_path(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) | None => unreachable!(),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        validate_path(path)?;
        let canonical = normalize(path, &self.cwd);
        let resolved = self.resolve_path(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                let mut children = Vec::new();
                for (child_path, entry) in &self.entries {
                    if child_path == "/" {
                        continue;
                    }
                    let (parent, name) = parent_and_name(child_path);
                    if parent == resolved {
                        children.push((name, entry.kind()));
                    }
                }
                children.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(children)
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => unreachable!(),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        validate_path(path)?;
        let canonical = normalize(path, &self.cwd);
        let resolved = self.resolve_path(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => unreachable!(),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        validate_path(from)?;
        validate_path(to)?;
        let source = normalize(from, &self.cwd);
        let destination = normalize(to, &self.cwd);

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source)?;
        let source_path = join_child(&source_parent, &source_name);
        let Some(source_entry) = self.entries.get(&source_path).cloned() else {
            return Err(Error::Missing(source_path));
        };

        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent, destination_name) = self.resolve_parent(&destination)?;
        let destination_path = join_child(&destination_parent, &destination_name);

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && is_within(&self.cwd, &destination_path)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && is_within(&destination_path, &source_path) {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(existing) = &destination_entry {
            let source_is_dir = matches!(source_entry, Entry::Dir);
            let destination_is_dir = matches!(existing, Entry::Dir);
            if source_is_dir && !destination_is_dir {
                return Err(Error::NotDir(destination_path));
            }
            if !source_is_dir && destination_is_dir {
                return Err(Error::IsDir(destination_path));
            }
            if source_is_dir && destination_is_dir && has_children(&self.entries, &destination_path) {
                return Err(Error::NotEmpty(destination_path));
            }
        }

        let mut moved = Vec::new();
        for (old_path, entry) in &self.entries {
            if old_path == &source_path || is_within(old_path, &source_path) {
                let suffix = &old_path[source_path.len()..];
                moved.push((format!("{destination_path}{suffix}"), entry.clone()));
            }
        }

        self.entries.retain(|old_path, _| {
            old_path != &source_path && !is_within(old_path, &source_path)
        });

        if destination_entry.is_some() {
            self.entries.retain(|old_path, _| {
                old_path != &destination_path && !is_within(old_path, &destination_path)
            });
        }

        for (new_path, entry) in moved {
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        validate_path(path)?;
        let canonical = normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Root);
        }

        let (parent, name) = self.resolve_parent(&canonical)?;
        let resolved = join_child(&parent, &name);
        let Some(entry) = self.entries.get(&resolved) else {
            return Err(Error::Missing(resolved));
        };

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &resolved) {
            return Err(Error::Busy(resolved));
        }

        if matches!(entry, Entry::Dir) && has_children(&self.entries, &resolved) && !recursive {
            return Err(Error::NotEmpty(resolved));
        }

        self.entries.retain(|candidate, _| {
            candidate != &resolved && !is_within(candidate, &resolved)
        });
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        validate_path(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        let canonical = normalize(link, &self.cwd);
        let (parent, name) = parent_and_name(&canonical);
        let resolved_parent = self.resolve_path(&parent)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }

        let destination = join_child(&resolved_parent, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries
            .insert(destination, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        validate_path(path)?;
        let canonical = normalize(path, &self.cwd);
        let (parent, name) = parent_and_name(&canonical);
        let resolved_parent = self.resolve_path(&parent)?;
        let final_path = join_child(&resolved_parent, &name);

        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }

        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        validate_path(path)?;
        let canonical = normalize(path, &self.cwd);
        let resolved = self.resolve_path(&canonical)?;

        let mut rows = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path != &resolved && !is_within(entry_path, &resolved) {
                continue;
            }

            let size = match entry {
                Entry::Dir => 0,
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
            };
            rows.push((entry_path.clone(), size));
        }

        let total: u64 = rows.iter().map(|(_, size)| *size).sum();
        if let Some(row) = rows.iter_mut().find(|(entry_path, _)| entry_path == &resolved) {
            row.1 = total;
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}

fn main() {}
```
