#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Specifier {
    name_start: usize,
    name_end: usize,
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<u8>,
    specifiers: Vec<Specifier>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(relative) = bytes[open + 1..close]
        .iter()
        .position(|&byte| byte == b'{')
    {
        return Err(Error::NestedOpen {
            pos: open + 1 + relative,
        });
    }

    let mut content_start = open + 1;
    let operator = if content_start < close
        && matches!(
            bytes[content_start],
            b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'
        )
    {
        let op = bytes[content_start];
        content_start += 1;
        if !matches!(op, b'+' | b'#') {
            return Err(Error::UnsupportedOperator {
                pos: content_start - 1,
            });
        }
        Some(op)
    } else {
        None
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specifiers: Vec<Specifier> = Vec::new();
    let mut start = content_start;

    loop {
        let end = bytes[start..close]
            .iter()
            .position(|&byte| byte == b',')
            .map(|relative| start + relative)
            .unwrap_or(close);

        if start == end {
            return Err(Error::EmptyVariable { pos: start });
        }

        let mut name_end = start;
        if !is_name_start(bytes[name_end]) {
            return Err(Error::InvalidName { pos: name_end });
        }
        name_end += 1;

        while name_end < end && is_name_continue(bytes[name_end]) {
            name_end += 1;
        }

        if name_end < end {
            return Err(Error::InvalidName { pos: name_end });
        }

        for previous in &specifiers {
            if template[previous.name_start..previous.name_end] == template[start..name_end] {
                return Err(Error::DuplicateVariable { pos: start });
            }
        }

        specifiers.push(Specifier {
            name_start: start,
            name_end,
        });

        if end == close {
            break;
        }
        start = end + 1;
    }

    Ok(Expression {
        start: open,
        end: close,
        operator,
        specifiers,
    })
}

fn parse_template(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let Some(relative_close) = bytes[index + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                else {
                    return Err(Error::Unclosed { pos: index });
                };

                let close = index + 1 + relative_close;
                expressions.push(parse_expression(template, index, close)?);
                index = close + 1;
            }
            _ => index += 1,
        }
    }

    Ok(expressions)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> u8 {
    if value < 10 {
        b'0' + value
    } else {
        b'A' + value - 10
    }
}

fn encode_value(value: &str, allow_reserved: bool, output: &mut String) {
    let bytes = value.as_bytes();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) =
                (hex_value(bytes[index + 1]), hex_value(bytes[index + 2]))
            {
                output.push('%');
                output.push(hex_digit(high) as char);
                output.push(hex_digit(low) as char);
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4) as char);
            output.push(hex_digit(byte & 0x0f) as char);
        }

        index += 1;
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let allow_reserved = matches!(expression.operator, Some(b'+' | b'#'));
        let mut fields = Vec::new();

        for specifier in expression.specifiers {
            let name = &template[specifier.name_start..specifier.name_end];
            if let Some(value) = vars
                .iter()
                .rev()
                .find(|(binding_name, _)| *binding_name == name)
                .map(|(_, value)| *value)
            {
                fields.push(value);
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some(b'#') {
                output.push('#');
            }

            for (index, value) in fields.into_iter().enumerate() {
                if index > 0 {
                    output.push(',');
                }
                encode_value(value, allow_reserved, &mut output);
            }
        }

        literal_start = expression.end + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn main() {}
