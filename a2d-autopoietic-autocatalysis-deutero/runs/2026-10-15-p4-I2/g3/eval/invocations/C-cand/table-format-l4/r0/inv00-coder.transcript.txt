fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {

        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {

            return Err(TableError::EmptyRow { row });

                row,
                expected: n,
                actual: cells.len(),

        }

            expected: n,
            actual: options.alignments.len(),

            expected: n,
            actual: options.overflow.len(),

            provided: options.max_width,

                    for _ in 2..widths[column] {

                    }
                    out.push(':');
                }
            }
            out.push(' ');
            out.push('|');
        }
        out.push('\n');
    };

        emit_grid_border(&mut output, &widths);

                    widths[column],
                    &overflows[column],

                        .cloned()
                        .unwrap_or_default()
