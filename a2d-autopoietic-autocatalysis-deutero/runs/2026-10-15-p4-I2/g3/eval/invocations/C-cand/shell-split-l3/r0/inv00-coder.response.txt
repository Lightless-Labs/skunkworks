#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn append_literal(word: &mut Word, c: char, active_glob: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            }
        } else {
            result.push(c);
        }
    }

    result
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let after_open = dollar + 2;
    let mut cursor = after_open;
    let mut close = None;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += c.len_utf8();
            if cursor < input.len() {
                let next = input[cursor..].chars().next().unwrap();
                cursor += next.len_utf8();
            }
        } else if c == '}' {
            close = Some(cursor);
            break;
        } else {
            cursor += c.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[after_open..close];

    let name_end = contents
        .char_indices()
        .find_map(|(i, c)| (!is_name_continue(c)).then_some(i))
        .unwrap_or(contents.len());

    let name = &contents[..name_end];
    let valid_name = !name.is_empty()
        && name.chars().next().is_some_and(is_name_start)
        && name.chars().all(is_name_continue);

    if !valid_name {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let remainder = &contents[name_end..];
    let value = if remainder.is_empty() {
        lookup(env, name).unwrap_or("").to_string()
    } else if let Some(default) = remainder.strip_prefix(":-") {
        match lookup(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close + 1, value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.bytes().position(|byte| byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut cursor = 0usize;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let next_cursor = cursor + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_literal(&mut word, c, false);
                }
                cursor = next_cursor;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    cursor = next_cursor;
                } else if c == '\\' {
                    let escape_pos = cursor;
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    let next = input[next_cursor..].chars().next().unwrap();
                    let after_next = next_cursor + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        append_literal(&mut word, next, false);
                        cursor = after_next;
                    } else if next == '\n' {
                        cursor = after_next;
                    } else {
                        append_literal(&mut word, '\\', false);
                        cursor = next_cursor;
                    }
                } else if c == '$' {
                    if input[next_cursor..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[next_cursor..].starts_with('{') {
                        active = true;
                        let (end, value) = parse_braced(input, cursor, env)?;
                        append_expansion(&mut word, &value, false);
                        cursor = end;
                    } else if let Some((end, name)) = parse_unbraced_name(input, next_cursor) {
                        active = true;
                        let value = lookup(env, name).unwrap_or("");
                        append_expansion(&mut word, value, false);
                        cursor = end;
                    } else {
                        append_literal(&mut word, '$', false);
                        cursor = next_cursor;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else {
                    append_literal(&mut word, c, false);
                    cursor = next_cursor;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    cursor = next_cursor;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single;
                    quote_start = cursor;
                    cursor = next_cursor;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double;
                    quote_start = cursor;
                    cursor = next_cursor;
                } else if c == '\\' {
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let next = input[next_cursor..].chars().next().unwrap();
                    let after_next = next_cursor + next.len_utf8();
                    if next == '\n' {
                        cursor = after_next;
                    } else {
                        active = true;
                        append_literal(&mut word, next, false);
                        cursor = after_next;
                    }
                } else if c == '$' {
                    if input[next_cursor..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[next_cursor..].starts_with('{') {
                        active = true;
                        let (end, value) = parse_braced(input, cursor, env)?;
                        append_expansion(&mut word, &value, true);
                        cursor = end;
                    } else if let Some((end, name)) = parse_unbraced_name(input, next_cursor) {
                        active = true;
                        let value = lookup(env, name).unwrap_or("");
                        append_expansion(&mut word, value, true);
                        cursor = end;
                    } else {
                        active = true;
                        append_literal(&mut word, '$', false);
                        cursor = next_cursor;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else {
                    active = true;
                    append_literal(&mut word, c, true);
                    cursor = next_cursor;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}

fn append_expansion(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append_literal(word, c, active_glob);
    }
}

fn parse_unbraced_name(input: &str, start: usize) -> Option<(usize, &str)> {
    if start >= input.len() {
        return None;
    }
    let first = input[start..].chars().next()?;
    if !is_name_start(first) {
        return None;
    }

    let mut end = start + first.len_utf8();
    while end < input.len() {
        let c = input[end..].chars().next()?;
        if !is_name_continue(c) {
            break;
        }
        end += c.len_utf8();
    }

    Some((end, &input[start..end]))
}

fn main() {}
