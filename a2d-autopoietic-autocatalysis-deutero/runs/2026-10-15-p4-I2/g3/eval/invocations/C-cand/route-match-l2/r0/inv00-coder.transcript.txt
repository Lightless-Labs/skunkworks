#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct ParsedPattern {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<ParsedPattern>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let parsed = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(parsed);
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (segments, _) = parse_path(head)?;

        let mut best: Option<(usize, Vec<u8>, Vec<(String, String)>)> = None;

        for (route_id, route) in self.routes.iter().enumerate() {
            let Some(params) = match_route(&route.segments, &segments) else {
                continue;
            };

            let replace = match &best {
                None => true,
                Some((best_id, best_rank, _)) => {
                    route.rank > *best_rank
                        || (route.rank == *best_rank && route_id < *best_id)
                }
            };
            if replace {
                best = Some((route_id, route.rank.clone(), params));
            }
        }

        match best {
            Some((route, _, params)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(input: &str) -> Result<ParsedPattern, Error> {
    let (raw_segments, trailing) = split_segments(input, true)?;
    let mut segments = Vec::with_capacity(raw_segments.len());
    let mut names = Vec::<String>::new();
    let mut rank = Vec::with_capacity(raw_segments.len() + 1);
    let mut saw_wildcard = false;

    for (index, (raw, start)) in raw_segments.iter().enumerate() {
        if saw_wildcard {
            return Err(Error::InvalidPattern { at: *start });
        }

        let bytes = raw.as_bytes();
        if bytes.first() == Some(&b':') || bytes.first() == Some(&b'*') {
            let marker = bytes[0];
            let name = &raw[1..];
            if !valid_name(name) || names.iter().any(|n| n == name) {
                return Err(Error::InvalidPattern { at: *start });
            }
            names.push(name.to_owned());
            if marker == b':' {
                segments.push(Segment::Param(name.to_owned()));
                rank.push(1);
            } else {
                if index + 1 != raw_segments.len() {
                    return Err(Error::InvalidPattern { at: *start });
                }
                saw_wildcard = true;
                segments.push(Segment::Wildcard(name.to_owned()));
                rank.push(0);
            }
        } else {
            let decoded = decode_segment(raw, *start, true)?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }
    }

    if trailing {
        return Err(Error::InvalidPattern {
            at: input.len() - 1,
        });
    }
    if !saw_wildcard {
        rank.push(3);
    }

    Ok(ParsedPattern { segments, rank })
}

fn parse_path(input: &str) -> Result<(Vec<String>, bool), Error> {
    let (raw_segments, trailing) = split_segments(input, false)?;
    let mut segments = Vec::with_capacity(raw_segments.len());
    for (raw, start) in raw_segments {
        segments.push(decode_segment(raw, start, false)?);
    }
    if trailing {
        return Err(Error::InvalidPath {
            at: input.len() - 1,
        });
    }
    Ok((segments, trailing))
}

// Returns nonempty segments and whether the input had a trailing slash.
fn split_segments(input: &str, pattern: bool) -> Result<(Vec<(&str, usize)>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { input.len() - 1 } else { input.len() };
    let mut result = Vec::new();
    let mut start = 1;

    while start < end {
        let next = input[start..end]
            .find('/')
            .map(|offset| start + offset)
            .unwrap_or(end);
        if next == start {
            return Err(invalid(start));
        }
        result.push((&input[start..next], start));
        start = next + 1;
    }

    if !trailing && start == input.len() {
        return Err(invalid(input.len() - 1));
    }

    Ok((result, trailing))
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, base: usize, pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(segment_error(pattern, base + i));
            }
            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(segment_error(pattern, base + i));
            }
            decoded.push(value);
            i += 3;
            continue;
        }

        if !is_unreserved(byte) {
            return Err(segment_error(pattern, base + i));
        }
        decoded.push(byte);
        i += 1;
    }

    // Decoded bytes are constrained to printable ASCII.
    Ok(String::from_utf8(decoded).unwrap())
}

fn segment_error(pattern: bool, at: usize) -> Error {
    if pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(byte: u8) -> bool {
    byte.is_ascii_digit() || (b'A'..=b'F').contains(&byte)
}

fn hex_value(byte: u8) -> u8 {
    match byte {
        b'0'..=b'9' => byte - b'0',
        b'A'..=b'F' => byte - b'A' + 10,
        _ => 0,
    }
}

fn match_route(pattern: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for (pattern_index, segment) in pattern.iter().enumerate() {
        match segment {
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                return Some(params);
            }
            Segment::Static(expected) => {
                if path.get(path_index)? != expected {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
        }

        if pattern_index + 1 == pattern.len() && path_index != path.len() {
            return None;
        }
    }

    if path_index == path.len() {
        Some(params)
    } else {
        None
    }
}
