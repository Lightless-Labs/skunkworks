            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

        self.limbs.is_empty()

                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }

        if factor == 0 || self.is_zero() {
            self.limbs.clear();
            self.negative = false;
            return;
        }

        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = (*limb as u64) * factor as u64 + carry;
            *limb = value as u32;
            carry = value >> 32;
        }
        if carry != 0 {

        }

        let mut carry = addend as u64;
        let mut i = 0;

            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let value = self.limbs[i] as u64 + carry;
            self.limbs[i] = value as u32;
            carry = value >> 32;
            i += 1;
        }

        let mut remainder = 0u64;
        for limb in self.limbs.iter_mut().rev() {
            let value = (remainder << 32) | *limb as u64;
            *limb = (value / divisor as u64) as u32;
            remainder = value % divisor as u64;
        }
        self.normalize();
        remainder as u32

        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        if text.is_empty() {
            return Err(Error::Empty);
        }

        let bytes = text.as_bytes();
        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),

        if start == bytes.len() {
            return Err(Error::SignOnly);

                Some(value) if value < radix => {}
                _ => {
                    invalid = Some(index);
                    break;
                }
            }

            return Err(Error::InvalidDigit { index });

                let next_is_digit = index + 1 < bytes.len()

                        .is_some_and(|value| value < radix);

                    separator = Some(index);
                    break;
                }

            return Err(Error::Separator { index });

                index: first_digit_index,

        radix: u32,

            return Err(Error::InvalidRadix { radix });

        let mut magnitude = self.abs();

            }
            output.push(*ch);
        }
        Ok(output)
    }

            !quotient.is_zero() && (self.negative != rhs.negative);
        remainder.negative = !remainder.is_zero() && self.negative;
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > 1024 {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: 1024,
            });
        }

        let mut result = Self::one();
        let mut base = self.clone();
        let mut power = exponent;
        while power != 0 {
            if power & 1 != 0 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let remainder = Self::div_rem_abs(&a, &b).1;
            a = b;
            b = remainder;
        }
        a.negative = false;
        a
    }
}
