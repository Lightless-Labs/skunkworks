    } else {

            len: effective.len(),

    }

    Ok(effective)
}

fn cross(a: Point, b: Point, q: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let aqx = q.x as i128 - a.x as i128;
    let aqy = q.y as i128 - a.y as i128;

fn squared_distance(a: Point, b: Point) -> u128 {

    let dy = (b.y as i128 - a.y as i128) as u128;
    dx * dx + dy * dy
}

fn rounded_sqrt(value: u128) -> u64 {
    let mut low = 0u128;
    let mut high = 1u128 << 64;

    while low + 1 < high {
        let mid = low + (high - low) / 2;
        if mid <= value / mid {
            low = mid;
        } else {
            high = mid;
        }
    }

    let floor = low;
    let remainder = value - floor * floor;
    if remainder >= floor + 1 {
        (floor + 1) as u64
    } else {
        floor as u64
    }
}

    rounded_sqrt(squared_distance(a, b) * 1_000_000)
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let points = prepare(points)?;
    let n = points.len();

    let mut signed_area2 = 0i128;
    let mut edge_milli = Vec::with_capacity(n);
    let mut perimeter_milli = 0u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];

        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;

        let length = edge_length_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let replace = match longest_edge {
            None => true,
            Some(j) => {
                let previous_length = edge_milli[j];
                if length != previous_length {
                    length > previous_length
                } else {
                    let previous_a: Point = points[j];
                    let previous_b: Point = points[(j + 1) % n];
                    let previous_pair =
                        (previous_a.min(previous_b), previous_a.max(previous_b));
                    let current_pair = (a.min(b), a.max(b));
                    current_pair < previous_pair || (current_pair == previous_pair && i < j)
                }
            }
        };

        if replace {
            longest_edge = Some(i);
        }
    }

    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let points = prepare(points)?;

    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }

    for (index, &query) in queries.iter().enumerate() {
        if !valid_point(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let n = points.len();
    let mut results = Vec::with_capacity(queries.len());

    for &query in queries {
        let mut boundary_edge = None;
        let mut toggles = false;

        for i in 0..n {
            let a = points[i];
            let b = points[(i + 1) % n];

            if on_segment(a, b, query) {
                boundary_edge = Some(i);
                break;
            }

            if (a.y > query.y) != (b.y > query.y) {
                let side = cross(a, b, query);
                if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                    toggles = !toggles;
                }
            }
        }

        results.push(match boundary_edge {
            Some(edge) => Location::Boundary { edge },
            None if toggles => Location::Inside,
            None => Location::Outside,
        });
    }

    Ok(results)
}
