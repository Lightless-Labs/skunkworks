```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    UrlSafe,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Forbidden,
    Optional,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Whitespace {
    None,
    IgnoreAscii,
    Wrapped { width: usize, ending: LineEnding },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Options {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub whitespace: Whitespace,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidByte { offset: usize, byte: u8 },
    BadPadding { offset: usize },
    Truncated { offset: usize },
    AlreadyFinished,
    InvalidWidth { width: usize },
    BadWrap { offset: usize },
}

fn validate_options(options: Options) -> Result<(), Error> {
    if let Whitespace::Wrapped { width, .. } = options.whitespace {
        if !(4..=120).contains(&width) || width % 4 != 0 {
            return Err(Error::InvalidWidth { width });
        }
    }
    Ok(())
}

fn alphabet_byte(index: u8, alphabet: Alphabet) -> u8 {
    match index {
        0..=25 => b'A' + index,
        26..=51 => b'a' + (index - 26),
        52..=61 => b'0' + (index - 52),
        62 => match alphabet {
            Alphabet::Standard => b'+',
            Alphabet::UrlSafe => b'-',
        },
        _ => match alphabet {
            Alphabet::Standard => b'/',
            Alphabet::UrlSafe => b'_',
        },
    }
}

fn digit_value(byte: u8, alphabet: Alphabet) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'a'..=b'z' => Some(byte - b'a' + 26),
        b'0'..=b'9' => Some(byte - b'0' + 52),
        b'+' if alphabet == Alphabet::Standard => Some(62),
        b'/' if alphabet == Alphabet::Standard => Some(63),
        b'-' if alphabet == Alphabet::UrlSafe => Some(62),
        b'_' if alphabet == Alphabet::UrlSafe => Some(63),
        _ => None,
    }
}

fn append_encoded_symbols(
    symbols: &[u8],
    options: Options,
    row_symbols: &mut usize,
    output: &mut String,
) {
    let ending = match options.whitespace {
        Whitespace::Wrapped { ending, .. } => Some(ending),
        _ => None,
    };
    let width = match options.whitespace {
        Whitespace::Wrapped { width, .. } => width,
        _ => usize::MAX,
    };

    for &symbol in symbols {
        if let Some(ending) = ending {
            if *row_symbols == width {
                match ending {
                    LineEnding::Lf => output.push('\n'),
                    LineEnding::CrLf => output.push_str("\r\n"),
                }
                *row_symbols = 0;
            }
        }
        output.push(symbol as char);
        *row_symbols += 1;
    }
}

pub fn encode(input: &[u8], options: Options) -> Result<String, Error> {
    validate_options(options)?;
    let mut encoder = Encoder::new(options)?;
    let mut output = encoder.push(input)?;
    output.push_str(&encoder.finish()?);
    Ok(output)
}

pub fn decode(input: &[u8], options: Options) -> Result<Vec<u8>, Error> {
    validate_options(options)?;
    let mut decoder = Decoder::new(options)?;
    let mut output = decoder.push(input)?;
    output.extend(decoder.finish()?);
    Ok(output)
}

pub struct Encoder {
    options: Options,
    tail: Vec<u8>,
    row_symbols: usize,
    finished: bool,
}

impl Encoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            tail: Vec::new(),
            row_symbols: 0,
            finished: false,
        })
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        let mut combined = Vec::with_capacity(self.tail.len() + input.len());
        combined.extend_from_slice(&self.tail);
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 3 * 3;
        let mut output = String::new();
        for chunk in combined[..complete_len].chunks_exact(3) {
            let symbols = [
                alphabet_byte(chunk[0] >> 2, self.options.alphabet),
                alphabet_byte(((chunk[0] & 0x03) << 4) | (chunk[1] >> 4), self.options.alphabet),
                alphabet_byte(((chunk[1] & 0x0f) << 2) | (chunk[2] >> 6), self.options.alphabet),
                alphabet_byte(chunk[2] & 0x3f, self.options.alphabet),
            ];
            append_encoded_symbols(&symbols, self.options, &mut self.row_symbols, &mut output);
        }
        self.tail.clear();
        self.tail.extend_from_slice(&combined[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, Error> {
        if self.finished {
            return Err(Error::AlreadyFinished);
        }
        self.finished = true;
        let mut output = String::new();
        let mut symbols = Vec::with_capacity(4);

        match self.tail.len() {
            0 => {}
            1 => {
                let byte = self.tail[0];
                symbols.push(alphabet_byte(byte >> 2, self.options.alphabet));
                symbols.push(alphabet_byte((byte & 0x03) << 4, self.options.alphabet));
                if self.options.padding != Padding::Forbidden {
                    symbols.extend_from_slice(b"==");
                }
            }
            2 => {
                let first = self.tail[0];
                let second = self.tail[1];
                symbols.push(alphabet_byte(first >> 2, self.options.alphabet));
                symbols.push(alphabet_byte(
                    ((first & 0x03) << 4) | (second >> 4),
                    self.options.alphabet,
                ));
                symbols.push(alphabet_byte((second & 0x0f) << 2, self.options.alphabet));
                if self.options.padding != Padding::Forbidden {
                    symbols.push(b'=');
                }
            }
            _ => unreachable!(),
        }
        append_encoded_symbols(&symbols, self.options, &mut self.row_symbols, &mut output);
        self.tail.clear();
        Ok(output)
    }
}

pub struct Decoder {
    options: Options,
    group: Vec<(u8, usize)>,
    terminal_padding: bool,
    offset: usize,
    row_symbols: usize,
    wrap_stage: u8,
    wrap_start: usize,
    after_separator: bool,
    closed: bool,
    sticky_error: Option<Error>,
}

impl Decoder {
    pub fn new(options: Options) -> Result<Self, Error> {
        validate_options(options)?;
        Ok(Self {
            options,
            group: Vec::with_capacity(4),
            terminal_padding: false,
            offset: 0,
            row_symbols: 0,
            wrap_stage: 0,
            wrap_start: 0,
            after_separator: false,
            closed: false,
            sticky_error: None,
        })
    }

    fn record_error(&mut self, error: Error) -> Error {
        self.sticky_error = Some(error.clone());
        error
    }

    fn decode_complete_group(&mut self, output: &mut Vec<u8>) -> Result<(), Error> {
        let group = &self.group;
        let a = digit_value(group[0].0, self.options.alphabet).unwrap();
        let b = digit_value(group[1].0, self.options.alphabet).unwrap();

        if group[2].0 == b'=' {
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset: group[2].1 });
            }
            output.push((a << 2) | (b >> 4));
            self.terminal_padding = true;
        } else {
            let c = digit_value(group[2].0, self.options.alphabet).unwrap();
            if group[3].0 == b'=' {
                if self.options.padding == Padding::Forbidden {
                    return Err(Error::BadPadding { offset: group[3].1 });
                }
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                self.terminal_padding = true;
            } else {
                let d = digit_value(group[3].0, self.options.alphabet).unwrap();
                output.push((a << 2) | (b >> 4));
                output.push((b << 4) | (c >> 2));
                output.push((c << 6) | d);
            }
        }
        self.group.clear();
        Ok(())
    }

    fn process_symbol(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        if self.terminal_padding {
            return Err(Error::BadPadding { offset });
        }

        if self.group.len() == 3 && self.group[2].0 == b'=' && byte != b'=' {
            return Err(Error::BadPadding { offset });
        }

        let position = self.group.len();
        if byte == b'=' {
            if position < 2 {
                return Err(Error::BadPadding { offset });
            }
            if position == 2 {
                self.group.push((byte, offset));
                return Ok(());
            }
            if self.group[2].0 != b'=' {
                if self.options.padding == Padding::Forbidden {
                    return Err(Error::BadPadding { offset });
                }
                if digit_value(self.group[2].0, self.options.alphabet).is_none() {
                    return Err(Error::InvalidByte {
                        offset: self.group[2].1,
                        byte: self.group[2].0,
                    });
                }
                self.group.push((byte, offset));
                return self.decode_complete_group(output);
            }
            if self.options.padding == Padding::Forbidden {
                return Err(Error::BadPadding { offset });
            }
            self.group.push((byte, offset));
            return self.decode_complete_group(output);
        }

        let value = digit_value(byte, self.options.alphabet)
            .ok_or(Error::InvalidByte { offset, byte })?;
        let _ = value;
        self.group.push((byte, offset));
        if self.group.len() == 4 {
            self.decode_complete_group(output)?;
        }
        Ok(())
    }

    fn process_wrapped_byte(
        &mut self,
        byte: u8,
        offset: usize,
        output: &mut Vec<u8>,
    ) -> Result<(), Error> {
        let (width, ending) = match self.options.whitespace {
            Whitespace::Wrapped { width, ending } => (width, ending),
            _ => unreachable!(),
        };

        if self.wrap_stage != 0 {
            match (self.wrap_stage, ending, byte) {
                (1, LineEnding::Lf, b'\n') => {
                    self.wrap_stage = 0;
                    self.after_separator = true;
                    return Ok(());
                }
                (1, LineEnding::CrLf, b'\n') => {
                    self.wrap_stage = 0;
                    self.after_separator = true;
                    return Ok(());
                }
                (1, LineEnding::CrLf, _) => return Err(Error::BadWrap { offset }),
                _ => return Err(Error::BadWrap { offset }),
            }
        }

        if self.row_symbols == width {
            match ending {
                LineEnding::Lf if byte == b'\n' => {
                    self.wrap_start = offset;
                    self.wrap_stage = 1;
                    return Ok(());
                }
                LineEnding::CrLf if byte == b'\r' => {
                    self.wrap_start = offset;
                    self.wrap_stage = 1;
                    return Ok(());
                }
                _ => return Err(Error::BadWrap { offset }),
            }
        }

        if byte == b'\r' || byte == b'\n' {
            return Err(Error::BadWrap { offset });
        }

        self.after_separator = false;
        self.row_symbols += 1;
        self.process_symbol(byte, offset, output)
    }

    pub fn push(&mut self, input: &[u8]) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(self.record_error(Error::AlreadyFinished));
        }

        let mut output = Vec::new();
        for &byte in input {
            let offset = self.offset;
            self.offset += 1;

            let result = match self.options.whitespace {
                Whitespace::None => self.process_symbol(byte, offset, &mut output),
                Whitespace::IgnoreAscii => {
                    if matches!(byte, 9 | 10 | 11 | 12 | 13 | 32) {
                        Ok(())
                    } else {
                        self.process_symbol(byte, offset, &mut output)
                    }
                }
                Whitespace::Wrapped { .. } => {
                    self.process_wrapped_byte(byte, offset, &mut output)
                }
            };
            if let Err(error) = result {
                return Err(self.record_error(error));
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, Error> {
        if let Some(error) = &self.sticky_error {
            return Err(error.clone());
        }
        if self.closed {
            return Err(self.record_error(Error::AlreadyFinished));
        }

        if let Whitespace::Wrapped { .. } = self.options.whitespace {
            if self.wrap_stage != 0 {
                return Err(self.record_error(Error::BadWrap {
                    offset: self.wrap_start,
                }));
            }
            if self.after_separator {
                return Err(self.record_error(Error::BadWrap {
                    offset: self.wrap_start,
                }));
            }
        }

        if self.group.len() == 3 && self.group[2].0 == b'=' {
            return Err(self.record_error(Error::BadPadding {
                offset: self.group[2].1,
            }));
        }

        let mut output = Vec::new();
        if !self.group.is_empty() {
            match self.group.len() {
                1 => {
                    return Err(self.record_error(Error::Truncated {
                        offset: self.group[0].1,
                    }));
                }
                2 | 3 if self.options.padding != Padding::Required => {
                    let a = digit_value(self.group[0].0, self.options.alphabet).unwrap();
                    let b = digit_value(self.group[1].0, self.options.alphabet).unwrap();
                    output.push((a << 2) | (b >> 4));
                    if self.group.len() == 3 {
                        let c = digit_value(self.group[2].0, self.options.alphabet).unwrap();
                        output.push((b << 4) | (c >> 2));
                    }
                }
                _ => {
                    return Err(self.record_error(Error::Truncated {
                        offset: self.group[0].1,
                    }));
                }
            }
        }

        self.group.clear();
        self.closed = true;
        Ok(output)
    }
}

fn main() {}
```
