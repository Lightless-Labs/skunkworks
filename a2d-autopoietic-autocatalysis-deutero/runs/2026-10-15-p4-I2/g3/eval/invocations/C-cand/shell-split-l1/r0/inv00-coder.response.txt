#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.bytes().position(|byte| byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    #[derive(Clone, Copy)]
    enum State {
        Outside,
        Single { opening: usize },
        Double { opening: usize },
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut state = State::Outside;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match state {
            State::Outside => match ch {
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                '\'' => {
                    active = true;
                    state = State::Single { opening: pos };
                }
                '"' => {
                    active = true;
                    state = State::Double { opening: pos };
                }
                '\\' => match chars.next() {
                    None => return Err(SplitError::TrailingEscape { pos }),
                    Some((_, '\n')) => {}
                    Some((_, escaped)) => {
                        active = true;
                        word.push(escaped);
                    }
                },
                other => {
                    active = true;
                    word.push(other);
                }
            },
            State::Single { opening } => {
                if ch == '\'' {
                    state = State::Outside;
                } else {
                    let _ = opening;
                    word.push(ch);
                }
            }
            State::Double { opening } => match ch {
                '"' => state = State::Outside,
                '\\' => match chars.next() {
                    None => return Err(SplitError::TrailingEscape { pos }),
                    Some((_, next @ ('"' | '\\' | '$'))) => word.push(next),
                    Some((_, '\n')) => {}
                    Some((next_pos, next)) => {
                        word.push('\\');
                        word.push(next);
                        let _ = next_pos;
                    }
                },
                other => {
                    let _ = opening;
                    word.push(other);
                }
            },
        }
    }

    match state {
        State::Outside => {}
        State::Single { opening } => {
            return Err(SplitError::UnterminatedSingle { pos: opening });
        }
        State::Double { opening } => {
            return Err(SplitError::UnterminatedDouble { pos: opening });
        }
    }

    if active {
        words.push(word);
    }

    Ok(words)
}

fn main() {}
