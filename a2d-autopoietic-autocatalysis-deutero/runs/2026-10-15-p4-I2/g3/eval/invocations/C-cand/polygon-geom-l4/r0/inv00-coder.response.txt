#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
    Duplicate { first: usize, second: usize },
    ZeroArea,
    AdjacentOverlap { vertex: usize },
    Intersection { edge_a: usize, edge_b: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Validated {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub measures: Measures,
}

const COORD_LIMIT: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective.first() == effective.last() {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn dot_from_vertex(prev: Point, current: Point, next: Point) -> i128 {
    (prev.x as i128 - current.x as i128) * (next.x as i128 - current.x as i128)
        + (prev.y as i128 - current.y as i128) * (next.y as i128 - current.y as i128)
}

fn signed_area2(points: &[Point]) -> i128 {
    let n = points.len();
    let mut sum = 0i128;
    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn isqrt(n: u128) -> u128 {
    if n < 2 {
        return n;
    }
    let mut x = n;
    let mut y = (x + 1) / 2;
    while y < x {
        x = y;
        y = (x + n / x) / 2;
    }
    x
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let d2 = (dx * dx + dy * dy) as u128;
    let scaled = d2 * 1_000_000;
    let root = isqrt(scaled);
    let rem = scaled - root * root;
    let rounded = if rem >= root + 1 { root + 1 } else { root };
    rounded as u64
}

fn measures_for(points: &[Point]) -> Measures {
    let signed = signed_area2(points);
    let orientation = if signed > 0 {
        Orientation::CounterClockwise
    } else if signed < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    let n = points.len();
    let mut edge_milli = Vec::with_capacity(n);
    let mut perimeter_milli = 0u64;
    let mut longest_edge = None;
    let mut longest_pair: Option<(Point, Point)> = None;

    for i in 0..n {
        let a = points[i];
        let b = points[(i + 1) % n];
        let length = rounded_edge_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let low = a.min(b);
        let high = a.max(b);
        let pair = (low, high);
        let replace = match longest_edge {
            None => true,
            Some(current) => {
                length > edge_milli[current]
                    || (length == edge_milli[current]
                        && (pair < longest_pair.unwrap()
                            || (pair == longest_pair.unwrap() && i < current)))
            }
        };
        if replace {
            longest_edge = Some(i);
            longest_pair = Some(pair);
        }
    }

    Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let effective = prepare_polygon(points)?;
    Ok(measures_for(&effective))
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;

    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let n = polygon.len();
    let mut results = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        for i in 0..n {
            if on_segment(polygon[i], polygon[(i + 1) % n], q) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            results.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..n {
            let a = polygon[i];
            let b = polygon[(i + 1) % n];
            if (a.y > q.y) != (b.y > q.y) {
                let side = cross(a, b, q);
                if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                    inside = !inside;
                }
            }
        }
        results.push(if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(results)
}

fn monotone_chain(points: &[(Point, usize)], include_collinear: bool) -> Vec<(Point, usize)> {
    if points.len() <= 1 {
        return points.to_vec();
    }

    let all_collinear = points
        .iter()
        .skip(2)
        .all(|&(p, _)| cross(points[0].0, points[1].0, p) == 0);

    if all_collinear {
        if include_collinear {
            return points.to_vec();
        }
        return vec![points[0], *points.last().unwrap()];
    }

    let mut lower: Vec<(Point, usize)> = Vec::new();
    for &item in points {
        while lower.len() >= 2 {
            let c = cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, item.0);
            if if include_collinear { c < 0 } else { c <= 0 } {
                lower.pop();
            } else {
                break;
            }
        }
        lower.push(item);
    }

    let mut upper: Vec<(Point, usize)> = Vec::new();
    for &item in points.iter().rev() {
        while upper.len() >= 2 {
            let c = cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, item.0);
            if if include_collinear { c < 0 } else { c <= 0 } {
                upper.pop();
            } else {
                break;
            }
        }
        upper.push(item);
    }

    lower.pop();
    upper.pop();
    lower.extend(upper);
    lower
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut first_indices = std::collections::BTreeMap::<Point, usize>::new();
    for (index, &point) in points.iter().enumerate() {
        first_indices.entry(point).or_insert(index);
    }
    let unique: Vec<(Point, usize)> = first_indices.into_iter().collect();
    let hull = monotone_chain(&unique, mode == Collinear::Include);
    let vertices: Vec<Point> = hull.iter().map(|&(p, _)| p).collect();
    let source_indices: Vec<usize> = hull.iter().map(|&(_, i)| i).collect();

    let area = if vertices.len() >= 3 {
        signed_area2(&vertices).unsigned_abs()
    } else {
        0
    };
    let perimeter = if vertices.len() <= 1 {
        0
    } else {
        let mut total = 0u64;
        for i in 0..vertices.len() {
            total += rounded_edge_milli(vertices[i], vertices[(i + 1) % vertices.len()]);
        }
        total
    };

    Ok(Hull {
        vertices,
        source_indices,
        area2: area,
        perimeter_milli: perimeter,
    })
}

fn segments_intersect(a: Point, b: Point, c: Point, d: Point) -> bool {
    let ab_c = cross(a, b, c);
    let ab_d = cross(a, b, d);
    let cd_a = cross(c, d, a);
    let cd_b = cross(c, d, b);

    if ab_c == 0 && on_segment(a, b, c) {
        return true;
    }
    if ab_d == 0 && on_segment(a, b, d) {
        return true;
    }
    if cd_a == 0 && on_segment(c, d, a) {
        return true;
    }
    if cd_b == 0 && on_segment(c, d, b) {
        return true;
    }
    (ab_c > 0) != (ab_d > 0) && (cd_a > 0) != (cd_b > 0)
}

pub fn validate(points: &[Point]) -> Result<Validated, GeomError> {
    let polygon = prepare_polygon(points)?;
    let n = polygon.len();

    let mut best_duplicate = None;
    for i in 0..n {
        for j in i + 1..n {
            if polygon[i] == polygon[j] {
                best_duplicate = Some((i, j));
                break;
            }
        }
        if best_duplicate.is_some() {
            break;
        }
    }
    if let Some((first, second)) = best_duplicate {
        return Err(GeomError::Duplicate { first, second });
    }

    let area = signed_area2(&polygon);
    if area == 0 {
        return Err(GeomError::ZeroArea);
    }

    for i in 0..n {
        let prev = polygon[(i + n - 1) % n];
        let current = polygon[i];
        let next = polygon[(i + 1) % n];
        if cross(prev, current, next) == 0 && dot_from_vertex(prev, current, next) > 0 {
            return Err(GeomError::AdjacentOverlap { vertex: i });
        }
    }

    for i in 0..n {
        for j in (i + 1)..n {
            if j == i + 1 || (i == 0 && j == n - 1) {
                continue;
            }
            if segments_intersect(
                polygon[i],
                polygon[(i + 1) % n],
                polygon[j],
                polygon[(j + 1) % n],
            ) {
                return Err(GeomError::Intersection {
                    edge_a: i,
                    edge_b: j,
                });
            }
        }
    }

    let mut min_index = 0;
    for i in 1..n {
        if polygon[i] < polygon[min_index] {
            min_index = i;
        }
    }

    let mut order = Vec::with_capacity(n);
    if area > 0 {
        for offset in 0..n {
            order.push((min_index + offset) % n);
        }
    } else {
        for offset in 0..n {
            order.push((min_index + n - offset) % n);
        }
    }

    let vertices: Vec<Point> = order.iter().map(|&i| polygon[i]).collect();
    let source_indices = order;
    let measures = measures_for(&vertices);

    Ok(Validated {
        vertices,
        source_indices,
        measures,
    })
}
