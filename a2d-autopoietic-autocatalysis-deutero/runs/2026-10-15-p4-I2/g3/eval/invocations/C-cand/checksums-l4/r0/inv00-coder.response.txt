#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }

    Ok(())
}

fn crc32_step(mut register: u32, byte: u8) -> u32 {
    register ^= byte as u32;
    for _ in 0..8 {
        register = (register >> 1)
            ^ if register & 1 != 0 { 0xedb88320 } else { 0 };
    }
    register
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut register = 0xffff_ffff;
    for &byte in data {
        register = crc32_step(register, byte);
    }
    register ^ 0xffff_ffff
}

fn adler32_update(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *a = (*a + byte as u32) % 65_521;
        *b = (*b + *a) % 65_521;
    }
}

pub fn adler32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (1, 0);
    adler32_update(&mut a, &mut b, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn crc32_raw_spans(data: &[u8], spans: &[Span], mut register: u32) -> u32 {
    for span in spans {
        for &byte in &data[span.start..span.end] {
            register = crc32_step(register, byte);
        }
    }
    register
}

fn adler32_spans(data: &[u8], spans: &[Span], mut a: u32, mut b: u32) -> (u32, u32) {
    for span in spans {
        adler32_update(&mut a, &mut b, &data[span.start..span.end]);
    }
    (a, b)
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data.len(), spans)?;
    let crc = crc32_raw_spans(data, spans, 0xffff_ffff) ^ 0xffff_ffff;
    let (a, b) = adler32_spans(data, spans, 1, 0);
    Ok(Hashes {
        crc32: crc,
        adler32: (b << 16) | a,
    })
}

#[derive(Debug, Clone)]
pub struct Stream {
    crc_raw: u32,
    adler_a: u32,
    adler_b: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {
    Shape,
    Hex { offset: usize },
    Adler { component: usize },
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_raw: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_raw ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_raw: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        self.crc_raw = crc32_raw_spans(
            data,
            &[Span {
                start: 0,
                end: data.len(),
            }],
            self.crc_raw,
        );
        adler32_update(&mut self.adler_a, &mut self.adler_b, data);
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        let crc_raw = crc32_raw_spans(data, spans, self.crc_raw);
        let (a, b) = adler32_spans(data, spans, self.adler_a, self.adler_b);
        self.crc_raw = crc_raw;
        self.adler_a = a;
        self.adler_b = b;
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut crc = 0u32;
        for offset in 1..9 {
            crc = (crc << 4)
                | hex_value(bytes[offset]).ok_or(SnapshotError::Hex { offset })? as u32;
        }
        let mut adler = 0u32;
        for offset in 11..19 {
            adler = (adler << 4)
                | hex_value(bytes[offset]).ok_or(SnapshotError::Hex { offset })? as u32;
        }

        Self::from_seed(crc, adler)
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let (mut a, mut b) = (0u32, 0u32);
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

fn fletcher32_word(a: &mut u32, b: &mut u32, word: u16) {
    *a = (*a + word as u32) % 65_535;
    *b = (*b + *a) % 65_535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (0u32, 0u32);
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        fletcher32_word(&mut a, &mut b, u16::from_be_bytes([pair[0], pair[1]]));
    }
    if let Some(&last) = chunks.remainder().first() {
        fletcher32_word(&mut a, &mut b, (last as u16) << 8);
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let (mut a, mut b) = (0u32, 0u32);
    let mut pending_high: Option<u8> = None;

    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending_high.take() {
                fletcher32_word(&mut a, &mut b, u16::from_be_bytes([high, byte]));
            } else {
                pending_high = Some(byte);
            }
        }
    }

    if let Some(high) = pending_high {
        fletcher32_word(&mut a, &mut b, (high as u16) << 8);
    }

    Ok((b << 16) | a)
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {
    Xmodem,
    False,
    X25,
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            let mut register: u16 = if variant == Crc16::Xmodem {
                0
            } else {
                0xffff
            };
            for &byte in data {
                register ^= (byte as u16) << 8;
                for _ in 0..8 {
                    register = register.wrapping_shl(1)
                        ^ if register & 0x8000 != 0 { 0x1021 } else { 0 };
                }
            }
            register
        }
        Crc16::X25 => {
            let mut register: u16 = 0xffff;
            for &byte in data {
                register ^= byte as u16;
                for _ in 0..8 {
                    register = (register >> 1)
                        ^ if register & 1 != 0 { 0x8408 } else { 0 };
                }
            }
            register ^ 0xffff
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Multi {
    pub base: Hashes,
    pub fletcher16: u16,
    pub fletcher32: u32,
    pub crc16: u16,
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut crc32_raw = 0xffff_ffff;
    let (mut adler_a, mut adler_b) = (1u32, 0u32);
    let (mut f16_a, mut f16_b) = (0u32, 0u32);
    let (mut f32_a, mut f32_b) = (0u32, 0u32);
    let mut f32_pending: Option<u8> = None;
    let mut crc16_register: u16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        for &byte in &data[span.start..span.end] {
            crc32_raw = crc32_step(crc32_raw, byte);
            adler32_update(&mut adler_a, &mut adler_b, &[byte]);

            f16_a = (f16_a + byte as u32) % 255;
            f16_b = (f16_b + f16_a) % 255;

            if let Some(high) = f32_pending.take() {
                fletcher32_word(
                    &mut f32_a,
                    &mut f32_b,
                    u16::from_be_bytes([high, byte]),
                );
            } else {
                f32_pending = Some(byte);
            }

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    crc16_register ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        crc16_register = crc16_register.wrapping_shl(1)
                            ^ if crc16_register & 0x8000 != 0 {
                                0x1021
                            } else {
                                0
                            };
                    }
                }
                Crc16::X25 => {
                    crc16_register ^= byte as u16;
                    for _ in 0..8 {
                        crc16_register = (crc16_register >> 1)
                            ^ if crc16_register & 1 != 0 { 0x8408 } else { 0 };
                    }
                }
            }
        }
    }

    if let Some(high) = f32_pending {
        fletcher32_word(&mut f32_a, &mut f32_b, (high as u16) << 8);
    }

    let crc16_result = if variant == Crc16::X25 {
        crc16_register ^ 0xffff
    } else {
        crc16_register
    };

    Ok(Multi {
        base: Hashes {
            crc32: crc32_raw ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: ((f16_b << 8) | f16_a) as u16,
        fletcher32: (f32_b << 16) | f32_a,
        crc16: crc16_result,
    })
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    let mut odd = [0u32; 32];
    odd[0] = 0xedb88320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }

    let mut even = [0u32; 32];
    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut length = right_len;
    let mut crc = left;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if length & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }

        gf2_matrix_square(&mut odd, &even);
        if length & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        length >>= 1;
        if length == 0 {
            break;
        }
    }

    crc ^ right
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Part {
    pub crc32: u32,
    pub len: u64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CombineError {
    LengthOverflow { part: usize },
    EmptyCrc { part: usize },
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total_len = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total_len = total_len
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }

    Ok(Part {
        crc32: crc,
        len: total_len,
    })
}

fn selected_hashes_and_len(data: &[u8], spans: &[Span]) -> Result<(Hashes, u64), SpanError> {
    validate_spans(data.len(), spans)?;

    let mut total = 0u64;
    let mut crc_raw = 0xffff_ffff;
    let (mut a, mut b) = (1u32, 0u32);

    for span in spans {
        total += (span.end - span.start) as u64;
        let selected = &data[span.start..span.end];
        for &byte in selected {
            crc_raw = crc32_step(crc_raw, byte);
        }
        adler32_update(&mut a, &mut b, selected);
    }

    Ok((
        Hashes {
            crc32: crc_raw ^ 0xffff_ffff,
            adler32: (b << 16) | a,
        },
        total,
    ))
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    let (hashes, len) = selected_hashes_and_len(data, spans)?;
    Ok(format!(
        "len={len:016x};crc={:08x};adler={:08x}",
        hashes.crc32, hashes.adler32
    ))
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VerifyError {
    Span(SpanError),
    Shape,
    Hex { offset: usize },
    Length,
    Mismatch {
        field: usize,
        expected: u32,
        actual: u32,
    },
}

fn parse_hex_u64(bytes: &[u8], start: usize, end: usize) -> Result<u64, VerifyError> {
    let mut value = 0u64;
    for offset in start..end {
        let digit = hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
        value = (value << 4) | digit as u64;
    }
    Ok(value)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let len = parse_hex_u64(bytes, 4, 20)?;
    let crc = parse_hex_u64(bytes, 25, 33)? as u32;
    let adler = parse_hex_u64(bytes, 40, 48)? as u32;

    let (actual, actual_len) =
        selected_hashes_and_len(data, spans).map_err(VerifyError::Span)?;
    if len != actual_len {
        return Err(VerifyError::Length);
    }
    if crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: crc,
            actual: actual.crc32,
        });
    }
    if adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
