#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

fn encode_under_thousand(n: u16) -> String {
    if n < 20 {
        return SMALL[n as usize].to_owned();
    }

    if n < 100 {
        let tens = TENS[(n / 10 - 2) as usize];
        let units = n % 10;
        if units == 0 {
            tens.to_owned()
        } else {
            format!("{}-{}", tens, SMALL[units as usize])
        }
    } else {
        let hundreds = format!("{} hundred", SMALL[(n / 100) as usize]);
        let remainder = n % 100;
        if remainder == 0 {
            hundreds
        } else {
            format!("{} and {}", hundreds, encode_under_thousand(remainder))
        }
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_owned());
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in &mut groups {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let mut parts = Vec::new();
    for index in (0..7).rev() {
        let group = groups[index];
        if group == 0 {
            continue;
        }

        let mut part = encode_under_thousand(group);
        if index > 0 {
            part.push(' ');
            part.push_str(SCALES[index - 1]);
        }
        parts.push((index, group, part));
    }

    let units_needs_and = groups[0] > 0
        && groups[0] < 100
        && parts.iter().any(|(index, _, _)| *index > 0);

    let mut output = String::new();
    if value < 0 {
        output.push_str("minus ");
    }
    for (position, (index, _, part)) in parts.iter().enumerate() {
        if position > 0 {
            output.push(' ');
            if *index == 0 && units_needs_and {
                output.push_str("and ");
            }
        }
        output.push_str(part);
    }
    Ok(output)
}

#[derive(Clone, Copy)]
enum Word {
    Minus,
    Zero,
    Value(u128),
    Hundred,
    Scale(u128),
    And,
}

fn lookup_word(word: &str) -> Option<Word> {
    if word == "minus" {
        return Some(Word::Minus);
    }
    if word == "zero" {
        return Some(Word::Zero);
    }
    if word == "hundred" {
        return Some(Word::Hundred);
    }
    if word == "and" {
        return Some(Word::And);
    }

    for (value, spelling) in SMALL.iter().enumerate().skip(1) {
        if word == *spelling {
            return Some(Word::Value(value as u128));
        }
    }
    for (index, spelling) in TENS.iter().enumerate() {
        if word == *spelling {
            return Some(Word::Value(((index + 2) * 10) as u128));
        }
    }
    for (index, spelling) in SCALES.iter().enumerate() {
        if word == *spelling {
            return Some(Word::Scale(1000u128.pow((index + 1) as u32)));
        }
    }
    None
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_lowercase() && byte != b' ' && byte != b'-' {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    let mut words = Vec::with_capacity(tokens.len());
    for &(token, index) in &tokens {
        match lookup_word(token) {
            Some(Word::Minus) if index != 0 => {
                return Err(WordsError::InvalidWord { index });
            }
            Some(word) => words.push(word),
            None => return Err(WordsError::InvalidWord { index }),
        }
    }

    let negative = matches!(words.first(), Some(Word::Minus));
    let arithmetic_start = usize::from(negative);
    let mut total = 0u128;
    let mut group = 0u128;

    for word in words.iter().skip(arithmetic_start) {
        match *word {
            Word::Minus => unreachable!(),
            Word::Zero | Word::And => {}
            Word::Value(value) => group += value,
            Word::Hundred => group *= 100,
            Word::Scale(scale) => {
                total += group * scale;
                group = 0;
            }
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(value)?;
    if canonical.as_bytes() != input.as_bytes() {
        let differing_index = canonical
            .as_bytes()
            .iter()
            .zip(input.as_bytes())
            .position(|(left, right)| left != right)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical {
            index: differing_index,
        });
    }

    Ok(value)
}

fn main() {}
