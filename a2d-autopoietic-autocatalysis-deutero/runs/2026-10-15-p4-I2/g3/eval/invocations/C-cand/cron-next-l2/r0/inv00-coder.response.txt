#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_is_exact_star: bool,
    day_of_week_is_exact_star: bool,
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_bounds(index: usize) -> (u8, u8) {
    match index {
        0 => (0, 59),
        1 => (0, 23),
        2 => (1, 31),
        3 => (1, 12),
        _ => (0, 6),
    }
}

fn make_parse_error(kind: u8, field: Field, offset: usize) -> ParseError {
    match kind {
        0 => ParseError::InvalidCharacter { field, offset },
        1 => ParseError::EmptyItem { field, offset },
        2 => ParseError::InvalidSyntax { field, offset },
        3 => ParseError::LeadingZero { field, offset },
        4 => ParseError::NumberOverflow { field, offset },
        5 => ParseError::OutOfRange { field, offset },
        6 => ParseError::ReversedRange { field, offset },
        _ => ParseError::BadStep { field, offset },
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }
    Ok(value)
}

fn parse_cron_field(
    text: &str,
    index: usize,
    field: Field,
) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (minimum, maximum) = field_bounds(index);
    let mut selected = vec![false; maximum as usize + 1];
    let mut item_start = 0usize;

    loop {
        let mut item_end = item_start;
        while item_end < bytes.len() && bytes[item_end] != b',' {
            item_end += 1;
        }

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let mut slash = None;
        let mut hyphen = None;
        for pos in item_start..item_end {
            match bytes[pos] {
                b'/' => {
                    if slash.is_some() {
                        return Err(ParseError::InvalidSyntax {
                            field,
                            offset: item_start,
                        });
                    }
                    slash = Some(pos);
                }
                b'-' => {
                    if hyphen.is_some() {
                        return Err(ParseError::InvalidSyntax {
                            field,
                            offset: item_start,
                        });
                    }
                    hyphen = Some(pos);
                }
                _ => {}
            }
        }

        let base_end = slash.unwrap_or(item_end);
        let base_start = item_start;
        let base_len = base_end - base_start;
        let has_star = base_len == 1 && bytes[base_start] == b'*';

        let (start_num, end_num) = if has_star {
            if hyphen.is_some() {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            (None, None)
        } else if let Some(hyphen_pos) = hyphen {
            if hyphen_pos < base_start
                || hyphen_pos >= base_end
                || hyphen_pos == base_start
                || hyphen_pos + 1 == base_end
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            let valid_left = bytes[base_start..hyphen_pos]
                .iter()
                .all(u8::is_ascii_digit);
            let valid_right = bytes[hyphen_pos + 1..base_end]
                .iter()
                .all(u8::is_ascii_digit);
            if !valid_left || !valid_right {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            (Some((base_start, hyphen_pos)), Some((hyphen_pos + 1, base_end)))
        } else {
            if base_len == 0
                || !bytes[base_start..base_end]
                    .iter()
                    .all(u8::is_ascii_digit)
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            (Some((base_start, base_end)), None)
        };

        if let Some(step_pos) = slash {
            if step_pos + 1 == item_end {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            if !bytes[step_pos + 1..item_end]
                .iter()
                .all(u8::is_ascii_digit)
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }

        let (base_start_value, base_end_value) = if let Some((s, e)) = start_num {
            let start_value = parse_number(bytes, s, e, field)?;
            if start_value < minimum as u32 || start_value > maximum as u32 {
                return Err(ParseError::OutOfRange { field, offset: s });
            }
            let end_value = if let Some((es, ee)) = end_num {
                let value = parse_number(bytes, es, ee, field)?;
                if value < minimum as u32 || value > maximum as u32 {
                    return Err(ParseError::OutOfRange { field, offset: es });
                }
                if start_value > value {
                    return Err(ParseError::ReversedRange { field, offset: es });
                }
                value
            } else if slash.is_some() {
                maximum as u32
            } else {
                start_value
            };
            (start_value, end_value)
        } else {
            (minimum as u32, maximum as u32)
        };

        let step = if let Some(step_pos) = slash {
            let value = parse_number(bytes, step_pos + 1, item_end, field)?;
            let width = (maximum - minimum + 1) as u32;
            if value == 0 || value > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_pos + 1,
                });
            }
            value
        } else {
            1
        };

        let mut value = base_start_value;
        while value <= base_end_value {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) if next > value => value = next,
                _ => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = [
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
    ];
    for index in 0..5 {
        selected[index] = parse_cron_field(pieces[index], index, fields[index])?;
    }

    Ok(Cron {
        selected,
        day_of_month_is_exact_star: pieces[2].as_bytes() == b"*",
        day_of_week_is_exact_star: pieces[4].as_bytes() == b"*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        let (minimum, maximum) = field_bounds(index);
        (minimum..=maximum)
            .filter(|&value| self.selected[index][value as usize])
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.selected[0][at.minute as usize]
            && self.selected[1][at.hour as usize]
            && self.selected[2][at.day as usize]
            && self.selected[3][at.month as usize]
            && self.selected[4][weekday as usize])
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if !increment_minute(&mut current) {
                return Ok(None);
            }
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap_year(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn increment_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return true;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return true;
    }
    at.month = 1;
    if at.year < 2399 {
        at.year += 1;
        return true;
    }
    false
}
