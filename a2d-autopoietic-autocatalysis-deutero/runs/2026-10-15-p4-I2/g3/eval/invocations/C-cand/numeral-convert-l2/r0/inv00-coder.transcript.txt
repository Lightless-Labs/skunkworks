#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

struct BigNat {
    // Little-endian base-1_000_000_000 limbs.
    limbs: Vec<u32>,
}

impl BigNat {
    fn zero() -> Self {
        Self { limbs: vec![0] }
    }

    fn multiply_add(&mut self, multiplier: u32, addend: u32) {
        let mut carry = addend as u64;

        for limb in &mut self.limbs {
            let value = (*limb as u64) * (multiplier as u64) + carry;
            *limb = (value % 1_000_000_000) as u32;
            carry = value / 1_000_000_000;
        }

        while carry != 0 {
            self.limbs.push((carry % 1_000_000_000) as u32);
            carry /= 1_000_000_000;
        }
    }

    fn add_assign(&mut self, other: &Self) {
        let max_len = self.limbs.len().max(other.limbs.len());
        self.limbs.resize(max_len, 0);

        let mut carry = 0u64;
        for i in 0..max_len {
            let rhs = other.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = self.limbs[i] as u64 + rhs + carry;
            self.limbs[i] = (sum % 1_000_000_000) as u32;
            carry = sum / 1_000_000_000;
        }

        if carry != 0 {
            self.limbs.push(carry as u32);
        }
    }

    fn divide_small(&mut self, divisor: u32) -> u32 {
        let mut remainder = 0u64;

        for limb in self.limbs.iter_mut().rev() {
            let value = remainder * 1_000_000_000 + *limb as u64;
            *limb = (value / divisor as u64) as u32;
            remainder = value % divisor as u64;
        }

        while self.limbs.len() > 1 && self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }

        remainder as u32
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn to_digits(&self, base: u8, upper: bool) -> Vec<u8> {
        let mut value = Self {
            limbs: self.limbs.clone(),
        };
        let mut digits = Vec::new();

        if value.is_zero() {
            digits.push(b'0');
            return digits;
        }

        while !value.is_zero() {
            let digit = value.divide_small(base as u32) as u8;
            let byte = if digit < 10 {
                b'0' + digit
            } else if upper {
                b'A' + (digit - 10)
            } else {
                b'a' + (digit - 10)
            };
            digits.push(byte);
        }

        digits.reverse();
        digits
    }
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<BigNat, BaseError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let mut payload_start = 0;
    if let Some(hash) = bytes.iter().position(|&byte| byte == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = hash + 1;
    }

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digit_count = 0usize;
    let mut first_digit = None;

    // Check all digits before checking separator placement.
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }

        let value = match byte {
            b'0'..=b'9' => byte - b'0',
            b'A'..=b'Z' => byte - b'A' + 10,
            b'a'..=b'z' => byte - b'a' + 10,
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        };

        if value >= radix {
            return Err(BaseError::InvalidDigit {
                operand,
                index: payload_start + offset,
            });
        }

        if first_digit.is_none() {
            first_digit = Some((value, payload_start + offset));
        }
        digit_count += 1;
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let left_is_digit = offset > 0 && payload[offset - 1] != b'_';
            let right_is_digit = offset + 1 < payload.len() && payload[offset + 1] != b'_';

            if !left_is_digit || !right_is_digit {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    if digit_count >= 2 && first_digit.map(|(value, _)| value) == Some(0) {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    let mut result = BigNat::zero();
    for &byte in payload {
        if byte == b'_' {
            continue;
        }

        let value = match byte {
            b'0'..=b'9' => byte - b'0',
            b'A'..=b'Z' => byte - b'A' + 10,
            b'a'..=b'z' => byte - b'a' + 10,
            _ => unreachable!(),
        };
        result.multiply_add(radix as u32, value as u32);
    }

    Ok(result)
}

fn render(mut digits: Vec<u8>, style: BaseStyle) -> String {
    if digits.len() < style.min_width {
        let padding = style.min_width - digits.len();
        let mut padded = Vec::with_capacity(style.min_width);
        padded.resize(padding, b'0');
        padded.extend_from_slice(&digits);
        digits = padded;
    }

    if style.group == 0 || digits.len() <= style.group {
        return String::from_utf8(digits).unwrap();
    }

    let separators = (digits.len() - 1) / style.group;
    let mut output = Vec::with_capacity(digits.len() + separators);

    for (index, &digit) in digits.iter().enumerate() {
        if index > 0 && (digits.len() - index) % style.group == 0 {
            output.push(b'_');
        }
        output.push(digit);
    }

    String::from_utf8(output).unwrap()
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;

    let value = parse_operand(input, from, 0)?;
    Ok(render(value.to_digits(to, style.upper), style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;

    let mut left_value = parse_operand(left, base, 0)?;
    let right_value = parse_operand(right, base, 1)?;
    left_value.add_assign(&right_value);

    Ok(render(left_value.to_digits(base, style.upper), style))
}

fn main() {}
