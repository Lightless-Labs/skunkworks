```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-10 digits. Zero is represented as [0] and is never negative.
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' || bytes[0] == b'-' {
            1
        } else {
            0
        };

        if start == bytes.len() && start == 1 {
            return Err(Error::SignOnly);
        }

        let negative = start == 1 && bytes[0] == b'-';

        // Invalid bytes take precedence over malformed separators.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        // Check separators only after the full invalid-byte scan succeeds.
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index });
            }
        }

        let mut first_digit_index = None;
        let mut digit_count = 0usize;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte.is_ascii_digit() {
                if first_digit_index.is_none() {
                    first_digit_index = Some(index);
                }
                digit_count += 1;
            }
        }

        let first_digit_index = first_digit_index.expect("nonempty magnitude has a digit");
        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            for (index, &byte) in bytes.iter().enumerate().skip(start) {
                if byte.is_ascii_digit() {
                    seen += 1;
                    if seen == excess_number {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index,
                        });
                    }
                }
            }
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes.iter().skip(start) {
            if byte.is_ascii_digit() {
                digits.push(byte - b'0');
            }
        }
        digits.reverse();

        let mut result = Self { negative, digits };
        result.normalize();
        Ok(result)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude_len = self.digits.len().max(1);
        let width = magnitude_len.max(min_digits);
        let padding = width - magnitude_len;

        let separators = if group > 0 { (width - 1) / group } else { 0 };
        let mut output = String::with_capacity(
            width.saturating_add(separators)
                .saturating_add(usize::from(self.negative)),
        );

        if self.negative {
            output.push('-');
        }

        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                output.push('_');
            }

            if position < padding {
                output.push('0');
            } else {
                let digit_index = width - 1 - position;
                let digit = self.digits.get(digit_index).copied().unwrap_or(0);
                output.push(char::from(b'0' + digit));
            }
        }

        output
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => compare_magnitude(&self.digits, &rhs.digits),
            (true, true) => compare_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self {
                negative: self.negative,
                digits: add_magnitude(&self.digits, &rhs.digits),
            };
            result.normalize();
            result
        } else {
            match compare_magnitude(&self.digits, &rhs.digits) {
                Ordering::Greater => {
                    let mut result = Self {
                        negative: self.negative,
                        digits: subtract_magnitude(&self.digits, &rhs.digits),
                    };
                    result.normalize();
                    result
                }
                Ordering::Less => {
                    let mut result = Self {
                        negative: rhs.negative,
                        digits: subtract_magnitude(&rhs.digits, &self.digits),
                    };
                    result.normalize();
                    result
                }
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            Self::zero()
        } else {
            Self {
                negative: !self.negative,
                digits: self.digits.clone(),
            }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut result = Self {
            negative: self.negative != rhs.negative,
            digits: multiply_magnitude(&self.digits, &rhs.digits),
        };
        result.normalize();
        result
    }

    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn normalize(&mut self) {
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.is_empty() {
            self.digits.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }
}

fn compare_magnitude(lhs: &[u8], rhs: &[u8]) -> Ordering {
    match lhs.len().cmp(&rhs.len()) {
        Ordering::Equal => {
            for (&a, &b) in lhs.iter().rev().zip(rhs.iter().rev()) {
                match a.cmp(&b) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_magnitude(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(lhs.len().max(rhs.len()).saturating_add(1));
    let mut carry = 0u8;

    for index in 0..lhs.len().max(rhs.len()) {
        let a = lhs.get(index).copied().unwrap_or(0);
        let b = rhs.get(index).copied().unwrap_or(0);
        let sum = a + b + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }

    if carry != 0 {
        result.push(carry);
    }
    result
}

// Requires lhs >= rhs.
fn subtract_magnitude(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(lhs.len());
    let mut borrow = 0i8;

    for (index, &a) in lhs.iter().enumerate() {
        let mut digit = i16::from(a) - i16::from(borrow as u8);
        let b = i16::from(rhs.get(index).copied().unwrap_or(0));
        if digit < b {
            digit += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push((digit - b) as u8);
    }

    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

fn multiply_magnitude(lhs: &[u8], rhs: &[u8]) -> Vec<u8> {
    let mut result = vec![0u8; lhs.len().saturating_add(rhs.len())];

    for (i, &a) in lhs.iter().enumerate() {
        let mut carry = 0u16;
        for (j, &b) in rhs.iter().enumerate() {
            let index = i + j;
            let value = u16::from(result[index]) + u16::from(a) * u16::from(b) + carry;
            result[index] = (value % 10) as u8;
            carry = value / 10;
        }

        let mut index = i + rhs.len();
        while carry > 0 {
            if index == result.len() {
                result.push(0);
            }
            let value = u16::from(result[index]) + carry;
            result[index] = (value % 10) as u8;
            carry = value / 10;
            index += 1;
        }
    }

    while result.len() > 1 && result.last() == Some(&0) {
        result.pop();
    }
    result
}

fn main() {}
```
