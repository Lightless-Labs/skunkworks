use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_stop(b: Option<u8>) -> bool {
    matches!(b, None | Some(b'.' | b'-' | b'+'))
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if start == *pos {
        return Err(if is_core_stop(bytes.get(*pos).copied()) {
            VersionError::MissingCore { at: *pos }
        } else {
            VersionError::InvalidChar { at: *pos }
        });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut n = 0u64;
    for &b in &bytes[start..*pos] {
        n = n.checked_mul(10)
            .and_then(|x| x.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(n)
}

fn valid_identifier_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut id_start = start;
    loop {
        let mut i = id_start;
        while i < end && bytes[i] != b'.' {
            if !valid_identifier_byte(bytes[i]) {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        if prerelease {
            let numeric = bytes[id_start..i].iter().all(u8::is_ascii_digit);
            if numeric && i - id_start > 1 && bytes[id_start] == b'0' {
                return Err(VersionError::LeadingZero { at: id_start });
            }
        }
        result.push(s[id_start..i].to_string());
        if i == end {
            break;
        }
        id_start = i + 1;
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for _ in 0..2 {
        if s.as_bytes().get(pos) != Some(&b'.') {
            return Err(if is_core_stop(s.as_bytes().get(pos).copied()) {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
        pos += 1;
        let n = parse_core(s, &mut pos)?;
        if pos == 0 {
            unreachable!();
        }
        if _ == 0 {
            // Store the second field for parsing the final field.
            let minor = n;
            if s.as_bytes().get(pos) == Some(&b'.') {
                pos += 1;
                let patch = parse_core(s, &mut pos)?;
                return parse_version_tail(s, pos, major, minor, patch);
            }
            return Err(if is_core_stop(s.as_bytes().get(pos).copied()) {
                VersionError::MissingCore { at: pos }
            } else {
                VersionError::InvalidChar { at: pos }
            });
        }
    }
    unreachable!()
}

fn parse_version_tail(
    s: &str,
    pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if pos == bytes.len() {
        return Ok(Version { major, minor, patch, pre: vec![], build: vec![] });
    }
    if bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }
    let mut pre = Vec::new();
    let mut build = Vec::new();
    let mut cursor = pos;
    if bytes[cursor] == b'-' {
        cursor += 1;
        let plus = bytes[cursor..].iter().position(|&b| b == b'+').map(|x| cursor + x);
        let end = plus.unwrap_or(bytes.len());
        pre = parse_identifiers(s, cursor, end, true)?;
        cursor = end;
    }
    if cursor < bytes.len() && bytes[cursor] == b'+' {
        cursor += 1;
        build = parse_identifiers(s, cursor, bytes.len(), false)?;
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}
fn cmp_numeric_str(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}
pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ord = match (numeric_id(x), numeric_id(y)) {
                        (true, true) => cmp_numeric_str(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone)]
struct Token { text: String, at: usize }
fn lex(s: &str) -> Vec<Token> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && spaces(b[i]) { i += 1; }
        if i == b.len() { break; }
        let start = i;
        while i < b.len() && !spaces(b[i]) { i += 1; }
        out.push(Token { text: s[start..i].to_string(), at: start });
    }
    out
}
fn syntax(at: usize) -> RangeError { RangeError::Syntax { at } }
fn wrap_version(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}
fn next_tuple(t: (u64,u64,u64), field: usize) -> Option<(u64,u64,u64)> {
    match field {
        0 => Some((t.0.checked_add(1)?,0,0)),
        1 => Some((t.0,t.1.checked_add(1)?,0)),
        _ => Some((t.0,t.1,t.2.checked_add(1)?)),
    }
}

#[derive(Clone)]
struct Partial { v: Version, k: usize, full: bool }
fn parse_partial(raw: &str, at: usize) -> Result<Partial, RangeError> {
    if raw.contains(['-', '+']) {
        return Ok(Partial { v: wrap_version(raw, at)?, k: 3, full: true });
    }
    let fields: Vec<&str> = raw.split('.').collect();
    if fields.len() >= 3 && !fields.iter().any(|x| matches!(*x, "*"|"x"|"X")) {
        return Ok(Partial { v: wrap_version(raw, at)?, k: 3, full: true });
    }
    if fields.is_empty() || fields.len() > 3 { return Err(syntax(at)); }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if matches!(f, "*"|"x"|"X") { wildcard_seen = true; continue; }
        if wildcard_seen || f.is_empty() || !f.bytes().all(|b| b.is_ascii_digit()) ||
            (f.len() > 1 && f.starts_with('0')) {
            return Err(syntax(at));
        }
        let n = f.parse::<u64>().map_err(|_| syntax(at))?;
        nums.push(n);
    }
    let k = nums.len();
    let v = Version {
        major: *nums.first().unwrap_or(&0),
        minor: *nums.get(1).unwrap_or(&0),
        patch: *nums.get(2).unwrap_or(&0),
        pre: vec![], build: vec![],
    };
    Ok(Partial { v, k, full:false })
}

fn add_cmp(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() { arm.anchors.push(tuple(&v)); }
    arm.comps.push(Comparator { op, version:v });
}
fn add_expanded(arm: &mut Arm, op: Op, raw: &str, at: usize, kind: u8) -> Result<(),RangeError> {
    let p = parse_partial(raw, at)?;
    if kind != 0 && p.k == 0 { return Err(syntax(at)); }
    if kind == 0 {
        if p.full { add_cmp(arm, Op::Eq, p.v, true); }
        else if p.k == 1 || p.k == 2 {
            add_cmp(arm, Op::Ge, p.v.clone(), true);
            let field = if p.k == 1 {0} else {1};
            let upper = next_tuple(tuple(&p.v),field).ok_or(syntax(at))?;
            add_cmp(arm, Op::Lt, version_at(upper,vec!["0".into()]), false);
        }
        return Ok(());
    }
    let lower = p.v.clone();
    add_cmp(arm, Op::Ge, lower.clone(), true);
    let t = tuple(&lower);
    let field = if kind == 1 {
        if lower.major > 0 || p.k == 1 { 0 } else if lower.minor > 0 || p.k == 2 { 1 } else { 2 }
    } else if p.k == 1 { 0 } else { 1 };
    let upper = next_tuple(t,field).ok_or(syntax(at))?;
    add_cmp(arm, Op::Lt, version_at(upper,vec!["0".into()]),false);
    let _ = op;
    Ok(())
}

fn parse_arm(tokens: &[Token], s: &str) -> Result<Arm, RangeError> {
    if tokens.is_empty() { return Err(RangeError::Empty { at:s.len() }); }
    let mut arm = Arm { comps:vec![], anchors:vec![] };
    let hyphens: Vec<usize> = tokens.iter().enumerate().filter(|(_,t)| t.text=="-").map(|(i,_)|i).collect();
    if !hyphens.is_empty() {
        let h = hyphens[0];
        if tokens.len()!=3 || h!=1 { return Err(syntax(tokens[h].at)); }
        let lo = parse_partial(&tokens[0].text,tokens[0].at)?;
        let hi = parse_partial(&tokens[2].text,tokens[2].at)?;
        if lo.k==0 { return Err(syntax(tokens[0].at)); }
        if hi.k==0 { return Err(syntax(tokens[2].at)); }
        add_cmp(&mut arm,Op::Ge,lo.v.clone(),true);
        if hi.full { add_cmp(&mut arm,Op::Le,hi.v,true); }
        else {
            let field=if hi.k==1 {0}else{1};
            let up=next_tuple(tuple(&hi.v),field).ok_or(syntax(tokens[2].at))?;
            add_cmp(&mut arm,Op::Lt,version_at(up,vec!["0".into()]),false);
        }
        return Ok(arm);
    }
    for t in tokens {
        let raw=t.text.as_str();
        if raw=="||" { return Err(syntax(t.at)); }
        if raw.starts_with('^') || raw.starts_with('~') {
            if raw.len()==1 { return Err(syntax(t.at)); }
            add_expanded(&mut arm,Op::Ge,&raw[1..],t.at+1,if raw.starts_with('^'){1}else{2})?;
            continue;
        }
        let mut prefix=0;
        let mut op=None;
        for candidate in ["<=",">=","=","<",">"] {
            if raw.starts_with(candidate) { prefix=candidate.len(); op=Some(match candidate {"<="=>Op::Le,">="=>Op::Ge,"<"=>Op::Lt,">"=>Op::Gt,_=>Op::Eq}); break; }
        }
        if let Some(op)=op {
            let operand=&raw[prefix..];
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() { return Err(syntax(t.at)); }
            let v=wrap_version(operand,t.at+prefix)?;
            add_cmp(&mut arm,op,v,true);
            continue;
        }
        let b=raw.as_bytes()[0];
        if !(b.is_ascii_digit() || matches!(b,b'*'|b'x'|b'X')) { return Err(syntax(t.at)); }
        add_expanded(&mut arm,Op::Eq,raw,t.at,0)?;
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens=lex(s);
    if tokens.is_empty() { return Err(RangeError::Empty { at:0 }); }
    let mut arms=Vec::new();
    let mut current=Vec::new();
    let mut i=0;
    while i<tokens.len() {
        if tokens[i].text=="||" {
            if current.is_empty() { return Err(RangeError::Empty { at:tokens[i].at }); }
            arms.push(parse_arm(&current,s)?);
            current.clear();
        } else { current.push(tokens[i].clone()); }
        i+=1;
    }
    if current.is_empty() {
        return Err(RangeError::Empty { at:s.len() });
    }
    arms.push(parse_arm(&current,s)?);
    Ok(Range{arms})
}

fn cmp_one(v:&Version,c:&Comparator)->bool {
    match (c.op,compare_versions(v,&c.version)) {
        (Op::Eq,Ordering::Equal)|(Op::Lt,Ordering::Less)|(Op::Le,Ordering::Less|Ordering::Equal)|
        (Op::Gt,Ordering::Greater)|(Op::Ge,Ordering::Greater|Ordering::Equal)=>true,
        _=>false
    }
}
fn arm_match(v:&Version,a:&Arm,include:bool)->bool {
    if !v.pre.is_empty() && !include && !a.anchors.contains(&tuple(v)) { return false; }
    a.comps.iter().all(|c|cmp_one(v,c))
}
pub fn matches_with(v:&Version,r:&Range,include_prerelease:bool)->bool {
    r.arms.iter().any(|a|arm_match(v,a,include_prerelease))
}
pub fn matches(v:&Version,r:&Range)->bool { matches_with(v,r,false) }

#[derive(Clone,Debug,PartialEq,Eq)]
pub struct CandidateError { pub index:usize, pub error:VersionError }
fn extremum(candidates:&[String],r:&Range,max:bool)->Result<Option<String>,CandidateError> {
    let parsed:Vec<Version>=candidates.iter().enumerate().map(|(i,s)|parse_version(s).map_err(|error|CandidateError{index:i,error})).collect::<Result<_,_>>()?;
    let mut best:Option<usize>=None;
    for (i,v) in parsed.iter().enumerate() {
        if !matches(v,r) { continue; }
        if let Some(j)=best {
            let ord=compare_versions(v,&parsed[j]);
            if (max && ord==Ordering::Greater)||(!max&&ord==Ordering::Less) { best=Some(i); }
        } else { best=Some(i); }
    }
    Ok(best.map(|i|candidates[i].clone()))
}
pub fn max_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extremum(candidates,r,true)}
pub fn min_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extremum(candidates,r,false)}

pub fn intersects(_a:&Range,_b:&Range,_include_prerelease:bool)->bool { false }

fn display_version(v:&Version)->String {
    let mut s=format!("{}.{}.{}",v.major,v.minor,v.patch);
    if !v.pre.is_empty(){s.push('-');s.push_str(&v.pre.join("."));}
    s
}
pub fn canonical(r:&Range)->String {
    let mut out=Vec::new();
    for a in &r.arms {
        let mut lower:Option<(Version,bool)>=None;
        let mut upper:Option<(Version,bool)>=None;
        let mut eq:Option<Version>=None;
        for c in &a.comps {
            match c.op {
                Op::Eq=>eq=Some(c.version.clone()),
                Op::Ge|Op::Gt=>{
                    let strict=c.op==Op::Gt;
                    if lower.as_ref().map_or(true,|(v,incl)|compare_versions(&c.version,v)==Ordering::Greater || (compare_versions(&c.version,v)==Ordering::Equal && strict && *incl)) {lower=Some((c.version.clone(),!strict));}
                }
                Op::Le|Op::Lt=>{
                    let strict=c.op==Op::Lt;
                    if upper.as_ref().map_or(true,|(v,incl)|compare_versions(&c.version,v)==Ordering::Less || (compare_versions(&c.version,v)==Ordering::Equal && strict && *incl)) {upper=Some((c.version.clone(),!strict));}
                }
            }
        }
        if let Some(v)=eq { lower=Some((v.clone(),true)); upper=Some((v,true)); }
        let lo=lower.map(|(v,i)|format!("{}{}",if i{">="}else{">"},display_version(&v))).unwrap_or("*".into());
        let hi=upper.map(|(v,i)|format!("{}{}",if i{"<="}else{"<"},display_version(&v))).unwrap_or("*".into());
        let mut anchors=a.anchors.clone(); anchors.sort();
        anchors.dedup();
        let anc=if anchors.is_empty(){"-".into()}else{anchors.iter().map(|t|format!("{}.{}.{}",t.0,t.1,t.2)).collect::<Vec<_>>().join(",")};
        out.push(format!("[{lo};{hi};pre={anc}]"));
    }
    out.sort();out.dedup();
    if out.is_empty(){"EMPTY".into()}else{out.join(" || ")}
}

pub enum SelectError {
    DuplicatePackage { name:String },
    DuplicateLabel { package:String,label:String },
    InvalidCandidate { package:String,index:usize,error:VersionError },

#[derive(Clone)]
enum CheckedExpr { Range(Range), Union(Vec<CheckedExpr>), Intersection(Vec<CheckedExpr>) }
fn validate_expr(e:&SetExpr,p:&str,l:&str,path:&mut Vec<usize>)->Result<CheckedExpr,SelectError>{
    match e {
        SetExpr::Range(s)=>parse_range(s).map(CheckedExpr::Range).map_err(|error|SelectError::InvalidRange{package:p.into(),label:l.into(),path:path.clone(),error}),
        SetExpr::Union(xs)=>{let mut ys=Vec::new();for(i,x)in xs.iter().enumerate(){path.push(i);ys.push(validate_expr(x,p,l,path)?);path.pop();}Ok(CheckedExpr::Union(ys))}
        SetExpr::Intersection(xs)=>{let mut ys=Vec::new();for(i,x)in xs.iter().enumerate(){path.push(i);ys.push(validate_expr(x,p,l,path)?);path.pop();}Ok(CheckedExpr::Intersection(ys))}
    }
}
fn expr_match(v:&Version,e:&CheckedExpr,inc:bool)->bool{match e{CheckedExpr::Range(r)=>matches_with(v,r,inc),CheckedExpr::Union(xs)=>xs.iter().any(|x|expr_match(v,x,inc)),CheckedExpr::Intersection(xs)=>xs.iter().all(|x|expr_match(v,x,inc))}}
fn subsets_conflict(indices:&[usize], candidates:&[Version], exprs:&[CheckedExpr],inc:bool)->bool{
    candidates.iter().all(|v|indices.iter().all(|&i|expr_match(v,&exprs[i],inc))==false)
}
fn choose_conflict(candidates:&[Version], exprs:&[CheckedExpr], labels:&[String],inc:bool)->Vec<String>{
    let n=exprs.len();
    for size in 1..=n {
        let mut found:Vec<Vec<String>>=Vec::new();
        fn rec(start:usize,left:usize,cur:&mut Vec<usize>,n:usize,candidates:&[Version],exprs:&[CheckedExpr],labels:&[String],inc:bool,found:&mut Vec<Vec<String>>){
            if left==0 {
                if candidates.iter().all(|v|cur.iter().all(|&i|expr_match(v,&exprs[i],inc))==false){
                    let mut ls:Vec<String>=cur.iter().map(|&i|labels[i].clone()).collect();ls.sort();found.push(ls);
                } return;
            }
            for i in start..=n-left {cur.push(i);rec(i+1,left-1,cur,n,candidates,exprs,labels,inc,found);cur.pop();}
        }
        rec(0,size,&mut Vec::new(),n,candidates,exprs,labels,inc,&mut found);
        if !found.is_empty(){found.sort();return found.remove(0);}
    }
    Vec::new()
}
pub fn select_packages(packages:&[Package],preference:Preference,include_prerelease:bool)->Result<Vec<Selection>,SelectError>{
    let mut names=std::collections::HashSet::new();
    for p in packages {if !names.insert(p.name.clone()){return Err(SelectError::DuplicatePackage{name:p.name.clone()});}}
    for p in packages {
        let mut labels=std::collections::HashSet::new();
        for c in &p.constraints {if !labels.insert(c.label.clone()){return Err(SelectError::DuplicateLabel{package:p.name.clone(),label:c.label.clone()});}}
    }
    let mut parsed=Vec::new();
    for p in packages {
        let mut vs=Vec::new();
        for(i,s)in p.candidates.iter().enumerate(){vs.push(parse_version(s).map_err(|error|SelectError::InvalidCandidate{package:p.name.clone(),index:i,error})?);}
        parsed.push(vs);
    }
    let mut checked=Vec::new();
    for p in packages {
        let mut es=Vec::new();
        for c in &p.constraints {es.push(validate_expr(&c.expr,&p.name,&c.label,&mut Vec::new())?);}
        checked.push(es);
    }
    let mut result=Vec::new();
    for (pi,p) in packages.iter().enumerate() {
        let es=&checked[pi];let vs=&parsed[pi];
        let good:Vec<usize>=(0..vs.len()).filter(|&i|es.iter().all(|e|expr_match(&vs[i],e,include_prerelease))).collect();
        if good.is_empty(){
            let labels=if vs.is_empty(){vec![]}else{choose_conflict(vs,es,&p.constraints.iter().map(|c|c.label.clone()).collect::<Vec<_>>(),include_prerelease)};
            return Err(SelectError::NoSolution{package:p.name.clone(),labels});
        }
        let stable_exists=matches!(preference,Preference::StableFirst)&&good.iter().any(|&i|vs[i].pre.is_empty());
        let mut best=good[0];
        for &i in &good[1..] {
            let eligible=if stable_exists{vs[i].pre.is_empty()}else{true};
            let current_eligible=if stable_exists{vs[best].pre.is_empty()}else{true};
            if eligible && (!current_eligible || compare_versions(&vs[i],&vs[best])==Ordering::Greater && !matches!(preference,Preference::Lowest)) ||
                (matches!(preference,Preference::Lowest) && compare_versions(&vs[i],&vs[best])==Ordering::Less) {best=i;}
        }
        result.push(Selection{package:p.name.clone(),version:p.candidates[best].clone()});
    }
    Ok(result)
}
