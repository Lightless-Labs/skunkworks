#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Timing {
    pub start: u64,
    pub finish: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Plan {
    pub order: Vec<usize>,
    pub times: Vec<Timing>,
    pub makespan: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Overflow {
        task: usize,
    },
}

pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let n = tasks.len();

    // Validate every dependency before doing any graph computation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    // Build predecessor counts and child lists.
    let mut indegree: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut children = vec![Vec::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest available task ID selected each time.
    let mut available = vec![false; n];
    for task_id in 0..n {
        if indegree[task_id] == 0 {
            available[task_id] = true;
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut removed = vec![false; n];
    while let Some(task_id) = (0..n).find(|&id| available[id]) {
        available[task_id] = false;
        removed[task_id] = true;
        order.push(task_id);

        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available[child] = true;
            }
        }
    }

    if order.len() != n {
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute earliest timings in Kahn order.
    let mut times: Vec<Option<Timing>> = vec![None; n];
    let mut makespan = 0;

    for &task_id in &order {
        let mut start = 0;
        for &predecessor in &tasks[task_id].deps {
            let finish = times[predecessor]
                .as_ref()
                .expect("predecessor must precede task")
                .finish;
            start = start.max(finish);
        }

        let finish = start
            .checked_add(tasks[task_id].duration)
            .ok_or(Error::Overflow { task: task_id })?;

        times[task_id] = Some(Timing { start, finish });
        makespan = makespan.max(finish);
    }

    let times = times
        .into_iter()
        .map(|timing| timing.expect("all tasks are scheduled"))
        .collect();

    Ok(Plan {
        order,
        times,
        makespan,
    })
}

fn main() {}
