use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_missing_byte(b: u8) -> bool {
    b == b'.' || b == b'-' || b == b'+'
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }

    if *pos == start {
        if start == bytes.len() || is_core_missing_byte(bytes[start]) {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }

    let digits = &bytes[start..*pos];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value = 0u64;
    for &digit in digits {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((digit - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }

    Ok(value)
}

fn parse_identifiers(
    s: &str,
    pos: &mut usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();

    loop {
        let start = *pos;
        while *pos < end && bytes[*pos] != b'.' {
            let b = bytes[*pos];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: *pos });
            }
            *pos += 1;
        }

        if *pos == start {
            return Err(VersionError::EmptyIdentifier { at: start });
        }

        if is_pre {
            let ident = &bytes[start..*pos];
            if ident.len() > 1
                && ident[0] == b'0'
                && ident.iter().all(u8::is_ascii_digit)
            {
                return Err(VersionError::LeadingZero { at: start });
            }
        }

        result.push(s[start..*pos].to_string());

        if *pos == end {
            break;
        }

        // The current byte is a dot; the next iteration reports an empty
        // identifier if this dot is trailing or doubled.
        *pos += 1;
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0;

    let major = parse_core(s, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        if pos == bytes.len() || is_core_missing_byte(bytes[pos]) {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let minor = parse_core(s, &mut pos)?;
    if pos >= bytes.len() || bytes[pos] != b'.' {
        if pos == bytes.len() || is_core_missing_byte(bytes[pos]) {
            return Err(VersionError::MissingCore { at: pos });
        }
        return Err(VersionError::InvalidChar { at: pos });
    }
    pos += 1;

    let patch = parse_core(s, &mut pos)?;
    if pos < bytes.len() && bytes[pos] != b'-' && bytes[pos] != b'+' {
        return Err(VersionError::InvalidChar { at: pos });
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        let pre_end = bytes[pos..]
            .iter()
            .position(|&b| b == b'+')
            .map(|i| pos + i)
            .unwrap_or(bytes.len());

        pre = parse_identifiers(s, &mut pos, pre_end, true)?;

        if pos < bytes.len() && bytes[pos] == b'+' {
            pos += 1;
        }
    } else if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
    }

    if pos < bytes.len() {
        build = parse_identifiers(s, &mut pos, bytes.len(), false)?;
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let a_num = is_numeric_identifier(a);
    let b_num = is_numeric_identifier(b);

    match (a_num, b_num) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a
            .len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (left, right) in a.pre.iter().zip(&b.pre) {
                let ordering = compare_identifiers(left, right);
                if ordering != Ordering::Equal {
                    return ordering;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug)]
struct Comparator {
    op: CompareOp,
    version: Version,
}

#[derive(Clone, Debug)]
enum CompareOp {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len()
            && matches!(bytes[pos], b' ' | b'\t' | b'\r' | b'\n')
        {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        let mut op = CompareOp::Equal;

        match bytes[pos] {
            b'=' => {
                pos += 1;
            }
            b'<' => {
                pos += 1;
                if pos < bytes.len() && bytes[pos] == b'=' {
                    pos += 1;
                    op = CompareOp::LessEqual;
                } else {
                    op = CompareOp::Less;
                }
            }
            b'>' => {
                pos += 1;
                if pos < bytes.len() && bytes[pos] == b'=' {
                    pos += 1;
                    op = CompareOp::GreaterEqual;
                } else {
                    op = CompareOp::Greater;
                }
            }
            _ => {}
        }

        let operand_start = pos;
        if pos >= bytes.len() || !bytes[pos].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        while pos < bytes.len()
            && !matches!(bytes[pos], b' ' | b'\t' | b'\r' | b'\n')
        {
            pos += 1;
        }

        let operand = &s[operand_start..pos];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        comparators.push(Comparator { op, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        arms: vec![comparators],
    })
}

fn same_core(a: &Version, b: &Version) -> bool {
    a.major == b.major && a.minor == b.minor && a.patch == b.patch
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let prerelease_admitted = v.pre.is_empty()
            || arm.iter().any(|comparator| {
                !comparator.version.pre.is_empty()
                    && same_core(v, &comparator.version)
            });

        prerelease_admitted
            && arm.iter().all(|comparator| {
                let ordering = compare_versions(v, &comparator.version);
                match comparator.op {
                    CompareOp::Equal => ordering == Ordering::Equal,
                    CompareOp::Less => ordering == Ordering::Less,
                    CompareOp::LessEqual => ordering != Ordering::Greater,
                    CompareOp::Greater => ordering == Ordering::Greater,
                    CompareOp::GreaterEqual => ordering != Ordering::Less,
                }
            })
    })
}

fn main() {}
