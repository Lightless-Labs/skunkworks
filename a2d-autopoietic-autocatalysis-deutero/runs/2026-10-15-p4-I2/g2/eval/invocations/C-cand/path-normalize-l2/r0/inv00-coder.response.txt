#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.bytes().position(|b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.bytes().position(|b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .bytes()
        .position(|b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn push_component(components: &mut Vec<&str>, component: &str, absolute: bool, at: usize) -> Result<(), PathError> {
    match component {
        "" | "." => Ok(()),
        ".." => {
            if components.last().is_some_and(|last| *last != "..") {
                components.pop();
                Ok(())
            } else if absolute {
                Err(PathError::Escape { at })
            } else {
                components.push("..");
                Ok(())
            }
        }
        _ => {
            components.push(component);
            Ok(())
        }
    }
}

fn reduce(path: &str, absolute: bool, mut components: Vec<&str>, offset: usize) -> Result<String, PathError> {
    let mut cursor = 0;
    for component in path.split('/') {
        let start = cursor;
        cursor += component.len() + 1;
        if !component.is_empty() {
            push_component(&mut components, component, absolute, offset + start)?;
        }
    }

    let mut result = String::new();
    if absolute {
        result.push('/');
    }
    result.push_str(&components.join("/"));

    if result.is_empty() {
        result.push('.');
    }

    if directory_intent(path) && result != "/" && result != "." {
        result.push('/');
    }
    Ok(result)
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    reduce(path, absolute, Vec::new(), 0)
}

fn normalized_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(normalized
        .trim_matches('/')
        .split('/')
        .filter(|part| !part.is_empty() && *part != ".")
        .map(str::to_owned)
        .collect())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = normalized_context(cwd).map_err(|_| PathError::InvalidCwd)?;

    let home_components = match home {
        Some(home_path) => {
            if !home_path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(normalized_context(home_path).map_err(|_| PathError::InvalidHome)?)
        }
        None => None,
    };

    validate_raw(path)?;

    let (absolute, initial, request, offset) = if path.starts_with('/') {
        (true, Vec::new(), path, 0)
    } else if path.starts_with('~') {
        if path == "~" || path.starts_with("~/") {
            let initial = home_components.ok_or(PathError::MissingHome)?;
            let remainder = if path == "~" {
                ""
            } else {
                &path[2..]
            };
            (true, initial, remainder, if path == "~" { path.len() } else { 2 })
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else {
        (true, cwd_components, path, 0)
    };

    let request_intent = if path.starts_with('~') {
        path.starts_with("~/") || directory_intent(&path[1..])
    } else {
        directory_intent(path)
    };

    let mut result = String::from("/");
    result.push_str(&reduce(request, false, initial, offset)?
        .trim_start_matches('/')
        .trim_end_matches('/'));

    if result.len() == 1 {
        result = "/".to_owned();
    } else if request_intent {
        result.push('/');
    }
    let _ = absolute;
    Ok(result)
}

fn main() {}
