fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.bytes().position(|b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.bytes().position(|b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .bytes()
        .position(|b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn push_component(components: &mut Vec<&str>, component: &str, absolute: bool, at: usize) -> Result<(), PathError> {
    match component {
        "" | "." => Ok(()),
        ".." => {

                Ok(())

            Ok(())

}

fn reduce(path: &str, absolute: bool, mut components: Vec<&str>, offset: usize) -> Result<String, PathError> {
