fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> u8 {
    match c {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn constraint_cmp(a: &Constraint, b: &Constraint) -> std::cmp::Ordering {
    constraint_rank(*a).cmp(&constraint_rank(*b)).then_with(|| {
        let version = |c| match c {
            Constraint::Exact(v)
            | Constraint::AtLeast(v)
            | Constraint::Before(v)
            | Constraint::Caret(v) => v,
        };
        version(*a).cmp(&version(*b))
    })
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort_by(constraint_cmp);
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical_constraints(constraints.to_vec())
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn matches_all(constraints: &[Constraint], v: Version) -> bool {
    constraints.iter().all(|c| matches(*c, v))
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for &b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_cmp(a: &Evidence, b: &Evidence) -> std::cmp::Ordering {
    constraint_cmp(&a.constraint, &b.constraint).then_with(|| a.path.cmp(&b.path))
}

fn append_evidence(out: &mut String, evidence: &[Evidence]) {
    let mut sorted = evidence.to_vec();
    sorted.sort_by(evidence_cmp);
    for item in sorted {
        out.push_str(&format!(
            "\n  C:{} via {}",
            constraint_text(item.constraint),
            item.path.join(" -> ")
        ));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut out = format!("missing|package={}", encode(package));
            append_evidence(&mut out, evidence);
            out
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut out = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            append_evidence(&mut out, evidence);
            out
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut out = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            append_evidence(&mut out, evidence);
            out
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: Vec<Origin>,
    requested: std::collections::BTreeSet<String>,
    activated: std::collections::BTreeSet<String>,
}

fn platform_matches(platform: Platform, target: Target) -> bool {
    matches!(platform, Platform::Any)
        || matches!(
            (platform, target),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn sorted_features(features: &[String]) -> String {
    let mut features = features.to_vec();
    features.sort();
    features.dedup();
    features.join(",")
}

fn root_token(root: &Requirement) -> String {
    format!(
        "root:{}@{}+{{{}}}",
        root.package,
        constraint_text(root.constraint),
        sorted_features(&root.features)
    )
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        sorted_features(&dep.features),
        dep.gate.as_deref().unwrap_or("*"),
        match dep.platform {
            Platform::Any => "any",
            Platform::Linux => "linux",
            Platform::Mac => "mac",
            Platform::Windows => "windows",
        }
    )
}

fn best_path(paths: impl Iterator<Item = Vec<String>>) -> Option<Vec<String>> {
    paths.min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.cmp(b)))
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    attempts: usize,
    budget: usize,
    best: Option<(usize, Vec<Selection>, std::collections::BTreeMap<String, Active>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn package(&self, name: &str) -> Option<&'a Package> {
        self.catalog.iter().find(|p| p.name == name)
    }

    fn release(&self, package: &str, version: Version) -> Option<&'a Release> {
        self.package(package)?
            .releases
            .iter()
            .find(|r| r.version == version)
    }

    fn lock_version(&self, package: &str) -> Option<Version> {
        self.locks
            .iter()
            .find(|lock| lock.package == package)
            .map(|lock| lock.version)
    }

    fn active(
        &self,
        selected: &std::collections::BTreeMap<String, Version>,
    ) -> std::collections::BTreeMap<String, Active> {
        let mut active = std::collections::BTreeMap::<String, Active>::new();
        for root in self.roots {
            let entry = active.entry(root.package.clone()).or_default();
            entry.origins.push(Origin {
                constraint: root.constraint,
                path: vec![root_token(root)],
            });
            entry.requested.extend(root.features.iter().cloned());
        }

        loop {
            let mut changed = false;
            let names: Vec<String> = active.keys().cloned().collect();

            for name in names {
                let Some(&version) = selected.get(&name) else {
                    continue;
                };
                let Some(release) = self.release(&name, version) else {
                    continue;
                };

                let requested: Vec<String> =
                    active[&name].requested.iter().cloned().collect();
                for feature_name in requested {
                    if active
                        .get_mut(&name)
                        .unwrap()
                        .activated
                        .insert(feature_name.clone())
                    {
                        changed = true;
                    }
                    if let Some(feature) = release.features.iter().find(|f| f.name == feature_name)
                    {
                        for enabled in &feature.enables {
                            if active.get_mut(&name).unwrap().requested.insert(enabled.clone()) {
                                changed = true;
                            }
                        }
                    }
                }

                let activated = active[&name].activated.clone();
                for (index, dep) in release.deps.iter().enumerate() {
                    if !platform_matches(dep.platform, self.target)
                        || dep
                            .gate
                            .as_ref()
                            .is_some_and(|gate| !activated.contains(gate))
                    {
                        continue;
                    }

                    let Some(mut path) = best_path(
                        active[&name].origins.iter().map(|origin| origin.path.clone()),
                    ) else {
                        continue;
                    };
                    path.push(dep_token(&name, version, index, dep));

                    let entry = active.entry(dep.package.clone()).or_default();
                    if !entry
                        .origins
                        .iter()
                        .any(|origin| origin.constraint == dep.constraint && origin.path == path)
                    {
                        entry.origins.push(Origin {
                            constraint: dep.constraint,
                            path,
                        });
                        changed = true;
                    }
                    for feature in &dep.features {
                        if entry.requested.insert(feature.clone()) {
                            changed = true;
                        }
                    }
                }
            }

            if !changed {
                break;
            }
        }
        active
    }

    fn evidence(active: &Active) -> Vec<Evidence> {
        let mut evidence: Vec<Evidence> = active
            .origins
            .iter()
            .map(|origin| Evidence {
                constraint: origin.constraint,
                path: origin.path.clone(),
            })
            .collect();
        evidence.sort_by(evidence_cmp);
        evidence
    }

    fn constraints(active: &Active) -> Vec<Constraint> {
        canonical_constraints(active.origins.iter().map(|o| o.constraint).collect())
    }

    fn candidates(&self, name: &str, constraints: &[Constraint]) -> Vec<Version> {
        let Some(package) = self.package(name) else {
            return Vec::new();
        };
        let mut versions: Vec<Version> = package
            .releases
            .iter()
            .map(|release| release.version)
            .filter(|version| matches_all(constraints, *version))
            .collect();
        versions.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.lock_version(name) {
            if let Some(index) = versions.iter().position(|version| *version == lock) {
                let version = versions.remove(index);
                versions.insert(0, version);
            }
        }
        versions
    }

    fn inspect(
        &self,
        selected: &std::collections::BTreeMap<String, Version>,
    ) -> Result<
        (
            std::collections::BTreeMap<String, Active>,
            Option<String>,
            Vec<Version>,
        ),
        ResolveError,
    > {
        let active = self.active(selected);

        for name in active.keys() {
            if self.package(name).is_none() {
                return Err(ResolveError::Missing {
                    package: name.clone(),
                    evidence: Self::evidence(&active[name]),
                });
            }
        }

        for (name, version) in selected {
            if let Some(entry) = active.get(name) {
                let constraints = Self::constraints(entry);
                if !matches_all(&constraints, *version) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints,
                        evidence: Self::evidence(entry),
                    });
                }
            }
        }

        for (name, entry) in &active {
            if let Some(&version) = selected.get(name) {
                if let Some(release) = self.release(name, version) {
                    for feature in &entry.activated {
                        if !release.features.iter().any(|declared| declared.name == *feature) {
                            let path = best_path(
                                entry.origins.iter().map(|origin| origin.path.clone()),
                            )
                            .unwrap_or_default();
                            return Err(ResolveError::MissingFeature {
                                package: name.clone(),
                                selected: version,
                                feature: feature.clone(),
                                path,
                            });
                        }
                    }
                }
            }
        }

        for (name, entry) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let constraints = Self::constraints(entry);
            if self.candidates(name, &constraints).is_empty() {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints,
                    evidence: Self::evidence(entry),
                });
            }
        }

        let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
        let candidates = next
            .as_ref()
            .map(|name| self.candidates(name, &Self::constraints(&active[name])))
            .unwrap_or_default();
        Ok((active, next, candidates))
    }

    fn solution_cost(&self, selected: &std::collections::BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|lock| selected.get(&lock.package) != Some(&lock.version))
            .count()
    }

    fn solution_better(
        &self,
        cost: usize,
        selected: &std::collections::BTreeMap<String, Version>,
    ) -> bool {
        let Some((best_cost, best, _)) = &self.best else {
            return true;
        };
        if cost != *best_cost {
            return cost < *best_cost;
        }

        let mut names: Vec<&str> = self.catalog.iter().map(|p| p.name.as_str()).collect();
        names.sort();
        names.dedup();
        for name in names {
            match (
                selected.get(name),
                best.iter().find(|selection| selection.package == name),
            ) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(a), Some(b)) if *a != b.version => return *a > b.version,
                _ => {}
            }
        }
        false
    }

    fn visit(
        &mut self,
        selected: &mut std::collections::BTreeMap<String, Version>,
    ) -> Result<(), ResolveError> {
        let (active, next, candidates) = match self.inspect(selected) {
            Ok(value) => value,
            Err(error) => {
                if self.first_error.is_none() {
                    self.first_error = Some(error.clone());
                }
                return Err(error);
            }
        };

        let Some(package) = next else {
            let solution: Vec<Selection> = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            let cost = self.solution_cost(selected);
            if self.solution_better(cost, selected) {
                self.best = Some((cost, solution, active));
            }
            return Ok(());
        };

        for candidate in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            let result = self.visit(selected);
            selected.remove(&package);
            if let Err(ResolveError::Limit { attempts }) = result {
                return Err(ResolveError::Limit { attempts });
            }
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut package_names = std::collections::BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = std::collections::BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        target,
        attempts: 0,
        budget: max_attempts,
        best: None,
        first_error: None,
    };
    let mut selected = std::collections::BTreeMap::new();
    if let Err(ResolveError::Limit { attempts }) = search.visit(&mut selected) {
        return Err(ResolveError::Limit { attempts });
    }

    let Some((_, chosen, final_active)) = search.best.take() else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let after: std::collections::BTreeMap<String, Version> = chosen
        .iter()
        .map(|selection| (selection.package.clone(), selection.version))
        .collect();
    let before: std::collections::BTreeMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();
    let mut names = std::collections::BTreeSet::new();
    names.extend(before.keys().cloned());
    names.extend(after.keys().cloned());

    let mut changes = Vec::new();
    for package in names {
        let b = before.get(&package).copied();
        let a = after.get(&package).copied();
        if b == a {
            continue;
        }

        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(locked), Some(_)) => {
                let exists = catalog.iter().any(|p| {
                    p.name == package && p.releases.iter().any(|r| r.version == locked)
                });
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active
                    .get(&package)
                    .is_some_and(|active| !matches_all(&Search::constraints(active), locked))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution {
        selected: chosen,
        changes,
    })
}

fn main() {}
