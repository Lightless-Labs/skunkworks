fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> u8 {
    match c {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn constraint_cmp(a: &Constraint, b: &Constraint) -> std::cmp::Ordering {
    constraint_rank(*a)
        .cmp(&constraint_rank(*b))
        .then_with(|| match (*a, *b) {
            (Constraint::Exact(x), Constraint::Exact(y))
            | (Constraint::AtLeast(x), Constraint::AtLeast(y))
            | (Constraint::Before(x), Constraint::Before(y))
            | (Constraint::Caret(x), Constraint::Caret(y)) => x.cmp(&y),
            _ => std::cmp::Ordering::Equal,
        })
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort_by(constraint_cmp);
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut items = canonical_constraints(constraints.to_vec());
    items
        .drain(..)
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn matches_all(constraints: &[Constraint], v: Version) -> bool {
    constraints.iter().all(|c| matches(*c, v))
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || *b == b'.' || *b == b'_' || *b == b'-' {
            out.push(*b as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", b));
        }
    }
    out
}

fn path_text(path: &[String]) -> String {
    path.join(" -> ")
}

fn evidence_cmp(a: &Evidence, b: &Evidence) -> std::cmp::Ordering {
    constraint_cmp(&a.constraint, &b.constraint).then_with(|| a.path.cmp(&b.path))
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut out = format!("missing|package={}", encode(package));
            let mut sorted = evidence.clone();
            sorted.sort_by(evidence_cmp);
            for item in sorted {
                out.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(item.constraint),
                    path_text(&item.path)
                ));
            }
            out
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut out = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            let mut sorted = evidence.clone();
            sorted.sort_by(evidence_cmp);
            for item in sorted {
                out.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(item.constraint),
                    path_text(&item.path)
                ));
            }
            out
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut out = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            let mut sorted = evidence.clone();
            sorted.sort_by(evidence_cmp);
            for item in sorted {
                out.push_str(&format!(
                    "\n  C:{} via {}",
                    constraint_text(item.constraint),
                    path_text(&item.path)
                ));
            }
            out
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path_text(path)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: Vec<Origin>,
    requested: std::collections::BTreeSet<String>,
    activated: std::collections::BTreeSet<String>,
}

struct Search<'a> {
    catalog: &'a [Package],
    target: Target,
    locks: &'a [Lock],
    attempts: usize,
    budget: usize,
    best: Option<(usize, Vec<Selection>, std::collections::BTreeMap<String, Active>)>,
    first_error: Option<ResolveError>,
}

fn platform_matches(platform: Platform, target: Target) -> bool {
    matches!(platform, Platform::Any)
        || matches!(
            (platform, target),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn root_token(root: &Requirement) -> String {
    let mut features = root.features.clone();
    features.sort();
    features.dedup();
    format!(
        "root:{}@{}+{{{}}}",
        root.package,
        constraint_text(root.constraint),
        features.join(",")
    )
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let mut features = dep.features.clone();
    features.sort();
    features.dedup();
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        features.join(","),
        dep.gate.as_deref().unwrap_or("*"),
        match dep.platform {
            Platform::Any => "any",
            Platform::Linux => "linux",
            Platform::Mac => "mac",
            Platform::Windows => "windows",
        }
    )
}

fn better_path(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

impl<'a> Search<'a> {
    fn package(&self, name: &str) -> Option<&'a Package> {
        self.catalog.iter().find(|p| p.name == name)
    }

    fn release(&self, package: &str, version: Version) -> Option<&'a Release> {
        self.package(package)?
            .releases
            .iter()
            .find(|r| r.version == version)
    }

    fn lock_version(&self, package: &str) -> Option<Version> {
        self.locks
            .iter()
            .find(|l| l.package == package)
            .map(|l| l.version)
    }

    fn active(&self, roots: &[Requirement], selected: &std::collections::BTreeMap<String, Version>) -> std::collections::BTreeMap<String, Active> {
        let mut active = std::collections::BTreeMap::<String, Active>::new();
        for root in roots {
            let token = root_token(root);
            let entry = active.entry(root.package.clone()).or_default();
            entry.origins.push(Origin {
                constraint: root.constraint,
                path: vec![token],
            });
            entry.requested.extend(root.features.iter().cloned());
        }

        loop {
            let mut changed = false;

            let names: Vec<String> = active.keys().cloned().collect();
            for name in names {
                let Some(version) = selected.get(&name).copied() else {
                    continue;
                };
                let Some(release) = self.release(&name, version) else {
                    continue;
                };

                let requested: Vec<String> = active[&name].requested.iter().cloned().collect();
                for feature_name in requested {
                    if active.get_mut(&name).unwrap().activated.insert(feature_name.clone()) {
                        changed = true;
                    }
                    if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                        for enabled in &feature.enables {
                            if active.get_mut(&name).unwrap().requested.insert(enabled.clone()) {
                                changed = true;
                            }
                        }
                    }
                }

                let activated = active[&name].activated.clone();
                for (index, dep) in release.deps.iter().enumerate() {
                    if !platform_matches(dep.platform, self.target)
                        || dep.gate.as_ref().is_some_and(|g| !activated.contains(g))
                    {
                        continue;
                    }
                    let source_path = active[&name]
                        .origins
                        .iter()
                        .map(|o| o.path.clone())
                        .min_by(|a, b| {
                            a.len().cmp(&b.len()).then_with(|| a.cmp(b))
                        });
                    let Some(mut path) = source_path else {
                        continue;
                    };
                    path.push(dep_token(&name, version, index, dep));
                    let entry = active.entry(dep.package.clone()).or_default();
                    if !entry.origins.iter().any(|o| {
                        o.constraint == dep.constraint && o.path == path
                    }) {
                        entry.origins.push(Origin {
                            constraint: dep.constraint,
                            path,
                        });
                        changed = true;
                    }
                    for feature in &dep.features {
                        if entry.requested.insert(feature.clone()) {
                            changed = true;
                        }
                    }
                }
            }

            if !changed {
                break;
            }

            let mut best_paths = std::collections::BTreeMap::<String, Vec<String>>::new();
            for (name, entry) in &active {
                if let Some(path) = entry
                    .origins
                    .iter()
                    .map(|o| o.path.clone())
                    .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.cmp(b)))
                {
                    best_paths.insert(name.clone(), path);
                }
            }
            for (name, entry) in active.iter_mut() {
                if let Some(best) = best_paths.get(name) {
                    entry.origins.retain(|o| {
                        o.path.len() <= best.len()
                            || !o.path.starts_with(best.as_slice())
                    });
                }
            }
        }

        active
    }

    fn evidence(active: &Active) -> Vec<Evidence> {
        let mut evidence: Vec<Evidence> = active
            .origins
            .iter()
            .map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            })
            .collect();
        evidence.sort_by(evidence_cmp);
        evidence
    }

    fn constraints(active: &Active) -> Vec<Constraint> {
        canonical_constraints(active.origins.iter().map(|o| o.constraint).collect())
    }

    fn inspect(
        &mut self,
        roots: &[Requirement],
        selected: &std::collections::BTreeMap<String, Version>,
    ) -> Result<(std::collections::BTreeMap<String, Active>, Option<String>, Vec<Version>), ResolveError> {
        let active = self.active(roots, selected);

        for name in active.keys() {
            if self.package(name).is_none() {
                return Err(ResolveError::Missing {
                    package: name.clone(),
                    evidence: Self::evidence(&active[name]),
                });
            }
        }

        for (name, version) in selected {
            if let Some(a) = active.get(name) {
                let constraints = Self::constraints(a);
                if !matches_all(&constraints, *version) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints,
                        evidence: Self::evidence(a),
                    });
                }
            }
        }

        for (name, a) in &active {
            if selected.contains_key(name) {
                let Some(version) = selected.get(name).copied() else {
                    continue;
                };
                let Some(release) = self.release(name, version) else {
                    continue;
                };
                for feature in &a.activated {
                    if !release.features.iter().any(|f| f.name == *feature) {
                        let path = a
                            .origins
                            .iter()
                            .map(|o| o.path.clone())
                            .min_by(|x, y| {
                                x.len().cmp(&y.len()).then_with(|| x.cmp(y))
                            })
                            .unwrap_or_default();
                        return Err(ResolveError::MissingFeature {
                            package: name.clone(),
                            selected: version,
                            feature: feature.clone(),
                            path,
                        });
                    }
                }
            }
        }

        for (name, a) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let constraints = Self::constraints(a);
            let candidates = self.candidates(name, &constraints);
            if candidates.is_empty() {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints,
                    evidence: Self::evidence(a),
                });
            }
        }

        let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
        let candidates = if let Some(name) = &next {
            self.candidates(name, &Self::constraints(&active[name]))
        } else {
            Vec::new()
        };
        Ok((active, next, candidates))
    }

    fn candidates(&self, name: &str, constraints: &[Constraint]) -> Vec<Version> {
        let Some(package) = self.package(name) else {
            return Vec::new();
        };
        let mut versions: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| matches_all(constraints, *v))
            .collect();
        versions.sort_by(|a, b| b.cmp(a));
        versions.dedup();

        if let Some(lock) = self.lock_version(name) {
            if let Some(pos) = versions.iter().position(|v| *v == lock) {
                let v = versions.remove(pos);
                versions.insert(0, v);
            }
        }
        versions
    }

    fn solution_cost(&self, selected: &std::collections::BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|lock| selected.get(&lock.package) != Some(&lock.version))
            .count()
    }

    fn solution_better(
        &self,
        cost: usize,
        selected: &std::collections::BTreeMap<String, Version>,
    ) -> bool {
        let Some((best_cost, best, _)) = &self.best else {
            return true;
        };
        if cost != *best_cost {
            return cost < *best_cost;
        }
        let mut all_names: Vec<&str> = self.catalog.iter().map(|p| p.name.as_str()).collect();
        all_names.sort();
        all_names.dedup();
        for name in all_names {
            match (selected.get(name), best.iter().find(|s| s.package == name)) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(a), Some(b)) if *a != b.version => return *a > b.version,
                _ => {}
            }
        }
        false
    }

    fn visit(
        &mut self,
        roots: &[Requirement],
        selected: &mut std::collections::BTreeMap<String, Version>,
    ) -> Result<(), ResolveError> {
        let inspected = self.inspect(roots, selected);
        let (active, next, candidates) = match inspected {
            Ok(v) => v,
            Err(error) => {
                if !matches!(error, ResolveError::Limit { .. }) && self.first_error.is_none() {
                    self.first_error = Some(error.clone());
                }
                return Err(error);
            }
        };

        let Some(package) = next else {
            let mut solution: Vec<Selection> = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect();
            solution.sort_by(|a, b| a.package.cmp(&b.package));
            let cost = self.solution_cost(selected);
            if self.solution_better(cost, selected) {
                self.best = Some((cost, solution, active));
            }
            return Ok(());
        };

        for candidate in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            let result = self.visit(roots, selected);
            selected.remove(&package);
            if let Err(ResolveError::Limit { attempts }) = result {
                return Err(ResolveError::Limit { attempts });
            }
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = std::collections::BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        target,
        locks,
        attempts: 0,
        budget: max_attempts,
        best: None,
        first_error: None,
    };
    let mut selected = std::collections::BTreeMap::new();
    let result = search.visit(roots, &mut selected);

    if let Err(ResolveError::Limit { attempts }) = result {
        return Err(ResolveError::Limit { attempts });
    }

    let Some((_, chosen, final_active)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let after: std::collections::BTreeMap<String, Version> = chosen
        .iter()
        .map(|s| (s.package.clone(), s.version))
        .collect();
    let before: std::collections::BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();
    let mut names = std::collections::BTreeSet::new();
    names.extend(before.keys().cloned());
    names.extend(after.keys().cloned());

    let mut changes = Vec::new();
    for package in names {
        let b = before.get(&package).copied();
        let a = after.get(&package).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(locked), Some(_)) => {
                let exists = search
                    .package(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == locked));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active
                    .get(&package)
                    .is_some_and(|active| !matches_all(&Search::constraints(active), locked))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution {
        selected: chosen,
        changes,
    })
}

fn main() {}
