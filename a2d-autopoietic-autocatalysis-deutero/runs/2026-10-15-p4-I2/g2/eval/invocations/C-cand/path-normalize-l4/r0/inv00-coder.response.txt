#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.as_bytes().last() == Some(&b'/')
        || path
            .split('/')
            .rev()
            .find(|component| !component.is_empty())
            .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|component| !component.is_empty()).collect()
}

fn push_component(
    stack: &mut Vec<String>,
    component: &str,
    absolute: bool,
    at: usize,
) -> Result<(), PathError> {
    match component {
        "." => Ok(()),
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
                Ok(())
            } else if absolute {
                Err(PathError::Escape { at })
            } else {
                stack.push("..".to_string());
                Ok(())
            }
        }
        _ => {
            stack.push(component.to_string());
            Ok(())
        }
    }
}

fn render(stack: &[String], absolute: bool, intent: bool) -> String {
    if absolute {
        if stack.is_empty() {
            return "/".to_string();
        }
        let mut result = format!("/{}", stack.join("/"));
        if intent {
            result.push('/');
        }
        result
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        let mut result = stack.join("/");
        if intent {
            result.push('/');
        }
        result
    }
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();

    let mut offset = 0;
    for part in path.split('/') {
        if !part.is_empty() {
            push_component(&mut stack, part, absolute, offset)?;
        }
        offset += part.len() + 1;
    }

    Ok(render(&stack, absolute, intent))
}

fn normalized_context(path: &str, cwd: bool) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(if cwd {
            PathError::InvalidCwd
        } else {
            PathError::InvalidHome
        });
    }
    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized).into_iter().map(str::to_string).collect()),
        Err(_) => Err(if cwd {
            PathError::InvalidCwd
        } else {
            PathError::InvalidHome
        }),
    }
}

fn request_start<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(bool, Vec<String>, &'a str, usize), PathError> {
    if path.starts_with('~') {
        let rest = if path == "~" {
            ""
        } else if path.as_bytes().get(1) == Some(&b'/') {
            &path[2..]
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        };
        let home = home.ok_or(PathError::MissingHome)?;
        let normalized = normalize(home).map_err(|_| PathError::InvalidHome)?;
        let start = components(&normalized).into_iter().map(str::to_string).collect();
        let consumed = path.len() - rest.len();
        Ok((true, start, rest, consumed))
    } else {
        Ok((path.starts_with('/'), Vec::new(), path, 0))
    }
}

fn resolve_impl(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = normalized_context(cwd, true)?;
    let home_components = match home {
        Some(value) => Some(normalized_context(value, false)?),
        None => None,
    };

    validate_raw(path)?;

    let intent = directory_intent(path);
    if path.starts_with('~') {
        let (absolute, mut stack, rest, base_offset) =
            request_start(path, home).map_err(|error| match error {
                PathError::InvalidHome => PathError::InvalidHome,
                other => other,
            })?;
        let _ = absolute;
        let mut offset = base_offset;
        for part in rest.split('/') {
            if !part.is_empty() {
                push_component(&mut stack, part, true, offset)?;
            }
            offset += part.len() + 1;
        }
        return Ok(render(&stack, true, intent));
    }

    let absolute = path.starts_with('/');
    let mut stack = if absolute { Vec::new() } else { cwd_components };
    let mut offset = 0;
    for part in path.split('/') {
        if !part.is_empty() {
            push_component(&mut stack, part, true, offset)?;
        }
        offset += part.len() + 1;
    }
    let _ = home_components;
    Ok(render(&stack, true, intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    resolve_impl(path, cwd, home)
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((_, first)) = seen.iter().find(|(path, _)| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }
    Ok(())
}

fn absolute_name(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", stack.join("/"))
    }
}

#[derive(Clone)]
struct WorkItem {
    text: String,
    offset: usize,
    source: Source,
}

#[derive(Clone, Copy)]
enum Source {
    Request,
    Target,
    Context,
}

fn traverse_links(
    initial: Vec<String>,
    request: &str,
    request_offset: usize,
    links: &[Link],
    max_expansions: usize,
) -> Result<Vec<String>, PathError> {
    let mut stack = initial;
    let mut work = Vec::new();

    for (part, offset) in split_with_offsets(request, request_offset) {
        if !part.is_empty() {
            work.push(WorkItem {
                text: part.to_string(),
                offset,
                source: Source::Request,
            });
        }
    }

    let mut index = 0;
    let mut active = Vec::<usize>::new();
    let mut frames = Vec::<(usize, Vec<WorkItem>)>::new();
    let mut expansions = 0usize;

    while index < work.len() {
        let item = work[index].clone();
        index += 1;

        match item.text.as_str() {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|component| component != "..") {
                    stack.pop();
                } else {
                    return Err(PathError::Escape { at: item.offset });
                }
            }
            _ => {
                stack.push(item.text);
                let current = absolute_name(&stack);
                if let Some((link_index, link)) = links
                    .iter()
                    .enumerate()
                    .find(|(_, link)| link.path == current)
                {
                    if active.contains(&link_index) {
                        return Err(PathError::Loop { index: link_index });
                    }
                    if expansions >= max_expansions {
                        return Err(PathError::Limit { index: link_index });
                    }
                    expansions += 1;

                    stack.pop();
                    if link.target.starts_with('/') {
                        stack.clear();
                    }

                    let target_items = split_with_offsets(&link.target, 0)
                        .into_iter()
                        .filter(|(part, _)| !part.is_empty())
                        .map(|(part, offset)| WorkItem {
                            text: part.to_string(),
                            offset,
                            source: Source::Target,
                        })
                        .collect::<Vec<_>>();

                    let remaining = work[index..].to_vec();
                    frames.push((link_index, remaining));
                    active.push(link_index);
                    work.truncate(index);
                    work.extend(target_items);
                    index = 0;
                }
            }
        }

        while index >= work.len() && !frames.is_empty() {
            let (link_index, remaining) = frames.pop().unwrap();
            if active.last() == Some(&link_index) {
                active.pop();
            }
            work = remaining;
            index = 0;
        }
    }
    Ok(stack)
}

fn split_with_offsets(path: &str, base: usize) -> Vec<(&str, usize)> {
    let mut result = Vec::new();
    let mut start = 0;
    for part in path.split('/') {
        result.push((part, base + start));
        start += part.len() + 1;
    }
    result
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = normalized_context(cwd, true)?;
    let home_components = match home {
        Some(value) => Some(normalized_context(value, false)?),
        None => None,
    };
    validate_links(links)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (absolute, initial, rest, offset) = if path.starts_with('~') {
        let (is_abs, home_stack, rest, offset) = request_start(path, home)?;
        (is_abs, home_stack, rest, offset)
    } else if path.starts_with('/') {
        (true, Vec::new(), path, 0)
    } else {
        (false, cwd_components, path, 0)
    };

    let initial = if absolute { initial } else { initial };
    let stack = traverse_links(initial, rest, offset, links, max_expansions)?;
    let mut result = render(&stack, true, intent);
    if let Some(home) = home_components {
        let _ = home;
    }
    Ok(result)
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target)
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = vec!["..".to_string(); base.len() - common];
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    }
}

fn result_components(path: &str) -> Vec<String> {
    components(path).into_iter().map(str::to_string).collect()
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_components = result_components(&resolved_base);
    let target_components = result_components(&resolved_target);
    let target_directory = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_components, &target_components)];
    for link in links {
        let probe = resolve_links(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else {
            continue;
        };
        let probe_components = result_components(&probe);
        if probe_components.len() <= target_components.len()
            && probe_components
                .iter()
                .zip(&target_components)
                .all(|(a, b)| a == b)
        {
            let key_components = result_components(&link.path);
            let mut candidate = route(&base_components, &key_components);
            let suffix = &target_components[probe_components.len()..];
            if !suffix.is_empty() {
                if candidate == "." {
                    candidate = suffix.join("/");
                } else {
                    candidate.push('/');
                    candidate.push_str(&suffix.join("/"));
                }
            }
            candidates.push(candidate);
        }
    }

    let mut valid = Vec::<String>::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_directory && candidate != "." {
            candidate.push('/');
        }
        if let Ok(resolved) = resolve_links(
            &candidate,
            &resolved_base,
            home,
            links,
            max_expansions,
        ) {
            if result_components(&resolved) == target_components && !valid.contains(&candidate) {
                valid.push(candidate);
            }
        }
    }

    valid
        .into_iter()
        .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())))
        .ok_or(PathError::Escape { at: 0 })
}

fn main() {}
