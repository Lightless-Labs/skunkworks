fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort();
    out.dedup();
    out
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn pct(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(e: &[Evidence]) -> Vec<Evidence> {
    let mut v = e.to_vec();
    v.sort_by(|a, b| a.constraint.cmp(&b.constraint).then(a.path.cmp(&b.path)));
    v
}

fn constraints_line(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn evidence_lines(es: &[Evidence]) -> String {
    evidence_sorted(es)
        .iter()
        .map(|e| {
            format!(
                "\n  C:{} via {}",
                constraint_text(e.constraint),
                e.path.join(" -> ")
            )
        })
        .collect::<String>()
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", pct(site), pct(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", pct(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            pct(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            pct(package),
            version_text(*version),
            pct(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", pct(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            pct(package),
            evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            pct(package),
            constraints_line(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            pct(package),
            version_text(*selected),
            constraints_line(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            pct(package),
            version_text(*selected),
            pct(feature),
            pct(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone, Default)]
struct NodeData {
    constraints: std::collections::BTreeMap<String, Vec<Constraint>>,
    origins: std::collections::BTreeMap<String, Vec<Origin>>,
    requested: std::collections::BTreeMap<String, std::collections::BTreeSet<String>>,
    activated: std::collections::BTreeMap<String, std::collections::BTreeSet<String>>,
    paths: std::collections::BTreeMap<String, Vec<String>>,
}

fn platform_matches(p: Platform, t: Target) -> bool {
    matches!(p, Platform::Any)
        || matches!(
            (p, t),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn root_token(i: usize, r: &Requirement) -> String {
    let mut fs = r.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        fs.join(",")
    )
}

fn dep_token(source: &str, version: Version, j: usize, d: &Dependency) -> String {
    let mut fs = d.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        j,
        d.package,
        constraint_text(d.constraint),
        fs.join(","),
        d.gate.as_deref().unwrap_or("*"),
        match d.platform {
            Platform::Any => "any",
            Platform::Linux => "linux",
            Platform::Mac => "mac",
            Platform::Windows => "windows",
        }
    )
}

fn release_for<'a>(
    catalog: &'a [Package],
    index: &std::collections::BTreeMap<String, usize>,
    name: &str,
    version: Version,
) -> Option<&'a Release> {
    catalog
        .get(*index.get(name)?)
        .and_then(|p| p.releases.iter().find(|r| r.version == version))
}

fn closure(
    catalog: &[Package],
    index: &std::collections::BTreeMap<String, usize>,
    roots: &[Requirement],
    selected: &std::collections::BTreeMap<String, Version>,
    target: Target,
) -> NodeData {
    let mut data = NodeData::default();

    for (i, root) in roots.iter().enumerate() {
        let token = root_token(i, root);
        data.constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        data.origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: vec![token],
            });
        data.requested
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
    }

    loop {
        let mut changed = false;

        for (package, version) in selected {
            let Some(release) = release_for(catalog, index, package, *version) else {
                continue;
            };

            let mut activated = data.activated.get(package).cloned().unwrap_or_default();
            let requests = data.requested.get(package).cloned().unwrap_or_default();
            for feature in requests {
                if activated.insert(feature) {
                    changed = true;
                }
            }

            loop {
                let before = activated.len();
                let current: Vec<String> = activated.iter().cloned().collect();
                for feature in current {
                    if let Some(decl) = release.features.iter().find(|f| f.name == feature) {
                        for enable in &decl.enables {
                            if activated.insert(enable.clone()) {
                                changed = true;
                            }
                        }
                    }
                }
                if activated.len() == before {
                    break;
                }
            }

            if data.activated.get(package) != Some(&activated) {
                data.activated.insert(package.clone(), activated.clone());
                changed = true;
            }

            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !activated.contains(g))
                {
                    continue;
                }

                let token = dep_token(package, *version, j, dep);
                data.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                let mut path = data.paths.get(package).cloned().unwrap_or_default();
                path.push(token.clone());
                data.origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path,
                    });

                let requested = data.requested.entry(dep.package.clone()).or_default();
                for feature in &dep.features {
                    if requested.insert(feature.clone()) {
                        changed = true;
                    }
                }
            }
        }

        let mut paths = std::collections::BTreeMap::<String, Vec<String>>::new();
        let mut queue = std::collections::VecDeque::<String>::new();
        for (i, root) in roots.iter().enumerate() {
            let candidate = vec![root_token(i, root)];
            let replace = paths
                .get(&root.package)
                .is_none_or(|old| (candidate.len(), &candidate) < (old.len(), old));
            if replace {
                paths.insert(root.package.clone(), candidate);
                queue.push_back(root.package.clone());
            }
        }

        while let Some(source) = queue.pop_front() {
            let Some(version) = selected.get(&source).copied() else {
                continue;
            };
            let Some(release) = release_for(catalog, index, &source, version) else {
                continue;
            };
            let activated = data.activated.get(&source).cloned().unwrap_or_default();
            let source_path = paths.get(&source).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !activated.contains(g))
                {
                    continue;
                }

                let mut candidate = source_path.clone();
                candidate.push(dep_token(&source, version, j, dep));
                let replace = paths
                    .get(&dep.package)
                    .is_none_or(|old| (candidate.len(), &candidate) < (old.len(), old));
                if replace {
                    paths.insert(dep.package.clone(), candidate);
                    queue.push_back(dep.package.clone());
                }
            }
        }

        if data.paths != paths {
            data.paths = paths;
            changed = true;
        }

        if !changed {
            break;
        }
    }

    for origins in data.origins.values_mut() {
        for origin in origins {
            if origin.path.len() > 1 {
                let token = origin.path.last().cloned().unwrap_or_default();
                if let Some(source) = token.split('@').next() {
                    if let Some(path) = data.paths.get(source) {
                        let mut canonical = path.clone();
                        canonical.push(token);
                        origin.path = canonical;
                    }
                }
            }
        }
    }
    data
}

fn canonical_evidence(data: &NodeData, package: &str) -> Vec<Evidence> {
    let mut out = data
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect::<Vec<_>>();
    out.sort_by(|a, b| a.constraint.cmp(&b.constraint).then(a.path.cmp(&b.path)));
    out
}

fn canonical_reqs(data: &NodeData, package: &str) -> Vec<Constraint> {
    canonical_constraints(
        data.constraints
            .get(package)
            .map(Vec::as_slice)
            .unwrap_or(&[]),
    )
}

fn dynamic_error(
    catalog: &[Package],
    index: &std::collections::BTreeMap<String, usize>,
    selected: &std::collections::BTreeMap<String, Version>,
    data: &NodeData,
) -> Option<ResolveError> {
    for package in data.constraints.keys() {
        if !index.contains_key(package) {
            return Some(ResolveError::Missing {
                package: package.clone(),
                evidence: canonical_evidence(data, package),
            });
        }
    }

    for (package, version) in selected {
        let reqs = canonical_reqs(data, package);
        if reqs.iter().any(|c| !matches(*version, *c)) {
            return Some(ResolveError::Conflict {
                package: package.clone(),
                selected: *version,
                requirements: reqs,
                evidence: canonical_evidence(data, package),
            });
        }
    }

    for (package, version) in selected {
        let Some(release) = release_for(catalog, index, package, *version) else {
            continue;
        };
        let activated = data.activated.get(package).cloned().unwrap_or_default();
        for feature in activated {
            if !release.features.iter().any(|f| f.name == feature) {
                return Some(ResolveError::MissingFeature {
                    package: package.clone(),
                    selected: *version,
                    feature,
                    path: data.paths.get(package).cloned().unwrap_or_default(),
                });
            }
        }
    }

    for package in data.constraints.keys() {
        if selected.contains_key(package) {
            continue;
        }
        let reqs = canonical_reqs(data, package);
        let has_match = index
            .get(package)
            .and_then(|i| catalog.get(*i))
            .is_some_and(|p| {
                p.releases
                    .iter()
                    .any(|r| reqs.iter().all(|c| matches(r.version, *c)))
            });
        if !has_match {
            return Some(ResolveError::NoMatch {
                package: package.clone(),
                requirements: reqs,
                evidence: canonical_evidence(data, package),
            });
        }
    }
    None
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    index: std::collections::BTreeMap<String, usize>,
    attempts: usize,
    budget: usize,
    first_error: Option<ResolveError>,
    best: Option<(usize, Vec<Selection>, std::collections::BTreeMap<String, Version>)>,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn visit(
        &mut self,
        selected: &mut std::collections::BTreeMap<String, Version>,
    ) -> Result<(), ResolveError> {
        let data = closure(self.catalog, &self.index, self.roots, selected, self.target);
        if let Some(error) = dynamic_error(self.catalog, &self.index, selected, &data) {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        let next = data
            .constraints
            .keys()
            .find(|p| !selected.contains_key(*p))
            .cloned();

        let Some(package) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|l| selected.get(&l.package) != Some(&l.version))
                .count();
            let selections = selected
                .iter()
                .map(|(package, version)| Selection {
                    package: package.clone(),
                    version: *version,
                })
                .collect::<Vec<_>>();

            let replace = match &self.best {
                None => true,
                Some((best_cost, _, _)) if cost < *best_cost => true,
                Some((best_cost, _, best_map)) if cost == *best_cost => self
                    .catalog_names
                    .iter()
                    .find_map(|name| match (selected.get(name), best_map.get(name)) {
                        (Some(a), Some(b)) if a != b => Some(a > b),
                        (Some(_), None) => Some(true),
                        (None, Some(_)) => Some(false),
                        _ => None,
                    })
                    .unwrap_or(false),
                _ => false,
            };
            if replace {
                self.best = Some((cost, selections, selected.clone()));
            }
            return Ok(());
        };

        let reqs = canonical_reqs(&data, &package);
        let mut candidates = self.catalog[self.index[&package]]
            .releases
            .iter()
            .filter(|r| reqs.iter().all(|c| matches(r.version, *c)))
            .map(|r| r.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.iter().find(|l| l.package == package) {
            if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
                let value = candidates.remove(pos);
                candidates.insert(0, value);
            }
        }

        for candidate in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            self.visit(selected)?;
            selected.remove(&package);
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enable) in feature.enables.iter().enumerate() {
                    if !valid_name(enable) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enable.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut names = std::collections::BTreeSet::new();
            for feature in &release.features {
                if !names.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let index = catalog
        .iter()
        .enumerate()
        .map(|(i, p)| (p.name.clone(), i))
        .collect::<std::collections::BTreeMap<_, _>>();
    let mut catalog_names = catalog.iter().map(|p| p.name.clone()).collect::<Vec<_>>();
    catalog_names.sort();

    let mut search = Search {
        catalog,
        roots,
        locks,
        target,
        index: index.clone(),
        attempts: 0,
        budget: max_attempts,
        first_error: None,
        best: None,
        catalog_names,
    };
    search.visit(&mut std::collections::BTreeMap::new())?;

    let Some((_, selected, selected_map)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_data = closure(catalog, &index, roots, &selected_map, target);
    let before = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect::<std::collections::BTreeMap<_, _>>();
    let after = selected_map.clone();
    let mut names = std::collections::BTreeSet::new();
    names.extend(before.keys().cloned());
    names.extend(after.keys().cloned());

    let mut changes = Vec::new();
    for package in names {
        let b = before.get(&package).copied();
        let a = after.get(&package).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = index
                    .get(&package)
                    .and_then(|i| catalog.get(*i))
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_data
                    .constraints
                    .get(&package)
                    .into_iter()
                    .flatten()
                    .any(|c| !matches(old, *c))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

fn main() {}
