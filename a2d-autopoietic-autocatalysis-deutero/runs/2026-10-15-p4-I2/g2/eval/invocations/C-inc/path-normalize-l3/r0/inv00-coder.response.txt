#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<(usize, &str)> {
    let mut result = Vec::new();
    let bytes = path.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((start, &path[start..i]));
    }
    result
}

fn render_absolute(stack: &[String], intent: bool) -> String {
    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    if intent && !stack.is_empty() {
        result.push('/');
    }
    result
}

fn render_normalized(stack: &[String], absolute: bool, intent: bool) -> String {
    if absolute {
        return render_absolute(stack, intent);
    }
    if stack.is_empty() {
        return ".".to_string();
    }
    let mut result = stack.join("/");
    if intent {
        result.push('/');
    }
    result
}

fn reduce_component(
    stack: &mut Vec<String>,
    component: &str,
    absolute: bool,
    at: usize,
) -> Result<(), PathError> {
    match component {
        "." => {}
        ".." => {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_string());
            }
        }
        _ => stack.push(component.to_string()),
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes()[0] == b'/';
    let intent = directory_intent(path);
    let mut stack = Vec::new();

    for (at, component) in components(path) {
        reduce_component(&mut stack, component, absolute, at)?;
    }

    Ok(render_normalized(&stack, absolute, intent))
}

fn normalized_context(path: &str, invalid: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(invalid);
    }
    let normalized = normalize(path).map_err(|_| invalid)?;
    Ok(components(&normalized)
        .into_iter()
        .map(|(_, component)| component.to_string())
        .collect())
}

fn expand_tilde<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(Option<Vec<String>>, &'a str, usize), PathError> {
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let home = home.ok_or(PathError::MissingHome)?;
            let normalized = normalize(home).map_err(|_| PathError::InvalidHome)?;
            let base = components(&normalized)
                .into_iter()
                .map(|(_, component)| component.to_string())
                .collect();
            let consumed = if bytes.len() > 1 { 2 } else { 1 };
            return Ok((Some(base), &path[consumed..], consumed));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    Ok((None, path, 0))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_stack = normalized_context(cwd, PathError::InvalidCwd)?;
    let home_stack = match home {
        Some(home) => Some(normalized_context(home, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let intent = directory_intent(path);
    let (tilde_stack, request, offset) = expand_tilde(path, home)?;
    let absolute = path.starts_with('/');
    let mut stack = if absolute {
        Vec::new()
    } else if let Some(tilde_stack) = tilde_stack {
        tilde_stack
    } else {
        cwd_stack
    };

    for (at, component) in components(request) {
        reduce_component(&mut stack, component, true, at + offset)?;
    }

    Ok(render_absolute(&stack, intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();

    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }

        if let Some((_, first)) = seen.iter().find(|(path, _)| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }
    Ok(())
}

struct LinkTraversal<'a> {
    links: &'a [Link],
    active: Vec<usize>,
    expansions: usize,
    max_expansions: usize,
    stack: Vec<String>,
}

impl LinkTraversal<'_> {
    fn push_component(&mut self, component: &str, absolute: bool, at: usize) -> Result<(), PathError> {
        match component {
            "." => return Ok(()),
            ".." => {
                if self.stack.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
                return Ok(());
            }
            _ => {}
        }

        self.stack.push(component.to_string());
        let matched_path = render_absolute(&self.stack, false);
        if let Some((index, link)) = self
            .links
            .iter()
            .enumerate()
            .find(|(_, link)| link.path == matched_path)
        {
            if self.active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if self.expansions >= self.max_expansions {
                return Err(PathError::Limit { index });
            }
            self.expansions += 1;
            self.active.push(index);

            let target = &link.target;
            let target_absolute = target.starts_with('/');
            if target_absolute {
                self.stack.clear();
            } else {
                self.stack.pop();
            }
            for (target_at, target_component) in components(target) {
                self.push_component(target_component, true, target_at)?;
            }
            self.active.pop();
        }
        let _ = absolute;
        Ok(())
    }

    fn process(&mut self, path: &str, offset: usize) -> Result<(), PathError> {
        for (at, component) in components(path) {
            self.push_component(component, true, at + offset)?;
        }
        Ok(())
    }
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_stack = normalized_context(cwd, PathError::InvalidCwd)?;
    let home_stack = match home {
        Some(home) => Some(normalized_context(home, PathError::InvalidHome)?),
        None => None,
    };

    validate_links(links)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (tilde_stack, request, offset) = expand_tilde(path, home)?;
    let absolute = path.starts_with('/');
    let mut traversal = LinkTraversal {
        links,
        active: Vec::new(),
        expansions: 0,
        max_expansions,
        stack: if absolute {
            Vec::new()
        } else if let Some(base) = tilde_stack {
            base
        } else {
            cwd_stack
        },
    };

    if !absolute && path.starts_with('~') && offset > 0 {
        // Expanded tilde already supplied the normalized home components.
    } else if !absolute && home_stack.is_some() {
        // The validated home is deliberately unused for ordinary relative paths.
    }

    traversal.process(request, offset)?;
    Ok(render_absolute(&traversal.stack, intent))
}

fn main() {}
