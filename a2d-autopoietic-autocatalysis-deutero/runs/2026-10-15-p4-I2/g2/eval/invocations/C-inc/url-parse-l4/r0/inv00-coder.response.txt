use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn scan_ascii_url(input: &str) -> Result<(), ParseError> {
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

fn is_ascii_alpha(byte: u8) -> bool {
    byte.is_ascii_alphabetic()
}

fn is_scheme_byte(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')
}

fn is_host_byte(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    origins.push(i);
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match String::from_utf8(decoded) {
        Ok(text) => Ok(text),
        Err(error) => {
            let at = error.utf8_error().valid_up_to();
            Err(CodecError::InvalidUtf8 {
                at: origins.get(at).copied().unwrap_or(input.len()),
            })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for byte in text.bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };

        if safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn map_codec_parse(error: CodecError, base: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

fn validate_component(
    component: Component,
    value: &str,
    base: usize,
) -> Result<(), ParseError> {
    decode_component(component, value)
        .map(|_| ())
        .map_err(|error| map_codec_parse(error, base))
}

fn parse_port(port_text: &str, port_start: usize) -> Result<u16, ParseError> {
    if port_text.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    if let Some((offset, _)) = port_text
        .bytes()
        .enumerate()
        .find(|(_, byte)| !byte.is_ascii_digit())
    {
        return Err(ParseError::InvalidPort {
            at: port_start + offset,
        });
    }

    let bytes = port_text.as_bytes();
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for byte in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|current| current.checked_add((byte - b'0') as u32))
        {
            Some(value) => value,
            None => return Err(ParseError::InvalidPort { at: port_start }),
        };
    }

    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

fn find_byte(bytes: &[u8], start: usize, end: usize, target: u8) -> Option<usize> {
    (start..end).find(|&i| bytes[i] == target)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    scan_ascii_url(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(colon) => colon,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !is_ascii_alpha(bytes[0]) {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !is_scheme_byte(byte) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    let authority_start = colon + 1;
    if bytes.get(authority_start) != Some(&b'/')
        || bytes.get(authority_start + 1) != Some(&b'/')
    {
        return Err(ParseError::MissingAuthority { at: authority_start });
    }
    let authority_start = authority_start + 2;
    let authority_end = (authority_start..bytes.len())
        .find(|&i| matches!(bytes[i], b'/' | b'?' | b'#'))
        .unwrap_or(bytes.len());

    let mut at_sign = None;
    for i in authority_start..authority_end {
        if bytes[i] == b'@' {
            if at_sign.is_some() {
                return Err(ParseError::InvalidUserInfo { at: i });
            }
            at_sign = Some(i);
        }
    }

    let (userinfo, host_start) = match at_sign {
        Some(at) => {
            if at == authority_start {
                return Err(ParseError::InvalidUserInfo { at });
            }
            (Some(input[authority_start..at].to_owned()), at + 1)
        }
        None => (None, authority_start),
    };

    let mut ipv6 = false;
    let (host, port) = if bytes.get(host_start) == Some(&b'[') {
        let close = match find_byte(bytes, host_start + 1, authority_end, b']') {
            Some(close) => close,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &input[host_start + 1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }
        ipv6 = true;

        let suffix_start = close + 1;
        let port = if suffix_start == authority_end {
            None
        } else if bytes[suffix_start] == b':' {
            Some(parse_port(
                &input[suffix_start + 1..authority_end],
                suffix_start + 1,
            )?)
        } else {
            return Err(ParseError::InvalidHost { at: suffix_start });
        };
        (interior.to_ascii_lowercase(), port)
    } else {
        let host_port_colon = find_byte(bytes, host_start, authority_end, b':');
        let host_end = host_port_colon.unwrap_or(authority_end);

        if host_start == host_end {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        if let Some((offset, _)) = bytes[host_start..host_end]
            .iter()
            .enumerate()
            .find(|(_, byte)| !is_host_byte(**byte))
        {
            return Err(ParseError::InvalidHost {
                at: host_start + offset,
            });
        }

        let port = match host_port_colon {
            Some(port_colon) => Some(parse_port(
                &input[port_colon + 1..authority_end],
                port_colon + 1,
            )?),
            None => None,
        };
        (input[host_start..host_end].to_ascii_lowercase(), port)
    };

    let fragment_start = find_byte(bytes, authority_end, bytes.len(), b'#');
    let component_end = fragment_start.unwrap_or(bytes.len());
    let query_start = find_byte(bytes, authority_end, component_end, b'?');
    let path_end = query_start.unwrap_or(component_end);

    let path = if path_end == authority_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };

    let query = query_start.map(|start| input[start + 1..component_end].to_owned());
    let fragment = fragment_start.map(|start| input[start + 1..].to_owned());

    let userinfo_start = authority_start;
    if let (Some(raw), Some(at)) = (userinfo.as_deref(), at_sign) {
        validate_component(Component::UserInfo, raw, userinfo_start)
            .map_err(|error| match error {
                ParseError::BadEscape { .. } | ParseError::InvalidUtf8 { .. } => error,
                _ => error,
            })?;
        let _ = at;
    }

    if path_end != authority_end {
        validate_component(Component::Path, &path, authority_end)?;
    }
    if let (Some(raw), Some(start)) = (query.as_deref(), query_start) {
        validate_component(Component::Query, raw, start + 1)?;
    }
    if let (Some(raw), Some(start)) = (fragment.as_deref(), fragment_start) {
        validate_component(Component::Fragment, raw, start + 1)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn map_codec_path(error: CodecError) -> PathError {
    match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(map_codec_path)?;

    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_navigation = false;

    for (index, segment) in segments.iter().enumerate() {
        let decoded =
            decode_component(Component::Path, segment).map_err(map_codec_path)?;
        let is_last = index + 1 == segments.len();

        match decoded.as_str() {
            "." => {
                if is_last {
                    terminal_navigation = true;
                }
            }
            ".." => {
                if !stack.is_empty() {
                    stack.pop();
                }
                if is_last {
                    terminal_navigation = true;
                }
            }
            _ => stack.push(segment),
        }
    }

    if terminal_navigation {
        stack.push("");
    }

    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn normalized_url(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

fn split_reference(reference: &str) -> (usize, Option<usize>, Option<usize>) {
    let bytes = reference.as_bytes();
    let fragment = find_byte(bytes, 0, bytes.len(), b'#');
    let component_end = fragment.unwrap_or(bytes.len());
    let query = find_byte(bytes, 0, component_end, b'?');
    let path_end = query.unwrap_or(component_end);
    (path_end, query, fragment)
}

fn has_absolute_scheme(reference: &str) -> bool {
    let bytes = reference.as_bytes();
    let end = bytes
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(bytes.len());
    bytes[..end].contains(&b':')
}

fn raw_authority(url: &Url) -> String {
    let mut authority = String::new();
    if let Some(userinfo) = &url.userinfo {
        authority.push_str(userinfo);
        authority.push('@');
    }
    if url.ipv6 {
        authority.push('[');
        authority.push_str(&url.host);
        authority.push(']');
    } else {
        authority.push_str(&url.host);
    }
    if let Some(port) = url.port {
        authority.push(':');
        authority.push_str(&port.to_string());
    }
    authority
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_url = normalized_url(base_url)?;
    scan_ascii_url(reference).map_err(ResolveError::Reference)?;

    if has_absolute_scheme(reference) {
        let url = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalized_url(url);
    }

    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let url = parse_url(&constructed).map_err(ResolveError::Reference)?;
        return normalized_url(url);
    }

    let bytes = reference.as_bytes();
    let fragment_start = find_byte(bytes, 0, bytes.len(), b'#');
    let before_fragment_end = fragment_start.unwrap_or(bytes.len());
    let query_start = find_byte(bytes, 0, before_fragment_end, b'?');
    let path_end = query_start.unwrap_or(before_fragment_end);

    let reference_path = &reference[..path_end];
    let reference_query = query_start.map(|start| {
        reference[start + 1..before_fragment_end].to_owned()
    });
    let reference_fragment =
        fragment_start.map(|start| reference[start + 1..].to_owned());

    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_url.path.clone(),
            if query_start.is_some() {
                reference_query
            } else {
                base_url.query.clone()
            },
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = base_url.path.rfind('/').unwrap_or(0);
            format!("{}{}", &base_url.path[..=slash], reference_path)
        };
        (merged, reference_query)
    };

    let mut constructed = format!(
        "{}://{}{}",
        base_url.scheme,
        raw_authority(&base_url),
        selected_path
    );
    if let Some(query) = selected_query.as_deref() {
        constructed.push('?');
        constructed.push_str(query);
    }
    if let Some(fragment) = reference_fragment.as_deref() {
        constructed.push('#');
        constructed.push_str(fragment);
    }

    let url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    normalized_url(url)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct QueryMap {
    pub entries: Vec<(String, Vec<String>)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },
}

fn map_codec_query(error: CodecError, base: usize) -> QueryError {
    match error {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: base + at },
    }
}

fn replace_plus(value: &str) -> String {
    value.replace('+', " ")
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut field_start = 0;

    for field in input.split('&') {
        let field_len = field.len();
        if !field.is_empty() {
            let (raw_key, raw_value, value_offset) = match field.find('=') {
                Some(equal) => (&field[..equal], &field[equal + 1..], equal + 1),
                None => (field, "", field.len()),
            };
            let key_input = replace_plus(raw_key);
            let key = decode_component(Component::Query, &key_input)
                .map_err(|error| map_codec_query(error, field_start))?;

            if key.is_empty() {
                return Err(QueryError::EmptyKey { at: field_start });
            }

            let value_input = replace_plus(raw_value);
            let value = decode_component(Component::Query, &value_input)
                .map_err(|error| {
                    map_codec_query(error, field_start + value_offset)
                })?;

            if let Some((_, values)) =
                entries.iter_mut().find(|(existing, _)| existing == &key)
            {
                values.push(value);
            } else {
                entries.push((key, vec![value]));
            }
        }
        field_start += field_len + 1;
    }

    Ok(QueryMap { entries })
}

fn form_encode(text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for byte in text.bytes() {
        if byte == b' ' {
            output.push('+');
        } else if byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
        {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(&str, Vec<&str>)> = Vec::new();

    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, combined)) =
            merged.iter_mut().find(|(existing, _)| *existing == key)
        {
            combined.extend(values.iter().map(String::as_str));
        } else {
            merged.push((
                key.as_str(),
                values.iter().map(String::as_str).collect(),
            ));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = form_encode(key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, form_encode(value)));
        }
    }
    fields.join("&")
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CanonicalError {
    Parse(ParseError),
    Path(PathError),
    Query(QueryError),
}

fn encode_unreserved(text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for byte in text.bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let mut url = parse_url(input).map_err(CanonicalError::Parse)?;
    url.path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match url.query.as_deref() {
        Some(query) => Some(parse_query(query).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut output = format!("{}://", url.scheme);

    if let Some(userinfo) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, userinfo)
            .expect("parse_url validated userinfo");
        output.push_str(&encode_component(Component::UserInfo, &decoded));
        output.push('@');
    }

    if url.ipv6 {
        let address = Ipv6Addr::from_str(&url.host)
            .expect("parse_url validated IPv6 host");
        output.push('[');
        output.push_str(&address.to_string());
        output.push(']');
    } else {
        output.push_str(&url.host);
    }

    if let Some(port) = url.port {
        let omit = (url.scheme == "http" && port == 80)
            || (url.scheme == "https" && port == 443);
        if !omit {
            output.push(':');
            output.push_str(&port.to_string());
        }
    }

    let encoded_segments: Vec<String> = url
        .path
        .split('/')
        .map(|segment| {
            let decoded = decode_component(Component::Path, segment)
                .expect("parse_url validated path");
            encode_unreserved(&decoded)
        })
        .collect();
    output.push_str(&encoded_segments.join("/"));

    if let (Some(raw_query), Some(map)) = (url.query.as_deref(), query_map.as_ref()) {
        let _ = raw_query;
        output.push('?');
        output.push_str(&serialise_query(map));
    }

    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment)
            .expect("parse_url validated fragment");
        output.push('#');
        output.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(output)
}
