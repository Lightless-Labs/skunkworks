#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
    reserved_ids: Vec<String>,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(&bytes[..split], 0, 60, &mut minutes)?;
        parse_field(&bytes[split + 1..], split + 1, 24, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

fn syntax_at(base: usize) -> CronError {
    CronError::Syntax(base)
}

fn parse_number(fragment: &[u8], start: usize) -> Result<u64, CronError> {
    if fragment.is_empty() {
        return Err(CronError::Syntax(start));
    }
    if let Some((i, _)) = fragment.iter().enumerate().find(|(_, b)| !b.is_ascii_digit()) {
        return Err(CronError::Syntax(start + i));
    }
    if fragment.len() > 1 && fragment[0] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0u64;
    for &b in fragment {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u64))
            .ok_or(CronError::Bounds(start))?;
    }
    Ok(value)
}

fn parse_field(
    field: &[u8],
    field_offset: usize,
    max_value: u64,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut item_start = 0usize;

    loop {
        let item_end = field[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(field.len());
        let item = &field[item_start..item_end];
        let absolute_start = field_offset + item_start;

        if item.is_empty() {
            return Err(CronError::Syntax(absolute_start));
        }

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(CronError::Syntax(absolute_start + slash_positions[1]));
        }

        let slash = slash_positions.first().copied();
        let (base, step_fragment, step_start) = if let Some(pos) = slash {
            (&item[..pos], Some(&item[pos + 1..]), absolute_start + pos + 1)
        } else {
            (item, None, 0)
        };

        let is_star = base == b"*";
        let dash = base.iter().position(|&b| b == b'-');

        if slash.is_some() && !is_star && dash.is_none() {
            return Err(CronError::Syntax(absolute_start + slash.unwrap()));
        }

        let (start, end, dash_pos) = if is_star {
            (0u64, max_value, None)
        } else if let Some(dash_pos) = dash {
            let left = parse_number(&base[..dash_pos], absolute_start)?;
            if left > max_value {
                return Err(CronError::Bounds(absolute_start));
            }
            let right_start = absolute_start + dash_pos + 1;
            let right = parse_number(&base[dash_pos + 1..], right_start)?;
            if right > max_value {
                return Err(CronError::Bounds(right_start));
            }
            if left > right {
                return Err(CronError::Descending(absolute_start + dash_pos));
            }
            (left, right, Some(dash_pos))
        } else {
            let value = parse_number(base, absolute_start)?;
            if value > max_value {
                return Err(CronError::Bounds(absolute_start));
            }
            (value, value, None)
        };

        let step = if let Some(fragment) = step_fragment {
            let value = parse_number(fragment, step_start)?;
            let maximum_step = max_value + 1;
            if value == 0 || value > maximum_step {
                return Err(CronError::Step(step_start));
            }
            value
        } else {
            1
        };

        let _ = dash_pos;
        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) if next > value => value = next,
                _ => break,
            }
        }

        if item_end == field.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(())
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
            reserved_ids: Vec::new(),
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved_ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;

        let order = self.jobs.len();
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order,
        });
        self.reserved_ids.push(id.to_owned());
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower_exclusive = self.last_poll;
        let first_minute = match lower_exclusive {
            None => 0,
            Some(last) => last / 60_000 + 1,
        };
        let last_minute = now / 60_000;

        let mut tagged: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let mut matching = Vec::new();
            if first_minute <= last_minute {
                for minute in first_minute..=last_minute {
                    if job.cron.matches(minute) {
                        let at = minute * 60_000;
                        if lower_exclusive.map_or(true, |last| at > last) && at <= now {
                            matching.push(at);
                        }
                    }
                }
            }

            let emitted: Vec<u64> = match job.policy {
                Policy::All => matching,
                Policy::Once => matching.last().copied().into_iter().collect(),
                Policy::Skip => matching
                    .into_iter()
                    .filter(|&at| at == now)
                    .collect(),
            };

            for at in emitted {
                tagged.push((
                    Occurrence {
                        id: job.id.clone(),
                        at,
                    },
                    job.priority,
                    job.order,
                ));
            }
        }

        tagged.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(tagged.into_iter().map(|(occurrence, _, _)| occurrence).collect())
    }
}

impl Default for Calendar {
    fn default() -> Self {
        Self::new()
    }
}

fn main() {}
