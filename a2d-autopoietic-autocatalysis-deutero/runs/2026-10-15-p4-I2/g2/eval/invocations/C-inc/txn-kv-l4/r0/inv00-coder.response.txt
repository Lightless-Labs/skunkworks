```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Entry>>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Entry {
    value: String,
    deadline: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Version {
    revision: u64,
    entry: Option<Entry>,
}

struct Transaction {
    revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Option<Entry>>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn check_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn visible_entry(&self, key: &str, tx: Option<u64>) -> Option<Entry> {
        if let Some(id) = tx {
            let transaction = self.transactions.get(&id)?;
            if let Some(entry) = transaction.writes.get(key) {
                return entry.clone();
            }
            self.data.get(key).and_then(|versions| {
                versions
                    .iter()
                    .rev()
                    .find(|version| version.revision <= transaction.revision)
                    .and_then(|version| version.entry.clone())
            })
        } else {
            self.data
                .get(key)
                .and_then(|versions| versions.last())
                .and_then(|version| version.entry.clone())
        }
    }

    fn is_live(entry: &Entry, now: u64) -> bool {
        entry.deadline.map_or(true, |deadline| deadline > now)
    }

    fn append_version(&mut self, key: String, entry: Option<Entry>, revision: u64) {
        self.data
            .entry(key)
            .or_default()
            .push(Version { revision, entry });
    }

    fn snapshot_name_error(name: &str) -> Option<Error> {
        if name.is_empty() {
            return Some(Error::NameEmpty);
        }
        for (at, ch) in name.char_indices() {
            let valid = if at == 0 {
                ch.is_ascii_lowercase()
            } else {
                ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
            };
            if !valid {
                return Some(Error::NameInvalid { at });
            }
        }
        None
    }

    fn matching_entries(
        &self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Vec<(String, String)> {
        let mut result = Vec::new();
        for key in self.data.keys() {
            if key.as_str() < start
                || end.map_or(false, |end| key.as_str() >= end)
                || prefix.map_or(false, |prefix| !key.starts_with(prefix))
            {
                continue;
            }
            if let Some(entry) = self.visible_entry(key, tx) {
                if Self::is_live(&entry, now) {
                    result.push((key.clone(), entry.value));
                }
            }
        }
        if let Some(id) = tx {
            if let Some(transaction) = self.transactions.get(&id) {
                for (key, entry) in &transaction.writes {
                    if self.data.contains_key(key)
                        || key.as_str() < start
                        || end.map_or(false, |end| key.as_str() >= end)
                        || prefix.map_or(false, |prefix| !key.starts_with(prefix))
                    {
                        continue;
                    }
                    if let Some(entry) = entry {
                        if Self::is_live(entry, now) {
                            result.push((key.clone(), entry.value.clone()));
                        }
                    }
                }
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        result
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(
            id,
            Transaction {
                revision: self.revision,
                read_only,
                writes: BTreeMap::new(),
            },
        );
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let transaction = self
            .transactions
            .get(&id)
            .ok_or(Error::UnknownTransaction(id))?;

        for key in transaction.writes.keys() {
            let changed = self
                .data
                .get(key)
                .and_then(|versions| versions.last())
                .map_or(false, |version| version.revision > transaction.revision);
            if changed {
                return Err(Error::Conflict {
                    tx: id,
                    key: key.clone(),
                });
            }
        }

        if transaction.writes.is_empty() {
            self.transactions.remove(&id);
            return Ok(self.revision);
        }

        self.revision += 1;
        let revision = self.revision;
        let writes = self
            .transactions
            .remove(&id)
            .expect("transaction was checked")
            .writes;
        for (key, entry) in writes {
            self.append_version(key, entry, revision);
        }
        Ok(revision)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.remove(&id).is_none() {
            return Err(Error::UnknownTransaction(id));
        }
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }

        let entry = match ttl {
            Some(0) => None,
            Some(duration) => Some(Entry {
                value: value.to_string(),
                deadline: Some(now.checked_add(duration).ok_or(Error::DeadlineOverflow)?),
            }),
            None => Some(Entry {
                value: value.to_string(),
                deadline: None,
            }),
        };

        if let Some(id) = tx {
            self.transactions
                .get_mut(&id)
                .unwrap()
                .writes
                .insert(key.to_string(), entry);
        } else {
            self.revision += 1;
            self.append_version(key.to_string(), entry, self.revision);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }

        let was_live = self
            .visible_entry(key, tx)
            .map_or(false, |entry| Self::is_live(&entry, now));

        if let Some(id) = tx {
            self.transactions
                .get_mut(&id)
                .unwrap()
                .writes
                .insert(key.to_string(), None);
        } else {
            self.revision += 1;
            self.append_version(key.to_string(), None, self.revision);
        }
        Ok(was_live)
    }

    pub fn get(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self
            .visible_entry(key, tx)
            .filter(|entry| Self::is_live(entry, now))
            .map(|entry| entry.value))
    }

    pub fn count(&mut self, now: u64, tx: Option<u64>, value: &str) -> Result<usize, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Ok(self
            .matching_entries(now, tx, "", None, None)
            .iter()
            .filter(|(_, stored)| stored == value)
            .count())
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if end.map_or(false, |end| end <= start) {
            return Err(Error::InvalidRange);
        }
        if let Some(cursor) = cursor {
            if cursor.start != start
                || cursor.end.as_deref() != end
                || cursor.prefix.as_deref() != prefix
                || cursor.offset != offset
                || cursor.last.is_empty()
                || cursor.last.as_str() < start
                || end.map_or(false, |end| cursor.last.as_str() >= end)
                || prefix.map_or(false, |prefix| !cursor.last.starts_with(prefix))
            {
                return Err(Error::CursorMismatch);
            }
        }

        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }

        let mut matching = self.matching_entries(now, tx, start, end, prefix);
        if let Some(cursor) = cursor {
            matching.retain(|(key, _)| key.as_str() > cursor.last.as_str());
        } else if offset > 0 {
            if offset >= matching.len() {
                return Ok(Page {
                    items: Vec::new(),
                    next: None,
                });
            }
            matching.drain(..offset);
        }

        let has_more = matching.len() > limit;
        let items: Vec<_> = matching.into_iter().take(limit).collect();
        let next = if has_more {
            items.last().map(|(last, _)| Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: last.clone(),
            })
        } else {
            None
        };
        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if end.map_or(false, |end| end <= start) {
            return Err(Error::InvalidRange);
        }

        let entries = self.matching_entries(now, tx, start, end, prefix);
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in entries {
            let parsed = parse_canonical_i64(&value).ok_or_else(|| Error::InvalidNumber {
                key: key.clone(),
                value: value.clone(),
            })?;
            sum = sum.checked_add(parsed).ok_or(Error::SumOverflow { key })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if let Some(error) = Self::snapshot_name_error(name) {
            return Err(error);
        }
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }

        let mut saved = BTreeMap::new();
        for (key, versions) in &self.data {
            if let Some(entry) = versions.last().and_then(|version| version.entry.as_ref()) {
                if Self::is_live(entry, now) {
                    saved.insert(key.clone(), entry.clone());
                }
            }
        }
        self.snapshots.insert(name.to_string(), saved);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        let saved = self
            .snapshots
            .get(name)
            .cloned()
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let live_saved: BTreeMap<_, _> = saved
            .into_iter()
            .filter(|(_, entry)| Self::is_live(entry, now))
            .collect();
        let mut union = BTreeSet::new();
        for (key, versions) in &self.data {
            if versions
                .last()
                .and_then(|version| version.entry.as_ref())
                .map_or(false, |entry| Self::is_live(entry, now))
            {
                union.insert(key.clone());
            }
        }
        union.extend(live_saved.keys().cloned());

        if union.is_empty() {
            return Ok(self.revision);
        }

        self.revision += 1;
        let revision = self.revision;
        for key in union {
            let entry = live_saved.get(&key).cloned();
            self.append_version(key, entry, revision);
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_string()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let mut removed = 0;
        for versions in self.data.values_mut() {
            let mut keep = BTreeSet::new();
            if !versions.is_empty() {
                keep.insert(versions.len() - 1);
            }
            for transaction in self.transactions.values() {
                if let Some(index) = versions
                    .iter()
                    .rposition(|version| version.revision <= transaction.revision)
                {
                    keep.insert(index);
                }
            }

            let mut index = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let keys = self.data.len();
        let mut live_keys = 0;
        let mut versions = 0;
        let mut tombstones = 0;
        for history in self.data.values() {
            versions += history.len();
            for version in history {
                if version.entry.is_none() {
                    tombstones += 1;
                }
            }
            if history
                .last()
                .and_then(|version| version.entry.as_ref())
                .map_or(false, |entry| Self::is_live(entry, now))
            {
                live_keys += 1;
            }
        }

        let active_transactions = self.transactions.len();
        let read_only_transactions = self
            .transactions
            .values()
            .filter(|transaction| transaction.read_only)
            .count();
        let staged_writes = self
            .transactions
            .values()
            .map(|transaction| transaction.writes.len())
            .sum();

        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys,
            live_keys,
            versions,
            tombstones,
            active_transactions,
            read_only_transactions,
            staged_writes,
            snapshots: self.snapshots.len(),
        })
    }
}

fn parse_canonical_i64(value: &str) -> Option<i64> {
    if value.is_empty() {
        return None;
    }
    let digits = if let Some(rest) = value.strip_prefix('-') {
        if rest.is_empty() || rest == "0" {
            return None;
        }
        rest
    } else {
        value
    };

    if digits == "0" {
        return value.parse::<i64>().ok();
    }
    if digits.starts_with('0') || !digits.bytes().all(|byte| byte.is_ascii_digit()) {
        return None;
    }
    value.parse::<i64>().ok()
}
```
