use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative { i += 1; }
        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
        if i == numerator_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }
        let numerator_end = i;

        let mut denominator_span = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            let slash = i;
            i += 1;
            let den_start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
            if i == den_start {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            if i < bytes.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            denominator_span = Some((slash, den_start, i));
        }

        if numerator_end - numerator_start >= 2 && bytes[numerator_start] == b'0' {
            return Err(parse_err(numerator_start + 1, ParseKind::NonCanonical));
        }
        if negative && numerator_end - numerator_start == 1 && bytes[numerator_start] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }
        if let Some((_, start, end)) = denominator_span {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_err(start + 1, ParseKind::NonCanonical));
            }
        }

        let nmag = parse_magnitude(&bytes[numerator_start..numerator_end]);
        let nlimit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if nmag.is_none() || nmag.unwrap() > nlimit {
            return Err(parse_err(0, ParseKind::OutOfRange));
        }
        let nmag = nmag.unwrap();
        let num = if negative {
            if nmag == (1u128 << 63) { i64::MIN } else { -(nmag as i64) }
        } else {
            nmag as i64
        };

        let Some((slash, den_start, den_end)) = denominator_span else {
            return Ok(Self { num, den: 1 });
        };
        let dmag = parse_magnitude(&bytes[den_start..den_end]);
        if dmag.is_none() || dmag.unwrap() > i64::MAX as u128 {
            return Err(parse_err(den_start, ParseKind::OutOfRange));
        }
        let dmag = dmag.unwrap() as i64;
        if dmag == 0 {
            return Err(parse_err(den_start, ParseKind::ZeroDenominator));
        }
        if dmag == 1 {
            return Err(parse_err(slash, ParseKind::NonCanonical));
        }
        if num == 0 {
            return Err(parse_err(slash, ParseKind::NonCanonical));
        }
        if gcd_u128(num.unsigned_abs(), dmag as u128) > 1 {
            return Err(parse_err(slash, ParseKind::NonCanonical));
        }
        Ok(Self { num, den: dmag })
    }
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn parse_magnitude(bytes: &[u8]) -> Option<u128> {
    let mut value = 0u128;
    for &b in bytes {
        value = value.checked_mul(10)?.checked_add((b - b'0') as u128)?;
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let zero = Rational { num: 0, den: 1 };
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let product = self.data[r * self.cols + k]
                        .mul(&rhs.data[k * rhs.cols + c])
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&product)
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self, self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let zero = Rational { num: 0, den: 1 };
        let one = Rational { num: 1, den: 1 };
        let mut data = vec![zero; len];
        for r in 0..n {
            for c in 0..n {
                data[r * width + c] = self.data[r * n + c];
            }
            data[r * width + n + r] = one;
        }
        let augmented = Self { rows: n, cols: width, data };
        let reduction = eliminate(&augmented, n)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular { rank: reduction.pivots.len() });
        }
        let mut result = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                result.push(reduction.matrix.data[r * width + n + c]);
            }
        }
        Ok(Self { rows: n, cols: n, data: result })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize { self.pivots.len() }
}

fn eliminate(matrix: &Matrix, pivot_cols: usize) -> Result<Reduction, MatrixError> {
    let mut result = matrix.clone();
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;
    let zero = Rational { num: 0, den: 1 };

    for col in 0..pivot_cols {
        if p == result.rows { break; }
        let mut chosen: Option<usize> = None;
        for row in p..result.rows {
            let value = result.data[row * result.cols + col];
            if value.num == 0 { continue; }
            if chosen.map_or(true, |old| {
                pivot_cmp(value, result.data[old * result.cols + col], row, old) == Ordering::Less
            }) {
                chosen = Some(row);
            }
        }
        let Some(q) = chosen else { continue };
        if q != p {
            for c in 0..result.cols {
                result.data.swap(p * result.cols + c, q * result.cols + c);
            }
            swaps.push((p, q));
        }

        let pivot = result.data[p * result.cols + col];
        let mut normalized = Vec::with_capacity(result.cols);
        for c in 0..result.cols {
            let value = result.data[p * result.cols + c]
                .div(&pivot)
                .map_err(|source| MatrixError::Arithmetic { row: p, col: c, step: p, source })?;
            normalized.push(value);
        }
        for c in 0..result.cols {
            result.data[p * result.cols + c] = normalized[c];
        }

        for row in 0..result.rows {
            if row == p { continue; }
            let factor = result.data[row * result.cols + col];
            if factor == zero { continue; }
            for c in 0..result.cols {
                let product = factor.mul(&normalized[c])
                    .map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                let value = result.data[row * result.cols + c].sub(&product)
                    .map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                result.data[row * result.cols + c] = value;
            }
        }
        pivots.push(col);
        p += 1;
    }

    Ok(Reduction { matrix: result, pivots, swaps })
}

fn pivot_cmp(a: Rational, b: Rational, row_a: usize, row_b: usize) -> Ordering {
    a.den.cmp(&b.den)
        .then_with(|| a.num.unsigned_abs().cmp(&b.num.unsigned_abs()))
        .then_with(|| row_a.cmp(&row_b))
}
