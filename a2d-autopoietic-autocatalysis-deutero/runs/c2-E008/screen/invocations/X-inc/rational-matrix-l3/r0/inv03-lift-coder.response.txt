#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let value = Self::normalized(num as i128, den as i128)?;
        Ok(value)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            + (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        Self::normalized(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            - (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        Self::normalized(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::normalized(
            (self.num as i128) * (rhs.num as i128),
            (self.den as i128) * (rhs.den as i128),
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::normalized(
            (self.num as i128) * (rhs.den as i128),
            (self.den as i128) * (rhs.num as i128),
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::normalized(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let mut slash = None;
        let mut i = 0;
        while i < bytes.len() {
            let b = bytes[i];
            if b == b'/' {
                if slash.is_some() {
                    return Err(parse_error(i, ParseKind::InvalidByte));
                }
                slash = Some(i);
                i += 1;
                continue;
            }
            if b.is_ascii_digit() {
                i += 1;
                continue;
            }
            if b == b'-' && i == 0 {
                i += 1;
                continue;
            }
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let numerator_end = slash.unwrap_or(bytes.len());
        let numerator_start = usize::from(bytes.first() == Some(&b'-'));
        if numerator_start == numerator_end {
            return Err(parse_error(bytes.len(), ParseKind::InvalidByte));
        }
        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(parse_error(bytes.len(), ParseKind::InvalidByte));
            }
        }

        let negative = numerator_start == 1;
        let numerator_digits = &bytes[numerator_start..numerator_end];
        let denominator_digits = slash.map(|s| &bytes[s + 1..]);

        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(parse_error(numerator_start + 1, ParseKind::NonCanonical));
        }
        if negative && numerator_digits == b"0" {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }
        if let Some(digits) = denominator_digits {
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(parse_error(numerator_end + 2, ParseKind::NonCanonical));
            }
        }

        let num_limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let num_mag = parse_magnitude(numerator_digits);
        if num_mag > num_limit {
            return Err(parse_error(0, ParseKind::OutOfRange));
        }

        let num = if negative {
            if num_mag == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };

        let Some(den_digits) = denominator_digits else {
            return Ok(Self { num, den: 1 });
        };
        let den_mag = parse_magnitude(den_digits);
        let den_at = numerator_end + 1;
        if den_mag > i64::MAX as u128 {
            return Err(parse_error(den_at, ParseKind::OutOfRange));
        }
        if den_mag == 0 {
            return Err(parse_error(den_at, ParseKind::ZeroDenominator));
        }
        if den_mag == 1 {
            return Err(parse_error(numerator_end, ParseKind::NonCanonical));
        }
        if num == 0 {
            return Err(parse_error(numerator_end, ParseKind::NonCanonical));
        }
        if gcd_u128(num.unsigned_abs() as u128, den_mag) > 1 {
            return Err(parse_error(numerator_end, ParseKind::NonCanonical));
        }
        Ok(Self {
            num,
            den: den_mag as i64,
        })
    }

    fn normalized(mut num: i128, mut den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        if den < 0 {
            num = -num;
            den = -den;
        }
        let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
        num /= g;
        den /= g;
        if num < i64::MIN as i128
            || num > i64::MAX as i128
            || den < 1
            || den > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: num as i64,
            den: den as i64,
        })
    }
}

impl std::fmt::Display for Rational {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

fn parse_error(at: usize, kind: ParseKind) -> RationalError {
    RationalError::Parse { at, kind }
}

fn parse_magnitude(digits: &[u8]) -> u128 {
    let mut value = 0u128;
    for &b in digits {
        value = value.saturating_mul(10).saturating_add((b - b'0') as u128);
    }
    value
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols, self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }
        let augmented = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let reduction = augmented.eliminate(n, width)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: reduction.pivots.len(),
            });
        }
        let mut inverse = Vec::with_capacity(n * n);
        for r in 0..n {
            inverse.extend_from_slice(&reduction.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: inverse,
        })
    }

    fn eliminate(&self, pivot_limit: usize, update_cols: usize) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..pivot_limit {
            if p == matrix.rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for row in p..matrix.rows {
                let value = matrix.data[row * matrix.cols + col];
                if value.num == 0 {
                    continue;
                }
                match chosen {
                    None => chosen = Some(row),
                    Some(old) => {
                        let previous = matrix.data[old * matrix.cols + col];
                        let key = (
                            value.den,
                            value.num.unsigned_abs(),
                            row,
                        );
                        let old_key = (
                            previous.den,
                            previous.num.unsigned_abs(),
                            old,
                        );
                        if key < old_key {
                            chosen = Some(row);
                        }
                    }
                }
            }
            let Some(q) = chosen else {
                continue;
            };
            if q != p {
                for c in 0..matrix.cols {
                    matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = matrix.data[p * matrix.cols + col];
            let start = p * matrix.cols;
            for c in 0..update_cols {
                let index = start + c;
                matrix.data[index] = matrix.data[index]
                    .div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic {
                        row: p,
                        col: c,
                        step: p,
                        source,
                    })?;
            }

            for row in 0..matrix.rows {
                if row == p {
                    continue;
                }
                let row_start = row * matrix.cols;
                let factor = matrix.data[row_start + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..update_cols {
                    let pivot_value = matrix.data[start + c];
                    let product = factor.mul(&pivot_value).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        }
                    })?;
                    let index = row_start + c;
                    matrix.data[index] = matrix.data[index].sub(&product).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        }
                    })?;
                }
            }
            pivots.push(col);
            p += 1;
        }

        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}
