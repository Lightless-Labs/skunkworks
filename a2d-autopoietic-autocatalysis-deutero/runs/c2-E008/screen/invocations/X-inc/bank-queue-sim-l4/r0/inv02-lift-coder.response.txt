use std::cmp::Ordering;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };
    Some(quotient + u128::from(round_up))
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    remaining: u64,
    deadline: Option<u64>,
    entry_time: u64,
    waited: u64,
    budget: Option<u64>,
    started: bool,
    paused: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Active {
    index: usize,
    dispatch_time: u64,
    work_at: u64,
    finish: u64,
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event {
        time,
        customer,
        kind,
    });
}

fn precedes(a: &Job, b: &Job) -> bool {
    if a.customer.priority != b.customer.priority {
        return a.customer.priority < b.customer.priority;
    }

    match (a.paused, b.paused) {
        (true, false) => true,
        (false, true) => false,
        (true, true) => (a.pause_time, a.customer.id) < (b.pause_time, b.customer.id),
        (false, false) => {
            (a.customer.arrival, std::cmp::Reverse(a.customer.id))
                < (b.customer.arrival, std::cmp::Reverse(b.customer.id))
        }
    }
}

fn time_overflow(customer: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow {
        customer,
        time,
        stage,
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index]
            .iter()
            .any(|previous| previous.id == customer.id)
        {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
        if let Some(patience) = customer.patience {
            if customer.arrival.checked_add(patience).is_none() {
                return Err(Error::DeadlineOverflow {
                    index,
                    id: customer.id,
                });
            }
        }
    }

    let mut arrival_order: Vec<usize> = (0..customers.len()).collect();
    arrival_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then(customers[a].priority.cmp(&customers[b].priority))
            .then(customers[b].id.cmp(&customers[a].id))
    });

    let mut jobs: Vec<Job> = customers
        .iter()
        .map(|customer| Job {
            customer: customer.clone(),
            remaining: customer.service,
            deadline: customer
                .patience
                .map(|patience| customer.arrival + patience),
            entry_time: customer.arrival,
            waited: 0,
            budget: customer.patience,
            started: false,
            paused: false,
            pause_time: 0,
        })
        .collect();

    let mut events = Vec::new();
    let (mut accepted, mut balked, mut reneged, mut completed, mut started, mut preemptions) =
        (0usize, 0usize, 0usize, 0usize, 0usize, 0usize);
    let (mut busy_ticks, mut wait_ticks, mut turnaround_ticks) = (0u128, 0u128, 0u128);
    let mut completion_order = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut last_dispatched: Option<u32> = None;
    let (mut arrival_index, mut now) = (0usize, 0u64);

    loop {
        // Phase A: finish before processing expirations or arrivals.
        if active.as_ref().is_some_and(|active| active.finish == now) {
            let finished = active.take().unwrap();
            let id = jobs[finished.index].customer.id;
            emit(&mut events, now, id, Kind::Finish);
            completed += 1;
            completion_order.push(id);
            turnaround_ticks += (now - jobs[finished.index].customer.arrival) as u128;
            busy_ticks += (now - finished.dispatch_time) as u128;
        }

        // Phase B: expire customers who were waiting at the start of this tick.
        let mut expired: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&index| jobs[index].deadline.is_some_and(|deadline| deadline <= now))
            .collect();
        expired.sort_by_key(|&index| (jobs[index].deadline.unwrap(), jobs[index].customer.id));

        for index in expired {
            waiting.retain(|&waiting_index| waiting_index != index);
            let segment = now - jobs[index].entry_time;
            jobs[index].waited += segment;
            wait_ticks += segment as u128;
            emit(
                &mut events,
                now,
                jobs[index].customer.id,
                Kind::Renege {
                    waited: jobs[index].waited,
                },
            );
            reneged += 1;
        }

        // Phase C: admit or refuse the whole arrival cohort without dispatching.
        while arrival_index < arrival_order.len()
            && customers[arrival_order[arrival_index]].arrival == now
        {
            let index = arrival_order[arrival_index];
            arrival_index += 1;
            emit(
                &mut events,
                now,
                jobs[index].customer.id,
                Kind::Arrive,
            );

            let gate_size = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= gate_size {
                emit(&mut events, now, jobs[index].customer.id, Kind::Balk);
                balked += 1;
            } else {
                accepted += 1;
                jobs[index].entry_time = now;
                waiting.push(index);
            }
        }

        // Phase D: an unprotected running customer may be preempted.
        if let Some(current) = active.as_ref() {
            if now >= current.work_at {
                let best_waiter = waiting.iter().copied().min_by(|&a, &b| {
                    if precedes(&jobs[a], &jobs[b]) {
                        Ordering::Less
                    } else if precedes(&jobs[b], &jobs[a]) {
                        Ordering::Greater
                    } else {
                        Ordering::Equal
                    }
                });

                if let Some(best) = best_waiter {
                    if jobs[best].customer.priority < jobs[current.index].customer.priority {
                        let paused = active.take().unwrap();
                        let remaining = paused.finish - now;
                        let id = jobs[paused.index].customer.id;
                        emit(
                            &mut events,
                            now,
                            id,
                            Kind::Preempt {
                                remaining,
                                by: jobs[best].customer.id,
                            },
                        );
                        preemptions += 1;
                        busy_ticks += (now - paused.dispatch_time) as u128;

                        jobs[paused.index].remaining = remaining;
                        jobs[paused.index].entry_time = now;
                        jobs[paused.index].paused = true;
                        jobs[paused.index].pause_time = now;
                        jobs[paused.index].deadline = match jobs[paused.index].budget {
                            Some(budget) => Some(
                                now.checked_add(budget).ok_or_else(|| {
                                    time_overflow(id, now, OverflowStage::Patience)
                                })?,
                            ),
                            None => None,
                        };
                        waiting.push(paused.index);
                    }
                }
            }
        }

        // Phase E: dispatch while idle, including inline zero-time completions.
        loop {
            if active.is_some() {
                break;
            }

            let best_position = waiting
                .iter()
                .enumerate()
                .min_by(|(_, &a), (_, &b)| {
                    if precedes(&jobs[a], &jobs[b]) {
                        Ordering::Less
                    } else if precedes(&jobs[b], &jobs[a]) {
                        Ordering::Greater
                    } else {
                        Ordering::Equal
                    }
                })
                .map(|(position, _)| position);

            let Some(position) = best_position else {
                break;
            };

            let index = waiting.remove(position);
            let id = jobs[index].customer.id;
            let segment = now - jobs[index].entry_time;
            jobs[index].waited += segment;
            wait_ticks += segment as u128;
            if let Some(budget) = jobs[index].budget {
                jobs[index].budget = Some(budget - segment);
            }

            let switch_cost = if last_dispatched.is_some_and(|last| last != id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now
                .checked_add(switch_cost)
                .ok_or_else(|| time_overflow(id, now, OverflowStage::WorkAt))?;
            let finish = work_at
                .checked_add(jobs[index].remaining)
                .ok_or_else(|| time_overflow(id, now, OverflowStage::Finish))?;

            if jobs[index].started {
                emit(
                    &mut events,
                    now,
                    id,
                    Kind::Resume {
                        remaining: jobs[index].remaining,
                        work_at,
                        finish,
                    },
                );
            } else {
                emit(
                    &mut events,
                    now,
                    id,
                    Kind::Start {
                        remaining: jobs[index].remaining,
                        work_at,
                        finish,
                    },
                );
                started += 1;
                jobs[index].started = true;
            }

            last_dispatched = Some(id);
            active = Some(Active {
                index,
                dispatch_time: now,
                work_at,
                finish,
            });

            if finish == now {
                active = None;
                emit(&mut events, now, id, Kind::Finish);
                completed += 1;
                completion_order.push(id);
                turnaround_ticks += (now - jobs[index].customer.arrival) as u128;
                continue;
            }
        }

        // Phase F: expire customers added or paused during this tick.
        let mut expired: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&index| jobs[index].deadline.is_some_and(|deadline| deadline <= now))
            .collect();
        expired.sort_by_key(|&index| (jobs[index].deadline.unwrap(), jobs[index].customer.id));

        for index in expired {
            waiting.retain(|&waiting_index| waiting_index != index);
            let segment = now - jobs[index].entry_time;
            jobs[index].waited += segment;
            wait_ticks += segment as u128;
            emit(
                &mut events,
                now,
                jobs[index].customer.id,
                Kind::Renege {
                    waited: jobs[index].waited,
                },
            );
            reneged += 1;
        }

        if arrival_index == arrival_order.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time: Option<u64> = None;
        if arrival_index < arrival_order.len() {
            next_time = Some(customers[arrival_order[arrival_index]].arrival);
        }

        if let Some(current) = active.as_ref() {
            for time in [
                Some(current.work_at).filter(|&time| time > now),
                Some(current.finish).filter(|&time| time > now),
            ]
            .into_iter()
            .flatten()
            {
                next_time = Some(next_time.map_or(time, |next| next.min(time)));
            }
        }

        for &index in &waiting {
            if let Some(deadline) = jobs[index].deadline.filter(|&deadline| deadline > now) {
                next_time = Some(next_time.map_or(deadline, |next| next.min(deadline)));
            }
        }

        let Some(time) = next_time else {
            break;
        };
        now = time;
    }

    let elapsed = events.last().map_or(0, |event| event.time);
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(busy_ticks * 10_000, elapsed as u128).unwrap() as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}
