```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
}

struct State {
    level: usize,
    burst_pos: usize,
    remaining: u64,
    completion: Option<u64>,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut events = Vec::with_capacity(jobs.len() * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: job.arrival as u64,
            kind: 0,
            index,
        });
    }
    events.sort_by_key(|event| (event.time, event.kind, event.index));

    let mut states: Vec<State> = jobs
        .iter()
        .map(|job| State {
            level: job.level,
            burst_pos: 0,
            remaining: job.bursts[0] as u64,
            completion: None,
        })
        .collect();
    let mut queues: Vec<VecDeque<usize>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();
    let mut event_pos = 0;
    let mut time = 0u64;
    let mut last_dispatch: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    while states.iter().any(|state| state.completion.is_none()) {
        while event_pos < events.len() && events[event_pos].time <= time {
            let event = events[event_pos];
            queues[states[event.index].level].push_back(event.index);
            event_pos += 1;
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_pos >= events.len() {
                break;
            }
            let next_time = events[event_pos].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch = None;
            }
            continue;
        };

        let index = queues[level].pop_front().unwrap();
        let job_id = jobs[index].id;

        if last_dispatch.is_some_and(|previous| previous != job_id) {
            time += switch_cost as u64;
        }

        while event_pos < events.len() && events[event_pos].time <= time {
            let event = events[event_pos];
            queues[states[event.index].level].push_back(event.index);
            event_pos += 1;
        }

        let quantum = quanta[level] as u64;
        let consumed = states[index].remaining.min(quantum);
        let start = time;
        time += consumed;
        let end = time;
        states[index].remaining -= consumed;

        let burst_finished = states[index].remaining == 0;
        let job_finished = burst_finished && states[index].burst_pos + 1 == jobs[index].bursts.len();

        runs.push(Run {
            id: job_id,
            start,
            end,
            finished: job_finished,
            level,
        });

        if burst_finished && !job_finished {
            states[index].burst_pos += 1;
            let io_delay = jobs[index].bursts[states[index].burst_pos] as u64;
            states[index].burst_pos += 1;
            states[index].remaining = jobs[index].bursts[states[index].burst_pos] as u64;
            states[index].level = level.saturating_sub(1);

            events.push(Event {
                time: end + io_delay,
                kind: 1,
                index,
            });
            events[event_pos..].sort_by_key(|event| (event.time, event.kind, event.index));
        } else if job_finished {
            states[index].completion = Some(end);
        } else {
            states[index].level = (level + 1).min(quanta.len() - 1);
            queues[states[index].level].push_back(index);
        }

        while event_pos < events.len() && events[event_pos].time <= end {
            let event = events[event_pos];
            queues[states[event.index].level].push_back(event.index);
            event_pos += 1;
        }

        last_dispatch = Some(job_id);
    }

    let stats: Vec<Stat> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| {
            let completion = states[index].completion.unwrap_or(time);
            let turnaround = completion - job.arrival as u64;
            let cpu: u64 = job
                .bursts
                .iter()
                .step_by(2)
                .map(|&length| length as u64)
                .sum();
            let io: u64 = job
                .bursts
                .iter()
                .skip(1)
                .step_by(2)
                .map(|&length| length as u64)
                .sum();
            Stat {
                id: job.id,
                completion,
                turnaround,
                waiting: turnaround - cpu - io,
            }
        })
        .collect();

    let count = stats.len() as u64;
    let total_waiting: u64 = stats.iter().map(|stat| stat.waiting).sum();
    let total_turnaround: u64 = stats.iter().map(|stat| stat.turnaround).sum();

    Ok(Schedule {
        runs,
        idle,
        mean_waiting_x100: (total_waiting * 100 + count / 2) / count,
        mean_turnaround_x100: (total_turnaround * 100 + count / 2) / count,
        end: time,
        stats,
    })
}

use std::collections::VecDeque;
```
