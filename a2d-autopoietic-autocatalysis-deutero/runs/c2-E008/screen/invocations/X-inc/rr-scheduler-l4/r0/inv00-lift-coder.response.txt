```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Process {
    id: u32,
    arrival: u64,
    bursts: Vec<u32>,
    level: usize,
    burst_index: usize,
    remaining: u32,
    completion: Option<u64>,
    total_cpu: u64,
    total_io: u64,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8, // 0 = external arrival, 1 = I/O wake
    index: usize,
    level: usize,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let mut processes: Vec<Process> = jobs
        .iter()
        .map(|job| Process {
            id: job.id,
            arrival: job.arrival as u64,
            bursts: job.bursts.clone(),
            level: job.level,
            burst_index: 0,
            remaining: job.bursts[0],
            completion: None,
            total_cpu: job
                .bursts
                .iter()
                .step_by(2)
                .map(|&value| value as u64)
                .sum(),
            total_io: job
                .bursts
                .iter()
                .skip(1)
                .step_by(2)
                .map(|&value| value as u64)
                .sum(),
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: job.arrival as u64,
            kind: 0,
            index,
            level: job.level,
        })
        .collect();

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut now = 0u64;
    let mut last_dispatch_id: Option<u32> = None;
    let mut completed_count = 0usize;

    fn admit_through(
        through: u64,
        events: &mut Vec<Event>,
        queues: &mut [std::collections::VecDeque<usize>],
    ) {
        events.sort_by_key(|event| (event.time, event.kind, event.index));
        let count = events.partition_point(|event| event.time <= through);
        for event in events.drain(..count) {
            queues[event.level].push_back(event.index);
        }
    }

    while completed_count < jobs.len() {
        admit_through(now, &mut events, &mut queues);

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(selected_level) = selected_level else {
            if events.is_empty() {
                break;
            }
            events.sort_by_key(|event| (event.time, event.kind, event.index));
            let next_time = events[0].time;
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let process_index = queues[selected_level].pop_front().unwrap();
        let selected_id = processes[process_index].id;
        if last_dispatch_id.is_some_and(|last_id| last_id != selected_id) {
            now += switch_cost as u64;
        }
        admit_through(now, &mut events, &mut queues);

        let level = processes[process_index].level;
        let quantum = quanta[level];
        let consumed = processes[process_index].remaining.min(quantum);
        let start = now;
        now += consumed as u64;
        let end = now;
        let burst_finished = consumed == processes[process_index].remaining;

        let mut finished = false;
        if burst_finished {
            if processes[process_index].burst_index + 1 == processes[process_index].bursts.len() - 1 {
                finished = true;
                processes[process_index].completion = Some(end);
                completed_count += 1;
            }
        }

        runs.push(Run {
            id: selected_id,
            start,
            end,
            finished,
            level,
        });

        admit_through(end, &mut events, &mut queues);

        if finished {
            // The process has no further events.
        } else if !burst_finished {
            let process = &mut processes[process_index];
            process.remaining -= consumed;
            process.level = (level + 1).min(quanta.len() - 1);
            queues[process.level].push_back(process_index);
        } else {
            let process = &mut processes[process_index];
            let io_position = process.burst_index + 1;
            let io_delay = process.bursts[io_position] as u64;
            process.burst_index += 2;
            process.remaining = process.bursts[process.burst_index];
            let wake_level = level.saturating_sub(1);
            events.push(Event {
                time: end + io_delay,
                kind: 1,
                index: process_index,
                level: wake_level,
            });
            // Zero-delay wakes are admitted at this endpoint before the next selection.
            admit_through(end, &mut events, &mut queues);
        }

        last_dispatch_id = Some(selected_id);
    }

    let stats: Vec<Stat> = processes
        .iter()
        .map(|process| {
            let completion = process.completion.unwrap_or(0);
            let turnaround = completion - process.arrival;
            Stat {
                id: process.id,
                completion,
                turnaround,
                waiting: turnaround - process.total_cpu - process.total_io,
            }
        })
        .collect();

    let (mean_waiting_x100, mean_turnaround_x100) = if stats.is_empty() {
        (0, 0)
    } else {
        let count = stats.len() as u64;
        let waiting_sum: u64 = stats.iter().map(|stat| stat.waiting).sum();
        let turnaround_sum: u64 = stats.iter().map(|stat| stat.turnaround).sum();
        (
            (waiting_sum * 100 + count / 2) / count,
            (turnaround_sum * 100 + count / 2) / count,
        )
    };

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end: now,
    })
}
```
