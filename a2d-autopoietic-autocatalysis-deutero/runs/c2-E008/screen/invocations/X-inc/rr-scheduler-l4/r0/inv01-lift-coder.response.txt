```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Debug)]
struct Process {
    level: usize,
    burst_index: usize,
    remaining: u32,
    completion: Option<u64>,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: u8,
    index: usize,
    level: usize,
}

fn rounded_mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        return 0;
    }
    let numerator = total * 100;
    let divisor = count as u64;
    (numerator + divisor / 2) / divisor
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut processes: Vec<Process> = jobs
        .iter()
        .map(|job| Process {
            level: job.level,
            burst_index: 0,
            remaining: job.bursts[0],
            completion: None,
        })
        .collect();

    let mut queues: Vec<VecDeque<usize>> = (0..quanta.len()).map(|_| VecDeque::new()).collect();

    let mut events = Vec::new();
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: job.arrival as u64,
            kind: 0,
            index,
            level: job.level,
        });
    }
    events.sort_by_key(|event| (event.time, event.kind, event.index));
    let mut event_cursor = 0;

    let mut time = 0u64;
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut last_dispatch_id: Option<u32> = None;
    let mut last_positive_dispatch_id: Option<u32> = None;

    loop {
        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor];
            queues[event.level].push_back(event.index);
            event_cursor += 1;
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(selected_level) = selected_level else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = events[event_cursor].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
                last_positive_dispatch_id = None;
            } else {
                continue;
            }
            continue;
        };

        let job_index = queues[selected_level].pop_front().unwrap();
        let job = &jobs[job_index];
        let id = job.id;
        let dispatch_level = processes[job_index].level;

        if last_positive_dispatch_id.is_some_and(|last_id| last_id != id) {
            time += switch_cost as u64;
        }

        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor];
            queues[event.level].push_back(event.index);
            event_cursor += 1;
        }

        let quantum = quanta[dispatch_level];
        let consumed = processes[job_index].remaining.min(quantum);
        let start = time;
        time += consumed as u64;
        let end = time;
        processes[job_index].remaining -= consumed;

        let burst_finished = processes[job_index].remaining == 0;
        let final_burst = burst_finished && processes[job_index].burst_index + 1 == job.bursts.len() - 1;

        runs.push(Run {
            id,
            start,
            end,
            finished: final_burst,
            level: dispatch_level,
        });

        while event_cursor < events.len() && events[event_cursor].time <= end {
            let event = events[event_cursor];
            queues[event.level].push_back(event.index);
            event_cursor += 1;
        }

        if !burst_finished {
            processes[job_index].level = (dispatch_level + 1).min(quanta.len() - 1);
            queues[processes[job_index].level].push_back(job_index);
        } else if final_burst {
            processes[job_index].completion = Some(end);
        } else {
            processes[job_index].burst_index += 2;
            processes[job_index].remaining = jobs[job_index].bursts[processes[job_index].burst_index];
            let wake_time = end + jobs[job_index].bursts[processes[job_index].burst_index - 1] as u64;
            let wake_level = dispatch_level.saturating_sub(1);
            let wake = Event {
                time: wake_time,
                kind: 1,
                index: job_index,
                level: wake_level,
            };
            let insert_at = events[event_cursor..]
                .iter()
                .position(|pending| (pending.time, pending.kind, pending.index) > (wake.time, wake.kind, wake.index))
                .map(|offset| event_cursor + offset)
                .unwrap_or(events.len());
            events.insert(insert_at, wake);
        }

        last_dispatch_id = Some(id);
        if consumed > 0 {
            last_positive_dispatch_id = Some(id);
        }
        let _ = last_dispatch_id;
    }

    let end = processes
        .iter()
        .filter_map(|process| process.completion)
        .max()
        .unwrap_or(0);

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = processes[index].completion.unwrap_or(0);
        let turnaround = completion - job.arrival as u64;
        let cpu_total: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&burst| burst as u64)
            .sum();
        let io_total: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&burst| burst as u64)
            .sum();
        let waiting = turnaround - cpu_total - io_total;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: rounded_mean_x100(total_waiting, jobs.len()),
        mean_turnaround_x100: rounded_mean_x100(total_turnaround, jobs.len()),
        end,
    })
}

use std::collections::VecDeque;
```
