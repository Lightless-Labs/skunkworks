#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
    id: usize,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .last()
            .is_some_and(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { segments, rank, id });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let segments = parse_path(&path[..head_end])?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;
        for route in &self.routes {
            let Some(params) = match_route(route, &segments) else {
                continue;
            };

            let replace = match &best {
                None => true,
                Some((current, _)) => route.rank > current.rank,
            };
            if replace {
                best = Some((route, params));
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let bytes = pattern.as_bytes();
    let mut raw_segments = Vec::new();
    let mut start = 1;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                // Validate preceding segments before reporting trailing slash.
                if start == i {
                    return Err(Error::InvalidPattern { at: start });
                }
                raw_segments.push((start, i));
                let parsed = parse_pattern_segments(pattern, &raw_segments)?;
                let _ = parsed;
                return Err(Error::InvalidPattern { at: i });
            }
            if start == i {
                return Err(Error::InvalidPattern { at: start });
            }
            raw_segments.push((start, i));
            start = i + 1;
        }
    }
    if start == bytes.len() {
        return Err(Error::InvalidPattern { at: start });
    }
    raw_segments.push((start, bytes.len()));
    parse_pattern_segments(pattern, &raw_segments)
}

fn parse_pattern_segments(
    pattern: &str,
    ranges: &[(usize, usize)],
) -> Result<Vec<Segment>, Error> {
    let mut segments = Vec::with_capacity(ranges.len());
    let mut names = std::collections::HashSet::new();

    for (index, &(start, end)) in ranges.iter().enumerate() {
        let raw = &pattern[start..end];
        if let Some(name) = raw.strip_prefix(':') {
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(Segment::Param(name.to_owned()));
        } else if let Some(name) = raw.strip_prefix('*') {
            if !valid_name(name)
                || !names.insert(name.to_owned())
                || index + 1 != ranges.len()
            {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(Segment::Wildcard(name.to_owned()));
        } else {
            segments.push(Segment::Static(decode_segment(raw, start, true)?));
        }
    }

    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut ranges = Vec::new();
    let mut start = 1;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                if start == i {
                    return Err(Error::InvalidPath { at: start });
                }
                ranges.push((start, i));
                let parsed = parse_path_segments(path, &ranges)?;
                let _ = parsed;
                return Err(Error::InvalidPath { at: i });
            }
            if start == i {
                return Err(Error::InvalidPath { at: start });
            }
            ranges.push((start, i));
            start = i + 1;
        }
    }
    if start == bytes.len() {
        return Err(Error::InvalidPath { at: start });
    }
    ranges.push((start, bytes.len()));
    parse_path_segments(path, &ranges)
}

fn parse_path_segments(path: &str, ranges: &[(usize, usize)]) -> Result<Vec<String>, Error> {
    ranges
        .iter()
        .map(|&(start, end)| decode_segment(&path[start..end], start, false))
        .collect()
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'_')
}

fn decode_segment(raw: &str, offset: usize, is_pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            let fail = || {
                if is_pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                }
            };
            if i + 2 >= bytes.len() {
                return Err(fail());
            }
            let Some(high) = uppercase_hex(bytes[i + 1]) else {
                return Err(fail());
            };
            let Some(low) = uppercase_hex(bytes[i + 2]) else {
                return Err(fail());
            };
            let value = high * 16 + low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(fail());
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return if is_pattern {
                Err(Error::InvalidPattern { at: offset + i })
            } else {
                Err(Error::InvalidPath { at: offset + i })
            };
        }
    }

    // Every accepted byte is ASCII, including percent-decoded bytes.
    Ok(String::from_utf8(decoded).unwrap())
}

fn uppercase_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let has_wildcard = route
        .segments
        .last()
        .is_some_and(|segment| matches!(segment, Segment::Wildcard(_)));
    let fixed_len = route.segments.len() - usize::from(has_wildcard);
    if (!has_wildcard && path.len() != fixed_len) || (has_wildcard && path.len() < fixed_len) {
        return None;
    }

    let mut params = Vec::new();
    for (index, segment) in route.segments[..fixed_len].iter().enumerate() {
        match segment {
            Segment::Static(value) if value != &path[index] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), path[index].clone())),
            Segment::Wildcard(_) => return None,
        }
    }

    if has_wildcard {
        if let Segment::Wildcard(name) = &route.segments[fixed_len] {
            params.push((name.clone(), path[fixed_len..].join("/")));
        }
    }

    Some(params)
}
