#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|part| !part.is_empty())
            .last()
            .is_some_and(|part| part == "." || part == "..")
}

fn append_directory_slash(result: &mut String, intent: bool) {
    if intent && result != "/" && result != "." {
        result.push('/');
    }
}

fn reduce_component(
    stack: &mut Vec<(String, usize)>,
    component: &str,
    at: usize,
    absolute: bool,
) -> Result<(), PathError> {
    if component.is_empty() || component == "." {
        return Ok(());
    }
    if component == ".." {
        if stack.last().is_some_and(|(name, _)| name != "..") {
            stack.pop();
        } else if absolute {
            return Err(PathError::Escape { at });
        } else {
            stack.push(("..".to_string(), at));
        }
    } else {
        stack.push((component.to_string(), at));
    }
    Ok(())
}

fn split_components(path: &str, offset: usize) -> Vec<(String, usize)> {
    let mut components = Vec::new();
    let mut start = 0;
    for (i, part) in path.split('/').enumerate() {
        if i > 0 {
            start += 1;
        }
        if !part.is_empty() {
            components.push((part.to_string(), offset + start));
        }
        start += part.len();
    }
    components
}

fn stack_to_absolute(stack: &[(String, usize)]) -> String {
    if stack.is_empty() {
        return "/".to_string();
    }
    let mut result = String::new();
    for (i, (component, _)) in stack.iter().enumerate() {
        if i > 0 {
            result.push('/');
        }
        result.push_str(component);
    }
    format!("/{result}")
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let mut stack = Vec::new();

    for (component, at) in split_components(path, 0) {
        reduce_component(&mut stack, &component, at, absolute)?;
    }

    let mut result = if absolute {
        stack_to_absolute(&stack)
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack
            .iter()
            .map(|(component, _)| component.as_str())
            .collect::<Vec<_>>()
            .join("/")
    };
    append_directory_slash(&mut result, directory_intent(path));
    Ok(result)
}

fn normalized_context(path: &str, invalid: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(invalid);
    }
    let normalized = normalize(path).map_err(|_| invalid)?;
    Ok(normalized
        .split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_string)
        .collect())
}

fn request_start<'a>(
    path: &'a str,
    home: Option<&'a [String]>,
) -> Result<(bool, Vec<(String, usize)>, usize), PathError> {
    if path.starts_with('/') {
        return Ok((true, Vec::new(), 0));
    }
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() == 1 || bytes[1] == b'/' {
            let base = home.ok_or(PathError::MissingHome)?.to_vec();
            let consumed = if bytes.len() > 1 { 2 } else { 1 };
            return Ok((true, base.into_iter().map(|s| (s, 0)).collect(), consumed));
        }
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    Ok((true, Vec::new(), 0))
}

fn resolve_core(path: &str, cwd: &[String], home: Option<&[String]>) -> Result<String, PathError> {
    validate_raw(path)?;
    let (absolute, base, consumed) = request_start(path, home)?;
    let mut stack: Vec<(String, usize)> = if path.starts_with('/') {
        Vec::new()
    } else if path.starts_with('~') {
        base
    } else {
        cwd.iter().cloned().map(|s| (s, 0)).collect()
    };

    for (component, at) in split_components(&path[consumed..], consumed) {
        reduce_component(&mut stack, &component, at, absolute)?;
    }

    let mut result = stack_to_absolute(&stack);
    append_directory_slash(&mut result, directory_intent(path));
    Ok(result)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd = normalized_context(cwd, PathError::InvalidCwd)?;
    let home = match home {
        Some(path) => Some(normalized_context(path, PathError::InvalidHome)?),
        None => None,
    };
    resolve_core(path, &cwd, home.as_deref())
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !path_valid || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((_, first)) = seen.iter().find(|(path, _)| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }
    Ok(())
}

#[derive(Clone)]
struct Component {
    name: String,
    at: usize,
    absolute_escape: bool,
}

enum WorkItem {
    Component(Component),
    Finish(usize),
}

fn append_stack_component(stack: &[(String, usize)], component: &str) -> String {
    let mut copy = stack.to_vec();
    copy.push((component.to_string(), 0));
    stack_to_absolute(&copy)
}

fn push_component(
    stack: &mut Vec<(String, usize)>,
    component: &Component,
    links: &[Link],
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
    work: &mut Vec<WorkItem>,
) -> Result<(), PathError> {
    if component.name.is_empty() || component.name == "." {
        return Ok(());
    }
    if component.name == ".." {
        if stack.last().is_some_and(|(name, _)| name != "..") {
            stack.pop();
        } else if component.absolute_escape {
            return Err(PathError::Escape { at: component.at });
        } else {
            stack.push(("..".to_string(), component.at));
        }
        return Ok(());
    }

    stack.push((component.name.clone(), component.at));
    let current = stack_to_absolute(stack);
    if let Some((index, link)) = links
        .iter()
        .enumerate()
        .find(|(_, link)| link.path == current)
    {
        if active.contains(&index) {
            return Err(PathError::Loop { index });
        }
        if *expansions >= max_expansions {
            return Err(PathError::Limit { index });
        }
        *expansions += 1;
        active.push(index);
        let target_absolute = link.target.starts_with('/');
        if target_absolute {
            stack.clear();
        } else {
            stack.pop();
        }
        work.push(WorkItem::Finish(index));
        let target_components = split_components(&link.target, 0);
        for (name, at) in target_components.into_iter().rev() {
            work.push(WorkItem::Component(Component {
                name,
                at,
                absolute_escape: true,
            }));
        }
    }
    Ok(())
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd = normalized_context(cwd, PathError::InvalidCwd)?;
    let home = match home {
        Some(path) => Some(normalized_context(path, PathError::InvalidHome)?),
        None => None,
    };
    validate_links(links)?;
    validate_raw(path)?;

    let (absolute, base, consumed) = request_start(path, home.as_deref())?;
    let mut stack: Vec<(String, usize)> = if path.starts_with('/') {
        Vec::new()
    } else if path.starts_with('~') {
        base
    } else {
        cwd.iter().cloned().map(|s| (s, 0)).collect()
    };

    let mut work = Vec::new();
    for (name, at) in split_components(&path[consumed..], consumed)
        .into_iter()
        .rev()
    {
        work.push(WorkItem::Component(Component {
            name,
            at,
            absolute_escape: absolute,
        }));
    }

    let mut active = Vec::new();
    let mut expansions = 0;
    while let Some(item) = work.pop() {
        match item {
            WorkItem::Finish(index) => {
                if let Some(position) = active.iter().rposition(|&value| value == index) {
                    active.remove(position);
                }
            }
            WorkItem::Component(component) => push_component(
                &mut stack,
                &component,
                links,
                &mut active,
                &mut expansions,
                max_expansions,
                &mut work,
            )?,
        }
    }

    let mut result = stack_to_absolute(&stack);
    append_directory_slash(&mut result, directory_intent(path));
    Ok(result)
}
