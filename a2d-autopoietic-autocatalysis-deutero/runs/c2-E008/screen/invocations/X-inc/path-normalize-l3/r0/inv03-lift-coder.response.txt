use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Mode {
    Normalize,
    Resolve,
    Links,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str) -> Vec<(usize, &str)> {
    let bytes = path.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }

        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }

        result.push((start, &path[start..i]));
    }

    result
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }

    components(path)
        .last()
        .map(|(_, c)| *c == "." || *c == "..")
        .unwrap_or(false)
}

fn format_absolute(stack: &[String], intent: bool) -> String {
    if stack.is_empty() {
        return "/".to_owned();
    }

    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    if intent {
        result.push('/');
    }
    result
}

fn format_relative(stack: &[String], intent: bool) -> String {
    if stack.is_empty() {
        return ".".to_owned();
    }

    let mut result = stack.join("/");
    if intent {
        result.push('/');
    }
    result
}

fn reduce_plain(path: &str, absolute: bool) -> Result<Vec<String>, PathError> {
    let mut stack: Vec<String> = Vec::new();

    for (at, component) in components(path) {
        match component {
            "." => {}
            ".." => {
                if stack.last().map(String::as_str).is_some_and(|c| c != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            _ => stack.push(component.to_owned()),
        }
    }

    Ok(stack)
}

fn normalize_impl(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let stack = reduce_plain(path, absolute)?;

    if absolute {
        Ok(format_absolute(&stack, intent))
    } else {
        Ok(format_relative(&stack, intent))
    }
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    normalize_impl(path)
}

fn validated_context(value: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !value.starts_with('/') {
        return Err(error);
    }

    match normalize_impl(value) {
        Ok(normalized) => Ok(components(&normalized)
            .into_iter()
            .map(|(_, c)| c.to_owned())
            .collect()),
        Err(_) => Err(error),
    }
}

fn tilde_expansion<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(bool, usize), PathError> {
    if !path.starts_with('~') {
        return Ok((false, 0));
    }

    if path.len() > 1 && !path.as_bytes()[1..].starts_with(b"/") {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }

    if home.is_none() {
        return Err(PathError::MissingHome);
    }

    let consumed = if path == "~" { 1 } else { 2 };
    Ok((true, consumed))
}

fn resolve_impl(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: Option<&[Link]>,
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = validated_context(cwd, PathError::InvalidCwd)?;
    let home_components = match home {
        Some(h) => Some(validated_context(h, PathError::InvalidHome)?),
        None => None,
    };

    let link_map = if let Some(table) = links {
        let mut map: HashMap<String, (usize, String)> = HashMap::new();

        for (index, link) in table.iter().enumerate() {
            let valid_path = link.path.starts_with('/')
                && !link.path.ends_with('/')
                && link.path != "/"
                && normalize_impl(&link.path)
                    .map(|normalized| normalized == link.path)
                    .unwrap_or(false);
            let valid_target = validate_raw(&link.target).is_ok();

            if !valid_path || !valid_target {
                return Err(PathError::InvalidLink { index });
            }

            if let Some((first, _)) = map.get(&link.path) {
                return Err(PathError::DuplicateLink {
                    first: *first,
                    second: index,
                });
            }

            map.insert(link.path.clone(), (index, link.target.clone()));
        }

        Some(map)
    } else {
        None
    };

    validate_raw(path)?;
    let intent = directory_intent(path);
    let (expanded_tilde, consumed) = tilde_expansion(path, home)?;

    let absolute = if expanded_tilde {
        false
    } else {
        path.starts_with('/')
    };

    let mut stack = if absolute {
        Vec::new()
    } else if expanded_tilde {
        home_components.clone().unwrap_or_default()
    } else {
        cwd_components.clone()
    };

    if links.is_none() {
        let remainder = &path[consumed..];
        for (at, component) in components(remainder) {
            let original_at = at + consumed;
            match component {
                "." => {}
                ".." => {
                    if stack.last().is_some() {
                        stack.pop();
                    } else {
                        return Err(PathError::Escape { at: original_at });
                    }
                }
                _ => stack.push(component.to_owned()),
            }
        }

        return Ok(format_absolute(&stack, intent));
    }

    let mut active: Vec<usize> = Vec::new();
    let mut expansions = 0usize;
    let request_start = if absolute { 0 } else { consumed };
    let request_components = components(&path[request_start..])
        .into_iter()
        .map(|(at, c)| (at + request_start, c.to_owned(), false))
        .collect::<Vec<_>>();

    struct Pending {
        items: Vec<(usize, String, bool)>,
        pos: usize,
        active_index: Option<usize>,
    }

    let mut pending = vec![Pending {
        items: request_components,
        pos: 0,
        active_index: None,
    }];

    while let Some(frame) = pending.last_mut() {
        if frame.pos >= frame.items.len() {
            let finished = pending.pop().unwrap();
            if let Some(index) = finished.active_index {
                if let Some(pos) = active.iter().rposition(|&i| i == index) {
                    active.remove(pos);
                }
            }
            continue;
        }

        let (at, component, is_context) = frame.items[frame.pos].clone();
        frame.pos += 1;

        match component.as_str() {
            "." => continue,
            ".." => {
                if stack.is_empty() {
                    return Err(PathError::Escape { at });
                }
                stack.pop();
            }
            _ => {
                stack.push(component);

                let Some(map) = link_map.as_ref() else {
                    continue;
                };

                let matched = format_absolute(&stack, false);
                if let Some((index, target)) = map.get(&matched) {
                    if active.contains(index) {
                        return Err(PathError::Loop { index: *index });
                    }

                    if expansions >= max_expansions {
                        return Err(PathError::Limit { index: *index });
                    }

                    expansions += 1;
                    active.push(*index);

                    // The link name is replaced by its target, which is
                    // interpreted relative to the matched link's parent.
                    stack.pop();

                    let target_absolute = target.starts_with('/');
                    if target_absolute {
                        stack.clear();
                    }

                    let target_items = components(target)
                        .into_iter()
                        .map(|(target_at, target_component)| {
                            (target_at, target_component.to_owned(), false)
                        })
                        .collect();

                    pending.push(Pending {
                        items: target_items,
                        pos: 0,
                        active_index: Some(*index),
                    });
                } else if is_context {
                    // Context components cannot contain link-triggering
                    // request offsets; lookups are still performed here.
                }
            }
        }
    }

    Ok(format_absolute(&stack, intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    resolve_impl(path, cwd, home, None, 0)
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_impl(path, cwd, home, Some(links), max_expansions)
}
