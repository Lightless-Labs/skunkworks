#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn in_range(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<&[Point], GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective = if points.len() >= 2 && points[points.len() - 1] == points[0] {
        &points[..points.len() - 1]
    } else {
        points
    };

    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }

    Ok(effective)
}

fn cross(a: Point, b: Point, q: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let aqx = q.x as i128 - a.x as i128;
    let aqy = q.y as i128 - a.y as i128;
    abx * aqy - aby * aqx
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && (q.x as i128) >= (a.x.min(b.x) as i128)
        && (q.x as i128) <= (a.x.max(b.x) as i128)
        && (q.y as i128) >= (a.y.min(b.y) as i128)
        && (q.y as i128) <= (a.y.max(b.y) as i128)
}

fn edge_length_milli(a: Point, b: Point) -> u64 {
    let dx = (a.x as i128 - b.x as i128) as u128;
    let dy = (a.y as i128 - b.y as i128) as u128;
    let squared = dx * dx + dy * dy;
    let target = squared * 1_000_000;

    // Integer square root by binary search. The result is at most 2^51
    // for supported coordinates and therefore fits in u64.
    let mut low = 0u128;
    let mut high = 1u128 << 52;
    while low + 1 < high {
        let mid = low + (high - low) / 2;
        if mid * mid <= target {
            low = mid;
        } else {
            high = mid;
        }
    }

    let lower_square = low * low;
    let next = low + 1;
    let upper_square = next * next;
    let rounded = if target - lower_square >= upper_square - target {
        next
    } else {
        low
    };
    rounded as u64
}

fn shoelace_signed(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn perimeter_and_longest(points: &[Point]) -> (u64, Option<usize>, Vec<u64>) {
    if points.is_empty() {
        return (0, None, Vec::new());
    }

    let mut lengths = Vec::with_capacity(points.len());
    let mut perimeter = 0u64;
    let mut longest: Option<usize> = None;

    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        let length = edge_length_milli(a, b);
        lengths.push(length);
        perimeter += length;

        let replace = match longest {
            None => true,
            Some(j) if length > lengths[j] => true,
            Some(j) if length == lengths[j] => {
                let current_pair = ordered_pair(a, b);
                let best_pair = ordered_pair(points[j], points[(j + 1) % points.len()]);
                current_pair < best_pair || (current_pair == best_pair && i < j)
            }
            _ => false,
        };
        if replace {
            longest = Some(i);
        }
    }

    (perimeter, longest, lengths)
}

fn ordered_pair(a: Point, b: Point) -> (Point, Point) {
    if a <= b {
        (a, b)
    } else {
        (b, a)
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let points = prepare_polygon(points)?;
    let signed_area2 = shoelace_signed(points);
    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };
    let (perimeter_milli, longest_edge, edge_milli) = perimeter_and_longest(points);

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;

    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &query in queries {
        let mut boundary = None;
        let mut inside = false;

        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];

            if on_segment(a, b, query) {
                boundary = Some(i);
                break;
            }

            if (a.y > query.y) != (b.y > query.y) {
                let side = cross(a, b, query);
                if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                    inside = !inside;
                }
            }
        }

        results.push(match boundary {
            Some(edge) => Location::Boundary { edge },
            None if inside => Location::Inside,
            None => Location::Outside,
        });
    }

    Ok(results)
}

fn orient(a: Point, b: Point, c: Point) -> i128 {
    cross(a, b, c)
}

fn between_collinear(a: Point, b: Point, p: Point) -> bool {
    orient(a, b, p) == 0
        && p.x >= a.x.min(b.x)
        && p.x <= a.x.max(b.x)
        && p.y >= a.y.min(b.y)
        && p.y <= a.y.max(b.y)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut indexed: Vec<(Point, usize)> = Vec::with_capacity(points.len());
    for (index, &point) in points.iter().enumerate() {
        if let Some(existing) = indexed.iter_mut().find(|entry| entry.0 == point) {
            existing.1 = existing.1.min(index);
        } else {
            indexed.push((point, index));
        }
    }
    indexed.sort_by_key(|entry| entry.0);

    if indexed.is_empty() {
        return Ok(Hull {
            vertices: Vec::new(),
            source_indices: Vec::new(),
            area2: 0,
            perimeter_milli: 0,
        });
    }

    if indexed.len() == 1 {
        return Ok(Hull {
            vertices: vec![indexed[0].0],
            source_indices: vec![indexed[0].1],
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let all_collinear = indexed[2..]
        .iter()
        .all(|entry| orient(indexed[0].0, indexed[1].0, entry.0) == 0);

    let hull_entries: Vec<(Point, usize)> = if all_collinear {
        match mode {
            Collinear::Include => indexed,
            Collinear::Exclude => vec![indexed[0], *indexed.last().unwrap()],
        }
    } else if mode == Collinear::Exclude {
        let mut lower: Vec<(Point, usize)> = Vec::new();
        for &entry in &indexed {
            while lower.len() >= 2
                && orient(lower[lower.len() - 2].0, lower[lower.len() - 1].0, entry.0) <= 0
            {
                lower.pop();
            }
            lower.push(entry);
        }

        let mut upper: Vec<(Point, usize)> = Vec::new();
        for &entry in indexed.iter().rev() {
            while upper.len() >= 2
                && orient(upper[upper.len() - 2].0, upper[upper.len() - 1].0, entry.0) <= 0
            {
                upper.pop();
            }
            upper.push(entry);
        }

        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower
    } else {
        let mut extremes_lower: Vec<(Point, usize)> = Vec::new();
        for &entry in &indexed {
            while extremes_lower.len() >= 2
                && orient(
                    extremes_lower[extremes_lower.len() - 2].0,
                    extremes_lower[extremes_lower.len() - 1].0,
                    entry.0,
                ) <= 0
            {
                extremes_lower.pop();
            }
            extremes_lower.push(entry);
        }

        let mut extremes_upper: Vec<(Point, usize)> = Vec::new();
        for &entry in indexed.iter().rev() {
            while extremes_upper.len() >= 2
                && orient(
                    extremes_upper[extremes_upper.len() - 2].0,
                    extremes_upper[extremes_upper.len() - 1].0,
                    entry.0,
                ) <= 0
            {
                extremes_upper.pop();
            }
            extremes_upper.push(entry);
        }

        extremes_lower.pop();
        extremes_upper.pop();
        extremes_lower.extend(extremes_upper);

        let mut expanded: Vec<(Point, usize)> = Vec::new();
        for i in 0..extremes_lower.len() {
            let a = extremes_lower[i].0;
            let b = extremes_lower[(i + 1) % extremes_lower.len()].0;
            let mut edge_points: Vec<(Point, usize)> = indexed
                .iter()
                .copied()
                .filter(|entry| between_collinear(a, b, entry.0))
                .collect();

            if a.x != b.x {
                edge_points.sort_by_key(|entry| {
                    if a.x < b.x {
                        (entry.0.x, entry.0.y)
                    } else {
                        (-entry.0.x, -entry.0.y)
                    }
                });
            } else if a.y <= b.y {
                edge_points.sort_by_key(|entry| entry.0.y);
            } else {
                edge_points.sort_by_key(|entry| -entry.0.y);
            }

            for entry in edge_points {
                if expanded.last().map(|last| last.0) != Some(entry.0) {
                    expanded.push(entry);
                }
            }
        }

        if expanded.len() > 1 && expanded.first().map(|entry| entry.0) == expanded.last().map(|entry| entry.0) {
            expanded.pop();
        }
        expanded
    };

    let vertices: Vec<Point> = hull_entries.iter().map(|entry| entry.0).collect();
    let source_indices: Vec<usize> = hull_entries.iter().map(|entry| entry.1).collect();

    let area2 = if vertices.len() >= 3 {
        shoelace_signed(&vertices).unsigned_abs()
    } else {
        0
    };
    let (perimeter_milli, _, _) = perimeter_and_longest(&vertices);

    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}
