#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_unrestricted: bool,
    day_of_week_unrestricted: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn domain_width(field: Field) -> u32 {
    let (min, max) = bounds(field);
    max - min + 1
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }
    Ok(value)
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u32> {
    let names: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };

    names.iter().position(|name| {
        name.as_bytes()
            .iter()
            .zip(bytes)
            .all(|(a, b)| a.eq_ignore_ascii_case(b))
            && name.len() == bytes.len()
    }).map(|i| {
        if field == Field::Month {
            (i + 1) as u32
        } else {
            i as u32
        }
    })
}

fn parse_atom(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    item_start: usize,
) -> Result<(u32, bool), ParseError> {
    if start == end {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    if bytes[start..end].iter().all(u8::is_ascii_digit) {
        let value = parse_number(bytes, start, end, field)?;
        let (min, max) = bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
        Ok((value, true))
    } else if let Some(value) = name_value(field, &bytes[start..end]) {
        Ok((value, false))
    } else {
        Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        })
    }
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item_start == item_end {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }

    let mut slash = None;
    let mut hyphen = None;
    for i in item_start..item_end {
        match bytes[i] {
            b'/' => {
                if slash.is_some() {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                slash = Some(i);
            }
            b'-' => {
                if hyphen.is_some() {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                hyphen = Some(i);
            }
            _ => {}
        }
    }

    let base_end = slash.unwrap_or(item_end);
    if base_end == item_start {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let (start, end, numeric_start, numeric_end) = if bytes[item_start] == b'*' {
        if base_end != item_start + 1 || hyphen.is_some() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let (min, max) = bounds(field);
        (min, max, None, None)
    } else if let Some(h) = hyphen {
        if h >= base_end || h == item_start || h + 1 >= base_end {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let (a, a_numeric) = parse_atom(bytes, item_start, h, field, item_start)?;
        let (b, b_numeric) = parse_atom(bytes, h + 1, base_end, field, item_start)?;

        if a > b {
            return Err(ParseError::ReversedRange {
                field,
                offset: h + 1,
            });
        }

        let nstart = if a_numeric { Some(item_start) } else { None };
        let nend = if b_numeric { Some(h + 1) } else { None };
        (a, b, nstart, nend)
    } else {
        let (value, numeric) = parse_atom(bytes, item_start, base_end, field, item_start)?;
        let end = if slash.is_some() {
            bounds(field).1
        } else {
            value
        };
        (
            value,
            end,
            if numeric { Some(item_start) } else { None },
            None,
        )
    };

    let step = if let Some(slash_pos) = slash {
        let step_start = slash_pos + 1;
        if step_start == item_end
            || !bytes[step_start..item_end]
                .iter()
                .all(u8::is_ascii_digit)
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let value = parse_number(bytes, step_start, item_end, field)?;
        if value == 0 || value > domain_width(field) {
            return Err(ParseError::BadStep {
                field,
                offset: step_start,
            });
        }
        value
    } else {
        1
    };

    let _ = (numeric_start, numeric_end);
    let mut value = start;
    while value <= end {
        selected[(value - bounds(field).0) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (_, max) = bounds(field);
    let mut selected = vec![false; (max - bounds(field).0 + 1) as usize];

    let mut item_start = 0;
    for i in 0..=bytes.len() {
        if i == bytes.len() || bytes[i] == b',' {
            parse_item(bytes, item_start, i, field, &mut selected)?;
            item_start = i + 1;
        }
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected = Vec::with_capacity(5);
    for (i, field) in FIELDS.iter().copied().enumerate() {
        selected.push(parse_field(parts[i], field)?);
    }

    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        day_of_month_unrestricted: parts[2] == "*",
        day_of_week_unrestricted: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, _) = bounds(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(min as u8 + i as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;

        let minute = self.selected[0][at.minute as usize];
        let hour = self.selected[1][at.hour as usize];
        let month = self.selected[3][(at.month - 1) as usize];
        if !(minute && hour && month) {
            return Ok(false);
        }

        let dom = self.selected[2][(at.day - 1) as usize];
        let dow = self.selected[4][weekday(at.year, at.month, at.day) as usize];

        let day_matches = match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        };

        Ok(day_matches)
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            current = match next_minute(current) {
                Some(value) => value,
                None => return Ok(None),
            };
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn next_minute(at: CivilDateTime) -> Option<CivilDateTime> {
    if at.year == 2399 && at.month == 12 && at.day == 31 && at.hour == 23 && at.minute == 59 {
        return None;
    }

    let mut next = at;
    if next.minute < 59 {
        next.minute += 1;
    } else {
        next.minute = 0;
        if next.hour < 23 {
            next.hour += 1;
        } else {
            next.hour = 0;
            if next.day < days_in_month(next.year, next.month) {
                next.day += 1;
            } else {
                next.day = 1;
                if next.month < 12 {
                    next.month += 1;
                } else {
                    next.month = 1;
                    next.year += 1;
                }
            }
        }
    }
    Some(next)
}
