fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }
    Ok(value)
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u32> {

            i as u32
        }
    })
}

fn parse_atom(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    item_start: usize,
) -> Result<(u32, bool), ParseError> {
    if start == end {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    if bytes[start..end].iter().all(u8::is_ascii_digit) {
        let value = parse_number(bytes, start, end, field)?;
        let (min, max) = bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
        Ok((value, true))
    } else if let Some(value) = name_value(field, &bytes[start..end]) {
        Ok((value, false))
    } else {
        Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        })
    }
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item_start == item_end {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }

    let mut slash = None;
    let mut hyphen = None;
    for i in item_start..item_end {
        match bytes[i] {
            b'/' => {
                if slash.is_some() {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                slash = Some(i);
            }
            b'-' => {
                if hyphen.is_some() {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                hyphen = Some(i);
            }
            _ => {}
        }
    }

    let base_end = slash.unwrap_or(item_end);
    if base_end == item_start {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let (start, end, numeric_start, numeric_end) = if bytes[item_start] == b'*' {
        if base_end != item_start + 1 || hyphen.is_some() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let (min, max) = bounds(field);
        (min, max, None, None)
    } else if let Some(h) = hyphen {
        if h >= base_end || h == item_start || h + 1 >= base_end {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let (a, a_numeric) = parse_atom(bytes, item_start, h, field, item_start)?;
        let (b, b_numeric) = parse_atom(bytes, h + 1, base_end, field, item_start)?;

        if a > b {
            return Err(ParseError::ReversedRange {
                field,
                offset: h + 1,
            });
        }

        let nstart = if a_numeric { Some(item_start) } else { None };
        let nend = if b_numeric { Some(h + 1) } else { None };
        (a, b, nstart, nend)
    } else {
        let (value, numeric) = parse_atom(bytes, item_start, base_end, field, item_start)?;
        let end = if slash.is_some() {
            bounds(field).1
        } else {
            value
        };
        (
            value,
            end,
            if numeric { Some(item_start) } else { None },
            None,
        )
    };

    let step = if let Some(slash_pos) = slash {
        let step_start = slash_pos + 1;
        if step_start == item_end
            || !bytes[step_start..item_end]
                .iter()
                .all(u8::is_ascii_digit)
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let value = parse_number(bytes, step_start, item_end, field)?;
        if value == 0 || value > domain_width(field) {
            return Err(ParseError::BadStep {
                field,
                offset: step_start,
            });
        }
        value
    } else {
        1
    };

    let _ = (numeric_start, numeric_end);
    let mut value = start;
    while value <= end {
        selected[(value - bounds(field).0) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (_, max) = bounds(field);
    let mut selected = vec![false; (max - bounds(field).0 + 1) as usize];

    let mut item_start = 0;
    for i in 0..=bytes.len() {
        if i == bytes.len() || bytes[i] == b',' {
            parse_item(bytes, item_start, i, field, &mut selected)?;
            item_start = i + 1;
        }
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected = Vec::with_capacity(5);
    for (i, field) in FIELDS.iter().copied().enumerate() {
        selected.push(parse_field(parts[i], field)?);
    }

    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        day_of_month_unrestricted: parts[2] == "*",
        day_of_week_unrestricted: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, _) = bounds(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(min as u8 + i as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;

        let minute = self.selected[0][at.minute as usize];
        let hour = self.selected[1][at.hour as usize];
        let month = self.selected[3][(at.month - 1) as usize];
        if !(minute && hour && month) {
            return Ok(false);
        }

        let dom = self.selected[2][(at.day - 1) as usize];
        let dow = self.selected[4][weekday(at.year, at.month, at.day) as usize];

        let day_matches = match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        };

        Ok(day_matches)
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            current = match next_minute(current) {
                Some(value) => value,
                None => return Ok(None),
            };
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn next_minute(at: CivilDateTime) -> Option<CivilDateTime> {
    if at.year == 2399 && at.month == 12 && at.day == 31 && at.hour == 23 && at.minute == 59 {
        return None;
    }

    let mut next = at;
    if next.minute < 59 {
        next.minute += 1;
    } else {
        next.minute = 0;
        if next.hour < 23 {
            next.hour += 1;
        } else {
            next.hour = 0;
            if next.day < days_in_month(next.year, next.month) {
                next.day += 1;
            } else {
                next.day = 1;
                if next.month < 12 {
                    next.month += 1;
                } else {
                    next.month = 1;
                    next.year += 1;
                }
            }
        }
    }
    Some(next)
}

    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if is_step {
        let (min, max) = bounds(field);
        let width = max - min + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else {
        let (min, max) = bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
    }
    Ok(value)
}

fn name_value(bytes: &[u8], field: Field) -> Option<u32> {
    let names: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };

    for (i, name) in names.iter().enumerate() {
        if bytes.len() == 3
            && bytes
                .iter()
                .zip(name.as_bytes())
                .all(|(&a, &b)| a.to_ascii_uppercase() == b)
        {
            return Some(if field == Field::Month {
                (i + 1) as u32
            } else {
                i as u32
            });
        }
    }
    None
}

fn parse_base_value(
    bytes: &[u8],
    start: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if !bytes.is_empty() && bytes.iter().all(u8::is_ascii_digit) {
        return parse_number(bytes, start, field, false);
    }
    name_value(bytes, field).ok_or_else(|| syntax_error(field, start))
}

fn parse_item(
    item: &[u8],
    item_start: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(syntax_error(field, item_start));
    }

    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        if slash == 0 || slash + 1 == item.len() {
            return Err(syntax_error(field, item_start));
        }
        (&item[..slash], Some(&item[slash + 1..]), item_start + slash + 1)
    } else {
        (item, None, item_start)
    };

    let range_positions: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();
    if range_positions.len() > 1 {
        return Err(syntax_error(field, item_start));
    }

    let is_star = base == b"*";
    if is_star && !range_positions.is_empty() {
        return Err(syntax_error(field, item_start));
    }

    let (start, end, start_offset, end_offset) = if is_star {
        let (min, max) = bounds(field);
        (min, max, item_start, item_start)
    } else if let Some(&hyphen) = range_positions.first() {
        if hyphen == 0 || hyphen + 1 == base.len() {
            return Err(syntax_error(field, item_start));
        }
        let left = &base[..hyphen];
        let right = &base[hyphen + 1..];

        let left_valid = (left.iter().all(u8::is_ascii_digit) && !left.is_empty())
            || name_value(left, field).is_some();
        let right_valid = (right.iter().all(u8::is_ascii_digit) && !right.is_empty())
            || name_value(right, field).is_some();
        if !left_valid || !right_valid {
            return Err(syntax_error(field, item_start));
        }

        let left_value = parse_base_value(left, item_start, field)?;
        let right_start = item_start + hyphen + 1;
        let right_value = parse_base_value(right, right_start, field)?;
        if left_value > right_value {
            return Err(ParseError::ReversedRange {
                field,
                offset: right_start,
            });
        }
        (left_value, right_value, item_start, right_start)
    } else {
        let valid =
            (base.iter().all(u8::is_ascii_digit) && !base.is_empty())
                || name_value(base, field).is_some();
        if !valid {
            return Err(syntax_error(field, item_start));
        }
        let value = parse_base_value(base, item_start, field)?;
        let end = if step_bytes.is_some() {
            bounds(field).1
        } else {
            value
        };
        (value, end, item_start, item_start)
    };

    let step = if let Some(bytes) = step_bytes {
        if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
            return Err(syntax_error(field, item_start));
        }
        Some(parse_number(bytes, step_offset, field, true)?)
    } else {
        None
    };

    let increment = step.unwrap_or(1);
    let mut value = start;
    while value <= end {
        let (min, _) = bounds(field);
        selected[(value - min) as usize] = true;
        match value.checked_add(increment) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }

    let _ = (start_offset, end_offset);
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let field = FIELDS[i];
        let (min, max) = bounds(field);
        vec![false; (max - min + 1) as usize]
    });

    for (index, text) in parts.iter().enumerate() {
        let field = FIELDS[index];
        let bytes = text.as_bytes();

        for (offset, &byte) in bytes.iter().enumerate() {
            if !(byte.is_ascii_digit()
                || byte.is_ascii_alphabetic()
                || matches!(byte, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        let field_selected = &mut selected[index];
        let mut item_start = 0;
        for item in bytes.split(|&b| b == b',') {
            parse_item(item, item_start, field, field_selected)?;
            item_start += item.len() + 1;
        }
    }

    Ok(Cron {
        selected,
        unrestricted_dom: parts[2] == "*",
        unrestricted_dow: parts[4] == "*",
    })
}

fn leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(at: CivilDateTime) -> u8 {
    let mut days = 0u32;
    for year in 2000..at.year {
        days += if leap_year(year) { 366 } else { 365 };
    }
    for month in 1..at.month {
        days += days_in_month(at.year, month) as u32;
    }
    days += (at.day - 1) as u32;
    ((6 + days) % 7) as u8
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, _) = bounds(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(min as u8 + i as u8))
            .collect()
    }

    fn selected_value(&self, field: Field, value: u8) -> bool {
        let (min, max) = bounds(field);
        value as u32 >= min
            && value as u32 <= max
            && self.selected[field_index(field)][(value as u32 - min) as usize]
    }

    fn date_matches(&self, at: CivilDateTime) -> bool {
        let dom = self.selected_value(Field::DayOfMonth, at.day);
        let dow = self.selected_value(Field::DayOfWeek, weekday(at));
        match (self.unrestricted_dom, self.unrestricted_dow) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                let last_day = at.day == days_in_month(at.year, at.month);
                if last_day {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.selected_value(Field::Minute, at.minute)
            && self.selected_value(Field::Hour, at.hour)
            && self.selected_value(Field::Month, at.month)
            && self.date_matches(at))
    }

    pub fn next_fire(
        &self,
        after: CivilDateTime,
    ) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let minute_values = self.allowed(Field::Minute);
        let hour_values = self.allowed(Field::Hour);
        let month_values = self.allowed(Field::Month);
        if minute_values.is_empty() || hour_values.is_empty() || month_values.is_empty() {
            return Ok(None);
        }

        let mut year = after.year;
        let mut month = after.month;
        let mut day = after.day;
        let mut hour = after.hour;
        let mut minute = after.minute;

        loop {
            let current = CivilDateTime {
                year,
                month,
                day,
                hour,
                minute,
            };
            if current > after && self.matches(current)? {
                return Ok(Some(current));
            }

            if minute < 59 {
                minute += 1;
            } else {
                minute = 0;
                if hour < 23 {
                    hour += 1;
                } else {
                    hour = 0;
                    if day < days_in_month(year, month) {
                        day += 1;
                    } else {
                        day = 1;
                        if month < 12 {
                            month += 1;
                        } else {
                            month = 1;
                            if year == 2399 {
                                return Ok(None);
                            }
                            year += 1;
                        }
                    }
                }
            }
        }
    }
}

#[derive(Clone, Copy)]
struct Domain {

    max: u8,

impl Domain {
    fn width(self) -> u8 {
        self.max - self.min + 1
    }

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }

fn parse_error(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::BadStep { field, offset },
        5 => ParseError::ReversedRange { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u8> {
    if !bytes.is_ascii() {
        return None;
    }
    let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
    match field {
        Field::Month => match upper.as_slice() {
            b"JAN" => Some(1),
            b"FEB" => Some(2),
            b"MAR" => Some(3),
            b"APR" => Some(4),
            b"MAY" => Some(5),
            b"JUN" => Some(6),
            b"JUL" => Some(7),
            b"AUG" => Some(8),
            b"SEP" => Some(9),
            b"OCT" => Some(10),
            b"NOV" => Some(11),
            b"DEC" => Some(12),
            _ => None,
        },
        Field::DayOfWeek => match upper.as_slice() {
            b"SUN" => Some(0),
            b"MON" => Some(1),
            b"TUE" => Some(2),
            b"WED" => Some(3),
            b"THU" => Some(4),
            b"FRI" => Some(5),
            b"SAT" => Some(6),
            _ => None,
        },
        _ => None,
    }
}

fn parse_number(
    field: Field,
    bytes: &[u8],
    offset: usize,
    step: bool,
) -> Result<u8, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(parse_error(field, offset, 0));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(parse_error(field, offset, 1));
    }
    let mut value: u32 = 0;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| parse_error(field, offset, 2))?;
    }
    if step {
        if value == 0 || value > domain(field).width() as u32 {
            return Err(parse_error(field, offset, 4));
        }
    } else {
        let d = domain(field);
        if value < d.min as u32 || value > d.max as u32 {
            return Err(parse_error(field, offset, 3));
        }
    }
    Ok(value as u8)
}

fn parse_base_value(
    field: Field,
    bytes: &[u8],
    offset: usize,
) -> Result<u8, ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) && !bytes.is_empty() {
        parse_number(field, bytes, offset, false)
    } else if let Some(value) = name_value(field, bytes) {
        Ok(value)
    } else {
        Err(parse_error(field, offset, 0))
    }
}

fn parse_item(
    field: Field,
    item: &[u8],
    item_offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(parse_error(field, item_offset, 0));
    }
    let (base, step_part, step_offset) = if let Some(&pos) = slash_positions.first() {
        (&item[..pos], Some(&item[pos + 1..]), item_offset + pos + 1)
    } else {
        (item, None, item_offset)
    };

    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    let syntax_ok = if base == b"*" {
        true
    } else if hyphens.is_empty() {
        !base.is_empty()
    } else {
        hyphens.len() == 1
            && hyphens[0] > 0
            && hyphens[0] + 1 < base.len()
            && base[0] != b'*'
            && base[hyphens[0] + 1] != b'*'
    };
    if !syntax_ok || step_part.is_some_and(|s| s.is_empty() || !s.iter().all(u8::is_ascii_digit)) {
        return Err(parse_error(field, item_offset, 0));
    }

    let is_star = base == b"*";
    let (start, end, end_offset) = if is_star {
        let d = domain(field);
        (d.min, d.max, item_offset)
    } else if let Some(&pos) = hyphens.first() {
        let left = &base[..pos];
        let right = &base[pos + 1..];
        let start = parse_base_value(field, left, item_offset)?;
        let right_offset = item_offset + pos + 1;
        let end = parse_base_value(field, right, right_offset)?;
        if start > end {
            return Err(parse_error(field, right_offset, 5));
        }
        (start, end, right_offset)
    } else {
        let start = parse_base_value(field, base, item_offset)?;
        let end = if step_part.is_some() {
            domain(field).max
        } else {
            start
        };
        (start, end, item_offset)
    };

    let step = match step_part {
        Some(bytes) => parse_number(field, bytes, step_offset, true)?,
        None => 1,
    };

    let _ = end_offset;
    let mut value = start;
    loop {
        selected[(value - domain(field).min) as usize] = true;
        match value.checked_add(step) {
            Some(next) if next <= end => value = next,
            _ => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut all = [
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
        Vec::new(),
    ];

    for (index, (&field, text)) in fields.iter().zip(pieces.iter()).enumerate() {
        let bytes = text.as_bytes();
        for (offset, &b) in bytes.iter().enumerate() {
            let legal = b.is_ascii_digit()
                || b.is_ascii_alphabetic()
                || matches!(b, b'*' | b'-' | b'/' | b',');
            if !legal {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        let d = domain(field);
        let mut selected = vec![false; d.width() as usize];
        let mut start = 0usize;
        for item in bytes.split(|&b| b == b',') {
            parse_item(field, item, start, &mut selected)?;
            start += item.len() + 1;
        }
        all[index] = selected
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(d.min + i as u8))
            .collect();
    }

    Ok(Cron {
        allowed: all,
        day_of_month_unrestricted: pieces[2] == "*",
        day_of_week_unrestricted: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.allowed[field_index(field)].clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        if after == max_datetime() {
            return Ok(None);
        }

        let mut current = after;
        loop {
            current = increment_minute(current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
            if current == max_datetime() {
                return Ok(None);
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.allowed[0].contains(&at.minute)
            || !self.allowed[1].contains(&at.hour)
            || !self.allowed[3].contains(&at.month)
        {
            return false;
        }

        let dom = self.allowed[2].contains(&at.day);
        let dow = self.allowed[4].contains(&weekday(at.year, at.month, at.day));
        match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {

    if !(1..=12).contains(&at.month) {

fn increment_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}

fn max_datetime() -> CivilDateTime {
    CivilDateTime {
        year: 2399,
        month: 12,
        day: 31,
        hour: 23,
        minute: 59,
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {

    Minute,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {

#[derive(Clone, Copy)]
struct Domain {
    min: u32,
    max: u32,
}

impl Domain {
    fn width(self) -> u32 {
        self.max - self.min + 1
    }
}

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn name_value(field: Field, s: &str) -> Option<u32> {
    let names: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN",
            "JUL", "AUG", "SEP", "OCT", "NOV", "DEC",
        ],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    names
        .iter()
        .position(|name| name.eq_ignore_ascii_case(s))
        .map(|i| if field == Field::DayOfWeek { i as u32 } else { i as u32 + 1 })
}

fn is_atom_for_field(field: Field, s: &str) -> bool {
    !s.is_empty()
        && (s.bytes().all(|b| b.is_ascii_digit()) || name_value(field, s).is_some())
}

fn parse_number(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn parse_endpoint(
    s: &str,
    field: Field,
    offset: usize,
    check_domain: bool,
) -> Result<u32, ParseError> {
    let value = if s.bytes().all(|b| b.is_ascii_digit()) {
        parse_number(s, field, offset)?
    } else {
        name_value(field, s).ok_or(ParseError::InvalidSyntax { field, offset })?
    };
    if check_domain {
        let d = domain(field);
        if value < d.min || value > d.max {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }
    Ok(value)
}

fn parse_item(
    item: &str,
    field: Field,
    item_offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_count = item.bytes().filter(|&b| b == b'/').count();
    if slash_count > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (base, step_text, step_offset) = if let Some(slash) = item.find('/') {
        (&item[..slash], Some(&item[slash + 1..]), item_offset + slash + 1)
    } else {
        (item, None, item_offset)
    };

    let d = domain(field);
    let (kind, first_text, second_text, second_offset) = if base == "*" {
        (0u8, "", None, item_offset)
    } else {
        let hyphen_count = base.bytes().filter(|&b| b == b'-').count();
        if hyphen_count > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        if hyphen_count == 1 {
            let hyphen = base.find('-').unwrap();
            let left = &base[..hyphen];
            let right = &base[hyphen + 1..];
            if !is_atom_for_field(field, left) || !is_atom_for_field(field, right) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_offset,
                });
            }
            (2u8, left, Some(right), item_offset + hyphen + 1)
        } else {
            if !is_atom_for_field(field, base) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_offset,
                });
            }
            (1u8, base, None, item_offset)
        }
    };

    if let Some(step) = step_text {
        if step.is_empty() || !step.bytes().all(|b| b.is_ascii_digit()) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
    }

    let mut start = d.min;
    let mut end = d.max;

    match kind {
        0 => {}
        1 => {
            start = parse_endpoint(first_text, field, item_offset, true)?;
            end = if step_text.is_some() { d.max } else { start };
        }
        2 => {
            start = parse_endpoint(first_text, field, item_offset, true)?;
            end = parse_endpoint(
                second_text.unwrap(),
                field,
                second_offset,
                true,
            )?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: second_offset,
                });
            }
        }
        _ => unreachable!(),
    }

    let step = if let Some(step_text) = step_text {
        let value = parse_number(step_text, field, step_offset)?;
        if value == 0 || value > d.width() {
            return Err(ParseError::BadStep {
                field,
                offset: step_offset,
            });
        }
        value
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected[(value - d.min) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = [
        vec![false; 60],
        vec![false; 24],
        vec![false; 31],
        vec![false; 12],
        vec![false; 7],
    ];

    for (i, field_text) in pieces.iter().enumerate() {
        let field = fields[i];
        for (offset, b) in field_text.bytes().enumerate() {
            if !(b.is_ascii_digit()
                || b.is_ascii_alphabetic()
                || matches!(b, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        let mut item_start = 0;
        for item in field_text.split(',') {
            parse_item(item, field, item_start, &mut selected[i])?;
            item_start += item.len() + 1;
        }
    }

    Ok(Cron {
        selected,
        day_of_month_unrestricted: pieces[2] == "*",
        day_of_week_unrestricted: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let d = domain(field);
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &is_selected)| {
                is_selected.then_some((d.min as usize + i) as u8)
            })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        if after.year == 2399 && after.month == 12 && after.day == 31
            && after.hour == 23 && after.minute == 59
        {
            return Ok(None);
        }

        let mut date = after;
        loop {
            let month_days = days_in_month(date.year, date.month);
            if date.day > month_days {
                date.day = 1;
                if date.month == 12 {
                    if date.year == 2399 {
                        return Ok(None);
                    }
                    date.year += 1;
                    date.month = 1;
                } else {
                    date.month += 1;
                }
                continue;
            }

            if self.date_matches(date) {
                let start_minute = if date.year == after.year
                    && date.month == after.month
                    && date.day == after.day
                {
                    date.hour as u16 * 60 + date.minute as u16 + 1
                } else {
                    0
                };

                if let Some((hour, minute)) = self.next_time_on_date(start_minute) {
                    return Ok(Some(CivilDateTime {
                        year: date.year,
                        month: date.month,
                        day: date.day,
                        hour,
                        minute,
                    }));
                }
            }

            if date.year == 2399 && date.month == 12 && date.day == 31 {
                return Ok(None);
            }
            date.day += 1;
            date.hour = 0;
            date.minute = 0;
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.date_matches(at)
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        let d = domain(field);
        let index = value as i32 - d.min as i32;
        index >= 0 && self.selected[field_index(field)][index as usize]
    }

    fn date_matches(&self, at: CivilDateTime) -> bool {
        let dom = self.contains(Field::DayOfMonth, at.day);
        let dow = self.contains(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn next_time_on_date(&self, start_minute: u16) -> Option<(u8, u8)> {
        for total in start_minute..(24 * 60) {
            let hour = (total / 60) as u8;
            let minute = (total % 60) as u8;
            if self.contains(Field::Hour, hour) && self.contains(Field::Minute, minute) {
                return Some((hour, minute));
            }
        }
        None
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day < 1 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap_year(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}
