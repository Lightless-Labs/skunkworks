#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

fn append_group(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let rem = n % 100;
        if rem != 0 {
            out.push_str(" and ");
            append_under_hundred(rem, out);
        }
    } else {
        append_under_hundred(n, out);
    }
}

fn append_under_hundred(n: u16, out: &mut String) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        out.push_str(TENS[(n / 10 - 2) as usize]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(SMALL[(n % 10) as usize]);
        }
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut rest = magnitude;
    for group in &mut groups {
        *group = (rest % 1000) as u16;
        rest /= 1000;
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let mut higher_present = false;
    for i in (1..groups.len()).rev() {
        if groups[i] == 0 {
            continue;
        }
        if !out.is_empty() && !(value < 0 && out == "minus ") {
            out.push(' ');
        }
        append_group(groups[i], &mut out);
        out.push(' ');
        out.push_str(SCALES[i - 1]);
        higher_present = true;
    }

    let units = groups[0];
    if units != 0 {
        if higher_present {
            out.push_str(if units < 100 { " and " } else { " " });
        } else if value < 0 {
            // The sign already ends in a space.
        }
        append_group(units, &mut out);
    }

    Ok(out)
}

fn dictionary_value(word: &str) -> Option<(u128, TokenKind)> {
    if word == "zero" {
        return Some((0, TokenKind::Add));
    }
    for (i, candidate) in SMALL.iter().enumerate().skip(1) {
        if word == *candidate {
            return Some((i as u128, TokenKind::Add));
        }
    }
    for (i, candidate) in TENS.iter().enumerate() {
        if word == *candidate {
            return Some(((20 + i * 10) as u128, TokenKind::Add));
        }
    }
    if word == "hundred" {
        return Some((100, TokenKind::Hundred));
    }
    for (i, candidate) in SCALES.iter().enumerate() {
        if word == *candidate {
            return Some((1000u128.pow((i + 1) as u32), TokenKind::Scale));
        }
    }
    if word == "and" {
        return Some((0, TokenKind::And));
    }
    None
}

#[derive(Clone, Copy)]
enum TokenKind {
    Add,
    Hundred,
    Scale,
    And,
}

#[derive(Clone, Copy)]
struct Token<'a> {
    word: &'a str,
    index: usize,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push(Token {
                word: &input[start..index],
                index: start,
            });
            start = index + 1;
        }
    }
    tokens.push(Token {
        word: &input[start..],
        index: start,
    });

    for (i, token) in tokens.iter().enumerate() {
        if token.word == "minus" {
            if i != 0 {
                return Err(WordsError::InvalidWord { index: token.index });
            }
        } else if dictionary_value(token.word).is_none() {
            return Err(WordsError::InvalidWord { index: token.index });
        }
    }

    let negative = tokens.first().map(|t| t.word == "minus").unwrap_or(false);
    let arithmetic_tokens = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total = 0u128;
    let mut group = 0u128;
    for token in arithmetic_tokens {
        let (value, kind) = dictionary_value(token.word).unwrap();
        match kind {
            TokenKind::Add => group += value,
            TokenKind::Hundred => group *= 100,
            TokenKind::Scale => {
                total += group * value;
                group = 0;
            }
            TokenKind::And => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let decoded = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(decoded)?;
    if canonical.as_bytes() != input.as_bytes() {
        let mismatch = canonical
            .bytes()
            .zip(input.bytes())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(input.len()));
        return Err(WordsError::NonCanonical { index: mismatch });
    }
    Ok(decoded)
}
