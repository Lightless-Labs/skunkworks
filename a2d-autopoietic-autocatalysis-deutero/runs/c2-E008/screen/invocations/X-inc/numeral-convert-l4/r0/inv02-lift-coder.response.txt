#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }

    let mut parts = Vec::new();
    let mut groups = [0u16; 7];
    let mut remaining = magnitude;

    for group in groups.iter_mut().rev() {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    for index in 0..groups.len() {
        let group = groups[index];
        if group == 0 {
            continue;
        }

        let is_units = index == groups.len() - 1;
        let scale_index = groups.len() - 2 - index;
        let mut spelling = spell_group(group);

        if !is_units {
            spelling.push(' ');
            spelling.push_str(SCALES[scale_index]);
        }

        parts.push((spelling, is_units, group));
    }

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }

    if parts.is_empty() {
        result.push_str("zero");
        return Ok(result);
    }

    for (i, (part, is_units, group)) in parts.iter().enumerate() {
        if i > 0 {
            result.push(' ');
            if *is_units && *group < 100 {
                result.push_str("and ");
            }
        }
        result.push_str(part);
        let _ = is_units;
    }

    Ok(result)
}

fn spell_group(value: u16) -> String {
    let mut result = String::new();
    let hundreds = value / 100;
    let remainder = value % 100;

    if hundreds > 0 {
        result.push_str(SMALL[hundreds as usize]);
        result.push_str(" hundred");
        if remainder > 0 {
            result.push_str(" and ");
        }
    }

    if remainder > 0 {
        if remainder < 20 {
            result.push_str(SMALL[remainder as usize]);
        } else {
            let tens = remainder / 10;
            let units = remainder % 10;
            result.push_str(TENS[(tens - 2) as usize]);
            if units > 0 {
                result.push('-');
                result.push_str(SMALL[units as usize]);
            }
        }
    }

    result
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let tokens: Vec<(usize, &str)> = input
        .split(|c| c == ' ' || c == '-')
        .scan(0usize, |offset, token| {
            let current = *offset;
            *offset += token.len() + 1;
            Some((current, token))
        })
        .collect();

    for (index, token) in &tokens {
        if *token == "minus" {
            continue;
        }
        if !is_dictionary_word(token) {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let negative = tokens.first().is_some_and(|(_, token)| *token == "minus");
    for (position, (index, token)) in tokens.iter().enumerate() {
        if *token == "minus" && position != 0 {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let arithmetic = if negative { &tokens[1..] } else { &tokens[..] };
    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for (_, token) in arithmetic {
        match *token {
            "zero" | "and" => {}
            "one" => group += 1,
            "two" => group += 2,
            "three" => group += 3,
            "four" => group += 4,
            "five" => group += 5,
            "six" => group += 6,
            "seven" => group += 7,
            "eight" => group += 8,
            "nine" => group += 9,
            "ten" => group += 10,
            "eleven" => group += 11,
            "twelve" => group += 12,
            "thirteen" => group += 13,
            "fourteen" => group += 14,
            "fifteen" => group += 15,
            "sixteen" => group += 16,
            "seventeen" => group += 17,
            "eighteen" => group += 18,
            "nineteen" => group += 19,
            "twenty" => group += 20,
            "thirty" => group += 30,
            "forty" => group += 40,
            "fifty" => group += 50,
            "sixty" => group += 60,
            "seventy" => group += 70,
            "eighty" => group += 80,
            "ninety" => group += 90,
            "hundred" => group *= 100,
            "thousand" => apply_scale(&mut total, &mut group, 1_000),
            "million" => apply_scale(&mut total, &mut group, 1_000_000),
            "billion" => apply_scale(&mut total, &mut group, 1_000_000_000),
            "trillion" => apply_scale(&mut total, &mut group, 1_000_000_000_000),
            "quadrillion" => apply_scale(&mut total, &mut group, 1_000_000_000_000_000),
            "quintillion" => apply_scale(&mut total, &mut group, LIMIT),
            _ => unreachable!(),
        }

        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let signed = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    if canonical.as_bytes() != bytes {
        let mismatch = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: mismatch });
    }

    Ok(signed)
}

fn apply_scale(total: &mut u128, group: &mut u128, scale: u128) {
    *total += *group * scale;
    *group = 0;
}

fn is_dictionary_word(word: &str) -> bool {
    matches!(
        word,
        "zero"
            | "one"
            | "two"
            | "three"
            | "four"
            | "five"
            | "six"
            | "seven"
            | "eight"
            | "nine"
            | "ten"
            | "eleven"
            | "twelve"
            | "thirteen"
            | "fourteen"
            | "fifteen"
            | "sixteen"
            | "seventeen"
            | "eighteen"
            | "nineteen"
            | "twenty"
            | "thirty"
            | "forty"
            | "fifty"
            | "sixty"
            | "seventy"
            | "eighty"
            | "ninety"
            | "hundred"
            | "thousand"
            | "million"
            | "billion"
            | "trillion"
            | "quadrillion"
            | "quintillion"
            | "and"
            | "minus"
    )
}
